\section{Browder's bridge}
% AimPaper: 1.3 theorem 117 123
\begin{theorem}[Browder's criterion for Kervaire invariant one]
\label{paper:1.3}
\source{AimPaper Theorem 1.3}
\level{coarse}
\textit{(From \cite{LWX126}, Theorem 1.3.)}
% BEGIN AIMPAPER STATEMENT
[Browder \cite{Browder}] \label{thm:browder}
   There exist framed manifolds in dimension $n$ of Kervaire invariant one if and only if
   \begin{enumerate}
       \item $n=2^{j+1}-2$, and
       \item the element $h_j^2$ in the Adams spectral sequence survives to the $E_\infty$-page.
   \end{enumerate}
% END AIMPAPER STATEMENT
% Mathematical dependencies: reviewed
\uses{sem:AdamsSynthetic:standard-classes,sem:SpectralSequence:permanent}
% Blueprint note: Mathematical dependency review: [AimPaper source, lines 117-126](paper/main.tex.html#L117). Browder is an external input; standard Adams classes and permanence specify its criterion, rather than proving the geometric bridge.

% Correspondence: Not formalized: geometric bridge.
% Blueprint note: Browder's equivalence between geometric existence and Adams permanence has no public interface here. The theta-detection results in the Literature chapter do not supply this geometric equivalence.
\end{theorem}
