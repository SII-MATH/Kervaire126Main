\section{Leibniz and Mahowald formulas}
\begin{theorem}[Generalized Leibniz and Mahowald laws]
\label{frontier:ProofTools:003e9deb7ec5}
\level{fine}
(From \cite{LWX126}, Theorems 6.1 and 6.12.)\par
Work in the fixed classical and synthetic Adams model.
\begin{enumerate}
\item Let $X,Y$ be classical spectra and $f:X\to Y$, and put $e=e(f)$. For every $p\in\mathbb N$ and $r,m,l,s,t\in\mathbb Z$ with
\[2\leq p\leq r,\qquad e\leq m\leq p-2+e,\qquad e\leq l,\]
let
\[x\in Z_{r-1}^{s,t}(X),\qquad y\in Z_{r-1-m+e}^{s+m,t+m}(Y),\]
\[x_\infty\in Z_\infty^{s+r,t+r-1}(X),\qquad y_\infty\in Z_\infty^{s+r+l,t+r+l-1}(Y).
\]
Suppose that
\[d_r(x)=x_\infty,\qquad d_m^{f,E_p}(x)=y,\qquad d_l^{f,E_\infty}(x_\infty)=y_\infty.
\]
Assume that the first differential has no crossing on page $p$ or that the finite extension has no crossing, and that the infinite extension has no crossing. Then
\[d_{r+l-m}(y)=y_\infty.
\]
Thus the generalized Leibniz law of \ref{sem:ProofTools:leibniz-law} holds for every $X,Y,f$.

\item Let $X\xrightarrow fY\xrightarrow gZ\xrightarrow h\Sigma X$ be a distinguished triangle with
\[e(f)+e(g)+e(h)=1,
\]
and assume that its normalized synthetic maps form a compatible triangle, as in \ref{sem:ProofTools:mahowald-law}. For every $n,m,l,s,t\in\mathbb Z$, put
\[n_1=n-e(f)\geq1,\quad m_1=m-e(g)\geq0,\quad l_1=l-e(h)\geq0,\quad r=n+m+l,\quad r'=r-m_1.
\]
Let
\[x\in Z_{n_1}^{s+l,t+l-1}(X),\qquad y\in Z_{m_1+1}^{s+l+n,t+l+n-1}(Y),\]
\[\bar x\in Z_{r-1}^{s,t}(Z),\qquad\bar y\in Z_\infty^{s+r,t+r-1}(Z).
\]
Let $\widetilde x\in E_2^{s+l,t+l}(\Sigma X)$ desuspend to $x$ under the classical tower suspension comparison. Suppose that
\[d_l^{h,E_{r'}}(\bar x)=\widetilde x,\qquad d_r(\bar x)=\bar y,\qquad d_m^{g,E_{m_1+2}}(y)=\bar y.
\]
Assume that the $h$-extension has no crossing or that the classical differential has no crossing on page $r'$. Then
\[x\in Z_{n+m+e(h)}^{s+l,t+l-1}(X)
\]
and there is $y'\in E_2^{s+l+n,t+l+n-1}(Y)$ such that
\[d_n^{f,E_{n+m+1+e(h)}}(x)=y',\qquad y'-y\in B_{r'}^{s+l+n,t+l+n-1}(Y).
\]
\end{enumerate}
\lean{KIP126.ProofTools.Standard.generalizedLeibniz,KIP126.ProofTools.Standard.generalizedMahowald}
\end{theorem}
