\begin{theorem}[Classification of every incoming differential to T]
\label{frontier:AdamsCalculations:f6a9ba54d96c}
\level{fine}
Let $r\in\mathbb Z$. If an $r$th-page differential hits a nonzero representative of $T\in E_2^{14,139}(S^0)$, then either
\[r=6\quad\hbox{and}\quad d_6(W)=T,\]
or
\[r=12\quad\hbox{and}\quad d_{12}(h_6^2)=T.\]
In either alternative, the displayed differential is nonzero. Its source bidegree is respectively $(8,134)$ or $(2,128)$.
\lean{KIP126.Computation.Standard.t_only_incoming}
\end{theorem}

\begin{theorem}[Certification of the finite Adams table]
\label{frontier:AdamsCalculations:84a3eb0053cf}
\level{fine}
Every row in the fixed Adams claim table has a successfully decoded source coordinate sum; differential and refutation rows also have a successfully decoded target coordinate sum. A row of type ``reaches'' admits a representative on its stated page; a row of type ``boundary by'' belongs to the boundary subobject by its stated page; a differential row satisfies its encoded differential equation; and a refutation row fails its encoded differential equation. These assertions concern the spectrum, bidegrees, page and coordinate lists of that row.
\lean{KIP126.Computation.Table.result}
\end{theorem}
