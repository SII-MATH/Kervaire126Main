\chapter{Main — The permanence argument}
The argument takes place on the fixed classical and synthetic sphere systems. It reduces the last possible differential on $h_6^2$ to three detection conditions, proves that two of these conditions cannot hold together, and concludes nonzero permanence.
In the statements quoted from \cite{LWX126}, spectra are assumed to be $2$-completed, connective, and of finite type. The original statements and their context are in \href{paper/main.tex.html}{the article source} and \href{paper/2412.10879.pdf}{the article PDF}.

\begin{remark}[Choice independence and the total boundary]

\level{coarse}

For every $\theta\in\pi_{62,64}S^{0,0}$ satisfying $\operatorname{ThetaChoice}(\theta)$,
\[
 \lambda\eta\theta^2=\lambda\eta(\theta_5^{\mathrm{BX}})^2.
\]
In particular, this expression has the same value for any two admissible choices. It is the expression, rather than $\theta$ itself, that is independent of the choice.

\begin{theorem}[Normalization and choice transport of the literature criterion]
\label{sem:Main:any-choice-criterion}
\level{coarse}
From \cite{LWX126}, Theorem 7.3 and Remarks 7.4--7.5; \cite{BurklundXu}, Proposition 7.19.\par
The literature criterion is recalled in \ref{paper:7.3}. On the fixed system its normalized form is as follows.
For every admissible $\theta$ and every natural number $r\geq0$,
\[
 h_6^2\text{ survives to }E_{r+3}
 \quad\Longleftrightarrow\quad
 [\lambda\eta\theta^2]=0\text{ in }S^{0,0}/\lambda^{r+1}.
\]
Moreover,
\[
 P\quad\Longleftrightarrow\quad\lambda\eta\theta^2=0
 \text{ in the untruncated synthetic sphere}.
\]

\lean{KIP126.Main.any_choice_criterion}
% Blueprint note: This transports the distinguished-choice Literature interfaces. The final permanence proof calls those interfaces directly and does not depend on this exported any-choice theorem. Registered upstream obligations remain.
\end{theorem}
\begin{proof}
From \cite{LWX126}, Remark 7.4.\par
Let $q_n$ denote the quotient map to $S^{0,0}/\lambda^n$. For every $a\in\pi_{125,130}S^{0,0}$ and $n\geq0$,
\[
 q_n(a)=0\quad\Longleftrightarrow\quad q_{n+1}(\lambda a)=0.
\]
In particular, with $a=\eta(\theta_5^{\mathrm{BX}})^2$ this identifies the two quotient conditions
\[
 q_r\bigl(\eta(\theta_5^{\mathrm{BX}})^2\bigr)=0
 \quad\Longleftrightarrow\quad
 q_{r+1}\bigl(\lambda\eta(\theta_5^{\mathrm{BX}})^2\bigr)=0.
\]

Exactness identifies $\ker q_n$ with the image of $\lambda^n$. In this degree multiplication by $\lambda$ is injective, so $\lambda a=\lambda^{n+1}b$ is equivalent to $a=\lambda^n b$. This proves the normalization used in Remark 7.4.

For $r\geq1$, apply the distinguished-choice Literature criterion, this normalization, and Remark~\ref{sem:Main:remark-7.5}. At $r=0$, $h_6^2$ survives to $E_3$ and every lambda multiple maps to zero in the first quotient. The untruncated equivalence follows from the distinguished-choice Literature criterion and the same choice independence.
\end{proof}
