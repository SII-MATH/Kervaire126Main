\section{Adams constructions and their identities}

\begin{definition}[Auxiliary maps and a detecting spectrum]
\label{frontier:AdamsSynthetic:0895d4999173}
\level{fine}
Fix maps $\eta\colon S^1\to S$ and $\nu_3\colon S^3\to S$, a spectrum $D$, and a map $S\to D$. These determine the classical object family $\mathcal C_0$ and the synthetic object family $\mathcal S_0$ defined above.
\lean{KIP126.AdamsSynthetic.Standard.auxiliary}
\end{definition}

\begin{definition}[Classical coefficient-unit towers]
\label{frontier:AdamsSynthetic:14d5e6606bc5}
\level{fine}
For every $X\in\mathrm{Sp}$, let $T(X)$ be its Adams tower for the unit $S\to H\mathbb F_2$. The towers are functorial in $X$. Their cycle and boundary quotients identify with the classical Adams pages, including their differentials and infinite cycles and boundaries. Under the ambient-group comparisons, induced maps agree with the maps on tower layers.
\lean{KIP126.AdamsSynthetic.Standard.classicalTower}
\end{definition}

\begin{definition}[Synthetic coefficient-unit towers]
\label{frontier:AdamsSynthetic:1b85f123880c}
\level{fine}
For every $Y\in\mathrm{Syn}$, let $T_{\mathrm{syn}}(Y)$ be its Adams tower for the image under $\nu$ of the mod-two coefficient unit. At weight $w$, the synthetic Adams sequence identifies with the tower sequence of $\Sigma^{0,-w}Y$, including its differentials, cycles, and boundaries. These identifications are functorial on $E_2$: the two ambient representatives associated to a map differ by an initial tower boundary.
\lean{KIP126.AdamsSynthetic.Standard.syntheticTower}
\end{definition}

\begin{definition}[Classical functorial cofibers]
\label{frontier:AdamsSynthetic:23334467e67d}
\level{fine}
For $f\colon X\to Y$ in $\mathrm{Sp}$, choose a cofiber $C(f)$ with a distinguished triangle $X\xrightarrow f Y\to C(f)\to\Sigma X$. Every commutative square between such maps induces a map of cofibers commuting with the inclusions and boundaries.
\lean{KIP126.AdamsSynthetic.Standard.spectrumCofiber}
\end{definition}

\begin{definition}[The synthetic Adams functor]
\label{frontier:AdamsSynthetic:46d2c66c5a63}
\level{fine}
Let $E^{\mathrm{syn}}$ be the functor from $\mathrm{Syn}$ to trigraded spectral sequences of abelian groups. Its first page is $E_2$, and its page-$r$ differential has degree $(r,r-1,0)$.
\lean{KIP126.AdamsSynthetic.Standard.family}
\end{definition}

\begin{definition}[The classical Adams functor]
\label{frontier:AdamsSynthetic:48bf5d838430}
\level{fine}
Let $E$ be the functor from $\mathrm{Sp}$ to bigraded spectral sequences of abelian groups. Its first page is $E_2$, and its page-$r$ differential has degree $(r,r-1)$.
\lean{KIP126.AdamsSynthetic.Standard.classical}
\end{definition}

\begin{definition}[The objects of the classical category]
\label{frontier:AdamsSynthetic:5833df0c640f}
\level{fine}
Let $\mathrm{Sp}$ denote the type of classical spectra, in universe $\mathrm{Type}_1$.
\lean{KIP126.AdamsSynthetic.Standard.Spectrum}
\end{definition}

\begin{definition}[The classical stable category]
\label{frontier:AdamsSynthetic:6793885d9e9c}
\level{fine}
Equip $\mathrm{Sp}$ with a category structure with small hom sets, a preadditive structure, integer suspension functors, a monoidal structure, a zero object, and a pretriangulated structure. Each integer suspension functor is additive.
\lean{KIP126.AdamsSynthetic.Standard.spectrumCategory}
\end{definition}

\begin{definition}[Recovery by lambda localization]
\label{frontier:AdamsSynthetic:68731f808d69}
\level{fine}
The full subcategory of $\mathrm{Syn}$ on objects for which $\lambda$ is invertible is reflective. Let $L$ be its reflector. There is an equivalence $\mathrm{Syn}[\lambda^{-1}]\simeq\mathrm{Sp}$ under which $L\nu$ is naturally isomorphic to the inverse equivalence.
\lean{KIP126.AdamsSynthetic.Standard.recovery}
\end{definition}

\begin{definition}[Homotopy of the first lambda quotient]
\label{frontier:AdamsSynthetic:740de17a7d73}
\level{fine}
For $X\in\mathcal C_0$ and $a,s,t\in\mathbb Z$, there is a $\mathbb Z$-linear isomorphism
\[
\pi_{t-s,t+a}\bigl((\Sigma^{0,a}\nu X)/\lambda\bigr)\cong E_2^{s,t}(X).
\]
\lean{KIP126.AdamsSynthetic.Standard.firstQuotient}
\end{definition}



\begin{definition}[Normalized synthetic lifts]
\label{frontier:AdamsSynthetic:8e642612876d}
\level{fine}
Let $X,Y\in\mathcal C_0$ and $f\colon X\to Y$. Put $\epsilon(f)=1$ if $f$ factors through the first stage of the target Adams tower, and put $\epsilon(f)=0$ otherwise. A normalized lift is a map
\[
\widehat f\colon\Sigma^{0,\epsilon(f)}\nu X\longrightarrow\nu Y
\]
such that $\widehat f\circ\lambda^{\epsilon(f)}=\nu(f)$, using the positive-target form of the power of $\lambda$.
\lean{KIP126.AdamsSynthetic.Standard.normalizedMap}
\end{definition}

\begin{definition}[Classical convergence and suspension]
\label{frontier:AdamsSynthetic:9a4b26f47efd}
\level{fine}
For $X\in\mathcal C_0$, the classical Adams sequence has canonical isomorphisms
\[
E_\infty^{s,t}(X)\cong\operatorname{gr}_F^s\pi_{t-s}X.
\]
The isomorphisms identify the class of an infinite cycle represented by a tower lift with the associated graded class of the projection of that same lift. The tower of $\Sigma X$ is isomorphic stagewise and layerwise to the suspension of the tower of $X$. These isomorphisms commute with tower maps, the stage-zero comparison, and layer inclusions; the layer connecting maps commute with a minus sign under the suspension-interchange comparison.
\lean{KIP126.AdamsSynthetic.Standard.classicalConvergence,KIP126.AdamsSynthetic.Standard.classicalSuspension}
\end{definition}

\begin{definition}[Classical and synthetic second-page comparisons]
\label{frontier:AdamsSynthetic:a5b004b4748d}
\level{fine}
For $X\in\mathcal C_0$, $a,s,t\in\mathbb Z$, and $k\in\mathbb N$, there are linear isomorphisms
\[
K_{X,a,k}\colon E_2^{s,t}(X)\cong E_2^{s,t,t+a-k}(\Sigma^{0,a}\nu X),\qquad
K_{S,k}\colon E_2^{s,t}(S)\cong E_2^{s,t,t-k}(S^{0,0}).
\]
They commute with maps between objects of $\mathcal C_0$. The sphere comparison agrees with the $X=S$, $a=0$ comparison under $\nu S\cong S^{0,0}$, and the map induced by $\lambda$ sends $K_{X,-1,k}(x)$ to $K_{X,0,k+1}(x)$.
\lean{KIP126.AdamsSynthetic.Standard.comparison}
\end{definition}

\begin{definition}[The synthetic stable category]
\label{frontier:AdamsSynthetic:a9091f3f5cf2}
\level{fine}
Equip $\mathrm{Syn}$ with a preadditive, monoidal, pretriangulated category structure, a zero object, and additive integer suspension. For $p\in\mathbb Z^2$, the fully faithful functor $\Sigma^p$ has comparisons $\Sigma^q\Sigma^p\cong\Sigma^{p+q}$, $\Sigma^{0,0}\cong\operatorname{id}$, and $\Sigma^{n,0}\cong\Sigma^n$. There is a natural transformation $\lambda\colon\Sigma^{0,-1}\to\operatorname{id}$, and natural comparisons $\Sigma^p(X\otimes Y)\cong(\Sigma^pX)\otimes Y$.
\lean{KIP126.AdamsSynthetic.Standard.syntheticCategory}
\end{definition}

\begin{definition}[The second page in Milnor coordinates]
\label{frontier:AdamsSynthetic:8b66ee46e41e}
\level{fine}
For $s,t\in\mathbb N$, there is a $\mathbb Z$-linear isomorphism $\Phi_{s,t}\colon E_2^{s,t}(S)\cong H^{s,t}_{\mathrm{Milnor}}$, where the group on the right is the cohomology of the normalized Milnor cobar complex.
\lean{KIP126.AdamsSynthetic.Standard.sphereE2_milnorEquiv}

\end{definition}

\begin{definition}[Synthetic functorial cofibers]
\label{frontier:AdamsSynthetic:c264cb4f3291}
\level{fine}
For $f\colon X\to Y$ in $\mathrm{Syn}$, choose a cofiber $C(f)$ and a distinguished triangle $X\xrightarrow f Y\to C(f)\to\Sigma X$. A commutative square induces a map of these cofibers commuting with inclusion and boundary.
\lean{KIP126.AdamsSynthetic.Standard.syntheticCofiber}
\end{definition}

\begin{definition}[Exact shifts and realization]
\label{frontier:AdamsSynthetic:d505f4e20090}
\level{fine}
The categories $\mathrm{Sp}$ and $\mathrm{Syn}$ are triangulated. Each bidegree shift on $\mathrm{Syn}$ is additive, commutes with integer suspension, and is exact for those suspension comparisons. Realization $R\colon\mathrm{Syn}\to\mathrm{Sp}$ commutes with suspension and is exact for its suspension comparison.
\lean{KIP126.AdamsSynthetic.Standard.exactness}
\end{definition}

\begin{definition}[The mod-two coefficient object]
\label{frontier:AdamsSynthetic:e9439d4312f5}
\level{fine}
Choose $H\mathbb F_2\in\mathrm{Sp}$ together with an additive isomorphism $\pi_0H\mathbb F_2\cong\mathbb Z/2$ and $\pi_nH\mathbb F_2=0$ for every nonzero integer $n$. Its coefficient unit $S\to H\mathbb F_2$ is the class corresponding to $1\in\mathbb Z/2$.
\lean{KIP126.AdamsSynthetic.Standard.coefficients}
\end{definition}

\label{frontier:AdamsSynthetic:eaf500baffc5}

Let $\mathrm{Syn}$ denote the type of synthetic spectra, in universe $\mathrm{Type}_1$.
\lean{KIP126.AdamsSynthetic.Standard.SyntheticSpectrum}

\begin{definition}[The sphere second-page algebra]
\label{frontier:AdamsSynthetic:f05463b82afe}
\level{fine}
Each group $E_2^{s,t}(S)$ is an $\mathbb F_2$-module. The sphere second page has a $\mathbb Z$-bilinear product
\[
E_2^p(S)\times E_2^q(S)\longrightarrow E_2^{p+q}(S)
\]
and a unit in degree $(0,0)$. Under the identifications of equal degree sums, the product is associative and the unit is both a left and a right identity.
\lean{KIP126.AdamsSynthetic.Standard.sphereAlgebra}
\end{definition}

\begin{definition}[Synthetic convergence]
\label{frontier:AdamsSynthetic:f083536a377e}
\level{fine}
For $Y\in\mathcal S_0$, there are canonical isomorphisms
\[
E_\infty^{s,t,w}(Y)\cong\operatorname{gr}_F^s\pi_{t-s,w}Y.
\]
The filtration agrees with the tower filtration after weight desuspension. If an infinite cycle is represented by a tower lift, its image under this isomorphism is the associated graded class of the projection of that same lift.
\lean{KIP126.AdamsSynthetic.Standard.convergence}
\end{definition}

\begin{definition}[The closed symmetric tensor structure]
\label{frontier:AdamsSynthetic:6a2402d27721}
\level{fine}
The monoidal category $\mathrm{Sp}$ is symmetric and closed. For each $W$, tensoring on the right with $W$ and taking the internal hom $F(W,-)$ commute with integer suspension and preserve distinguished triangles for these suspension comparisons.
\lean{KIP126.AdamsSynthetic.Standard.closedBackground}
\end{definition}











\begin{definition}[Filtered synthetic shift comparisons]
\label{frontier:AdamsSynthetic:63b3daf63e20}
\level{fine}
For $X\in\mathcal S_0$ and $p,\delta\in\mathbb Z^2$, there is an isomorphism between the filtered homotopy group in degree $p$ of $X$ and that in degree $p+\delta$ of $\Sigma^\delta X$. It identifies the ambient groups and every filtration stage.
\lean{KIP126.AdamsSynthetic.Standard.shift_filtered}
\end{definition}







\begin{definition}[Finite-page synthetic shift comparisons]
\label{frontier:AdamsSynthetic:f3257e955b3d}

For $X\in\mathcal S_0$, $p,\delta\in\mathbb Z^2$, $s\in\mathbb Z$, and $r\geq2$, there is an isomorphism
\[
E_r^{s,p_1+s,p_2}(X)\cong E_r^{s,(p+\delta)_1+s,(p+\delta)_2}(\Sigma^\delta X).
\]
\lean{KIP126.AdamsSynthetic.Standard.shift_page}
\end{definition}















\begin{theorem}[Detection, products, and exactness of the synthetic analogue]
\label{frontier:AdamsSynthetic:791f3c0e6a4a}
\level{fine}
For $X\in\mathcal C_0$, $a,s,t\in\mathbb Z$, and $x\in E_2^{s,t}(X)$, the image of $K_{X,a,0}(x)$ under the first-quotient inclusion detects the inverse image of $x$ in $\pi_{t-s,t+a}((\Sigma^{0,a}\nu X)/\lambda)$.

For a distinguished triangle $X\to Y\to Z\to\Sigma X$ whose mod-two homology is short exact in every degree, there is a boundary $\nu Z\to\Sigma\nu X$ making the images of its first two maps a distinguished synthetic triangle.

For $s,t\in\mathbb Z$, $k\in\mathbb N$, and $\alpha\in\pi_{t-s,t}S^{0,0}$, the class $K_{S,k}(q\alpha)$ detects $\lambda^k\alpha$, where $q\alpha$ denotes the classical $E_2$ class obtained from $\alpha$ by the first-quotient comparison.

For $s,t,s',t',k,l\in\mathbb N$, $x\in E_2^{s,t}(S)$, and $y\in E_2^{s',t'}(S)$, if $K_{S,k}(x)$ detects $\alpha\in\pi_{t-s,t-k}S^{0,0}$ and $K_{S,l}(y)$ detects $\beta\in\pi_{t'-s',t'-l}S^{0,0}$, then $K_{S,k+l}(xy)$ detects $\alpha\beta$ after regrading the degree sums. For $\alpha\in\pi_{m,n}S^{0,0}$ and $\beta\in\pi_{k,l}S^{0,0}$, where $m,n,k,l\in\mathbb Z$, one has $\alpha\beta=(-1)^{mk}\beta\alpha$ after the sum-degree identifications.
\lean{KIP126.AdamsSynthetic.Standard.first_quotient,KIP126.AdamsSynthetic.Standard.sphereProductCommutative,KIP126.AdamsSynthetic.Standard.nu_preserves_triangle,KIP126.AdamsSynthetic.Standard.multiplicationCompatible,KIP126.AdamsSynthetic.Standard.sphere_homotopy_lambda}
\end{theorem}



\begin{theorem}[Infinite filtration of maps between finite spectra]
\label{frontier:AdamsSynthetic:1ae83419acab}
\level{fine}
Let $X,Y$ be finite spectra and $f\colon X\to Y$. If $f$ factors through every integer stage of the Adams tower of $Y$, then $f$ is odd torsion.
\lean{KIP126.AdamsSynthetic.Standard.finite_map_separation}
\end{theorem}











\begin{theorem}[Shift comparison on ambient homotopy]
\label{frontier:AdamsSynthetic:b321b7897dca}
\level{fine}
Let $X\in\mathcal S_0$, $p,\delta\in\mathbb Z^2$, and $\alpha\in\pi_{p_1,p_2}X$. The ambient homomorphism of the filtered shift comparison sends $\alpha$ to its shifted homotopy class in $\pi_{(p+\delta)_1,(p+\delta)_2}(\Sigma^\delta X)$.
\lean{KIP126.AdamsSynthetic.Standard.shift_ambient_eq}
\end{theorem}











\begin{theorem}[First-quotient labels under positive weight shift]
\label{frontier:AdamsSynthetic:894dd8ba400f}
\level{fine}
Let $x\in E_2^{s,t}(X)$ and let $\epsilon$ identify the first lambda quotient of $\Sigma^{0,1}\nu X$ with $\Sigma^{0,1}(\nu X/\lambda)$. Suppose that $\epsilon$ commutes with the quotient inclusion. Shifting the first-quotient label of $x$ by $(0,1)$ and transporting it through $\epsilon^{-1}$ gives the fixed first-quotient label of $x$ in weight shift one.
\lean{KIP126.AdamsSynthetic.Standard.firstQuotient_positive_shift}

\begin{theorem}[First-quotient labels under classical suspension]
\label{frontier:AdamsSynthetic:837d666cb875}
\level{fine}
Let $x\in E_2^{s,t}(X)$ and let $sx\in E_2^{s,t+1}(\Sigma X)$ desuspend to $x$ under the fixed classical suspension comparison. Suppose that $\epsilon$ identifies the first lambda quotient of $\Sigma^{1,1}\nu X$ with the shifted quotient and commutes with the quotient inclusion. Transporting the label of $sx$ through the fixed nu-suspension comparison and $\epsilon$ agrees with shifting the label of $x$ by $(1,1)$.
\lean{KIP126.AdamsSynthetic.Standard.firstQuotient_suspension}
