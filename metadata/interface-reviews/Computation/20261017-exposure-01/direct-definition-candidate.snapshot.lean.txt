local syntax (name := directInlineDef) "inline_definition%" "public" "noncomputable" "def" ident ":" term ":=" term : command
@[local command_elab directInlineDef]
private meta def KIP126.Computation.Internal.elabDirectInlineDef : Lean.Elab.Command.CommandElab := fun stx => do
  match stx with
  | `(command| inline_definition% public
      noncomputable def $id:ident : $ty:term := $val:term) =>
    let name := (← Lean.getCurrNamespace) ++ id.getId
    Lean.Elab.Command.liftTermElabM do
      let type ← Lean.Meta.withFreshCache <| Lean.withoutExporting do
        Lean.instantiateMVars (← Lean.Elab.Term.withSynthesize <| Lean.Elab.Term.elabType ty)
      let value ← Lean.Meta.withFreshCache <| Lean.withoutExporting <|
        Lean.Elab.Term.elabTermAndSynthesize val none
      let value ← Lean.withoutExporting <| Lean.Meta.deltaExpand value Lean.isPrivateName (allowOpaque := true)
      let remaining := (type.getUsedConstants ++ value.getUsedConstants).filter Lean.isPrivateName
      unless remaining.isEmpty do throwError "Private constants remain: {remaining}"
      unless !type.hasSorry && !value.hasSorry do throwError "Unproved term in direct definition"
      let value ← Lean.Meta.mkExpectedTypeHint value type
      Lean.Meta.checkWithKernel value
      unless !(Lean.debug.skipKernelTC.get (← Lean.getOptions)) do
        throwError "Kernel checking must remain enabled"
      Lean.addDecl (.defnDecl {
        name := name
        levelParams := []
        type := type
        value := value
        hints := .regular 0
        safety := .safe }) (forceExpose := true)
      Lean.modifyEnv (Lean.addNoncomputable · name)
  | _ => throwError "Invalid directInlineDef syntax"


local syntax (name := kernelCheckedTerm) "kernel_checked% " term : term
@[local term_elab kernelCheckedTerm]
private meta def KIP126.Computation.Internal.elabKernelCheckedTerm : Lean.Elab.Term.TermElab :=
  fun stx expectedType? => Lean.Meta.withFreshCache <| Lean.withoutExporting do
    unless !(Lean.debug.skipKernelTC.get (← Lean.getOptions)) do
      throwError "Kernel checking must remain enabled"
    let value ← Lean.Elab.Term.elabTermAndSynthesize stx[1] none
    unless !value.hasSorry do throwError "Unproved term in kernel conversion"
    let value ← match expectedType? with
      | some expectedType => Lean.Meta.mkExpectedTypeHint value expectedType
      | none => pure value
    Lean.Meta.checkWithKernel value
    return value

/-- Direct evaluation of the typed monomial encoding; no parser fallback. -/
def polynomialOfPowers (powers : List (Generator × ℕ)) : Poly :=
  (powers.map fun p => MvPolynomial.X p.1 ^ p.2).prod

def relationPolynomial (terms : List (List (Generator × ℕ))) : Poly :=
  (terms.map polynomialOfPowers).sum

def definingIdeal : Ideal Poly :=
  Ideal.span {p | ∃ terms ∈ relationPowers, p = relationPolynomial terms}

abbrev E2 := Poly ⧸ definingIdeal

def projection : Poly →+* E2 := Ideal.Quotient.mk definingIdeal

def generator (i : Generator) : E2 := projection (MvPolynomial.X i)

def monomialDegree (m : Generator →₀ ℕ) : ℕ × ℕ :=
  m.sum fun i a => (a * (generatorDegree i).1, a * (generatorDegree i).2)

def homogeneousPart (s t : ℕ) : Submodule (ZMod 2) E2 :=
  Submodule.span (ZMod 2) {x | ∃ m : Generator →₀ ℕ,
    monomialDegree m = (s, t) ∧ x = projection (MvPolynomial.monomial m 1)}

abbrev E2At (s t : ℕ) := ↥(homogeneousPart s t)
