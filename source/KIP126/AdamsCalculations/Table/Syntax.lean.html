   8 namespace KIP126.Computation.Table.Raw
   9 public inductive Kind where
  10   | equation | reaches | boundaryBy | refutation
  11   deriving DecidableEq, BEq, Repr
  12 public structure Claim where
  13   spectrum : Spectrum
  14   kind : Kind
  15   r : Nat
  16   s : Nat
  17   t : Nat
  18   x : List Nat
  19   ts : Nat
  20   tt : Nat
  21   /-- Unused for `reaches`/`boundaryBy`; [] there is not a zero differential. -/
  22   y : List Nat
  23   origin : String
  24   record : Nat
  25   deriving DecidableEq, BEq, Repr
  26 /-- Syntax/degree checking only, not certification of any differential. -/
  27 public def Claim.valid (degrees : List Degree) (c : Claim) : Bool :=
  28   2 ≤ c.r && coordinatesValid degrees c.spectrum c.s c.t c.x &&
  29     match c.kind with
  30     | .reaches | .boundaryBy => true
  31     | .equation | .refutation => c.ts == c.s + c.r && c.tt == c.t + c.r - 1 &&
  32         coordinatesValid degrees c.spectrum c.ts c.tt c.y
  33 end KIP126.Computation.Table.Raw
