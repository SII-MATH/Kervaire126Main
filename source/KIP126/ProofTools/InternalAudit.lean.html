 112 /-! Trust audit for article-owned proof and construction responsibilities. Only the exact individually
 113 registered proof/construction holes are permitted; project axioms and synthetic
 114 (elaboration-error) holes are always rejected. SpectralSequence stays sorry-free. -/
 115 open Lean Elab Command
 116 set_option maxHeartbeats 0
 117 set_option maxRecDepth 8192
 118 run_cmd do
 119   let ledger ← Lean.ofExcept <| Json.parse proofObligationLedger
 120   let entries ← Lean.ofExcept <| ledger.getObjValAs? (Array Json) "obligations"
 121   let expected : Array Json ← entries.filterM fun (row : Json) => do
 122     let mod ← Lean.ofExcept <| row.getObjValAs? String "module"
 123     pure ((`KIP126.ProofTools).isPrefixOf mod.toName)
 124   let expectedNames : Array String ← expected.mapM fun (row : Json) => Lean.ofExcept <| row.getObjValAs? String "name"
 125   let env ← getEnv
 126   let mut checked : Nat := 0
 127   let mut holes : Array String := #[]
 128   for (name, info) in env.constants.toList do
 129     let some idx := env.getModuleIdxFor? name | continue
 130     let mod := env.header.moduleNames[idx.toNat]!
 131     unless (`KIP126.ProofTools).isPrefixOf mod do continue
 132     if info.isAxiom then throwError "Project axiom declaration: {name}"
 133     let body := info.value? (allowOpaque := true)
 134     if info.type.hasSyntheticSorry || body.any (·.hasSyntheticSorry) then
 135       throwError "Synthetic sorry from failed elaboration: {name}"
 136     if info.type.hasSorry || body.any (·.hasSorry) then
 137       holes := holes.push name.toString
 138       let some entry := expected.find? (fun e => e.getObjValAs? String "name" == .ok name.toString)
 139         | throwError "Unregistered proof/construction obligation: {name}"
 140       let expectedType ← Lean.ofExcept <| entry.getObjValAs? String "type_fingerprint"
 141       let expectedBody ← Lean.ofExcept <| entry.getObjValAs? String "body_fingerprint"
 142       unless expectedType == toString (hash (toString info.type)) &&
 143           expectedBody == toString (hash (toString body)) do
 144         throwError "Registered obligation changed; review its exact type/body: {name}"
 145     let axioms ← collectAxioms name
 146     let unexpected := axioms.filter fun n =>
 147       n != `propext && n != `Classical.choice && n != `Quot.sound && n != `sorryAx
 148     unless unexpected.isEmpty do
 149       throwError "Unexpected axiom dependency in {name}: {unexpected}"
 150     checked := checked + 1
 151   unless holes.qsort (· < ·) == expectedNames.qsort (· < ·) do
 152     throwError "Proof obligation registry has stale or missing declarations: {holes}"
 153   let originalGapDeclarations := #[
 154     "KIP126.ProofTools.Standard.generalizedLeibniz",
