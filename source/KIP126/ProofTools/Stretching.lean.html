  71     right
  72     by_contra hnot

  77     have module_map_zero {A B : ModuleCat ℤ} (f : A ⟶ B) : f (0 : A) = 0 := f.hom.map_zero
  78     have hln' : l ≤ s + n := by omega

  97         ((model.quotients.tower ((extensionSourceObject model X Y f).obj model.nu model.auxiliary)).rho
  98           (rEarlier-1) (rLater-1) (Nat.sub_le_sub_right hle 1))).eInftyMap
  99             (s+a,p.1+(s+a),p.2)).hom) := by
 100       intro hu
 101       obtain ⟨ε,hε,hres⟩ := (infinity_image_iff_lift (model.convergentMap
 102         (.quotient (rLater-1) (extensionSourceObject model X Y f))

 106       apply hlift

 134         (ModuleCat.subobjectModule _) ((model.syntheticTower.filtration Bearly).F (s+a+(n-a-b)+1) p) at hnext
 135       rw [module_map_zero,sub_zero,hdb] at hnext

 137         simpa only [SyntheticTowerPresentation.filtration, OrderIso.apply_symm_apply] using hnext
 138       exact hgnext hnext'
 139     refine ⟨hle,a,b,ha,haupper,hab,u,v,hu,?_,hv⟩
 140     apply (tower_extension_iff (model.convergentMap
 141       (.quotient (rEarlier-1) (extensionSourceObject model X Y f))
 142       (.quotient (rEarlier-1) (.nu Y)) (finiteExtensionMap model X Y f (rEarlier-1))) p (s+a) (n-a-b) u v).mpr
 143     refine ⟨by omega,da,db,?_,?_,?_⟩
 144     · change ((model.syntheticTower.filtration Aearly).F (s+a) p).arrow da ≫
 145         finiteExtensionMap model X Y f (rEarlier-1) =
 146         ((model.syntheticTower.filtration Bearly).F (s+a+(n-a-b)) p).arrow db
 147       rw [hda,hdb]
 148       rfl
 149     · exact ((infinityGradedIso
 150         (model.convergence (.quotient (rEarlier-1) (extensionSourceObject model X Y f))) p (s+a)).toLinearEquiv.apply_symm_apply _).symm
