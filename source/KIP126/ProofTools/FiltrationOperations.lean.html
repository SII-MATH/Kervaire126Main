  19 /-- Equality in the graded piece retains exactly the next-level ambiguity. -/
  20 theorem element_grade_equal_iff (F : Filtration A) (s : ℤ) (i : ι)
  21     (a b : (Subobject.underlying.obj (F.F s i) : ModuleCat R)) :
  22     F.toAssociatedGraded s i a = F.toAssociatedGraded s i b ↔
  23       (F.F s i).arrow a - (F.F s i).arrow b ∈
  24         (ModuleCat.subobjectModule (A i)) (F.F (s+1) i) := by
  25   simpa only [map_sub, sub_eq_zero] using
  26     F.toAssociatedGraded_eq_zero_iff s i (a-b)
  27 
  28 theorem graded_map_representative (f : FilteredMorphism F G) (s : ℤ) (i : ι)
  29     (a : (Subobject.underlying.obj (F.F s i) : ModuleCat R)) :
  30     f.inducedGrMap s i (F.toAssociatedGraded s i a) =
  31       G.toAssociatedGraded s i ((f.compat s i).choose a) := by
  32   have h : F.toAssociatedGraded s i ≫ f.inducedGrMap s i =
  33       (f.compat s i).choose ≫ G.toAssociatedGraded s i := by
  34     unfold Filtration.toAssociatedGraded FilteredMorphism.inducedGrMap
  35       Filtration.inducedAssocGradedMap
  36     dsimp only [cokernel.map]
  37     exact cokernel.π_desc _ _ _
  38   exact ConcreteCategory.congr_hom h a

  40 /-- Being in the image on the graded piece means lifting modulo the next level. -/
  41 theorem graded_image_iff_lift (f : FilteredMorphism F G) (s : ℤ) (i : ι)
  42     (x : (Subobject.underlying.obj (G.F s i) : ModuleCat R)) :
  43     (∃ z : F.associatedGraded s i,
  44       f.inducedGrMap s i z = G.toAssociatedGraded s i x) ↔
  45     ∃ a : A i, a ∈ (ModuleCat.subobjectModule (A i)) (F.F s i) ∧
  46       (G.F s i).arrow x - f.map i a ∈
  47         (ModuleCat.subobjectModule (B i)) (G.F (s+1) i) := by
  48   have hc (a : (Subobject.underlying.obj (F.F s i) : ModuleCat R)) :
  49       f.map i ((F.F s i).arrow a) = (G.F s i).arrow ((f.compat s i).choose a) :=
  50     (ConcreteCategory.congr_hom (f.compat s i).choose_spec a).symm
  51   constructor
  52   · rintro ⟨z, hz⟩
  53     haveI : Epi (F.toAssociatedGraded s i) := inferInstanceAs (Epi (cokernel.π _))
  54     obtain ⟨a, ha⟩ := (ModuleCat.epi_iff_surjective (F.toAssociatedGraded s i)).mp inferInstance z
  55     refine ⟨(F.F s i).arrow a, ⟨a, rfl⟩, ?_⟩
  56     rw [hc]
  57     apply (element_grade_equal_iff G s i x _).mp
  58     rw [← graded_map_representative f s i a, ha, hz]
  59   · rintro ⟨_, ⟨a, rfl⟩, ha⟩
  60     refine ⟨F.toAssociatedGraded s i a, ?_⟩
  61     rw [hc] at ha
  62     rw [graded_map_representative]
  63     exact ((element_grade_equal_iff G s i x _).mpr ha).symm
  64 
  65 /-- Graded injectivity reflects the next filtration level, without ambient
  66 injectivity or any assumption of separatedness. -/
  67 theorem next_filtration_mem_of_graded_mono (f : FilteredMorphism F G)
  68     (s : ℤ) (i : ι) [Mono (f.inducedGrMap s i)]
  69     (a : A i) (ha : a ∈ (ModuleCat.subobjectModule (A i)) (F.F s i))
  70     (hb : f.map i a ∈ (ModuleCat.subobjectModule (B i)) (G.F (s+1) i)) :
  71     a ∈ (ModuleCat.subobjectModule (A i)) (F.F (s+1) i) := by
  72   obtain ⟨x, rfl⟩ := ha
  73   have hy : (G.F s i).arrow ((f.compat s i).choose x) =
  74       f.map i ((F.F s i).arrow x) :=
  75     ConcreteCategory.congr_hom (f.compat s i).choose_spec x
  76   have hzero : G.toAssociatedGraded s i ((f.compat s i).choose x) = 0 := by
  77     apply (G.toAssociatedGraded_eq_zero_iff s i _).mpr
  78     rwa [hy]
  79   have hsource : F.toAssociatedGraded s i x = 0 := by
  80     apply (ModuleCat.mono_iff_injective (f.inducedGrMap s i)).mp inferInstance
  81     rw [map_zero, graded_map_representative, hzero]
  82   exact (F.toAssociatedGraded_eq_zero_iff s i x).mp hsource
  83 
  84 end
