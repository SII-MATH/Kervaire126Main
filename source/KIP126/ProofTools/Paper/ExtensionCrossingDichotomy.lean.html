  27 /-- This is the cycle-set difference branch, retaining the specified original
  28 target. It is separate from the classical crossing predicate of Definition 5.9. -/
  29 @[expose] public def FiniteExtensionNonliftableShortTo (X Y : ClassicalObject)
  30     (f : X.obj model.auxiliary ⟶ Y.obj model.auxiliary)
  31     (rEarlier rLater : ℕ) (n s t : ℤ) (x y)
  32     (Wearly : FiniteExtensionWitness model X Y f rEarlier n s t x y) : Prop :=
  33   ∃ a : ℕ, 0 < a ∧ (a:ℤ) ≤ n-normalizedExponent model.classicalTower f ∧
  34     ∃ x' : Ambient model X (s+a,(t-s)+(s+a)),
  35       x' ∈ cycles model X (((rEarlier-1:ℕ):ℤ)-a) (s+a,(t-s)+(s+a)) ∧
  36       x' ∉ cycles model X (((rLater-1:ℕ):ℤ)-a) (s+a,(t-s)+(s+a)) ∧
  37       ∃ z,
  38         HasInfinityRepresentative _ 2 _
  39           (finiteExtensionCorrectionLabel X Y f (rEarlier-1) s t a x') z ∧
  40         TowerExtensionEquation (model.convergentMap
  41           (.quotient (rEarlier-1) (extensionSourceObject model X Y f))
  42           (.quotient (rEarlier-1) (.nu Y)) (finiteExtensionMap model X Y f (rEarlier-1)))
  43           (t-s,t+normalizedExponent model.classicalTower f) (s+a) (n-a) z
  44           ((eqToHom (congrArg (fun d =>
  45             ((model.family.obj ((SyntheticObject.quotient (rEarlier-1) (.nu Y)).obj
  46               model.nu model.auxiliary)).sequence.ssData
  47               (d,(t-s)+d,t+normalizedExponent model.classicalTower f)).eInfty)
  48             (show s+n = (s+a)+(n-a) by omega))) Wearly.target)

  50 /-- A leading source that lifts through the actual square cannot support an
  51 essential short early extension when all late higher-source crossings vanish. -/
  52 private theorem essential_short_source_not_in_range
  53     (Al Bl Ae Be : SyntheticObject)
  54     (gl : Al.obj model.nu model.auxiliary ⟶ Bl.obj model.nu model.auxiliary)
  55     (ge : Ae.obj model.nu model.auxiliary ⟶ Be.obj model.nu model.auxiliary)
  56     (ra : Al.obj model.nu model.auxiliary ⟶ Ae.obj model.nu model.auxiliary)
  57     (rb : Bl.obj model.nu model.auxiliary ⟶ Be.obj model.nu model.auxiliary)
  58     (hsquare : ra ≫ ge = gl ≫ rb) (p : ℤ × ℤ) (s n u d : ℤ)
  59     (hn : 0 ≤ n) (hu : s < u) (htarget : u+d=s+n)
  60     (hcross : ExtensionNoCrossingRange (model.convergentMap Al Bl gl) p s n s)
  61     (xe : ((model.family.obj (Ae.obj model.nu model.auxiliary)).sequence.ssData
  62       (u,p.1+u,p.2)).eInfty)
  63     (ye : ((model.family.obj (Be.obj model.nu model.auxiliary)).sequence.ssData
  64       (u+d,p.1+(u+d),p.2)).eInfty)
  65     (he : TowerEssentialExtension (model.convergentMap Ae Be ge) p u d xe ye)
  66     (xl : ((model.family.obj (Al.obj model.nu model.auxiliary)).sequence.ssData
  67       (u,p.1+u,p.2)).eInfty)
  68     (hxl : (model.family.map ra).eInftyMap (u,p.1+u,p.2) xl = xe) : False := by
  69   let FL := model.syntheticTower.filtration (Al.obj model.nu model.auxiliary)
  70   let FE := model.syntheticTower.filtration (Ae.obj model.nu model.auxiliary)
  71   let GE := model.syntheticTower.filtration (Be.obj model.nu model.auxiliary)
  72   let IL := infinityGradedIso (model.convergence Al) p u
  73   let IE := infinityGradedIso (model.convergence Ae) p u
  74   haveI : Epi (FL.toAssociatedGraded u p) := inferInstanceAs (Epi (cokernel.π _))
  75   obtain ⟨ar,har⟩ := (ModuleCat.epi_iff_surjective (FL.toAssociatedGraded u p)).mp
  76     inferInstance (IL.hom xl)
  77   let α := (FL.F u p).arrow ar
  78   have hα : α ∈ model.syntheticTower.filtrationSubmodule (Al.obj model.nu model.auxiliary) u p := by
  79     have hm : α ∈ ModuleCat.subobjectModule _ (FL.F u p) := ⟨ar,rfl⟩
  80     simpa only [FL,SyntheticTowerPresentation.filtration,OrderIso.apply_symm_apply] using hm
  81   have hαnext := model.syntheticTower.filtrationSubmodule_antitone _ p
  82     (show s+1 ≤ u by omega) hα
  83   have hgl := image_correction_in_next (model.convergentMap Al Bl gl) p s n s
  84     (by omega) hcross α hαnext
  85     (map_preserves_filtration (model.convergentMap Al Bl gl) p s α
  86       (model.syntheticTower.filtrationSubmodule_antitone _ p (by omega) hα))
  87   have hαe := map_preserves_filtration (model.convergentMap Al Ae ra) p u α hα
  88   obtain ⟨ae,hae⟩ := filtration_representative_exists model.syntheticTower
  89     (Ae.obj model.nu model.auxiliary) p u (α≫ra) hαe
  90   have hg := representative_grade_naturality (model.convergentMap Al Ae ra) p u ar ae hae.symm
  91   rw [har] at hg
  92   have hc : IL.inv (IL.hom xl)=xl := IL.toLinearEquiv.symm_apply_apply xl
  93   rw [hc] at hg
  94   have hge : IE.inv (FE.toAssociatedGraded u p ae)=xe := hg.trans hxl
  95   have hax : FE.toAssociatedGraded u p ae = IE.hom xe :=
  96     (IE.toLinearEquiv.apply_symm_apply _).symm.trans (congrArg IE.hom hge)
  97   have hhigh : (α≫ra)≫ge ∈
  98       model.syntheticTower.filtrationSubmodule (Be.obj model.nu model.auxiliary) (u+d+1) p := by
  99     have hh := map_preserves_filtration (model.convergentMap Bl Be rb) p (s+n+1) (α≫gl) hgl
 100     simpa only [Category.assoc,hsquare,htarget] using hh
 101   have hlow := model.syntheticTower.filtrationSubmodule_antitone _ p
 102     (show u+d ≤ u+d+1 by omega) hhigh
 103   obtain ⟨be,hbe⟩ := filtration_representative_exists model.syntheticTower
 104     (Be.obj model.nu model.auxiliary) p (u+d) ((α≫ra)≫ge) hlow
 105   have hzero : GE.toAssociatedGraded (u+d) p be = 0 := by
 106     have hh := (element_grade_equal_iff GE (u+d) p be 0).mpr (by
 107       rw [hbe,map_zero,sub_zero]
 108       simpa only [GE,SyntheticTowerPresentation.filtration,OrderIso.apply_symm_apply] using hhigh)
 109     simpa only [map_zero] using hh
 110   apply ((tower_essential_iff_no_zero _ _ _ _ _ _ he.1).mp he)
 111   apply (tower_extension_iff (model.convergentMap Ae Be ge) p u d xe 0).mpr
 112   refine ⟨((tower_extension_iff _ _ _ _ _ _).mp he.1).1,ae,be,?_,hax,?_⟩
 113   · rw [hae,hbe]
 114   · simpa only [map_zero] using hzero

 116 private theorem infinity_map_eq_of_homotopy_eq
 117     (A B : SyntheticObject)
 118     (f g : A.obj model.nu model.auxiliary ⟶ B.obj model.nu model.auxiliary)
 119     (p : ℤ × ℤ) (s : ℤ)
 120     (hfg : ∀ a : BiHom p.1 p.2 (A.obj model.nu model.auxiliary), a≫f=a≫g)
 121     (z : ((model.family.obj (A.obj model.nu model.auxiliary)).sequence.ssData
 122       (s,p.1+s,p.2)).eInfty) :
 123     (model.family.map f).eInftyMap (s,p.1+s,p.2) z =
 124       (model.family.map g).eInftyMap (s,p.1+s,p.2) z := by
 125   let FA := model.syntheticTower.filtration (A.obj model.nu model.auxiliary)
 126   let FB := model.syntheticTower.filtration (B.obj model.nu model.auxiliary)

 129   obtain ⟨ar,har⟩ := (ModuleCat.epi_iff_surjective (FA.toAssociatedGraded s p)).mp
 130     inferInstance (IA.hom z)
 131   let α := (FA.F s p).arrow ar
 132   have hα : α ∈ model.syntheticTower.filtrationSubmodule (A.obj model.nu model.auxiliary) s p := by
 133     have hh : α ∈ ModuleCat.subobjectModule _ (FA.F s p) := ⟨ar,rfl⟩

 137     (map_preserves_filtration (model.convergentMap A B f) p s α hα)
 138   have hf := representative_grade_naturality (model.convergentMap A B f) p s ar br hbr.symm
 139   have hg := representative_grade_naturality (model.convergentMap A B g) p s ar br

 141   rw [har] at hf hg
 142   have hc : IA.inv (IA.hom z)=z := IA.toLinearEquiv.symm_apply_apply z
 143   rw [hc] at hf hg
 144   exact hf.symm.trans hg
