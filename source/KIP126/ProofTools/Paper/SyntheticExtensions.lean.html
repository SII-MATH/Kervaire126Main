 248 /-- Corollary 4.10, absolute target: every λ-weight 0≤a≤n,
 249 including both r≥n+1 and r<n+1, gives the actual δ_n extension equation.
 250 The endpoint a=n is included and has both labels zero. The source and
 251 target are labels in the fixed Adams systems. -/
 252 public theorem syntheticDeltaExtension_absolute (X : ClassicalObject)
 253     (n k a : ℕ) (hn : 0<n) (ha : a≤n) (hnk : n≤a+k) (s t : ℤ)
 254     (x : Ambient model X (s,t)) (y : Ambient model X (s+k+1,t+k))
 255     (hd : HasDifferential (model.classical.obj (X.obj model.auxiliary))
 256       (k+1) (s,t) (s+k+1,t+k) x y)
 257     (hy : y∈permanentCycles model X (s+k+1,t+k)) :
 258     ∃ ex ey,
 259       HasInfinityRepresentative _ 2 (s,(t-s)+s,t-a)
 260         (syntheticDeltaSourceLabel X n s t a x) ex ∧
 261       HasInfinityRepresentative _ 2 (s+(k+1),(t-s)+(s+(k+1)),t-a)
 262         (syntheticDeltaNuLabel X n k a s t hnk y) ey ∧
 263       TowerExtensionEquation (model.convergentMap (.quotient n (.nu X))
 264         (.shift (1,-(n:ℤ)) (.nu X)) (syntheticAbsoluteBoundary X n))
 265         (t-s,t-a) s (k+1) ex ey := by
 266   obtain ⟨b,hb,hδ⟩ := weighted_absolute_boundary_closed X n k a hn ha hnk s t x y hd hy
 267   have hb' := detects_transport (show (s,t,t-a)=(s,(t-s)+s,t-a) by simp) _ _ hb
 268   have hδ' := detects_transport
 269     (show (s+k+1,t+k,t-a+n)=(s+(k+1),(t-s)-1+(s+(k+1)),t-a+n) by ext <;> dsimp <;> omega)
 270     _ _ hδ
 271   have hδ' := boundary_detects_shifted (.nu X) n (t-s) (t-a) (s+(k+1)) b _ (by
 272     convert hδ' using 1
 273     simp only [SyntheticObject.obj,regrade_precomp,←Category.assoc,eqToHom_trans])
 274   have he := extension_equation_of_detected_pair
 275     (model.convergentMap (.quotient n (.nu X)) (.shift (1,-(n:ℤ)) (.nu X))
 276       (syntheticAbsoluteBoundary X n)) (t-s,t-a) s (k+1) (by omega)
 277       (syntheticDeltaSourceLabel X n s t a x) _ b (b≫syntheticAbsoluteBoundary X n)
 278       (by exact hb') (by
 279           simpa only [syntheticAbsoluteBoundary, absoluteBoundary,Operations.biSuspensionIso,
 280             SyntheticObject.obj] using hδ') rfl
 281   exact he

 314 /-- Proposition 4.6 and Corollary 4.10 for finite m: the map is the actual
 315 δ_(n,m), not a separately chosen extension relation. No bound r≤m is imposed;
 316 the out-of-strip target is consequently zero. -/
 317 public theorem syntheticDeltaExtension_finite (X : ClassicalObject)
 318     (n m k a : ℕ) (hn : 0<n) (ha : a≤n) (hnm : n<m) (hnk : n≤a+k) (s t : ℤ)
 319     (x : Ambient model X (s,t)) (y : Ambient model X (s+k+1,t+k))
 320     (hd : HasDifferential (model.classical.obj (X.obj model.auxiliary))
 321       (k+1) (s,t) (s+k+1,t+k) x y)
 322     (hy : y∈permanentCycles model X (s+k+1,t+k)) :
 323     ∃ ex ey,
 324       HasInfinityRepresentative _ 2 (s,(t-s)+s,t-a)
 325         (syntheticDeltaSourceLabel X n s t a x) ex ∧
 326       HasInfinityRepresentative _ 2 (s+(k+1),(t-s)+(s+(k+1)),t-a)
 327         (syntheticDeltaFiniteLabel X n m k a s t hnk y) ey ∧
 328       TowerExtensionEquation (model.convergentMap (.quotient n (.nu X))
 329         (.shift (1,-(n:ℤ)) (.quotient (m-n) (.nu X)))
 330         (syntheticFiniteBoundary X n m (by omega) hnm))
 331         (t-s,t-a) s (k+1) ex ey := by
 332   obtain ⟨b,hb,hδ⟩ := weighted_absolute_boundary_closed X n k a hn ha hnk s t x y hd hy
 333   have hb' := detects_transport (show (s,t,t-a)=(s,(t-s)+s,t-a) by simp) _ _ hb
 334   have hδ' := detects_transport
