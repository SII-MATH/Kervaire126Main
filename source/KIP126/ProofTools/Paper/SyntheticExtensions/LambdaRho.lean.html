  51 private theorem not_essential_of_source_injective
  52     (M : ConvergentMap E g ca.toConvergence cb.toConvergence)
  53     (p : ℤ × ℤ) (s d : ℤ) (hd : 0 < d)
  54     (hf : Function.Injective ((E.functor.map g).eInftyMap (s,p.1+s,p.2)).hom)
  55     (x : ((E.obj A).sequence.ssData (s,p.1+s,p.2)).eInfty)
  56     (y : ((E.obj B).sequence.ssData (s+d,p.1+(s+d),p.2)).eInfty) :
  57     ¬ TowerEssentialExtension M p s d x y := by
  58   intro h
  59   obtain ⟨_,ar,br,hab,hx,_⟩ := (tower_extension_iff M p s d x y).mp h.1
  60   have ha : ((P.filtration A).F s p).arrow ar ∈ P.filtrationSubmodule A s p := by
  61     have hh : ((P.filtration A).F s p).arrow ar ∈
  62         ModuleCat.subobjectModule _ ((P.filtration A).F s p) := ⟨ar,rfl⟩
  63     simpa only [SyntheticTowerPresentation.filtration,OrderIso.apply_symm_apply] using hh
  64   have hb : ((P.filtration A).F s p).arrow ar ≫ g ∈ P.filtrationSubmodule B (s+1) p := by
  65     apply P.filtrationSubmodule_antitone B p (show s+1 ≤ s+d by omega)
  66     have hh : ((P.filtration A).F s p).arrow ar ≫ g ∈
  67         ModuleCat.subobjectModule _ ((P.filtration B).F (s+d) p) := ⟨br,hab.symm⟩
  68     simpa only [SyntheticTowerPresentation.filtration,OrderIso.apply_symm_apply] using hh
  69   have hn := next_filtration_of_infinity_injective M p s hf _ ha hb
  70   have hz : (P.filtration A).toAssociatedGraded s p ar = 0 := by
  71     have hh := (element_grade_equal_iff (P.filtration A) s p ar 0).mpr (by
  72       simpa only [map_zero,sub_zero,SyntheticTowerPresentation.filtration,
  73         OrderIso.apply_symm_apply] using hn)
  74     simpa only [map_zero] using hh
  75   have hx0 : x = 0 := (infinityGradedIso ca p s).toLinearEquiv.injective
  76     (hx.symm.trans (hz.trans (map_zero _).symm))
  77   subst x
  78   exact zero_source_not_essential M p s d (by omega) y h

  80 private theorem not_essential_of_target_surjective
  81     (M : ConvergentMap E g ca.toConvergence cb.toConvergence)
  82     (p : ℤ × ℤ) (s d : ℤ) (hd : 0 < d)
  83     (hf : Function.Surjective ((E.functor.map g).eInftyMap (s+d,p.1+(s+d),p.2)).hom)
  84     (x : ((E.obj A).sequence.ssData (s,p.1+s,p.2)).eInfty)
  85     (y : ((E.obj B).sequence.ssData (s+d,p.1+(s+d),p.2)).eInfty) :
  86     ¬ TowerEssentialExtension M p s d x y := by
  87   intro h
  88   obtain ⟨_,ar,br,hab,hx,hy⟩ := (tower_extension_iff M p s d x y).mp h.1
  89   have hrange : (infinityGradedIso cb p (s+d)).inv
  90       ((P.filtration B).toAssociatedGraded (s+d) p br) ∈
  91         Set.range ((E.functor.map g).eInftyMap (s+d,p.1+(s+d),p.2)).hom :=
  92     hf _
  93   obtain ⟨c,hc,hbc⟩ := (infinity_image_iff_lift M p (s+d) br).mp hrange
  94   have hcs := P.filtrationSubmodule_antitone A p (show s ≤ s+d by omega) hc
  95   have hcs1 := P.filtrationSubmodule_antitone A p (show s+1 ≤ s+d by omega) hc
  96   let a : BiHom p.1 p.2 A := ((P.filtration A).F s p).arrow ar
  97   have ha : a ∈ P.filtrationSubmodule A s p := by
  98     have hh : a ∈ ModuleCat.subobjectModule _ ((P.filtration A).F s p) := ⟨ar,rfl⟩
  99     simpa only [SyntheticTowerPresentation.filtration,OrderIso.apply_symm_apply] using hh
 100   obtain ⟨ar',har'⟩ := filtration_representative_exists P A p s (a-c)
 101     ((P.filtrationSubmodule A s p).sub_mem ha hcs)
 102   have hnew : (a-c) ≫ g ∈ P.filtrationSubmodule B (s+d+1) p := by
 103     simpa only [a,Preadditive.sub_comp,hab] using hbc
 104   obtain ⟨br',hbr'⟩ := filtration_representative_exists P B p (s+d) ((a-c)≫g)
 105     (P.filtrationSubmodule_antitone B p (show s+d≤s+d+1 by omega) hnew)
 106   apply ((tower_essential_iff_no_zero M p s d x y h.1).mp h)
 107   apply (tower_extension_iff M p s d x 0).mpr
 108   refine ⟨by omega,ar',br',by rw [har',hbr'],?_,?_⟩
 109   · have hdiff : ((P.filtration A).F s p).arrow ar -
 110         ((P.filtration A).F s p).arrow ar' ∈
 111         ModuleCat.subobjectModule _ ((P.filtration A).F (s+1) p) := by
 112       rw [har']
 113       simpa only [a,sub_sub_cancel,SyntheticTowerPresentation.filtration,
 114         OrderIso.apply_symm_apply] using hcs1
 115     exact ((element_grade_equal_iff (P.filtration A) s p ar ar').mpr hdiff).symm.trans hx
 116   · have hdiff : ((P.filtration B).F (s+d) p).arrow br' -
 117         ((P.filtration B).F (s+d) p).arrow
 118           (0 : (Subobject.underlying.obj ((P.filtration B).F (s+d) p) : ModuleCat.{v} ℤ)) ∈
 119         ModuleCat.subobjectModule _ ((P.filtration B).F (s+d+1) p) := by
 120       rw [hbr',map_zero,sub_zero]
 121       simpa only [SyntheticTowerPresentation.filtration,OrderIso.apply_symm_apply] using hnew
 122     have hh := (element_grade_equal_iff (P.filtration B) (s+d) p br' 0).mpr hdiff
 123     simpa only [map_zero] using hh

 311 /-- AimPaper Proposition 4.2 for `0 < n < m < ∞`. The first two clauses
 312 exclude every positive-length essential λ/ρ extension; the last two clauses
 313 identify their length-zero equations with their actual E∞ maps; the final
 314 two clauses exclude crossings of those length-zero equations.
 315 A crossing would require the empty interval `s < u ≤ v ≤ s`. -/
 316 public theorem lambda_rho_extension_collapse_finite (X : ClassicalObject)
 317     (n m : ℕ) (hn : 0 < n) (hnm : n < m) :
 318     (∀ (p : ℤ × ℤ) (s d : ℤ)
 319       (x : ((model.family.obj ((SyntheticObject.shift (0,-(n:ℤ))
 320         (.quotient (m-n) (.nu X))).obj model.nu model.auxiliary)).sequence.ssData
 321         (s,p.1+s,p.2)).eInfty)
 322       (y : ((model.family.obj ((SyntheticObject.quotient m (.nu X)).obj model.nu model.auxiliary)).sequence.ssData
 323         (s+d,p.1+(s+d),p.2)).eInfty),
 324       TowerEssentialExtension
 325         (model.convergentMap (.shift (0,-(n:ℤ)) (.quotient (m-n) (.nu X))) (.quotient m (.nu X))
 326           ((model.quotients.tower _).triangle hn hnm).lambdaMap) p s d x y → d=0) ∧
 327     (∀ (p : ℤ × ℤ) (s d : ℤ)
 328       (x : ((model.family.obj ((SyntheticObject.quotient m (.nu X)).obj model.nu model.auxiliary)).sequence.ssData
 329         (s,p.1+s,p.2)).eInfty)
 330       (y : ((model.family.obj ((SyntheticObject.quotient n (.nu X)).obj model.nu model.auxiliary)).sequence.ssData
 331         (s+d,p.1+(s+d),p.2)).eInfty),
 332       TowerEssentialExtension
 333         (model.convergentMap (.quotient m (.nu X)) (.quotient n (.nu X))
 334           ((model.quotients.tower _).rho n m (Nat.le_of_lt hnm))) p s d x y → d=0) ∧
 335     (∀ (p : ℤ × ℤ) (s : ℤ)
 336       (x : ((model.family.obj ((SyntheticObject.shift (0,-(n:ℤ))
 337         (.quotient (m-n) (.nu X))).obj model.nu model.auxiliary)).sequence.ssData
