  44 private theorem differential_target_permanent
  45     (X : ClassicalObject) (k : ℕ) (hk : 0<k) (s t : ℤ)
  46     (x : Ambient model X (s,t)) (y : Ambient model X (s+k+1,t+k))
  47     (hd : HasDifferential (model.classical.obj (X.obj model.auxiliary))
  48       (k+1) (s,t) (s+k+1,t+k) x y) :
  49     y∈permanentCycles model X (s+k+1,t+k) := by
  50   have hy := differential_target_boundary X (k+1) (by omega) _ _ x y hd

  54     (E.B_mono le_top).trans (E.B_le_Z ⊤)

  56   change (Subobject.ofLE _ _ h ≫ cycleLabel model X ⊤ (s+k+1,t+k)) b = _
  57   simp only [cycleLabel,boundaryLabel,E,←Category.assoc,Subobject.ofLE_comp_ofLE]

  59 private theorem nonzero_differential_target_not_boundary
  60     (X : ClassicalObject) (r b : ℤ) (p q : ℤ×ℤ)
  61     (hb : 1≤b) (hbr : b≤r-1)
  62     (x : Ambient model X p) (y : Ambient model X q)
  63     (hd : HasNonzeroDifferential (model.classical.obj (X.obj model.auxiliary)) r p q x y) :
  64     y∉boundaries model X b q := by
  65   intro hy
  66   obtain ⟨hdeg,xr,yr,hx,hyrep,heq,hnonzero⟩ := hd
  67   have hd0 := boundary_target_zero model X r b p q hx.1 hbr x y hy
  68     ⟨hdeg,xr,yr,hx,hyrep,heq⟩
  69   obtain ⟨_,xr0,yr0,hx0,hy0,heq0⟩ := hd0
  70   have hxx := hx.unique hx0
  71   have hyy : yr=yr0 := heq.symm.trans (by simpa only [hxx] using heq0)
  72   have hz : yr0=0 := hy0.unique (RepresentsOnPage.zero (by omega))
  73   exact hnonzero (hyy.trans hz)

 162 /-- Actual synthetic crossing of the absolute boundary for the quotient
 163 `νX/λ^(page-1)`, in the fixed stem and weight of a classical differential.
 164 The essentiality and all source and target grades are taken in that one ESS. -/
 165 @[expose] public def DeltaHasCrossing (X : ClassicalObject) (page : ℕ) (r s t : ℤ) : Prop :=
 166   ∃ u v : ℤ, s < u ∧ u ≤ v ∧ v ≤ s+r ∧ ∃ x y,
 167     TowerEssentialExtension (model.convergentMap
 168       (.quotient (page-1) (.nu X)) (.shift (1,-((page-1:ℕ):ℤ)) (.nu X))
 169       (XModLambdaN.proj (model.nu.functor.obj (X.obj model.auxiliary)) (page-1) ≫
 170         (Operations.biSuspensionIso (page-1:ℕ)
 171           (model.nu.functor.obj (X.obj model.auxiliary))).inv))
 172       (t-s,t) u (v-u) x y

 174 private theorem delta_no_crossing_implies_classical
 175     (X : ClassicalObject) (page : ℕ) (r s t : ℤ)
 176     (hpage : 2 ≤ page) (hpr : (page:ℤ) ≤ r)
 177     (hnc : TowerNoCrossing (model.convergentMap
 178       (.quotient (page-1) (.nu X)) (.shift (1,-((page-1:ℕ):ℤ)) (.nu X))
 179       (absoluteBoundary (model.nu.functor.obj (X.obj model.auxiliary)) (page-1)))
 180       (t-s,t) s r) :
 181     NoCrossingOn model X r page (s,t) := by
 182   refine ⟨by omega,hpr,?_⟩
 183   rintro ⟨a,b,ha,har,hb,hbr,x,y,hd⟩
 184   let q : ℕ := page-1
 185   let i : ℕ := a.toNat
 186   let k : ℕ := (r-a-b-1).toNat
 187   have hqZ : (q:ℤ)=(page:ℤ)-1 := by dsimp [q]; omega
 188   have hiZ : (i:ℤ)=a := by dsimp [i]; omega
 189   have hkZ : (k:ℤ)=r-a-b-1 := by dsimp [k]; omega
 190   have H := crossing_differential_contradicts_delta_no_crossing X q i k
 191     (by omega) (by omega) (by omega) r s t (by omega) (by omega) hnc
 192   have hs : s+a+(k:ℤ)+1=s+r-b := by omega
 193   have ht : t+a+(k:ℤ)=t+r-b-1 := by omega
 194   have hl : (k:ℤ)+1=r-a-b := by omega
 195   rw [hiZ,hs,ht,hl] at H
 196   exact H x y hd

 198 /-- AimPaper Proposition 4.16 (`prop:cross-dr-En`), with both directions
 199 and the full classical range `2 ≤ page ≤ r`. The quotient exponent is
 200 `page-1`, so `page=n+1` in the paper. -/
 201 public theorem classical_crossing_iff_delta_crossing
 202     (X : ClassicalObject) (page : ℕ) (r s t : ℤ)
 203     (hpage : 2 ≤ page) (hpr : (page:ℤ) ≤ r) :
 204     HasCrossingOn model X r page (s,t) ↔ DeltaHasCrossing X page r s t := by
 205   classical
 206   have hdef : DeltaHasCrossing X page r s t ↔
 207       ¬TowerNoCrossing (model.convergentMap
 208         (.quotient (page-1) (.nu X)) (.shift (1,-((page-1:ℕ):ℤ)) (.nu X))
 209         (absoluteBoundary (model.nu.functor.obj (X.obj model.auxiliary)) (page-1)))
 210         (t-s,t) s r := by
 211     constructor
 212     · rintro ⟨u,v,hu,huv,hv,x,y,hess⟩ hn
 213       exact hn u v hu huv hv x y
 214         (by simpa only [Operations.biSuspensionIso,absoluteBoundary] using hess)
 215     · intro hn
 216       by_contra hnot
 217       apply hn
 218       intro u v hu huv hv x y hess
 219       apply hnot
 220       exact ⟨u,v,hu,huv,hv,x,y,
 221         by simpa only [Operations.biSuspensionIso,absoluteBoundary] using hess⟩
 222   rw [hdef]
 223   constructor
 224   · intro hcross hnc
 225     exact (delta_no_crossing_implies_classical X page r s t hpage hpr hnc).2.2 hcross
 226   · intro hcross
 227     by_contra hnc
 228     exact hcross (classical_noCrossing_absolute_boundary X page r s t ⟨by omega,hpr,hnc⟩)
