  26 /-- Among all higher source representatives hitting a fixed nonzero target,
  27 choose one of maximal source filtration. It gives an essential extension. -/
  28 private theorem tower_essential_shorter_same_target
  29     (M : ConvergentMap F g cx.toConvergence cy.toConvergence)
  30     (p : ℤ × ℤ) (s n : ℤ) (hn : 0<n)
  31     (x : ((F.obj X).sequence.ssData (s,p.1+s,p.2)).eInfty)
  32     (y : ((F.obj Y).sequence.ssData (s+n,p.1+(s+n),p.2)).eInfty)
  33     (h : TowerExtensionEquation M p s n x y)
  34     (hy : y≠0) (hne : ¬TowerEssentialExtension M p s n x y) :
  35     ∃ a : ℕ, 0<a ∧ (a:ℤ)≤n ∧
  36       ∃ z : ((F.obj X).sequence.ssData (s+a,p.1+(s+a),p.2)).eInfty,
  37         TowerEssentialExtension M p (s+a) (n-a) z
  38           ((eqToHom (congrArg (fun d =>
  39             ((F.obj Y).sequence.ssData (d,p.1+d,p.2)).eInfty)
  40             (show s+n=(s+a)+(n-a) by omega))) y) := by
  41   classical
  42   let Pred (a : ℕ) : Prop := 0<a ∧ a≤n.toNat ∧ ∃
  43     (ar : (Subobject.underlying.obj ((P.filtration X).F (s+a) p) : ModuleCat.{v} ℤ))
  44     (br : (Subobject.underlying.obj ((P.filtration Y).F (s+n) p) : ModuleCat.{v} ℤ)),
  45     ((P.filtration X).F (s+a) p).arrow ar ≫ g =
  46       ((P.filtration Y).F (s+n) p).arrow br ∧
  47     (P.filtration Y).toAssociatedGraded (s+n) p br =
  48       (infinityGradedIso cy p (s+n)).hom y
  49   obtain ⟨_,ar1,br1,heq1,hbr1⟩ :=
  50     (tower_inessential_iff_higher_source_same_target M p s n x y h).mp hne
  51   have hp1 : Pred 1 := ⟨by decide,by omega,ar1,br1,heq1,hbr1⟩
  52   let a := Nat.findGreatest Pred n.toNat
  53   obtain ⟨ha,han,ar,br,heq,hbr⟩ := Nat.findGreatest_spec (P:=Pred)
  54     (show 1≤n.toNat by omega) hp1
  55   let I := infinityGradedIso cx p (s+a)
  56   let z := I.inv ((P.filtration X).toAssociatedGraded (s+a) p ar)
  57   have haz : (P.filtration X).toAssociatedGraded (s+a) p ar = I.hom z :=
  58     (I.toLinearEquiv.apply_symm_apply _).symm
  59   have he : TowerExtensionEquation M p (s+a) (n-a) z
  60       ((eqToHom (congrArg (fun d =>
  61         ((F.obj Y).sequence.ssData (d,p.1+d,p.2)).eInfty)
  62         (show s+n=(s+a)+(n-a) by omega))) y) :=
  63     (tower_extension_reindexed M p (s+a) (n-a) (s+n) (by omega) z y).mpr
  64       ⟨by omega,ar,br,heq,haz,hbr⟩
  65   refine ⟨a,ha,by omega,z,?_⟩
  66   by_contra hnot
  67   have he0 := (tower_inessential_iff_zero_source_same_target M p (s+a) (n-a)
  68     z _ he).mp hnot
  69   obtain ⟨_,ar0,br0,heq0,har0,hbr0⟩ :=
  70     (tower_extension_reindexed M p (s+a) (n-a) (s+n) (by omega) 0 y).mp he0
  71   have hzero : (P.filtration X).toAssociatedGraded (s+a) p ar0=0 := by
  72     simpa only [map_zero] using har0
  73   have hmem : ((P.filtration X).F (s+a) p).arrow ar0 ∈
  74       P.filtrationSubmodule X (s+a+1) p := by
  75     have hh := ((P.filtration X).toAssociatedGraded_eq_zero_iff (s+a) p ar0).mp hzero
  76     simpa only [SyntheticTowerPresentation.filtration,OrderIso.apply_symm_apply] using hh
  77   have hanext : a+1≤n.toNat := by
  78     by_contra hgt
  79     have hae : (a:ℤ)=n := by omega
  80     have hhigh := map_preserves_filtration M p (s+a+1)
  81       (((P.filtration X).F (s+a) p).arrow ar0) hmem
  82     have htarget : ((P.filtration Y).F (s+n) p).arrow br0 ∈
  83         ModuleCat.subobjectModule _ ((P.filtration Y).F (s+n+1) p) := by
  84       rw [←heq0]
  85       simpa only [hae,SyntheticTowerPresentation.filtration,OrderIso.apply_symm_apply] using hhigh
  86     have hbzero := ((P.filtration Y).toAssociatedGraded_eq_zero_iff (s+n) p br0).mpr htarget
  87     apply hy
  88     apply (infinityGradedIso cy p (s+n)).toLinearEquiv.injective
  89     change (infinityGradedIso cy p (s+n)).hom y =
  90       (infinityGradedIso cy p (s+n)).hom 0
  91     rw [←hbr0,map_zero,hbzero]
  92   have hmem' : ((P.filtration X).F (s+a) p).arrow ar0 ∈
  93       P.filtrationSubmodule X (s+((a+1:ℕ):ℤ)) p := by
  94     simpa only [Nat.cast_add,Nat.cast_one,add_assoc] using hmem
  95   obtain ⟨arn,harn⟩ := filtration_representative_exists P X p (s+((a+1:ℕ):ℤ))
  96     (((P.filtration X).F (s+a) p).arrow ar0) hmem'
  97   have hpnext : Pred (a+1) := by
  98     refine ⟨by omega,hanext,arn,br0,?_,hbr0⟩
  99     rw [harn]
 100     exact heq0
 101   have hamax := Nat.le_findGreatest (P:=Pred) hanext hpnext
 102   change a+1≤a at hamax
 103   omega
 104 
 105 end Tower

 110 /-- The actual finite normalized map admits an essential higher-source
 111 extension to the same nonzero target. Its source is labelled by a classical
 112 cycle in the precise finite quotient window. -/
 113 private theorem finiteExtension_essential_shorter_source (X Y : ClassicalObject)
 114     (f : X.obj model.auxiliary ⟶ Y.obj model.auxiliary)
 115     (r : ℕ) (n s t : ℤ) (x y)
 116     (W : FiniteExtensionWitness model X Y f r n s t x y)
 117     (hy : W.target≠0) (hne : ¬W.Essential) :
 118     ∃ a : ℕ, 0<a ∧
 119       (a:ℤ)≤n-normalizedExponent model.classicalTower f ∧
 120       ∃ x' : Ambient model X (s+a,(t-s)+(s+a)),
 121         x' ∈ cycles model X (((r-1:ℕ):ℤ)-(a:ℤ)) (s+a,(t-s)+(s+a)) ∧
 122         ∃ z,
 123           HasInfinityRepresentative _ 2 _
 124             (finiteExtensionCorrectionLabel X Y f (r-1) s t a x') z ∧
 125           TowerEssentialExtension (model.convergentMap
 126             (.quotient (r-1) (extensionSourceObject model X Y f))
 127             (.quotient (r-1) (.nu Y)) (finiteExtensionMap model X Y f (r-1)))
 128             (t-s,t+normalizedExponent model.classicalTower f) (s+a) (n-a) z
 129             ((eqToHom (congrArg (fun d =>
 130               ((model.family.obj ((SyntheticObject.quotient (r-1) (.nu Y)).obj
 131                 model.nu model.auxiliary)).sequence.ssData
 132                 (d,(t-s)+d,t+normalizedExponent model.classicalTower f)).eInfty)
 133               (show s+n=(s+a)+(n-a) by omega))) W.target) := by
 134   let p : ℤ × ℤ := (t-s,t+normalizedExponent model.classicalTower f)
 135   let M := model.convergentMap
 136     (.quotient (r-1) (extensionSourceObject model X Y f))
 137     (.quotient (r-1) (.nu Y)) (finiteExtensionMap model X Y f (r-1))
 138   have hstrict := inessential_length_strict X Y f r n s t x y W hy hne
 139   have hn : 0<n := by omega
 140   obtain ⟨a,ha,han,z,hess⟩ := tower_essential_shorter_same_target M p s n hn
 141     W.source W.target W.equation hy hne
 142   have hq : 0<r-1 := by have := W.page_ge_two; omega
 143   have htargetbound : s+(a:ℤ)+normalizedExponent model.classicalTower f ≤ s+n := by
 144     have H := finite_crossing_target_bound X Y f (r-1) hq s t (s+a) (s+n)
 145       (by omega) (by omega)
 146     have hlen : (s+n)-(s+(a:ℤ))=n-a := by omega
 147     rw [hlen] at H
 148     exact H z _ hess
 149   have han' : (a:ℤ)≤n-normalizedExponent model.classicalTower f := by omega
 150   have haq : a<r-1 := by have := W.length_upper; have := W.page_ge_two; omega
 151   obtain ⟨x',hx',hz⟩ := correction_label_exists X Y f (r-1) hq s t a haq z
 152   exact ⟨a,ha,han',x',hx',z,hz,hess⟩
 153 
 154 end
