  25 /-- Inessentiality for a known equation means that the same target is a
  26 homogeneous solution for the same filtered map. -/
  27 private theorem inessential_iff_zero_source_same_target
  28     (f : FilteredMorphism F G) (k : ι) (s n : ℤ) {T : Acat}
  29     (x : T ⟶ F.associatedGraded s k)
  30     (y : T ⟶ G.associatedGraded (s+n) k)
  31     (h : FilteredMapExtension.Relation f k s n x y) :
  32     ¬ FilteredMapExtension.Essential f k s n x y ↔
  33       FilteredMapExtension.Relation f k s n 0 y := by
  34   classical
  35   constructor
  36   · intro hne
  37     have hz : FilteredMapExtension.Relation f k s n x 0 := by
  38       by_contra hnz
  39       exact hne ((FilteredMapExtension.essential_iff h).mpr hnz)
  40     simpa only [sub_self, sub_zero] using h.sub hz
  41   · intro hz he
  42     have hh : FilteredMapExtension.Relation f k s n x 0 := by
  43       simpa only [sub_zero, sub_self] using h.sub hz
  44     exact ((FilteredMapExtension.essential_iff h).mp he) hh

  46 /-- This is the actual higher-source representative form: the specified
  47 target is retained, with a source lift in filtration s+1. -/
  48 private theorem inessential_iff_higher_source_same_target
  49     (f : FilteredMorphism F G) (k : ι) (s n : ℤ) {T : Acat}
  50     (x : T ⟶ F.associatedGraded s k)
  51     (y : T ⟶ G.associatedGraded (s+n) k)
  52     (h : FilteredMapExtension.Relation f k s n x y) :
  53     ¬ FilteredMapExtension.Essential f k s n x y ↔
  54       0 ≤ n ∧ ∃ (a : T ⟶ Subobject.underlying.obj (F.F (s+1) k))
  55         (b : T ⟶ Subobject.underlying.obj (G.F (s+n) k)),
  56         b ≫ G.toAssociatedGraded (s+n) k = y ∧
  57         a ≫ (F.F (s+1) k).arrow ≫ f.map k =
  58           b ≫ (G.F (s+n) k).arrow := by
  59   rw [inessential_iff_zero_source_same_target f k s n x y h]
  60   exact FilteredMapExtension.mem_indeterminacy_iff
  61 end Filtered

  97     (M : ConvergentMap F g cx.toConvergence cy.toConvergence)
  98     (p : ℤ × ℤ) (s n : ℤ)

 100     (y : ((F.obj Y).sequence.ssData (s+n,p.1+(s+n),p.2)).eInfty)
 101     (h : TowerExtensionEquation M p s n x y) :
 102     ¬ TowerEssentialExtension M p s n x y ↔
 103       0 ≤ n ∧ ∃
 104         (a : (Subobject.underlying.obj ((P.filtration X).F (s+1) p) : ModuleCat.{v} ℤ))
 105         (b : (Subobject.underlying.obj ((P.filtration Y).F (s+n) p) : ModuleCat.{v} ℤ)),
 106         ((P.filtration X).F (s+1) p).arrow a ≫ g =
 107           ((P.filtration Y).F (s+n) p).arrow b ∧
 108         (P.filtration Y).toAssociatedGraded (s+n) p b =
 109           (KIP126.ProofTools.Operations.infinityGradedIso cy p (s+n)).hom y := by
 110   have hrel : FilteredMapExtension.Relation M.filteredMap p s n
 111       (toGraded cx.toConvergence (by simp [SyntheticTowerConvergence.toConvergence])
 112         (elementMap x))
 113       (toGraded cy.toConvergence (by simp [SyntheticTowerConvergence.toConvergence])
 114         (elementMap y)) := h
 115   rw [show (¬ TowerEssentialExtension M p s n x y) ↔ _ from
 116     inessential_iff_higher_source_same_target _ _ _ _ _ _ hrel]
 117   simp only [KIP126.ProofTools.Internal.toGraded_element]
 118   constructor
 119   · rintro ⟨hn,a,b,hb,he⟩
 120     refine ⟨hn,a (ULift.up 1),b (ULift.up 1),?_,?_⟩
 121     · have hh := ConcreteCategory.congr_hom he (ULift.up (1 : ℤ))
 122       have hf := ConcreteCategory.congr_hom (M.filteredMap_map p)
 123         (((P.filtration X).F (s+1) p).arrow (a (ULift.up 1)))
 124       exact hf.symm.trans hh
 125     · have hh := ConcreteCategory.congr_hom hb (ULift.up (1 : ℤ))
 126       change (P.filtration Y).toAssociatedGraded (s+n) p (b (ULift.up 1)) =
 127         elementMap ((KIP126.ProofTools.Internal.infinityGradedIso cy p (s+n)).hom y)
 128           (ULift.up (1 : ℤ)) at hh
 129       rw [elementMap_one] at hh
 130       exact hh
 131   · rintro ⟨hn,a,b,he,hb⟩
 132     refine ⟨hn,elementMap a,elementMap b,?_,?_⟩
 133     · rw [← KIP126.AdamsSynthetic.elementMap_comp]
 134       exact congrArg elementMap hb
 135     · rw [← Category.assoc,← KIP126.AdamsSynthetic.elementMap_comp,
 136         ← KIP126.AdamsSynthetic.elementMap_comp,
 137         ← KIP126.AdamsSynthetic.elementMap_comp]
 138       have hf := ConcreteCategory.congr_hom (M.filteredMap_map p)
 139         (((P.filtration X).F (s+1) p).arrow a)

 204 private theorem correction_label_exists (X Y : ClassicalObject)
 205     (f : X.obj model.auxiliary ⟶ Y.obj model.auxiliary)
 206     (q : ℕ) (hq : 0 < q) (s t : ℤ) (a : ℕ) (ha : a < q)
 207     (z : ((model.family.obj ((SyntheticObject.quotient q (extensionSourceObject model X Y f)).obj
 208       model.nu model.auxiliary)).sequence.ssData
 209       (s+a,(t-s)+(s+a),t+normalizedExponent model.classicalTower f)).eInfty) :
 210     ∃ x : Ambient model X (s+a,(t-s)+(s+a)),
 211       x ∈ cycles model X ((q:ℤ)-a) (s+a,(t-s)+(s+a)) ∧
 212       HasInfinityRepresentative _ 2 _ (finiteExtensionCorrectionLabel X Y f q s t a x) z := by
 213   let e : ℤ := normalizedExponent model.classicalTower f
 214   let u : ℤ := s+a
 215   let v : ℤ := (t-s)+(s+a)
 216   let p : ℤ × ℤ := (t-s,v-a)
 217   let Q := correctionShiftIso X Y f q
 218   let I := towerInfinityIso (model.convergence (.quotient q (.nu X)))
 219     (model.convergence (.shift (0,e) (.quotient q (.nu X))))
 220     ((essShift (0,e)).filtered (.quotient q (.nu X)) p) u
 221   let hdeg : (u,(p+(0,e)).1+u,(p+(0,e)).2) =
 222       (s+a,(t-s)+(s+a),t+normalizedExponent model.classicalTower f) := by
 223     ext <;> dsimp [p,u,v,e] <;> omega
 224   let zs := ((model.family.map Q.hom).eInftyMap _ z)
 225   let zu := I.inv ((eqToHom (congrArg (fun i =>
 226     ((model.family.obj ((SyntheticObject.shift (0,e) (.quotient q (.nu X))).obj
 227       model.nu model.auxiliary)).sequence.ssData i).eInfty) hdeg.symm)) zs)
 228   obtain ⟨x,hx,hrep,_⟩ := CrossingLabels.finite_grade_label X q hq u v (v-a)
 229     (by dsimp [v]; constructor <;> omega) zu
 230   have hlabel : nuWeightLabel model X u v (v-a) (by omega) x =
 231       targetNuLabel model X u v a x := by
 232     have hk : (v-(v-(a:ℤ))).toNat=a := by omega
 233     have transport (b c : ℕ) (he : b=c) (hi : (u,v,v-b)=(u,v,v-c)) :
 234         regradeE2 hi (targetNuLabel model X u v b x) = targetNuLabel model X u v c x := by
 235       subst c
 236       rfl
 237     exact transport _ a hk _
 238   rw [hlabel] at hrep
 239   refine ⟨x,?_,?_⟩
 240   · convert hx using 1 <;> dsimp [u,v] <;> congr 1 <;> omega
 241   have hs := ((essShift (0,e)).representative_iff (.quotient q (.nu X)) 2
 242     (by decide) p u (finiteTargetLabel model X q u v a x) zu).mp hrep
 243   have hs' := infinity_representative_regrade hdeg _ _ hs
 244   have hz : ((eqToHom (congrArg (fun i =>
 245       ((model.family.obj ((SyntheticObject.shift (0,e) (.quotient q (.nu X))).obj
 246         model.nu model.auxiliary)).sequence.ssData i).eInfty) hdeg)) (I.hom zu)) = zs := by
 247     dsimp only [zu]
 248     simp only [←ModuleCat.comp_apply,Category.assoc,Iso.inv_hom_id_assoc,
 249       eqToHom_trans,eqToHom_refl,ConcreteCategory.id_apply]
 250   rw [hz] at hs'
 251   have hm := family_infinity_representative_map Q.inv _ _ zs hs'
 252   have hc : (model.family.map Q.inv).eInftyMap _ zs = z := by
 253     dsimp only [zs]
 254     rw [←ModuleCat.comp_apply,←SpectralSequenceMorphism.eInftyMap_comp,
 255       ←SyntheticAdamsFamily.map_comp,Q.hom_inv_id,SyntheticAdamsFamily.map_id,
 256       SpectralSequenceMorphism.eInftyMap_id,ConcreteCategory.id_apply]
 257   rw [hc] at hm
 258   exact hm

 297 /-- Proposition 6.20 after removing the source cycle-set exclusion. The
 298 original target is retained, with b=0. The shorter extension may itself be
 299 inessential and its source class is allowed to vanish. -/
 300 public theorem finiteExtension_inessential_iff_shorter (X Y : ClassicalObject)
 301     (f : X.obj model.auxiliary ⟶ Y.obj model.auxiliary)
 302     (rEarlier rLater : ℕ) (n s t : ℤ) (x y)
 303     (Wlater : FiniteExtensionWitness model X Y f rLater n s t x y)
 304     (Wearly : FiniteExtensionWitness model X Y f rEarlier n s t x y)
 305     (hle : rEarlier ≤ rLater) (hess : Wlater.Essential) :
 306     ¬ Wearly.Essential ↔ ∃ a : ℕ, 0 < a ∧
 307       (a:ℤ) ≤ n-normalizedExponent model.classicalTower f ∧
 308       ∃ x' : Ambient model X (s+a,(t-s)+(s+a)),
 309         x' ∈ cycles model X (((rEarlier-1:ℕ):ℤ)- (a:ℤ)) (s+a,(t-s)+(s+a)) ∧
 310         ∃ z,
 311           HasInfinityRepresentative _ 2 _
 312             (finiteExtensionCorrectionLabel X Y f (rEarlier-1) s t a x') z ∧
 313           TowerExtensionEquation (model.convergentMap
 314             (.quotient (rEarlier-1) (extensionSourceObject model X Y f))
 315             (.quotient (rEarlier-1) (.nu Y)) (finiteExtensionMap model X Y f (rEarlier-1)))
 316             (t-s,t+normalizedExponent model.classicalTower f) (s+a) (n-a) z
 317             ((eqToHom (congrArg (fun d =>
 318               ((model.family.obj ((SyntheticObject.quotient (rEarlier-1) (.nu Y)).obj
 319                 model.nu model.auxiliary)).sequence.ssData
 320                 (d,(t-s)+d,t+normalizedExponent model.classicalTower f)).eInfty)
 321               (show s+n = (s+a)+(n-a) by omega))) Wearly.target) := by
 322   let A := (SyntheticObject.quotient (rEarlier-1) (extensionSourceObject model X Y f)).obj
 323     model.nu model.auxiliary
 324   let B := (SyntheticObject.quotient (rEarlier-1) (.nu Y)).obj model.nu model.auxiliary
 325   let p : ℤ × ℤ := (t-s,t+normalizedExponent model.classicalTower f)
 326   let M := model.convergentMap
 327     (.quotient (rEarlier-1) (extensionSourceObject model X Y f))
 328     (.quotient (rEarlier-1) (.nu Y)) (finiteExtensionMap model X Y f (rEarlier-1))
 329   have hy := earlier_target_nonzero X Y f rEarlier rLater n s t x y Wlater Wearly hle hess
 330   constructor
 331   · intro hne
 332     have hn := inessential_length_strict X Y f rEarlier n s t x y Wearly hy hne
 333     have hq : 1 < rEarlier-1 := by have := Wearly.length_upper; omega
 334     obtain ⟨hn0,ar,br,heq,hbr⟩ :=
 335       (tower_inessential_iff_higher_source_same_target M p s n
 336         Wearly.source Wearly.target Wearly.equation).mp hne
 337     let I := infinityGradedIso (model.convergence
 338       (.quotient (rEarlier-1) (extensionSourceObject model X Y f))) p (s+1)
 339     let z := I.inv ((model.syntheticTower.filtration A).toAssociatedGraded (s+1) p ar)
 340     obtain ⟨x',hx',hz⟩ := correction_label_exists X Y f (rEarlier-1)
 341       (by omega) s t 1 hq z
 342     refine ⟨1,by decide,by omega,x',hx',z,hz,?_⟩
 343     apply (tower_extension_reindexed M p (s+1) (n-1) (s+n) (by omega) z Wearly.target).mpr
 344     refine ⟨by omega,ar,br,heq,?_,hbr⟩
 345     exact (I.toLinearEquiv.apply_symm_apply _).symm
 346   · rintro ⟨a,ha,han,x',hx',z,hz,he⟩
 347     obtain ⟨_,ar,br,heq,_,hbr⟩ :=
 348       (tower_extension_reindexed M p (s+a) (n-a) (s+n) (by omega) z Wearly.target).mp he
 349     have hα : ((model.syntheticTower.filtration A).F (s+a) p).arrow ar ∈
 350         model.syntheticTower.filtrationSubmodule A (s+a) p := by
 351       have hm : ((model.syntheticTower.filtration A).F (s+a) p).arrow ar ∈
 352           ModuleCat.subobjectModule _ ((model.syntheticTower.filtration A).F (s+a) p) := ⟨ar,rfl⟩
 353       simpa only [SyntheticTowerPresentation.filtration,OrderIso.apply_symm_apply] using hm
 354     have hα' := model.syntheticTower.filtrationSubmodule_antitone A p
 355       (show s+1 ≤ s+a by omega) hα
 356     obtain ⟨ar',har'⟩ := filtration_representative_exists model.syntheticTower A p (s+1)
 357       (((model.syntheticTower.filtration A).F (s+a) p).arrow ar) hα'
 358     apply (tower_inessential_iff_higher_source_same_target M p s n
 359       Wearly.source Wearly.target Wearly.equation).mpr
 360     refine ⟨by have := Wearly.length_lower; omega,ar',br,?_,hbr⟩
 361     rw [har']
 362     exact heq
