  27 /-- The homotopy class has the prescribed E∞ label under this convergence. -/
  28 @[expose] public def RepresentsInfinityClass (c : SyntheticTowerConvergence P X)
  29     (p : ℤ × ℤ) (s : ℤ)
  30     (x : ((E.obj X).sequence.ssData (s,p.1+s,p.2)).eInfty)
  31     (a : BiHom p.1 p.2 X) : Prop :=
  32   ∃ ar : (Subobject.underlying.obj ((P.filtration X).F s p) : ModuleCat.{v} ℤ),
  33     ((P.filtration X).F s p).arrow ar = a ∧
  34     (P.filtration X).toAssociatedGraded s p ar =
  35       (infinityGradedIso c p s).hom x

  37 /-- No higher-source extension with nonzero target lands between the specified threshold
  38 and the original target filtration. -/

  40     (M : ConvergentMap E g cx.toConvergence cy.toConvergence)

  45 private theorem image_correction_in_next
  46     (M : ConvergentMap E g cx.toConvergence cy.toConvergence)
  47     (p : ℤ × ℤ) (s n threshold : ℤ) (hthreshold : threshold ≤ s+n)
  48     (hcross : ExtensionNoCrossingRange M p s n threshold)
  49     (a : BiHom p.1 p.2 X)
  50     (ha : a ∈ P.filtrationSubmodule X (s+1) p)
  51     (hga : a ≫ g ∈ P.filtrationSubmodule Y threshold p) :
  52     a ≫ g ∈ P.filtrationSubmodule Y (s+n+1) p := by
  53   let F : ℤ → AddSubgroup (BiHom p.1 p.2 X) :=
  54     fun k => (P.filtrationSubmodule X k p).toAddSubgroup
  55   let G : ℤ → AddSubgroup (BiHom p.1 p.2 Y) :=
  56     fun k => (P.filtrationSubmodule Y k p).toAddSubgroup
  57   let f : BiHom p.1 p.2 X →+ BiHom p.1 p.2 Y :=
  58     (syntheticHomotopyMap g p).hom.toAddMonoidHom
  59   by_contra hnot
  60   have hnot' : f a ∉ G (threshold + (s+n+1-threshold).toNat) := by
  61     have hindex : threshold + ((s+n+1-threshold).toNat : ℤ) = s+n+1 := by
  62       rw [Int.toNat_of_nonneg (show 0 ≤ s+n+1-threshold by omega)]
  63       omega
  64     rw [hindex]
  65     change ¬ a ≫ g ∈ P.filtrationSubmodule Y (s+n+1) p
  66     exact hnot
  67   obtain ⟨b,hb,hbn,hgb,hgbnext⟩ :=
  68     first_nonzero_filtration_grade G threshold (s+n+1-threshold).toNat (f a)
  69       hga hnot'
  70   by_cases hbs : b ≤ s
  71   · apply hgbnext
  72     exact P.filtrationSubmodule_antitone Y p (by omega)
  73       (map_preserves_filtration M p (s+1) a ha)
  74   · apply hgbnext
  75     exact Internal.Leibniz.higher_source_image_of_nonessential f F G
  76       (fun i j hij z hz => P.filtrationSubmodule_antitone Y p hij hz)
  77       (fun k z hz => map_preserves_filtration M p k z hz) s b (by omega)
  78       (fun k hsk hkb z hz hgz => by
  79         change z ≫ g ∈ P.filtrationSubmodule Y b p at hgz
  80         have hno : ∀ x y, ¬TowerEssentialExtension M p k (b-k) x y := by
  81           intro x y hess
  82           have hy := hcross k b hsk hkb hb (by omega) x y hess.1
  83           subst y
  84           exact (Internal.Leibniz.tower_essential_iff_no_zero M p k (b-k) x 0 hess.1).mp hess hess.1
  85         have hh := Internal.Leibniz.tower_correction_of_nonessential M p k (b-k)
  86           (by omega) hno z hz (by
  87             simpa only [add_sub_cancel] using hgz)
  88         change ∃ b', b' ∈ P.filtrationSubmodule X k p ∧
  89           z-b' ∈ P.filtrationSubmodule X (k+1) p ∧
  90           b' ≫ g ∈ P.filtrationSubmodule Y (b+1) p
  91         simpa only [add_sub_cancel] using hh) a ha hgb

 119     (p : ℤ × ℤ) (s : ℤ)
 120     (x : ((E.obj X).sequence.ssData (s,p.1+s,p.2)).eInfty)
 121     (a b : BiHom p.1 p.2 X)

 138 /-- AimPaper Proposition 2.10: absence of crossings in the prescribed target
 139 range is equivalent to independence of all source representatives whose image
 140 has the required filtration.  This is the representative formulation for the
 141 same actual map and the same two convergences. -/
 142 public theorem extension_noCrossingRange_iff
 143     (M : ConvergentMap E g cx.toConvergence cy.toConvergence)
 144     (p : ℤ × ℤ) (s n threshold : ℤ) (hthreshold : threshold ≤ s+n)
 145     (x : ((E.obj X).sequence.ssData (s,p.1+s,p.2)).eInfty)
 146     (y : ((E.obj Y).sequence.ssData (s+n,p.1+(s+n),p.2)).eInfty)
 147     (he : TowerExtensionEquation M p s n x y) :
 148     ExtensionNoCrossingRange M p s n threshold ↔
 149       ∀ a : BiHom p.1 p.2 X, RepresentsInfinityClass cx p s x a →
 150         a ≫ g ∈ P.filtrationSubmodule Y threshold p →
 151         RepresentsInfinityClass cy p (s+n) y (a ≫ g) := by
 152   obtain ⟨hn,ar₀,br₀,he₀,hx₀,hy₀⟩ := (tower_extension_iff M p s n x y).mp he
 153   let a₀ : BiHom p.1 p.2 X := ((P.filtration X).F s p).arrow ar₀
 154   let b₀ : BiHom p.1 p.2 Y := ((P.filtration Y).F (s+n) p).arrow br₀
 155   have hx : RepresentsInfinityClass cx p s x a₀ := ⟨ar₀,rfl,hx₀⟩
 156   have hy : RepresentsInfinityClass cy p (s+n) y b₀ := ⟨br₀,rfl,hy₀⟩
 157   have hbase : a₀ ≫ g = b₀ := he₀
 158   constructor
 159   · intro hc a ha hga
 160     have hdiff := representatives_difference p s x a a₀ ha hx
 161     have himage : (a-a₀) ≫ g ∈ P.filtrationSubmodule Y threshold p := by
 162       rw [Preadditive.sub_comp,hbase]
 163       exact (P.filtrationSubmodule Y threshold p).sub_mem hga
 164         (P.filtrationSubmodule_antitone Y p hthreshold (represents_mem p (s+n) y b₀ hy))
 165     have hh := image_correction_in_next M p s n threshold hthreshold hc (a-a₀) hdiff himage
 166     apply represents_of_difference p (s+n) y (a≫g) b₀ hy
 167     simpa only [Preadditive.sub_comp,hbase] using hh
 168   · intro hall u v hsu huv htv hvs xx yy he'
 169     obtain ⟨_,ar,br,heq,hxx,hyy⟩ := (tower_extension_iff M p u (v-u) xx yy).mp he'
 170     let a : BiHom p.1 p.2 X := ((P.filtration X).F u p).arrow ar
 171     have ha : a ∈ P.filtrationSubmodule X u p := by
 172       have h : a ∈ ModuleCat.subobjectModule _ ((P.filtration X).F u p) := ⟨ar,rfl⟩
 173       simpa only [SyntheticTowerPresentation.filtration,OrderIso.apply_symm_apply] using h
 174     have hfa : a≫g ∈ P.filtrationSubmodule Y v p := by
 175       have h : a≫g ∈ ModuleCat.subobjectModule _ ((P.filtration Y).F (u+(v-u)) p) :=
 176         ⟨br,heq.symm⟩
 177       simpa only [SyntheticTowerPresentation.filtration,OrderIso.apply_symm_apply,
 178         add_sub_cancel] using h
 179     have hsum : RepresentsInfinityClass cx p s x (a₀+a) := by
 180       apply represents_of_difference p s x (a₀+a) a₀ hx
 181       simpa only [add_sub_cancel_left] using
 182         (P.filtrationSubmodule_antitone X p (show s+1 ≤ u by omega) ha)
 183     have hsumg : (a₀+a)≫g ∈ P.filtrationSubmodule Y threshold p := by
 184       rw [Preadditive.add_comp,hbase]
 185       exact (P.filtrationSubmodule Y threshold p).add_mem
 186         (P.filtrationSubmodule_antitone Y p hthreshold (represents_mem p (s+n) y b₀ hy))
 187         (P.filtrationSubmodule_antitone Y p htv hfa)
 188     have hdet := hall (a₀+a) hsum hsumg
 189     have hdiff := representatives_difference p (s+n) y ((a₀+a)≫g) b₀ hdet hy
 190     have hnext : a≫g ∈ P.filtrationSubmodule Y (v+1) p := by
 191       have h : a≫g ∈ P.filtrationSubmodule Y (s+n+1) p := by
 192         simpa only [Preadditive.add_comp,hbase,add_sub_cancel_left] using hdiff
 193       exact P.filtrationSubmodule_antitone Y p (by omega) h
 194     have hzero := (Internal.element_grade_equal_iff (P.filtration Y) (u+(v-u)) p br 0).mpr
 195       (by
 196         rw [←heq,map_zero,sub_zero]
 197         simpa only [SyntheticTowerPresentation.filtration,OrderIso.apply_symm_apply,
 198           add_sub_cancel, a] using hnext)
 199     apply (ModuleCat.mono_iff_injective (infinityGradedIso cy p (u+(v-u))).hom).mp inferInstance
 200     rw [map_zero]
 201     rw [map_zero] at hzero
 202     exact hyy.symm.trans hzero

 204 /-- Proposition 2.10(1).  Higher-source crossings automatically have target
 205 filtration at least s+1; the threshold s therefore excludes every crossing. -/
 206 public theorem extension_noCrossing_iff
 207     (M : ConvergentMap E g cx.toConvergence cy.toConvergence)
 208     (p : ℤ × ℤ) (s n : ℤ)
 209     (x : ((E.obj X).sequence.ssData (s,p.1+s,p.2)).eInfty)
 210     (y : ((E.obj Y).sequence.ssData (s+n,p.1+(s+n),p.2)).eInfty)
 211     (he : TowerExtensionEquation M p s n x y) :
 212     ExtensionNoCrossingRange M p s n s ↔
 213       ∀ a : BiHom p.1 p.2 X, RepresentsInfinityClass cx p s x a →
 214         RepresentsInfinityClass cy p (s+n) y (a ≫ g) := by
 215   have hn := ((tower_extension_iff M p s n x y).mp he).1
 216   have h := extension_noCrossingRange_iff M p s n s (by omega) x y he
 217   constructor
 218   · intro hc a ha
 219     exact h.mp hc a ha (map_preserves_filtration M p s a (represents_mem p s x a ha))
 220   · intro hall
 221     exact h.mpr (fun a ha _ => hall a ha)

 223 private theorem represents_zero_iff (p : ℤ × ℤ) (s : ℤ) (a : BiHom p.1 p.2 X) :
 224     RepresentsInfinityClass cx p s 0 a ↔ a ∈ P.filtrationSubmodule X (s+1) p := by
 225   have hzero : RepresentsInfinityClass cx p s 0 (0 : BiHom p.1 p.2 X) :=
 226     ⟨0,by simp only [map_zero],by simp only [map_zero]⟩
 227   constructor
 228   · intro ha
 229     simpa only [sub_zero] using representatives_difference p s 0 a 0 ha hzero
 230   · intro ha
 231     exact represents_of_difference p s 0 a 0 hzero (by simpa only [sub_zero] using ha)

 233 /-- Proposition 2.10(2): the zero target means strictly higher filtration,
 234 uniformly for every representative of the specified source label. -/
 235 public theorem extension_zero_noCrossing_iff
 236     (M : ConvergentMap E g cx.toConvergence cy.toConvergence)
 237     (p : ℤ × ℤ) (s n : ℤ)
 238     (x : ((E.obj X).sequence.ssData (s,p.1+s,p.2)).eInfty)
 239     (he : TowerExtensionEquation M p s n x 0) :
 240     ExtensionNoCrossingRange M p s n s ↔
 241       ∀ a : BiHom p.1 p.2 X, RepresentsInfinityClass cx p s x a →
 242         a ≫ g ∈ P.filtrationSubmodule Y (s+n+1) p := by
 243   rw [extension_noCrossing_iff M p s n x 0 he]
 244   constructor
 245   · intro hall a ha
 246     exact (represents_zero_iff p (s+n) (a≫g)).mp (hall a ha)
 247   · intro hall a ha
 248     exact (represents_zero_iff p (s+n) (a≫g)).mpr (hall a ha)

 250 /-- Proposition 2.10(3).  An actual filtration-n map has no length-n
 251 crossing.  Only the filtration lower bound is needed, so the statement also
 252 applies when the actual filtration is larger. -/
 253 public theorem extension_noCrossing_of_filtration_bound
 254     (M : ConvergentMap E g cx.toConvergence cy.toConvergence)
 255     (p : ℤ × ℤ) (s n : ℤ)
 256     (x : ((E.obj X).sequence.ssData (s,p.1+s,p.2)).eInfty)
 257     (y : ((E.obj Y).sequence.ssData (s+n,p.1+(s+n),p.2)).eInfty)
 258     (he : TowerExtensionEquation M p s n x y)
 259     (hmap : ∀ k : ℤ, ∀ a : BiHom p.1 p.2 X,
 260       a ∈ P.filtrationSubmodule X k p →
 261       a ≫ g ∈ P.filtrationSubmodule Y (k+n) p) :
 262     ExtensionNoCrossingRange M p s n s := by
 263   apply (extension_noCrossing_iff M p s n x y he).mpr
 264   obtain ⟨_,ar₀,br₀,he₀,hx₀,hy₀⟩ := (tower_extension_iff M p s n x y).mp he
 265   let a₀ : BiHom p.1 p.2 X := ((P.filtration X).F s p).arrow ar₀
 266   let b₀ : BiHom p.1 p.2 Y := ((P.filtration Y).F (s+n) p).arrow br₀
 267   have hx : RepresentsInfinityClass cx p s x a₀ := ⟨ar₀,rfl,hx₀⟩
 268   have hy : RepresentsInfinityClass cy p (s+n) y b₀ := ⟨br₀,rfl,hy₀⟩
 269   have hbase : a₀ ≫ g = b₀ := he₀
 270   intro a ha
 271   apply represents_of_difference p (s+n) y (a≫g) b₀ hy
 272   have hdiff := hmap (s+1) (a-a₀) (representatives_difference p s x a a₀ ha hx)
 273   simpa only [Preadditive.sub_comp,hbase,add_assoc,add_comm,add_left_comm] using hdiff

 280 /-- Crossing in ordinary filtered homotopy groups, with all nonzero target
 281 grades retained.  A represents any leading source class in filtration u. -/
 282 @[expose] public def FilteredNoCrossingRange (f : A →+ B)
 283     (F : ℤ → AddSubgroup A) (G : ℤ → AddSubgroup B)
 284     (s n threshold : ℤ) : Prop :=
 285   ∀ u v : ℤ, s < u → u ≤ v → threshold ≤ v → v ≤ s+n →
 286     ∀ a : A, a ∈ F u → f a ∈ G v → f a ∈ G (v+1)

 288 private theorem filtered_correction_in_next (f : A →+ B)
 289     (F : ℤ → AddSubgroup A) (G : ℤ → AddSubgroup B) (hG : Antitone G)
 290     (hf : ∀ u a, a ∈ F u → f a ∈ G u)
 291     (s n threshold : ℤ) (ht : threshold ≤ s+n)
 292     (hc : FilteredNoCrossingRange f F G s n threshold)
 293     (a : A) (ha : a ∈ F (s+1)) (hfa : f a ∈ G threshold) :
 294     f a ∈ G (s+n+1) := by
 295   by_contra hnot
 296   have hi : threshold + ((s+n+1-threshold).toNat : ℤ) = s+n+1 := by
 297     rw [Int.toNat_of_nonneg (show 0 ≤ s+n+1-threshold by omega)]
 298     omega
 299   obtain ⟨v,hv,hvn,hfv,hfvn⟩ := first_nonzero_filtration_grade G threshold
 300     (s+n+1-threshold).toNat (f a) hfa (by rwa [hi])
 301   apply hfvn
 302   by_cases hvs : v ≤ s
 303   · exact hG (show v+1 ≤ s+1 by omega) (hf (s+1) a ha)
 304   · exact hc (s+1) v (by omega) (by omega) hv (by omega) a ha hfv

 306 /-- Proposition 2.10 for arbitrary filtered abelian groups.  Equality of source
 307 labels is a-a0 ∈ F^(s+1); equality of target labels is f(a)-f(a0) ∈ G^(s+n+1).
 308 Thus this formulation applies to classical homotopy groups as well as to the
 309 synthetic representative version above.  a0 is a witness of the given extension. -/
 310 public theorem filtered_extension_noCrossingRange_iff (f : A →+ B)
 311     (F : ℤ → AddSubgroup A) (G : ℤ → AddSubgroup B)
 312     (hF : Antitone F) (hG : Antitone G)
 313     (hf : ∀ u a, a ∈ F u → f a ∈ G u)
 314     (s n threshold : ℤ) (ht : threshold ≤ s+n)
 315     (a0 : A) (ha0 : a0 ∈ F s) (hfa0 : f a0 ∈ G (s+n)) :
 316     FilteredNoCrossingRange f F G s n threshold ↔
 317       ∀ a : A, a ∈ F s → a-a0 ∈ F (s+1) → f a ∈ G threshold →
 318         f a ∈ G (s+n) ∧ f a-f a0 ∈ G (s+n+1) := by
 319   constructor
 320   · intro hc a _ hd hfa
 321     have hh := filtered_correction_in_next f F G hG hf s n threshold ht hc
 322       (a-a0) hd (by rw [map_sub]; exact (G threshold).sub_mem hfa (hG ht hfa0))
 323     have hdiff : f a-f a0 ∈ G (s+n+1) := by simpa only [map_sub] using hh
 324     refine ⟨?_,hdiff⟩
 325     have h := (G (s+n)).add_mem (hG (show s+n ≤ s+n+1 by omega) hdiff) hfa0
 326     simpa only [sub_add_cancel] using h
 327   · intro hall u v hsu huv htv hvs a ha hfa
 328     have ha' : a ∈ F (s+1) := hF (by omega) ha
 329     have hsum : a0+a ∈ F s := (F s).add_mem ha0 (hF (by omega) ha)
 330     have hdiff : a0+a-a0 ∈ F (s+1) := by simpa only [add_sub_cancel_left] using ha'
 331     have hsumf : f (a0+a) ∈ G threshold := by
 332       rw [map_add]
 333       exact (G threshold).add_mem (hG ht hfa0) (hG htv hfa)
 334     have hh := (hall (a0+a) hsum hdiff hsumf).2
 335     have hh' : f a ∈ G (s+n+1) := by simpa only [map_add,add_sub_cancel_left] using hh
 336     exact hG (by omega) hh'

 338 /-- Proposition 2.10(1), with all representatives and no target threshold. -/

 353     exact h.mpr (fun a ha hd _ => hall a ha hd)

 355 /-- Proposition 2.10(2): zero target label is precisely strictly higher
 356 filtration for the image of every source representative. -/
 357 public theorem filtered_zero_extension_noCrossing_iff (f : A →+ B)
 358     (F : ℤ → AddSubgroup A) (G : ℤ → AddSubgroup B)
 359     (hF : Antitone F) (hG : Antitone G)
 360     (hf : ∀ u a, a ∈ F u → f a ∈ G u)
 361     (s n : ℤ) (hn : 0 ≤ n) (a0 : A)
 362     (ha0 : a0 ∈ F s) (hfa0 : f a0 ∈ G (s+n+1)) :
 363     FilteredNoCrossingRange f F G s n s ↔
 364       ∀ a : A, a ∈ F s → a-a0 ∈ F (s+1) → f a ∈ G (s+n+1) := by
 365   rw [filtered_extension_noCrossing_iff f F G hF hG hf s n hn a0 ha0
 366     (hG (show s+n ≤ s+n+1 by omega) hfa0)]
 367   constructor
 368   · intro hall a ha hd
 369     have h := (G (s+n+1)).add_mem (hall a ha hd).2 hfa0
