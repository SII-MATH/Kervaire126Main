  13 /-- A finite correction argument with the actual residual kept throughout. -/
  14 public theorem finite_filtration_correction
  15     {A A' B : Type*} [AddCommGroup A] [AddCommGroup A'] [AddCommGroup B]
  16     (ρ : A' →+ A) (g : A →+ B)
  17     (E : ℤ → AddSubgroup A) (L : ℤ → AddSubgroup A')
  18     (T : AddSubgroup B) (hL : Antitone L)
  19     (s : ℤ) (N : ℕ)
  20     (hstep : ∀ k : ℤ, s ≤ k → k < s + N → ∀ δ : A, δ ∈ E k →
  21       (∃ ε : A', ε ∈ L k ∧ δ - ρ ε ∈ E (k + 1)) ∨ g δ ∈ T)
  22     (hend : ∀ δ : A, δ ∈ E (s + N) → g δ ∈ T)
  23     (α₀ : A) (α : A') (hδ : α₀ - ρ α ∈ E s) :
  24     ∃ α' : A', α' - α ∈ L s ∧ g (α₀ - ρ α') ∈ T := by
  25   induction N generalizing s α with
  26   | zero =>
  27     refine ⟨α, ?_, hend _ ?_⟩
  28     · simpa only [sub_self] using (L s).zero_mem
  29     · simpa only [Nat.cast_zero, add_zero] using hδ
  30   | succ N ih =>
  31     rcases hstep s le_rfl (by omega) _ hδ with hcorrect | hstop
  32     · obtain ⟨ε, hε, hδε⟩ := hcorrect
  33       have hδ' : α₀ - ρ (α + ε) ∈ E (s + 1) := by
  34         simpa only [map_add, sub_add_eq_sub_sub] using hδε
  35       obtain ⟨α', hα', hfinal⟩ := ih (s + 1)
  36         (fun k hk hkn δ hδ => hstep k (by omega) (by omega) δ hδ)
  37         (fun δ hδ => hend δ (by simpa only [Nat.cast_succ, add_assoc,
  38           add_comm, add_left_comm] using hδ)) (α + ε) hδ'
  39       refine ⟨α', ?_, hfinal⟩
  40       have hmem := (L s).add_mem (hL (by omega) hα') hε
  41       convert hmem using 1
  42       abel
  43     · exact ⟨α, by simpa only [sub_self] using (L s).zero_mem, hstop⟩

  46 /-- Injectivity of successive graded maps reflects membership through a finite
  47 filtration interval; it does not require injectivity of the ambient map. -/
  48 public theorem finite_filtration_reflection
  49     {B' B : Type*} [AddCommGroup B'] [AddCommGroup B]
  50     (ρ : B' →+ B) (L : ℤ → AddSubgroup B') (E : ℤ → AddSubgroup B)
  51     (hE : Antitone E) (s : ℤ) (N : ℕ)
  52     (hgraded : ∀ k : ℤ, s ≤ k → k < s + N → ∀ b : B', b ∈ L k →
  53       ρ b ∈ E (k + 1) → b ∈ L (k + 1))
  54     (b : B') (hb : b ∈ L s) (hρb : ρ b ∈ E (s + N)) :
  55     b ∈ L (s + N) := by
  56   have h : ∀ k : ℕ, k ≤ N → b ∈ L (s + k) := by
  57     intro k hk
  58     induction k with
  59     | zero => simpa only [Nat.cast_zero, add_zero] using hb
  60     | succ k ih =>
  61       have hkN : k ≤ N := by omega
  62       have hnext := hgraded (s + k) (by omega) (by omega) b (ih hkN)
  63         (hE (by omega) hρb)
  64       simpa only [Nat.cast_succ, add_assoc] using hnext
  65   exact h N le_rfl

  67 /-- Failure to lie in the final layer is witnessed by a nonzero leading grade
  68 in the finite interval. -/
  69 public theorem first_nonzero_filtration_grade
  70     {B : Type*} [AddCommGroup B] (E : ℤ → AddSubgroup B)
  71     (s : ℤ) (N : ℕ) (b : B) (hb : b ∈ E s) (hnot : b ∉ E (s + N)) :
  72     ∃ k : ℤ, s ≤ k ∧ k < s + N ∧ b ∈ E k ∧ b ∉ E (k + 1) := by
  73   induction N generalizing s with
  74   | zero => exact (hnot (by simpa only [Nat.cast_zero, add_zero] using hb)).elim
  75   | succ N ih =>
  76     by_cases hnext : b ∈ E (s + 1)
  77     · have hnot' : b ∉ E ((s + 1) + N) := by
  78         simpa only [Nat.cast_succ, add_assoc, add_comm, add_left_comm] using hnot
  79       obtain ⟨k, hk, hkN, hb, hnot⟩ := ih (s + 1) hnext hnot'
  80       exact ⟨k, by omega, by omega, hb, hnot⟩
  81     · exact ⟨s, le_rfl, by omega, hb, hnext⟩
