  25 def infinityGradedIso (c : SyntheticTowerConvergence P X) (p : ℤ × ℤ) (s : ℤ) :
  26     ((F.obj X).sequence.ssData (s,p.1+s,p.2)).eInfty ≅
  27       (P.filtration X).associatedGraded s p :=
  28   c.identification (s,p.1+s,p.2) ≪≫ eqToIso (by simp)

  30 theorem elementMap_injective {A : ModuleCat.{v} ℤ} :
  31     Function.Injective (elementMap (A := A)) := by
  32   intro a b h
  33   have h' := ConcreteCategory.congr_hom h (ULift.up (1 : ℤ))
  34   simpa [elementMap, one_smul] using h'

  43 theorem tower_extension_iff
  44     (M : ConvergentMap F g cx.toConvergence cy.toConvergence)
  45     (p : ℤ × ℤ) (s n : ℤ)
  46     (x : ((F.obj X).sequence.ssData (s,p.1+s,p.2)).eInfty)
  47     (y : ((F.obj Y).sequence.ssData (s+n,p.1+(s+n),p.2)).eInfty) :
  48     TowerExtensionEquation M p s n x y ↔
  49       0 ≤ n ∧ ∃ (a : (Subobject.underlying.obj ((P.filtration X).F s p) : ModuleCat.{v} ℤ))
  50         (b : (Subobject.underlying.obj ((P.filtration Y).F (s+n) p) : ModuleCat.{v} ℤ)),
  51         ((P.filtration X).F s p).arrow a ≫ g = ((P.filtration Y).F (s+n) p).arrow b ∧
  52         (P.filtration X).toAssociatedGraded s p a = (infinityGradedIso cx p s).hom x ∧
  53         (P.filtration Y).toAssociatedGraded (s+n) p b = (infinityGradedIso cy p (s+n)).hom y := by
  54   unfold TowerExtensionEquation
  55   rw [M.extensionEquation_iff]
  56   simp only [toGraded_element]
  57   constructor
  58   · rintro ⟨hn,a,b,ha,hb,he⟩
  59     refine ⟨hn,a (ULift.up 1),b (ULift.up 1),?_,?_,?_⟩
  60     · exact ConcreteCategory.congr_hom he (ULift.up 1)
  61     · simpa [elementMap, one_smul] using ConcreteCategory.congr_hom ha (ULift.up (1 : ℤ))
  62     · simpa [elementMap, one_smul] using ConcreteCategory.congr_hom hb (ULift.up (1 : ℤ))
  63   · rintro ⟨hn,a,b,he,ha,hb⟩
  64     refine ⟨hn,elementMap a,elementMap b,?_,?_,?_⟩
  65     · rw [← elementMap_comp,ha]
  66     · rw [← elementMap_comp,hb]
  67     · rw [← Category.assoc,← elementMap_comp,← elementMap_comp,← elementMap_comp]
  68       exact congrArg elementMap he

  70 theorem filtration_representative_exists (P : SyntheticTowerPresentation unit F)
  71     (A : Syn) (p : ℤ × ℤ) (s : ℤ) (a : BiHom p.1 p.2 A)
  72     (ha : a ∈ P.filtrationSubmodule A s p) :
  73     ∃ a' : (Subobject.underlying.obj ((P.filtration A).F s p) : ModuleCat.{v} ℤ),
  74       ((P.filtration A).F s p).arrow a' = a := by
  75   have hm : a ∈ ModuleCat.subobjectModule _ ((P.filtration A).F s p) := by
  76     simpa [SyntheticTowerPresentation.filtration, syntheticHomotopy] using ha
  77   exact hm

  79 theorem map_preserves_filtration
  80     (M : ConvergentMap F g cx.toConvergence cy.toConvergence)
  81     (p : ℤ × ℤ) (s : ℤ) (a : BiHom p.1 p.2 X)
  82     (ha : a ∈ P.filtrationSubmodule X s p) :
  83     a ≫ g ∈ P.filtrationSubmodule Y s p := by
  84   obtain ⟨a',ha'⟩ := filtration_representative_exists P X p s a ha
  85   have h := ConcreteCategory.congr_hom (M.filteredMap.compat s p).choose_spec a'
  86   have hm : a ≫ g ∈ ModuleCat.subobjectModule _ ((P.filtration Y).F s p) := by
  87     refine ⟨(M.filteredMap.compat s p).choose a',?_⟩
  88     change ((P.filtration Y).F s p).arrow ((M.filteredMap.compat s p).choose a') = a ≫ g
  89     change ((P.filtration Y).F s p).arrow ((M.filteredMap.compat s p).choose a') =
  90       M.filteredMap.map p (((P.filtration X).F s p).arrow a') at h
  91     have hh := ConcreteCategory.congr_hom (M.filteredMap_map p)
  92       (((P.filtration X).F s p).arrow a')
  93     change M.filteredMap.map p (((P.filtration X).F s p).arrow a') =
  94       ((P.filtration X).F s p).arrow a' ≫ g at hh
  95     exact h.trans (hh.trans (congrArg (fun z => z ≫ g) ha'))
  96   simpa [SyntheticTowerPresentation.filtration, syntheticHomotopy] using hm

  98 theorem infinity_graded_naturality
  99     (M : ConvergentMap F g cx.toConvergence cy.toConvergence)
 100     (p : ℤ × ℤ) (s : ℤ) :
 101     (infinityGradedIso cx p s).hom ≫ M.filteredMap.inducedGrMap s p =
 102       (F.functor.map g).eInftyMap (s,p.1+s,p.2) ≫ (infinityGradedIso cy p s).hom := by
 103   have h := M.infinity_action_graded (s,p.1+s,p.2)
 104   have transport {ι : Type} (A B : ι → ModuleCat.{v} ℤ)
 105       (f : ∀ i, A i ⟶ B i) {i j : ι} (hij : i=j)
 106       {U V : ModuleCat.{v} ℤ} (a : U ⟶ A i) (b : V ⟶ B i)
 107       (q : U ⟶ V) (hq : a ≫ f i = q ≫ b) :
 108       (a ≫ eqToHom (congrArg A hij)) ≫ f j =
 109         q ≫ (b ≫ eqToHom (congrArg B hij)) := by
 110     subst j
 111     simpa only [eqToHom_refl,Category.comp_id] using hq
 112   exact transport
 113     (fun p => (P.filtration X).associatedGraded s p)
 114     (fun p => (P.filtration Y).associatedGraded s p)
 115     (fun p => M.filteredMap.inducedGrMap s p)
 116     (show (p.1+s-s,p.2)=p by ext <;> simp) _ _ _ h

 118 theorem infinity_image_iff_lift
 119     (M : ConvergentMap F g cx.toConvergence cy.toConvergence)
 120     (p : ℤ × ℤ) (s : ℤ)
 121     (x : (Subobject.underlying.obj ((P.filtration Y).F s p) : ModuleCat.{v} ℤ)) :
 122     (infinityGradedIso cy p s).inv ((P.filtration Y).toAssociatedGraded s p x) ∈
 123       Set.range ((F.functor.map g).eInftyMap (s,p.1+s,p.2)).hom ↔
 124     ∃ a : BiHom p.1 p.2 X, a ∈ P.filtrationSubmodule X s p ∧
 125       @HSub.hSub (BiHom p.1 p.2 Y) (BiHom p.1 p.2 Y) (BiHom p.1 p.2 Y)
 126         inferInstance (((P.filtration Y).F s p).arrow x) (a ≫ g) ∈
 127         P.filtrationSubmodule Y (s+1) p := by
 128   have hn := infinity_graded_naturality M p s
 129   have hi : (∃ z, (F.functor.map g).eInftyMap (s,p.1+s,p.2) z =
 130       (infinityGradedIso cy p s).inv ((P.filtration Y).toAssociatedGraded s p x)) ↔
 131       ∃ z, M.filteredMap.inducedGrMap s p z = (P.filtration Y).toAssociatedGraded s p x := by
 132     constructor
 133     · rintro ⟨z,hz⟩
 134       refine ⟨(infinityGradedIso cx p s).hom z,?_⟩
 135       have he := ConcreteCategory.congr_hom hn z
 136       change M.filteredMap.inducedGrMap s p ((infinityGradedIso cx p s).hom z) =
 137         (infinityGradedIso cy p s).hom ((F.functor.map g).eInftyMap (s,p.1+s,p.2) z) at he
 138       rw [hz] at he
 139       exact he.trans ((infinityGradedIso cy p s).toLinearEquiv.apply_symm_apply _)
 140     · rintro ⟨z,hz⟩
 141       refine ⟨(infinityGradedIso cx p s).inv z,?_⟩
 142       apply (infinityGradedIso cy p s).toLinearEquiv.injective
 143       have he := ConcreteCategory.congr_hom hn ((infinityGradedIso cx p s).inv z)
 144       simp only [ModuleCat.comp_apply] at he
 145       have ha := (infinityGradedIso cx p s).toLinearEquiv.apply_symm_apply z
 146       change (infinityGradedIso cx p s).hom ((infinityGradedIso cx p s).inv z) = z at ha
 147       rw [ha,hz] at he
 148       exact he.symm.trans ((infinityGradedIso cy p s).toLinearEquiv.apply_symm_apply _).symm
 149   change (∃ z, _ = _) ↔ _
 150   refine hi.trans ((graded_image_iff_lift M.filteredMap s p x).trans ?_)
 151   simp only [ConvergentMap.filteredMap_map,SyntheticTowerPresentation.filtration,
 152     OrderIso.apply_symm_apply]
 153   rfl

 155 theorem next_filtration_of_infinity_injective
 156     (M : ConvergentMap F g cx.toConvergence cy.toConvergence)
 157     (p : ℤ × ℤ) (s : ℤ)
 158     (hf : Function.Injective ((F.functor.map g).eInftyMap (s,p.1+s,p.2)).hom)
 159     (a : BiHom p.1 p.2 X) (ha : a ∈ P.filtrationSubmodule X s p)
 160     (hb : a ≫ g ∈ P.filtrationSubmodule Y (s+1) p) :
 161     a ∈ P.filtrationSubmodule X (s+1) p := by
 162   have hn := infinity_graded_naturality M p s
 163   have hmono : Mono (M.filteredMap.inducedGrMap s p) := by
 164     rw [ModuleCat.mono_iff_injective]
 165     intro x y hxy
 166     have he (z : (P.filtration X).associatedGraded s p) :
 167         M.filteredMap.inducedGrMap s p z =
 168         (infinityGradedIso cy p s).hom
 169           ((F.functor.map g).eInftyMap (s,p.1+s,p.2) ((infinityGradedIso cx p s).inv z)) := by
 170       have h := ConcreteCategory.congr_hom hn ((infinityGradedIso cx p s).inv z)
 171       simp only [ModuleCat.comp_apply] at h
 172       have hc := (infinityGradedIso cx p s).toLinearEquiv.apply_symm_apply z
 173       change (infinityGradedIso cx p s).hom ((infinityGradedIso cx p s).inv z) = z at hc
 174       rwa [hc] at h
 175     rw [he,he] at hxy
 176     exact (infinityGradedIso cx p s).symm.toLinearEquiv.injective
 177       (hf ((infinityGradedIso cy p s).toLinearEquiv.injective hxy))
 178   have h := next_filtration_mem_of_graded_mono M.filteredMap s p a
 179   simp only [SyntheticTowerPresentation.filtration,OrderIso.apply_symm_apply,
 180     ConvergentMap.filteredMap_map] at h
 181   exact h ha hb

 183 theorem representative_grade_naturality
 184     (M : ConvergentMap F g cx.toConvergence cy.toConvergence)
 185     (p : ℤ × ℤ) (s : ℤ)
 186     (a : (Subobject.underlying.obj ((P.filtration X).F s p) : ModuleCat.{v} ℤ))
 187     (b : (Subobject.underlying.obj ((P.filtration Y).F s p) : ModuleCat.{v} ℤ))
 188     (hab : ((P.filtration X).F s p).arrow a ≫ g = ((P.filtration Y).F s p).arrow b) :
 189     (infinityGradedIso cy p s).inv ((P.filtration Y).toAssociatedGraded s p b) =
 190       (F.functor.map g).eInftyMap (s,p.1+s,p.2)
 191         ((infinityGradedIso cx p s).inv ((P.filtration X).toAssociatedGraded s p a)) := by
 192   have hb : (M.filteredMap.compat s p).choose a = b := by
 193     apply (ModuleCat.mono_iff_injective ((P.filtration Y).F s p).arrow).mp inferInstance
 194     have h := ConcreteCategory.congr_hom (M.filteredMap.compat s p).choose_spec a
 195     change ((P.filtration Y).F s p).arrow ((M.filteredMap.compat s p).choose a) =
 196       M.filteredMap.map p (((P.filtration X).F s p).arrow a) at h
 197     have hh := ConcreteCategory.congr_hom (M.filteredMap_map p)
 198       (((P.filtration X).F s p).arrow a)
 199     change M.filteredMap.map p (((P.filtration X).F s p).arrow a) =
 200       ((P.filtration X).F s p).arrow a ≫ g at hh
 201     exact h.trans (hh.trans hab)
 202   have hp := graded_map_representative M.filteredMap s p a
 203   rw [hb] at hp
 204   apply (infinityGradedIso cy p s).toLinearEquiv.injective
 205   have he := ConcreteCategory.congr_hom (infinity_graded_naturality M p s)
 206     ((infinityGradedIso cx p s).inv ((P.filtration X).toAssociatedGraded s p a))
 207   simp only [ModuleCat.comp_apply] at he
 208   have hA := (infinityGradedIso cx p s).toLinearEquiv.apply_symm_apply
 209     ((P.filtration X).toAssociatedGraded s p a)
 210   have hB := (infinityGradedIso cy p s).toLinearEquiv.apply_symm_apply
 211     ((P.filtration Y).toAssociatedGraded s p b)
 212   exact hB.trans (hp.symm.trans ((congrArg (M.filteredMap.inducedGrMap s p) hA.symm).trans he))

 214 theorem grade_difference
 215     (M : ConvergentMap F g cx.toConvergence cy.toConvergence)
 216     (p : ℤ × ℤ) (s : ℤ)
 217     (a : (Subobject.underlying.obj ((P.filtration X).F s p) : ModuleCat.{v} ℤ))
 218     (b : (Subobject.underlying.obj ((P.filtration Y).F s p) : ModuleCat.{v} ℤ))
 219     (he : (F.functor.map g).eInftyMap (s,p.1+s,p.2)
 220       ((infinityGradedIso cx p s).inv ((P.filtration X).toAssociatedGraded s p a)) =
 221       (infinityGradedIso cy p s).inv ((P.filtration Y).toAssociatedGraded s p b)) :
 222     @HSub.hSub (BiHom p.1 p.2 Y) (BiHom p.1 p.2 Y) (BiHom p.1 p.2 Y)
 223       inferInstance (((P.filtration Y).F s p).arrow b)
 224       (((P.filtration X).F s p).arrow a ≫ g) ∈
 225       P.filtrationSubmodule Y (s+1) p := by
 226   let b' := (M.filteredMap.compat s p).choose a
 227   have hb' : ((P.filtration Y).F s p).arrow b' = ((P.filtration X).F s p).arrow a ≫ g := by
 228     have h := ConcreteCategory.congr_hom (M.filteredMap.compat s p).choose_spec a
 229     change ((P.filtration Y).F s p).arrow b' =
 230       M.filteredMap.map p (((P.filtration X).F s p).arrow a) at h
 231     rw [ConvergentMap.filteredMap_map] at h
 232     exact h
 233   have hh := representative_grade_naturality M p s a b' hb'.symm
 234   have heq := (infinityGradedIso cy p s).symm.toLinearEquiv.injective (he.symm.trans hh.symm)
 235   have hm := (element_grade_equal_iff (P.filtration Y) s p b b').mp heq
 236   rw [hb'] at hm
 237   simpa only [SyntheticTowerPresentation.filtration,OrderIso.apply_symm_apply,
 238     syntheticHomotopy] using hm
