 124 private theorem lambdaRawInfinity_zero (A : SyntheticObject) (p : ℤ×ℤ) (s : ℤ)
 125     (e : ((model.family.obj (A.obj model.nu model.auxiliary)).sequence.ssData (s,p.1+s,p.2)).eInfty) :
 126     HEq (lambdaRawInfinity A 0 p s e) e := by
 127   let F := model.syntheticTower.filtration (A.obj model.nu model.auxiliary)
 128   haveI : Epi (F.toAssociatedGraded s p) := inferInstanceAs (Epi (cokernel.π _))
 129   obtain ⟨ar,har⟩ := (ModuleCat.epi_iff_surjective (F.toAssociatedGraded s p)).mp
 130     inferInstance ((infinityGradedIso (model.convergence A) p s).hom e)
 131   obtain ⟨br,hbr,hgbr⟩ := shift_frame A (0,0) p s e ar har
 132   obtain ⟨cr,hcr,hgcr⟩ := tower_map_frame
 133     (model.convergentMap (.shift (0,0) A) A (lambdaPow 0 (A.obj model.nu model.auxiliary)))
 134     (p+(0,0)) s _ br hgbr
 135   have hactual := shift_zero_lambda (A.obj model.nu model.auxiliary) p ((F.F s p).arrow ar)
 136   rw [hbr] at hcr
 137   have hh : homotopyRegrade (by simp) (by simp)
 138       (((model.syntheticTower.filtration (A.obj model.nu model.auxiliary)).F s (p+(0,0))).arrow cr) =
 139       (F.F s p).arrow ar := by
 140     rw [←hcr,regrade_comp]
 141     exact hactual
 142   have transport_frame (p q : ℤ×ℤ) (h : p=q)
 143       (a : (Subobject.underlying.obj (F.F s p) : ModuleCat ℤ))
 144       (b : (Subobject.underlying.obj (F.F s q) : ModuleCat ℤ))
 145       (ha : homotopyRegrade (congrArg Prod.fst h.symm) (congrArg Prod.snd h.symm)
 146         ((F.F s q).arrow b) = (F.F s p).arrow a) :
 147       HEq (F.toAssociatedGraded s q b) (F.toAssociatedGraded s p a) := by
 148     subst q
 149     have hb : b=a := (ModuleCat.mono_iff_injective (F.F s p).arrow).mp inferInstance ha
 150     subst b
 151     rfl
 152   have hgr := transport_frame p (p+(0,0)) (by simp) ar cr hh
 153   rw [har,hgcr] at hgr
 154   have transport_grade (p q : ℤ×ℤ) (h : p=q)
 155       (z : ((model.family.obj (A.obj model.nu model.auxiliary)).sequence.ssData (s,p.1+s,p.2)).eInfty)
 156       (z' : ((model.family.obj (A.obj model.nu model.auxiliary)).sequence.ssData (s,q.1+s,q.2)).eInfty)
 157       (hz : HEq ((infinityGradedIso (model.convergence A) q s).hom z')
 158         ((infinityGradedIso (model.convergence A) p s).hom z)) : HEq z' z := by
 159     subst q
 160     exact heq_of_eq ((infinityGradedIso (model.convergence A) p s).toLinearEquiv.injective (eq_of_heq hz))
 161   exact transport_grade p (p+(0,0)) (by simp) e _ hgr
