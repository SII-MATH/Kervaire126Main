  37     (a : BiHom (t-s) (t-d) (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) i))
  38     (ha : Detects (model.convergence (.quotient i (.nu X))) (s,t,t-d)

  48   let A := model.nu.functor.obj (X.obj model.auxiliary)

  68   obtain ⟨z,hz⟩ := nu_first_label_exists X (s+(q-d:ℕ)+1) (t+(q-d:ℕ)) β
  69   have hdz := adjacent_boundary_differential X (q-d) hk s t x0 z c hx0 hz

  71     homotopyRegrade (by omega) (by omega)
  72       (biSuspensionHomEquiv (t-s) (t-d) q A (b ≫ XModLambdaN.proj A q))
  73   have hα : α ≫ XModLambdaN.incl A 1 = firstLabel model X (s+(q-d:ℕ)+1) (t+(q-d:ℕ)) z ≫
  74       XModLambdaN.map (SyntheticCategory.biShift_zero.hom.app A) 1 := by
  75     have hdiv := divided_boundary_first_quotient X d q hd hdq (t-s) t c
  76     dsimp only at hdiv

  78     have hdiv := congrArg (homotopyRegrade (by omega : (t-s)-1=(t+(q-d:ℕ))-(s+(q-d:ℕ)+1)) rfl) hdiv
  79     have hαβ : α ≫ XModLambdaN.incl A 1 = β ≫ XModLambdaN.incl A 1 := by

  82   have hp := first_label_lift_permanent X (s+(q-d:ℕ)+1) (t+(q-d:ℕ)) z α hα

  84       (nuWeightLabel model X (s+(q-d:ℕ)+1) (t+(q-d:ℕ)) (t-d+i) (by omega) z)

  95       have hdet := nu_weight_detects_transport X (s+(q-d:ℕ)+1) (t+(q-d:ℕ))

 112 private theorem finite_detected_boundary_differential_weighted (X : ClassicalObject)
 113     (i k d : ℕ) (hdi : d < i) (hik : i-d ≤ k) (s t : ℤ)
 114     (x : Ambient model X (s,t)) (y : Ambient model X (s+k+1,t+k))
 115     (a : BiHom (t-s) (t-d) (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) i))
 116     (ha : Detects (model.convergence (.quotient i (.nu X))) (s,t,t-d)
 117       ((model.family.quotientProjection (model.nu.functor.obj (X.obj model.auxiliary)) i).pageMap 2 (s,t,t-d)
 118         (nuWeightLabel model X s t (t-d) (by omega) x)) a)
 119     (hy : Detects (model.convergence (.nu X)) (s+k+1,t+k,t-d+i)
 120       (nuWeightLabel model X (s+k+1) (t+k) (t-d+i) (by omega) y)
 121       (homotopyRegrade (by dsimp; omega) rfl
 122         (biSuspensionHomEquiv (t-s) (t-d) i (model.nu.functor.obj (X.obj model.auxiliary))
 123           (a ≫ XModLambdaN.proj _ i)))) :
 124     HasDifferential (model.classical.obj (X.obj model.auxiliary))
 125       (k+1) (s,t) (s+k+1,t+k) x y := by
 126   by_cases hd : d=0
 127   · subst d
 128     simp only [Nat.sub_zero] at hik
 129     have H (w : ℤ) (hw : w=t)
 130         (a0 : BiHom (t-s) w (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) i))
 131         (ha0 : Detects (model.convergence (.quotient i (.nu X))) (s,t,w)
 132           ((model.family.quotientProjection (model.nu.functor.obj (X.obj model.auxiliary)) i).pageMap 2 (s,t,w)
 133             (nuWeightLabel model X s t w (by omega) x)) a0)
 134         (hy0 : Detects (model.convergence (.nu X)) (s+k+1,t+k,w+i)
 135           (nuWeightLabel model X (s+k+1) (t+k) (w+i) (by omega) y)
 136           (homotopyRegrade (by dsimp; omega) rfl
 137             (biSuspensionHomEquiv (t-s) w i (model.nu.functor.obj (X.obj model.auxiliary))
 138               (a0 ≫ XModLambdaN.proj _ i)))) :
 139         HasDifferential (model.classical.obj (X.obj model.auxiliary))
 140           (k+1) (s,t) (s+k+1,t+k) x y := by
 141       subst w
 142       exact finite_detected_boundary_differential X i k (by omega) hik s t x y a0 ha0 hy0
 143     exact H (t-(0:ℕ)) (by omega) a ha hy
 144   · generalize hq : k+d=q at *
 145     have he : q-d=k := by omega
 146     subst k
 147     exact finite_detected_boundary_differential_weighted_pos X i q d
 148       (by omega) hdi (by omega) s t x y a ha hy
