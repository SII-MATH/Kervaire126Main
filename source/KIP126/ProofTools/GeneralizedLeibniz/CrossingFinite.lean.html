   5 /-! Finite-quotient bounds used in the crossing comparison of AimPaper
   6 Proposition `prop:cross-f-Er`. Actual λ division of a higher-filtration source
   7 forces its normalized image into filtration at least `u + e(f)`. Thus crossing
   8 is impossible at lengths at most `e(f)`; at length at most `e(f) + 1` its
   9 position is forced. The general finite no-crossing conversion is not asserted. -/

  23 private theorem normalized_finite_source_division (X : ClassicalObject)
  24     (e : ℤ) (i q : ℕ) (hi : 0 < i) (hiq : i < q) (m w u : ℤ)
  25     (hs : (i:ℤ) ≤ m+u-(w-e))
  26     (a : BiHom m w (XModLambdaN ((SyntheticCategory.biShift (0,e)).obj
  27       (model.nu.functor.obj (X.obj model.auxiliary))) q))
  28     (ha : FiltrationAtLeast model.syntheticTower u a) :
  29     ∃ b : BiHom m w ((SyntheticCategory.biShift (0,-(i:ℤ))).obj
  30         (XModLambdaN ((SyntheticCategory.biShift (0,e)).obj
  31           (model.nu.functor.obj (X.obj model.auxiliary))) (q-i))),
  32       b ≫ ((model.quotients.tower _).triangle hi hiq).lambdaMap = a := by
  33   let A : SyntheticObject := .shift (0,e) (.nu X)
  34   let T := (model.quotients.tower (A.obj model.nu model.auxiliary)).triangle hi hiq
  35   letI := model.exactness.shiftCommShift (0,e)
  36   letI := model.exactness.shiftExact (0,e)
  37   obtain ⟨Q,_,_⟩ := quotient_shift_exists model.quotients.coherence (0,e) i
  38     (model.nu.functor.obj (X.obj model.auxiliary))
  39   have harho : FiltrationAtLeast model.syntheticTower u (a≫T.rho) :=
  40     map_preserves_filtration (model.convergentMap (.quotient q A) (.quotient i A) T.rho)
  41       (m,w) u a ha
  42   have hQ : FiltrationAtLeast model.syntheticTower u (a≫T.rho≫Q.hom) := by
  43     change (a≫T.rho≫Q.hom) ∈ model.syntheticTower.filtrationSubmodule
  44       ((SyntheticObject.shift (0,e) (.quotient i (.nu X))).obj model.nu model.auxiliary) u (m,w)
  45     simpa only [Category.assoc] using map_preserves_filtration (model.convergentMap (.quotient i A)
  46       (.shift (0,e) (.quotient i (.nu X))) Q.hom) (m,w) u (a≫T.rho) harho
  47   have hz := shifted_finite_high_filtration_zero X i hi e m w u hs _ hQ
  48   have hrho : a≫T.rho=0 := by
  49     have he := congrArg (fun z => z≫Q.inv) hz
  50     simpa only [Category.assoc,Iso.hom_inv_id,Category.comp_id,zero_comp] using he
  51   obtain ⟨b,hb⟩ := (Triangle.mk T.lambdaMap T.rho T.delta).coyoneda_exact₂
  52     T.distinguished a hrho
  53   exact ⟨b,hb.symm⟩

  55 private theorem shifted_quotient_bottom_filtration (X : ClassicalObject)
  56     (q : ℕ) (hq : 0 < q) (i m w : ℤ)
  57     (a : BiHom m w ((SyntheticCategory.biShift (0,i)).obj
  58       (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) q))) :
  59     FiltrationAtLeast model.syntheticTower (w-m-i) a := by
  60   let A : SyntheticObject := .shift (0,i) (.quotient q (.nu X))
  61   let u : ℤ := w-m-i
  62   have hzero (j : ℤ) (hj : j < u) :
  63       IsZero (((model.family.obj (A.obj model.nu model.auxiliary)).sequence.ssData (j,m+j,w)).eInfty) := by
  64     have hz := finite_infinity_zero_outside X q hq j (m+j) (w-i) (by dsimp [u] at hj; omega)
  65     let e := towerInfinityIso (model.convergence (.quotient q (.nu X))) (model.convergence A)
  66       ((essShift (0,i)).filtered (.quotient q (.nu X)) (m,w-i)) j
  67     have he := hz.of_iso e.symm
  68     simpa only [e,A,Prod.fst_add,Prod.snd_add,add_zero,sub_add_cancel] using he
  69   have ha0 := filtration_at_zero model.syntheticTower _ m w a
  70   by_cases hu : u≤0
  71   · exact model.syntheticTower.filtrationSubmodule_antitone _ (m,w) hu ha0
  72   · exact (filtration_iff_of_infinity_zero (model.convergence A) 0 u m w (by omega)
  73       (fun j _ hj => hzero j hj) a).mp ha0

  75 private theorem finite_normalized_image_filtration (X Y : ClassicalObject)
  76     (f : X.obj model.auxiliary ⟶ Y.obj model.auxiliary)
  77     (q i : ℕ) (hi : 0 < i) (hiq : i < q) (s t : ℤ)
  78     (a : BiHom (t-s) (t+normalizedExponent model.classicalTower f)
  79       (XModLambdaN ((extensionSourceObject model X Y f).obj model.nu model.auxiliary) q))
  80     (ha : FiltrationAtLeast model.syntheticTower (s+i) a) :
  81     FiltrationAtLeast model.syntheticTower (s+i+normalizedExponent model.classicalTower f)
  82       (a≫finiteExtensionMap model X Y f q) := by
  83   let e : ℤ := normalizedExponent model.classicalTower f
  84   let A := (extensionSourceObject model X Y f).obj model.nu model.auxiliary
  85   let B := model.nu.functor.obj (Y.obj model.auxiliary)
  86   let F := SyntheticCategory.biShift (Syn:=SyntheticSpectrum) (0,-(i:ℤ))
  87   let Ta := (model.quotients.tower A).triangle hi hiq
  88   let Tb := (model.quotients.tower B).triangle hi hiq
  89   obtain ⟨b,hb⟩ := normalized_finite_source_division X e i q hi hiq (t-s) (t+e) (s+i)
  90     (by omega) a ha
  91   let c := b≫F.map (finiteExtensionMap model X Y f (q-i))
  92   have hc : FiltrationAtLeast model.syntheticTower (s+i+e) c := by
  93     have h := shifted_quotient_bottom_filtration Y (q-i) (by omega) (-(i:ℤ)) (t-s) (t+e) c
  94     convert h using 1; omega
  95   have hcomm := (model.quotients.natural (model.normalizedMap X Y f).map).lambda_naturality hi hiq
  96   change F.map (finiteExtensionMap model X Y f (q-i))≫Tb.lambdaMap =
  97     Ta.lambdaMap≫finiteExtensionMap model X Y f q at hcomm
  98   change b≫Ta.lambdaMap=a at hb
  99   have he : c≫Tb.lambdaMap=a≫finiteExtensionMap model X Y f q := by
 100     dsimp only [c]
 101     rw [Category.assoc,hcomm,←Category.assoc,hb]
 102   rw [←he]
 103   exact map_preserves_filtration (model.convergentMap (.shift (0,-(i:ℤ)) (.quotient (q-i) (.nu Y)))
 104     (.quotient q (.nu Y)) Tb.lambdaMap) (t-s,t+e) (s+i+e) c hc

 106 private theorem finite_crossing_target_bound (X Y : ClassicalObject)
 107     (f : X.obj model.auxiliary ⟶ Y.obj model.auxiliary) (q : ℕ) (hq : 0 < q)
 108     (s t u v : ℤ) (hs : s < u) (huv : u ≤ v)
 109     (x : ((model.family.obj ((SyntheticObject.quotient q (extensionSourceObject model X Y f)).obj model.nu model.auxiliary)).sequence.ssData
 110       (u,(t-s)+u,t+normalizedExponent model.classicalTower f)).eInfty)
 111     (y : ((model.family.obj ((SyntheticObject.quotient q (.nu Y)).obj model.nu model.auxiliary)).sequence.ssData
 112       (u+(v-u),(t-s)+(u+(v-u)),t+normalizedExponent model.classicalTower f)).eInfty)
 113     (he : TowerEssentialExtension (model.convergentMap
 114       (.quotient q (extensionSourceObject model X Y f)) (.quotient q (.nu Y))
 115       (finiteExtensionMap model X Y f q))
 116       (t-s,t+normalizedExponent model.classicalTower f) u (v-u) x y) :
 117     u+normalizedExponent model.classicalTower f ≤ v := by
 118   have hiq := finite_crossing_source_bound X Y f q hq s t u v huv x y he
 119   let i := (u-s).toNat
 120   have hi : 0 < i := by dsimp only [i]; omega
 121   have hiz : (i:ℤ)=u-s := Int.toNat_of_nonneg (by omega)
 122   have hui : s+(i:ℤ)=u := by omega
 123   let A := (SyntheticObject.quotient q (extensionSourceObject model X Y f)).obj model.nu model.auxiliary
 124   let B := (SyntheticObject.quotient q (.nu Y)).obj model.nu model.auxiliary
 125   let p := (t-s,t+normalizedExponent model.classicalTower f)
 126   have hrel : TowerExtensionEquation (model.convergentMap
 127       (.quotient q (extensionSourceObject model X Y f)) (.quotient q (.nu Y))
 128       (finiteExtensionMap model X Y f q)) p u (v-u) x y := he.1
 129   obtain ⟨_,ar,br,hab,hax,hby⟩ := (tower_extension_iff _ _ _ _ _ _).mp hrel
 130   let α := ((model.syntheticTower.filtration A).F u p).arrow ar
 131   have hα : FiltrationAtLeast model.syntheticTower u α := by
 132     have hm : α ∈ ModuleCat.subobjectModule _ ((model.syntheticTower.filtration A).F u p) := ⟨ar,rfl⟩
 133     simpa only [SyntheticTowerPresentation.filtration,OrderIso.apply_symm_apply,FiltrationAtLeast] using hm
 134   have himage := finite_normalized_image_filtration X Y f q i hi (by omega) s t α (by
 135     rw [hui]
 136     exact hα)
 137   by_contra hv
 138   have hhigh : FiltrationAtLeast model.syntheticTower (u+(v-u)+1)
 139       (α≫finiteExtensionMap model X Y f q) :=
 140     model.syntheticTower.filtrationSubmodule_antitone _ p (by omega) himage
 141   have hdiff : ((model.syntheticTower.filtration B).F (u+(v-u)) p).arrow br -
 142       ((model.syntheticTower.filtration B).F (u+(v-u)) p).arrow 0 ∈
 143       ModuleCat.subobjectModule _ ((model.syntheticTower.filtration B).F (u+(v-u)+1) p) := by
 144     rw [map_zero,sub_zero,←hab]
 145     change (α≫finiteExtensionMap model X Y f q) ∈
 146       ModuleCat.subobjectModule _ ((model.syntheticTower.filtration B).F (u+(v-u)+1) p)
 147     simp only [SyntheticTowerPresentation.filtration,OrderIso.apply_symm_apply]
 148     obtain ⟨c,hc⟩ := hhigh
 149     exact ⟨c,hc⟩
 150   have hzero := (element_grade_equal_iff (model.syntheticTower.filtration B)
 151     (u+(v-u)) p br 0).mpr hdiff
 152   simp only [map_zero] at hzero
 153   have hy : y=0 := by
 154     apply (infinityGradedIso (model.convergence (.quotient q (.nu Y))) p (u+(v-u))).toLinearEquiv.injective
 155     change (infinityGradedIso (model.convergence (.quotient q (.nu Y))) p (u+(v-u))).hom y =
 156       (infinityGradedIso (model.convergence (.quotient q (.nu Y))) p (u+(v-u))).hom 0
 157     rw [←hby,map_zero]
 158     exact hzero
 159   have hzrel : TowerExtensionEquation (model.convergentMap
 160       (.quotient q (extensionSourceObject model X Y f)) (.quotient q (.nu Y))
 161       (finiteExtensionMap model X Y f q)) p u (v-u) x 0 := by
 162     rw [←hy]
 163     exact hrel
 164   exact ((tower_essential_iff_no_zero _ _ _ _ _ _ hrel).mp he) hzrel

 166 private theorem minimal_length_finite_tower_no_crossing (X Y : ClassicalObject)
 167     (f : X.obj model.auxiliary ⟶ Y.obj model.auxiliary) (q : ℕ) (hq : 0 < q)
 168     (s t n : ℤ) (hn : n ≤ normalizedExponent model.classicalTower f) :
 169     TowerNoCrossing (model.convergentMap
 170       (.quotient q (extensionSourceObject model X Y f)) (.quotient q (.nu Y))
 171       (finiteExtensionMap model X Y f q))
 172       (t-s,t+normalizedExponent model.classicalTower f) s n := by
 173   intro u v hs huv hv x y he
 174   have hh := finite_crossing_target_bound X Y f q hq s t u v hs huv x y he
 175   omega

 177 private theorem short_finite_crossing_position (X Y : ClassicalObject)
 178     (f : X.obj model.auxiliary ⟶ Y.obj model.auxiliary) (q : ℕ) (hq : 0 < q)
 179     (s t n u v : ℤ) (hn : n ≤ normalizedExponent model.classicalTower f+1)
 180     (hs : s < u) (huv : u ≤ v) (hv : v ≤ s+n)
 181     (x : ((model.family.obj ((SyntheticObject.quotient q (extensionSourceObject model X Y f)).obj model.nu model.auxiliary)).sequence.ssData
 182       (u,(t-s)+u,t+normalizedExponent model.classicalTower f)).eInfty)
 183     (y : ((model.family.obj ((SyntheticObject.quotient q (.nu Y)).obj model.nu model.auxiliary)).sequence.ssData
 184       (u+(v-u),(t-s)+(u+(v-u)),t+normalizedExponent model.classicalTower f)).eInfty)
 185     (he : TowerEssentialExtension (model.convergentMap
 186       (.quotient q (extensionSourceObject model X Y f)) (.quotient q (.nu Y))
 187       (finiteExtensionMap model X Y f q))
 188       (t-s,t+normalizedExponent model.classicalTower f) u (v-u) x y) :
 189     u=s+1 ∧ v=s+normalizedExponent model.classicalTower f+1 ∧
 190       n=normalizedExponent model.classicalTower f+1 := by
 191   have hh := finite_crossing_target_bound X Y f q hq s t u v hs huv x y he
 192   omega
