  28     (a : BiHom (t-s) t (model.nu.functor.obj (X.obj model.auxiliary))) :
  29     FiltrationAtLeast model.syntheticTower s a := by
  30   have hzero := filtration_at_zero model.syntheticTower _ (t-s) t a
  31   by_cases hs : s ≤ 0

  34       0 s (t-s) t (by omega)
  35       (fun k _ hk => nu_infinity_zero_above X k (t-s+k) t (by omega)) a).mp hzero

  82 private theorem adjacent_absolute_boundary (X : ClassicalObject)
  83     (k : ℕ) (hk : 0<k) (s t : ℤ)
  84     (x : Ambient model X (s,t)) (y : Ambient model X (s+k+1,t+k))
  85     (hd : HasDifferential (model.classical.obj (X.obj model.auxiliary))
  86       (k+1) (s,t) (s+k+1,t+k) x y) :
  87     ∃ a : BiHom (t-s) t (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) k),
  88       a ≫ (model.quotients.tower _).rho 1 k hk =
  89         firstLabel model X s t x ≫ XModLambdaN.map (SyntheticCategory.biShift_zero.hom.app _) 1 ∧
  90       (homotopyRegrade (by omega) rfl
  91         (biSuspensionHomEquiv (t-s) t k (model.nu.functor.obj (X.obj model.auxiliary))
  92           (a ≫ XModLambdaN.proj _ k))) ≫ XModLambdaN.incl _ 1 =
  93         firstLabel model X (s+k+1) (t+k) y ≫ XModLambdaN.map (SyntheticCategory.biShift_zero.hom.app _) 1 := by
  94   let A := model.nu.functor.obj (X.obj model.auxiliary)
  95   obtain ⟨a,ha,had⟩ := adjacent_mahowald_case X k hk s t x y hd
  96   dsimp only at had
  97   let δ : XModLambdaN A k ⟶ ((SyntheticCategory.biShift (0,-(k:ℤ))).obj (XModLambdaN A 1))⟦(1:ℤ)⟧ := by
  98     simpa only [show k.succ-k=1 by omega] using ((model.quotients.tower A).triangle hk (Nat.lt_succ_self k)).delta
  99   have hp : biSuspensionHomEquiv (t-s) t k (XModLambdaN A 1) (a≫δ) =
 100       biSuspensionHomEquiv (t-s) t k A (a≫XModLambdaN.proj A k) ≫ XModLambdaN.incl A 1 := by
 101     have h := cofiber_boundary_projection X k (k+1) hk (Nat.lt_succ_self k) (t-s) t a
 102     have transport (q : ℕ) (hq : q=1)
 103         (d : XModLambdaN A k ⟶ ((SyntheticCategory.biShift (0,-(k:ℤ))).obj (XModLambdaN A q))⟦(1:ℤ)⟧)
 104         (he : (XModLambdaN A k ⟶ ((SyntheticCategory.biShift (0,-(k:ℤ))).obj (XModLambdaN A q))⟦(1:ℤ)⟧) =
 105           (XModLambdaN A k ⟶ ((SyntheticCategory.biShift (0,-(k:ℤ))).obj (XModLambdaN A 1))⟦(1:ℤ)⟧))
 106         (hd : biSuspensionHomEquiv (t-s) t k (XModLambdaN A q) (a≫d) =
 107           biSuspensionHomEquiv (t-s) t k A (a≫XModLambdaN.proj A k) ≫ XModLambdaN.incl A q) :
 108         biSuspensionHomEquiv (t-s) t k (XModLambdaN A 1) (a≫he.mp d) =
 109           biSuspensionHomEquiv (t-s) t k A (a≫XModLambdaN.proj A k) ≫ XModLambdaN.incl A 1 := by
 110       subst q
 111       exact hd
 112     exact transport (k+1-k) (by omega) _ _ h
 113   let b := firstLabel model X (s+k+1) (t+k) y ≫ XModLambdaN.map (SyntheticCategory.biShift_zero.hom.app A) 1
 114   have hb := first_label_detects_unshifted X 1 (by omega) (s+k+1) (t+k) y b (by
 115     rw [(model.quotients.tower _).rho_id,Category.comp_id])
 116   have heq := finite_top_detection_unique X 1 (by omega) (s+k+1) (t+k) (t+k) (by omega)
 117     _ _ b had hb
 118   have hdet : homotopyRegrade (show (t-s)-1=(t+k)-(s+k+1) by omega) rfl
 119       (biSuspensionHomEquiv (t-s) t k (XModLambdaN A 1) (a≫δ)) = b := heq
 120   rw [hp,regrade_comp] at hdet
 121   exact ⟨a,ha,hdet⟩

 123 private theorem nu_lambda_detects (X : ClassicalObject) (k : ℕ) (hk : 0 < k)
 124     (s t : ℤ) (y : Ambient model X (s,t))
 125     (hy : y ∈ permanentCycles model X (s,t))
 126     (α : BiHom (t-s) t (model.nu.functor.obj (X.obj model.auxiliary)))
 127     (hα : α ≫ XModLambdaN.incl _ 1 = firstLabel model X s t y ≫
 128       XModLambdaN.map (SyntheticCategory.biShift_zero.hom.app _) 1) :
 129     Detects (model.convergence (.nu X)) (s,t,t-k)
 130       (nuWeightLabel model X s t (t-k) (by omega) y) (lambdaMultiply k α) := by
 131   let A := model.nu.functor.obj (X.obj model.auxiliary)
 132   have ha := nu_first_label_detects X s t y hy α hα
 133   have hsource : (s,t,t)=(s,(t-s)+s,t) := by simp
 134   have hstem := detects_transport hsource _ _ ha
 135   let L : ((SyntheticObject.shift (0,-(k:ℤ)) (.nu X)).obj model.nu model.auxiliary) ⟶ A := lambdaPow k A
 136   have hmap := shifted_map_detects (.nu X) (.nu X) (0,-(k:ℤ)) (t-s,t) s L
 137     (regradeE2 hsource (nuWeightLabel model X s t t (by omega) y)) α hstem
 138   have hindex : (s,((t-s,t)+(0,-(k:ℤ))).1+s,((t-s,t)+(0,-(k:ℤ))).2) =
 139       (s,t,t-k) := by ext <;> simp [sub_eq_add_neg]
 140   have hmap := detects_transport hindex _ _ hmap
 141   obtain ⟨e,he,b,hb,hgb⟩ := hmap
 142   obtain ⟨e0,he0,_⟩ := ha
 143   have heStem := representative_transport hsource _ _ he0
 144   have heMap := shifted_representative_map (.nu X) (.nu X)
 145     (0,-(k:ℤ)) (t-s,t) s L _ _ heStem
 146   have heMap := representative_transport hindex _ _ heMap
 147   have heq := he.unique heMap
 148   have heq' : e = KIP126.Literature.Standard.infinityLambda (.nu X) k (s,t) t e0 := by
 149     convert heq using 1
 150     simp only [KIP126.Literature.Standard.infinityLambda,
 151       InfinityShiftCoordinates.lambdaMap, show k ≠ 0 by omega, ↓reduceDIte,
 152       InfinityShiftCoordinates.shift, infinityShift, Iso.trans_hom, ModuleCat.comp_apply,
 153       eqToIso.hom, KIP126.Literature.Standard.infinityShiftData,
 154       KIP126.Literature.Standard.infinityShift_comparison]
 155     exact (infinity_map_regrade L hindex _).symm
 156   have hlabel : HasInfinityRepresentative _ 2 (s,t,t-k)
 157       (nuWeightLabel model X s t (t-k) (by omega) y)
 158       (KIP126.Literature.Standard.infinityLambda (.nu X) k (s,t) t e0) := by
 159     apply (KIP126.Literature.Standard.eInfinity_nu_labels X s t (t-k) (by omega) y _).mpr
 160     refine ⟨hy,?_⟩
 161     rw [KIP126.Literature.Standard.eInfinity_lambda_nu X k (s,t) t (by omega)]
 162     have he0' := (KIP126.Literature.Standard.eInfinity_nu_labels X s t t (by omega) y e0).mp he0
 163     rw [he0'.2]
 164     rfl
 165   have heFinal : HasInfinityRepresentative _ 2 (s,t,t-k)
 166       (nuWeightLabel model X s t (t-k) (by omega) y) e := by
 167     rw [heq']
 168     exact hlabel
 169   refine ⟨e,heFinal,b,?_,hgb⟩
 170   convert hb using 1
 171   simp only [regrade_precomp, L, shiftHomotopyClass, lambdaMultiply,
 172     Category.assoc]
 173   simp only [A]
 174   erw [lambdaPow_naturality k α]
 175   simp only [←Category.assoc,eqToHom_trans]

 177 private theorem absolute_boundary_restriction (X : ClassicalObject)
 178     (i k : ℕ) (hik : i ≤ k) (m w : ℤ)
 179     (a : BiHom m w (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) k)) :

 181       ((a ≫ (model.quotients.tower _).rho i k hik) ≫ XModLambdaN.proj _ i) =

 183       (lambdaMultiply (k-i) (biSuspensionHomEquiv m w k

 188     (biSuspensionHomEquiv m w k A (a ≫ XModLambdaN.proj A k))

 191 private theorem lambda_regrade {m m' w w' : ℤ} {A : SyntheticSpectrum}
 192     (hm : m=m') (hw : w=w') (k : ℕ) (a : BiHom m w A) :
 193     lambdaMultiply k (homotopyRegrade hm hw a) =
 194       homotopyRegrade hm (congrArg (fun z : ℤ => z-(k:ℤ)) hw) (lambdaMultiply k a) := by
 195   subst m'; subst w'
 196   rfl

 198 private theorem differential_absolute_boundary (X : ClassicalObject)
 199     (i k : ℕ) (hi : 0 < i) (hik : i ≤ k) (s t : ℤ)
 200     (x : Ambient model X (s,t)) (y : Ambient model X (s+k+1,t+k))
 201     (hd : HasDifferential (model.classical.obj (X.obj model.auxiliary))
 202       (k+1) (s,t) (s+k+1,t+k) x y)
 203     (hy : y ∈ permanentCycles model X (s+k+1,t+k)) :
 204     ∃ a : BiHom (t-s) t (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) i),
 205       a ≫ (model.quotients.tower _).rho 1 i hi =
 206         firstLabel model X s t x ≫ XModLambdaN.map (SyntheticCategory.biShift_zero.hom.app _) 1 ∧
 207       Detects (model.convergence (.nu X)) (s+k+1,t+k,t+i)
 208         (nuWeightLabel model X (s+k+1) (t+k) (t+i) (by omega) y)
 209         (homotopyRegrade (by dsimp; omega) rfl
 210           (biSuspensionHomEquiv (t-s) t i (model.nu.functor.obj (X.obj model.auxiliary))
 211             (a ≫ XModLambdaN.proj _ i))) := by
 212   obtain ⟨a,ha,had⟩ := adjacent_absolute_boundary X k (by omega) s t x y hd
 213   let A := model.nu.functor.obj (X.obj model.auxiliary)
 214   let α : BiHom ((t+k)-(s+k+1)) (t+k) A :=
 215     homotopyRegrade (by omega) rfl
 216       (biSuspensionHomEquiv (t-s) t k A (a ≫ XModLambdaN.proj A k))
 217   have hα : α ≫ XModLambdaN.incl A 1 =
 218       firstLabel model X (s+k+1) (t+k) y ≫ XModLambdaN.map (SyntheticCategory.biShift_zero.hom.app A) 1 := had
 219   refine ⟨a ≫ (model.quotients.tower A).rho i k hik,?_,?_⟩
 220   · rw [Category.assoc,(model.quotients.tower A).rho_comp hi hik]
 221     exact ha
 222   by_cases hik' : i=k
 223   · subst i
 224     have hz := nu_first_label_detects X (s+k+1) (t+k) y hy α hα
 225     simp only [A, (model.quotients.tower _).rho_id,Category.comp_id]
 226     exact hz
 227   have hlam := nu_lambda_detects X (k-i) (by omega) (s+k+1) (t+k) y hy α hα
 228   have hind : (s+k+1,t+k,(t+k)-(k-i:ℕ))=(s+k+1,t+k,t+i) := Prod.ext rfl (Prod.ext rfl (by dsimp; omega))
 229   have hlam := detects_transport hind _ _ hlam
 230   convert hlam using 1
 231   · have transport (w w' : ℤ) (hw : w=w') (hw₁ : w≤t+k) (hw₂ : w'≤t+k)
 232         (h : (s+k+1,t+k,w)=(s+k+1,t+k,w')) :
 233         regradeE2 h (nuWeightLabel model X (s+k+1) (t+k) w hw₁ y) =
 234           nuWeightLabel model X (s+k+1) (t+k) w' hw₂ y := by
 235       subst w'
 236       rfl
 237     exact (transport _ _ (by omega) _ _ hind).symm
 238   · rw [absolute_boundary_restriction]
 239     dsimp only [α]
 240     rw [lambda_regrade]
 241     simp only [regrade_precomp,←Category.assoc,eqToHom_trans]
 242     rfl
