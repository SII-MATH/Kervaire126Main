  22 private theorem lambdaMultiply_quotient_zero (A : SyntheticSpectrum)
  23     (n : ℕ) (m w : ℤ) (a : BiHom m w A) :
  24     lambdaMultiply n a ≫ XModLambdaN.incl A n = 0 := by
  25   have hz : lambdaPow n (Smn m w) ≫ a ≫ XModLambdaN.incl A n = 0 := by
  26     rw [←Category.assoc]
  27     erw [←lambdaPow_naturality n a]
  28     rw [Category.assoc,XModLambdaN.lambdaPow_comp_incl,comp_zero]
  29   simp only [lambdaMultiply,Category.assoc,hz,comp_zero]

  31 private theorem absolute_boundary_filtration_lift (X : ClassicalObject)
  32     (i j : ℕ) (hi : 0 < i) (hij : i < j) (s t : ℤ)
  33     (a : BiHom (t-s) t (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) i))
  34     (hfil : FiltrationAtLeast model.syntheticTower (s+j+1)
  35       (biSuspensionHomEquiv (t-s) t i (model.nu.functor.obj (X.obj model.auxiliary))
  36         (a ≫ XModLambdaN.proj _ i))) :
  37     ∃ b : BiHom (t-s) t (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) j),
  38       b ≫ (model.quotients.tower _).rho i j hij.le = a := by
  39   let A := model.nu.functor.obj (X.obj model.auxiliary)
  40   let α := biSuspensionHomEquiv (t-s) t i A (a ≫ XModLambdaN.proj A i)
  41   obtain ⟨β,hβ⟩ := (KIP126.Literature.Standard.selected_filtration_lambda X
  42     (t-s-1) (t+i) (s+j+1) (by omega) α).mp hfil
  43   have hexp : ((t-s-1)+(s+j+1)-(t+i)).toNat = j-i := by omega
  44   have hzero : α ≫ XModLambdaN.incl A (j-i) = 0 := by
  45     rw [←hβ]
  46     have hz (n q : ℕ) (hn : n = q) : lambdaMultiply n β ≫ XModLambdaN.incl A q = 0 := by
  47       subst q
  48       exact lambdaMultiply_quotient_zero A n _ _ β
  49     simp only [regrade_precomp,Category.assoc,hz _ _ hexp,comp_zero]
  50   let T := (model.quotients.tower A).triangle hi hij
  51   have hz : a ≫ T.delta = 0 := by
  52     apply (biSuspensionHomEquiv (t-s) t i (XModLambdaN A (j-i))).injective
  53     rw [map_zero,cofiber_boundary_projection]
  54     exact hzero
  55   obtain ⟨b,hb⟩ := (Triangle.mk T.lambdaMap T.rho T.delta).coyoneda_exact₃ T.distinguished a hz
  56   refine ⟨b,?_⟩
  57   simpa only [T,Triangle.mk,(model.quotients.tower A).triangle_rho hi hij] using hb.symm

  59 private theorem first_label_lift_cycles (X : ClassicalObject)
  60     (j : ℕ) (hj : 0 < j) (s t : ℤ) (x : Ambient model X (s,t))
  61     (a : BiHom (t-s) t (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) j))
  62     (ha : a ≫ (model.quotients.tower _).rho 1 j hj =
  63       firstLabel model X s t x ≫ XModLambdaN.map (SyntheticCategory.biShift_zero.hom.app _) 1) :
  64     x ∈ cycles model X j (s,t) := by
  65   let A := model.nu.functor.obj (X.obj model.auxiliary)
  66   let inv : A ⟶ nuZero model X := SyntheticCategory.biShift_zero.inv.app A
  67   have hfirst : (a ≫ XModLambdaN.map inv j) ≫ (model.quotients.tower _).rho 1 j hj =
  68       firstLabel model X s t x := by
  69     rw [Category.assoc,←(model.quotients.natural inv).rho_naturality hj,←Category.assoc,ha,
  70       Category.assoc,←model.quotients.functoriality.map_comp]
  71     simp only [inv,A,Iso.hom_inv_id_app,model.quotients.functoriality.map_id,Category.comp_id]
  72   have hx := (KIP126.Literature.Standard.bhs_finite_lift X j hj s t x).mpr
  73     ⟨a ≫ XModLambdaN.map inv j,hfirst⟩
  74   have hc := ((isCycle_iff_reaches model X (j+1) (by omega) (s,t) x).mpr hx).2
  75   simpa only [add_sub_cancel_right] using hc

  77 private theorem absolute_boundary_filtration_cycles (X : ClassicalObject)
  78     (i j : ℕ) (hi : 0 < i) (hij : i ≤ j) (s t : ℤ) (x : Ambient model X (s,t))
  79     (a : BiHom (t-s) t (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) i))
  80     (ha : a ≫ (model.quotients.tower _).rho 1 i hi =
  81       firstLabel model X s t x ≫ XModLambdaN.map (SyntheticCategory.biShift_zero.hom.app _) 1)
  82     (hfil : FiltrationAtLeast model.syntheticTower (s+j+1)
  83       (biSuspensionHomEquiv (t-s) t i (model.nu.functor.obj (X.obj model.auxiliary))
  84         (a ≫ XModLambdaN.proj _ i))) :
  85     x ∈ cycles model X j (s,t) := by
  86   rcases hij.eq_or_lt with rfl | hij
  87   · exact first_label_lift_cycles X i hi s t x a ha
  88   obtain ⟨b,hb⟩ := absolute_boundary_filtration_lift X i j hi hij s t a hfil
  89   apply first_label_lift_cycles X j (by omega) s t x b
  90   rw [←(model.quotients.tower _).rho_comp hi hij.le,←Category.assoc,hb]
  91   exact ha
