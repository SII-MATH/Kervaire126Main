   6 /-! AimPaper's boundary indeterminacy after finite lambda division.
   7 The actual lambda map carries the first label of a chosen smaller-quotient
   8 representative to its finite target label. Equal detected representatives
   9 therefore identify the original label modulo B_(d+1). This comparison concerns
  10 the source label. Detection supplies an actual lambda preimage, including
  11 the zero-exponent case; its first label is chosen internally, and source
  12 boundary ambiguity preserves the classical differential. Compatibility of
  13 the divided absolute-boundary target remains a separate reverse GL step. -/

  36 private theorem finite_quotient_lambda_detects (X : ClassicalObject)
  37     (q d : ℕ) (hd : 0 < d) (hdq : d < q) (s t : ℤ)
  38     (x : Ambient model X (s,t))
  39     (b : BiHom (t-s) t (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) (q-d)))
  40     (hb : b ≫ (model.quotients.tower _).rho 1 (q-d) (by omega) =
  41       firstLabel model X s t x ≫ XModLambdaN.map (SyntheticCategory.biShift_zero.hom.app _) 1) :
  42     Detects (model.convergence (.quotient q (.nu X))) (s,t,t-d)
  43       (finiteTargetLabel model X q s t d x) (finiteLambdaClass X q d hd hdq s t b) := by
  44   let A := model.nu.functor.obj (X.obj model.auxiliary)
  45   let n := q-d
  46   have hn : 0 < n := by dsimp [n]; omega
  47   have ha := first_label_detects_unshifted X n hn s t x b hb
  48   have hx := first_label_lift_cycles X n hn s t x b hb
  49   have hsource : (s,t,t)=(s,(t-s)+s,t) := by simp
  50   have hstem := detects_transport hsource _ _ ha
  51   let L : ((SyntheticObject.shift (0,-(d:ℤ)) (.quotient n (.nu X))).obj model.nu model.auxiliary) ⟶
  52       ((SyntheticObject.quotient q (.nu X)).obj model.nu model.auxiliary) :=
  53     ((model.quotients.tower A).triangle hd hdq).lambdaMap
  54   have hmap := shifted_map_detects (.quotient n (.nu X)) (.quotient q (.nu X))
  55     (0,-(d:ℤ)) (t-s,t) s L
  56     (regradeE2 hsource (regradeE2 (by simp) (finiteTargetLabel model X n s t 0 x))) b hstem
  57   have hindex : (s,((t-s,t)+(0,-(d:ℤ))).1+s,((t-s,t)+(0,-(d:ℤ))).2) =
  58       (s,t,t-d) := by ext <;> simp [sub_eq_add_neg]
  59   have hmap := detects_transport hindex _ _ hmap
  60   obtain ⟨e,he,ar,har,hgar⟩ := hmap
  61   obtain ⟨e0,he0,_⟩ := ha
  62   have heStem := representative_transport hsource _ _ he0
  63   have heMap := shifted_representative_map (.quotient n (.nu X)) (.quotient q (.nu X))
  64     (0,-(d:ℤ)) (t-s,t) s L _ _ heStem
  65   have heMap := representative_transport hindex _ _ heMap
  66   have heq := he.unique heMap
  67   have heq' : e = KIP126.Literature.Standard.infinityFiniteLambda X q d hdq (s,t) t e0 := by
  68     convert heq using 1
  69     simp only [KIP126.Literature.Standard.infinityFiniteLambda,
  70       InfinityShiftCoordinates.finiteLambdaMap, show d ≠ 0 by omega, ↓reduceDIte,
  71       InfinityShiftCoordinates.shift, infinityShift, Iso.trans_hom, ModuleCat.comp_apply,
  72       eqToIso.hom, KIP126.Literature.Standard.infinityShiftData,
  73       KIP126.Literature.Standard.infinityShift_comparison]
  74     rw [infinity_map_regrade L hindex]
  75     rfl
  76   have hlabel : HasInfinityRepresentative _ 2 (s,t,t-d)
  77       (finiteTargetLabel model X q s t d x)
  78       (KIP126.Literature.Standard.infinityFiniteLambda X q d hdq (s,t) t e0) := by
  79     apply (finite_target_label_iff X q (by omega) s t d hdq
  80       ⟨x,by convert hx using 1; congr 1; dsimp [n]; omega⟩ _).mpr
  81     rw [KIP126.Literature.Standard.eInfinity_lambda_finite X q d hdq (s,t) t (by dsimp [n] at hn; omega)]
  82     have he0' := (KIP126.Literature.Standard.eInfinity_finite_labels X n hn s t t
  83       (by constructor <;> dsimp [n] at * <;> omega) x e0).mp
  84       ((congrArg (fun z => HasInfinityRepresentative _ 2 (s,t,t) z e0)
  85         (zero_weight_label X n s t x)).mpr he0)
  86     rw [he0'.2]
  87     rfl
  88   have heFinal : HasInfinityRepresentative _ 2 (s,t,t-d)
  89       (finiteTargetLabel model X q s t d x) e := by
  90     simpa only [heq'] using hlabel
  91   refine ⟨e,heFinal,ar,?_,hgar⟩
  92   convert har using 1
  93   simp only [finiteLambdaClass,regrade_precomp,Category.assoc]
  94   simp only [←Category.assoc,eqToHom_trans]
  95   rfl

  97 private theorem finite_detected_labels_congruent (X : ClassicalObject)
  98     (q : ℕ) (hq : 0 < q) (s t w : ℤ) (hw : 0 ≤ t-w ∧ t-w < q)
  99     (x y : Ambient model X (s,t))
 100     (a : BiHom (t-s) w (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) q))
 101     (hx : Detects (model.convergence (.quotient q (.nu X))) (s,t,w)
 102       ((model.family.quotientProjection (model.nu.functor.obj (X.obj model.auxiliary)) q).pageMap 2 (s,t,w)
 103         (nuWeightLabel model X s t w (by omega) x)) a)
 104     (hy : Detects (model.convergence (.quotient q (.nu X))) (s,t,w)
 105       ((model.family.quotientProjection (model.nu.functor.obj (X.obj model.auxiliary)) q).pageMap 2 (s,t,w)
 106         (nuWeightLabel model X s t w (by omega) y)) a) :
 107     x-y ∈ boundaries model X (1+t-w) (s,t) := by
 108   obtain ⟨ex,hex,ax,hax,hgx⟩ := hx
 109   obtain ⟨ey,hey,ay,hay,hgy⟩ := hy
 110   have hab : ax=ay := (ModuleCat.mono_iff_injective
 111     ((model.syntheticTower.filtration (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) q)).F s (t-s,w)).arrow).mp
 112       inferInstance (hax.trans hay.symm)
 113   have heq : ex=ey := ((model.convergence (.quotient q (.nu X))).identification (s,t,w)).toLinearEquiv.injective (by
 114     change ((model.convergence (.quotient q (.nu X))).identification (s,t,w)).hom ex =
 115       ((model.convergence (.quotient q (.nu X))).identification (s,t,w)).hom ey
 116     rw [hgx,hgy,hab])
 117   obtain ⟨hxc,hxl⟩ := (KIP126.Literature.Standard.eInfinity_finite_labels X q hq s t w hw x ex).mp hex
 118   obtain ⟨hyc,hyl⟩ := (KIP126.Literature.Standard.eInfinity_finite_labels X q hq s t w hw y ey).mp hey
 119   have hequal : subquotientClass (cycles model X (q-t+w) (s,t)) (boundaries model X (1+t-w) (s,t)) ⟨x,hxc⟩ =
 120       subquotientClass (cycles model X (q-t+w) (s,t)) (boundaries model X (1+t-w) (s,t)) ⟨y,hyc⟩ := by
 121     rw [←hxl,←hyl,heq]
 122   have hzero : subquotientClass (cycles model X (q-t+w) (s,t)) (boundaries model X (1+t-w) (s,t))
 123       (⟨x,hxc⟩-⟨y,hyc⟩) = 0 := by rw [map_sub,hequal,sub_self]
 124   exact (CrossingLabels.subquotient_zero_iff _ _ _).mp hzero

 128     (x x0 : Ambient model X (s,t))

 131       firstLabel model X s t x0 ≫ XModLambdaN.map (SyntheticCategory.biShift_zero.hom.app _) 1)

 141       targetNuLabel model X s t d x0 := by
 142     have hd' : (t-(t-d)).toNat = d := by omega

 144         (h : (s,t,t-a)=(s,t,t-b)) :
 145         regradeE2 h (targetNuLabel model X s t a x0) = targetNuLabel model X s t b x0 := by
 146       subst b
 147       rfl
 148     exact transport _ d hd' _
 149   have hbdet' : Detects (model.convergence (.quotient q (.nu X))) (s,t,t-d)
 150       ((model.family.quotientProjection (model.nu.functor.obj (X.obj model.auxiliary)) q).pageMap 2 (s,t,t-d)

 153     exact hbdet
 154   have h := finite_detected_labels_congruent X q (by omega) s t (t-d)
 155     (by omega) x x0 a ha hbdet'
 156   convert h using 1
 157   congr 1
 158   omega

 194 private theorem differential_of_source_boundary_congruent (X : ClassicalObject)
 195     (r b : ℤ) (p q : Bidegree)
 196     (x x0 : Ambient model X p) (y : Ambient model X q)
 197     (hd : HasDifferential (model.classical.obj (X.obj model.auxiliary)) r p q x0 y)
 198     (hx : x-x0 ∈ boundaries model X b p) :
 199     HasDifferential (model.classical.obj (X.obj model.auxiliary)) r p q x y := by
 200   let E := model.classical.obj (X.obj model.auxiliary)
 201   obtain ⟨hdeg,xr,yr,hxr,hyr,hdr⟩ := hd
 202   have hr : 2 ≤ r := hxr.1
 203   have hnext : ReachesPage E (r+1) p (x-x0) :=
 204     (isCycle_iff_reaches model X (r+1) (by omega) p (x-x0)).mp
 205       ⟨by omega, by
 206         simpa only [add_sub_cancel_right] using boundaries_le_cycles model X p b r hx⟩
 207   obtain ⟨zr,hzr⟩ := reaches_earlier hr (by omega : r≤r+1) hnext
 208   have hdz : E.d r p zr = 0 := represents_d_zero_of_later
 209     (by change (model.classical.obj (X.obj model.auxiliary)).r₀ ≤ r
 210         rw [model.classical.firstPage]
 211         exact hr) (by omega : r<r+1) hzr hnext
 212   refine ⟨hdeg,xr+zr,yr,?_,hyr,?_⟩
 213   · have hh := middle_add hxr hzr
 214     have he : x0+(x-x0)=x := by abel
 215     simpa only [he] using hh
 216   · rw [map_add,hdr]
 217     change yr + (eqToHom (congrArg (E.Page r) hdeg)) (E.d r p zr) = yr
 218     rw [hdz,map_zero,add_zero]

 315     (q d : ℕ) (hdq : d < q) (s t : ℤ)
 316     (x : Ambient model X (s,t))
 317     (a : BiHom (t-s) (t-d) (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) q))
 318     (ha : Detects (model.convergence (.quotient q (.nu X))) (s,t,t-d)
 319       ((model.family.quotientProjection (model.nu.functor.obj (X.obj model.auxiliary)) q).pageMap 2 (s,t,t-d)
 320         (nuWeightLabel model X s t (t-d) (by omega) x)) a) :
 321     ∃ b : BiHom (t-s) t (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) (q-d)),
 322       finiteLambdaDivisionClass X q d hdq s t b = a := by
 323   by_cases hd : d=0
 324   · subst d
 325     refine ⟨homotopyRegrade rfl (by simp) a,?_⟩
 326     simp only [finiteLambdaDivisionClass, ↓reduceDIte]
 327     have cancel (A : SyntheticSpectrum) (m w w' : ℤ) (hw : w=w') (hw' : w'=w)
 328         (c : BiHom m w A) :
 329         homotopyRegrade rfl hw' (homotopyRegrade rfl hw c)=c := by subst w'; rfl
 330     exact cancel _ _ _ _ _ _ a
 331   · have hd' : 0<d := by omega
 332     obtain ⟨b,hb⟩ := finite_detected_source_division X q d hd' hdq s t x a ha
 333     refine ⟨b,?_⟩
 334     simp only [finiteLambdaDivisionClass,dif_neg hd]
 335     exact (finiteLambdaClass_vertical X q d hd' hdq s t b).trans hb

 338     (q d k : ℕ) (hdq : d < q) (hqdk : q-d ≤ k) (s t : ℤ)
 339     (x : Ambient model X (s,t)) (y : Ambient model X (s+k+1,t+k))

 347       (nuWeightLabel model X (s+k+1) (t+k) (t+(q-d:ℕ)) (by omega) y)

 349         (biSuspensionHomEquiv (t-s) t (q-d:ℕ) (model.nu.functor.obj (X.obj model.auxiliary))

 353   by_cases hd : d=0
 354   · subst d

 356     have hp : (s,t,t-(0:ℕ))=(s,t,t) := by simp

 365             ((model.family.quotientProjection (model.nu.functor.obj (X.obj model.auxiliary)) q).pageMap 2 (s,t,w')

 372       exact (cancel _ _ _ _ _ _ b).symm
