  26 private theorem detects_of_shift (A : SyntheticObject) (δ p : ℤ × ℤ) (s : ℤ)
  27     (x : (model.family.obj (A.obj model.nu model.auxiliary)).E₂ (s,p.1+s,p.2))
  28     (a : BiHom p.1 p.2 (A.obj model.nu model.auxiliary))
  29     (ha : Detects (model.convergence (.shift δ A)) (s,(p+δ).1+s,(p+δ).2)
  30       (((essShift δ).page A 2 (by decide) p s).hom x)
  31       (homotopyRegrade (by dsimp; omega) rfl (shiftHomotopyClass δ p a))) :
  32     Detects (model.convergence A) (s,p.1+s,p.2) x
  33       (homotopyRegrade (by dsimp; omega) rfl a) := by
  34   obtain ⟨eS,heS,b,hb,hgb⟩ := frame_of_source_detects
  35     (model.convergence (.shift δ A)) (p+δ) s _ (shiftHomotopyClass δ p a) ha
  36   let S := (essShift δ).filtered A p
  37   let I := towerInfinityIso (model.convergence A) (model.convergence (.shift δ A)) S s
  38   let e := I.inv eS
  39   let ar := (S.level s).inv b
  40   have har : ((model.syntheticTower.filtration (A.obj model.nu model.auxiliary)).F s p).arrow ar = a := by
  41     apply S.ambient.toLinearEquiv.injective
  42     have h := ConcreteCategory.congr_hom (S.level_arrow s) ar
  43     have hlevel : (S.level s).hom ar=b := (S.level s).toLinearEquiv.apply_symm_apply b
  44     simp only [ModuleCat.comp_apply] at h
  45     rw [hlevel] at h
  46     exact h.symm.trans (hb.trans ((essShift δ).ambient_eq A p a).symm)
  47   have hgar : (model.syntheticTower.filtration (A.obj model.nu model.auxiliary)).toAssociatedGraded s p ar =
  48       (infinityGradedIso (model.convergence A) p s).hom e := by
  49     have h := ConcreteCategory.congr_hom (S.projection_inv_comm s) b
  50     simp only [ModuleCat.comp_apply] at h
  51     rw [hgb] at h
  52     rw [←h]
  53     change (S.gradedIso s).inv
  54         ((towerGradedIso (model.convergence (.shift δ A)) (p+δ) s).hom eS) =
  55       (towerGradedIso (model.convergence A) p s).hom
  56         ((towerInfinityIso (model.convergence A) (model.convergence (.shift δ A)) S s).inv eS)
  57     simp only [towerInfinityIso,Iso.trans_inv,Iso.symm_inv,←ModuleCat.comp_apply,
  58       Category.assoc,Iso.inv_hom_id,Category.comp_id]
  59   have he : HasInfinityRepresentative _ 2 (s,p.1+s,p.2) x e := by
  60     apply ((essShift δ).representative_iff A 2 (by decide) p s x e).mpr
  61     have hI : I.hom e=eS := I.toLinearEquiv.apply_symm_apply eS
  62     change HasInfinityRepresentative _ 2 _ _ (I.hom e)
  63     rw [hI]
  64     exact heS
  65   have hdet := source_frame_detects (model.convergence A) p s e x he ar hgar
  66   rw [har] at hdet
  67   exact hdet

  69 private theorem absolute_boundary_detects_unshifted (B : SyntheticObject)
  70     (q : ℕ) (m w v : ℤ)
  71     (a : BiHom m w ((SyntheticObject.quotient q B).obj model.nu model.auxiliary))
  72     (x : (model.family.obj (B.obj model.nu model.auxiliary)).E₂ (v,m-1+v,w+q))
  73     (hdet : Detects (model.convergence (.shift (1,-(q:ℤ)) B)) (v,m+v,w)
  74       (regradeE2 (by ext <;> dsimp <;> omega)
  75         (((essShift (1,-(q:ℤ))).page B 2 (by decide) (m-1,w+q) v).hom x))
  76       (homotopyRegrade (by dsimp; omega) rfl (a ≫ absoluteBoundary (B.obj model.nu model.auxiliary) q))) :
  77     Detects (model.convergence B) (v,m-1+v,w+q) x
  78       (homotopyRegrade (by dsimp; omega) rfl
  79         (biSuspensionHomEquiv m w q (B.obj model.nu model.auxiliary)
  80           (a ≫ XModLambdaN.proj _ q))) := by
  81   let β := biSuspensionHomEquiv m w q (B.obj model.nu model.auxiliary)
  82     (a ≫ XModLambdaN.proj _ q)
  83   have he : a ≫ absoluteBoundary (B.obj model.nu model.auxiliary) q =
  84       homotopyRegrade (sub_add_cancel m 1) (add_neg_cancel_right w (q:ℤ))
  85         (shiftHomotopyClass (1,-(q:ℤ)) (m-1,w+q) β) := by
  86     have h := congrArg (fun z => z ≫ (biSuspensionIso q (B.obj model.nu model.auxiliary)).inv)
  87       (bi_susp_symm_eq m w q (B.obj model.nu model.auxiliary) β)
  88     simpa [β,absoluteBoundary,shiftHomotopyClass,susp_invariance,Iso.homCongr] using h
  89   have hp : (v,((m-1,w+q)+(1,-(q:ℤ))).1+v,((m-1,w+q)+(1,-(q:ℤ))).2) =
  90       (v,m+v,w) := by ext <;> dsimp <;> omega
  91   have H := detects_transport hp.symm _ _ hdet
  92   apply detects_of_shift B (1,-(q:ℤ)) (m-1,w+q) v x β
  93   convert H using 1
  94   · have cancel {i j : Tridegree} (h : i=j)
  95         (z : (model.family.obj ((SyntheticObject.shift (1,-(q:ℤ)) B).obj model.nu model.auxiliary)).E₂ i) :
  96         regradeE2 h.symm (regradeE2 h z)=z := by subst j; rfl
  97     exact (cancel hp _).symm
  98   · rw [he]
  99     simp only [regrade_precomp,←Category.assoc,eqToHom_trans]

 101 private theorem absolute_boundary_square_detects (A B : SyntheticObject)
 102     (g : A.obj model.nu model.auxiliary ⟶ B.obj model.nu model.auxiliary)
 103     (q : ℕ) (p : ℤ × ℤ) (s m r l : ℤ)
 104     (x : ((model.family.obj ((SyntheticObject.quotient q A).obj model.nu model.auxiliary)).sequence.ssData
 105       (s,p.1+s,p.2)).eInfty)
 106     (y : ((model.family.obj ((SyntheticObject.quotient q B).obj model.nu model.auxiliary)).sequence.ssData
 107       (s+m,p.1+(s+m),p.2)).eInfty)
 108     (z : ((model.family.obj ((SyntheticObject.shift (1,-(q:ℤ)) A).obj model.nu model.auxiliary)).sequence.ssData
 109       (s+r,p.1+(s+r),p.2)).eInfty)
 110     (w : ((model.family.obj ((SyntheticObject.shift (1,-(q:ℤ)) B).obj model.nu model.auxiliary)).sequence.ssData
 111       (s+r+l,p.1+(s+r+l),p.2)).eInfty)
 112     (hleft : TowerExtensionEquation
 113       (model.convergentMap (.quotient q A) (.shift (1,-(q:ℤ)) A)
 114         (absoluteBoundary (A.obj model.nu model.auxiliary) q)) p s r x z)
 115     (htop : TowerExtensionEquation
 116       (model.convergentMap (.quotient q A) (.quotient q B) (XModLambdaN.map g q)) p s m x y)
 117     (hbottom : TowerExtensionEquation
 118       (model.convergentMap (.shift (1,-(q:ℤ)) A) (.shift (1,-(q:ℤ)) B)
 119         ((SyntheticCategory.biShift (1,-(q:ℤ))).map g)) p (s+r) l z w)
 120     (hsourceCrossing :
 121       TowerNoCrossing (model.convergentMap (.quotient q A) (.shift (1,-(q:ℤ)) A)
 122         (absoluteBoundary (A.obj model.nu model.auxiliary) q)) p s r ∨
 123       TowerNoCrossing (model.convergentMap (.quotient q A) (.quotient q B)
 124         (XModLambdaN.map g q)) p s m)
 125     (hbottomCrossing : TowerNoCrossing
 126       (model.convergentMap (.shift (1,-(q:ℤ)) A) (.shift (1,-(q:ℤ)) B)
 127         ((SyntheticCategory.biShift (1,-(q:ℤ))).map g)) p (s+r) l)
 128     (y0 : (model.family.obj ((SyntheticObject.quotient q B).obj model.nu model.auxiliary)).E₂
 129       (s+m,p.1+(s+m),p.2))
 130     (hy : HasInfinityRepresentative _ 2 (s+m,p.1+(s+m),p.2) y0 y)
 131     (w0 : (model.family.obj ((SyntheticObject.shift (1,-(q:ℤ)) B).obj model.nu model.auxiliary)).E₂
 132       (s+r+l,p.1+(s+r+l),p.2))
 133     (hw : HasInfinityRepresentative _ 2 (s+r+l,p.1+(s+r+l),p.2) w0 w) :
 134     ∃ aRight : BiHom p.1 p.2 ((SyntheticObject.quotient q B).obj model.nu model.auxiliary),
 135       Detects (model.convergence (.quotient q B)) (s+m,p.1+(s+m),p.2) y0
 136         (homotopyRegrade (by dsimp; omega) rfl aRight) ∧
 137       Detects (model.convergence (.shift (1,-(q:ℤ)) B)) (s+r+l,p.1+(s+r+l),p.2) w0
 138         (homotopyRegrade (by dsimp; omega) rfl
 139           (aRight ≫ absoluteBoundary (B.obj model.nu model.auxiliary) q)) := by
 140   obtain ⟨ar,br,cr,dr,har,hbr,hcr,hdr,hab,hac,hcd,hbd⟩ := square_frames
 141     (model.convergentMap (.quotient q A) (.quotient q B) (XModLambdaN.map g q))
 142     (model.convergentMap (.quotient q A) (.shift (1,-(q:ℤ)) A)
 143       (absoluteBoundary (A.obj model.nu model.auxiliary) q))
 144     (model.convergentMap (.shift (1,-(q:ℤ)) A) (.shift (1,-(q:ℤ)) B)
 145       ((SyntheticCategory.biShift (1,-(q:ℤ))).map g))
 146     (absoluteBoundary_natural g q) p s m r l x y z w htop hleft hbottom
 147     (hsourceCrossing.elim Or.inr Or.inl) hbottomCrossing
 148   have hright := source_frame_detects (model.convergence (.quotient q B)) p (s+m) y y0 hy br hbr
 149   have hrightBoundary := source_frame_detects (model.convergence (.shift (1,-(q:ℤ)) B))
 150     p (s+r+l) w w0 hw dr hdr
 151   rw [←hbd] at hrightBoundary
 152   exact ⟨_,hright,hrightBoundary⟩
