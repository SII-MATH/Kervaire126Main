  23 private theorem first_label_lift_permanent (X : ClassicalObject) (s t : ℤ)
  24     (y : Ambient model X (s,t))
  25     (α : BiHom (t-s) t (model.nu.functor.obj (X.obj model.auxiliary)))
  26     (hα : α ≫ XModLambdaN.incl _ 1 = firstLabel model X s t y ≫
  27       XModLambdaN.map (SyntheticCategory.biShift_zero.hom.app _) 1) :
  28     y ∈ permanentCycles model X (s,t) := by
  29   let A := model.nu.functor.obj (X.obj model.auxiliary)
  30   let inv : A ⟶ nuZero model X := SyntheticCategory.biShift_zero.inv.app A
  31   have hfirst : quotientClass 1 (α ≫ inv) = firstLabel model X s t y := by
  32     dsimp only [quotientClass]
  33     rw [Category.assoc,XModLambdaN.incl_naturality,←Category.assoc,hα,
  34       Category.assoc,←model.quotients.functoriality.map_comp]
  35     simp only [inv,A,Iso.hom_inv_id_app,model.quotients.functoriality.map_id,Category.comp_id]
  36   obtain ⟨z,hz⟩ := (KIP126.Literature.Standard.selected_permanent_lift X s t y).mpr
  37     ⟨α ≫ inv,hfirst⟩
  38   refine ⟨z,?_⟩
  39   have h := normalizeInitialProjection ((model.classical.obj (X.obj model.auxiliary)).ssData (s,t))
  40     (↑(2-(model.classical.obj (X.obj model.auxiliary)).r₀).toNat)
  41     ⊤ (by rw [model.classical.firstPage]; rfl) le_top
  42   exact (congrArg (fun f => f z) h).symm.trans hz

  44 private theorem detected_zero_infinity (A : SyntheticObject) (p : Tridegree)
  45     (x : (model.family.obj (A.obj model.nu model.auxiliary)).E₂ p)
  46     (h : Detects (model.convergence A) p x 0) :
  47     HasInfinityRepresentative _ 2 p x 0 := by

  51       inferInstance (ha.trans (map_zero _).symm)

  53     change ((model.convergence A).identification p).hom e =
  54       ((model.convergence A).identification p).hom 0
  55     rw [hga,haz,map_zero,map_zero])
  56   rwa [hez] at he

  58 private theorem first_label_lambda_zero_boundary (X : ClassicalObject) (s t : ℤ)
  59     (k : ℕ) (hk : 0 < k) (y : Ambient model X (s,t))
  60     (α : BiHom (t-s) t (model.nu.functor.obj (X.obj model.auxiliary)))
  61     (hα : α ≫ XModLambdaN.incl _ 1 = firstLabel model X s t y ≫
  62       XModLambdaN.map (SyntheticCategory.biShift_zero.hom.app _) 1)
  63     (hz : lambdaMultiply k α = 0) :
  64     y ∈ boundaries model X (k+1) (s,t) := by
  65   have hy := first_label_lift_permanent X s t y α hα
  66   have hd := nu_lambda_detects X k hk s t y hy α hα
  67   rw [hz] at hd
  68   have he := detected_zero_infinity (.nu X) (s,t,t-k) _ hd
  69   have hclass := (KIP126.Literature.Standard.eInfinity_nu_labels X s t (t-k) (by omega) y 0).mp he
  70   have hb := (CrossingLabels.subquotient_zero_iff _ _ ⟨y,hclass.1⟩).mp (by
  71     rw [←hclass.2,map_zero])
  72   convert hb using 1
  73   congr 1
  74   omega

  76 private theorem firstLabel_sub (X : ClassicalObject) (s t : ℤ)
  77     (x y : Ambient model X (s,t)) :
  78     firstLabel model X s t (x-y) = firstLabel model X s t x - firstLabel model X s t y := by
  79   have cast_sub (m w w' : ℤ) (A : SyntheticSpectrum) (hw : w = w')
  80       (he : BiHom m w A = BiHom m w' A) (a b : BiHom m w A) :
  81       he.mp (a-b) = he.mp a - he.mp b := by
  82     subst w'; rfl
  83   simp only [firstLabel,map_sub]
  84   exact cast_sub _ _ _ _ (Int.add_zero t) _ _ _

  86 private theorem first_label_zero (X : ClassicalObject) (s t : ℤ)

  89       (SyntheticCategory.biShift_zero.hom.app _) 1 = 0) : y = 0 := by

  91       XModLambdaN.incl _ 1 = firstLabel model X s t y ≫ XModLambdaN.map
  92       (SyntheticCategory.biShift_zero.hom.app _) 1 := by rw [zero_comp,hy]

  95   have he := detected_zero_infinity (.nu X) (s,t,t) _ hd

 113 private theorem first_restriction_kernel_boundary_lambda_zero (X : ClassicalObject)
 114     (k : ℕ) (hk : 0 < k) (s t : ℤ)
 115     (a : BiHom (t-s) t (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) k))
 116     (ha : a ≫ (model.quotients.tower _).rho 1 k hk = 0) :
 117     lambdaMultiply (k-1) (boundaryRepresentative X k s t a) = 0 := by

 121       (biSuspensionHomEquiv (t-s) t k (model.nu.functor.obj (X.obj model.auxiliary))

 128 private theorem adjacent_boundary_labels_congruent (X : ClassicalObject)
 129     (k : ℕ) (hk : 0 < k) (s t : ℤ)
 130     (a b : BiHom (t-s) t (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) k))
 131     (hab : a ≫ (model.quotients.tower _).rho 1 k hk =
 132       b ≫ (model.quotients.tower _).rho 1 k hk)
 133     (y z : Ambient model X (s+k+1,t+k))
 134     (hy : boundaryRepresentative X k s t a ≫ XModLambdaN.incl _ 1 =
 135       firstLabel model X (s+k+1) (t+k) y ≫ XModLambdaN.map (SyntheticCategory.biShift_zero.hom.app _) 1)
 136     (hz : boundaryRepresentative X k s t b ≫ XModLambdaN.incl _ 1 =
 137       firstLabel model X (s+k+1) (t+k) z ≫ XModLambdaN.map (SyntheticCategory.biShift_zero.hom.app _) 1) :
 138     y-z ∈ boundaries model X k (s+k+1,t+k) := by
 139   have hfirst : boundaryRepresentative X k s t (a-b) ≫ XModLambdaN.incl _ 1 =
 140       firstLabel model X (s+k+1) (t+k) (y-z) ≫
 141         XModLambdaN.map (SyntheticCategory.biShift_zero.hom.app _) 1 := by
 142     rw [boundaryRepresentative_sub,Preadditive.sub_comp,hy,hz,firstLabel_sub,Preadditive.sub_comp]
 143   by_cases hk1 : k=1
 144   · subst k
 145     have hab' : a=b := by simpa only [(model.quotients.tower _).rho_id,Category.comp_id] using hab
 146     have hzero : y-z=0 := first_label_zero X (s+(1:ℕ)+1) (t+(1:ℕ)) (y-z) (by
 147       rw [←hfirst,hab',sub_self]
 148       simp only [boundaryRepresentative,zero_comp,map_zero,regrade_precomp,comp_zero])
 149     rw [hzero]
 150     exact Submodule.zero_mem _
 151   have hmul := first_restriction_kernel_boundary_lambda_zero X k hk s t (a-b) (by
 152     rw [Preadditive.sub_comp,hab,sub_self])
 153   have hb := first_label_lambda_zero_boundary X (s+k+1) (t+k) (k-1) (by omega) (y-z)
 154     (boundaryRepresentative X k s t (a-b)) hfirst hmul
 155   simpa only [Nat.cast_sub hk,Nat.cast_one,sub_add_cancel] using hb

 157 private theorem adjacent_boundary_differential (X : ClassicalObject)
 158     (k : ℕ) (hk : 0 < k) (s t : ℤ)
 159     (x : Ambient model X (s,t)) (y : Ambient model X (s+k+1,t+k))
 160     (a : BiHom (t-s) t (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) k))
 161     (ha : a ≫ (model.quotients.tower _).rho 1 k hk =
 162       firstLabel model X s t x ≫ XModLambdaN.map (SyntheticCategory.biShift_zero.hom.app _) 1)
 163     (hy : boundaryRepresentative X k s t a ≫ XModLambdaN.incl _ 1 =
 164       firstLabel model X (s+k+1) (t+k) y ≫ XModLambdaN.map (SyntheticCategory.biShift_zero.hom.app _) 1) :
 165     HasDifferential (model.classical.obj (X.obj model.auxiliary))
 166       (k+1) (s,t) (s+k+1,t+k) x y := by
 167   have hx := first_label_lift_cycles X k hk s t x a ha
 168   have hreach := (isCycle_iff_reaches model X (k+1) (by omega) (s,t) x).mp
 169     ⟨by omega,by simpa only [add_sub_cancel_right] using hx⟩
 170   obtain ⟨b,hb,hdb⟩ := KIP126.Literature.Standard.bhs_bockstein X k hk s t x hreach
 171   let z := bocksteinLabel model X k hk s t b
 172   obtain ⟨a0,ha0,hz⟩ := adjacent_absolute_boundary X k hk s t x z hdb
 173   have hcong := adjacent_boundary_labels_congruent X k hk s t a a0 (ha.trans ha0.symm) y z hy hz
 174   exact (differentialAt_of_congruent model X
 175     (⟨by omega,hdb⟩ : DifferentialAt model X (k+1) (s,t) (s+k+1,t+k) x z)
 176     ⟨by omega,by simpa only [add_sub_cancel_right] using hcong⟩).2

 201 private theorem nu_detected_labels_congruent (X : ClassicalObject) (s t w : ℤ)
 202     (hw : w ≤ t) (y z : Ambient model X (s,t))
 203     (α : BiHom (t-s) w (model.nu.functor.obj (X.obj model.auxiliary)))
 204     (hy : Detects (model.convergence (.nu X)) (s,t,w) (nuWeightLabel model X s t w hw y) α)
 205     (hz : Detects (model.convergence (.nu X)) (s,t,w) (nuWeightLabel model X s t w hw z) α) :
 206     y-z ∈ boundaries model X (1+t-w) (s,t) := by
 207   obtain ⟨ey,hey,a,ha,hga⟩ := hy
 208   obtain ⟨ez,hez,b,hb,hgb⟩ := hz
 209   have hab : a = b := (ModuleCat.mono_iff_injective
 210     ((model.syntheticTower.filtration (model.nu.functor.obj (X.obj model.auxiliary))).F s (t-s,w)).arrow).mp
 211       inferInstance (ha.trans hb.symm)
 212   have heq : ey = ez := ((model.convergence (.nu X)).identification (s,t,w)).toLinearEquiv.injective (by
 213     change ((model.convergence (.nu X)).identification (s,t,w)).hom ey =
 214       ((model.convergence (.nu X)).identification (s,t,w)).hom ez
 215     rw [hga,hgb,hab])
 216   obtain ⟨hyp,hycl⟩ := (KIP126.Literature.Standard.eInfinity_nu_labels X s t w hw y ey).mp hey
 217   obtain ⟨hzp,hzcl⟩ := (KIP126.Literature.Standard.eInfinity_nu_labels X s t w hw z ez).mp hez
 218   have hequal : subquotientClass (permanentCycles model X (s,t)) (boundaries model X (1+t-w) (s,t)) ⟨y,hyp⟩ =
 219       subquotientClass (permanentCycles model X (s,t)) (boundaries model X (1+t-w) (s,t)) ⟨z,hzp⟩ := by
 220     rw [←hycl,←hzcl,heq]
 221   have hzero : subquotientClass (permanentCycles model X (s,t)) (boundaries model X (1+t-w) (s,t))
 222       (⟨y,hyp⟩ - ⟨z,hzp⟩) = 0 := by rw [map_sub,hequal,sub_self]
 223   exact (CrossingLabels.subquotient_zero_iff _ _ _).mp hzero

 234 private theorem boundary_detects_differential (X : ClassicalObject)
 235     (i k : ℕ) (hi : 0 < i) (hik : i ≤ k) (s t : ℤ)
 236     (x : Ambient model X (s,t)) (y : Ambient model X (s+k+1,t+k))
 237     (a : BiHom (t-s) t (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) i))
 238     (ha : a ≫ (model.quotients.tower _).rho 1 i hi =
 239       firstLabel model X s t x ≫ XModLambdaN.map (SyntheticCategory.biShift_zero.hom.app _) 1)
 240     (hy : Detects (model.convergence (.nu X)) (s+k+1,t+k,t+i)
 241       (nuWeightLabel model X (s+k+1) (t+k) (t+i) (by omega) y)
 242       (homotopyRegrade (by dsimp; omega) rfl
 243         (biSuspensionHomEquiv (t-s) t i (model.nu.functor.obj (X.obj model.auxiliary))
 244           (a ≫ XModLambdaN.proj _ i)))) :
 245     HasDifferential (model.classical.obj (X.obj model.auxiliary))
 246       (k+1) (s,t) (s+k+1,t+k) x y := by
 247   let A := model.nu.functor.obj (X.obj model.auxiliary)
 248   have hfil : FiltrationAtLeast model.syntheticTower (s+k+1)
 249       (biSuspensionHomEquiv (t-s) t i A (a ≫ XModLambdaN.proj A i)) := by
 250     have hf := detects_filtration (.nu X) (s+k+1,t+k,t+i) _ _ hy
 251     have transport (m m' w s : ℤ) (hm : m=m') (a : BiHom m w A) :
 252         FiltrationAtLeast model.syntheticTower s (homotopyRegrade hm rfl a) ↔
 253           FiltrationAtLeast model.syntheticTower s a := by subst m'; rfl
 254     exact (transport _ _ _ _ _ _).mp hf
 255   obtain ⟨b,hb⟩ : ∃ b : BiHom (t-s) t (XModLambdaN A k),
 256       b ≫ (model.quotients.tower A).rho i k hik = a := by
 257     rcases hik.eq_or_lt with rfl | hlt
 258     · exact ⟨a,by rw [(model.quotients.tower A).rho_id,Category.comp_id]⟩
 259     · exact absolute_boundary_filtration_lift X i k hi hlt s t a hfil
 260   have hbfirst : b ≫ (model.quotients.tower A).rho 1 k (by omega) =
 261       firstLabel model X s t x ≫ XModLambdaN.map (SyntheticCategory.biShift_zero.hom.app A) 1 := by
 262     rw [←(model.quotients.tower A).rho_comp hi hik,←Category.assoc,hb]
 263     exact ha
 264   let β := boundaryRepresentative X k s t b
 265   obtain ⟨z,hz⟩ := nu_first_label_exists X (s+k+1) (t+k) β
 266   have hp := first_label_lift_permanent X (s+k+1) (t+k) z β hz
 267   have hdz := adjacent_boundary_differential X k (by omega) s t x z b hbfirst hz
 268   have hzdet : Detects (model.convergence (.nu X)) (s+k+1,t+k,t+i)
 269       (nuWeightLabel model X (s+k+1) (t+k) (t+i) (by omega) z)
 270       (homotopyRegrade (by dsimp; omega) rfl
 271         (biSuspensionHomEquiv (t-s) t i A (a ≫ XModLambdaN.proj A i))) := by
 272     rw [←hb]
 273     by_cases hik' : i=k
 274     · subst i
 275       simpa only [β,boundaryRepresentative,A,(model.quotients.tower A).rho_id,Category.comp_id] using
 276         nu_first_label_detects X (s+k+1) (t+k) z hp β hz
 277     have hlam := nu_lambda_detects X (k-i) (by omega) (s+k+1) (t+k) z hp β hz
 278     have hind : (s+k+1,t+k,(t+k)-(k-i:ℕ))=(s+k+1,t+k,t+i) :=
 279       Prod.ext rfl (Prod.ext rfl (by dsimp; omega))
 280     have hlam := detects_transport hind _ _ hlam
 281     convert hlam using 1
 282     · have transport (w w' : ℤ) (hw : w=w') (hw₁ : w≤t+k) (hw₂ : w'≤t+k)
 283           (h : (s+k+1,t+k,w)=(s+k+1,t+k,w')) :
 284           regradeE2 h (nuWeightLabel model X (s+k+1) (t+k) w hw₁ z) =
 285             nuWeightLabel model X (s+k+1) (t+k) w' hw₂ z := by subst w'; rfl
 286       exact (transport _ _ (by omega) _ _ hind).symm
 287     · rw [absolute_boundary_restriction]
 288       dsimp only [β,boundaryRepresentative]
 289       rw [lambda_regrade]
 290       simp only [regrade_precomp,←Category.assoc,eqToHom_trans]
 291   have hcong := nu_detected_labels_congruent X (s+k+1) (t+k) (t+i) (by omega) y z _ hy hzdet
 292   have hboundary : y-z ∈ boundaries model X k (s+k+1,t+k) :=
 293     boundaries_monotone model X (s+k+1,t+k) (by omega) hcong
 294   exact (differentialAt_of_congruent model X
 295     (⟨by omega,hdz⟩ : DifferentialAt model X (k+1) (s,t) (s+k+1,t+k) x z)
 296     ⟨by omega,by simpa only [add_sub_cancel_right] using hboundary⟩).2
