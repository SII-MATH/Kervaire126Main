  24 private abbrev lambdaBoundaryPoint (k q : ℕ) (p : ℤ × ℤ) : ℤ × ℤ :=

  27 private def lambdaBoundaryInfinity (A : SyntheticObject) (k q : ℕ)
  28     (p : ℤ × ℤ) (s : ℤ)
  29     (x : ((model.family.obj (A.obj model.nu model.auxiliary)).sequence.ssData
  30       (s,p.1+s,p.2)).eInfty) :
  31     ((model.family.obj ((SyntheticObject.shift (1,-(q:ℤ)) A).obj
  32       model.nu model.auxiliary)).sequence.ssData
  33         (s,(lambdaBoundaryPoint k q p).1+s,(lambdaBoundaryPoint k q p).2)).eInfty := by
  34   let p' : ℤ × ℤ := p+(0,-(k:ℤ))
  35   let I : ((model.family.obj (A.obj model.nu model.auxiliary)).sequence.ssData
  36       (s,p.1+s,p.2)).eInfty ≅
  37       ((model.family.obj ((SyntheticObject.shift (0,-(k:ℤ)) A).obj model.nu model.auxiliary)).sequence.ssData
  38       (s,p'.1+s,p'.2)).eInfty :=
  39     towerInfinityIso (model.convergence A) (model.convergence (.shift (0,-(k:ℤ)) A))
  40       ((essShift (0,-(k:ℤ))).filtered A p) s
  41   let z : ((model.family.obj ((SyntheticObject.shift (0,-(k:ℤ)) A).obj model.nu model.auxiliary)).sequence.ssData
  42       (s,p'.1+s,p'.2)).eInfty := I.hom x
  43   let L : ((model.family.obj ((SyntheticObject.shift (0,-(k:ℤ)) A).obj model.nu model.auxiliary)).sequence.ssData
  44       (s,p'.1+s,p'.2)).eInfty ⟶
  45       ((model.family.obj (A.obj model.nu model.auxiliary)).sequence.ssData
  46       (s,p'.1+s,p'.2)).eInfty :=
  47     (model.family.map (lambdaPow k (A.obj model.nu model.auxiliary))).eInftyMap
  48       (s,p'.1+s,p'.2)
  49   let y : ((model.family.obj (A.obj model.nu model.auxiliary)).sequence.ssData
  50       (s,p'.1+s,p'.2)).eInfty := L z
  51   let J : ((model.family.obj (A.obj model.nu model.auxiliary)).sequence.ssData
  52       (s,p'.1+s,p'.2)).eInfty ≅
  53       ((model.family.obj ((SyntheticObject.shift (1,-(q:ℤ)) A).obj model.nu model.auxiliary)).sequence.ssData
  54       (s,(p'+(1,-(q:ℤ))).1+s,(p'+(1,-(q:ℤ))).2)).eInfty :=
  55     towerInfinityIso (model.convergence A) (model.convergence (.shift (1,-(q:ℤ)) A))
  56       ((essShift (1,-(q:ℤ))).filtered A p') s
  57   exact J.hom y

  60     (p : ℤ × ℤ) (s : ℤ)
  61     (x : (model.family.obj (A.obj model.nu model.auxiliary)).E₂ (s,p.1+s,p.2)) :
  62     (model.family.obj ((SyntheticObject.shift (1,-(q:ℤ)) A).obj
  63       model.nu model.auxiliary)).E₂
  64         (s,(lambdaBoundaryPoint k q p).1+s,(lambdaBoundaryPoint k q p).2) := by
  65   let p' : ℤ × ℤ := p+(0,-(k:ℤ))
  66   let I : (model.family.obj (A.obj model.nu model.auxiliary)).E₂ (s,p.1+s,p.2) ≅
  67       (model.family.obj ((SyntheticObject.shift (0,-(k:ℤ)) A).obj model.nu model.auxiliary)).E₂
  68         (s,p'.1+s,p'.2) := (essShift (0,-(k:ℤ))).page A 2 (by decide) p s
  69   let z : (model.family.obj ((SyntheticObject.shift (0,-(k:ℤ)) A).obj model.nu model.auxiliary)).E₂
  70       (s,p'.1+s,p'.2) := I.hom x
  71   let L : (model.family.obj ((SyntheticObject.shift (0,-(k:ℤ)) A).obj model.nu model.auxiliary)).E₂
  72       (s,p'.1+s,p'.2) ⟶
  73       (model.family.obj (A.obj model.nu model.auxiliary)).E₂ (s,p'.1+s,p'.2) :=
  74     familyPageMap model.family (lambdaPow k (A.obj model.nu model.auxiliary)) 2 (s,p'.1+s,p'.2)
  75   let y : (model.family.obj (A.obj model.nu model.auxiliary)).E₂ (s,p'.1+s,p'.2) := L z
  76   let J : (model.family.obj (A.obj model.nu model.auxiliary)).E₂ (s,p'.1+s,p'.2) ≅
  77       (model.family.obj ((SyntheticObject.shift (1,-(q:ℤ)) A).obj model.nu model.auxiliary)).E₂
  78         (s,(p'+(1,-(q:ℤ))).1+s,(p'+(1,-(q:ℤ))).2) :=
  79     (essShift (1,-(q:ℤ))).page A 2 (by decide) p' s
  80   exact J.hom y

  83     (p : ℤ × ℤ) (s : ℤ)
  84     (x : (model.family.obj (A.obj model.nu model.auxiliary)).E₂ (s,p.1+s,p.2))

  86       (s,p.1+s,p.2)).eInfty)
  87     (hz : HasInfinityRepresentative (model.family.obj (A.obj model.nu model.auxiliary))
  88       2 (s,p.1+s,p.2) x z) :

  90       (model.family.obj ((SyntheticObject.shift (1,-(q:ℤ)) A).obj model.nu model.auxiliary))

  97     (p+(0,-(k:ℤ))) s _ _).mp hm
  98   exact hf

 100 /-- Push the entire actual extension through lambda^k and the absolute-boundary
 101 shift. This gives precisely the third edge needed by square_frames. -/
 102 private theorem extension_lambda_boundary_shift (A B : SyntheticObject)
 103     (g : A.obj model.nu model.auxiliary ⟶ B.obj model.nu model.auxiliary)
 104     (k q : ℕ) (p : ℤ × ℤ) (s n : ℤ)
 105     (x : ((model.family.obj (A.obj model.nu model.auxiliary)).sequence.ssData
 106       (s,p.1+s,p.2)).eInfty)
 107     (y : ((model.family.obj (B.obj model.nu model.auxiliary)).sequence.ssData
 108       (s+n,p.1+(s+n),p.2)).eInfty)
 109     (he : TowerExtensionEquation (model.convergentMap A B g) p s n x y) :
 110     TowerExtensionEquation
 111       (model.convergentMap (.shift (1,-(q:ℤ)) A) (.shift (1,-(q:ℤ)) B)
 112         ((SyntheticCategory.biShift (1,-(q:ℤ))).map g))
 113       (lambdaBoundaryPoint k q p) s n
 114       (lambdaBoundaryInfinity A k q p s x)
 115       (lambdaBoundaryInfinity B k q p (s+n) y) := by
 116   have hs := ((essShift (0,-(k:ℤ))).towerExtensionEquation_iff A B g p s n x y).mp he
 117   have hl := tower_extension_map
 118     (model.convergentMap (.shift (0,-(k:ℤ)) A) (.shift (0,-(k:ℤ)) B)
 119       ((SyntheticCategory.biShift (0,-(k:ℤ))).map g))
 120     (model.convergentMap A B g)
 121     (model.convergentMap (.shift (0,-(k:ℤ)) A) A
 122       (lambdaPow k (A.obj model.nu model.auxiliary)))
 123     (model.convergentMap (.shift (0,-(k:ℤ)) B) B
 124       (lambdaPow k (B.obj model.nu model.auxiliary)))
 125     (lambdaPow_naturality k g) (p+(0,-(k:ℤ))) s n _ _ hs
 126   exact ((essShift (1,-(q:ℤ))).towerExtensionEquation_iff A B g
 127     (p+(0,-(k:ℤ))) s n _ _).mp hl

 129 /-- Witness-level interface for the GL bottom edge. Both labels are explicitly
 130 the actual E2 shift/lambda images of W's original labels, and both infinity
 131 classes are exactly the corresponding images of W.source and W.target. -/
 132 private theorem infinite_extension_lambda_boundary_shift
 133     (X Y : ClassicalObject) (f : X.obj model.auxiliary ⟶ Y.obj model.auxiliary)
 134     (k q : ℕ) (n s t : ℤ)
 135     (x : Ambient model X (s,t)) (y : Ambient model Y (s+n,t+n))
 136     (W : InfiniteExtensionWitness model X Y f n s t x y) :
 137     let A := extensionSourceObject model X Y f
 138     let B := SyntheticObject.nu Y
 139     let p : ℤ × ℤ := (t-s,t+normalizedExponent model.classicalTower f)
 140     TowerExtensionEquation
 141       (model.convergentMap (.shift (1,-(q:ℤ)) A) (.shift (1,-(q:ℤ)) B)
 142         ((SyntheticCategory.biShift (1,-(q:ℤ))).map (model.normalizedMap X Y f).map))
 143       (lambdaBoundaryPoint k q p) s n
 144       (lambdaBoundaryInfinity A k q p s W.source)
 145       (lambdaBoundaryInfinity B k q p (s+n) W.target) ∧
 146     HasInfinityRepresentative _ 2 _
 147       (lambdaBoundaryLabel A k q p s (infiniteSourceForExtension model X Y f s t x))
 148       (lambdaBoundaryInfinity A k q p s W.source) ∧
 149     HasInfinityRepresentative _ 2 _
 150       (lambdaBoundaryLabel B k q p (s+n)
 151         (infiniteTargetForExtension model X Y f n s t W.length_lower y))
 152       (lambdaBoundaryInfinity B k q p (s+n) W.target) := by
 153   refine ⟨extension_lambda_boundary_shift _ _ _ k q _ s n _ _ W.equation, ?_, ?_⟩
 154   · exact lambdaBoundary_representative _ k q _ s _ _ W.source_rep
 155   · exact lambdaBoundary_representative _ k q _ (s+n) _ _ W.target_rep

 157 /-- The transformed bottom stem/weight is exactly the GL square's stem/weight. -/
 158 private theorem gl_bottom_point (page : ℕ) (r s t e : ℤ)
 159     (hpage : 2 ≤ page) (hpr : (page:ℤ) ≤ r) :
 160     lambdaBoundaryPoint (r-page).toNat (page-1)
 161       ((t+r-1)-(s+r),(t+r-1)+e) = (t-s,t+e) := by
 162   have hk : ((r-page).toNat:ℤ) = r-page := Int.toNat_of_nonneg (by omega)
 163   have hq : ((page-1:ℕ):ℤ) = (page:ℤ)-1 := by omega
 164   ext <;> dsimp [lambdaBoundaryPoint] <;> omega
