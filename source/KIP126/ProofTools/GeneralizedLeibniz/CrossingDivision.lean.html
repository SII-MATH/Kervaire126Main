   5 /-! Strict finite-quotient source division for the absolute-boundary crossing
   6 comparison in AimPaper Proposition `prop:cross-dr-En`. Filtration above the
   7 smaller quotient window kills restriction; exactness gives an actual λ lift.
   8 The boundary formula uses the selected λ arrow, with its sphere evaluation
   9 identified by finite no-return. It does not assert λ division for an arbitrary
  10 normalized-map extension. -/

  23 private theorem finite_source_division (X : ClassicalObject)
  24     (i q : ℕ) (hi : 0 < i) (hiq : i < q) (m w s : ℤ)
  25     (hs : (i : ℤ) ≤ m+s-w)
  26     (a : BiHom m w (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) q))
  27     (ha : FiltrationAtLeast model.syntheticTower s a) :
  28     ∃ b : BiHom m w ((SyntheticCategory.biShift (0,-(i:ℤ))).obj
  29         (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) (q-i))),
  30       b ≫ ((model.quotients.tower _).triangle hi hiq).lambdaMap = a := by
  31   let A := model.nu.functor.obj (X.obj model.auxiliary)
  32   let T := (model.quotients.tower A).triangle hi hiq
  33   have harho : FiltrationAtLeast model.syntheticTower s (a ≫ T.rho) := by
  34     exact map_preserves_filtration (model.convergentMap
  35       (.quotient q (.nu X)) (.quotient i (.nu X)) T.rho) (m,w) s a ha
  36   have hz : a ≫ T.rho = 0 := finite_high_filtration_zero X i hi m w s hs _ harho
  37   obtain ⟨b,hb⟩ := (Triangle.mk T.lambdaMap T.rho T.delta).coyoneda_exact₂
  38     T.distinguished a hz
  39   exact ⟨b,hb.symm⟩

  42 private theorem shifted_quotient_maps_equal_on_spheres
  43     (X Y : ClassicalObject) (i q : ℕ) (hiq : i < q)
  44     (a b : (SyntheticCategory.biShift (0,-(i:ℤ))).obj
  45         (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) (q-i)) ⟶
  46       XModLambdaN (model.nu.functor.obj (Y.obj model.auxiliary)) q)
  47     (hab : (SyntheticCategory.biShift (0,-(i:ℤ))).map (XModLambdaN.incl _ (q-i)) ≫ a =
  48       (SyntheticCategory.biShift (0,-(i:ℤ))).map (XModLambdaN.incl _ (q-i)) ≫ b)
  49     (m w : ℤ)
  50     (c : BiHom m w ((SyntheticCategory.biShift (0,-(i:ℤ))).obj
  51       (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) (q-i)))) :
  52     c ≫ a = c ≫ b := by
  53   letI := model.exactness.shiftCommShift (0,-(i:ℤ))
  54   letI := model.exactness.shiftExact (0,-(i:ℤ))
  55   let F := SyntheticCategory.biShift (Syn:=SyntheticSpectrum) (0,-(i:ℤ))
  56   let V := model.nu.functor.obj (X.obj model.auxiliary)
  57   let T := XModLambdaN.cofiberTriangle V (q-i)
  58   have hzero : (F.mapTriangle.obj T).mor₂ ≫ (a-b) = 0 := by
  59     change F.map (XModLambdaN.incl V (q-i)) ≫ (a-b) = 0
  60     rw [Preadditive.comp_sub,hab,sub_self]
  61   obtain ⟨z,hz⟩ := Triangle.yoneda_exact₃ (F.mapTriangle.obj T)
  62     (F.map_distinguished T (XModLambdaN.cofiberTriangle_distinguished V (q-i))) (a-b) hzero
  63   let ε : (F.obj ((SyntheticCategory.biShift (0,-((q-i:ℕ):ℤ))).obj V))⟦(1:ℤ)⟧ ≅
  64       (SyntheticCategory.biShift (0,-(q:ℤ)-1)).obj
  65         (model.nu.functor.obj ((ClassicalObject.shift 1 X).obj model.auxiliary)) :=
  66     (shiftFunctor SyntheticSpectrum (1:ℤ)).mapIso
  67       ((lambdaShiftAddIso (q-i) i q (Nat.sub_add_cancel hiq.le)).app V) ≪≫
  68     (SyntheticCategory.biShift_compat (Syn:=SyntheticSpectrum) 1).symm.app _ ≪≫
  69     (SyntheticCategory.biShift_comp (0,-(q:ℤ)) (1,0)).app V ≪≫
  70     eqToIso (congrArg (fun p => (SyntheticCategory.biShift p).obj V)
  71       (show (0,-(q:ℤ))+(1,0)=(1,1)+(0,-(q:ℤ)-1) by ext <;> simp)) ≪≫
  72     (SyntheticCategory.biShift_comp (1,1) (0,-(q:ℤ)-1)).symm.app V ≪≫
  73     (SyntheticCategory.biShift (0,-(q:ℤ)-1)).mapIso (model.nu.suspensionIso _).symm
  74   have h := negative_shift_nu_to_quotient_zero (.shift 1 X) Y q (by omega) m w
  75     ((c ≫ (F.mapTriangle.obj T).mor₃) ≫ ε.hom) (ε.inv ≫ z)
  76   have hcz : (c ≫ (F.mapTriangle.obj T).mor₃) ≫ z = 0 := by
  77     simpa only [Category.assoc,Iso.hom_inv_id_assoc] using h
  78   apply sub_eq_zero.mp
  79   rw [←Preadditive.comp_sub,hz,←Category.assoc]
  80   exact hcz

  82 private theorem lambda_cofiber_map_exists (A : SyntheticSpectrum)
  83     (i q : ℕ) (hiq : i ≤ q) :
  84     let F := SyntheticCategory.biShift (Syn:=SyntheticSpectrum) (0,-(i:ℤ))
  85     letI := model.exactness.shiftCommShift (0,-(i:ℤ))
  86     ∃ c : F.obj (XModLambdaN A (q-i)) ⟶ XModLambdaN A q,
  87       F.map (XModLambdaN.incl A (q-i)) ≫ c = lambdaPow i A ≫ XModLambdaN.incl A q ∧
  88       c ≫ XModLambdaN.proj A q =
  89         F.map (XModLambdaN.proj A (q-i)) ≫ (F.commShiftIso (1:ℤ)).hom.app _ ≫
  90           (shiftFunctor SyntheticSpectrum (1:ℤ)).map
  91             ((lambdaShiftAddIso (q-i) i q (Nat.sub_add_cancel hiq)).hom.app A) := by
  92   let F := SyntheticCategory.biShift (Syn:=SyntheticSpectrum) (0,-(i:ℤ))
  93   letI := model.exactness.shiftCommShift (0,-(i:ℤ))
  94   letI := model.exactness.shiftExact (0,-(i:ℤ))
  95   let T := XModLambdaN.cofiberTriangle A (q-i)
  96   let U := XModLambdaN.cofiberTriangle A q
  97   have hs : F.map (lambdaPow (q-i) A) ≫ lambdaPow i A =
  98       (lambdaShiftAddIso (q-i) i q (Nat.sub_add_cancel hiq)).hom.app A ≫ lambdaPow q A :=
  99     (lambdaPow_add_comparison model.quotients.coherence (q-i) i q
 100       (Nat.sub_add_cancel hiq) A).symm
 101   obtain ⟨c,hc,hd⟩ := complete_distinguished_triangle_morphism
 102     (F.mapTriangle.obj T) U
 103     (F.map_distinguished T (XModLambdaN.cofiberTriangle_distinguished A (q-i)))
 104     (XModLambdaN.cofiberTriangle_distinguished A q)
 105     ((lambdaShiftAddIso (q-i) i q (Nat.sub_add_cancel hiq)).hom.app A) (lambdaPow i A) hs
 106   refine ⟨c,hc,?_⟩
 107   change (F.map (XModLambdaN.proj A (q-i)) ≫ (F.commShiftIso (1:ℤ)).hom.app _) ≫
 108       (shiftFunctor SyntheticSpectrum (1:ℤ)).map
 109         ((lambdaShiftAddIso (q-i) i q (Nat.sub_add_cancel hiq)).hom.app A) =
 110     c ≫ XModLambdaN.proj A q at hd
 111   simpa only [Category.assoc] using hd.symm

 113 private theorem lambda_proj_evaluation
 114     (X : ClassicalObject) (i q : ℕ) (hi : 0 < i) (hiq : i < q)
 115     (m w : ℤ) (b : BiHom m w ((SyntheticCategory.biShift (0,-(i:ℤ))).obj
 116       (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) (q-i)))) :
 117     let A := model.nu.functor.obj (X.obj model.auxiliary)
 118     let F := SyntheticCategory.biShift (Syn:=SyntheticSpectrum) (0,-(i:ℤ))
 119     letI := model.exactness.shiftCommShift (0,-(i:ℤ))
 120     b ≫ ((model.quotients.tower A).triangle hi hiq).lambdaMap ≫ XModLambdaN.proj A q =
 121       b ≫ F.map (XModLambdaN.proj A (q-i)) ≫ (F.commShiftIso (1:ℤ)).hom.app _ ≫
 122         (shiftFunctor SyntheticSpectrum (1:ℤ)).map
 123           ((lambdaShiftAddIso (q-i) i q (Nat.sub_add_cancel hiq.le)).hom.app A) := by
 124   letI := model.exactness.shiftCommShift (0,-(i:ℤ))
 125   let A := model.nu.functor.obj (X.obj model.auxiliary)
 126   let T := (model.quotients.tower A).triangle hi hiq
 127   obtain ⟨c,hc,hp⟩ := lambda_cofiber_map_exists A i q hiq.le
 128   have he := shifted_quotient_maps_equal_on_spheres X X i q hiq
 129     T.lambdaMap c (T.lambda_quotient.trans hc.symm) m w b
 130   change b ≫ T.lambdaMap ≫ XModLambdaN.proj A q = _
 131   rw [←Category.assoc,he,Category.assoc,hp]

 134 private def boundaryDivisionIso (X : ClassicalObject) (i q : ℕ) (hiq : i ≤ q) :
 135     (SyntheticCategory.biShift (0,-(i:ℤ))).obj
 136       ((SyntheticCategory.biShift (1,-((q-i:ℕ):ℤ))).obj
 137         (model.nu.functor.obj (X.obj model.auxiliary))) ≅
 138     (SyntheticCategory.biShift (1,-(q:ℤ))).obj
 139       (model.nu.functor.obj (X.obj model.auxiliary)) := by
 140   let A := model.nu.functor.obj (X.obj model.auxiliary)
 141   let F := SyntheticCategory.biShift (Syn:=SyntheticSpectrum) (0,-(i:ℤ))
 142   letI := model.exactness.shiftCommShift (0,-(i:ℤ))
 143   exact F.mapIso (biSuspensionIso ((q-i:ℕ):ℤ) A) ≪≫
 144     (F.commShiftIso (1:ℤ)).app _ ≪≫
 145     (shiftFunctor SyntheticSpectrum (1:ℤ)).mapIso
 146       ((lambdaShiftAddIso (q-i) i q (Nat.sub_add_cancel hiq)).app A) ≪≫
 147     (biSuspensionIso q A).symm

 166 private theorem iso_filtration_iff (A B : SyntheticObject)
 167     (e : A.obj model.nu model.auxiliary ≅ B.obj model.nu model.auxiliary)
 168     (m w s : ℤ) (a : BiHom m w (A.obj model.nu model.auxiliary)) :
 169     FiltrationAtLeast model.syntheticTower s (a ≫ e.hom) ↔
 170       FiltrationAtLeast model.syntheticTower s a := by
 171   constructor
 172   · intro h
 173     have he := map_preserves_filtration (model.convergentMap B A e.inv)
 174       (m,w) s (a ≫ e.hom) h
 175     simpa only [FiltrationAtLeast,Category.assoc,e.hom_inv_id,Category.comp_id] using he
 176   · exact map_preserves_filtration (model.convergentMap A B e.hom) (m,w) s a

 178 private theorem shift_filtration_iff (A : SyntheticObject) (δ p : ℤ × ℤ)
 179     (s : ℤ) (a : BiHom p.1 p.2 (A.obj model.nu model.auxiliary)) :
 180     FiltrationAtLeast model.syntheticTower s (shiftHomotopyClass δ p a) ↔
 181       FiltrationAtLeast model.syntheticTower s a := by
 182   let e := (essShift δ).filtered A p
 183   have forward {A B : SyntheticSpectrum} {p q : ℤ × ℤ}
 184       (e : FilteredPieceIso (model.syntheticTower.filtration A)
 185         (model.syntheticTower.filtration B) p q)
 186       (a : BiHom p.1 p.2 A) (ha : FiltrationAtLeast model.syntheticTower s a) :
 187       FiltrationAtLeast model.syntheticTower s (e.ambient.hom a) := by
 188     obtain ⟨ar,har⟩ := filtration_representative_exists model.syntheticTower A p s a ha
 189     have hh := ConcreteCategory.congr_hom (e.level_arrow s) ar
 190     have hm : e.ambient.hom a ∈ ModuleCat.subobjectModule _
 191         ((model.syntheticTower.filtration B).F s q) := by
 192       refine ⟨(e.level s).hom ar,?_⟩
 193       simpa only [ModuleCat.comp_apply,har] using hh
 194     simpa only [SyntheticTowerPresentation.filtration,OrderIso.apply_symm_apply,
 195       FiltrationAtLeast] using hm
 196   rw [←(essShift δ).ambient_eq A p a]
 197   constructor
 198   · intro ha
 199     have h := forward e.symm _ ha
 200     change FiltrationAtLeast model.syntheticTower s (e.ambient.inv (e.ambient.hom a)) at h
 201     have he : e.ambient.inv (e.ambient.hom a) = a :=
 202       ConcreteCategory.congr_hom e.ambient.hom_inv_id a
 203     rwa [he] at h
 204   · exact forward e a

 206 private theorem divided_boundary_filtration_iff

 208     (m w v : ℤ) (b : BiHom m w ((SyntheticCategory.biShift (0,-(i:ℤ))).obj

 211       (b ≫ ((model.quotients.tower _).triangle hi hiq).lambdaMap ≫
 212         absoluteBoundary (model.nu.functor.obj (X.obj model.auxiliary)) q) ↔
 213     FiltrationAtLeast model.syntheticTower v
 214       (b ≫ (SyntheticCategory.biShift (0,-(i:ℤ))).map
 215         (absoluteBoundary (model.nu.functor.obj (X.obj model.auxiliary)) (q-i))) := by
 216   rw [absolute_boundary_division_evaluation X i q hi hiq m w b]
 217   simpa only [Category.assoc] using iso_filtration_iff

 219     (.shift (1,-(q:ℤ)) (.nu X)) (boundaryDivisionIso X i q hiq.le)

 221       (absoluteBoundary (model.nu.functor.obj (X.obj model.auxiliary)) (q-i)))
