   7 /-! AimPaper, the last step of the Generalized Leibniz Rule: the actual
   8 right-edge representative with high absolute boundary lifts through rho.
   9 The finite E-infinity window reflects filtration and shows that the original
  10 classical label is a later cycle, even when its weight is below its degree.
  11 No division by lambda or exact first-label choice is required. -/

  25 private theorem rho_lift_detects (X : ClassicalObject)
  26     (i j : ℕ) (hi : 0 < i) (hij : i ≤ j) (s t w : ℤ)
  27     (hw : 0 ≤ t-w ∧ t-w < i) (y : Ambient model X (s,t))
  28     (a : BiHom (t-s) w (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) i))
  29     (ha : Detects (model.convergence (.quotient i (.nu X))) (s,t,w)
  30       ((model.family.quotientProjection (model.nu.functor.obj (X.obj model.auxiliary)) i).pageMap 2 (s,t,w)
  31         (nuWeightLabel model X s t w (by omega) y)) a)
  32     (b : BiHom (t-s) w (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) j))
  33     (hb : b ≫ (model.quotients.tower _).rho i j hij = a) :
  34     Detects (model.convergence (.quotient j (.nu X))) (s,t,w)
  35       ((model.family.quotientProjection (model.nu.functor.obj (X.obj model.auxiliary)) j).pageMap 2 (s,t,w)
  36         (nuWeightLabel model X s t w (by omega) y)) b := by
  37   let A := model.nu.functor.obj (X.obj model.auxiliary)
  38   let ρ := (model.quotients.tower A).rho i j hij
  39   let p : ℤ × ℤ := (t-s,w)
  40   have haf : FiltrationAtLeast model.syntheticTower s a :=
  41     detects_filtration (.quotient i (.nu X)) (s,t,w) _ a ha
  42   have hbf : FiltrationAtLeast model.syntheticTower s b := by
  43     have hb0 := filtration_at_zero model.syntheticTower _ (t-s) w b
  44     by_cases hs : s ≤ 0
  45     · exact model.syntheticTower.filtrationSubmodule_antitone _ p hs hb0
  46     · let L : ℤ → AddSubgroup (BiHom (t-s) w (XModLambdaN A j)) :=
  47         fun k => (model.syntheticTower.filtrationSubmodule (XModLambdaN A j) k p).toAddSubgroup
  48       let E : ℤ → AddSubgroup (BiHom (t-s) w (XModLambdaN A i)) :=
  49         fun k => (model.syntheticTower.filtrationSubmodule (XModLambdaN A i) k p).toAddSubgroup
  50       have hsN : (0:ℤ) + (s.toNat:ℤ) = s := by omega
  51       have hρb : (syntheticHomotopyMap ρ p).hom.toAddMonoidHom b ∈ E (0+s.toNat) := by
  52         change b ≫ ρ ∈ E (0+s.toNat)
  53         rw [hsN,hb]
  54         exact haf
  55       have hh := finite_filtration_reflection (syntheticHomotopyMap ρ p).hom.toAddMonoidHom L E
  56         (fun k l hkl z hz => model.syntheticTower.filtrationSubmodule_antitone _ p hkl hz)
  57         0 s.toNat (fun k hk hks z hz hρz =>
  58           next_filtration_of_infinity_injective
  59             (model.convergentMap (.quotient j (.nu X)) (.quotient i (.nu X)) ρ)
  60             p k (rho_infty_injective X i j hi hij k (p.1+k) p.2 (by dsimp [p]; omega))
  61             z hz hρz) b hb0 hρb
  62       rw [hsN] at hh
  63       exact hh
  64   obtain ⟨e,he,a',ha',hga⟩ := ha
  65   obtain ⟨b',hb'⟩ := filtration_representative_exists model.syntheticTower
  66     (XModLambdaN A j) p s b hbf
  67   let e' := ((model.convergence (.quotient j (.nu X))).identification (s,t,w)).inv
  68     ((model.syntheticTower.filtration (XModLambdaN A j)).toAssociatedGraded s p b')
  69   have hg := grade_natural (.quotient j (.nu X)) (.quotient i (.nu X)) ρ (s,t,w) b' a'
  70     (by exact (congrArg (fun z => z ≫ ρ) hb').trans (hb.trans ha'.symm))
  71   have heq : (model.family.map ρ).eInftyMap (s,t,w) e' = e := by
  72     have hiGrade : ((model.convergence (.quotient i (.nu X))).identification (s,t,w)).inv
  73         ((model.syntheticTower.filtration (XModLambdaN A i)).toAssociatedGraded s p a') = e := by
  74       exact (congrArg ((model.convergence (.quotient i (.nu X))).identification (s,t,w)).inv hga.symm).trans
  75         (((model.convergence (.quotient i (.nu X))).identification (s,t,w)).toLinearEquiv.symm_apply_apply e)
  76     exact hg.symm.trans hiGrade
  77   obtain ⟨hy,hycl⟩ := (KIP126.Literature.Standard.eInfinity_finite_labels X i hi s t w hw y e).mp he
  78   obtain ⟨z,hz⟩ := CrossingLabels.subquotient_rep
  79     (cycles model X (j-t+w) (s,t)) (boundaries model X (1+t-w) (s,t))
  80     (KIP126.Literature.Standard.infinityFiniteWindow X j (by omega) (s,t) w (by constructor <;> omega) e')
  81   have hr := KIP126.Literature.Standard.eInfinity_rho_finite X i j hi hij (s,t) w hw e'
  82   rw [heq,hycl,←hz] at hr
  83   have hzold : z.val ∈ cycles model X (i-t+w) (s,t) :=
  84     cycles_antitone model X (s,t) (by omega : (i:ℤ)-t+w ≤ (j:ℤ)-t+w) z.property
  85   have hclass : subquotientClass (cycles model X (i-t+w) (s,t))
  86       (boundaries model X (1+t-w) (s,t)) ⟨y,hy⟩ =
  87     subquotientClass (cycles model X (i-t+w) (s,t))
  88       (boundaries model X (1+t-w) (s,t)) ⟨z.val,hzold⟩ := hr
  89   have hzero : subquotientClass (cycles model X (i-t+w) (s,t))
  90       (boundaries model X (1+t-w) (s,t)) (⟨y,hy⟩-⟨z.val,hzold⟩) = 0 := by
  91     rw [map_sub,hclass,sub_self]
  92   have hboundary : y-z.val ∈ boundaries model X (1+t-w) (s,t) :=
  93     (CrossingLabels.subquotient_zero_iff _ _ _).mp hzero
  94   have hdiff := boundaries_le_cycles model X (s,t) (1+t-w) (j-t+w) hboundary
  95   have hylater : y ∈ cycles model X (j-t+w) (s,t) := by
  96     simpa only [sub_add_cancel] using
  97       (cycles model X (j-t+w) (s,t)).add_mem hdiff z.property
  98   have hyz : subquotientClass (cycles model X (j-t+w) (s,t))
  99       (boundaries model X (1+t-w) (s,t)) ⟨y,hylater⟩ =
 100       subquotientClass (cycles model X (j-t+w) (s,t))
 101         (boundaries model X (1+t-w) (s,t)) z := by
 102     apply sub_eq_zero.mp
 103     rw [←map_sub]
 104     exact (CrossingLabels.subquotient_zero_iff _ _ _).mpr hboundary
 105   have he' : HasInfinityRepresentative _ 2 (s,t,w)
 106       ((model.family.quotientProjection (model.nu.functor.obj (X.obj model.auxiliary)) j).pageMap 2 (s,t,w)
 107         (nuWeightLabel model X s t w (by omega) y)) e' := by
 108     apply (KIP126.Literature.Standard.eInfinity_finite_labels X j (by omega) s t w
 109       (by constructor <;> omega) y e').mpr
 110     exact ⟨hylater,hz.symm.trans hyz.symm⟩
 111   refine ⟨e',he',b',hb',?_⟩
 112   exact ((model.convergence (.quotient j (.nu X))).identification (s,t,w)).toLinearEquiv.apply_symm_apply _

 115 private theorem rho_lift_detected_cycles (X : ClassicalObject)
 116     (i j : ℕ) (hi : 0 < i) (hij : i ≤ j) (s t w : ℤ)
 117     (hw : 0 ≤ t-w ∧ t-w < i) (y : Ambient model X (s,t))
 118     (a : BiHom (t-s) w (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) i))
 119     (ha : Detects (model.convergence (.quotient i (.nu X))) (s,t,w)
 120       ((model.family.quotientProjection (model.nu.functor.obj (X.obj model.auxiliary)) i).pageMap 2 (s,t,w)

 127     (by constructor <;> omega) y e).mp he).1

 130 private theorem absolute_boundary_filtration_lift_weighted (X : ClassicalObject)
 131     (i j : ℕ) (hi : 0 < i) (hij : i < j) (m w : ℤ)
 132     (a : BiHom m w (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) i))
 133     (hfil : FiltrationAtLeast model.syntheticTower (w-m+j+1)
 134       (biSuspensionHomEquiv m w i (model.nu.functor.obj (X.obj model.auxiliary))
 135         (a ≫ XModLambdaN.proj _ i))) :
 136     ∃ b : BiHom m w (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) j),
 137       b ≫ (model.quotients.tower _).rho i j hij.le = a := by
 138   have hm : m = w-(w-m) := by omega
 139   generalize hs : w-m = s0 at *
 140   subst m
 141   exact absolute_boundary_filtration_lift X i j hi hij s0 w a hfil

 158 private theorem finite_extension_target_weight_label (X Y : ClassicalObject)
 159     (f : X.obj model.auxiliary ⟶ Y.obj model.auxiliary)
 160     (q : ℕ) (n s t : ℤ) (hn : (normalizedExponent model.classicalTower f : ℤ) ≤ n)
 161     (y : Ambient model Y (s+n,t+n)) :
 162     regradeE2 (by ext <;> dsimp <;> omega)
 163       (finiteTargetForExtension model X Y f q n s t hn y) =
 164     (model.family.quotientProjection (model.nu.functor.obj (Y.obj model.auxiliary)) q).pageMap
 165       2 (s+n,t+n,t+normalizedExponent model.classicalTower f)
 166       (nuWeightLabel model Y (s+n) (t+n) (t+normalizedExponent model.classicalTower f) (by omega) y) := by
 167   have regrade_comp {E : SyntheticAdamsSS} {p₁ p₂ p₃ : Tridegree}
 168       (h₁ : p₁=p₂) (h₂ : p₂=p₃) (z : E.E₂ p₁) :
 169       regradeE2 h₂ (regradeE2 h₁ z) = regradeE2 (h₁.trans h₂) z := by
 170     subst p₂; subst p₃; rfl
 171   have hpow : ((t+n)-(t+normalizedExponent model.classicalTower f)).toNat =
 172       (n-normalizedExponent model.classicalTower f).toNat := by congr 1; omega
 173   unfold finiteTargetForExtension finiteTargetLabel nuWeightLabel
 174   rw [regrade_comp]
 175   change regradeE2 _ (familyPageMap model.family (XModLambdaN.incl _ q) 2 _ _) =
 176     familyPageMap model.family (XModLambdaN.incl _ q) 2 _ (regradeE2 _ _)
 177   rw [page_regrade]
 178   have labels (k k' : ℕ) (hk : k=k') (p : Tridegree)
 179       (h₁ : (s+n,t+n,t+n-k)=p) (h₂ : (s+n,t+n,t+n-k')=p) :
 180       regradeE2 h₁ (targetNuLabel model Y (s+n) (t+n) k y) =
 181         regradeE2 h₂ (targetNuLabel model Y (s+n) (t+n) k' y) := by
 182     subst k'; rfl
 183   exact congrArg (familyPageMap model.family (XModLambdaN.incl _ q) 2
 184     (s+n,t+n,t+normalizedExponent model.classicalTower f))
 185     (labels _ _ hpow.symm _ _ _)

 187 private theorem gl1_right_edge_survival (X Y : ClassicalObject)
 188     (f : X.obj model.auxiliary ⟶ Y.obj model.auxiliary)
 189     (page : ℕ) (r m l s t : ℤ) (hpage : 2 ≤ page) (hpr : (page:ℤ) ≤ r)
 190     (hm : (normalizedExponent model.classicalTower f : ℤ) ≤ m)
 191     (hmUpper : m ≤ (page:ℤ)-2+normalizedExponent model.classicalTower f)
 192     (hl : (normalizedExponent model.classicalTower f : ℤ) < l)
 193     (y : Ambient model Y (s+m,t+m))
 194     (a : BiHom ((t+m)-(s+m)) (t+normalizedExponent model.classicalTower f)
 195       (XModLambdaN (model.nu.functor.obj (Y.obj model.auxiliary)) (page-1)))
 196     (ha : Detects (model.convergence (.quotient (page-1) (.nu Y)))
 197       (s+m,t+m,t+normalizedExponent model.classicalTower f)
 198       (regradeE2 (by ext <;> dsimp <;> omega)
 199         (finiteTargetForExtension model X Y f (page-1) m s t hm y)) a)
 200     (hfil : FiltrationAtLeast model.syntheticTower (s+r+l)
 201       (biSuspensionHomEquiv ((t+m)-(s+m)) (t+normalizedExponent model.classicalTower f) ((page-1:ℕ):ℤ)
 202         (model.nu.functor.obj (Y.obj model.auxiliary)) (a ≫ XModLambdaN.proj _ (page-1)))) :
 203     IsCycle model Y (r+l-m-1) (s+m,t+m) y := by
 204   let j := (r+l-normalizedExponent model.classicalTower f-1).toNat
 205   have hj : (j:ℤ)=r+l-normalizedExponent model.classicalTower f-1 := by
 206     dsimp [j]; omega
 207   rw [finite_extension_target_weight_label] at ha
 208   have hfil' : FiltrationAtLeast model.syntheticTower
 209       ((t+normalizedExponent model.classicalTower f)-((t+m)-(s+m))+j+1)
 210       (biSuspensionHomEquiv ((t+m)-(s+m)) (t+normalizedExponent model.classicalTower f) ((page-1:ℕ):ℤ)
 211         (model.nu.functor.obj (Y.obj model.auxiliary)) (a ≫ XModLambdaN.proj _ (page-1))) := by
 212     convert hfil using 1
 213     omega
 214   have hcycle := absolute_boundary_detected_cycles_weighted Y (page-1) j (by omega) (by omega)
 215     (s+m) (t+m) (t+normalizedExponent model.classicalTower f) (by constructor <;> omega) y a ha hfil'
 216   refine ⟨by omega,?_⟩
 217   convert hcycle using 1
 218   congr 1
 219   omega
