   7 /-! Sphere automorphisms act trivially after the first lambda quotient.
   8 The actual sphere E2 coordinates in degree (0,0) identify its first quotient
   9 with the two-element additive group. Transport through the fixed quotient
  10 shift comparison proves the statement for every bigraded sphere and every
  11 target. This removes the choice of suspension comparison on Q1 labels in
  12 AimPaper Remark `rmk:qvoewfj`; no characteristic-two claim is made about
  13 unquotiented homotopy groups. -/

  25 private def sphere_first_quotient_f2 :
  26     ((S_0_0 : SyntheticSpectrum) ⟶ XModLambdaN S_0_0 1) ≃+ ZMod 2 := by
  27   let z : Smn (Syn:=SyntheticSpectrum) 0 0 ≅ S_0_0 := SyntheticCategory.biShift_zero.app _
  28   let e : ((S_0_0 : SyntheticSpectrum) ⟶ XModLambdaN S_0_0 1) ≃+
  29       BiHom 0 0 (XModLambdaN S_0_0 1) :=
  30     { toFun := fun a => z.hom ≫ a
  31       invFun := fun a => z.inv ≫ a
  32       left_inv := by intro a; simp
  33       right_inv := by intro a; simp
  34       map_add' := by intro a b; simp }
  35   let c := sphereFirstQuotient model.nu model.quotients.functoriality model.firstQuotient 0 0
  36   have hn : KIP126.Computation.Standard.archive.basisCount 0 0 = 1 :=
  37     (KIP126.Computation.Table.archive_dimension 0 0 (by decide)).trans (by rfl)
  38   let d : model.classical.sphere.Page 2 (0,0) ≃ₗ[ℤ] (Fin 1 →₀ ZMod 2) :=
  39     hn ▸ KIP126.Computation.Standard.coordinates 0 0 (by decide)
  40   exact e.trans (c.toAddEquiv.trans (d.toAddEquiv.trans
  41     (Finsupp.uniqueLinearEquiv ℤ (ZMod 2) (0 : Fin 1)).toAddEquiv))

  43 private theorem additive_automorphism_f2_identity {A : Type*} [AddGroup A]
  44     (e : A ≃+ ZMod 2) (f : A ≃+ A) (x : A) : f x=x := by
  45   let g : ZMod 2 ≃+ ZMod 2 := e.symm.trans (f.trans e)
  46   have hcases : ∀ z : ZMod 2, z=0 ∨ z=1 := by decide
  47   have h1 : g 1=1 := by
  48     rcases hcases (g 1) with h | h
  49     · have hz : (1:ZMod 2)=0 := g.injective (h.trans (map_zero g).symm)
  50       exact False.elim (by norm_num at hz)
  51     · exact h
  52   have hh : ∀ z : ZMod 2, g z=z := by
  53     intro z
  54     rcases hcases z with rfl | rfl
  55     · exact map_zero g
  56     · exact h1
  57   apply e.injective
  58   simpa only [g,AddEquiv.trans_apply,AddEquiv.symm_apply_apply] using hh (e x)

  60 private theorem sphere_automorphism_first_quotient
  61     (u : (S_0_0 : SyntheticSpectrum) ≅ S_0_0)
  62     (a : (S_0_0 : SyntheticSpectrum) ⟶ XModLambdaN S_0_0 1) :
  63     u.hom ≫ a = a := by
  64   let f : ((S_0_0 : SyntheticSpectrum) ⟶ XModLambdaN S_0_0 1) ≃+
  65       ((S_0_0 : SyntheticSpectrum) ⟶ XModLambdaN S_0_0 1) :=
  66     { toFun := fun a => u.hom ≫ a
  67       invFun := fun a => u.inv ≫ a
  68       left_inv := by intro a; simp
  69       right_inv := by intro a; simp
  70       map_add' := by intro a b; simp }
  71   exact additive_automorphism_f2_identity sphere_first_quotient_f2 f a

  74 private theorem bigraded_sphere_automorphism_first_quotient (m w : ℤ)
  75     (u : Smn (Syn:=SyntheticSpectrum) m w ≅ Smn m w) :
  76     u.hom ≫ XModLambdaN.incl (Smn m w) 1 = XModLambdaN.incl (Smn m w) 1 := by
  77   let F := SyntheticCategory.biShift (Syn:=SyntheticSpectrum) (m,w)
  78   let ff := SyntheticCategory.biShift_fullyFaithful (Syn:=SyntheticSpectrum) (m,w)
  79   let u0 : (S_0_0 : SyntheticSpectrum) ≅ S_0_0 := ff.preimageIso u
  80   have hu : F.map u0.hom = u.hom := ff.map_preimage _
  81   have h0 := sphere_automorphism_first_quotient u0 (XModLambdaN.incl S_0_0 1)
  82   letI := model.exactness.shiftCommShift (m,w)
  83   letI := model.exactness.shiftExact (m,w)
  84   obtain ⟨Q,hQ,_⟩ := KIP126.ProofTools.Internal.quotient_shift_exists
  85     model.quotients.coherence (m,w) 1 (S_0_0 : SyntheticSpectrum)
  86   apply (cancel_mono Q.hom).mp
  87   dsimp only [Smn] at u ⊢
  88   rw [Category.assoc,hQ,←hu,←F.map_comp,h0]

  90 private theorem sphere_automorphism_quotient_projection (m w : ℤ)
