  21 private def verticalTranslate (A : SyntheticSpectrum) (e m w : ℤ) :
  22     BiHom m w A ≃+ BiHom m (w+e) ((SyntheticCategory.biShift (0,e)).obj A) :=
  23   (suspInvarianceAddEquiv m w 0 e A).trans
  24     (regradeAddEquiv (add_zero m) rfl)
  25 
  26 private theorem vertical_translate_filtration_iff (A : SyntheticObject)
  27     (e m w v : ℤ) (a : BiHom m w (A.obj model.nu model.auxiliary)) :
  28     FiltrationAtLeast model.syntheticTower v (verticalTranslate _ e m w a) ↔
  29       FiltrationAtLeast model.syntheticTower v a := by
  30   change FiltrationAtLeast model.syntheticTower v
  31     (homotopyRegrade (add_zero m) rfl (susp_invariance m w 0 e _ a)) ↔ _
  32   rw [regrade_filtration_iff]
  33   have he : susp_invariance m w 0 e _ a = shiftHomotopyClass (0,e) (m,w) a := by
  34     simp [susp_invariance,shiftHomotopyClass,Iso.homCongr]
  35   rw [he]
  36   exact shift_filtration_iff A (0,e) (m,w) v a

  38 private theorem vertical_translate_comp (A B : SyntheticSpectrum) (f : A ⟶ B)
  39     (e m w : ℤ) (a : BiHom m w A) :
  40     verticalTranslate A e m w a ≫ (SyntheticCategory.biShift (0,e)).map f =
  41       verticalTranslate B e m w (a ≫ f) := by
  42   simp [verticalTranslate,AddEquiv.trans_apply,regradeAddEquiv,
  43     suspInvarianceAddEquiv,susp_invariance,Iso.homCongr,regrade_precomp,Functor.map_comp]

  45 private theorem quotient_boundary_shift_exists (A : SyntheticSpectrum) (e : ℤ) (q : ℕ) :
  46     ∃ (eQ : XModLambdaN ((SyntheticCategory.biShift (0,e)).obj A) q ≅
  47         (SyntheticCategory.biShift (0,e)).obj (XModLambdaN A q))
  48       (eD : (SyntheticCategory.biShift (0,e)).obj
  49           ((SyntheticCategory.biShift (1,-(q:ℤ))).obj A) ≅
  50         (SyntheticCategory.biShift (1,-(q:ℤ))).obj
  51           ((SyntheticCategory.biShift (0,e)).obj A)),
  52       eQ.hom ≫ (SyntheticCategory.biShift (0,e)).map (absoluteBoundary A q) ≫ eD.hom =
  53         absoluteBoundary ((SyntheticCategory.biShift (0,e)).obj A) q := by
  54   let F := SyntheticCategory.biShift (Syn:=SyntheticSpectrum) (0,e)
  55   letI := model.exactness.shiftCommShift (0,e)
  56   letI := model.exactness.shiftExact (0,e)
  57   obtain ⟨eQ,_,e₁,_,hδ⟩ := quotient_shift_exists model.quotients.coherence (0,e) q A
  58   let eD := F.mapIso (biSuspensionIso q A) ≪≫
  59     (F.commShiftIso (1:ℤ)).app _ ≪≫
  60     (shiftFunctor SyntheticSpectrum (1:ℤ)).mapIso e₁.symm ≪≫
  61     (biSuspensionIso q (F.obj A)).symm
  62   refine ⟨eQ,eD,?_⟩
  63   have h := congrArg (fun z => z ≫ (shiftFunctor SyntheticSpectrum (1:ℤ)).map e₁.inv ≫
  64     (biSuspensionIso q (F.obj A)).inv) hδ
  65   simpa only [absoluteBoundary,eD,F,Iso.trans_hom,Iso.symm_hom,Functor.mapIso_hom,
  66     Functor.map_comp,Category.assoc,Iso.map_inv_hom_id_assoc,Iso.map_hom_inv_id_assoc,
  67     Iso.app_hom] using h

  69 private theorem shifted_boundary_crossing_correction (X : ClassicalObject)
  70     (q : ℕ) (e r s t u v : ℤ) (hnc : NoCrossingOn model X r (q+1) (s,t))
  71     (hu : s < u) (hv : v ≤ s+r)
  72     (a : BiHom (t-s) (t+e) (XModLambdaN ((SyntheticCategory.biShift (0,e)).obj
  73       (model.nu.functor.obj (X.obj model.auxiliary))) q))
  74     (ha : FiltrationAtLeast model.syntheticTower u a)
  75     (hδ : FiltrationAtLeast model.syntheticTower v
  76       (a ≫ absoluteBoundary ((SyntheticCategory.biShift (0,e)).obj
  77         (model.nu.functor.obj (X.obj model.auxiliary))) q)) :
  78     ∃ a0 : BiHom (t-s) (t+e) (XModLambdaN ((SyntheticCategory.biShift (0,e)).obj
  79         (model.nu.functor.obj (X.obj model.auxiliary))) q),
  80       FiltrationAtLeast model.syntheticTower u a0 ∧
  81       FiltrationAtLeast model.syntheticTower (u+1) (a-a0) ∧
  82       FiltrationAtLeast model.syntheticTower (v+1)
  83         (a0 ≫ absoluteBoundary ((SyntheticCategory.biShift (0,e)).obj
  84           (model.nu.functor.obj (X.obj model.auxiliary))) q) := by
  85   let A := model.nu.functor.obj (X.obj model.auxiliary)
  86   let F := SyntheticCategory.biShift (Syn:=SyntheticSpectrum) (0,e)
  87   obtain ⟨eQ,eD,hsquare⟩ := quotient_boundary_shift_exists A e q
  88   let E := verticalTranslate (XModLambdaN A q) e (t-s) t
  89   let ED := verticalTranslate ((SyntheticCategory.biShift (1,-(q:ℤ))).obj A) e (t-s) t
  90   let b := E.symm (a ≫ eQ.hom)
  91   have hb : E b = a ≫ eQ.hom := E.apply_symm_apply _
  92   have hfil : ∀ (k : ℤ) (z : BiHom (t-s) t (XModLambdaN A q)),
  93       FiltrationAtLeast model.syntheticTower k (E z ≫ eQ.inv) ↔
  94         FiltrationAtLeast model.syntheticTower k z := by
  95     intro k z
  96     exact (iso_filtration_iff (.shift (0,e) (.quotient q (.nu X)))
  97       (.quotient q (.shift (0,e) (.nu X))) eQ.symm (t-s) (t+e) k (E z)).trans
  98         (vertical_translate_filtration_iff (.quotient q (.nu X)) e (t-s) t k z)
  99   have hboundary : ∀ z : BiHom (t-s) t (XModLambdaN A q),
 100       (E z ≫ eQ.inv) ≫ absoluteBoundary (F.obj A) q =
 101         ED (z ≫ absoluteBoundary A q) ≫ eD.hom := by
 102     intro z
 103     rw [←hsquare]
 104     simp only [Category.assoc,Iso.inv_hom_id_assoc]
 105     rw [←Category.assoc,vertical_translate_comp]
 106   have hboundary_fil : ∀ (k : ℤ) (z : BiHom (t-s) t (XModLambdaN A q)),
 107       FiltrationAtLeast model.syntheticTower k
 108           ((E z ≫ eQ.inv) ≫ absoluteBoundary (F.obj A) q) ↔
 109         FiltrationAtLeast model.syntheticTower k (z ≫ absoluteBoundary A q) := by
 110     intro k z
 111     rw [hboundary]
 112     exact (iso_filtration_iff (.shift (0,e) (.shift (1,-(q:ℤ)) (.nu X)))
 113       (.shift (1,-(q:ℤ)) (.shift (0,e) (.nu X))) eD (t-s) (t+e) k _).trans
 114         (vertical_translate_filtration_iff (.shift (1,-(q:ℤ)) (.nu X)) e (t-s) t k _)
 115   have hba : E b ≫ eQ.inv = a := by rw [hb]; simp
 116   obtain ⟨b0,hb0,hdiff,hδ0⟩ := boundary_crossing_correction X q r s t u v hnc hu hv b
 117     ((hfil u b).mp (by rwa [hba])) ((hboundary_fil v b).mp (by rwa [hba]))
 118   refine ⟨E b0 ≫ eQ.inv,(hfil u b0).mpr hb0,?_,(hboundary_fil (v+1) b0).mpr hδ0⟩
 119   have h := (hfil (u+1) (b-b0)).mpr hdiff
 120   simpa only [map_sub,Preadditive.sub_comp,hba] using h

 122 private theorem classical_noCrossing_shifted_absolute_boundary (X : ClassicalObject)
 123     (page : ℕ) (e r s t : ℤ) (hnc : NoCrossingOn model X r page (s,t)) :
 124     TowerNoCrossing (model.convergentMap
 125       (.quotient (page-1) (.shift (0,e) (.nu X)))
 126       (.shift (1,-((page-1:ℕ):ℤ)) (.shift (0,e) (.nu X)))
 127       (absoluteBoundary ((SyntheticCategory.biShift (0,e)).obj
 128         (model.nu.functor.obj (X.obj model.auxiliary))) (page-1)))
 129       (t-s,t+e) s r := by
 130   apply tower_no_crossing_of_correction
 131   intro u v hu _ hv a ha hδ
 132   have hnc' : NoCrossingOn model X r ((page-1:ℕ)+1) (s,t) := by
 133     convert hnc using 1
 134     have h := hnc.1
 135     omega
 136   exact shifted_boundary_crossing_correction X (page-1) e r s t u v hnc' hu hv a ha hδ
