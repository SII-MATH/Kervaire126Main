  15 /-- Raising the source filtration terminates at the target filtration.
  16 The correction premise is exactly nonessentiality of the fixed target grade. -/
  17 private theorem higher_source_image_of_nonessential
  18     {A B : Type*} [AddCommGroup A] [AddCommGroup B]
  19     (f : A →+ B) (F : ℤ → AddSubgroup A) (G : ℤ → AddSubgroup B)
  20     (hG : Antitone G)
  21     (hf : ∀ u a, a ∈ F u → f a ∈ G u)
  22     (s t : ℤ) (hst : s ≤ t)
  23     (hcorrect : ∀ u, s < u → u ≤ t → ∀ a, a ∈ F u → f a ∈ G t →
  24       ∃ b, b ∈ F u ∧ a - b ∈ F (u + 1) ∧ f b ∈ G (t + 1))
  25     (a : A) (ha : a ∈ F (s + 1)) (hfa : f a ∈ G t) :
  26     f a ∈ G (t + 1) := by
  27   have aux : ∀ N : ℕ, ∀ u : ℤ, s < u → u + N = t + 1 →
  28       ∀ a, a ∈ F u → f a ∈ G t → f a ∈ G (t + 1) := by
  29     intro N
  30     induction N with
  31     | zero =>
  32       intro u hu he a ha hfa
  33       have hut : u = t + 1 := by omega
  34       subst u
  35       exact hf _ _ ha
  36     | succ N ih =>
  37       intro u hu he a ha hfa
  38       obtain ⟨b, hb, hab, hfb⟩ := hcorrect u hu (by omega) a ha hfa
  39       have hsub : f (a - b) ∈ G t := by
  40         rw [map_sub]
  41         exact (G t).sub_mem hfa (hG (by omega) hfb)
  42       have hh := ih (u + 1) (by omega) (by omega) (a - b) hab hsub
  43       have h := (G (t + 1)).add_mem hh hfb
  44       simpa only [map_sub, sub_add_cancel] using h
  45   exact aux (t - s).toNat (s + 1) (by omega) (by omega) a ha hfa

  47 /-- A finite interval of essential-crossing exclusions makes every higher-source
  48 correction invisible at the selected target grade. No completion is used. -/
  49 private theorem higher_source_image_of_no_crossing
  50     {A B : Type*} [AddCommGroup A] [AddCommGroup B]
  51     (f : A →+ B) (F : ℤ → AddSubgroup A) (G : ℤ → AddSubgroup B)
  52     (hG : Antitone G)
  53     (hf : ∀ u a, a ∈ F u → f a ∈ G u)
  54     (s n : ℤ) (hn : 0 ≤ n)
  55     (hcorrect : ∀ u v, s < u → u ≤ v → v ≤ s + n →
  56       ∀ a, a ∈ F u → f a ∈ G v →
  57         ∃ b, b ∈ F u ∧ a - b ∈ F (u + 1) ∧ f b ∈ G (v + 1))
  58     (a : A) (ha : a ∈ F (s + 1)) : f a ∈ G (s + n + 1) := by
  59   by_contra hnot
  60   obtain ⟨v, hv, hvn, hva, hvnext⟩ :=
  61     first_nonzero_filtration_grade G (s + 1) n.toNat (f a)
  62       (hf _ _ ha) (by simpa only [Int.toNat_of_nonneg hn, add_assoc, add_comm, add_left_comm] using hnot)
  63   apply hvnext
  64   exact higher_source_image_of_nonessential f F G hG hf s v (by omega)
  65     (fun u hsu huv => hcorrect u v hsu huv (by omega)) a ha hva

  67 /-- The image of every representative has the same target grade once all
  68 higher-source corrections have zero target grade. -/
  69 private theorem image_same_target_of_no_crossing
  70     {A B : Type*} [AddCommGroup A] [AddCommGroup B]
  71     (f : A →+ B) (F : ℤ → AddSubgroup A) (G : ℤ → AddSubgroup B)
  72     (hG : Antitone G)
  73     (hf : ∀ u a, a ∈ F u → f a ∈ G u)
  74     (s n : ℤ) (hn : 0 ≤ n)
  75     (hcorrect : ∀ u v, s < u → u ≤ v → v ≤ s + n →
  76       ∀ a, a ∈ F u → f a ∈ G v →
  77         ∃ b, b ∈ F u ∧ a - b ∈ F (u + 1) ∧ f b ∈ G (v + 1))
  78     (a a₀ : A) (hrep : a - a₀ ∈ F (s + 1)) :
  79     f a - f a₀ ∈ G (s + n + 1) := by
  80   simpa only [map_sub] using
  81     higher_source_image_of_no_crossing f F G hG hf s n hn hcorrect (a-a₀) hrep

  94 private theorem tower_essential_iff_no_zero
  95     (M : ConvergentMap E g cx.toConvergence cy.toConvergence)
  96     (p : ℤ × ℤ) (s n : ℤ)
  97     (x : ((E.obj X).sequence.ssData (s,p.1+s,p.2)).eInfty)
  98     (y : ((E.obj Y).sequence.ssData (s+n,p.1+(s+n),p.2)).eInfty)
  99     (he : TowerExtensionEquation M p s n x y) :
 100     TowerEssentialExtension M p s n x y ↔
 101       ¬TowerExtensionEquation M p s n x 0 := by

 107       (by simp [SyntheticTowerConvergence.toConvergence])
 108       (by simp [SyntheticTowerConvergence.toConvergence]) (elementMap x) (elementMap y) he

 110 /-- Nonessentiality supplies a representative with the same leading source
 111 and zero leading target; the difference raises the source filtration. -/
 112 private theorem tower_correction_of_nonessential
 113     (M : ConvergentMap E g cx.toConvergence cy.toConvergence)
 114     (p : ℤ × ℤ) (s n : ℤ) (hn : 0 ≤ n)
 115     (hnot : ∀ x y, ¬TowerEssentialExtension M p s n x y)
 116     (a : BiHom p.1 p.2 X) (ha : a ∈ P.filtrationSubmodule X s p)
 117     (hfa : a ≫ g ∈ P.filtrationSubmodule Y (s+n) p) :
 118     ∃ b, b ∈ P.filtrationSubmodule X s p ∧
 119       a-b ∈ P.filtrationSubmodule X (s+1) p ∧
 120       b ≫ g ∈ P.filtrationSubmodule Y (s+n+1) p := by
 121   obtain ⟨ar, har⟩ := filtration_representative_exists P X p s a ha
 122   obtain ⟨br, hbr⟩ := filtration_representative_exists P Y p (s+n) (a≫g) hfa
 123   let x := (infinityGradedIso cx p s).inv ((P.filtration X).toAssociatedGraded s p ar)
 124   let y := (infinityGradedIso cy p (s+n)).inv ((P.filtration Y).toAssociatedGraded (s+n) p br)
 125   have hx : (P.filtration X).toAssociatedGraded s p ar = (infinityGradedIso cx p s).hom x :=
 126     ((infinityGradedIso cx p s).toLinearEquiv.apply_symm_apply _).symm
 127   have hy : (P.filtration Y).toAssociatedGraded (s+n) p br = (infinityGradedIso cy p (s+n)).hom y :=
 128     ((infinityGradedIso cy p (s+n)).toLinearEquiv.apply_symm_apply _).symm
 129   have he : TowerExtensionEquation M p s n x y :=
 130     (tower_extension_iff M p s n x y).mpr ⟨hn, ar, br, by rw [har,hbr], hx,hy⟩
 131   have he0 : TowerExtensionEquation M p s n x 0 := by
 132     by_contra hz
 133     exact hnot x y ((tower_essential_iff_no_zero M p s n x y he).mpr hz)
 134   obtain ⟨_,ar',br',he',hx',hy'⟩ := (tower_extension_iff M p s n x 0).mp he0
 135   let b := ((P.filtration X).F s p).arrow ar'
 136   refine ⟨b, ?_, ?_, ?_⟩
 137   · have hh : b ∈ ModuleCat.subobjectModule _ ((P.filtration X).F s p) := ⟨ar',rfl⟩
 138     simpa only [SyntheticTowerPresentation.filtration,OrderIso.apply_symm_apply] using hh
 139   · have hh := (element_grade_equal_iff (P.filtration X) s p ar ar').mp (hx.trans hx'.symm)
 140     rw [har] at hh
 141     simpa only [b,SyntheticTowerPresentation.filtration,OrderIso.apply_symm_apply] using hh
 142   · have hh := (element_grade_equal_iff (P.filtration Y) (s+n) p br' 0).mp (by
 143       rw [hy']; simp only [map_zero])
 144     have hz : ((P.filtration Y).F (s+n) p).arrow (0 : (Subobject.underlying.obj ((P.filtration Y).F (s+n) p) : ModuleCat.{v} ℤ)) = 0 := map_zero _
 145     rw [hz,sub_zero,←he'] at hh
 146     simpa only [b,SyntheticTowerPresentation.filtration,OrderIso.apply_symm_apply] using hh

 148 /-- Absence of essential higher-source extensions on the finite triangle
 149 forces all choices of a source representative to have the same target grade. -/
 150 private theorem tower_image_same_target_of_no_crossing
 151     (M : ConvergentMap E g cx.toConvergence cy.toConvergence)
 152     (p : ℤ × ℤ) (s n : ℤ) (hn : 0 ≤ n)
 153     (hnot : ∀ u v : ℤ, s < u → u ≤ v → v ≤ s+n →
 154       ∀ x y, ¬TowerEssentialExtension M p u (v-u) x y)
 155     (a a₀ : BiHom p.1 p.2 X)
 156     (ha : a-a₀ ∈ P.filtrationSubmodule X (s+1) p) :
 157     a≫g - a₀≫g ∈ P.filtrationSubmodule Y (s+n+1) p := by
 158   let F : ℤ → AddSubgroup (BiHom p.1 p.2 X) := fun s => (P.filtrationSubmodule X s p).toAddSubgroup
 159   let G : ℤ → AddSubgroup (BiHom p.1 p.2 Y) := fun s => (P.filtrationSubmodule Y s p).toAddSubgroup
 160   let f : BiHom p.1 p.2 X →+ BiHom p.1 p.2 Y := (syntheticHomotopyMap g p).hom.toAddMonoidHom
 161   apply image_same_target_of_no_crossing f F G
 162     (fun i j hij a ha => P.filtrationSubmodule_antitone Y p hij ha)
 163     (fun u a ha => map_preserves_filtration M p u a ha) s n hn ?_ a a₀ ha
 164   intro u v hsu huv hvs a ha hfa
 165   have huv' : u + (v-u) = v := by omega
 166   change a ≫ g ∈ P.filtrationSubmodule Y v p at hfa
 167   change ∃ b, b ∈ P.filtrationSubmodule X u p ∧ a-b ∈ P.filtrationSubmodule X (u+1) p ∧ b ≫ g ∈ P.filtrationSubmodule Y (v+1) p
 168   have hh := tower_correction_of_nonessential M p u (v-u) (by omega)
 169     (hnot u v hsu huv hvs) a ha (by simpa only [huv', f, G, syntheticHomotopyMap] using hfa)
 170   simpa only [huv',f,F,G,syntheticHomotopyMap] using hh

 172 /-- Every representative of a fixed source grade has the prescribed target
 173 when the extension has no essential crossing. -/
 174 private theorem tower_target_of_any_source_frame
 175     (M : ConvergentMap E g cx.toConvergence cy.toConvergence)
 176     (p : ℤ × ℤ) (s n : ℤ)
 177     (x : ((E.obj X).sequence.ssData (s,p.1+s,p.2)).eInfty)
 178     (y : ((E.obj Y).sequence.ssData (s+n,p.1+(s+n),p.2)).eInfty)
 179     (he : TowerExtensionEquation M p s n x y)
 180     (hnot : ∀ u v : ℤ, s < u → u ≤ v → v ≤ s+n →
 181       ∀ x y, ¬TowerEssentialExtension M p u (v-u) x y)
 182     (ar : (Subobject.underlying.obj ((P.filtration X).F s p) : ModuleCat.{v} ℤ))
 183     (har : (P.filtration X).toAssociatedGraded s p ar = (infinityGradedIso cx p s).hom x) :
 184     ∃ br : (Subobject.underlying.obj ((P.filtration Y).F (s+n) p) : ModuleCat.{v} ℤ),
 185       ((P.filtration X).F s p).arrow ar ≫ g = ((P.filtration Y).F (s+n) p).arrow br ∧
 186       (P.filtration Y).toAssociatedGraded (s+n) p br = (infinityGradedIso cy p (s+n)).hom y := by
 187   obtain ⟨hn,ar₀,br₀,he₀,hx₀,hy₀⟩ := (tower_extension_iff M p s n x y).mp he
 188   have hsource := (element_grade_equal_iff (P.filtration X) s p ar ar₀).mp (har.trans hx₀.symm)
 189   have hsource' : ((P.filtration X).F s p).arrow ar -
 190       ((P.filtration X).F s p).arrow ar₀ ∈ P.filtrationSubmodule X (s+1) p := by
 191     simpa only [SyntheticTowerPresentation.filtration,OrderIso.apply_symm_apply] using hsource
 192   have hdiff := tower_image_same_target_of_no_crossing M p s n hn hnot _ _ hsource'
 193   rw [he₀] at hdiff
 194   have hb₀ : ((P.filtration Y).F (s+n) p).arrow br₀ ∈ P.filtrationSubmodule Y (s+n) p := by
 195     have hh : ((P.filtration Y).F (s+n) p).arrow br₀ ∈
 196         ModuleCat.subobjectModule _ ((P.filtration Y).F (s+n) p) := ⟨br₀,rfl⟩
 197     simpa only [SyntheticTowerPresentation.filtration,OrderIso.apply_symm_apply] using hh
 198   have hb : ((P.filtration X).F s p).arrow ar ≫ g ∈ P.filtrationSubmodule Y (s+n) p := by
 199     have hh := (P.filtrationSubmodule Y (s+n) p).add_mem
 200       (P.filtrationSubmodule_antitone Y p (show s+n≤s+n+1 by omega) hdiff) hb₀
 201     simpa only [sub_add_cancel] using hh
 202   obtain ⟨br,hbr⟩ := filtration_representative_exists P Y p (s+n) _ hb
 203   refine ⟨br,hbr.symm,?_⟩
 204   have hdiff' : ((P.filtration Y).F (s+n) p).arrow br -
 205       ((P.filtration Y).F (s+n) p).arrow br₀ ∈
 206         ModuleCat.subobjectModule _ ((P.filtration Y).F (s+n+1) p) := by
 207     rw [hbr]
 208     simpa only [SyntheticTowerPresentation.filtration,OrderIso.apply_symm_apply] using hdiff
 209   exact ((element_grade_equal_iff (P.filtration Y) (s+n) p br br₀).mpr hdiff').trans hy₀

 219 /-- A correction with the same source grade and a higher target filtration
 220 excludes every essential extension in the specified crossing triangle. -/
 221 private theorem tower_no_crossing_of_correction
 222     (M : ConvergentMap E g cx.toConvergence cy.toConvergence)
 223     (p : ℤ × ℤ) (s n : ℤ)
 224     (hcorrect : ∀ u v : ℤ, s < u → u ≤ v → v ≤ s+n →
 225       ∀ a : BiHom p.1 p.2 X, a ∈ P.filtrationSubmodule X u p →
 226         a ≫ g ∈ P.filtrationSubmodule Y v p →
 227         ∃ b, b ∈ P.filtrationSubmodule X u p ∧
 228           a-b ∈ P.filtrationSubmodule X (u+1) p ∧
 229           b ≫ g ∈ P.filtrationSubmodule Y (v+1) p) :
 230     TowerNoCrossing M p s n := by
 231   intro u v hsu huv hvs x y hess
 232   have he : TowerExtensionEquation M p u (v-u) x y := hess.1
 233   apply (tower_essential_iff_no_zero M p u (v-u) x y he).mp hess
 234   obtain ⟨hn,ar,br,heq,hx,hy⟩ := (tower_extension_iff M p u (v-u) x y).mp he
 235   let a := ((P.filtration X).F u p).arrow ar
 236   have ha : a ∈ P.filtrationSubmodule X u p := by
 237     have hm : a ∈ ModuleCat.subobjectModule _ ((P.filtration X).F u p) := ⟨ar,rfl⟩
 238     simpa only [SyntheticTowerPresentation.filtration,OrderIso.apply_symm_apply] using hm
 239   have hfa : a ≫ g ∈ P.filtrationSubmodule Y (u+(v-u)) p := by
 240     have hm : a ≫ g ∈ ModuleCat.subobjectModule _ ((P.filtration Y).F (u+(v-u)) p) :=
 241       ⟨br,heq.symm⟩
 242     simpa only [SyntheticTowerPresentation.filtration,OrderIso.apply_symm_apply] using hm
 243   obtain ⟨b,hb,hab,hbg⟩ := hcorrect u (u+(v-u)) hsu (by omega) (by omega) a ha hfa
 244   obtain ⟨ar',har'⟩ := filtration_representative_exists P X p u b hb
 245   have hbg' := P.filtrationSubmodule_antitone Y p (show u+(v-u)≤u+(v-u)+1 by omega) hbg
 246   obtain ⟨br',hbr'⟩ := filtration_representative_exists P Y p (u+(v-u)) (b≫g) hbg'
 247   apply (tower_extension_iff M p u (v-u) x 0).mpr
 248   refine ⟨hn,ar',br',by rw [har',hbr'],?_,?_⟩
 249   · have hdiff : ((P.filtration X).F u p).arrow ar -
 250         ((P.filtration X).F u p).arrow ar' ∈
 251         ModuleCat.subobjectModule _ ((P.filtration X).F (u+1) p) := by
 252       rw [har']
 253       simpa only [a,SyntheticTowerPresentation.filtration,OrderIso.apply_symm_apply] using hab
 254     exact ((element_grade_equal_iff (P.filtration X) u p ar ar').mpr hdiff).symm.trans hx
 255   · rw [map_zero]
 256     have hdiff : ((P.filtration Y).F (u+(v-u)) p).arrow br' -
 257         ((P.filtration Y).F (u+(v-u)) p).arrow
 258           (0 : (Subobject.underlying.obj ((P.filtration Y).F (u+(v-u)) p) : ModuleCat.{v} ℤ)) ∈
 259         ModuleCat.subobjectModule _ ((P.filtration Y).F (u+(v-u)+1) p) := by
 260       rw [hbr',map_zero,sub_zero]
 261       simpa only [SyntheticTowerPresentation.filtration,OrderIso.apply_symm_apply] using hbg
 262     have hgrade := (element_grade_equal_iff (P.filtration Y) (u+(v-u)) p br' 0).mpr hdiff
 263     simpa only [map_zero] using hgrade

 265 private theorem square_frames
 266     {Z W : Syn}
 267     {cz : SyntheticTowerConvergence P Z} {cw : SyntheticTowerConvergence P W}
 268     {f : X ⟶ Y} {h : X ⟶ Z} {j : Z ⟶ W} {k : Y ⟶ W}
 269     (Mf : ConvergentMap E f cx.toConvergence cy.toConvergence)
 270     (Mh : ConvergentMap E h cx.toConvergence cz.toConvergence)
 271     (Mj : ConvergentMap E j cz.toConvergence cw.toConvergence)
 272     (hcomm : f ≫ k = h ≫ j)
 273     (p : ℤ × ℤ) (s n m l : ℤ)
 274     (x : ((E.obj X).sequence.ssData (s,p.1+s,p.2)).eInfty)
 275     (y : ((E.obj Y).sequence.ssData (s+n,p.1+(s+n),p.2)).eInfty)
 276     (z : ((E.obj Z).sequence.ssData (s+m,p.1+(s+m),p.2)).eInfty)
 277     (w : ((E.obj W).sequence.ssData (s+m+l,p.1+(s+m+l),p.2)).eInfty)
 278     (hf : TowerExtensionEquation Mf p s n x y)
 279     (hh : TowerExtensionEquation Mh p s m x z)
 280     (hj : TowerExtensionEquation Mj p (s+m) l z w)
 281     (hnc : TowerNoCrossing Mf p s n ∨ TowerNoCrossing Mh p s m)
 282     (hncj : TowerNoCrossing Mj p (s+m) l) :
 283     ∃ (ar : (Subobject.underlying.obj ((P.filtration X).F s p) : ModuleCat.{v} ℤ))
 284       (br : (Subobject.underlying.obj ((P.filtration Y).F (s+n) p) : ModuleCat.{v} ℤ))
 285       (cr : (Subobject.underlying.obj ((P.filtration Z).F (s+m) p) : ModuleCat.{v} ℤ))
 286       (dr : (Subobject.underlying.obj ((P.filtration W).F (s+m+l) p) : ModuleCat.{v} ℤ)),
 287       (P.filtration X).toAssociatedGraded s p ar = (infinityGradedIso cx p s).hom x ∧
 288       (P.filtration Y).toAssociatedGraded (s+n) p br = (infinityGradedIso cy p (s+n)).hom y ∧
 289       (P.filtration Z).toAssociatedGraded (s+m) p cr = (infinityGradedIso cz p (s+m)).hom z ∧
 290       (P.filtration W).toAssociatedGraded (s+m+l) p dr = (infinityGradedIso cw p (s+m+l)).hom w ∧
 291       ((P.filtration X).F s p).arrow ar ≫ f = ((P.filtration Y).F (s+n) p).arrow br ∧
 292       ((P.filtration X).F s p).arrow ar ≫ h = ((P.filtration Z).F (s+m) p).arrow cr ∧
 293       ((P.filtration Z).F (s+m) p).arrow cr ≫ j = ((P.filtration W).F (s+m+l) p).arrow dr ∧
 294       ((P.filtration Y).F (s+n) p).arrow br ≫ k = ((P.filtration W).F (s+m+l) p).arrow dr := by
 295   have hthree : ∃ (ar : (Subobject.underlying.obj ((P.filtration X).F s p) : ModuleCat.{v} ℤ))
 296       (br : (Subobject.underlying.obj ((P.filtration Y).F (s+n) p) : ModuleCat.{v} ℤ))
 297       (cr : (Subobject.underlying.obj ((P.filtration Z).F (s+m) p) : ModuleCat.{v} ℤ)),
 298       (P.filtration X).toAssociatedGraded s p ar = (infinityGradedIso cx p s).hom x ∧
 299       (P.filtration Y).toAssociatedGraded (s+n) p br = (infinityGradedIso cy p (s+n)).hom y ∧
 300       (P.filtration Z).toAssociatedGraded (s+m) p cr = (infinityGradedIso cz p (s+m)).hom z ∧
 301       ((P.filtration X).F s p).arrow ar ≫ f = ((P.filtration Y).F (s+n) p).arrow br ∧
 302       ((P.filtration X).F s p).arrow ar ≫ h = ((P.filtration Z).F (s+m) p).arrow cr := by
 303     rcases hnc with hnf | hnh
 304     · obtain ⟨_,ar,cr,he,ha,hc⟩ := (tower_extension_iff Mh p s m x z).mp hh
 305       obtain ⟨br,he',hb⟩ := tower_target_of_any_source_frame Mf p s n x y hf hnf ar ha
 306       exact ⟨ar,br,cr,ha,hb,hc,he',he⟩
 307     · obtain ⟨_,ar,br,he,ha,hb⟩ := (tower_extension_iff Mf p s n x y).mp hf
 308       obtain ⟨cr,he',hc⟩ := tower_target_of_any_source_frame Mh p s m x z hh hnh ar ha
 309       exact ⟨ar,br,cr,ha,hb,hc,he,he'⟩
 310   obtain ⟨ar,br,cr,ha,hb,hc,haf,hah⟩ := hthree
 311   obtain ⟨dr,hcj,hd⟩ := tower_target_of_any_source_frame Mj p (s+m) l z w hj hncj cr hc

 315 private theorem tower_map_frame
 316     (M : ConvergentMap E g cx.toConvergence cy.toConvergence)
 317     (p : ℤ × ℤ) (s : ℤ)
 318     (x : ((E.obj X).sequence.ssData (s,p.1+s,p.2)).eInfty)
 319     (ar : (Subobject.underlying.obj ((P.filtration X).F s p) : ModuleCat.{v} ℤ))
 320     (ha : (P.filtration X).toAssociatedGraded s p ar = (infinityGradedIso cx p s).hom x) :
 321     ∃ br : (Subobject.underlying.obj ((P.filtration Y).F s p) : ModuleCat.{v} ℤ),
 322       ((P.filtration X).F s p).arrow ar ≫ g = ((P.filtration Y).F s p).arrow br ∧
 323       (P.filtration Y).toAssociatedGraded s p br = (infinityGradedIso cy p s).hom
 324         ((E.functor.map g).eInftyMap (s,p.1+s,p.2) x) := by
 325   let br := (M.filteredMap.compat s p).choose ar
 326   have hab : ((P.filtration X).F s p).arrow ar ≫ g = ((P.filtration Y).F s p).arrow br := by
 327     have h := ConcreteCategory.congr_hom (M.filteredMap.compat s p).choose_spec ar
 328     change ((P.filtration Y).F s p).arrow br = M.filteredMap.map p (((P.filtration X).F s p).arrow ar) at h
 329     rw [ConvergentMap.filteredMap_map] at h
 330     exact h.symm
 331   refine ⟨br,hab,?_⟩
 332   have h := representative_grade_naturality M p s ar br hab
 333   rw [ha] at h
 334   have hx : (infinityGradedIso cx p s).inv ((infinityGradedIso cx p s).hom x) = x :=
 335     (infinityGradedIso cx p s).toLinearEquiv.symm_apply_apply x
 336   rw [hx] at h

 341     {X' Y' : Syn}
 342     {cx' : SyntheticTowerConvergence P X'} {cy' : SyntheticTowerConvergence P Y'}
 343     {g' : X' ⟶ Y'} {a : X ⟶ X'} {b : Y ⟶ Y'}
 344     (M : ConvergentMap E g cx.toConvergence cy.toConvergence)

 346     (Ma : ConvergentMap E a cx.toConvergence cx'.toConvergence)
 347     (Mb : ConvergentMap E b cy.toConvergence cy'.toConvergence)
 348     (hcomm : g ≫ b = a ≫ g')
 349     (p : ℤ × ℤ) (s n : ℤ)
 350     (x : ((E.obj X).sequence.ssData (s,p.1+s,p.2)).eInfty)
 351     (y : ((E.obj Y).sequence.ssData (s+n,p.1+(s+n),p.2)).eInfty)
 352     (he : TowerExtensionEquation M p s n x y) :
 353     TowerExtensionEquation M' p s n
 354       ((E.functor.map a).eInftyMap (s,p.1+s,p.2) x)
 355       ((E.functor.map b).eInftyMap (s+n,p.1+(s+n),p.2) y) := by

 358   obtain ⟨br',hbb,hb'⟩ := tower_map_frame Mb p (s+n) y br hb
 359   apply (tower_extension_iff M' p s n _ _).mpr
 360   refine ⟨hn,ar',br',?_,ha',hb'⟩
 361   rw [←haa,Category.assoc,←hcomm,←Category.assoc,hab,hbb]
