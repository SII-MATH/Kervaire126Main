   4 /-! The actual absolute-boundary square for AimPaper Theorem 6.1.
   5 The square is induced by quotient naturality and the fixed suspension
   6 comparison; its commutativity does not assume any crossing conclusion. -/
   7 
   8 @[expose] section
   9 namespace KIP126.ProofTools.Internal.Leibniz
  10 open CategoryTheory KIP126.StableHomotopy KIP126.Synthetic.Context
  11 open KIP126.AdamsSynthetic
  12 noncomputable section
  13 set_option backward.isDefEq.respectTransparency false
  14 
  15 private def absoluteBoundary {Syn : Type*} [SyntheticCategory Syn]
  16     [HasFunctorialCofiber (C := Syn)] (A : Syn) (q : ℕ) :
  17     XModLambdaN A q ⟶ (SyntheticCategory.biShift (1,-(q:ℤ))).obj A :=
  18   XModLambdaN.proj A q ≫ (Mahowald.biSuspensionIso q A).inv
  19 
  20 private theorem absoluteBoundary_natural {Syn : Type*} [SyntheticCategory Syn]
  21     [HasFunctorialCofiber (C := Syn)] {A B : Syn} (f : A ⟶ B) (q : ℕ) :
  22     XModLambdaN.map f q ≫ absoluteBoundary B q =
  23       absoluteBoundary A q ≫ (SyntheticCategory.biShift (1,-(q:ℤ))).map f := by
  24   apply (cancel_mono (Mahowald.biSuspensionIso q B).hom).mp
  25   simp only [absoluteBoundary,Category.assoc,Iso.inv_hom_id,Category.comp_id]
  26   rw [Mahowald.biSuspensionIso_natural]
  27   simpa only [Category.assoc,Iso.inv_hom_id_assoc] using XModLambdaN.proj_naturality f q
  28 
  29 end
  30 end KIP126.ProofTools.Internal.Leibniz
