   5 /-! The actual absolute boundaries before and after finite lambda division
   6 have equal first-quotient labels, as used in AimPaper Remark `rmk:qvoewfj`.
   7 The chosen exact shift and canonical bigraded suspension may differ. Their
   8 comparison is natural on representable homotopy groups, hence is induced by
   9 a sphere automorphism. Such automorphisms act trivially after Q1. -/

  23 private def homPostcompAddEquiv {A B C : SyntheticSpectrum} (e : B ≅ C) :
  24     (A ⟶ B) ≃+ (A ⟶ C) where
  25   toFun a := a ≫ e.hom
  26   invFun a := a ≫ e.inv
  27   left_inv a := by simp
  28   right_inv a := by simp
  29   map_add' a b := by simp only [Preadditive.add_comp]

  31 private def boundaryDivisionSuspIso (d q : ℕ) (hdq : d ≤ q) (A : SyntheticSpectrum) :
  32     (SyntheticCategory.biShift (0,-(d:ℤ))).obj
  33       (((SyntheticCategory.biShift (0,-((q-d:ℕ):ℤ))).obj A)⟦(1:ℤ)⟧) ≅
  34     ((SyntheticCategory.biShift (0,-(q:ℤ))).obj A)⟦(1:ℤ)⟧ := by
  35   let F := SyntheticCategory.biShift (Syn:=SyntheticSpectrum) (0,-(d:ℤ))
  36   letI := model.exactness.shiftCommShift (0,-(d:ℤ))
  37   exact (F.commShiftIso (1:ℤ)).app _ ≪≫
  38     (shiftFunctor SyntheticSpectrum (1:ℤ)).mapIso
  39       ((lambdaShiftAddIso (q-d) d q (Nat.sub_add_cancel hdq)).app A)

  41 private theorem boundaryDivisionSuspIso_natural (d q : ℕ) (hdq : d ≤ q)
  42     {A B : SyntheticSpectrum} (f : A ⟶ B) :
  43     (SyntheticCategory.biShift (0,-(d:ℤ))).map
  44         ((shiftFunctor SyntheticSpectrum (1:ℤ)).map
  45           ((SyntheticCategory.biShift (0,-((q-d:ℕ):ℤ))).map f)) ≫
  46       (boundaryDivisionSuspIso d q hdq B).hom =
  47     (boundaryDivisionSuspIso d q hdq A).hom ≫
  48       (shiftFunctor SyntheticSpectrum (1:ℤ)).map
  49         ((SyntheticCategory.biShift (0,-(q:ℤ))).map f) := by
  50   letI := model.exactness.shiftCommShift (0,-(d:ℤ))
  51   simp only [boundaryDivisionSuspIso,Iso.trans_hom,Functor.mapIso_hom,Iso.app_hom]
  52   rw [←Category.assoc,Functor.commShiftIso_hom_naturality]
  53   simp only [Category.assoc,←Functor.map_comp]
  54   congr 1
  55   exact congrArg ((shiftFunctor SyntheticSpectrum (1:ℤ)).map)
  56     ((lambdaShiftAddIso (q-d) d q (Nat.sub_add_cancel hdq)).hom.naturality f)

  58 private def boundaryDivisionReindex (d q : ℕ) (hdq : d ≤ q) (m t : ℤ)
  59     (A : SyntheticSpectrum) :
  60     BiHom (m-1) (t+(q-d:ℕ)) A ≃+ BiHom (m-1) (t+(q-d:ℕ)) A :=
  61   (biSuspensionHomEquiv m t ((q-d:ℕ):ℤ) A).symm |>.trans
  62     (regradeAddEquiv rfl (by omega)) |>.trans
  63     (verticalShiftEquiv _ d m (t-d)) |>.trans
  64     (homPostcompAddEquiv (boundaryDivisionSuspIso d q hdq A)) |>.trans
  65     (biSuspensionHomEquiv m (t-d) q A) |>.trans
  66     (regradeAddEquiv rfl (by omega))

  68 private theorem homotopyRegrade_comp {A B : SyntheticSpectrum}
  69     {m m' w w' : ℤ} (hm : m=m') (hw : w=w') (a : BiHom m w A) (f : A ⟶ B) :
  70     homotopyRegrade hm hw (a≫f) = homotopyRegrade hm hw a ≫ f := by
  71   subst m'; subst w'; rfl

  73 private theorem bi_susp_symm_natural (m w i : ℤ) {A B : SyntheticSpectrum}
  74     (f : A ⟶ B) (a : BiHom (m-1) (w+i) A) :

  76       (biSuspensionHomEquiv m w i A).symm a ≫

  78           ((SyntheticCategory.biShift (0,-i)).map f) := by

  83 private theorem boundaryDivisionReindex_natural (d q : ℕ) (hdq : d ≤ q) (m t : ℤ)
  84     {A B : SyntheticSpectrum} (f : A ⟶ B)
  85     (a : BiHom (m-1) (t+(q-d:ℕ)) A) :
  86     boundaryDivisionReindex d q hdq m t B (a≫f) =
  87       boundaryDivisionReindex d q hdq m t A a ≫ f := by
  88   dsimp only [boundaryDivisionReindex,AddEquiv.trans_apply,regradeAddEquiv,
  89     homPostcompAddEquiv,AddEquiv.coe_mk,Equiv.coe_fn_mk]
  90   rw [bi_susp_symm_natural,homotopyRegrade_comp,←vertical_shift_comp,Category.assoc,
  91     boundaryDivisionSuspIso_natural,←Category.assoc,
  92     bi_susp_natural model.quotients.coherence,homotopyRegrade_comp]

  94 private def boundaryDivisionSourceIso (d q : ℕ) (hdq : d ≤ q) (m t : ℤ) :
  95     Smn (m-1) (t+(q-d:ℕ)) ≅ (Smn (m-1) (t+(q-d:ℕ)) : SyntheticSpectrum) where
  96   hom := boundaryDivisionReindex d q hdq m t _ (𝟙 _)
  97   inv := (boundaryDivisionReindex d q hdq m t _).symm (𝟙 _)
  98   hom_inv_id := by
  99     rw [←boundaryDivisionReindex_natural,Category.id_comp,AddEquiv.apply_symm_apply]
 100   inv_hom_id := by
 101     apply (boundaryDivisionReindex d q hdq m t _).injective
 102     rw [boundaryDivisionReindex_natural,AddEquiv.apply_symm_apply,Category.id_comp]

 104 private theorem boundaryDivisionReindex_eq_precomp (d q : ℕ) (hdq : d ≤ q) (m t : ℤ)
 105     (A : SyntheticSpectrum) (a : BiHom (m-1) (t+(q-d:ℕ)) A) :
 106     boundaryDivisionReindex d q hdq m t A a =
 107       (boundaryDivisionSourceIso d q hdq m t).hom ≫ a := by
 108   simpa only [Category.id_comp,boundaryDivisionSourceIso] using
 109     boundaryDivisionReindex_natural d q hdq m t a (𝟙 _)

 111 private theorem divided_boundary_eq_reindex (X : ClassicalObject)
 112     (d q : ℕ) (hd : 0 < d) (hdq : d < q) (m t : ℤ)
 113     (b0 : BiHom m t (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) (q-d))) :
 114     let A := model.nu.functor.obj (X.obj model.auxiliary)
 115     let b0' : BiHom m ((t-d)+d) (XModLambdaN A (q-d)) :=
 116       homotopyRegrade rfl (by omega) b0
 117     let a := verticalShiftEquiv (XModLambdaN A (q-d)) d m (t-d) b0' ≫
 118       ((model.quotients.tower A).triangle hd hdq).lambdaMap
 119     homotopyRegrade rfl (by omega)
 120       (biSuspensionHomEquiv m (t-d) q A (a ≫ XModLambdaN.proj A q)) =
 121     boundaryDivisionReindex d q hdq.le m t A
 122       (biSuspensionHomEquiv m t ((q-d:ℕ):ℤ) A (b0 ≫ XModLambdaN.proj A (q-d))) := by
 123   dsimp only
 124   letI := model.exactness.shiftCommShift (0,-(d:ℤ))
 125   dsimp only [boundaryDivisionReindex,AddEquiv.trans_apply,regradeAddEquiv,
 126     homPostcompAddEquiv,AddEquiv.coe_mk,Equiv.coe_fn_mk]
 127   rw [AddEquiv.symm_apply_apply,homotopyRegrade_comp,←vertical_shift_comp,
 128     Category.assoc,Category.assoc,lambda_proj_evaluation X d q hd hdq]
 129   rfl

 133     (b0 : BiHom m t (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) (q-d))) :

 136       homotopyRegrade rfl (by omega) b0
