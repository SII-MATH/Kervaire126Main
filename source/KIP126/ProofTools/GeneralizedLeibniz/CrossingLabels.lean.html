  17 private theorem subquotient_rep {R A : Type*} [Ring R] [AddCommGroup A] [Module R A]
  18     (Z B : Submodule R A) (e : E2Subquotient Z B) :
  19     ∃ y : Z, subquotientClass Z B y = e :=
  20   Submodule.Quotient.mk_surjective _ e

  22 private theorem subquotient_zero_iff {R A : Type*} [Ring R] [AddCommGroup A] [Module R A]
  23     (Z B : Submodule R A) (y : Z) : subquotientClass Z B y = 0 ↔ y.val ∈ B :=
  24   Submodule.Quotient.mk_eq_zero _

  26 private theorem nu_grade_label (X : ClassicalObject) (s t w : ℤ) (hw : w ≤ t)
  27     (e : ((model.family.nu model.nu (X.obj model.auxiliary)).sequence.ssData (s,t,w)).eInfty) :
  28     ∃ y : Ambient model X (s,t), y ∈ permanentCycles model X (s,t) ∧
  29       HasInfinityRepresentative _ 2 (s,t,w) (nuWeightLabel model X s t w hw y) e ∧
  30       (y ∈ boundaries model X (1+t-w) (s,t) ↔ e = 0) := by
  31   obtain ⟨y,hy⟩ := subquotient_rep (permanentCycles model X (s,t)) (boundaries model X (1+t-w) (s,t))
  32     (KIP126.Literature.Standard.infinityNuWindow X (s,t) w hw e)
  33   have hrep : HasInfinityRepresentative _ 2 (s,t,w)
  34       (nuWeightLabel model X s t w hw y.val) e := by
  35     apply (KIP126.Literature.Standard.eInfinity_nu_labels X s t w hw y.val e).mpr
  36     exact ⟨y.property,hy.symm⟩
  37   refine ⟨y.val,y.property,hrep,?_⟩
  38   have hz : y.val ∈ boundaries model X (1+t-w) (s,t) ↔
  39       KIP126.Literature.Standard.infinityNuWindow X (s,t) w hw e = 0 := by
  40     rw [←hy]
  41     exact (subquotient_zero_iff _ _ y).symm
  42   exact hz.trans (map_eq_zero_iff _ (KIP126.Literature.Standard.infinityNuWindow X (s,t) w hw).injective)

  44 private theorem finite_grade_label (X : ClassicalObject) (q : ℕ) (hq : 0 < q)
  45     (s t w : ℤ) (hw : 0 ≤ t-w ∧ t-w < q)
  46     (e : ((model.family.quotient (model.nu.functor.obj (X.obj model.auxiliary)) q).sequence.ssData (s,t,w)).eInfty) :
  47     ∃ y : Ambient model X (s,t), y ∈ cycles model X (q-t+w) (s,t) ∧
  48       HasInfinityRepresentative _ 2 (s,t,w)
  49         ((model.family.quotientProjection (model.nu.functor.obj (X.obj model.auxiliary)) q).pageMap 2 (s,t,w)
  50           (nuWeightLabel model X s t w (by omega) y)) e ∧
  51       (y ∈ boundaries model X (1+t-w) (s,t) ↔ e = 0) := by
  52   obtain ⟨y,hy⟩ := subquotient_rep (cycles model X (q-t+w) (s,t)) (boundaries model X (1+t-w) (s,t))
  53     (KIP126.Literature.Standard.infinityFiniteWindow X q hq (s,t) w hw e)
  54   have hrep : HasInfinityRepresentative _ 2 (s,t,w)
  55       ((model.family.quotientProjection (model.nu.functor.obj (X.obj model.auxiliary)) q).pageMap 2 (s,t,w)
  56         (nuWeightLabel model X s t w (by omega) y.val)) e := by
  57     apply (KIP126.Literature.Standard.eInfinity_finite_labels X q hq s t w hw y.val e).mpr
  58     exact ⟨y.property,hy.symm⟩
  59   refine ⟨y.val,y.property,hrep,?_⟩
  60   have hz : y.val ∈ boundaries model X (1+t-w) (s,t) ↔
  61       KIP126.Literature.Standard.infinityFiniteWindow X q hq (s,t) w hw e = 0 := by
  62     rw [←hy]
  63     exact (subquotient_zero_iff _ _ y).symm
  64   exact hz.trans (map_eq_zero_iff _ (KIP126.Literature.Standard.infinityFiniteWindow X q hq (s,t) w hw).injective)

  66 private theorem lambda_nu_label_nonzero (X : ClassicalObject) (k : ℕ)
  67     (s t w : ℤ) (hw : w ≤ t) (y : Ambient model X (s,t))
  68     (e : ((model.family.nu model.nu (X.obj model.auxiliary)).sequence.ssData (s,t,w)).eInfty)
  69     (he : HasInfinityRepresentative _ 2 (s,t,w) (nuWeightLabel model X s t w hw y) e) :
  70     KIP126.Literature.Standard.infinityLambda (.nu X) k (s,t) w e ≠ 0 ↔
  71       y ∉ boundaries model X (1+t-(w-k)) (s,t) := by
  72   obtain ⟨hy,hlabel⟩ := (KIP126.Literature.Standard.eInfinity_nu_labels X s t w hw y e).mp he
  73   have hlam := KIP126.Literature.Standard.eInfinity_lambda_nu X k (s,t) w hw e
  74   rw [hlabel] at hlam
  75   change KIP126.Literature.Standard.infinityNuWindow X (s,t) (w-k) (by omega)
  76       (KIP126.Literature.Standard.infinityLambda (.nu X) k (s,t) w e) =
  77     subquotientClass (permanentCycles model X (s,t))
  78       (boundaries model X (1+t-(w-k)) (s,t)) ⟨y,hy⟩ at hlam
  79   apply not_congr
  80   constructor
  81   · intro hzero
  82     rw [hzero,map_zero] at hlam
  83     exact (subquotient_zero_iff _ _ ⟨y,hy⟩).mp hlam.symm
  84   · intro hboundary
  85     apply (KIP126.Literature.Standard.infinityNuWindow X (s,t) (w-k) (by omega)).injective
  86     rw [map_zero,hlam]
  87     exact (subquotient_zero_iff _ _ ⟨y,hy⟩).mpr hboundary

  89 private theorem subquotient_map_class_zero_iff {R A : Type*} [Ring R] [AddCommGroup A] [Module R A]
  90     {Z Z' B B' : Submodule R A} (hZ : Z ≤ Z') (hB : B ≤ B') (y : Z) :
  91     subquotientMap hZ hB (subquotientClass Z B y) = 0 ↔ y.val ∈ B' :=
  92   subquotient_zero_iff Z' B' (Submodule.inclusion hZ y)

  94 private theorem lambda_finite_label_nonzero (X : ClassicalObject) (q k : ℕ) (hkq : k < q)
  95     (s t w : ℤ) (hw : 0 ≤ t-w ∧ t-w < (q-k:ℕ)) (y : Ambient model X (s,t))
  96     (e : ((model.family.quotient (model.nu.functor.obj (X.obj model.auxiliary)) (q-k)).sequence.ssData (s,t,w)).eInfty)
  97     (he : HasInfinityRepresentative _ 2 (s,t,w)
  98       ((model.family.quotientProjection (model.nu.functor.obj (X.obj model.auxiliary)) (q-k)).pageMap 2 (s,t,w)
  99         (nuWeightLabel model X s t w (by omega) y)) e) :
 100     KIP126.Literature.Standard.infinityFiniteLambda X q k hkq (s,t) w e ≠ 0 ↔
 101       y ∉ boundaries model X (1+t-(w-k)) (s,t) := by
 102   obtain ⟨hy,hlabel⟩ := (KIP126.Literature.Standard.eInfinity_finite_labels X (q-k) (by omega)
 103     s t w hw y e).mp he
 104   have hlam := KIP126.Literature.Standard.eInfinity_lambda_finite X q k hkq (s,t) w hw e
 105   rw [hlabel] at hlam
 106   have hz := subquotient_map_class_zero_iff
 107     (cycles_antitone model X (s,t) (show (q:ℤ)-t+(w-k) ≤ (q-k:ℕ)-t+w by omega))
 108     (boundaries_monotone model X (s,t) (show 1+t-w ≤ 1+t-(w-k) by omega)) ⟨y,hy⟩
 109   apply not_congr
 110   constructor
 111   · intro hzero
 112     rw [hzero,map_zero] at hlam
 113     exact hz.mp hlam.symm
 114   · intro hboundary
 115     apply (KIP126.Literature.Standard.infinityFiniteWindow X q (by omega) (s,t) (w-k) (by omega)).injective
 116     rw [map_zero,hlam]
 117     exact hz.mpr hboundary
