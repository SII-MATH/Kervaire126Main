   7 `prop:cross-dr-En`. The source is divided in the actual finite quotient,

  10 has higher filtration. -/

  26     FiltrationAtLeast model.syntheticTower s (homotopyRegrade hm hw a) ↔
  27       FiltrationAtLeast model.syntheticTower s a := by

  30 private theorem absolute_boundary_filtration_iff (X : ClassicalObject)
  31     (q : ℕ) (m w v : ℤ)
  32     (a : BiHom m w (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) q)) :
  33     FiltrationAtLeast model.syntheticTower v
  34         (a ≫ absoluteBoundary (model.nu.functor.obj (X.obj model.auxiliary)) q) ↔
  35       FiltrationAtLeast model.syntheticTower v
  36         (biSuspensionHomEquiv m w q (model.nu.functor.obj (X.obj model.auxiliary))
  37           (a ≫ XModLambdaN.proj _ q)) := by
  38   let A := model.nu.functor.obj (X.obj model.auxiliary)
  39   let α := biSuspensionHomEquiv m w q A (a ≫ XModLambdaN.proj A q)
  40   have he : a ≫ absoluteBoundary A q =
  41       homotopyRegrade (sub_add_cancel m 1) (add_neg_cancel_right w (q:ℤ))
  42         (shiftHomotopyClass (1,-(q:ℤ)) (m-1,w+q) α) := by
  43     have h := congrArg (fun z => z ≫ (biSuspensionIso q A).inv)
  44       (bi_susp_symm_eq m w q A α)
  45     simpa [α, absoluteBoundary, shiftHomotopyClass, susp_invariance,
  46       Iso.homCongr] using h
  47   rw [he,regrade_filtration_iff]
  48   exact shift_filtration_iff (.nu X) (1,-(q:ℤ)) (m-1,w+q) v α

  50 private theorem quotient_first_label_exists (X : ClassicalObject)
  51     (q : ℕ) (hq : 0 < q) (s t : ℤ)
  52     (a : BiHom (t-s) t (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) q)) :
  53     ∃ x : Ambient model X (s,t),
  54       a ≫ (model.quotients.tower _).rho 1 q hq =
  55         firstLabel model X s t x ≫
  56           XModLambdaN.map (SyntheticCategory.biShift_zero.hom.app _) 1 := by
  57   let A := model.nu.functor.obj (X.obj model.auxiliary)
  58   obtain ⟨x,hx⟩ := firstLabel_surjective X s t
  59     (a ≫ (model.quotients.tower A).rho 1 q hq ≫
  60       XModLambdaN.map (SyntheticCategory.biShift_zero.inv.app A) 1)
  61   refine ⟨x,?_⟩
  62   rw [hx,Category.assoc,Category.assoc,←model.quotients.functoriality.map_comp,
  63     Iso.inv_hom_id_app,model.quotients.functoriality.map_id]
  64   simp only [Functor.id_obj,Category.comp_id]
  65   rfl

  67 private theorem no_crossing_differential_zero (X : ClassicalObject)
  68     (r page s t u v : ℤ) (hnc : NoCrossingOn model X r page (s,t))
  69     (hu : s < u) (hu' : u-s ≤ page-2) (hv : s+page ≤ v) (hv' : v ≤ s+r)
  70     (x : Ambient model X (u,t+(u-s)))
  71     (y : Ambient model X (v,t+(v-s)-1))
  72     (hd : HasDifferential (model.classical.obj (X.obj model.auxiliary))
  73       (v-u) (u,t+(u-s)) (v,t+(v-s)-1) x y) :
  74     HasDifferential (model.classical.obj (X.obj model.auxiliary))
  75       (v-u) (u,t+(u-s)) (v,t+(v-s)-1) x 0 := by
  76   obtain ⟨hdeg,xr,yr,hx,hy,hd⟩ := hd
  77   have hz : yr = 0 := by
  78     by_contra hz
  79     apply hnc.2.2
  80     refine ⟨u-s,s+r-v,by omega,hu',by omega,by omega,?_⟩
  81     have he : HasNonzeroDifferential (model.classical.obj (X.obj model.auxiliary))
  82         (v-u) (u,t+(u-s)) (v,t+(v-s)-1) x y := ⟨hdeg,xr,yr,hx,hy,hd,hz⟩
  83     have h₁ : s+(u-s)=u := by omega
  84     have h₂ : s+r-(s+r-v)=v := by omega
  85     have h₃ : t+r-(s+r-v)-1=t+(v-s)-1 := by omega
  86     have h₄ : r-(u-s)-(s+r-v)=v-u := by omega
  87     change ∃ (x : Ambient model X (s+(u-s),t+(u-s)))
  88       (y : Ambient model X (s+r-(s+r-v),t+r-(s+r-v)-1)),
  89       HasNonzeroDifferential (model.classical.obj (X.obj model.auxiliary))
  90         (r-(u-s)-(s+r-v)) (s+(u-s),t+(u-s))
  91         (s+r-(s+r-v),t+r-(s+r-v)-1) x y
  92     rw [h₁,h₂,h₃,h₄]
  93     exact ⟨x,y,he⟩
  94   exact ⟨hdeg,xr,0,hx,RepresentsOnPage.zero (by omega),hd.trans hz⟩

  96 private theorem divided_boundary_correction (X : ClassicalObject)
  97     (r page s t u v : ℤ) (hnc : NoCrossingOn model X r page (s,t))
  98     (hu : s < u) (hu' : u-s ≤ page-2) (hv : s+page ≤ v) (hv' : v ≤ s+r)
  99     (d : ℕ) (hd : 0 < d) (hdv : (d:ℤ) ≤ v-u-1)
 100     (a : BiHom ((t+(u-s))-u) (t+(u-s))
 101       (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) d))
 102     (hfil : FiltrationAtLeast model.syntheticTower v
 103       (a ≫ absoluteBoundary (model.nu.functor.obj (X.obj model.auxiliary)) d)) :
 104     ∃ a0 : BiHom ((t+(u-s))-u) (t+(u-s))
 105         (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) d),
 106       FiltrationAtLeast model.syntheticTower (u+1) (a-a0) ∧
 107       FiltrationAtLeast model.syntheticTower (v+1)
 108         (a0 ≫ absoluteBoundary (model.nu.functor.obj (X.obj model.auxiliary)) d) := by
 109   let w := t+(u-s)
 110   let k : ℕ := (v-u-1).toNat
 111   have hk : (k:ℤ)=v-u-1 := by dsimp [k]; omega
 112   obtain ⟨x,hx⟩ := quotient_first_label_exists X d hd u w a
 113   have hsurv := absolute_boundary_filtration_cycles X d k hd (by omega) u w x a hx
 114     (by simpa only [show u+(k:ℤ)+1=v by omega] using
 115       (absolute_boundary_filtration_iff X d _ _ v a).mp hfil)
 116   have hreach := (isCycle_iff_reaches model X (k+1) (by omega) (u,w) x).mp
 117     ⟨by omega,by simpa only [add_sub_cancel_right] using hsurv⟩
 118   obtain ⟨b,hb,hdb⟩ := KIP126.Literature.Standard.bhs_bockstein X k (by omega) u w x hreach
 119   have hzero : HasDifferential (model.classical.obj (X.obj model.auxiliary))
 120       (k+1) (u,w) (u+k+1,w+k) x 0 := by
 121     have H := no_crossing_differential_zero X r page s t u (u+k+1) hnc
 122       hu hu' (by omega) (by omega) x
 123     have hweight : t+(u+(k:ℤ)+1-s)-1=w+k := by dsimp [w]; omega
 124     have hlength : u+(k:ℤ)+1-u=k+1 := by omega
 125     rw [hweight,hlength] at H
 126     exact H _ hdb
 127   obtain ⟨a0,ha0,hδ⟩ := differential_absolute_boundary X d k hd (by omega) u w x 0
 128     hzero (Submodule.zero_mem _)
 129   refine ⟨a0,?_,?_⟩
 130   · exact detects_sub_filtration (model.convergence (.quotient d (.nu X))) (u,w,w)
 131       (first_label_detects_unshifted X d hd u w x a hx)
 132       (first_label_detects_unshifted X d hd u w x a0 ha0)
 133   · apply (absolute_boundary_filtration_iff X d _ _ (v+1) a0).mpr
 134     have hz : nuWeightLabel model X (u+k+1) (w+k) (w+d) (by omega) 0 = 0 := by
 135       have cast_zero (E : SyntheticAdamsSS.{0}) (s t w w' : ℤ) (hw : w=w')
 136           (he : (E.E₂ (s,t,w) : Type) = (E.E₂ (s,t,w') : Type)) :
 137           he.mp (0 : E.E₂ (s,t,w)) = 0 := by subst w'; rfl
 138       simp only [nuWeightLabel,regradeE2,targetNuLabel,map_zero]
 139       rw [cast_zero _ _ _ _ _ (by omega)]
 140       simp only [map_zero]
 141     rw [hz] at hδ
 142     have H := detects_zero_filtration (model.convergence (.nu X)) (u+k+1,w+k,w+d) hδ
 143     have H := (regrade_filtration_iff _ _ _ _).mp H
 144     simpa only [show u+(k:ℤ)+1+1=v+1 by omega] using H

 146 private def regradeAddEquiv {A : SyntheticSpectrum} {m m' w w' : ℤ}
 147     (hm : m=m') (hw : w=w') : BiHom m w A ≃+ BiHom m' w' A where
 148   toFun := homotopyRegrade hm hw
 149   invFun := homotopyRegrade hm.symm hw.symm
 150   left_inv := by subst m'; subst w'; intro a; rfl
 151   right_inv := by subst m'; subst w'; intro a; rfl
 152   map_add' := by subst m'; subst w'; intro a b; rfl

 154 private def verticalShiftEquiv (A : SyntheticSpectrum) (i : ℕ) (m w : ℤ) :
 155     BiHom m (w+i) A ≃+
 156       BiHom m w ((SyntheticCategory.biShift (0,-(i:ℤ))).obj A) :=
 157   (suspInvarianceAddEquiv m (w+i) 0 (-(i:ℤ)) A).trans
 158     (regradeAddEquiv (add_zero m) (add_neg_cancel_right w (i:ℤ)))

 160 private theorem vertical_shift_filtration_iff (A : SyntheticObject)
 161     (i : ℕ) (m w v : ℤ) (a : BiHom m (w+i) (A.obj model.nu model.auxiliary)) :
 162     FiltrationAtLeast model.syntheticTower v (verticalShiftEquiv _ i m w a) ↔
 163       FiltrationAtLeast model.syntheticTower v a := by
 164   change FiltrationAtLeast model.syntheticTower v
 165     (homotopyRegrade _ _ (susp_invariance m (w+i) 0 (-(i:ℤ)) _ a)) ↔ _
 166   rw [regrade_filtration_iff]
 167   have he : susp_invariance m (w+i) 0 (-(i:ℤ)) _ a =
 168       shiftHomotopyClass (0,-(i:ℤ)) (m,w+i) a := by
 169     simp [susp_invariance,shiftHomotopyClass,Iso.homCongr]
 170   rw [he]
 171   exact shift_filtration_iff A (0,-(i:ℤ)) (m,w+i) v a

 180 private theorem boundary_crossing_correction (X : ClassicalObject)
 181     (q : ℕ) (r s t u v : ℤ) (hnc : NoCrossingOn model X r (q+1) (s,t))
 182     (hu : s < u) (hv' : v ≤ s+r)
 183     (a : BiHom (t-s) t (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) q))
 184     (ha : FiltrationAtLeast model.syntheticTower u a)
 185     (hδ : FiltrationAtLeast model.syntheticTower v
 186       (a ≫ absoluteBoundary (model.nu.functor.obj (X.obj model.auxiliary)) q)) :
 187     ∃ a0 : BiHom (t-s) t (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) q),
 188       FiltrationAtLeast model.syntheticTower u a0 ∧
 189       FiltrationAtLeast model.syntheticTower (u+1) (a-a0) ∧
 190       FiltrationAtLeast model.syntheticTower (v+1)
 191         (a0 ≫ absoluteBoundary (model.nu.functor.obj (X.obj model.auxiliary)) q) := by
 192   let A := model.nu.functor.obj (X.obj model.auxiliary)
 193   have hq : 0 < q := by have := hnc.1; omega
 194   by_cases hv : s+(q:ℤ)+1 ≤ v
 195   swap
 196   · refine ⟨a,ha,?_,?_⟩
 197     · simpa only [FiltrationAtLeast,sub_self] using
 198         (model.syntheticTower.filtrationSubmodule _ (u+1) (t-s,t)).zero_mem
 199     · apply (absolute_boundary_filtration_iff X q (t-s) t (v+1) a).mpr
 200       let α := biSuspensionHomEquiv (t-s) t q A (a ≫ XModLambdaN.proj A q)
 201       have hα := nu_bottom_filtration X (s+q+1) (t+q)
 202         (homotopyRegrade (by omega) rfl α)
 203       have hα := (regrade_filtration_iff _ _ _ _).mp hα
 204       exact model.syntheticTower.filtrationSubmodule_antitone _ _ (by omega) hα
 205   by_cases hui : (q:ℤ) ≤ u-s
 206   · have haz := finite_high_filtration_zero X q hq (t-s) t u (by omega) a ha
 207     subst a
 208     refine ⟨0,?_,?_,?_⟩ <;>
 209       simp only [FiltrationAtLeast,sub_self,zero_comp] <;> exact Submodule.zero_mem _
 210   let i : ℕ := (u-s).toNat
 211   have hiZ : (i:ℤ)=u-s := by dsimp [i]; omega
 212   have hi : 0 < i := by omega
 213   have hiq : i < q := by omega
 214   let T := (model.quotients.tower A).triangle hi hiq
 215   obtain ⟨bS,hbS⟩ := finite_source_division X i q hi hiq (t-s) t u (by omega) a ha
 216   let e := verticalShiftEquiv (XModLambdaN A (q-i)) i (t-s) t
 217   let b := e.symm bS
 218   have hb : e b = bS := e.apply_symm_apply bS
 219   have hfilb : FiltrationAtLeast model.syntheticTower v (b ≫ absoluteBoundary A (q-i)) := by
 220     have h := (divided_boundary_filtration_iff X i q hi hiq (t-s) t v bS).mp (by
 221       rw [←Category.assoc,hbS]
 222       exact hδ)
 223     rw [←hb,vertical_shift_comp] at h
 224     exact (vertical_shift_filtration_iff (.shift (1,-((q-i:ℕ):ℤ)) (.nu X))
 225       i (t-s) t v (b ≫ absoluteBoundary A (q-i))).mp h
 226   have H := divided_boundary_correction X r (q+1) s t u v hnc hu (by omega)
 227     (by omega) hv' (q-i) (by omega) (by omega)
 228   have hw : t+(i:ℤ)=t+(u-s) := by omega
 229   have hm : t-s=(t+(i:ℤ))-u := by omega
 230   rw [←hw,←hm] at H
 231   obtain ⟨b0,hb0,hδ0⟩ := H b hfilb
 232   let a0 := e b0 ≫ T.lambdaMap
 233   have hdiff : FiltrationAtLeast model.syntheticTower (u+1) (a-a0) := by
 234     have hshift := (vertical_shift_filtration_iff (.quotient (q-i) (.nu X))
 235       i (t-s) t (u+1) (b-b0)).mpr hb0
 236     have hmap := map_preserves_filtration (model.convergentMap
 237       (.shift (0,-(i:ℤ)) (.quotient (q-i) (.nu X))) (.quotient q (.nu X)) T.lambdaMap)
 238       (t-s,t) (u+1) _ hshift
 239     change FiltrationAtLeast model.syntheticTower (u+1) (e (b-b0) ≫ T.lambdaMap) at hmap
 240     rw [map_sub,Preadditive.sub_comp,hb] at hmap
 241     change FiltrationAtLeast model.syntheticTower (u+1) (bS ≫ T.lambdaMap - a0) at hmap
 242     rwa [hbS] at hmap
 243   refine ⟨a0,?_,hdiff,?_⟩
 244   · have hdiff' := model.syntheticTower.filtrationSubmodule_antitone _ (t-s,t)
 245       (show u ≤ u+1 by omega) hdiff
 246     have H := (model.syntheticTower.filtrationSubmodule _ u (t-s,t)).sub_mem ha hdiff'
 247     simpa only [FiltrationAtLeast,sub_sub_cancel] using H
 248   · dsimp only [a0]
 249     rw [Category.assoc]
 250     apply (divided_boundary_filtration_iff X i q hi hiq (t-s) t (v+1) (e b0)).mpr
 251     rw [vertical_shift_comp]
 252     exact (vertical_shift_filtration_iff (.shift (1,-((q-i:ℕ):ℤ)) (.nu X))
 253       i (t-s) t (v+1) (b0 ≫ absoluteBoundary A (q-i))).mpr hδ0

 255 private theorem classical_noCrossing_absolute_boundary (X : ClassicalObject)
 256     (page : ℕ) (r s t : ℤ) (hnc : NoCrossingOn model X r page (s,t)) :
 257     TowerNoCrossing (model.convergentMap
 258       (.quotient (page-1) (.nu X))
 259       (.shift (1,-((page-1:ℕ):ℤ)) (.nu X))
 260       (absoluteBoundary (model.nu.functor.obj (X.obj model.auxiliary)) (page-1)))
 261       (t-s,t) s r := by
 262   apply tower_no_crossing_of_correction
 263   intro u v hu _ hv a ha hδ
 264   have hnc' : NoCrossingOn model X r ((page-1:ℕ)+1) (s,t) := by
 265     convert hnc using 1
 266     have h := hnc.1
 267     omega
 268   exact boundary_crossing_correction X (page-1) r s t u v hnc' hu hv a ha hδ
