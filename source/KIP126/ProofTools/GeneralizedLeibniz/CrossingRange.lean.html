  17 private theorem finite_extension_zero_source_not_essential (X Y : ClassicalObject)
  18     (f : X.obj model.auxiliary ⟶ Y.obj model.auxiliary) (q : ℕ)
  19     (p : ℤ × ℤ) (u d : ℤ) (hd : 0 ≤ d)
  20     (y : ((model.family.obj ((SyntheticObject.quotient q (.nu Y)).obj model.nu model.auxiliary)).sequence.ssData
  21       (u+d,p.1+(u+d),p.2)).eInfty) :
  22     ¬ TowerEssentialExtension (model.convergentMap
  23       (.quotient q (extensionSourceObject model X Y f)) (.quotient q (.nu Y))
  24       (finiteExtensionMap model X Y f q)) p u d 0 y := by
  25   intro he
  26   have hrel : TowerExtensionEquation (model.convergentMap
  27       (.quotient q (extensionSourceObject model X Y f)) (.quotient q (.nu Y))
  28       (finiteExtensionMap model X Y f q)) p u d 0 y := he.1
  29   have hzero : TowerExtensionEquation (model.convergentMap
  30       (.quotient q (extensionSourceObject model X Y f)) (.quotient q (.nu Y))
  31       (finiteExtensionMap model X Y f q)) p u d 0 0 := by
  32     apply (tower_extension_iff _ _ _ _ _ _).mpr
  33     refine ⟨hd,0,0,?_,?_,?_⟩ <;> simp
  34   exact ((tower_essential_iff_no_zero _ _ _ _ _ _ hrel).mp he) hzero

  36 private theorem finite_crossing_source_bound (X Y : ClassicalObject)
  37     (f : X.obj model.auxiliary ⟶ Y.obj model.auxiliary) (q : ℕ) (hq : 0 < q)
  38     (s t u v : ℤ) (huv : u ≤ v)
  39     (x : ((model.family.obj ((SyntheticObject.quotient q (extensionSourceObject model X Y f)).obj model.nu model.auxiliary)).sequence.ssData
  40       (u,(t-s)+u,t+normalizedExponent model.classicalTower f)).eInfty)
  41     (y : ((model.family.obj ((SyntheticObject.quotient q (.nu Y)).obj model.nu model.auxiliary)).sequence.ssData
  42       (u+(v-u),(t-s)+(u+(v-u)),t+normalizedExponent model.classicalTower f)).eInfty)
  43     (he : TowerEssentialExtension (model.convergentMap
  44       (.quotient q (extensionSourceObject model X Y f)) (.quotient q (.nu Y))
  45       (finiteExtensionMap model X Y f q))
  46       (t-s,t+normalizedExponent model.classicalTower f) u (v-u) x y) : u-s < q := by
  47   by_contra hqle
  48   have hz := shifted_quotient_eInfty_zero X q hq (normalizedExponent model.classicalTower f)
  49     u ((t-s)+u) (t+normalizedExponent model.classicalTower f) (by omega)
  50   have hsub := ModuleCat.isZero_iff_subsingleton.mp hz
  51   have hx : x = 0 := @Subsingleton.elim _ hsub _ _
  52   subst x
  53   exact finite_extension_zero_source_not_essential X Y f q
  54     (t-s,t+normalizedExponent model.classicalTower f) u (v-u) (by omega) y he

  56 private theorem page_two_tower_no_crossing (X Y : ClassicalObject)
  57     (f : X.obj model.auxiliary ⟶ Y.obj model.auxiliary) (s t n : ℤ) :
  58     TowerNoCrossing (model.convergentMap
  59       (.quotient 1 (extensionSourceObject model X Y f)) (.quotient 1 (.nu Y))
  60       (finiteExtensionMap model X Y f 1))
  61       (t-s,t+normalizedExponent model.classicalTower f) s n := by
  62   intro u v hsu huv hvs x y he
  63   have hh := finite_crossing_source_bound X Y f 1 (by omega) s t u v huv x y he
  64   omega
  65 end
