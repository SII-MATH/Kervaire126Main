  21 private theorem normalizeInitialProjection
  22     (P : SSData (ModuleCat.{v} ℤ)) (n m : WithTop ℕ)
  23     (hn : n = 0) (hnm : n ≤ m) :
  24     Subobject.ofLE (P.Z m) (P.Z n) (P.Z_anti hnm) ≫ P.pageπ n =
  25       Subobject.ofLE (P.Z m) (P.Z 0) (P.Z_anti bot_le) ≫ P.pageπ 0 ≫
  26         eqToHom (congrArg P.page hn.symm) := by
  27   subst n
  28   simp only [eqToHom_refl, Category.comp_id]

  30 private theorem initialProjection_eq_cycleLabel (X : ClassicalObject) (r : ℤ)
  31     (hr : 2 ≤ r) (p : ℤ × ℤ) :
  32     let E := D.classical.obj (X.obj D.auxiliary)
  33     let P := E.ssData p
  34     let n : WithTop ℕ := ↑(2-E.r₀).toNat
  35     let m : WithTop ℕ := ↑(r-E.r₀).toNat
  36     Subobject.ofLE (P.Z m) (P.Z n)
  37       (P.Z_anti (by
  38         change (↑(2-E.r₀).toNat : WithTop ℕ) ≤ ↑(r-E.r₀).toNat
  39         exact_mod_cast (show (2-E.r₀).toNat ≤ (r-E.r₀).toNat by omega))) ≫
  40       P.pageπ n = KIP126.Literature.cycleLabel D X m p := by
  41   dsimp only
  42   exact normalizeInitialProjection _ _ _ (by rw [D.classical.firstPage]; rfl)
  43     (by exact_mod_cast (show (2-(D.classical.obj (X.obj D.auxiliary)).r₀).toNat ≤
  44       (r-(D.classical.obj (X.obj D.auxiliary)).r₀).toNat by omega))

  46 public theorem isCycle_iff_reaches (X : ClassicalObject) (r : ℤ) (hr : 2 ≤ r) (p : ℤ × ℤ)
  47     (x : Ambient D X p) :
  48     IsCycle D X (r-1) p x ↔ ReachesPage (D.classical.obj (X.obj D.auxiliary)) r p x := by
  49   let E := D.classical.obj (X.obj D.auxiliary)
  50   let P := E.ssData p
  51   let m : WithTop ℕ := ↑(r-E.r₀).toNat
  52   have hm : (↑(r-1-1).toNat : WithTop ℕ) = m := by
  53     dsimp only [m, E]
  54     rw [D.classical.firstPage]
  55     congr 2; omega
  56   constructor
  57   · rintro ⟨_, hx⟩
  58     change ∃ z, KIP126.Literature.cycleLabel D X (↑(r-1-1).toNat) p z = x at hx
  59     rw [hm] at hx
  60     obtain ⟨z,hz⟩ := hx
  61     refine ⟨P.pageπ m z, hr, z, ?_, rfl⟩
  62     change (Subobject.ofLE (P.Z m) (P.Z _) _ ≫ P.pageπ _) z = x
  63     rw [initialProjection_eq_cycleLabel D X r hr p]
  64     exact hz
  65   · rintro ⟨y, _, z, hz, _⟩
  66     refine ⟨by omega, ?_⟩
  67     change ∃ z, KIP126.Literature.cycleLabel D X (↑(r-1-1).toNat) p z = x
  68     rw [hm]
  69     refine ⟨z, ?_⟩
  70     rw [initialProjection_eq_cycleLabel D X r hr p] at hz
  71     exact hz

  73 public theorem representsOnPage_of_congruent (X : ClassicalObject) {R : ℤ}
  74     (hR : 2 ≤ R) {p : ℤ × ℤ} {x y : Ambient D X p}
  75     {xr : (D.classical.obj (X.obj D.auxiliary)).Page R p}
  76     (hx : RepresentsOnPage (D.classical.obj (X.obj D.auxiliary)) R p x xr)
  77     (hy : Congruent D X (R - 1) p y x) :
  78     RepresentsOnPage (D.classical.obj (X.obj D.auxiliary)) R p y xr := by
  79   let E := D.classical.obj (X.obj D.auxiliary)
  80   let P := E.ssData p
  81   let k : WithTop ℕ := ↑(R - E.r₀).toNat
  82   obtain ⟨_, z, hzx, hzr⟩ := hx
  83   have hb : y - x ∈ KIP126.Literature.boundaries D X (R - 1) p := hy.2
  84   change ∃ b, KIP126.Literature.boundaryLabel D X (↑(R - 1 - 1).toNat) p b = y - x at hb
  85   have hk : (↑(R-1-1).toNat : WithTop ℕ) = k := by
  86     dsimp only [k,E]
  87     rw [D.classical.firstPage]
  88     congr 2; omega
  89   rw [hk] at hb
  90   obtain ⟨b, hb⟩ := hb
  91   let j := Subobject.ofLE (P.B k) (P.Z k) (P.B_le_Z k)
  92   have hlabel : j ≫ KIP126.Literature.cycleLabel D X k p =
  93       KIP126.Literature.boundaryLabel D X k p := by
  94     dsimp only [j, KIP126.Literature.cycleLabel, KIP126.Literature.boundaryLabel, P, E]
  95     simp only [← Category.assoc, Subobject.ofLE_comp_ofLE]
  96   have hboundary : (P.pageπ k) (j b) = 0 := by
  97     have hj : j ≫ P.pageπ k = 0 := cokernel.condition _
  98     exact congrArg (fun f => f b) hj
  99   refine ⟨hR, z + j b, ?_, ?_⟩
 100   · have hz0 : KIP126.Literature.cycleLabel D X k p z = x := by
 101       rw [initialProjection_eq_cycleLabel D X R hR p] at hzx
 102       exact hzx
 103     have hzb : KIP126.Literature.cycleLabel D X k p (j b) = y - x := by
 104       change (j ≫ KIP126.Literature.cycleLabel D X k p) b = _
 105       rw [hlabel]
 106       exact hb
 107     have hh : KIP126.Literature.cycleLabel D X k p (z + j b) = y := by
 108       rw [map_add, hz0, hzb]
 109       abel
 110     change (Subobject.ofLE (P.Z k) (P.Z _) _ ≫ P.pageπ _) (z+j b) = y
 111     rw [initialProjection_eq_cycleLabel D X R hR p]
 112     exact hh
 113   · change P.pageπ k (z + j b) = xr
 114     rw [map_add, hboundary, add_zero]
 115     exact hzr

 117 public theorem differentialAt_of_congruent (X : ClassicalObject) {R : ℤ}
 118     {p q : ℤ × ℤ} {x : Ambient D X p} {y z : Ambient D X q}
 119     (hd : DifferentialAt D X R p q x y)
