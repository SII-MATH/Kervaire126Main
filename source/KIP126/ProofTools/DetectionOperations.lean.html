  23 theorem grade_natural (A B : SyntheticObject)
  24     (f : A.obj model.nu model.auxiliary ⟶ B.obj model.nu model.auxiliary) (i : Tridegree)
  25     (a : (Subobject.underlying.obj ((model.syntheticTower.filtration
  26       (A.obj model.nu model.auxiliary)).F i.1 (i.2.1-i.1,i.2.2)) : ModuleCat ℤ))
  27     (b : (Subobject.underlying.obj ((model.syntheticTower.filtration
  28       (B.obj model.nu model.auxiliary)).F i.1 (i.2.1-i.1,i.2.2)) : ModuleCat ℤ))
  29     (hab : ((model.syntheticTower.filtration (A.obj model.nu model.auxiliary)).F
  30       i.1 (i.2.1-i.1,i.2.2)).arrow a ≫ f =
  31       ((model.syntheticTower.filtration (B.obj model.nu model.auxiliary)).F
  32         i.1 (i.2.1-i.1,i.2.2)).arrow b) :
  33     ((model.convergence B).identification i).inv
  34       ((model.syntheticTower.filtration (B.obj model.nu model.auxiliary)).toAssociatedGraded
  35         i.1 (i.2.1-i.1,i.2.2) b) =
  36       (model.family.map f).eInftyMap i (((model.convergence A).identification i).inv
  37         ((model.syntheticTower.filtration (A.obj model.nu model.auxiliary)).toAssociatedGraded
  38           i.1 (i.2.1-i.1,i.2.2) a)) := by
  39   let M := model.convergentMap A B f
  40   let FA := model.syntheticTower.filtration (A.obj model.nu model.auxiliary)
  41   let FB := model.syntheticTower.filtration (B.obj model.nu model.auxiliary)
  42   let p := (i.2.1-i.1,i.2.2)
  43   have hb : (M.filteredMap.compat i.1 p).choose a = b := by
  44     apply (ModuleCat.mono_iff_injective (FB.F i.1 p).arrow).mp inferInstance
  45     have h := ConcreteCategory.congr_hom (M.filteredMap.compat i.1 p).choose_spec a
  46     change (FB.F i.1 p).arrow ((M.filteredMap.compat i.1 p).choose a) =
  47       M.filteredMap.map p ((FA.F i.1 p).arrow a) at h
  48     have hm := ConcreteCategory.congr_hom (M.filteredMap_map p) ((FA.F i.1 p).arrow a)
  49     change M.filteredMap.map p ((FA.F i.1 p).arrow a) = (FA.F i.1 p).arrow a ≫ f at hm
  50     exact h.trans (hm.trans hab)
  51   have hp := graded_map_representative M.filteredMap i.1 p a
  52   rw [hb] at hp
  53   apply ((model.convergence B).identification i).toLinearEquiv.injective
  54   have hn := ConcreteCategory.congr_hom (M.infinity_action_graded i)
  55     (((model.convergence A).identification i).inv (FA.toAssociatedGraded i.1 p a))
  56   change M.filteredMap.inducedGrMap i.1 p
  57       (((model.convergence A).identification i).hom
  58         (((model.convergence A).identification i).inv (FA.toAssociatedGraded i.1 p a))) =
  59     ((model.convergence B).identification i).hom
  60       ((model.family.map f).eInftyMap i
  61         (((model.convergence A).identification i).inv (FA.toAssociatedGraded i.1 p a))) at hn
  62   have hA := ((model.convergence A).identification i).toLinearEquiv.apply_symm_apply
  63     (FA.toAssociatedGraded i.1 p a)
  64   have hB := ((model.convergence B).identification i).toLinearEquiv.apply_symm_apply
  65     (FB.toAssociatedGraded i.1 p b)
  66   exact hB.trans (hp.symm.trans
  67     ((congrArg (M.filteredMap.inducedGrMap i.1 p) hA.symm).trans hn))

  69 theorem detects_map (A B : SyntheticObject)
  70     (f : A.obj model.nu model.auxiliary ⟶ B.obj model.nu model.auxiliary) (i : Tridegree)
  71     (x : (model.family.obj (A.obj model.nu model.auxiliary)).E₂ i)
  72     (a : BiHom (i.2.1-i.1) i.2.2 (A.obj model.nu model.auxiliary))
  73     (ha : Detects (model.convergence A) i x a) :
  74     Detects (model.convergence B) i (familyPageMap model.family f 2 i x) (a ≫ f) := by
  75   obtain ⟨e,he,a',ha',hga⟩ := ha
  76   let M := model.convergentMap A B f
  77   let b := (M.filteredMap.compat i.1 (i.2.1-i.1,i.2.2)).choose a'
  78   have hb : ((model.syntheticTower.filtration (B.obj model.nu model.auxiliary)).F
  79       i.1 (i.2.1-i.1,i.2.2)).arrow b = a ≫ f := by
  80     have h := ConcreteCategory.congr_hom
  81       (M.filteredMap.compat i.1 (i.2.1-i.1,i.2.2)).choose_spec a'
  82     change _ = M.filteredMap.map (i.2.1-i.1,i.2.2) _ at h
  83     have hm := ConcreteCategory.congr_hom (M.filteredMap_map (i.2.1-i.1,i.2.2))
  84       (((model.syntheticTower.filtration (A.obj model.nu model.auxiliary)).F
  85         i.1 (i.2.1-i.1,i.2.2)).arrow a')
  86     change _ = _ ≫ f at hm
  87     exact h.trans (hm.trans (congrArg (fun z => z ≫ f) ha'))
  88   refine ⟨_,family_infinity_representative_map f i x e he,b,hb,?_⟩
  89   have h := grade_natural A B f i a' b (by rw [ha',hb])
  90   have haE : ((model.convergence A).identification i).inv
  91       ((model.syntheticTower.filtration (A.obj model.nu model.auxiliary)).toAssociatedGraded
  92         i.1 (i.2.1-i.1,i.2.2) a') = e := by
  93     rw [← hga]
  94     exact ((model.convergence A).identification i).toLinearEquiv.symm_apply_apply e
  95   rw [haE] at h
  96   exact (congrArg ((model.convergence B).identification i).hom h).symm.trans
  97     (((model.convergence B).identification i).toLinearEquiv.apply_symm_apply _)

  99 theorem detects_reflect (A B : SyntheticObject)
 100     (f : A.obj model.nu model.auxiliary ⟶ B.obj model.nu model.auxiliary) (i : Tridegree)
 101     (hf : Function.Injective ((model.family.map f).eInftyMap i).hom)
 102     (x : (model.family.obj (A.obj model.nu model.auxiliary)).E₂ i)
 103     (e : ((model.family.obj (A.obj model.nu model.auxiliary)).sequence.ssData i).eInfty)
 104     (he : HasInfinityRepresentative _ 2 i x e)
 105     (a : BiHom (i.2.1-i.1) i.2.2 (A.obj model.nu model.auxiliary))
 106     (ha : FiltrationAtLeast model.syntheticTower i.1 a)
 107     (hb : Detects (model.convergence B) i (familyPageMap model.family f 2 i x) (a ≫ f)) :
 108     Detects (model.convergence A) i x a := by
 109   obtain ⟨a',ha'⟩ := filtration_representative_exists model.syntheticTower
 110     (A.obj model.nu model.auxiliary) (i.2.1-i.1,i.2.2) i.1 a ha
 111   obtain ⟨eB,heB,b,hb,hgb⟩ := hb
 112   have huniq := heB.unique (family_infinity_representative_map f i x e he)
 113   have h := grade_natural A B f i a' b (by rw [ha',hb])
 114   have hB : ((model.convergence B).identification i).inv
 115       ((model.syntheticTower.filtration (B.obj model.nu model.auxiliary)).toAssociatedGraded
 116         i.1 (i.2.1-i.1,i.2.2) b) = (model.family.map f).eInftyMap i e := by
 117     rw [← hgb]
 118     exact (((model.convergence B).identification i).toLinearEquiv.symm_apply_apply eB).trans huniq
 119   have hA := hf (h.symm.trans hB)
 120   refine ⟨e,he,a',ha',?_⟩
 121   exact (congrArg ((model.convergence A).identification i).hom hA).symm.trans
 122     (((model.convergence A).identification i).toLinearEquiv.apply_symm_apply _)

 132 /-- A filtration frame, reindexed by the same stem, gives detection. -/
 133 theorem source_frame_detects (cx : SyntheticTowerConvergence P X)
 134     (p : ℤ × ℤ) (s : ℤ)
 135     (x : ((F.obj X).sequence.ssData (s,p.1+s,p.2)).eInfty)
 136     (x0 : (F.obj X).E₂ (s,p.1+s,p.2))
 137     (hx : HasInfinityRepresentative _ 2 _ x0 x)
 138     (a : (Subobject.underlying.obj ((P.filtration X).F s p) : ModuleCat.{v} ℤ))
 139     (ha : (P.filtration X).toAssociatedGraded s p a = (infinityGradedIso cx p s).hom x) :
 140     Detects cx (s,p.1+s,p.2) x0
 141       (homotopyRegrade (by dsimp; omega) rfl (((P.filtration X).F s p).arrow a)) := by
 142   have lift_frame (m m' w s : ℤ) (hm : m=m') {V : ModuleCat.{v} ℤ} (e : V)
 143       (I : V ≅ (P.filtration X).associatedGraded s (m',w))
 144       (a : (Subobject.underlying.obj ((P.filtration X).F s (m,w)) : ModuleCat.{v} ℤ))
 145       (ha : (P.filtration X).toAssociatedGraded s (m,w) a =
 146         (I ≪≫ eqToIso (congrArg
 147           (fun z => (P.filtration X).associatedGraded s (z,w)) hm.symm)).hom e) :
 148       ∃ a' : (Subobject.underlying.obj ((P.filtration X).F s (m',w)) : ModuleCat.{v} ℤ),
 149         ((P.filtration X).F s (m',w)).arrow a' =
 150           homotopyRegrade hm rfl (((P.filtration X).F s (m,w)).arrow a) ∧
 151         I.hom e = (P.filtration X).toAssociatedGraded s (m',w) a' := by
 152     subst m'
 153     refine ⟨a,rfl,?_⟩
 154     simpa only [eqToIso_refl, Iso.trans_refl] using ha.symm
 155   obtain ⟨a',ha',hga⟩ := lift_frame p.1 ((p.1+s)-s) p.2 s (by omega) x
 156     (cx.identification (s,p.1+s,p.2)) a ha
 157   exact ⟨x,hx,a',ha',hga⟩

 159 /-- Detection supplies a frame in the original stem coordinates. -/
 160 theorem frame_of_source_detects (cx : SyntheticTowerConvergence P X)
 161     (p : ℤ × ℤ) (s : ℤ) (x0 : (F.obj X).E₂ (s,p.1+s,p.2))
 162     (a0 : BiHom p.1 p.2 X)
 163     (ha0 : Detects cx (s,p.1+s,p.2) x0 (homotopyRegrade (by dsimp; omega) rfl a0)) :
 164     ∃ x : ((F.obj X).sequence.ssData (s,p.1+s,p.2)).eInfty,
 165       HasInfinityRepresentative _ 2 _ x0 x ∧
 166       ∃ a : (Subobject.underlying.obj ((P.filtration X).F s p) : ModuleCat.{v} ℤ),
 167         ((P.filtration X).F s p).arrow a = a0 ∧
 168         (P.filtration X).toAssociatedGraded s p a = (infinityGradedIso cx p s).hom x := by
 169   have lower_frame (m m' w s : ℤ) (hm : m=m') {V : ModuleCat.{v} ℤ} (e : V)
 170       (I : V ≅ (P.filtration X).associatedGraded s (m',w))
 171       (a0 : BiHom m w X)
 172       (a' : (Subobject.underlying.obj ((P.filtration X).F s (m',w)) : ModuleCat.{v} ℤ))
 173       (ha' : ((P.filtration X).F s (m',w)).arrow a' = homotopyRegrade hm rfl a0)
 174       (hga : I.hom e = (P.filtration X).toAssociatedGraded s (m',w) a') :
 175       ∃ a : (Subobject.underlying.obj ((P.filtration X).F s (m,w)) : ModuleCat.{v} ℤ),
 176         ((P.filtration X).F s (m,w)).arrow a = a0 ∧
 177         (P.filtration X).toAssociatedGraded s (m,w) a =
 178           (I ≪≫ eqToIso (congrArg
 179             (fun z => (P.filtration X).associatedGraded s (z,w)) hm.symm)).hom e := by
 180     subst m'
 181     exact ⟨a',ha',hga.symm⟩
 182   obtain ⟨e,he,a',ha',hga⟩ := ha0
 183   obtain ⟨a,ha,hg⟩ := lower_frame p.1 ((p.1+s)-s) p.2 s (by omega) e
 184     (cx.identification (s,p.1+s,p.2)) a0 a' ha' hga
 185   exact ⟨e,he,a,ha,hg⟩

 187 /-- Exact representatives of two detected classes determine the extension
 188 equation for this same map and pair of convergences. -/
 189 theorem extension_equation_of_detected_pair
 190     (M : ConvergentMap F g cx.toConvergence cy.toConvergence)
 191     (p : ℤ × ℤ) (s n : ℤ) (hn : 0 ≤ n)
 192     (x0 : (F.obj X).E₂ (s,p.1+s,p.2))
 193     (y0 : (F.obj Y).E₂ (s+n,p.1+(s+n),p.2))
 194     (a : BiHom p.1 p.2 X) (b : BiHom p.1 p.2 Y)
 195     (ha : Detects cx (s,p.1+s,p.2) x0 (homotopyRegrade (by dsimp; omega) rfl a))
 196     (hb : Detects cy (s+n,p.1+(s+n),p.2) y0 (homotopyRegrade (by dsimp; omega) rfl b))
 197     (hab : a ≫ g = b) :
 198     ∃ x y, HasInfinityRepresentative _ 2 _ x0 x ∧
 199       HasInfinityRepresentative _ 2 _ y0 y ∧ TowerExtensionEquation M p s n x y := by
 200   obtain ⟨x,hx,a',ha',hax⟩ := frame_of_source_detects cx p s x0 a ha
 201   obtain ⟨y,hy,b',hb',hby⟩ := frame_of_source_detects cy p (s+n) y0 b hb
 202   exact ⟨x,y,hx,hy,(tower_extension_iff M p s n x y).mpr
 203     ⟨hn,a',b',by rw [ha',hb']; exact hab,hax,hby⟩⟩
