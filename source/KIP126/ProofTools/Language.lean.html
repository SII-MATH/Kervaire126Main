  26 /-- Membership in paper Z_c, with its valid index range explicit. -/
  27 public def IsCycle (level : ℤ) (p : ℤ × ℤ) (x : Ambient D X p) : Prop :=
  28   1 ≤ level ∧ x ∈ KIP126.Literature.cycles D X level p

  30 /-- Membership in paper B_c; this does not assert a nonzero incoming hit. -/
  31 public def IsBoundary (level : ℤ) (p : ℤ × ℤ) (x : Ambient D X p) : Prop :=
  32   1 ≤ level ∧ x ∈ KIP126.Literature.boundaries D X level p

  34 /-- A common infinite-cycle representative, allowing the zero class. -/
  35 public def IsPermanent (p : ℤ × ℤ) (x : Ambient D X p) : Prop :=
  36   x ∈ KIP126.Literature.permanentCycles D X p

  38 /-- Equality modulo the actual paper boundary submodule. -/
  39 public def Congruent (level : ℤ) (p : ℤ × ℤ) (x y : Ambient D X p) : Prop :=
  40   IsBoundary D X level p (x - y)

  42 /-- The actual Adams differential on common E₂ labels, with the degree
  43 transport checked by `HasDifferential`. It is never a freely chosen operation. -/
  44 public def DifferentialAt (r : ℤ) (p q : ℤ × ℤ)
  45     (x : Ambient D X p) (y : Ambient D X q) : Prop :=
  46   2 ≤ r ∧ KIP126.Core.SpectralSequence.HasDifferential
  47     (D.classical.obj (X.obj D.auxiliary)) r p q x y

  49 /-- MainPaper Definition `classicalcrossdiff`: a crossing of a length-r
  50 Adams differential beginning at p, viewed on page `page`. Its witness is an
  51 essential differential in the actual Adams sequence, with the same stems.
  52 This is distinct from the direction used in Moss's theorem. -/
  53 public def HasCrossingOn (r page : ℤ) (p : ℤ × ℤ) : Prop :=
  54   ∃ a b : ℤ, 0 < a ∧ a ≤ page - 2 ∧ 0 ≤ b ∧ b ≤ r - page ∧
  55     ∃ (x : Ambient D X (p.1 + a, p.2 + a))
  56       (y : Ambient D X (p.1 + r - b, p.2 + r - b - 1)),
  57       KIP126.Core.SpectralSequence.HasNonzeroDifferential
  58         (D.classical.obj (X.obj D.auxiliary)) (r - a - b)
  59         (p.1 + a, p.2 + a) (p.1 + r - b, p.2 + r - b - 1) x y
  60 
  61 /-- Absence of paper-direction crossings in the stated, valid page range. -/
  62 public def NoCrossingOn (r page : ℤ) (p : ℤ × ℤ) : Prop :=
  63   2 ≤ page ∧ page ≤ r ∧ ¬ HasCrossingOn D X r page p

  68 /-- A crossing in the finite page-extension position `(r,n,s,t)`.
  69 The target degree is written as the actual shorter witness's target;
  70 it is `(s+n-b,t+n-b)`. Its source and target cycle conditions are supplied
  71 by that same witness, on page r-a and of length n-a-b. -/
  72 public def FiniteHasCrossing
  73     (r : ℕ) (n s t : ℤ) : Prop :=
  74   ∃ a b : ℕ, 0 < a ∧ a ≤ r - 2 ∧
  75     (b : ℤ) ≤ n - a - normalizedExponent D.classicalTower f ∧
  76     ∃ (x' : Ambient D X (s + a, t + a))
  77       (y' : Ambient D Y
  78         (s + a + (n - a - b), t + a + (n - a - b))),
  79       ∃ W' : FiniteExtensionWitness D X Y f (r - a) (n - a - b) (s + a) (t + a) x' y',
  80         W'.Essential ∧ y' ∉ KIP126.Literature.boundaries D Y
  81           (1 + n - b - normalizedExponent D.classicalTower f)
  82           (s + a + (n - a - b), t + a + (n - a - b))
  83 
  84 /-- Absence of crossings in this finite position. Applied to an actual
  85 extension witness, its page and length validity are already supplied. -/
  86 public def FiniteNoCrossing
  87     (r : ℕ) (n s t : ℤ) : Prop :=
  88   ¬ FiniteHasCrossing D X Y f r n s t
  89 
  90 /-- The infinite clause uses the same untruncated normalized map and actual
  91 permanent representatives. The bound a≤n-e follows from the length range,
  92 and is recorded explicitly; there is no finite-page upper bound on a. -/
  93 public def InfiniteHasCrossing
  94     (n s t : ℤ) : Prop :=
  95   ∃ a b : ℕ, 0 < a ∧ (a : ℤ) ≤ n - normalizedExponent D.classicalTower f ∧
  96     (b : ℤ) ≤ n - a - normalizedExponent D.classicalTower f ∧
  97     ∃ (x' : Ambient D X (s + a, t + a))
  98       (y' : Ambient D Y
  99         (s + a + (n - a - b), t + a + (n - a - b))),
 100       ∃ W' : InfiniteExtensionWitness D X Y f (n - a - b) (s + a) (t + a) x' y',
 101         W'.Essential ∧ y' ∉ KIP126.Literature.boundaries D Y
 102           (1 + n - b - normalizedExponent D.classicalTower f)
 103           (s + a + (n - a - b), t + a + (n - a - b))
 104 
 105 public def InfiniteNoCrossing
 106     (n s t : ℤ) : Prop :=
 107   ¬ InfiniteHasCrossing D X Y f n s t

 110 /-- 有限 stretching 的障碍候选，直接在同一个早期 λ 商的实际 ESS 中定义。
 111 源和靶的 synthetic weight 固定为原 extension 的 `t + e(f)`，而源的
 112 filtration 提高 `a`。源必须不在后期商到早期商的实际 `ρ` 所诱导的
 113 `E∞` 映射像中；这与某个普通 Adams 代表元不属于后页 cycle 层不同。
 114 允许 `b = 0` 及长度 `n - a - b = 0` 的 synthetic extension，包括
 115 `e(f) = 1` 的情形。要求同商中实际 extension equation 的靶非零，不要求
 116 essential 性，不换成较小 λ 商的 extension，也不额外施加普通 Adams
 117 boundary 排除条件。排除这些候选强于仅排除 essential crossings；这是
 118 修复后的充分条件，不声称与原普通 Adams 代表元谓词等价。 -/
 119 public def FiniteExtensionNonliftableCrossing
 120     (rEarlier rLater : ℕ) (n s t : ℤ) : Prop :=
 121   ∃ hle : rEarlier ≤ rLater, ∃ a b : ℕ,
 122     0 < a ∧ a ≤ rEarlier - 2 ∧ (a : ℤ) + b ≤ n ∧
 123     ∃ (source : ((D.family.obj ((SyntheticObject.quotient (rEarlier - 1)
 124           (extensionSourceObject D X Y f)).obj D.nu D.auxiliary)).sequence.ssData
 125           (s + a, (t - s) + (s + a), t + normalizedExponent D.classicalTower f)).eInfty)
 126       (target : ((D.family.obj ((SyntheticObject.quotient (rEarlier - 1)
 127           (.nu Y)).obj D.nu D.auxiliary)).sequence.ssData
 128           (s + a + (n - a - b), (t - s) + (s + a + (n - a - b)),
 129             t + normalizedExponent D.classicalTower f)).eInfty),
 130       source ∉ Set.range (((D.family.functor.map
 131         ((D.quotients.tower ((extensionSourceObject D X Y f).obj D.nu D.auxiliary)).rho
 132           (rEarlier - 1) (rLater - 1) (Nat.sub_le_sub_right hle 1))).eInftyMap
 133             (s + a, (t - s) + (s + a), t + normalizedExponent D.classicalTower f)).hom) ∧
 134       TowerExtensionEquation (D.convergentMap
 135         (.quotient (rEarlier - 1) (extensionSourceObject D X Y f))
 136         (.quotient (rEarlier - 1) (.nu Y))
 137         (finiteExtensionMap D X Y f (rEarlier - 1)))
 138         (t - s, t + normalizedExponent D.classicalTower f) (s + a) (n - a - b) source target ∧
 139       target ≠ 0
