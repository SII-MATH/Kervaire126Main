  21 /-- 本文待证工具（原 am7）：MainPaper Theorem 6.1 (`thm:e73f481e`) 的准确交付命题。
  22 所有页面、微分、有限／无限 extension 和 crossing 均来自同一 D。
  23 末端次数是 (s+r+l,t+r+l-1)，l-extension 不额外减一次 t。
  24 本定义只给出命题；证明使用同一 D 上的文献比较定理及
  25 论文中的 δ／λ／ρ 推理。不要求微分或 extension 非零。 -/
  26 public def GeneralizedLeibnizLaw (X Y : ClassicalObject) (f : X.obj D.auxiliary ⟶ Y.obj D.auxiliary) : Prop :=
  27   ∀ (page : ℕ) (r m l s t : ℤ)
  28     (x : Ambient D X (s, t)) (y : Ambient D Y (s + m, t + m))
  29     (xInfinity : Ambient D X (s + r, t + r - 1))
  30     (yInfinity : Ambient D Y (s + r + l, t + r - 1 + l)),
  31     2 ≤ page → (page : ℤ) ≤ r →
  32     (normalizedExponent D.classicalTower f : ℤ) ≤ m →
  33     m ≤ (page : ℤ) - 2 + normalizedExponent D.classicalTower f →
  34     (normalizedExponent D.classicalTower f : ℤ) ≤ l →
  35     IsCycle D X (r - 1) (s, t) x →
  36     IsCycle D Y (r - 1 - m + normalizedExponent D.classicalTower f) (s + m, t + m) y →
  37     IsPermanent D X (s + r, t + r - 1) xInfinity →
  38     IsPermanent D Y (s + r + l, t + r - 1 + l) yInfinity →
  39     DifferentialAt D X r (s, t) (s + r, t + r - 1) x xInfinity →
  40     FiniteExtension D X Y f page m s t x y →
  41     InfiniteExtension D X Y f l (s + r) (t + r - 1) xInfinity yInfinity →
  42     (NoCrossingOn D X r page (s, t) ∨
  43       FiniteNoCrossing D X Y f page m s t) →
  44     InfiniteNoCrossing D X Y f l (s + r) (t + r - 1) →
  45     DifferentialAt D Y (r + l - m) (s + m, t + m)
  46       (s + r + l, t + r - 1 + l) y yInfinity

  49 /-- 本文待证工具（原 am7）：MainPaper Theorem 6.12 (`thm:158d451a`) 的实际三角版本。
  50 第二个 no-crossing 分支属于 Z 上的 d_r，在 E_r′ 页检查；不属于 g。
  51 h 的 target 位于 ΣX，其标签通过 D.classicalSuspension 的实际 tower/layer 图和 raw-cycle
  52 商代表元与 x 关联。该比较的构造证明尚未交付，不能以任意 E₂ 等价代替。
  53 结论是存在同余于 y 的真实 f-extension target，模实际 B_r′；不是自由
  54 选取的 fTarget 操作。r′ 及结论页的正性由指数和与三个长度界保证。 -/
  55 public def GeneralizedMahowaldLaw (T : TriangleData D.auxiliary) : Prop :=
  56   ∀ he : (normalizedExponent D.classicalTower T.f : ℤ) + normalizedExponent D.classicalTower T.g +
  57       normalizedExponent D.classicalTower T.h = 1,
  58   KIP126.Literature.NormalizedTriangleCompatible D T he →
  59   ∀ (n m l s t : ℤ),
  60     let r := n + m + l
  61     let n₁ := n - normalizedExponent D.classicalTower T.f
  62     let m₁ := m - normalizedExponent D.classicalTower T.g
  63     let l₁ := l - normalizedExponent D.classicalTower T.h
  64     let r' := r - m₁
  65     ∀ (x : Ambient D (T.X) (s + l, t + l - 1))
  66       (y : Ambient D (T.Y) (s + l + n, t + l - 1 + n))
  67       (xBar : Ambient D (T.Z) (s, t))
  68       (yBar : Ambient D (T.Z) (s + l + n + m, t + l - 1 + n + m))
  69       (suspendedX : Ambient D (.shift 1 T.X) (s + l, t + l)),
  70       1 ≤ n₁ → 0 ≤ m₁ → 0 ≤ l₁ →
  71       IsCycle D (T.X) n₁ (s + l, t + l - 1) x →
  72       IsCycle D (T.Y) (m₁ + 1) (s + l + n, t + l - 1 + n) y →
  73       IsCycle D (T.Z) (r - 1) (s, t) xBar →
  74       IsPermanent D (T.Z) (s + l + n + m, t + l - 1 + n + m) yBar →
  75       (D.classicalSuspension T.X).DesuspendsClass (s + l) (t + l) suspendedX x →
  76       FiniteExtension D T.Z (.shift 1 T.X) T.h r'.toNat l s t xBar suspendedX →
  77       DifferentialAt D (T.Z) r (s, t)
  78         (s + l + n + m, t + l - 1 + n + m) xBar yBar →
  79       (FiniteNoCrossing D T.Z (.shift 1 T.X) T.h r'.toNat l s t ∨
  80         NoCrossingOn D (T.Z) r r' (s, t)) →
  81       FiniteExtension D T.Y T.Z T.g (m₁ + 2).toNat m (s + l + n) (t + l - 1 + n) y yBar →
  82       IsCycle D (T.X) (n + m + normalizedExponent D.classicalTower T.h) (s + l, t + l - 1) x ∧
  83         ∃ y' : Ambient D (T.Y) (s + l + n, t + l - 1 + n),
  84           FiniteExtension D T.X T.Y T.f (n + m + 1 + normalizedExponent D.classicalTower T.h).toNat
  85             n (s + l) (t + l - 1) x y' ∧
  86           Congruent D (T.Y) r' (s + l + n, t + l - 1 + n) y' y

  89 /-- 本文待证工具（原 am7）：保留 λ 层数的有限页 stretching 充分条件。
  90 参考 MainPaper Proposition `prop:dec738d3` 中同一有限商内的修正过程。
  91 源、靶须属于后页所需的 cycle 层；排除的障碍是在早页同一个有限商中，
  92 源不属于后页 ρ 的 E∞ 像、目标非零的实际较短扩张（包括非 essential 的扩张）。
  93 不能改为只排除较小商中的无层扩张：乘 λ 可能杀掉低滤过项。
  94 这比只排除 essential crossings 更强，不声称等价于原跨页候选或印出的推论。
  95 原仓库已证明有限滤过修正；当前迁移状态另行追踪，不将迁移缺口认定为新假设。
  96 结论只给出后页 extension 的存在，不声称任意指定的早期严格解可提升，
  97 也不蕴含相容解塔或未截断 extension 的存在。 -/
  98 public def FinitePageExtensionStretchingLaw (X Y : ClassicalObject) (f : X.obj D.auxiliary ⟶ Y.obj D.auxiliary) : Prop :=
  99   ∀ (rEarlier rLater : ℕ) (n s t : ℤ)
 100     (x : Ambient D X (s, t)) (y : Ambient D Y (s + n, t + n)),
 101     2 ≤ rEarlier → rEarlier ≤ rLater →
 102     (normalizedExponent D.classicalTower f : ℤ) ≤ n →
 103     n ≤ (rEarlier : ℤ) - 2 + normalizedExponent D.classicalTower f →
 104     IsCycle D X ((rLater : ℤ) - 1) (s, t) x →
 105     IsCycle D Y ((rLater : ℤ) - 1 - n + normalizedExponent D.classicalTower f) (s + n, t + n) y →
 106     FiniteExtension D X Y f rEarlier n s t x y →
 107     ¬ FiniteExtensionNonliftableCrossing D X Y f rEarlier rLater n s t →
 108     FiniteExtension D X Y f rLater n s t x y
