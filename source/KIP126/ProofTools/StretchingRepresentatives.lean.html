  22 private theorem quotient_shift_exists (coh : BiShiftCoherence SyntheticSpectrum)
  23     (a : ℤ × ℤ) (q : ℕ) (X : SyntheticSpectrum)
  24     [cs : (SyntheticCategory.biShift (Syn := SyntheticSpectrum) a).CommShift ℤ]
  25     [(SyntheticCategory.biShift (Syn := SyntheticSpectrum) a).IsTriangulated] :
  26     ∃ e : XModLambdaN ((SyntheticCategory.biShift a).obj X) q ≅
  27       (SyntheticCategory.biShift a).obj (XModLambdaN X q),
  28       XModLambdaN.incl ((SyntheticCategory.biShift a).obj X) q ≫ e.hom =
  29         (SyntheticCategory.biShift a).map (XModLambdaN.incl X q) ∧
  30       ∃ e₁ : (SyntheticCategory.biShift (lambdaDegree q)).obj
  31           ((SyntheticCategory.biShift a).obj X) ≅
  32         (SyntheticCategory.biShift a).obj
  33           ((SyntheticCategory.biShift (lambdaDegree q)).obj X),
  34         lambdaPow q ((SyntheticCategory.biShift a).obj X) =
  35           e₁.hom ≫ (SyntheticCategory.biShift a).map (lambdaPow q X) ∧
  36         e.hom ≫ (SyntheticCategory.biShift a).map (XModLambdaN.proj X q) ≫
  37             ((SyntheticCategory.biShift a).commShiftIso (1 : ℤ)).hom.app _ =
  38           XModLambdaN.proj ((SyntheticCategory.biShift a).obj X) q ≫
  39             (shiftFunctor SyntheticSpectrum (1 : ℤ)).map e₁.hom := by
  40   let oneComm (a : ℤ × ℤ) :
  41       SyntheticCategory.biShift (Syn := SyntheticSpectrum) a ⋙ SyntheticCategory.biShift (lambdaDegree 1) ≅
  42         SyntheticCategory.biShift (lambdaDegree 1) ⋙ SyntheticCategory.biShift a :=
  43     biShiftAddIso a (0,-1) (a+(0,-1)) rfl ≪≫
  44       (biShiftAddIso (0,-1) a (a+(0,-1)) (add_comm _ _)).symm
  45 
  46   let powerComm (a : ℤ × ℤ) : (q : ℕ) →
  47       SyntheticCategory.biShift (Syn := SyntheticSpectrum) a ⋙ SyntheticCategory.biShift (lambdaDegree q) ≅
  48         SyntheticCategory.biShift (lambdaDegree q) ⋙ SyntheticCategory.biShift a := fun q =>
  49     Nat.rec (motive := fun q =>
  50       SyntheticCategory.biShift (Syn := SyntheticSpectrum) a ⋙ SyntheticCategory.biShift (lambdaDegree q) ≅
  51         SyntheticCategory.biShift (lambdaDegree q) ⋙ SyntheticCategory.biShift a)
  52       (isoWhiskerLeft _ SyntheticCategory.biShift_zero ≪≫
  53         Functor.rightUnitor _ ≪≫ (Functor.leftUnitor _).symm ≪≫
  54         (isoWhiskerRight SyntheticCategory.biShift_zero _).symm)
  55       (fun q ih =>
  56         isoWhiskerLeft _ (lambdaShiftAddIso 1 q (q+1) (by omega)).symm ≪≫
  57         (Functor.associator _ _ _).symm ≪≫
  58         isoWhiskerRight (oneComm a) _ ≪≫ Functor.associator _ _ _ ≪≫
  59         isoWhiskerLeft _ ih ≪≫
  60         (Functor.associator _ _ _).symm ≪≫
  61         isoWhiskerRight (lambdaShiftAddIso 1 q (q+1) (by omega)) _) q
  62 
  63   have powerComm_square (coh : BiShiftCoherence SyntheticSpectrum) (a : ℤ × ℤ)
  64       (q : ℕ) (X : SyntheticSpectrum) :
  65       lambdaPow q ((SyntheticCategory.biShift a).obj X) =
  66         (powerComm a q).hom.app X ≫ (SyntheticCategory.biShift a).map (lambdaPow q X) := by
  67     induction q generalizing X with
  68     | zero =>
  69       simp [powerComm, lambdaPow, ← Functor.map_comp]
  70       exact (Category.comp_id _).symm
  71     | succ q ih =>
  72       have h1 : (oneComm a).hom.app X ≫
  73           (SyntheticCategory.biShift a).map (SyntheticCategory.lam.app X) =
  74           SyntheticCategory.lam.app ((SyntheticCategory.biShift a).obj X) := by
  75         simpa [oneComm, Category.assoc] using (coh.lambda_comm a X).symm
  76       apply (cancel_epi ((lambdaShiftAddIso 1 q (q+1) (by omega)).hom.app
  77         ((SyntheticCategory.biShift a).obj X))).mp
  78       rw [lambdaPow_succ_comparison]
  79       simp only [powerComm, Iso.trans_hom, Iso.symm_hom, NatTrans.comp_app,
  80         isoWhiskerLeft_hom, isoWhiskerRight_hom, whiskerLeft_app, whiskerRight_app,
  81         Functor.associator_inv_app, Functor.associator_hom_app, Category.id_comp, Category.assoc]
  82       rw [Iso.hom_inv_id_app_assoc]
  83       rw [← Functor.map_comp, lambdaPow_succ_comparison, Functor.map_comp]
  84       have hn := (powerComm a q).hom.naturality (SyntheticCategory.lam.app X)
  85       have hn' := congrArg (fun h => h ≫ (SyntheticCategory.biShift a).map (lambdaPow q X)) hn
  86       simp only [Functor.comp_map, Category.assoc] at hn'
  87       dsimp only [KIP126.Synthetic.Context.lambdaDegree, Functor.id_obj] at hn' ⊢
  88       erw [← hn']
  89       rw [← Category.assoc, ← Functor.map_comp, h1, ih]
  90 
  91   let F := SyntheticCategory.biShift (Syn := SyntheticSpectrum) a
  92   let T := Triangle.mk (lambdaPow q X) (XModLambdaN.incl X q) (XModLambdaN.proj X q)
  93   let T' := Triangle.mk (lambdaPow q (F.obj X))
  94     (XModLambdaN.incl (F.obj X) q) (XModLambdaN.proj (F.obj X) q)
  95   have hT : T ∈ distTriang SyntheticSpectrum := HasFunctorialCofiber.cofib_distinguished _
  96   have hT' : T' ∈ distTriang SyntheticSpectrum := HasFunctorialCofiber.cofib_distinguished _
  97   let e := isoTriangleOfIso₁₂ T' (F.mapTriangle.obj T) hT'
  98     (F.map_distinguished T hT) ((powerComm a q).app X) (Iso.refl _) (by
  99       change lambdaPow q (F.obj X) ≫ 𝟙 _ =
 100         (powerComm a q).hom.app X ≫ F.map (lambdaPow q X)
 101       simpa only [Category.comp_id] using powerComm_square coh a q X)
 102   refine ⟨Triangle.π₃.mapIso e, ?_, (powerComm a q).app X,
 103     powerComm_square coh a q X, ?_⟩
 104   · have h := e.hom.comm₂
 105     change XModLambdaN.incl (F.obj X) q ≫ e.hom.hom₃ =
 106       e.hom.hom₂ ≫ F.map (XModLambdaN.incl X q) at h
 107     change XModLambdaN.incl (F.obj X) q ≫ e.hom.hom₃ =
 108       F.map (XModLambdaN.incl X q)
 109     simpa only [e, isoTriangleOfIso₁₂_hom_hom₂, Iso.refl_hom, Category.id_comp] using h
 110   · have h := e.hom.comm₃
 111     have h₁ : e.hom.hom₁ = ((powerComm a q).app X).hom := by
 112       simp only [e, isoTriangleOfIso₁₂_hom_hom₁]
 113     rw [h₁] at h
 114     change XModLambdaN.proj (F.obj X) q ≫
 115         (shiftFunctor SyntheticSpectrum (1 : ℤ)).map ((powerComm a q).app X).hom =
 116       e.hom.hom₃ ≫ (F.map (XModLambdaN.proj X q) ≫
 117         (F.commShiftIso (1 : ℤ)).hom.app _) at h
 118     change e.hom.hom₃ ≫ F.map (XModLambdaN.proj X q) ≫
 119         (F.commShiftIso (1 : ℤ)).hom.app _ = _
 120     exact h.symm

 193     {Z Z' B : Submodule R A} (hZ : Z ≤ Z') :

 230 private theorem syntheticRep_map {A B : SyntheticAdamsSS}
 231     (f : A.sequence ⟶ B.sequence) {r : ℤ} {i : Tridegree}
 232     {x : A.E₂ i} {xr : A.Page r i}
 233     (hx : SyntheticRepresentsOnPage A r i x xr) :
 234     SyntheticRepresentsOnPage B r i (f.pageMap 2 i x) (f.pageMap r i xr) := by
 235   have hstart := f.r₀_eq
 236   rcases A with ⟨⟨⟨r₀, D, δ, d⟩, dd, Zs, Bs⟩, hA, hDA⟩
 237   rcases B with ⟨⟨⟨r₀', D', δ', d'⟩, dd', Zs', Bs'⟩, hB, hDB⟩
 238   dsimp at hstart
 239   subst r₀'

 242   let m : WithTop ℕ := ↑(r-r₀).toNat
 243   let F := f.toSSDataMorphism
 244   have hnm : n ≤ m := by
 245     dsimp [n,m]
 246     exact_mod_cast (show (2-r₀).toNat ≤ (r-r₀).toNat by omega)
 247   refine ⟨hr,F.cycleMap i m z,?_,?_⟩
 248   · have hs := F.cycleMap_ofLE_assoc i hnm ((D' i).pageπ n)
 249     have hp := F.pageπ_pageMap i n
 250     have eqn : (Subobject.ofLE ((D i).Z m) ((D i).Z n) ((D i).Z_anti hnm) ≫
 251         (D i).pageπ n) ≫ F.pageMap i n =
 252       F.cycleMap i m ≫ Subobject.ofLE ((D' i).Z m) ((D' i).Z n) ((D' i).Z_anti hnm) ≫ (D' i).pageπ n := by
 253       rw [Category.assoc,hp]
 254       exact hs
 255     have hh := congrArg (fun a => a z) eqn
 256     simpa only [ModuleCat.comp_apply, hz, SpectralSequenceMorphism.pageMap,
 257       SpectralSequenceMorphism.toPreSSMorphism, PreSSMorphism.pageMap, SSDataMorphism.pageMapAt,
 258       eqToHom_refl, Category.comp_id, n, m, F] using hh.symm
 259   · have hh := congrArg (fun a => a z) (F.pageπ_pageMap i m)
 260     simpa only [ModuleCat.comp_apply, he, SpectralSequenceMorphism.pageMap,
 261       SpectralSequenceMorphism.toPreSSMorphism, PreSSMorphism.pageMap, SSDataMorphism.pageMapAt,
 262       eqToHom_refl, Category.comp_id, n, m, F] using hh.symm

 414 private theorem finite_target_infinity_exists (X Y : ClassicalObject)

 420     (hy : y ∈ KIP126.Literature.cycles model Y

 433     simpa only [hlvl] using hy
 434   let c := KIP126.Literature.subquotientClass
 435     (KIP126.Literature.cycles model Y (q-p.2+(p.2-k)) p)
