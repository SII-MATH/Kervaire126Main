  17 /- A quotient comparison with ordinary suspension, requiring only its
  18    inclusion equation. No choice of exact commShift is identified with the
  19    canonical bigraded shift comparison. -/
  20 private theorem quotient_suspension_inclusion_exists
  21     (A : SyntheticSpectrum) (r : ℕ) :
  22     ∃ e : XModLambdaN (A⟦(1:ℤ)⟧) r ≅ (XModLambdaN A r)⟦(1:ℤ)⟧,
  23       XModLambdaN.incl (A⟦(1:ℤ)⟧) r ≫ e.hom =
  24         (shiftFunctor SyntheticSpectrum (1:ℤ)).map (XModLambdaN.incl A r) := by
  25   letI := model.exactness.shiftCommShift (1,0)
  26   letI := model.exactness.shiftExact (1,0)
  27   obtain ⟨ε,hε,_⟩ := quotient_shift_exists model.quotients.coherence (1,0) r A
  28   let κ := (SyntheticCategory.biShift_compat (Syn:=SyntheticSpectrum) 1).app A
  29   let κr := (SyntheticCategory.biShift_compat (Syn:=SyntheticSpectrum) 1).app (XModLambdaN A r)
  30   let e : XModLambdaN (A⟦(1:ℤ)⟧) r ≅ (XModLambdaN A r)⟦(1:ℤ)⟧ :=
  31     (model.quotients.functoriality.functor r).mapIso κ.symm ≪≫ ε ≪≫ κr
  32   refine ⟨e,?_⟩
  33   change XModLambdaN.incl _ r ≫ XModLambdaN.map κ.inv r ≫ ε.hom ≫ κr.hom = _
  34   rw [←Category.assoc,←XModLambdaN.incl_naturality,Category.assoc,←Category.assoc _ ε.hom,hε]
  35   have hn := (SyntheticCategory.biShift_compat (Syn:=SyntheticSpectrum) 1).hom.naturality
  36     (XModLambdaN.incl A r)
  37   change (SyntheticCategory.biShift (1,0)).map (XModLambdaN.incl A r) ≫ κr.hom =
  38     κ.hom ≫ (shiftFunctor SyntheticSpectrum (1:ℤ)).map (XModLambdaN.incl A r) at hn
  39   rw [hn,Iso.inv_hom_id_assoc]
