  19 private def normalized_g_source_iso (T : TriangleData model.auxiliary) :
  20     (SyntheticCategory.biShift (0,-(normalizedExponent model.classicalTower T.g : ℤ))).obj
  21       ((SyntheticCategory.biShift (0,(normalizedExponent model.classicalTower T.g : ℤ))).obj
  22         (model.nu.functor.obj (T.Y.obj model.auxiliary))) ≅
  23       model.nu.functor.obj (T.Y.obj model.auxiliary) :=
  24   (biShiftAddIso (0,(normalizedExponent model.classicalTower T.g : ℤ))
  25     (0,-(normalizedExponent model.classicalTower T.g : ℤ)) (0,0)
  26     (by ext <;> simp)).app _ ≪≫ SyntheticCategory.biShift_zero.app _

  29     (SyntheticCategory.biShift (0,-(normalizedExponent model.classicalTower T.g : ℤ))).map
  30       (model.normalizedMap T.Y T.Z T.g).map =
  31       (normalized_g_source_iso T).hom ≫
  32         negativeLift (normalizedExponent model.classicalTower T.g)
  33           (model.normalizedMap T.Y T.Z T.g).map := by

  35     Iso.app_hom, NatTrans.comp_app, eqToIso.hom, eqToHom_app, negativeLift,
  36     Category.assoc, Iso.hom_inv_id_app_assoc, eqToHom_trans_assoc]
  37   simp

  39 private def normalized_g_cofiber_triangle_iso (T : TriangleData model.auxiliary)
  40     (he : (normalizedExponent model.classicalTower T.f : ℤ) +
  41       normalizedExponent model.classicalTower T.g + normalizedExponent model.classicalTower T.h = 1)
  42     (hT : NormalizedTriangleCompatible model T he) :
  43     letI := model.exactness.shiftCommShift
  44       (0,-(normalizedExponent model.classicalTower T.g : ℤ))
  45     (SyntheticCategory.biShift (0,-(normalizedExponent model.classicalTower T.g : ℤ))).mapTriangle.obj
  46       (Triangle.mk (model.normalizedMap T.Y T.Z T.g).map
  47         (HasFunctorialCofiber.cofibι (model.normalizedMap T.Y T.Z T.g).map)
  48         (HasFunctorialCofiber.cofibδ (model.normalizedMap T.Y T.Z T.g).map)) ≅
  49       (normalizedTriangle model T he).rotate := by
  50   letI := model.exactness.shiftCommShift
  51     (0,-(normalizedExponent model.classicalTower T.g : ℤ))
  52   letI := model.exactness.shiftExact
  53     (0,-(normalizedExponent model.classicalTower T.g : ℤ))
  54   refine isoTriangleOfIso₁₂ _ _
  55     ((SyntheticCategory.biShift _).map_distinguished _
  56       (HasFunctorialCofiber.cofib_distinguished (model.normalizedMap T.Y T.Z T.g).map))
  57     (rot_of_distTriang _ hT) (normalized_g_source_iso T) (Iso.refl _) ?_
  58   change (SyntheticCategory.biShift
  59       (0,-(normalizedExponent model.classicalTower T.g : ℤ))).map
  60       (model.normalizedMap T.Y T.Z T.g).map ≫ 𝟙 _ =
  61     (normalized_g_source_iso T).hom ≫ negativeLift
  62       (normalizedExponent model.classicalTower T.g) (model.normalizedMap T.Y T.Z T.g).map
  63   simpa only [Category.comp_id] using normalized_g_source_square T

  70       (HasFunctorialCofiber.cofib (model.normalizedMap T.Y T.Z T.g).map) ≅
  71       ((SyntheticCategory.biShift (0,(normalizedExponent model.classicalTower T.f : ℤ))).obj

  75 private theorem normalized_g_shifted_cofiber_inclusion (T : TriangleData model.auxiliary)
  76     (he : (normalizedExponent model.classicalTower T.f : ℤ) +
  77       normalizedExponent model.classicalTower T.g + normalizedExponent model.classicalTower T.h = 1)
  78     (hT : NormalizedTriangleCompatible model T he) :
  79     (SyntheticCategory.biShift (0,-(normalizedExponent model.classicalTower T.g : ℤ))).map
  80       (HasFunctorialCofiber.cofibι (model.normalizedMap T.Y T.Z T.g).map) ≫
  81         (normalized_g_shifted_cofiber_iso T he hT).hom = normalizedConnecting model T he := by
  82   have h := (normalized_g_cofiber_triangle_iso T he hT).hom.comm₂
  83   have h₂ : (normalized_g_cofiber_triangle_iso T he hT).hom.hom₂ = 𝟙 _ := by
  84     simp only [normalized_g_cofiber_triangle_iso, isoTriangleOfIso₁₂_hom_hom₂,
  85       Iso.refl_hom]
  86   rw [h₂, Category.id_comp] at h
  87   exact h

  89 private theorem normalized_g_shifted_cofiber_boundary (T : TriangleData model.auxiliary)
  90     (he : (normalizedExponent model.classicalTower T.f : ℤ) +
  91       normalizedExponent model.classicalTower T.g + normalizedExponent model.classicalTower T.h = 1)
  92     (hT : NormalizedTriangleCompatible model T he) :
  93     letI := model.exactness.shiftCommShift
  94       (0,-(normalizedExponent model.classicalTower T.g : ℤ))
  95     let F := SyntheticCategory.biShift (Syn := SyntheticSpectrum)
  96       (0,-(normalizedExponent model.classicalTower T.g : ℤ))
  97     F.map (HasFunctorialCofiber.cofibδ (model.normalizedMap T.Y T.Z T.g).map) ≫
  98       (F.commShiftIso (1 : ℤ)).hom.app _ ≫
  99         (shiftFunctor SyntheticSpectrum (1 : ℤ)).map (normalized_g_source_iso T).hom =
 100       -((normalized_g_shifted_cofiber_iso T he hT).hom ≫
 101         (shiftFunctor SyntheticSpectrum (1 : ℤ)).map (model.normalizedMap T.X T.Y T.f).map) := by
 102   letI := model.exactness.shiftCommShift
 103     (0,-(normalizedExponent model.classicalTower T.g : ℤ))
 104   have h := (normalized_g_cofiber_triangle_iso T he hT).hom.comm₃
 105   have h₁ : (normalized_g_cofiber_triangle_iso T he hT).hom.hom₁ =
 106       (normalized_g_source_iso T).hom := by
 107     simp only [normalized_g_cofiber_triangle_iso, isoTriangleOfIso₁₂_hom_hom₁]
 108   rw [h₁] at h
 109   change (_ ≫ _) ≫ _ = (normalized_g_shifted_cofiber_iso T he hT).hom ≫
 110     (-((shiftFunctor SyntheticSpectrum (1 : ℤ)).map (model.normalizedMap T.X T.Y T.f).map)) at h
 111   simpa only [Category.assoc, Preadditive.comp_neg, Triangle.mk_mor₃,
 112     Triangle.mk_obj₁] using h

 114 private def normalized_connecting_tail_iso (T : TriangleData model.auxiliary)
 115     (he : (normalizedExponent model.classicalTower T.f : ℤ) +
 116       normalizedExponent model.classicalTower T.g + normalizedExponent model.classicalTower T.h = 1) :
 117     (SyntheticCategory.biShift (0,-(normalizedExponent model.classicalTower T.g : ℤ))).obj
 118       ((SyntheticCategory.biShift (0,-(normalizedExponent model.classicalTower T.h : ℤ))).obj
 119         ((SyntheticCategory.biShift (1,1)).obj
 120           (model.nu.functor.obj (T.X.obj model.auxiliary)))) ≅
 121       ((SyntheticCategory.biShift (0,(normalizedExponent model.classicalTower T.f : ℤ))).obj
 122         (model.nu.functor.obj (T.X.obj model.auxiliary)))⟦(1 : ℤ)⟧ := by
 123   let eg : ℤ := normalizedExponent model.classicalTower T.g
 124   let eh : ℤ := normalizedExponent model.classicalTower T.h
 125   let X := model.nu.functor.obj (T.X.obj model.auxiliary)
 126   exact (SyntheticCategory.biShift (0,-eg)).mapIso
 127       ((SyntheticCategory.biShift_comp (1,1) (0,-eh)).app X) ≪≫
 128     (SyntheticCategory.biShift_comp ((1,1)+(0,-eh)) (0,-eg)).app X ≪≫
 129     eqToIso (congrArg (fun p => (SyntheticCategory.biShift p).obj X)
 130       (show ((1,1)+(0,-eh))+(0,-eg) =
 131           (0,(normalizedExponent model.classicalTower T.f : ℤ))+(1,0) from by
 132         dsimp [eg,eh]; ext <;> simp only []; omega)) ≪≫
 133     (SyntheticCategory.biShift_comp (0,(normalizedExponent model.classicalTower T.f : ℤ))
 134       (1,0)).symm.app X ≪≫
 135     (SyntheticCategory.biShift_compat (Syn := SyntheticSpectrum) 1).app _

 137 private def normalized_g_raw_cofiber_iso (T : TriangleData model.auxiliary)
 138     (he : (normalizedExponent model.classicalTower T.f : ℤ) +
 139       normalizedExponent model.classicalTower T.g + normalizedExponent model.classicalTower T.h = 1)
 140     (hT : NormalizedTriangleCompatible model T he) :
 141     HasFunctorialCofiber.cofib (model.normalizedMap T.Y T.Z T.g).map ≅
 142       (SyntheticCategory.biShift (0,-(normalizedExponent model.classicalTower T.h : ℤ))).obj
 143         ((SyntheticCategory.biShift (1,1)).obj
 144           (model.nu.functor.obj (T.X.obj model.auxiliary))) :=
 145   (SyntheticCategory.biShift_fullyFaithful
 146     (0,-(normalizedExponent model.classicalTower T.g : ℤ))).preimageIso
 147       (normalized_g_shifted_cofiber_iso T he hT ≪≫ (normalized_connecting_tail_iso T he).symm)

 149 private theorem normalized_g_raw_cofiber_inclusion (T : TriangleData model.auxiliary)
 150     (he : (normalizedExponent model.classicalTower T.f : ℤ) +
 151       normalizedExponent model.classicalTower T.g + normalizedExponent model.classicalTower T.h = 1)
 152     (hT : NormalizedTriangleCompatible model T he) :
 153     HasFunctorialCofiber.cofibι (model.normalizedMap T.Y T.Z T.g).map ≫
 154       (normalized_g_raw_cofiber_iso T he hT).hom =
 155       negativeLift (normalizedExponent model.classicalTower T.h)
 156         (model.normalizedMap T.Z (.shift 1 T.X) T.h).map ≫
 157       (SyntheticCategory.biShift (0,-(normalizedExponent model.classicalTower T.h : ℤ))).map
 158         (model.nu.suspensionIso (T.X.obj model.auxiliary)).hom := by
 159   apply (SyntheticCategory.biShift_fullyFaithful
 160     (0,-(normalizedExponent model.classicalTower T.g : ℤ))).map_injective
 161   simp only [Functor.map_comp,normalized_g_raw_cofiber_iso,Functor.FullyFaithful.preimageIso_hom,
 162     Functor.FullyFaithful.map_preimage,Iso.trans_hom,Iso.symm_hom,←Category.assoc]
 163   rw [normalized_g_shifted_cofiber_inclusion]
 164   apply (cancel_mono (normalized_connecting_tail_iso T he).hom).mp
 165   simp only [Category.assoc,Iso.inv_hom_id,Category.comp_id]
 166   rfl

 169     (he : (normalizedExponent model.classicalTower T.f : ℤ) +

 177 private theorem normalized_g_raw_cofiber_boundary (T : TriangleData model.auxiliary)
 178     (he : (normalizedExponent model.classicalTower T.f : ℤ) +
 179       normalizedExponent model.classicalTower T.g + normalizedExponent model.classicalTower T.h = 1)
 180     (hT : NormalizedTriangleCompatible model T he) :
 181     letI := model.exactness.shiftCommShift
 182       (0,-(normalizedExponent model.classicalTower T.g : ℤ))
 183     let F := SyntheticCategory.biShift (Syn := SyntheticSpectrum)
 184       (0,-(normalizedExponent model.classicalTower T.g : ℤ))
 185     F.map (HasFunctorialCofiber.cofibδ (model.normalizedMap T.Y T.Z T.g).map) ≫
 186       (F.commShiftIso (1 : ℤ)).hom.app _ ≫
 187         (shiftFunctor SyntheticSpectrum (1 : ℤ)).map (normalized_g_source_iso T).hom =
 188       -(F.map (normalized_g_raw_cofiber_iso T he hT).hom ≫ (normalized_connecting_tail_iso T he).hom ≫
 189         (shiftFunctor SyntheticSpectrum (1 : ℤ)).map (model.normalizedMap T.X T.Y T.f).map) := by
 190   have h := normalized_g_shifted_cofiber_boundary T he hT
 191   rw [←normalized_g_raw_cofiber_shift T he hT] at h
 192   simpa only [Category.assoc] using h

 194 private def normalized_g_cofiber_iso (T : TriangleData model.auxiliary)
 195     (he : (normalizedExponent model.classicalTower T.f : ℤ) +
 196       normalizedExponent model.classicalTower T.g + normalizedExponent model.classicalTower T.h = 1)
 197     (hT : NormalizedTriangleCompatible model T he) :
 198     HasFunctorialCofiber.cofib (model.normalizedMap T.Y T.Z T.g).map ≅
 199       (SyntheticCategory.biShift
 200         (1,(normalizedExponent model.classicalTower T.f : ℤ) +
 201           normalizedExponent model.classicalTower T.g)).obj
 202         (model.nu.functor.obj (T.X.obj model.auxiliary)) :=
 203   normalized_g_raw_cofiber_iso T he hT ≪≫
 204     (biShiftAddIso (1,1) (0,-(normalizedExponent model.classicalTower T.h : ℤ))
 205       (1,(normalizedExponent model.classicalTower T.f : ℤ) +
 206         normalizedExponent model.classicalTower T.g) (by ext <;> simp only [Prod.fst_add, Prod.snd_add] <;> omega)).app _

 208 private theorem normalized_g_cofiber_inclusion (T : TriangleData model.auxiliary)
 209     (he : (normalizedExponent model.classicalTower T.f : ℤ) +
 210       normalizedExponent model.classicalTower T.g + normalizedExponent model.classicalTower T.h = 1)
 211     (hT : NormalizedTriangleCompatible model T he) :
 212     HasFunctorialCofiber.cofibι (model.normalizedMap T.Y T.Z T.g).map ≫
 213       (normalized_g_cofiber_iso T he hT).hom =
 214       negativeLift (normalizedExponent model.classicalTower T.h)
 215         (model.normalizedMap T.Z (.shift 1 T.X) T.h).map ≫
 216       (SyntheticCategory.biShift (0,-(normalizedExponent model.classicalTower T.h : ℤ))).map
 217         (model.nu.suspensionIso (T.X.obj model.auxiliary)).hom ≫
 218       (biShiftAddIso (1,1) (0,-(normalizedExponent model.classicalTower T.h : ℤ))
 219         (1,(normalizedExponent model.classicalTower T.f : ℤ) +
 220           normalizedExponent model.classicalTower T.g) (by ext <;> simp only [Prod.fst_add, Prod.snd_add] <;> omega)).hom.app _ := by
 221   simp only [normalized_g_cofiber_iso,Iso.trans_hom,Iso.app_hom,←Category.assoc,
 222     normalized_g_raw_cofiber_inclusion]
