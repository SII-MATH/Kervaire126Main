  59 private theorem quotient_boundary_lift_desuspended {A B C : SyntheticSpectrum}
  60     (f : A ⟶ B) (g : B ⟶ C) (h : f ≫ g = 0)
  61     (i q : ℕ) (hi : 0 < i) (hq : 0 < q) (m w : ℤ)
  62     (a : BiHom m w (XModLambdaN B i))
  63     (b : BiHom (m-1) (w+i) (XModLambdaN A q))
  64     (hab : b ≫ XModLambdaN.map f q =
  65       biSuspensionHomEquiv m w i (XModLambdaN B q)
  66         (by simpa only [Nat.add_sub_cancel_left] using
  67           a ≫ ((model.quotients.tower B).triangle hi (show i < i+q by omega)).delta)) :
  68     ∃ c : BiHom m w (XModLambdaN C (i+q)),
  69       c ≫ (model.quotients.tower C).rho i (i+q) (by omega) =
  70         a ≫ XModLambdaN.map g i := by
  71   have lift_source (d q : ℕ) (hd : d=q)
  72       (δ : BiHom m w (((SyntheticCategory.biShift (0,-(i:ℤ))).obj (XModLambdaN B d))⟦(1:ℤ)⟧))
  73       (b : BiHom (m-1) (w+i) (XModLambdaN A q))
  74       (hb : b ≫ XModLambdaN.map f q =
  75         biSuspensionHomEquiv m w i (XModLambdaN B q) (by simpa only [hd] using δ)) :
  76       ∃ b' : BiHom m w (((SyntheticCategory.biShift (0,-(i:ℤ))).obj (XModLambdaN A d))⟦(1:ℤ)⟧),
  77         δ = b' ≫ (shiftFunctor SyntheticSpectrum (1:ℤ)).map
  78           ((SyntheticCategory.biShift (0,-(i:ℤ))).map (XModLambdaN.map f d)) := by
  79     subst q
  80     refine ⟨(biSuspensionHomEquiv m w i (XModLambdaN A d)).symm b, ?_⟩
  81     apply (biSuspensionHomEquiv m w i (XModLambdaN B d)).injective
  82     rw [bi_susp_natural model.quotients.coherence, AddEquiv.apply_symm_apply]
  83     exact hb.symm
  84   obtain ⟨b',hb'⟩ := lift_source (i+q-i) q (Nat.add_sub_cancel_left i q)
  85     (a ≫ ((model.quotients.tower B).triangle hi (show i < i+q by omega)).delta) b hab
  86   exact quotient_boundary_lift f g h i (i+q) hi (by omega) a b' hb'

  91 private theorem quotient_lift_image_division {A B C S : SyntheticSpectrum}
  92     (g : A ⟶ B) (h : B ⟶ C) (hgh : g ≫ h = 0)
  93     (i j : ℕ) (hi : 0 < i) (hij : i < j)
  94     (a : S ⟶ XModLambdaN A i) (c : S ⟶ XModLambdaN B j)
  95     (hc : c ≫ (model.quotients.tower B).rho i j hij.le = a ≫ XModLambdaN.map g i) :
  96     ∃ b : S ⟶ (SyntheticCategory.biShift (0,-(i:ℤ))).obj (XModLambdaN C (j-i)),
  97       b ≫ ((model.quotients.tower C).triangle hi hij).lambdaMap =
  98         c ≫ XModLambdaN.map h j := by
  99   let T := (model.quotients.tower C).triangle hi hij
 100   have hz : (c ≫ XModLambdaN.map h j) ≫ T.rho = 0 := by
 101     dsimp only [T]
 102     rw [(model.quotients.tower C).triangle_rho hi hij]
 103     rw [Category.assoc,←(model.quotients.natural h).rho_naturality hij.le,
 104       ←Category.assoc,hc,Category.assoc,quotient_map_comp_zero g h hgh,comp_zero]
 105   obtain ⟨b,hb⟩ := (Triangle.mk T.lambdaMap T.rho T.delta).coyoneda_exact₂
 106     T.distinguished (c ≫ XModLambdaN.map h j) hz
 107   exact ⟨b,hb.symm⟩

 110 /-- The quotient filling and its complementary triangle from three actual
 111 octahedra. The two fillings share their middle cofiber, and the last triangle
 112 retains the signed comparison required by the common-lift argument. -/
 113 private theorem quotient_octahedron_fill {A B : SyntheticSpectrum}
 114     (f : A ⟶ B) (q : ℕ) :
 115     ∃ u : XModLambdaN A q ⟶ XModLambdaN B q,
 116       XModLambdaN.incl A q ≫ u = f ≫ XModLambdaN.incl B q ∧
 117       u ≫ XModLambdaN.proj B q = XModLambdaN.proj A q ≫
 118         (shiftFunctor SyntheticSpectrum (1:ℤ)).map
 119           ((SyntheticCategory.biShift (0,-(q:ℤ))).map f) ∧
 120       ∃ (l : HasFunctorialCofiber.cofib ((SyntheticCategory.biShift (0,-(q:ℤ))).map f) ⟶
 121             HasFunctorialCofiber.cofib f)
 122         (v : HasFunctorialCofiber.cofib f ⟶ HasFunctorialCofiber.cofib u)
 123         (w : HasFunctorialCofiber.cofib u ⟶
 124           (HasFunctorialCofiber.cofib ((SyntheticCategory.biShift (0,-(q:ℤ))).map f))⟦(1:ℤ)⟧),
 125         HasFunctorialCofiber.cofibι ((SyntheticCategory.biShift (0,-(q:ℤ))).map f) ≫ l =
 126           lambdaPow q B ≫ HasFunctorialCofiber.cofibι f ∧
 127         l ≫ HasFunctorialCofiber.cofibδ f =
 128           HasFunctorialCofiber.cofibδ ((SyntheticCategory.biShift (0,-(q:ℤ))).map f) ≫
 129             (shiftFunctor SyntheticSpectrum (1:ℤ)).map (lambdaPow q A) ∧
 130         HasFunctorialCofiber.cofibι f ≫ v =
 131           XModLambdaN.incl B q ≫ HasFunctorialCofiber.cofibι u ∧
 132         v ≫ HasFunctorialCofiber.cofibδ u =
 133           HasFunctorialCofiber.cofibδ f ≫ (shiftFunctor SyntheticSpectrum (1:ℤ)).map (XModLambdaN.incl A q) ∧
 134         HasFunctorialCofiber.cofibι u ≫ w = XModLambdaN.proj B q ≫
 135           (shiftFunctor SyntheticSpectrum (1:ℤ)).map
 136             (HasFunctorialCofiber.cofibι ((SyntheticCategory.biShift (0,-(q:ℤ))).map f)) ∧
 137         Triangle.mk v w (-(shiftFunctor SyntheticSpectrum (1:ℤ)).map l) ∈ distTriang SyntheticSpectrum := by
 138   letI := model.exactness.syntheticTriangulated
 139   let F := SyntheticCategory.biShift (Syn:=SyntheticSpectrum) (0,-(q:ℤ))
 140   let c := lambdaPow q A ≫ f
 141   let O := Triangulated.someOctahedron rfl
 142     (XModLambdaN.cofiberTriangle_distinguished A q)
 143     (HasFunctorialCofiber.cofib_distinguished f)
 144     (HasFunctorialCofiber.cofib_distinguished c)
 145   let O' := Triangulated.someOctahedron (lambdaPow_naturality q f)
 146     (HasFunctorialCofiber.cofib_distinguished (F.map f))
 147     (XModLambdaN.cofiberTriangle_distinguished B q)
 148     (HasFunctorialCofiber.cofib_distinguished c)
 149   let u := O.m₁ ≫ O'.m₃
 150   let l := O'.m₁ ≫ O.m₃
 151   let O'' := Triangulated.someOctahedron rfl O.mem
 152     (rot_of_distTriang O'.triangle O'.mem)
 153     (HasFunctorialCofiber.cofib_distinguished u)
 154   refine ⟨u, ?_, ?_, l, O''.m₁, O''.m₃, ?_, ?_, ?_, ?_, ?_, ?_⟩
 155   · dsimp [u]
 156     rw [← Category.assoc, O.comm₁, Category.assoc, O'.comm₃]
 157   · dsimp [u]
 158     rw [Category.assoc, ← O'.comm₄, ← Category.assoc, O.comm₂]
 159   · dsimp [l]
 160     rw [← Category.assoc, O'.comm₁, Category.assoc, O.comm₃]
 161   · dsimp [l]
 162     rw [Category.assoc, ← O.comm₄, ← Category.assoc, O'.comm₂]
 163   · calc
 164       _ = (HasFunctorialCofiber.cofibι c ≫ O.m₃) ≫ O''.m₁ := by rw [O.comm₃]
 165       _ = _ := by
 166         rw [Category.assoc, O''.comm₁]
 167         change HasFunctorialCofiber.cofibι c ≫ O'.m₃ ≫ _ = _
 168         rw [← Category.assoc, O'.comm₃]
 169   · exact O''.comm₂
 170   · exact O''.comm₃
 171   · simpa only [Triangulated.Octahedron.triangle, Triangle.rotate, Triangle.mk,
 172       Preadditive.neg_comp, ← Functor.map_comp] using O''.mem

 202 /- The two actual tensor equations, with the rotated minus signs expanded.
 203 Write N=(A--f→B--g→C--h→ΣA), L=(F_-i Qq1--λ→Qr1--ρ→Qi1).
 204 Then a is the suspended [y] slot B⊗ΣF_-i Qq1; b is [xBar] in C⊗Qi1.
 205 The result is c∈π_n(ΣA⊗Qr1), c(ΣA⊗ρ)=b(h⊗Qi1), and
 206 c(Σf⊗Qr1)κ_right = -a(B⊗Σλ)κ_left.
 207 The negative sign is genuine under the registered May conventions. -/
 208 private theorem may_tensor_common_lift
 209     (T : TriangleData model.auxiliary)
 210     (he : (normalizedExponent model.classicalTower T.f : ℤ) +
 211       normalizedExponent model.classicalTower T.g +
 212       normalizedExponent model.classicalTower T.h = 1)
 213     (hT : KIP126.Literature.NormalizedTriangleCompatible model T he)
 214     (i q : ℕ) (hi : 0 < i) (hq : 0 < q)
 215     (n : ℤ)
 216     (a : HomotopyGroup n ((may_normalized_rotated T he hT).X ⊗
 217       (may_relative_unit_rotated i q hi hq).Z))
 218     (b : HomotopyGroup n ((may_normalized_rotated T he hT).Y ⊗
 219       (may_relative_unit_rotated i q hi hq).Y))
 220     (hab : a ≫ ((may_normalized_rotated T he hT).f ▷
 221         (may_relative_unit_rotated i q hi hq).Z) =
 222       b ≫ ((may_normalized_rotated T he hT).Y ◁
 223         (may_relative_unit_rotated i q hi hq).g)) :
 224     let N := KIP126.Literature.normalizedTriangle model T he
 225     let L := (model.quotients.tower (𝟙_ SyntheticSpectrum)).triangle hi
 226       (show i < i+q by omega)
 227     let U := may_relative_unit_rotated i q hi hq
 228     letI := KIP126.Literature.Standard.may_leftShift
 229     letI := KIP126.Literature.Standard.may_rightShift
 230     ∃ c : HomotopyGroup n (N.obj₁⟦(1:ℤ)⟧ ⊗ U.X),
 231       c ≫ (N.obj₁⟦(1:ℤ)⟧ ◁ L.rho) = b ≫ (N.mor₃ ▷ U.Y) ∧
 232       c ≫ (tensorRight U.X).map ((shiftFunctor SyntheticSpectrum (1:ℤ)).map N.mor₁) ≫
 233           ((tensorRight U.X).commShiftIso (1:ℤ)).hom.app N.obj₂ =
 234         -(a ≫ (tensorLeft N.obj₂).map
 235           ((shiftFunctor SyntheticSpectrum (1:ℤ)).map L.lambdaMap) ≫
 236             ((tensorLeft N.obj₂).commShiftIso (1:ℤ)).hom.app U.X) := by
 237   letI := KIP126.Literature.Standard.may_leftShift
 238   letI := KIP126.Literature.Standard.may_rightShift
 239   letI := KIP126.Literature.Standard.may_leftExact
 240   letI := KIP126.Literature.Standard.may_rightExact
 241   obtain ⟨P⟩ := KIP126.Literature.Standard.may_pushpull
 242     (may_normalized_rotated T he hT) (may_relative_unit_rotated i q hi hq)
 243   obtain ⟨v,hva,hvb⟩ := P.lift n a b hab
 244   refine ⟨v ≫ P.j3, ?_, ?_⟩
 245   · change (v ≫ P.j3) ≫ ((may_normalized_rotated T he hT).Z ◁
 246       (may_relative_unit_rotated i q hi hq).f) =
 247         b ≫ ((may_normalized_rotated T he hT).g ▷ (may_relative_unit_rotated i q hi hq).Y)
 248     rw [←hvb]
 249     simp only [Category.assoc, P.other_square]
 250   · have hb := congrArg (fun z => v ≫ z) P.boundary
 251     rw [Preadditive.comp_neg, ←Category.assoc, hva] at hb
 252     dsimp only [HoCofiberSequence.map, may_normalized_rotated, may_relative_unit_rotated] at hb
 253     simp only [Functor.map_neg, Preadditive.neg_comp, Preadditive.comp_neg,
 254       neg_neg] at hb
 255     simpa only [may_relative_unit_rotated, Category.assoc] using hb.symm
