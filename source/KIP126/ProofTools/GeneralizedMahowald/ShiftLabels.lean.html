  18 private theorem finite_zero_shift_label (X : ClassicalObject) (q : ℕ)
  19     (s t : ℤ) (k : ℕ) (x : Ambient model X (s,t)) :
  20     familyPageMap model.family
  21       (XModLambdaN.map (SyntheticCategory.biShift_zero.hom.app
  22         (model.nu.functor.obj (X.obj model.auxiliary))) q) 2 (s,t,t-k)
  23       (regradeE2 (by simp) (finiteNuLabel model X q 0 s t k x)) =
  24         finiteTargetLabel model X q s t k x := by
  25   unfold finiteNuLabel finiteTargetLabel targetNuLabel
  26   rw [page_regrade]
  27   have regrade_cast {B : SyntheticAdamsSS.{0}} {i j : Tridegree} (h : i=j) (z : B.E₂ i) :
  28       regradeE2 h z = (congrArg (fun k => (B.E₂ k : Type)) h).mp z := by
  29     subst j
  30     rfl
  31   rw [regrade_cast]
  32   let A := model.nu.functor.obj (X.obj model.auxiliary)
  33   let z : (model.family.obj ((SyntheticCategory.biShift (0,0)).obj A)).E₂ (s,t,t-k) :=
  34     (congrArg (fun i => ((model.family.obj ((SyntheticCategory.biShift (0,0)).obj A)).E₂ i : Type))
  35       (show (s,t,t+0-k)=(s,t,t-k) by simp)).mp (model.comparison.nuE2 X 0 s t k x)
  36   change familyPageMap model.family (XModLambdaN.map (SyntheticCategory.biShift_zero.hom.app A) q) 2 (s,t,t-k)
  37       (familyPageMap model.family (XModLambdaN.incl _ q) 2 (s,t,t-k) z) =
  38     familyPageMap model.family (XModLambdaN.incl A q) 2 (s,t,t-k)
  39       (familyPageMap model.family (SyntheticCategory.biShift_zero.hom.app A) 2 (s,t,t-k) z)
  40   rw [← ModuleCat.comp_apply, family_page_comp, ← XModLambdaN.incl_naturality]
  41   rw [← family_page_comp, ModuleCat.comp_apply]
  42   rfl

  45 private theorem finite_zero_shift_detects (X : ClassicalObject) (q : ℕ)
  46     (s t : ℤ) (k : ℕ) (x : Ambient model X (s,t))
  47     (a : BiHom (t-s) (t-k) (XModLambdaN
  48       ((SyntheticCategory.biShift (0,0)).obj (model.nu.functor.obj (X.obj model.auxiliary))) q))
  49     (ha : Detects (model.convergence (.quotient q (.shift (0,0) (.nu X)))) (s,t,t-k)
  50       (regradeE2 (by simp) (finiteNuLabel model X q 0 s t k x)) a) :
  51     Detects (model.convergence (.quotient q (.nu X))) (s,t,t-k)
  52       (finiteTargetLabel model X q s t k x)
  53       (a ≫ XModLambdaN.map (SyntheticCategory.biShift_zero.hom.app _) q) := by
  54   have h := detects_map (.quotient q (.shift (0,0) (.nu X))) (.quotient q (.nu X))
  55     (XModLambdaN.map (SyntheticCategory.biShift_zero.hom.app _) q) (s,t,t-k) _ a ha
  56   exact (congrArg (fun z => Detects (model.convergence (.quotient q (.nu X)))
  57     (s,t,t-k) z (a ≫ XModLambdaN.map (SyntheticCategory.biShift_zero.hom.app _) q))
  58       (finite_zero_shift_label X q s t k x)).mp h
  59 
  60 end
