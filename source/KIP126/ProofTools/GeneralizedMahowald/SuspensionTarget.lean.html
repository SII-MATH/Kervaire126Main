  21 private theorem first_label_inline (X : ClassicalObject) (s t : ℤ)
  22     (x : Ambient model X (s,t)) :
  23     firstLabel model X s t x =
  24       homotopyRegrade rfl (by simp) ((model.firstQuotient X 0 s t).symm x) := by
  25   have cast (m w w' : ℤ) (h : w=w') (A : SyntheticSpectrum) (a : BiHom m w A) :
  26       homotopyRegrade rfl h a = (congrArg (fun v => BiHom m v A) h).mp a := by
  27     subst w'; rfl
  28   exact (cast _ _ _ _ _ _).symm

  30 private theorem suspension_first_unshift
  31     (X : ClassicalObject) (s t : ℤ)
  32     (x : Ambient model X (s,t)) (sx : Ambient model (.shift 1 X) (s,t+1))
  33     (ε : XModLambdaN ((SyntheticCategory.biShift (1,1)).obj
  34         (model.nu.functor.obj (X.obj model.auxiliary))) 1 ≅
  35       (SyntheticCategory.biShift (1,1)).obj
  36         (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) 1))
  37     (hε : XModLambdaN.incl _ 1 ≫ ε.hom =
  38       (SyntheticCategory.biShift (1,1)).map (XModLambdaN.incl _ 1))
  39     (hsx : (model.classicalSuspension X).DesuspendsClass s (t+1) sx (by simpa using x)) :
  40     quotientUnshift X (1,1) (t-s,t) 1 ε
  41       (homotopyRegrade (by dsimp; omega) rfl
  42         ((firstLabel model (.shift 1 X) s (t+1) sx ≫
  43           XModLambdaN.map (SyntheticCategory.biShift_zero.hom.app _) 1) ≫
  44           XModLambdaN.map (model.nu.suspensionIso (X.obj model.auxiliary)).hom 1)) =
  45       firstLabel model X s t x ≫ XModLambdaN.map (SyntheticCategory.biShift_zero.hom.app _) 1 := by
  46   have hf := firstQuotient_suspension X s t x sx ε hε hsx
  47   rw [←first_label_inline,←first_label_inline] at hf
  48   have hc : homotopyRegrade (show (t+1)-s=((t-s,t)+(1,1)).1 by dsimp; omega) rfl
  49       ((firstLabel model (.shift 1 X) s (t+1) sx ≫
  50         XModLambdaN.map (SyntheticCategory.biShift_zero.hom.app _) 1) ≫
  51         XModLambdaN.map (model.nu.suspensionIso (X.obj model.auxiliary)).hom 1) =
  52       shiftHomotopyClass (1,1) (t-s,t)
  53         (firstLabel model X s t x ≫ XModLambdaN.map (SyntheticCategory.biShift_zero.hom.app _) 1) ≫ ε.inv := by
  54     apply (cancel_mono ε.hom).mp
  55     rw [←regrade_comp,Category.assoc,hf,homotopy_regrade_inverse,Category.assoc,Iso.inv_hom_id,Category.comp_id]
  56   rw [hc,quotient_unshift_shift]

  58 private theorem suspension_finite_target_desuspend
  59     (X : ClassicalObject) (q : ℕ) (hq : 0<q) (s t : ℤ)
  60     (x : Ambient model X (s,t)) (sx : Ambient model (.shift 1 X) (s,t+1))
  61     (ε₁ : XModLambdaN ((SyntheticCategory.biShift (1,1)).obj
  62         (model.nu.functor.obj (X.obj model.auxiliary))) 1 ≅
  63       (SyntheticCategory.biShift (1,1)).obj
  64         (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) 1))
  65     (εq : XModLambdaN ((SyntheticCategory.biShift (1,1)).obj
  66         (model.nu.functor.obj (X.obj model.auxiliary))) q ≅
  67       (SyntheticCategory.biShift (1,1)).obj
  68         (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) q))
  69     (hε₁ : XModLambdaN.incl _ 1 ≫ ε₁.hom =
  70       (SyntheticCategory.biShift (1,1)).map (XModLambdaN.incl _ 1))
  71     (hεq : XModLambdaN.incl _ q ≫ εq.hom =
  72       (SyntheticCategory.biShift (1,1)).map (XModLambdaN.incl _ q))
  73     (hsx : (model.classicalSuspension X).DesuspendsClass s (t+1) sx (by simpa using x))
  74     (a : BiHom ((t+1)-s) (t+1)
  75       (XModLambdaN (model.nu.functor.obj ((ClassicalObject.shift 1 X).obj model.auxiliary)) q))
  76     (ha : a ≫ (model.quotients.tower _).rho 1 q hq =
  77       firstLabel model (.shift 1 X) s (t+1) sx ≫
  78         XModLambdaN.map (SyntheticCategory.biShift_zero.hom.app _) 1) :
  79     let a₀ := quotientUnshift X (1,1) (t-s,t) q εq
  80       (homotopyRegrade (by dsimp; omega) rfl
  81         (a ≫ XModLambdaN.map (model.nu.suspensionIso (X.obj model.auxiliary)).hom q))
  82     a₀ ≫ (model.quotients.tower _).rho 1 q hq =
  83       firstLabel model X s t x ≫ XModLambdaN.map (SyntheticCategory.biShift_zero.hom.app _) 1 ∧
  84     Detects (model.convergence (.quotient q (.nu X))) (s,t,t)
  85       (regradeE2 (by simp) (finiteTargetLabel model X q s t 0 x)) a₀ := by
  86   dsimp only
  87   have hfirst : (a ≫ XModLambdaN.map (model.nu.suspensionIso (X.obj model.auxiliary)).hom q) ≫
  88       (model.quotients.tower _).rho 1 q hq =
  89       (firstLabel model (.shift 1 X) s (t+1) sx ≫
  90         XModLambdaN.map (SyntheticCategory.biShift_zero.hom.app _) 1) ≫
  91         XModLambdaN.map (model.nu.suspensionIso (X.obj model.auxiliary)).hom 1 := by
  92     rw [Category.assoc,←(model.quotients.natural
  93       (model.nu.suspensionIso (X.obj model.auxiliary)).hom).rho_naturality,
  94       ←Category.assoc]
  95     erw [ha]
  96   have h : quotientUnshift X (1,1) (t-s,t) q εq
  97       (homotopyRegrade (by dsimp; omega) rfl
  98         (a ≫ XModLambdaN.map (model.nu.suspensionIso (X.obj model.auxiliary)).hom q)) ≫
  99       (model.quotients.tower _).rho 1 q hq =
 100       firstLabel model X s t x ≫ XModLambdaN.map (SyntheticCategory.biShift_zero.hom.app _) 1 := by
 101     rw [quotient_unshift_restriction X (1,1) (t-s,t) 1 q (by decide) hq ε₁ εq hε₁ hεq,
 102       ←regrade_comp,hfirst,suspension_first_unshift X s t x sx ε₁ hε₁ hsx]
 103   exact ⟨h,first_label_detects_unshifted X q hq s t x _ h⟩

 105 private theorem suspension_finite_target_unshifted
 106     (X : ClassicalObject) (q : ℕ) (hq : 0<q) (s t : ℤ)
 107     (x : Ambient model X (s,t)) (sx : Ambient model (.shift 1 X) (s,t+1))
 108     (hsx : (model.classicalSuspension X).DesuspendsClass s (t+1) sx (by simpa using x))
 109     (a : BiHom ((t+1)-s) (t+1)
 110       (XModLambdaN (model.nu.functor.obj ((ClassicalObject.shift 1 X).obj model.auxiliary)) q))
 111     (ha : a ≫ (model.quotients.tower _).rho 1 q hq =
 112       firstLabel model (.shift 1 X) s (t+1) sx ≫
 113         XModLambdaN.map (SyntheticCategory.biShift_zero.hom.app _) 1) :
 114     ∃ (εq : XModLambdaN ((SyntheticCategory.biShift (1,1)).obj
 115           (model.nu.functor.obj (X.obj model.auxiliary))) q ≅
 116         (SyntheticCategory.biShift (1,1)).obj
 117           (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) q)),
 118       XModLambdaN.incl _ q ≫ εq.hom =
 119         (SyntheticCategory.biShift (1,1)).map (XModLambdaN.incl _ q) ∧
 120       ∃ a₀ : BiHom (t-s) t (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) q),
 121         a₀ ≫ (model.quotients.tower _).rho 1 q hq =
 122           firstLabel model X s t x ≫ XModLambdaN.map (SyntheticCategory.biShift_zero.hom.app _) 1 ∧
 123         Detects (model.convergence (.quotient q (.nu X))) (s,t,t)
 124           (regradeE2 (by simp) (finiteTargetLabel model X q s t 0 x)) a₀ ∧
 125         homotopyRegrade (by dsimp; omega) rfl (shiftHomotopyClass (1,1) (t-s,t) a₀) ≫ εq.inv =
 126           a ≫ XModLambdaN.map (model.nu.suspensionIso (X.obj model.auxiliary)).hom q := by
 127   letI := model.exactness.shiftCommShift (1,1)
 128   letI := model.exactness.shiftExact (1,1)
 129   obtain ⟨ε₁,hε₁,_⟩ := quotient_shift_exists model.quotients.coherence (1,1) 1
 130     (model.nu.functor.obj (X.obj model.auxiliary))
 131   obtain ⟨εq,hεq,_⟩ := quotient_shift_exists model.quotients.coherence (1,1) q
 132     (model.nu.functor.obj (X.obj model.auxiliary))
 133   have h := suspension_finite_target_desuspend X q hq s t x sx ε₁ εq hε₁ hεq hsx a ha
 134   refine ⟨εq,hεq,_,h.1,h.2,?_⟩
 135   rw [←regrade_comp,shift_quotient_unshift,homotopy_regrade_inverse]
 136 
 137 end
