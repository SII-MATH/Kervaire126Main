  26 /- The final face of an octahedron is a homotopy pullback on represented
  27 Hom groups. The correction uses the same octahedron, not an arbitrary
  28 second preimage. -/
  29 private theorem octahedron_common_lift
  30     {A B C D E V S : SyntheticSpectrum}
  31     {u : A ⟶ B} {l : B ⟶ C} {f : A ⟶ C} {hcomp : u≫l=f}
  32     {j : B ⟶ D} {h : D ⟶ A⟦(1:ℤ)⟧}
  33     {rho : C ⟶ E} {delta : E ⟶ B⟦(1:ℤ)⟧}
  34     {incl : C ⟶ V} {proj : V ⟶ A⟦(1:ℤ)⟧}
  35     {hU : Triangle.mk u j h ∈ distTriang SyntheticSpectrum}
  36     {hL : Triangle.mk l rho delta ∈ distTriang SyntheticSpectrum}
  37     {hF : Triangle.mk f incl proj ∈ distTriang SyntheticSpectrum}
  38     (O : Triangulated.Octahedron hcomp hU hL hF)
  39     (a : S ⟶ E) (b : S ⟶ A⟦(1:ℤ)⟧)
  40     (hab : a≫delta=b≫(shiftFunctor SyntheticSpectrum (1:ℤ)).map u) :
  41     ∃ v : S ⟶ V, v≫O.m₃=a ∧ v≫proj=b := by
  42   have huj : u ≫ j = 0 := comp_distTriang_mor_zero₁₂ _ hU
  43   have haz : a ≫ (delta ≫ (shiftFunctor SyntheticSpectrum (1:ℤ)).map j)=0 := by
  44     rw [←Category.assoc,hab,Category.assoc,←Functor.map_comp,
  45       huj,Functor.map_zero,comp_zero]
  46   obtain ⟨v0,hv0⟩ := O.triangle.coyoneda_exact₃ O.mem a haz
  47   change S ⟶ V at v0
  48   change a = v0 ≫ O.m₃ at hv0
  49   let e := b-v0≫proj
  50   have he : e ≫ (-((shiftFunctor SyntheticSpectrum (1:ℤ)).map u))=0 := by
  51     simp only [e,Preadditive.comp_neg,Preadditive.sub_comp,Category.assoc,O.comm₄]
  52     rw [←Category.assoc,←hv0,←hab,sub_self,neg_zero]
  53   obtain ⟨c,hc⟩ := (Triangle.mk u j h).rotate.coyoneda_exact₃
  54     (rot_of_distTriang _ hU) e he
  55   change S ⟶ D at c
  56   change e=c≫h at hc
  57   have hzero : O.m₁ ≫ O.m₃ = 0 := comp_distTriang_mor_zero₁₂ _ O.mem
  58   refine ⟨v0+c≫O.m₁,?_,?_⟩
  59   · rw [Preadditive.add_comp,Category.assoc,
  60       hzero,comp_zero,add_zero]
  61     exact hv0.symm
  62   · rw [Preadditive.add_comp,Category.assoc,O.comm₂,←hc]
  63     dsimp only [e]
  64     abel

  66 /- Actual finite quotient instance. V uses only Q_q(G), lambda_i and the
  67 chosen cofiber; its pair of output maps is fixed before a and b are chosen. -/
  68 private theorem finite_quotient_octahedral_pullback {A B : SyntheticSpectrum}
  69     (G : A ⟶ B) (i q : ℕ) (hi : 0 < i) (hq : 0 < q) :
  70     let F := SyntheticCategory.biShift (Syn:=SyntheticSpectrum) (0,-(i:ℤ))
  71     let L := (model.quotients.tower B).triangle hi (show i < i+q by omega)
  72     let u := F.map (XModLambdaN.map G (i+q-i))
  73     let V := HasFunctorialCofiber.cofib (u≫L.lambdaMap)
  74     ∃ p : V ⟶ XModLambdaN B i,
  75       HasFunctorialCofiber.cofibι (u≫L.lambdaMap) ≫ p=L.rho ∧
  76       HasFunctorialCofiber.cofibδ (u≫L.lambdaMap) ≫
  77           (shiftFunctor SyntheticSpectrum (1:ℤ)).map u=p≫L.delta ∧
  78       ∀ {S : SyntheticSpectrum} (a : S ⟶ XModLambdaN B i)
  79         (b : S ⟶ (F.obj (XModLambdaN A (i+q-i)))⟦(1:ℤ)⟧),
  80         a≫L.delta=b≫(shiftFunctor SyntheticSpectrum (1:ℤ)).map u →
  81         ∃ v : S ⟶ V,
  82           v≫p=a ∧ v≫HasFunctorialCofiber.cofibδ (u≫L.lambdaMap)=b := by
  83   dsimp only
  84   letI := model.exactness.syntheticTriangulated
  85   let F := SyntheticCategory.biShift (Syn:=SyntheticSpectrum) (0,-(i:ℤ))
  86   let L := (model.quotients.tower B).triangle hi (show i < i+q by omega)
  87   let u := F.map (XModLambdaN.map G (i+q-i))
  88   let O := Triangulated.someOctahedron rfl
  89     (HasFunctorialCofiber.cofib_distinguished u) L.distinguished
  90     (HasFunctorialCofiber.cofib_distinguished (u≫L.lambdaMap))
  91   refine ⟨O.m₃,O.comm₃,O.comm₄,?_⟩
  92   intro S a b hab
  93   exact octahedron_common_lift O a b hab

 155 private theorem finite_quotient_octahedral_comparison
 156     {A B C D : SyntheticSpectrum} (G : A ⟶ B) (j : B ⟶ C) (h : C ⟶ D)
 157     (hGj : G≫j=0) (hjh : j≫h=0)

 159     let F := SyntheticCategory.biShift (Syn:=SyntheticSpectrum) (0,-(i:ℤ))
 160     let L := (model.quotients.tower B).triangle hi (show i < i+q by omega)
 161     let u := F.map (XModLambdaN.map G (i+q-i))
 162     let f := u≫L.lambdaMap

 170         HasFunctorialCofiber.cofibδ f≫K=φ≫XModLambdaN.map h (i+q) ∧
 171         φ≫(model.quotients.tower C).rho i (i+q) (by omega) -
 172           p≫XModLambdaN.map j i = HasFunctorialCofiber.cofibδ f≫z := by
 173   dsimp only
 174   intro p hp
 175   let F := SyntheticCategory.biShift (Syn:=SyntheticSpectrum) (0,-(i:ℤ))
 176   let L := (model.quotients.tower B).triangle hi (show i < i+q by omega)
 177   let u := F.map (XModLambdaN.map G (i+q-i))
 178   let f := u≫L.lambdaMap
 179   let T := Triangle.mk f (HasFunctorialCofiber.cofibι f) (HasFunctorialCofiber.cofibδ f)
 180   have hT := HasFunctorialCofiber.cofib_distinguished f
 181   have hzero : f≫XModLambdaN.map j (i+q)=0 := by
 182     dsimp only [f,u,F]
 183     rw [(model.quotients.natural G).lambda_naturality hi (show i < i+q by omega),

 187   change XModLambdaN.map j (i+q)=HasFunctorialCofiber.cofibι f≫φ at hφ
 188   have hz₂ : HasFunctorialCofiber.cofibι f≫(φ≫XModLambdaN.map h (i+q))=0 := by
 189     rw [←Category.assoc,←hφ,quotient_map_comp_zero j h hjh]
 190   obtain ⟨K,hK⟩ := T.yoneda_exact₃ hT (φ≫XModLambdaN.map h (i+q)) hz₂
 191   change φ≫XModLambdaN.map h (i+q)=HasFunctorialCofiber.cofibδ f≫K at hK
 192   have hz₃ : HasFunctorialCofiber.cofibι f≫
 193       (φ≫(model.quotients.tower C).rho i (i+q) (by omega)-p≫XModLambdaN.map j i)=0 := by
 194     rw [Preadditive.comp_sub,←Category.assoc,←hφ,←Category.assoc,hp,
 195       (model.quotients.tower B).triangle_rho hi (show i < i+q by omega),
 196       (model.quotients.natural j).rho_naturality,sub_self]
 197   obtain ⟨z,hz⟩ := T.yoneda_exact₃ hT _ hz₃
 198   exact ⟨φ,K,z,hφ.symm,hK.symm,hz⟩
