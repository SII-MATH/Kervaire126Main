  26       SyntheticCategory.biShift (1,0) := by

  68     (hε : XModLambdaN.incl _ q≫ε.hom=
  69       (SyntheticCategory.biShift (1,0)).map (XModLambdaN.incl _ q))
  70     (a : BiHom ((t-s,t-k)+(1,0)).1 ((t-s,t-k)+(1,0)).2
  71       (XModLambdaN (((SyntheticCategory.biShift (0,e) ⋙
  72         shiftFunctor SyntheticSpectrum (1:ℤ) ⋙ SyntheticCategory.biShift (0,-e)).obj
  73           (model.nu.functor.obj (X.obj model.auxiliary)))) q))
  74     (ha : Detects (model.convergence (.quotient q (.nu X))) (s,t,t-k)
  75       (finiteTargetLabel model X q s t k x)
  76       (quotientUnshift X (1,0) (t-s,t-k) q ε
  77         (a≫XModLambdaN.map ((canonicalSuspensionTransport e).hom.app _) q))) :
  78     Detects (model.convergence (.quotient q (.nu X))) (s,t,t-k)
  79       (finiteTargetLabel model X q s t k x)

  82   let η := (canonicalSuspensionTransport e).symm ≪≫ exactSuspensionTransport e

  84     (a≫XModLambdaN.map ((canonicalSuspensionTransport e).hom.app _) q) ha
  85   have hc : ((canonicalSuspensionTransport e).hom.app _) ≫ η.hom.app _ =
  86       (exactSuspensionTransport e).hom.app (model.nu.functor.obj (X.obj model.auxiliary)) := by
  87     simp [η]

 123 private theorem quotient_unshift_normalized_map
 124     (X Y : ClassicalObject) (k : ℕ) (hk : k≤1)
 125     (f : (SyntheticCategory.biShift (0,(k:ℤ))).obj
 126         (model.nu.functor.obj (X.obj model.auxiliary)) ⟶
 127       model.nu.functor.obj (Y.obj model.auxiliary))
 128     (δ p : ℤ×ℤ) (q : ℕ) (hq : 0<q)
 129     (εA : XModLambdaN ((SyntheticCategory.biShift δ).obj
 130         ((SyntheticCategory.biShift (0,(k:ℤ))).obj
 131           (model.nu.functor.obj (X.obj model.auxiliary)))) q ≅
 132       (SyntheticCategory.biShift δ).obj (XModLambdaN
 133         ((SyntheticCategory.biShift (0,(k:ℤ))).obj
 134           (model.nu.functor.obj (X.obj model.auxiliary))) q))
 135     (εB : XModLambdaN ((SyntheticCategory.biShift δ).obj
 136         (model.nu.functor.obj (Y.obj model.auxiliary))) q ≅
 137       (SyntheticCategory.biShift δ).obj
 138         (XModLambdaN (model.nu.functor.obj (Y.obj model.auxiliary)) q))
 139     (hεA : XModLambdaN.incl _ q≫εA.hom=
 140       (SyntheticCategory.biShift δ).map (XModLambdaN.incl _ q))
 141     (hεB : XModLambdaN.incl _ q≫εB.hom=
 142       (SyntheticCategory.biShift δ).map (XModLambdaN.incl _ q))
 143     (a : BiHom (p+δ).1 (p+δ).2
 144       (XModLambdaN ((SyntheticCategory.biShift δ).obj
 145         ((SyntheticCategory.biShift (0,(k:ℤ))).obj
 146           (model.nu.functor.obj (X.obj model.auxiliary)))) q)) :
 147     quotientUnshift Y δ p q εB
 148         (a≫XModLambdaN.map ((SyntheticCategory.biShift δ).map f) q) =
 149       quotientClassUnshift _ δ p q εA a ≫ XModLambdaN.map f q := by
 150   let F := SyntheticCategory.biShift (Syn:=SyntheticSpectrum) δ
 151   let ff := SyntheticCategory.biShift_fullyFaithful (Syn:=SyntheticSpectrum) δ
 152   let A := (SyntheticCategory.biShift (0,(k:ℤ))).obj
 153     (model.nu.functor.obj (X.obj model.auxiliary))
 154   let M := ff.preimage (εA.inv≫XModLambdaN.map (F.map f) q≫εB.hom)
 155   have hM : F.map M=εA.inv≫XModLambdaN.map (F.map f) q≫εB.hom := ff.map_preimage _
 156   dsimp only [F] at hM
 157   have hbase : XModLambdaN.incl A q≫M =
 158       XModLambdaN.incl A q≫XModLambdaN.map f q := by
 159     apply ff.map_injective
 160     rw [Functor.map_comp,Functor.map_comp,hM,←hεA]
 161     simp only [Category.assoc,Iso.hom_inv_id_assoc]
 162     rw [←Category.assoc,←XModLambdaN.incl_naturality,Category.assoc,hεB]
 163     rw [←Category.assoc,hεA,←Functor.map_comp,XModLambdaN.incl_naturality]
 164     simp only [Functor.map_comp]
 165   have heq := normalized_quotient_maps_equal_on_spheres X Y k hk q hq
 166     M (XModLambdaN.map f q) hbase p.1 p.2 (quotientClassUnshift A δ p q εA a)
 167   rw [←heq]
 168   apply ff.map_injective
 169   simp only [quotientUnshift,quotientClassUnshift,Functor.map_comp,
 170     Functor.FullyFaithful.map_preimage,hM,Category.assoc,Iso.hom_inv_id_assoc]

 189 private theorem normalized_g_boundary_suspended_f
 190     (T : TriangleData model.auxiliary)
