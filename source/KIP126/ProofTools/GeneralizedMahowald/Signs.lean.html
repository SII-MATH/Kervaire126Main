  17 private theorem cohomology_two (n : ℤ) (X : Spectrum)
  18     (a : ModTwoCohomology model n X) : a+a=0 := by
  19   apply (KIP126.Literature.Standard.universal_coefficient n X).injective
  20   simp only [map_add,map_zero]
  21   ext x
  22   simp only [AddMonoidHom.add_apply, AddMonoidHom.zero_apply]
  23   exact CharTwo.add_self_eq_zero _
  24 private theorem coefficient_identity_two :
  25     (𝟙 model.coefficients.object : model.coefficients.object ⟶ model.coefficients.object) +
  26       𝟙 model.coefficients.object = 0 := by
  27   let z : (shiftFunctor Spectrum (0:ℤ)).obj model.coefficients.object ⟶ model.coefficients.object :=
  28     (shiftFunctorZero Spectrum ℤ).hom.app model.coefficients.object
  29   have hz := cohomology_two 0 model.coefficients.object z
  30   apply (cancel_epi z).mp
  31   rw [Preadditive.comp_add,Category.comp_id,comp_zero]
  32   exact hz
  33 private theorem tensor_coefficient_identity_two (X : Spectrum) :
  34     (𝟙 (model.coefficients.object ⊗ X) : _ ⟶ _) + 𝟙 _ = 0 := by
  35   letI := closedBackground.tensorShift X
  36   letI := closedBackground.tensorExact X
  37   have h := congrArg (fun f => (tensorRight X).map f) coefficient_identity_two
  38   change (tensorRight X).map (𝟙 model.coefficients.object + 𝟙 model.coefficients.object) = (tensorRight X).map 0 at h
  39   rw [Functor.map_add,CategoryTheory.Functor.map_id,Functor.map_zero] at h
  40   exact h
  41 private theorem layer_identity_two (X : Spectrum) (s : ℤ) :
  42     (𝟙 ((model.classicalTower.tower X).layer s) : _ ⟶ _) + 𝟙 _ = 0 := by
  43   let T := model.classicalTower.tower X
  44   have cofib_transport (a b c d : ℕ) (ha : a=c) (hb : b=d)
  45       (h : a≤b) (h' : c≤d) :
  46       HasFunctorialCofiber.cofib (T.map h) = HasFunctorialCofiber.cofib (T.map h') := by
  47     subst c; subst d; rfl
  48   by_cases hs : 0 ≤ s
  49   · have he : T.layer s = HasFunctorialCofiber.cofib (T.map (Nat.le_succ s.toNat)) := by
  50       dsimp [AbstractAdamsTower.layer,AbstractAdamsTower.mapAt]
  51       have hs1 : (s+1).toNat=s.toNat+1 := by omega
  52       exact cofib_transport _ _ _ _ rfl hs1 _ _
  53     let e : T.layer s ≅ model.coefficients.object ⊗ T.stage s.toNat :=
  54       eqToIso he ≪≫ T.layerIso s.toNat
  55     apply (cancel_epi e.inv).mp
  56     apply (cancel_mono e.hom).mp
  57     change (e.inv ≫ (𝟙 (T.layer s) + 𝟙 (T.layer s))) ≫ e.hom = (e.inv ≫ 0) ≫ e.hom
  58     rw [Preadditive.comp_add,Preadditive.add_comp,Category.comp_id,
  59       Iso.inv_hom_id,comp_zero,zero_comp]
  60     exact tensor_coefficient_identity_two (T.stage s.toNat)
  61   · have hs0 : s.toNat=0 := by omega
  62     have hs10 : (s+1).toNat=0 := by omega
  63     have he : T.layer s = HasFunctorialCofiber.cofib (𝟙 (T.stage 0)) := by
  64       dsimp only [AbstractAdamsTower.layer,AbstractAdamsTower.mapAt]
  65       exact (cofib_transport _ _ _ _ hs0 hs10 _ (le_refl 0)).trans
  66         (congrArg HasFunctorialCofiber.cofib (T.map_self 0))
  67     have hz : IsZero (HasFunctorialCofiber.cofib (𝟙 (T.stage 0))) :=
  68       (Triangle.mk _ _ _).isZero₃_of_isIso₁ (HasFunctorialCofiber.cofib_distinguished (𝟙 (T.stage 0))) (by change IsIso (𝟙 (T.stage 0)); infer_instance)
  69     have hz' : IsZero (T.layer s) := he.symm ▸ hz
  70     exact hz'.eq_of_tgt _ _
  71 private theorem first_page_two (X : Spectrum) (p : ℤ×ℤ)
  72     (a : (model.classicalTower.tower X).FirstPage p) : a+a=0 := by
  73   have h := congrArg (fun f => a ≫ f) (layer_identity_two X p.1)
  74   simpa only [Preadditive.comp_add,Category.comp_id,comp_zero] using h
  75 private theorem ambient_two (X : Spectrum) (p : ℤ×ℤ)
  76     (a : ((model.classical.obj X).ssData p).V) : a+a=0 := by
  77   apply (ModuleCat.mono_iff_injective ((model.classicalTower.sequence X).ambient p).hom).mp inferInstance
  78   simp only [map_add,map_zero]
  79   apply Subtype.ext
  80   exact first_page_two X p (((model.classicalTower.sequence X).ambient p).hom a).val
  81 private theorem classical_page_two (X : Spectrum) (r : ℤ) (p : ℤ×ℤ)
  82     (a : (model.classical.obj X).Page r p) : a+a=0 := by
  83   let D := (model.classical.obj X).ssData p
  84   let j : WithTop ℕ := ↑(r-(model.classical.obj X).r₀).toNat
  85   haveI : Epi (D.pageπ j) := inferInstanceAs (Epi (cokernel.π _))
  86   obtain ⟨z,rfl⟩ := (ModuleCat.epi_iff_surjective (D.pageπ j)).mp inferInstance a
  87   have hz : z+z=0 := by
  88     apply (ModuleCat.mono_iff_injective (D.Z j).arrow).mp inferInstance
  89     simpa only [map_add,map_zero] using ambient_two X p ((D.Z j).arrow z)
  90   rw [← map_add,hz,map_zero]

  92 private theorem detects_neg {A : SyntheticObject} {i : Tridegree}
  93     {x : (model.family.obj (A.obj model.nu model.auxiliary)).E₂ i}
  94     {a : BiHom (i.2.1-i.1) i.2.2 (A.obj model.nu model.auxiliary)}
  95     (ha : Detects (model.convergence A) i x a) :
  96     Detects (model.convergence A) i (-x) (-a) := by
  97   obtain ⟨e, ⟨hr,z,hz,he⟩,a',ha',hga⟩ := ha
  98   refine ⟨-e,⟨hr,-z,?_,?_⟩,-a',?_,?_⟩
  99   · rw [map_neg,hz]
 100   · rw [map_neg,he]
 101   · rw [map_neg,ha']
 102   · rw [map_neg,map_neg,hga]
 103 private theorem finite_target_neg (X : ClassicalObject) (q : ℕ) (s t : ℤ) (k : ℕ)
 104     (x : Ambient model X (s,t)) :
 105     -(finiteTargetLabel model X q s t k x) = finiteTargetLabel model X q s t k x := by
 106   have hx : -x=x := neg_eq_of_add_eq_zero_left (classical_page_two (X.obj model.auxiliary) 2 (s,t) x)
 107   have hneg : finiteTargetLabel model X q s t k (-x) = -(finiteTargetLabel model X q s t k x) := by
 108     have cast_neg (A B : ModuleCat ℤ) (h : A=B) (he : (A:Type)=(B:Type)) (z : A) :
 109         he.mp (-z) = -(he.mp z) := by
 110       subst B
 111       rfl
 112     simp only [finiteTargetLabel,targetNuLabel,map_neg]
 113     rw [cast_neg _ _ (by congr 2; simp),map_neg,map_neg]
 114   exact hneg.symm.trans (congrArg (finiteTargetLabel model X q s t k) hx)
 115 private theorem finite_target_detects_neg (X : ClassicalObject) (q : ℕ) (s t : ℤ) (k : ℕ)
 116     (x : Ambient model X (s,t))
 117     (a : BiHom (t-s) (t-k) (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) q))
 118     (ha : Detects (model.convergence (.quotient q (.nu X))) (s,t,t-k)
 119       (finiteTargetLabel model X q s t k x) a) :
 120     Detects (model.convergence (.quotient q (.nu X))) (s,t,t-k)
 121       (finiteTargetLabel model X q s t k x) (-a) := by
 122   have h := detects_neg ha
 123   rw [finite_target_neg] at h
 124   exact h
