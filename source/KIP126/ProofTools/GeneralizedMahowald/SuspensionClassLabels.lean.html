  16 private theorem desuspend_bijective (X : C) (n : ℤ) :
  17     Function.Bijective (AbstractAdamsTower.desuspend X n) := by
  18   let e := (shiftFunctorAdd' C n (-1) (n - 1) (by omega)).app SphereSpectrum
  19   let d := (shiftFunctorCompIsoId C 1 (-1) (by omega)).app X
  20   constructor
  21   · intro a b h
  22     change e.hom ≫ (shiftFunctor C (-1 : ℤ)).map a ≫ d.hom =
  23       e.hom ≫ (shiftFunctor C (-1 : ℤ)).map b ≫ d.hom at h
  24     apply (shiftFunctor C (-1 : ℤ)).map_injective
  25     apply (cancel_epi e.hom).mp
  26     apply (cancel_mono d.hom).mp
  27     simpa only [Category.assoc] using h
  28   · intro b
  29     refine ⟨(shiftFunctor C (-1 : ℤ)).preimage (e.inv ≫ b ≫ d.inv), ?_⟩
  30     change e.hom ≫ (shiftFunctor C (-1 : ℤ)).map
  31       ((shiftFunctor C (-1 : ℤ)).preimage _) ≫ d.hom = b
  32     rw [Functor.map_preimage]
  33     simp only [Category.assoc, Iso.inv_hom_id, Category.comp_id, Iso.hom_inv_id_assoc]

  36 private theorem desuspend_postcompose {X Y : C} (f : X ⟶ Y) (n : ℤ)
  37     (a : HomotopyGroup n (X⟦(1 : ℤ)⟧)) :
  38     AbstractAdamsTower.desuspend Y n (a ≫ f⟦(1 : ℤ)⟧') = AbstractAdamsTower.desuspend X n a ≫ f := by
  39   change _ ≫ (shiftFunctor C (-1 : ℤ)).map (a ≫ f⟦(1 : ℤ)⟧') ≫ _ = _
  40   rw [Functor.map_comp]
  41   have h := (shiftFunctorCompIsoId C 1 (-1) (by omega)).hom.naturality f
  42   change (shiftFunctor C (-1 : ℤ)).map (f⟦(1 : ℤ)⟧') ≫ _ = _ ≫ f at h
  43   simp only [Category.assoc]
  44   rw [h]
  45   simp only [AbstractAdamsTower.desuspend, AddMonoidHom.coe_mk, ZeroHom.coe_mk, Category.assoc]

  47 private theorem raw_cycles_iff [HasFunctorialCofiber (C:=C)] {H : C} {unit : SphereSpectrum ⟶ H}
  48     {X : C} (T : AbstractAdamsTower unit X) (r : ℕ) (hr : 1≤r) (s t : ℤ)
  49     (a : T.FirstPage (s,t)) :
  50     a∈T.cycles r hr (s,t) ↔
  51     ∃ c : HomotopyGroup (t-s) ((T.stageAt (s+r))⟦(1:ℤ)⟧),
  52       c≫(T.mapAt (show s+1≤s+r by omega))⟦(1:ℤ)⟧' =
  53       a≫HasFunctorialCofiber.cofibδ (T.mapAt (show s≤s+1 by omega)) := by
  54   constructor
  55   · rintro ⟨b,hb⟩
  56     obtain ⟨c,hc⟩ := (desuspend_bijective (T.stageAt (s+r)) (t-s)).2 b
  57     refine ⟨c,?_⟩
  58     apply (desuspend_bijective (T.stageAt (s+1)) (t-s)).1
  59     rw [desuspend_postcompose,hc]
  60     exact hb
  61   · rintro ⟨c,hc⟩
  62     refine ⟨AbstractAdamsTower.desuspend (T.stageAt (s+r)) (t-s) c,?_⟩
  63     change _≫T.mapAt (show s+1≤s+r by omega)=
  64       AbstractAdamsTower.desuspend (T.stageAt (s+1)) (t-s)
  65         (a≫HasFunctorialCofiber.cofibδ (T.mapAt (show s≤s+1 by omega)))
  66     rw [←desuspend_postcompose,hc]

  68 private theorem desuspend_mem_raw_cycles (X : ClassicalObject)
  69     (r : ℕ) (hr : 1≤r) (s t : ℤ)
  70     (a : (model.classicalTower.tower ((X.obj model.auxiliary)⟦(1:ℤ)⟧)).FirstPage (s,t))
  71     (ha : a∈(model.classicalTower.tower ((X.obj model.auxiliary)⟦(1:ℤ)⟧)).cycles r hr (s,t)) :
  72     (model.classicalSuspension X).desuspendFirstPage s t a ∈
  73       (model.classicalTower.tower (X.obj model.auxiliary)).cycles r hr (s,t-1) := by
  74   let S := model.classicalSuspension X
  75   let T := model.classicalTower.tower (X.obj model.auxiliary)
  76   obtain ⟨c,hc⟩ := (raw_cycles_iff _ r hr s t a).mp ha
  77   let R := shiftFunctorComm Spectrum (1:ℤ) (1:ℤ)
  78   let e := c≫(S.tower (s+r)).hom⟦(1:ℤ)⟧'≫R.inv.app (T.stageAt (s+r))
  79   have he : e≫(T.mapAt (show s+1≤s+r by omega))⟦(1:ℤ)⟧'⟦(1:ℤ)⟧' =
  80       -(a≫(S.layer s).hom≫(HasFunctorialCofiber.cofibδ
  81         (T.mapAt (show s≤s+1 by omega)))⟦(1:ℤ)⟧') := by
  82     dsimp only [e]
  83     simp only [Category.assoc]
  84     erw [←R.inv.naturality]
  85     simp only [Functor.comp_map]
  86     rw [←Functor.map_comp_assoc,S.tower_comm,Functor.map_comp]
  87     simp only [←Category.assoc]
  88     erw [hc]
  89     simp only [Category.assoc]
  90     have hs := S.layer_connecting s
  91     have hs' := congrArg (fun f=>f≫R.inv.app (T.stageAt (s+1))) hs
  92     dsimp only [T,AbstractAdamsTower.layer] at hs'
  93     simp only [R,Category.assoc,Iso.hom_inv_id_app,Preadditive.neg_comp] at hs'
  94     erw [Category.comp_id] at hs'
  95     erw [hs']
  96     simp only [Preadditive.comp_neg,neg_neg]
  97     rfl
  98   apply (raw_cycles_iff T r hr s (t-1) _).mpr
  99   let j := eqToHom (congrArg (Sphere (C:=Spectrum)) (show (t-1)-s=(t-s)-1 by omega))
 100   refine ⟨-(j≫AbstractAdamsTower.desuspend ((T.stageAt (s+r))⟦(1:ℤ)⟧) (t-s) e),?_⟩
 101   rw [Preadditive.neg_comp,Category.assoc,←desuspend_postcompose,he,map_neg]
 102   simp only [Preadditive.comp_neg,neg_neg]
 103   rw [←Category.assoc a]
 104   erw [desuspend_postcompose]
 105   exact (Category.assoc _ _ _).symm

 137 private theorem lift_desuspend_bijective (X : ClassicalObject) (d n : ℤ) :
 138     Function.Bijective (lift_desuspend X d n) := by
 139   let S := model.classicalSuspension X
 140   let T := model.classicalTower.tower (X.obj model.auxiliary)
 141   let R := shiftFunctorComm Spectrum (1:ℤ) (1:ℤ)
 142   constructor
 143   · intro a b h
 144     have h' := (desuspend_bijective ((T.stageAt d)⟦(1:ℤ)⟧) n).1 h
 145     apply (cancel_mono ((S.tower d).hom⟦(1:ℤ)⟧')).mp
 146     apply (cancel_mono (R.inv.app (T.stageAt d))).mp
 147     simpa only [Category.assoc] using h'
 148   · intro b
 149     obtain ⟨e,he⟩ := (desuspend_bijective ((T.stageAt d)⟦(1:ℤ)⟧) n).2 b
 150     refine ⟨e ≫ R.hom.app (T.stageAt d) ≫ (S.tower d).inv⟦(1:ℤ)⟧', ?_⟩
 151     change AbstractAdamsTower.desuspend ((T.stageAt d)⟦(1:ℤ)⟧) n
 152       ((e ≫ R.hom.app (T.stageAt d) ≫ (S.tower d).inv⟦(1:ℤ)⟧') ≫
 153         (S.tower d).hom⟦(1:ℤ)⟧' ≫ R.inv.app (T.stageAt d))=b
 154     simp only [Category.assoc]
 155     let E := (shiftFunctor Spectrum (1:ℤ)).mapIso (S.tower d)
 156     change AbstractAdamsTower.desuspend ((T.stageAt d)⟦(1:ℤ)⟧) n
 157       (e ≫ R.hom.app (T.stageAt d) ≫ E.inv ≫ E.hom ≫ R.inv.app (T.stageAt d))=b
 158     simp only [Iso.inv_hom_id_assoc,Iso.hom_inv_id_app]
 159     erw [Category.comp_id]
 160     exact he

 208 private theorem desuspend_mem_raw_cycles_iff (X : ClassicalObject)
 209     (r : ℕ) (hr : 1≤r) (s t : ℤ)
 210     (a : (model.classicalTower.tower ((X.obj model.auxiliary)⟦(1:ℤ)⟧)).FirstPage (s,t)) :
 211     (model.classicalSuspension X).desuspendFirstPage s t a ∈

 228   simp only [←Category.assoc,Iso.inv_hom_id,Category.id_comp]

 351 private theorem desuspend_mem_raw_boundaries_iff (X : ClassicalObject)
 352     (r : ℕ) (hr : 1≤r) (s t : ℤ)
 353     (a : (model.classicalTower.tower ((X.obj model.auxiliary)⟦(1:ℤ)⟧)).FirstPage (s,t)) :
 354     (model.classicalSuspension X).desuspendFirstPage s t a ∈
 355       (model.classicalTower.tower (X.obj model.auxiliary)).boundaries r hr (s,t-1) ↔
 356     a∈(model.classicalTower.tower ((X.obj model.auxiliary)⟦(1:ℤ)⟧)).boundaries r hr (s,t) := by
 357   let S := model.classicalSuspension X
 358   let T := model.classicalTower.tower (X.obj model.auxiliary)
 359   let U := model.classicalTower.tower ((X.obj model.auxiliary)⟦(1:ℤ)⟧)
 360   have hi (b : HomotopyGroup (t-s) (U.stageAt s)) :
 361       boundary_desuspendHom (S.layer s) s t
 362         (b≫HasFunctorialCofiber.cofibι (U.mapAt (show s≤s+1 by omega))) =
 363       boundary_desuspendHom (S.tower s) s t b≫
 364         HasFunctorialCofiber.cofibι (T.mapAt (show s≤s+1 by omega)) :=
 365     boundary_desuspendHom_natural _ _ _ _ (S.layer_inclusion s).symm s t b
 366   have hm (b : HomotopyGroup (t-s) (U.stageAt s)) :
 367       boundary_desuspendHom (S.tower (s-r+1)) s t
 368         (b≫U.mapAt (show s-r+1≤s by omega)) =
 369       boundary_desuspendHom (S.tower s) s t b≫T.mapAt (show s-r+1≤s by omega) :=
 370     boundary_desuspendHom_natural _ _ _ _ (S.tower_comm _ _ _) s t b
 371   constructor
 372   · intro ha
 373     obtain ⟨b,hb,he⟩ := (raw_boundaries_iff T r hr s (t-1) _).mp ha
 374     obtain ⟨c,hc⟩ := (boundary_desuspendHom_bijective (S.tower s) s t).2 b
 375     apply (raw_boundaries_iff U r hr s t a).mpr
 376     refine ⟨c,?_,?_⟩
 377     · apply (boundary_desuspendHom_bijective (S.tower (s-r+1)) s t).1
 378       rw [hm,hc,hb,boundary_desuspendHom_zero]
 379     · apply (boundary_desuspendHom_bijective (S.layer s) s t).1
 380       rw [hi,hc]
 381       exact he
 382   · intro ha
 383     obtain ⟨b,hb,he⟩ := (raw_boundaries_iff U r hr s t a).mp ha
 384     apply (raw_boundaries_iff T r hr s (t-1) _).mpr
 385     refine ⟨boundary_desuspendHom (S.tower s) s t b,?_,?_⟩
 386     · rw [←hm,hb,boundary_desuspendHom_zero]
 387     · rw [←hi,he]
 388       rfl

 390 private theorem initial_class_boundary_iff (D : SSData (ModuleCat.{0} ℤ))
 391     (n : WithTop ℕ) (z : (Subobject.underlying.obj (D.Z 0) : ModuleCat ℤ)) :
 392     (∃ b : (Subobject.underlying.obj (D.B n) : ModuleCat ℤ),
 393       (Subobject.ofLE (D.B n) (D.Z 0)
 394         ((D.B_le_Z n).trans (D.Z_anti bot_le)) ≫ D.pageπ 0) b=D.pageπ 0 z) ↔
 395       (D.Z 0).arrow z∈(ModuleCat.subobjectModule D.V) (D.B n) := by
 396   let i := Subobject.ofLE (D.B n) (D.Z 0)
 397     ((D.B_le_Z n).trans (D.Z_anti bot_le))
 398   constructor
 399   · rintro ⟨b,hb⟩
 400     have hz : D.pageπ 0 (z-i b)=0 := by
 401       rw [map_sub]
 402       exact sub_eq_zero.mpr hb.symm
 403     have h0 := (subobject_cokernel_π_eq_zero_iff (D.B 0) (D.Z 0) (D.B_le_Z 0) _).mp hz
 404     have h0' : (D.Z 0).arrow (z-i b)∈(ModuleCat.subobjectModule D.V) (D.B n) :=
 405       (ModuleCat.subobjectModule D.V).monotone (D.B_mono bot_le) h0
 406     have hb' : (D.Z 0).arrow (i b)∈(ModuleCat.subobjectModule D.V) (D.B n) := by
 407       change (i≫(D.Z 0).arrow) b∈_
 408       rw [Subobject.ofLE_arrow]
 409       exact ⟨b,rfl⟩
 410     have h := ((ModuleCat.subobjectModule D.V) (D.B n)).add_mem h0' hb'
 411     rw [map_sub,sub_add_cancel] at h
 412     exact h
 413   · rintro ⟨b,hb⟩
 414     refine ⟨b,?_⟩
 415     have he : i b=z := by
 416       apply (ModuleCat.mono_iff_injective (D.Z 0).arrow).mp inferInstance
 417       change (i≫(D.Z 0).arrow) b=(D.Z 0).arrow z
 418       rw [Subobject.ofLE_arrow]
 419       exact hb
 420     change D.pageπ 0 (i b)=D.pageπ 0 z
 421     rw [he]
