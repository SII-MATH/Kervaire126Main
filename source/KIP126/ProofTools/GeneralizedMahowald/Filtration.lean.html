  18 variable {Syn : Type u} [SyntheticCategory.{u,v} Syn]

  23 theorem filtration_at_zero (P : SyntheticTowerPresentation unit F)
  24     (X : Syn) (m w : ℤ) (a : BiHom m w X) : FiltrationAtLeast P 0 a := by
  25   refine ⟨a ≫ (P.tower X).zeroIso.inv, ?_⟩
  26   change (a ≫ (P.tower X).zeroIso.inv) ≫ (P.tower X).projection 0 = a
  27   simp only [AbstractAdamsTower.projection, Int.toNat_zero, (P.tower X).map_self]
  28   erw [Category.assoc, Category.id_comp, Iso.inv_hom_id, Category.comp_id]
  29 
  30 theorem filtration_iff_of_infinity_zero (c : SyntheticTowerConvergence P X)
  31     (s t m w : ℤ) (hst : s ≤ t)
  32     (hz : ∀ j, s ≤ j → j < t → IsZero (((F.obj X).sequence.ssData (j,m+j,w)).eInfty))
  33     (a : BiHom m w X) : FiltrationAtLeast P s a ↔ FiltrationAtLeast P t a := by
  34   have hgraded (j : ℤ) (hj : s ≤ j) (hjt : j < t) :
  35       IsZero ((P.filtration X).associatedGraded j (m,w)) :=
  36     (hz j hj hjt).of_iso (towerGradedIso c (m,w) j).symm
  37   have he := (P.filtration X).eq_of_associatedGraded_isZero s t (m,w) hst hgraded
  38   have he' := congrArg (fun Q => a ∈ (ModuleCat.subobjectModule (syntheticHomotopy X (m,w))) Q) he
  39   simpa only [SyntheticTowerPresentation.filtration, OrderIso.apply_symm_apply,
  40     FiltrationAtLeast] using Iff.of_eq he'
  41 end General

  45 /- The shift comparison transports actual filtration levels, so separation on
  46    a finite quotient also holds after the specified bigraded shift. -/
  47 section ShiftSeparation
  48 universe u v w
  49 variable {C : Type u} [StableHomotopyCategory.{u,v} C]
  50   [HasFunctorialCofiber (C := C)]
  51   {Syn : Type w} [SyntheticCategory.{w,v} Syn]
  52   [HasFunctorialCofiber (C := Syn)]
  53 private theorem homotopySeparated_shift (D : Model C Syn) (X : SyntheticObject) (δ : ℤ × ℤ)
  54     (S : SyntheticESSShiftComparison D δ)
  55     (hsep : HomotopySeparated D.syntheticTower (X.obj D.nu D.auxiliary)) :
  56     HomotopySeparated D.syntheticTower ((SyntheticCategory.biShift δ).obj
  57       (X.obj D.nu D.auxiliary)) := by
  58   intro m w a ha
  59   let p : ℤ × ℤ := (m-δ.1,w-δ.2)
  60   have hp : p+δ=(m,w) := by ext <;> simp [p]
  61   let e : FilteredPieceIso (D.convergence X).filtration
  62       (D.convergence (.shift δ X)).filtration p (m,w) :=
  63     hp ▸ S.filtered X p
  64   have hb : e.ambient.inv a = 0 := by
  65     apply hsep p.1 p.2 (e.ambient.inv a)
  66     intro s
  67     have hmem : a ∈ ModuleCat.subobjectModule _
  68         ((D.convergence (.shift δ X)).filtration.F s (m,w)) := by
  69       simpa only [SyntheticTowerConvergence.filtration,
  70         SyntheticTowerPresentation.filtration, OrderIso.apply_symm_apply,
  71         FiltrationAtLeast, SyntheticObject.obj] using ha s
  72     obtain ⟨x,hx⟩ := hmem
  73     have hcomm := ConcreteCategory.congr_hom (e.symm.level_arrow s) x
  74     simp only [ModuleCat.comp_apply, FilteredPieceIso.symm] at hcomm
  75     have hm : e.ambient.inv a ∈ ModuleCat.subobjectModule _
  76         ((D.convergence X).filtration.F s p) := by
  77       refine ⟨(e.level s).inv x,?_⟩
  78       change ((D.convergence X).filtration.F (s:ℤ) p).arrow ((e.level s).inv x) = e.ambient.inv a
  79       simpa only [hx, Iso.symm_hom] using hcomm
  80     simpa only [SyntheticTowerConvergence.filtration,
  81       SyntheticTowerPresentation.filtration, OrderIso.apply_symm_apply,
  82       FiltrationAtLeast] using hm
  83   apply (ModuleCat.mono_iff_injective e.ambient.inv).mp inferInstance
  84   simpa only [map_zero] using hb
  85 
  86 end ShiftSeparation

  95 theorem finite_infinity_zero_outside (X : ClassicalObject) (q : ℕ) (hq : 0 < q)
  96     (s t w : ℤ) (hw : ¬ (0 ≤ t-w ∧ t-w < q)) :
  97     IsZero (((model.family.quotient (model.nu.functor.obj (X.obj model.auxiliary)) q).sequence.ssData
  98       (s,t,w)).eInfty) := by
  99   haveI : Subsingleton (KIP126.Literature.finiteEInftyModel model X q (s,t) w) := by
 100     simp only [KIP126.Literature.finiteEInftyModel, hw, ite_false]
 101     infer_instance
 102   haveI : Subsingleton (((model.family.quotient
 103       (model.nu.functor.obj (X.obj model.auxiliary)) q).sequence.ssData (s,t,w)).eInfty) :=
 104     (KIP126.Literature.Standard.eInfinity_finite X q hq (s,t) w).injective.subsingleton
 105   exact ModuleCat.isZero_of_subsingleton _

 107 theorem finite_high_filtration_zero (X : ClassicalObject) (q : ℕ) (hq : 0 < q)
 108     (m w s : ℤ) (hs : (q:ℤ) ≤ m+s-w)
 109     (a : BiHom m w (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) q))
 110     (ha : FiltrationAtLeast model.syntheticTower s a) : a = 0 := by
 111   apply synthetic_homotopy_separated (.nu X) q m w a
 112   intro j
 113   by_cases hj : s ≤ (j:ℤ)
 114   · exact (filtration_iff_of_infinity_zero (model.convergence (.quotient q (.nu X)))
 115       s j m w hj (fun k hk _ => finite_infinity_zero_outside X q hq k (m+k) w (by omega)) a).mp ha
 116   · exact model.syntheticTower.filtrationSubmodule_antitone _ (m,w) (by omega) ha

 118 theorem finite_top_detection_unique (X : ClassicalObject) (q : ℕ) (hq : 0 < q)
 119     (s t w : ℤ) (hwindow : (q:ℤ) ≤ t+1-w)
 120     (x : (model.family.quotient (model.nu.functor.obj (X.obj model.auxiliary)) q).E₂ (s,t,w))
 121     (a b : BiHom (t-s) w (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) q))
 122     (ha : Detects (model.convergence (.quotient q (.nu X))) (s,t,w) x a)
 123     (hb : Detects (model.convergence (.quotient q (.nu X))) (s,t,w) x b) : a = b := by
 124   apply sub_eq_zero.mp
 125   exact finite_high_filtration_zero X q hq (t-s) w (s+1) (by omega) (a-b)
 126     (detects_sub_filtration (model.convergence (.quotient q (.nu X))) (s,t,w) ha hb)

 128 theorem finite_bottom_filtration (X : ClassicalObject) (q : ℕ) (hq : 0 < q)
 129     (s t : ℤ)
 130     (a : BiHom (t-s) t (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) q)) :
 131     FiltrationAtLeast model.syntheticTower s a := by
 132   have hzero := filtration_at_zero model.syntheticTower _ (t-s) t a
 133   by_cases hs : s ≤ 0
 134   · exact model.syntheticTower.filtrationSubmodule_antitone _ (t-s,t) hs hzero
 135   · exact (filtration_iff_of_infinity_zero (model.convergence (.quotient q (.nu X)))
 136       0 s (t-s) t (by omega)
 137       (fun k _ hk => finite_infinity_zero_outside X q hq k (t-s+k) t (by omega)) a).mp hzero

 139 theorem shifted_finite_high_filtration_zero (X : ClassicalObject) (q : ℕ) (hq : 0 < q)
 140     (i m w s : ℤ) (hs : (q:ℤ) ≤ m+s-(w-i))
 141     (a : BiHom m w ((SyntheticCategory.biShift (0,i)).obj
 142       (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) q)))
 143     (ha : FiltrationAtLeast model.syntheticTower s a) : a = 0 := by
 144   let B : SyntheticObject := .shift (0,i) (.quotient q (.nu X))
 145   have hz (j : ℤ) (hj : s ≤ j) :
 146       IsZero (((model.family.obj (B.obj model.nu model.auxiliary)).sequence.ssData (j,m+j,w)).eInfty) := by
 147     have hbase := finite_infinity_zero_outside X q hq j (m+j) (w-i) (by omega)
 148     let e := towerInfinityIso (model.convergence (.quotient q (.nu X)))
 149       (model.convergence (.shift (0,i) (.quotient q (.nu X))))
 150       ((essShift (0,i)).filtered (.quotient q (.nu X)) (m,w-i)) j
 151     have he := hbase.of_iso e.symm
 152     simpa only [e, B, Prod.fst_add, Prod.snd_add, add_zero, sub_add_cancel] using he
 153   apply shifted_quotient_homotopy_separated (.nu X) q (0,i) m w a
 154   intro j
 155   by_cases hj : s ≤ (j:ℤ)
 156   · exact (filtration_iff_of_infinity_zero (model.convergence B) s j m w hj
 157       (fun k hk _ => hz k hk) a).mp ha
 158   · exact model.syntheticTower.filtrationSubmodule_antitone _ (m,w) (by omega) ha
