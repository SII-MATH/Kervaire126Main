  16 private theorem quotient_shift_desuspend_as_native_unshift
  17     (X : ClassicalObject) (δ p : ℤ×ℤ) (q : ℕ) (m w : ℤ)
  18     (hm : m-δ.1=p.1) (hw : w-δ.2=p.2)
  19     (ε : XModLambdaN ((SyntheticCategory.biShift δ).obj
  20       (model.nu.functor.obj (X.obj model.auxiliary))) q ≅
  21       (SyntheticCategory.biShift δ).obj
  22         (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) q))
  23     (a : BiHom m w (XModLambdaN ((SyntheticCategory.biShift δ).obj
  24       (model.nu.functor.obj (X.obj model.auxiliary))) q)) :
  25     homotopyRegrade hm hw (quotientShiftDesuspend X δ q ε m w a) =
  26       quotientUnshift X δ p q ε
  27         (homotopyRegrade (by dsimp; omega : m=(p+δ).1)
  28           (by dsimp; omega : w=(p+δ).2) a) := by
  29   rcases p with ⟨pm,pw⟩
  30   dsimp only [Prod.fst,Prod.snd] at hm hw
  31   subst pm; subst pw
  32   simp only [quotientShiftDesuspend,quotientUnshift,
  33     biShiftAddIso,Iso.trans_hom,eqToIso.hom,eqToHom_app,
  34     NatTrans.comp_app,regrade_precomp,Category.assoc,eqToHom_refl,Category.id_comp]

  36 private theorem normalized_f_raw_source_unshift_detects
  37     (T : TriangleData model.auxiliary)
  38     (he : (normalizedExponent model.classicalTower T.f : ℤ) +
  39       normalizedExponent model.classicalTower T.g + normalizedExponent model.classicalTower T.h = 1)
  40     (hT : NormalizedTriangleCompatible model T he)
  41     (r k : ℕ) (hkr : k<r) (sf tf m w : ℤ) (x : Ambient model T.X (sf,tf))
  42     (hm : m-1=tf-sf)
  43     (hw : w-((normalizedExponent model.classicalTower T.f : ℤ)+
  44       normalizedExponent model.classicalTower T.g)=tf-k)
  45     (εmain : XModLambdaN ((SyntheticCategory.biShift
  46       (1,(normalizedExponent model.classicalTower T.f : ℤ)+normalizedExponent model.classicalTower T.g)).obj
  47         (model.nu.functor.obj (T.X.obj model.auxiliary))) r ≅
  48       (SyntheticCategory.biShift
  49         (1,(normalizedExponent model.classicalTower T.f : ℤ)+normalizedExponent model.classicalTower T.g)).obj
  50           (XModLambdaN (model.nu.functor.obj (T.X.obj model.auxiliary)) r))
  51     (hεmain : XModLambdaN.incl _ r≫εmain.hom =
  52       (SyntheticCategory.biShift
  53         (1,(normalizedExponent model.classicalTower T.f : ℤ)+normalizedExponent model.classicalTower T.g)).map
  54           (XModLambdaN.incl _ r))
  55     (cGeometry : BiHom m w
  56       (XModLambdaN (HasFunctorialCofiber.cofib (model.normalizedMap T.Y T.Z T.g).map) r))
  57     (hc : Detects (model.convergence (.quotient r (.nu T.X))) (sf,tf,tf-k)
  58       (finiteTargetLabel model T.X r sf tf k x)
  59       (homotopyRegrade hm hw
  60         (quotientShiftDesuspend T.X
  61           (1,(normalizedExponent model.classicalTower T.f : ℤ)+normalizedExponent model.classicalTower T.g)
  62           r εmain m w
  63           (cGeometry≫XModLambdaN.map (normalized_g_cofiber_iso T he hT).hom r)))) :
  64     let cNative := homotopyRegrade
  65       (show m=((tf-sf,tf-k)+(1,(normalizedExponent model.classicalTower T.f : ℤ)+
  66         normalizedExponent model.classicalTower T.g)).1 by dsimp; omega)
  67       (show w=((tf-sf,tf-k)+(1,(normalizedExponent model.classicalTower T.f : ℤ)+
  68         normalizedExponent model.classicalTower T.g)).2 by dsimp; omega) cGeometry
  69     let ef : ℤ := normalizedExponent model.classicalTower T.f
  70     let eg : ℤ := normalizedExponent model.classicalTower T.g
  71     let p : ℤ×ℤ := (tf-sf,tf-k)
  72     let Cg := HasFunctorialCofiber.cofib (model.normalizedMap T.Y T.Z T.g).map
  73     ∃ (εRaw : XModLambdaN ((SyntheticCategory.biShift (0,-eg)).obj Cg) r ≅
  74         (SyntheticCategory.biShift (0,-eg)).obj (XModLambdaN Cg r))
  75       (εe : XModLambdaN ((SyntheticCategory.biShift (0,ef)).obj
  76           (model.nu.functor.obj (T.X.obj model.auxiliary))) r ≅
  77         (SyntheticCategory.biShift (0,ef)).obj
  78           (XModLambdaN (model.nu.functor.obj (T.X.obj model.auxiliary)) r))
  79       (εs : XModLambdaN ((SyntheticCategory.biShift (1,0)).obj
  80           ((SyntheticCategory.biShift (0,ef)).obj
  81             (model.nu.functor.obj (T.X.obj model.auxiliary)))) r ≅
  82         (SyntheticCategory.biShift (1,0)).obj
  83           (XModLambdaN ((SyntheticCategory.biShift (0,ef)).obj
  84             (model.nu.functor.obj (T.X.obj model.auxiliary))) r)),
  85       XModLambdaN.incl _ r≫εRaw.hom=(SyntheticCategory.biShift (0,-eg)).map (XModLambdaN.incl _ r) ∧
  86       XModLambdaN.incl _ r≫εe.hom=(SyntheticCategory.biShift (0,ef)).map (XModLambdaN.incl _ r) ∧
  87       XModLambdaN.incl _ r≫εs.hom=(SyntheticCategory.biShift (1,0)).map (XModLambdaN.incl _ r) ∧
  88       let aRaw := homotopyRegrade
  89         (show (((p+(1,ef+eg))+(0,-eg)).1)=(((p+(0,ef))+(1,0)).1) by dsimp; omega)
  90         (show (((p+(1,ef+eg))+(0,-eg)).2)=(((p+(0,ef))+(1,0)).2) by dsimp; omega)
  91         (shiftHomotopyClass (0,-eg) (p+(1,ef+eg)) cNative≫εRaw.inv)
  92       let α := quotientClassUnshift _ (1,0) (p+(0,ef)) r εs
  93         (aRaw≫XModLambdaN.map (-(normalized_f_suspended_source_iso T he hT).hom) r)
  94       Detects (model.convergence (.quotient r (.nu T.X))) (sf,tf,tf-k)
  95         (finiteTargetLabel model T.X r sf tf k x)
  96         (quotientUnshift T.X (0,ef) p r εe α) := by
  97   dsimp only
  98   let δ : ℤ×ℤ := (1,(normalizedExponent model.classicalTower T.f : ℤ)+
  99     normalizedExponent model.classicalTower T.g)
 100   let p : ℤ×ℤ := (tf-sf,tf-k)
 101   let cNative := homotopyRegrade
 102     (show m=(p+δ).1 by dsimp [p,δ]; omega)
 103     (show w=(p+δ).2 by dsimp [p,δ]; omega) cGeometry
 104   have hnative : Detects (model.convergence (.quotient r (.nu T.X))) (sf,tf,tf-k)
 105       (finiteTargetLabel model T.X r sf tf k x)
 106       (quotientUnshift T.X δ p r εmain
 107         (cNative≫XModLambdaN.map (normalized_g_cofiber_iso T he hT).hom r)) := by
 108     dsimp only [cNative]
 109     rw [←regrade_comp]
 110     rw [←quotient_shift_desuspend_as_native_unshift T.X δ p r m w hm hw εmain]
 111     exact hc
 112   exact normalized_f_actual_source_unshift_detects T he hT r k hkr sf tf x εmain hεmain cNative hnative
