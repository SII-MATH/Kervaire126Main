  15 theorem finite_top_extension_preimage
  16     (X Y : ClassicalObject)
  17     (f : X.obj model.auxiliary ⟶ Y.obj model.auxiliary)
  18     (r : ℕ) (n s t : ℤ)
  19     (x : Ambient model X (s,t))
  20     (y : Ambient model Y (s+n,t+n))
  21     (W : FiniteExtensionWitness model X Y f r n s t x y)
  22     (htop : (r:ℤ)-1 ≤ n-normalizedExponent model.classicalTower f+1)
  23     (b : BiHom (t-s) (t+normalizedExponent model.classicalTower f)
  24       (XModLambdaN (model.nu.functor.obj (Y.obj model.auxiliary)) (r-1)))
  25     (hb : Detects (model.convergence (.quotient (r-1) (.nu Y)))
  26       (s+n,(t-s)+(s+n),t+normalizedExponent model.classicalTower f)
  27       (finiteTargetForExtension model X Y f (r-1) n s t W.length_lower y)
  28       (homotopyRegrade (by dsimp; omega) rfl b)) :
  29     ∃ a : BiHom (t-s) (t+normalizedExponent model.classicalTower f)
  30       ((SyntheticObject.quotient (r-1) (extensionSourceObject model X Y f)).obj model.nu model.auxiliary),
  31       a ≫ finiteExtensionMap model X Y f (r-1) = b ∧
  32       Detects (model.convergence (.quotient (r-1) (extensionSourceObject model X Y f)))
  33         (s,(t-s)+s,t+normalizedExponent model.classicalTower f)
  34         (finiteSourceForExtension model X Y f (r-1) s t x)
  35         (homotopyRegrade (by dsimp; omega) rfl a) := by
  36   let A := (SyntheticObject.quotient (r-1) (extensionSourceObject model X Y f)).obj model.nu model.auxiliary
  37   let B := XModLambdaN (model.nu.functor.obj (Y.obj model.auxiliary)) (r-1)
  38   let p := (t-s,t+normalizedExponent model.classicalTower f)
  39   obtain ⟨_,a,b',hab,hax,hby⟩ := (tower_extension_iff
  40     (model.convergentMap (.quotient (r-1) (extensionSourceObject model X Y f))
  41       (.quotient (r-1) (.nu Y)) (finiteExtensionMap model X Y f (r-1)))
  42     p s n W.source W.target).mp W.equation
  43   have ha := source_frame_detects
  44     (model.convergence (.quotient (r-1) (extensionSourceObject model X Y f))) p s W.source
  45     _ W.source_rep a hax
  46   have hb' := source_frame_detects
  47     (model.convergence (.quotient (r-1) (.nu Y))) p (s+n) W.target
  48     _ W.target_rep b' hby
  49   have hr := W.page_ge_two
  50   have heq := finite_top_detection_unique Y (r-1) (by omega)
  51     (s+n) ((t-s)+(s+n)) (t+normalizedExponent model.classicalTower f) (by omega)
  52     _ _ _ hb' hb
  53   have heq' : ((model.syntheticTower.filtration B).F (s+n) p).arrow b' = b := by
  54     have hinj (u v w : ℤ) (huv : u=v) :
  55         Function.Injective (homotopyRegrade huv (rfl : w=w) (X:=B)) := by
  56       subst v
  57       exact Function.injective_id
  58     exact hinj _ _ _ _ heq
  59   refine ⟨((model.syntheticTower.filtration A).F s p).arrow a, hab.trans heq', ha⟩
  60 
  61 end
