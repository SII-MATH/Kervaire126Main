   5 /-!
   6 # Finite λ classes under suspension
   7 
   8 These private lemmas compare quotient shift isomorphisms on actual sphere
   9 representatives. Agreement on the quotient inclusion suffices: the remaining
  10 cofiber boundary has weight −q−1 and vanishes in the target q-window. This gives
  11 compatibility of finite λ classes with inverse suspension without requiring a
  12 new compatibility assumption for the quotient shift isomorphisms.
  13 -/

  27 private def shiftCommIso (a b : ℤ × ℤ) :
  28     SyntheticCategory.biShift (Syn:=SyntheticSpectrum) a ⋙ SyntheticCategory.biShift b ≅
  29       SyntheticCategory.biShift b ⋙ SyntheticCategory.biShift a :=
  30   biShiftAddIso a b (a+b) rfl ≪≫ (biShiftAddIso b a (a+b) (add_comm b a)).symm

  32 private theorem shiftCommIso_eq (a b c : ℤ × ℤ) (h : a+b=c) :
  33     shiftCommIso a b = biShiftAddIso a b c h ≪≫
  34       (biShiftAddIso b a c ((add_comm b a).trans h)).symm := by
  35   subst c
  36   rfl

  38 private theorem shift_comm_add (a b c d : ℤ × ℤ) (hbc : b+c=d) (X : SyntheticSpectrum) :
  39     (shiftCommIso a d).hom.app X =
  40       (biShiftAddIso b c d hbc).inv.app ((SyntheticCategory.biShift a).obj X) ≫
  41         (SyntheticCategory.biShift c).map ((shiftCommIso a b).hom.app X) ≫
  42         (shiftCommIso a c).hom.app ((SyntheticCategory.biShift b).obj X) ≫
  43         (SyntheticCategory.biShift a).map ((biShiftAddIso b c d hbc).hom.app X) := by
  44   let coh := model.quotients.coherence
  45   apply (cancel_mono ((biShiftAddIso d a (a+d) (add_comm d a)).hom.app X)).mp
  46   simp only [shiftCommIso,Iso.trans_hom,Iso.symm_hom,NatTrans.comp_app,Category.assoc]
  47   have h₁ := coh.associativity b c a d (a+c) (a+d) hbc (add_comm c a)
  48     (add_comm d a) (by rw [←hbc]; abel) X
  49   have h₂ := coh.associativity b a c (a+b) (a+c) (a+d) (add_comm b a) rfl
  50     (by rw [←hbc]; abel) (by rw [←hbc]; abel) X
  51   have h₃ := coh.associativity a b c (a+b) d (a+d) rfl hbc
  52     (by rw [←hbc]; abel) rfl X
  53   rw [h₁]
  54   simp only [Functor.map_comp,Category.assoc,Iso.inv_hom_id_app_assoc]
  55   rw [←h₂]
  56   rw [←Category.assoc ((SyntheticCategory.biShift c).map ((biShiftAddIso b a (a+b) (add_comm b a)).inv.app X)),
  57     ←Functor.map_comp,Iso.inv_hom_id_app (biShiftAddIso b a (a+b) (add_comm b a)) X,
  58     CategoryTheory.Functor.map_id,Category.id_comp]
  59   rw [h₃]
  60   simp

  62 private theorem lambdaPow_shift_comm (a : ℤ × ℤ) (n : ℕ) (X : SyntheticSpectrum) :
  63     lambdaPow n ((SyntheticCategory.biShift a).obj X) =
  64       (shiftCommIso a (lambdaDegree n)).hom.app X ≫
  65         (SyntheticCategory.biShift a).map (lambdaPow n X) := by
  66   induction n generalizing X with
  67   | zero =>
  68     change SyntheticCategory.biShift_zero.hom.app ((SyntheticCategory.biShift a).obj X) =
  69       (shiftCommIso a 0).hom.app X ≫
  70         (SyntheticCategory.biShift a).map (SyntheticCategory.biShift_zero.hom.app X)
  71     rw [shiftCommIso_eq a 0 a (add_zero a)]
  72     simp only [Iso.trans_hom,Iso.symm_hom,NatTrans.comp_app]
  73     rw [model.quotients.coherence.right_unit]
  74     have hl := model.quotients.coherence.left_unit a X
  75     rw [←hl]
  76     simp
  77   | succ n ih =>
  78     have h1 : (shiftCommIso a (0,-1)).hom.app X ≫
  79         (SyntheticCategory.biShift a).map (SyntheticCategory.lam.app X) =
  80         SyntheticCategory.lam.app ((SyntheticCategory.biShift a).obj X) := by
  81       simpa [shiftCommIso,Category.assoc] using (model.quotients.coherence.lambda_comm a X).symm
  82     apply (cancel_epi ((lambdaShiftAddIso 1 n (n+1) (by omega)).hom.app
  83       ((SyntheticCategory.biShift a).obj X))).mp
  84     rw [lambdaPow_succ_comparison]
  85     rw [shift_comm_add a (lambdaDegree 1) (lambdaDegree n) (lambdaDegree (n+1))
  86       (by ext <;> simp [lambdaDegree,Nat.cast_add]) X]
  87     simp only [lambdaShiftAddIso,Category.assoc,Iso.hom_inv_id_app_assoc]
  88     rw [←Functor.map_comp]
  89     erw [lambdaPow_succ_comparison]
  90     rw [Functor.map_comp]
  91     have hn := (shiftCommIso a (lambdaDegree n)).hom.naturality (SyntheticCategory.lam.app X)
  92     have hn' := congrArg (fun h => h ≫ (SyntheticCategory.biShift a).map (lambdaPow n X)) hn
  93     simp only [Functor.comp_map,Category.assoc] at hn'
  94     dsimp only [lambdaDegree,Functor.id_obj] at hn' ⊢
  95     erw [←hn']
  96     rw [←Category.assoc,←Functor.map_comp]
  97     erw [h1,ih]
  98     rfl

 100 private theorem finite_lambda_maps_equal_on_spheres
 101     (X Y : ClassicalObject) (q d : ℕ) (hdq : d < q)
 102     (f g : (SyntheticCategory.biShift (0,-(d:ℤ))).obj
 103         (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) (q-d)) ⟶
 104       XModLambdaN (model.nu.functor.obj (Y.obj model.auxiliary)) q)
 105     (hfg : (SyntheticCategory.biShift (0,-(d:ℤ))).map (XModLambdaN.incl _ (q-d)) ≫ f =
 106       (SyntheticCategory.biShift (0,-(d:ℤ))).map (XModLambdaN.incl _ (q-d)) ≫ g) :
 107     ∀ (m w : ℤ) (c : BiHom m w ((SyntheticCategory.biShift (0,-(d:ℤ))).obj
 108       (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) (q-d)))), c ≫ f = c ≫ g := by
 109   letI := model.exactness.shiftCommShift (0,-(d:ℤ))
 110   letI := model.exactness.shiftExact (0,-(d:ℤ))
 111   let F := SyntheticCategory.biShift (Syn:=SyntheticSpectrum) (0,-(d:ℤ))
 112   let V := model.nu.functor.obj (X.obj model.auxiliary)
 113   let T := XModLambdaN.cofiberTriangle V (q-d)
 114   have hzero : (F.mapTriangle.obj T).mor₂ ≫ (f-g) = 0 := by
 115     change F.map (XModLambdaN.incl V (q-d)) ≫ (f-g) = 0
 116     rw [Preadditive.comp_sub,hfg,sub_self]
 117   obtain ⟨z,hz⟩ := Triangle.yoneda_exact₃ (F.mapTriangle.obj T)
 118     (F.map_distinguished T (XModLambdaN.cofiberTriangle_distinguished V (q-d))) (f-g) hzero
 119   intro m w c
 120   let ε : (F.obj ((SyntheticCategory.biShift (0,-((q-d:ℕ):ℤ))).obj V))⟦(1:ℤ)⟧ ≅
 121       (SyntheticCategory.biShift (0,-(q:ℤ)-1)).obj
 122         (model.nu.functor.obj ((ClassicalObject.shift 1 X).obj model.auxiliary)) :=
 123     (shiftFunctor SyntheticSpectrum (1:ℤ)).mapIso
 124       ((lambdaShiftAddIso (q-d) d q (Nat.sub_add_cancel (by omega))).app V) ≪≫
 125       (SyntheticCategory.biShift_compat (Syn:=SyntheticSpectrum) 1).symm.app _ ≪≫
 126       (SyntheticCategory.biShift_comp (0,-(q:ℤ)) (1,0)).app V ≪≫
 127       eqToIso (congrArg (fun p => (SyntheticCategory.biShift p).obj V)
 128         (show (0,-(q:ℤ))+(1,0)=(1,1)+(0,-(q:ℤ)-1) by ext <;> simp)) ≪≫
 129       (SyntheticCategory.biShift_comp (1,1) (0,-(q:ℤ)-1)).symm.app V ≪≫
 130       (SyntheticCategory.biShift (0,-(q:ℤ)-1)).mapIso (model.nu.suspensionIso _).symm
 131   have h := negative_shift_nu_to_quotient_zero (.shift 1 X) Y q (by omega) m w
 132     ((c ≫ (F.mapTriangle.obj T).mor₃) ≫ ε.hom) (ε.inv ≫ z)
 133   have hcz : (c ≫ (F.mapTriangle.obj T).mor₃) ≫ z = 0 := by
 134     simpa only [Category.assoc,Iso.hom_inv_id_assoc] using h
 135   apply sub_eq_zero.mp
 136   rw [←Preadditive.comp_sub,hz,←Category.assoc]
 137   exact hcz

 139 private theorem quotient_shift_lambda_on_spheres
 140     (X : ClassicalObject) (δ : ℤ × ℤ) (q d : ℕ) (hd : 0 < d) (hdq : d < q)
 141     (εn : XModLambdaN ((SyntheticCategory.biShift δ).obj

 143       (SyntheticCategory.biShift δ).obj

 154       ((SyntheticCategory.biShift δ).obj

 167   let v : K.obj (XModLambdaN A (q-d)) ⟶ XModLambdaN A q :=
 168     ff.preimage (C.inv.app _ ≫ K.map εn.inv ≫ L' ≫ εq.hom)
 169   have hv : F.map v = C.inv.app _ ≫ K.map εn.inv ≫ L' ≫ εq.hom := ff.map_preimage _

 174     have hn := C.inv.naturality (XModLambdaN.incl A (q-d))

 176       C.inv.app A ≫ K.map (F.map (XModLambdaN.incl A (q-d))) at hn

 178     rw [←Category.assoc (K.map εn.hom),←Functor.map_comp,Iso.hom_inv_id,

 180     rw [←Category.assoc (K.map (XModLambdaN.incl (F.obj A) (q-d))),
 181       ((model.quotients.tower (F.obj A)).triangle hd hdq).lambda_quotient,
 182       Category.assoc,hεq]
 183     have hc := lambdaPow_shift_comm δ d A
 184     change lambdaPow d (F.obj A) = C.hom.app A ≫ F.map (lambdaPow d A) at hc
 185     rw [←Category.assoc,hc]

 188     exact congrArg F.map (((model.quotients.tower A).triangle hd hdq).lambda_quotient.symm)

 190   have hh := shift_preserves_sphere_congruence _ _ δ v L hb m w (c ≫ C.hom.app _)

 194 private theorem shift_hom_swap (p b c pb pc total : ℤ × ℤ)
 195     (hpb : p+b=pb) (hpc : p+c=pc) (hbc : pb+c=total) (hcb : pc+b=total)
 196     {A : SyntheticSpectrum} (α : BiHom p.1 p.2 A) :
 197     (biShiftAddIso pb c total hbc).inv.app S_0_0 ≫
 198         (SyntheticCategory.biShift c).map
 199           ((biShiftAddIso p b pb hpb).inv.app S_0_0 ≫ (SyntheticCategory.biShift b).map α) ≫
 200         (shiftCommIso b c).hom.app A =
 201       (biShiftAddIso pc b total hcb).inv.app S_0_0 ≫
 202         (SyntheticCategory.biShift b).map
 203           ((biShiftAddIso p c pc hpc).inv.app S_0_0 ≫ (SyntheticCategory.biShift c).map α) := by
 204   have ht : p+(b+c)=total := by rw [←add_assoc,hpb,hbc]
 205   have hL := shift_hom_associativity model.quotients.coherence p b c pb (b+c) total
 206     hpb rfl hbc ht α
 207   have hR := shift_hom_associativity model.quotients.coherence p c b pc (b+c) total
 208     hpc (add_comm c b) hcb ht α
 209   rw [←Category.assoc,←hL]
 210   simp only [shiftCommIso,Iso.trans_hom,Iso.symm_hom,NatTrans.comp_app,Category.assoc,
 211     Iso.inv_hom_id_app_assoc]
 212   exact hR
