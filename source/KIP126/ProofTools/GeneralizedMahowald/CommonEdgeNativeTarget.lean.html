  39 private theorem common_target_unshift_filtration
  40     (A : SyntheticObject) (δ p : ℤ×ℤ) (s : ℤ)
  41     (a : BiHom (p+δ).1 (p+δ).2 ((SyntheticCategory.biShift δ).obj (A.obj model.nu model.auxiliary)))
  42     (ha : FiltrationAtLeast model.syntheticTower s a) :
  43     FiltrationAtLeast model.syntheticTower s (commonTargetUnshift δ p _ a) := by
  44   let E := (essShift δ).filtered A p
  45   obtain ⟨b,hb⟩ := filtration_representative_exists model.syntheticTower _ (p+δ) s a ha
  46   have he : E.ambient.hom (commonTargetUnshift δ p _ a) = a := by
  47     rw [(essShift δ).ambient_eq,common_target_shift_unshift]
  48   have hi : E.ambient.inv a = commonTargetUnshift δ p _ a := by
  49     exact (congrArg E.ambient.inv he.symm).trans (E.ambient.toLinearEquiv.symm_apply_apply _)
  50   have hc := ConcreteCategory.congr_hom (E.symm.level_arrow s) b
  51   change ((model.syntheticTower.filtration _).F s p).arrow ((E.level s).inv b) =
  52     E.ambient.inv (((model.syntheticTower.filtration _).F s (p+δ)).arrow b) at hc
  53   erw [hb,hi] at hc
  54   have hm : commonTargetUnshift δ p _ a ∈ ModuleCat.subobjectModule _
  55       ((model.syntheticTower.filtration _).F s p) := ⟨(E.level s).inv b,hc⟩
  56   simpa only [SyntheticTowerPresentation.filtration,OrderIso.apply_symm_apply,
  57     FiltrationAtLeast] using hm

  59 private theorem common_target_double_vertical_unshift
  60     (a b c m w : ℤ) (hab : a+b=c) (V : SyntheticSpectrum)
  61     (v : BiHom ((m,w)+(0,a)).1 ((m,w)+(0,a)).2 ((SyntheticCategory.biShift (0,a)).obj V)) :

  65           (biShiftAddIso (0,a) (0,b) (0,c) (by ext <;> dsimp; omega)).hom.app V)) =
  66       commonTargetUnshift (0,a) (m,w) V v := by
  67   let z := commonTargetUnshift (0,a) (m,w) V v
  68   have hz : shiftHomotopyClass (0,a) (m,w) z = v :=
  69     common_target_shift_unshift (0,a) (m,w) V v
  70   have hshift : homotopyRegrade (m' := ((m,w)+(0,c)).1) (w' := ((m,w)+(0,c)).2)
  71         (by simp) (by dsimp; omega)
  72         (shiftHomotopyClass (0,b) ((m,w)+(0,a)) v ≫
  73           (biShiftAddIso (0,a) (0,b) (0,c) (by ext <;> dsimp; omega)).hom.app V) =
  74       shiftHomotopyClass (0,c) (m,w) z := by
  75     rw [←hz,regrade_comp,shift_regrade]
  76     have h := shift_hom_associativity model.quotients.coherence
  77       (m,w) (0,a) (0,b) ((m,w)+(0,a)) (0,c) ((m,w)+(0,c))
  78       rfl (by ext <;> dsimp; omega) (by ext <;> dsimp <;> omega) rfl z
  79     have h' := congrArg (fun v => v ≫
  80       (biShiftAddIso (0,a) (0,b) (0,c) (by ext <;> dsimp; omega)).hom.app V) h.symm
  81     simp only [Category.assoc,Iso.inv_hom_id_app,Category.comp_id] at h'
  82     simpa [shiftHomotopyClass,biShiftAddIso,Category.assoc] using h'
  83   rw [hshift,common_target_unshift_shift]
