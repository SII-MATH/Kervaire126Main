  16 /-- Compare the combined bigraded shift with the ordinary suspension of the
  17 specified vertical shift, using the existing composition and compatibility. -/
  18 noncomputable def biSuspensionIso (q : ℤ) (Q : Syn) :
  19     (SyntheticCategory.biShift (1, -q)).obj Q ≅
  20       ((SyntheticCategory.biShift (0, -q)).obj Q)⟦(1 : ℤ)⟧ := by
  21   have h : ((0 : ℤ), -q) + (1, 0) = (1, -q) := by
  22     ext <;> simp
  23   simpa only [h] using
  24     ((SyntheticCategory.biShift_comp (0, -q) (1, 0)).symm.app Q ≪≫
  25       (SyntheticCategory.biShift_compat (Syn := Syn) 1).app
  26         ((SyntheticCategory.biShift (0, -q)).obj Q))

  28 /-- The underlying equivalence for the Bockstein landing degree. It is the
  29 inverse of the existing suspension invariance, followed by the specified
  30 ordinary-suspension comparison and the integer degree equalities. -/
  31 noncomputable def biSuspensionHomEquivRaw (n w q : ℤ) (Q : Syn) :
  32     BiHom n w (((SyntheticCategory.biShift (0, -q)).obj Q)⟦(1 : ℤ)⟧) ≃
  33       BiHom (n - 1) (w + q) Q := by
  34   let e : BiHom (n - 1) (w + q) Q ≃
  35       BiHom n w (((SyntheticCategory.biShift (0, -q)).obj Q)⟦(1 : ℤ)⟧) := by
  36     simpa only [sub_add_cancel, add_neg_cancel_right] using
  37       ((susp_invariance (n - 1) (w + q) 1 (-q) Q).trans
  38         (Iso.homCongr (Iso.refl _) (biSuspensionIso q Q)))
  39   exact e.symm

  43     (SyntheticCategory.biShift (Syn := Syn) p).Additive := by

  53         SyntheticCategory.biShift_zero)

  65 /-- The specified inverse reindexing map is additive. -/
  66 theorem biSuspensionHomEquivRaw_add (n w q : ℤ) (Q : Syn)
  67     (f g : BiHom n w (((SyntheticCategory.biShift (0, -q)).obj Q)⟦(1 : ℤ)⟧)) :
  68     biSuspensionHomEquivRaw n w q Q (f + g) =
  69       biSuspensionHomEquivRaw n w q Q f + biSuspensionHomEquivRaw n w q Q g := by
  70   let e := biSuspensionHomEquivRaw n w q Q
  71   have h : ∀ a b, e.symm (a + b) = e.symm a + e.symm b := by
  72     intro a b
  73     let e0 := (susp_invariance (n - 1) (w + q) 1 (-q) Q).trans
  74       (Iso.homCongr (Iso.refl _) (biSuspensionIso q Q))
  75     have h0 : e0 (a + b) = e0 a + e0 b := by
  76       simp [e0, susp_invariance_add, Preadditive.comp_add, Preadditive.add_comp]
  77     have hc (n' w' : ℤ) (hn : n' = n) (hw : w' = w)
  78         (e' : BiHom (n - 1) (w + q) Q ≃
  79           BiHom n' w' (((SyntheticCategory.biShift (0, -q)).obj Q)⟦(1 : ℤ)⟧))
  80         (he' : e' (a + b) = e' a + e' b)
  81         (ht : (BiHom (n - 1) (w + q) Q ≃
  82           BiHom n' w' (((SyntheticCategory.biShift (0, -q)).obj Q)⟦(1 : ℤ)⟧)) =
  83           (BiHom (n - 1) (w + q) Q ≃
  84           BiHom n w (((SyntheticCategory.biShift (0, -q)).obj Q)⟦(1 : ℤ)⟧))) :
  85         (cast ht e') (a + b) = (cast ht e') a + (cast ht e') b := by
  86       subst n'
  87       subst w'
  88       exact he'
  89     dsimp only [e, biSuspensionHomEquivRaw, Equiv.symm_symm]
  90     exact hc (n - 1 + 1) (w + q + -q) (sub_add_cancel n 1)
  91       (add_neg_cancel_right w q) e0 h0 _
  92   apply e.symm.injective
  93   rw [h]
  94   simp [e]
