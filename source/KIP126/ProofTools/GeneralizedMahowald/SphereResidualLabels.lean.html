  90 private theorem actual_extension_source_native_detects
  91     (Y Z : ClassicalObject) (g : Y.obj model.auxiliary ⟶ Z.obj model.auxiliary)
  92     (q : ℕ) (s t : ℤ) (y : Ambient model Y (s,t))
  93     (ag : BiHom (t-s) (t+normalizedExponent model.classicalTower g)
  94       (XModLambdaN ((extensionSourceObject model Y Z g).obj model.nu model.auxiliary) q))
  95     (hag : Detects (model.convergence (.quotient q (extensionSourceObject model Y Z g)))
  96       (s,(t-s)+s,t+normalizedExponent model.classicalTower g)
  97       (finiteSourceForExtension model Y Z g q s t y)
  98       (homotopyRegrade (by simp) rfl ag)) :
  99     Detects (model.convergence (.quotient q (extensionSourceObject model Y Z g)))
 100       (s,t,t+normalizedExponent model.classicalTower g)
 101       (familyPageMap model.family
 102         (XModLambdaN.incl ((extensionSourceObject model Y Z g).obj model.nu model.auxiliary) q)
 103         2 (s,t,t+normalizedExponent model.classicalTower g)
 104         (regradeE2 (by simp) (infiniteSourceForExtension model Y Z g s t y))) ag := by
 105   let p : Tridegree := (s,(t-s)+s,t+normalizedExponent model.classicalTower g)
 106   have hsource : finiteSourceForExtension model Y Z g q s t y =
 107       familyPageMap model.family (XModLambdaN.incl _ q) 2 p
 108         (infiniteSourceForExtension model Y Z g s t y) := by
 109     unfold finiteSourceForExtension infiniteSourceForExtension finiteNuLabel
 110     rw [page_regrade]
 111     rfl
 112   have hag' := (congrArg (fun z => Detects
 113     (model.convergence (.quotient q (extensionSourceObject model Y Z g))) p z
 114     (homotopyRegrade (show t-s=((t-s)+s)-s by omega) rfl ag)) hsource).mp hag
 115   have hindex : p=(s,t,t+normalizedExponent model.classicalTower g) := by
 116     simp only [p,sub_add_cancel]
 117   have hnative := detects_transport hindex _ _ hag'
 118   have regrade_comp {m m' m'' w w' w'' : ℤ} {V : SyntheticSpectrum}
 119       (hm : m=m') (hw : w=w') (hm' : m'=m'') (hw' : w'=w'')
 120       (z : BiHom m w V) :
 121       homotopyRegrade hm' hw' (homotopyRegrade hm hw z) =
 122         homotopyRegrade (hm.trans hm') (hw.trans hw') z := by
 123     subst m'; subst m''; subst w'; subst w''; rfl
 124   simp only [page_regrade,regrade_comp] at hnative
 125   exact hnative
