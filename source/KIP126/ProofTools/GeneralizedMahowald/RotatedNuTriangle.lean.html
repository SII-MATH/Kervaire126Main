 107 private theorem quotient_one_negative (X : ClassicalObject) (m w : ℤ)

 206 private theorem homology_short_exact_of_boundary_zero
 207     (T : Triangle Spectrum) (hT : T ∈ distTriang Spectrum)
 208     (hz : ZeroOnModTwoHomology model.coefficients T.mor₃) :
 209     HomologyShortExact model.coefficients T := by
 210   letI := closedBackground.symmetric
 211   letI := closedBackground.tensorShift model.coefficients.object
 212   letI := closedBackground.tensorExact model.coefficients.object
 213   let e := (BraidedCategory.tensorLeftIsoTensorRight model.coefficients.object).symm
 214   letI := Functor.CommShift.ofIso e ℤ
 215   letI := Functor.CommShift.ofIso_compatibility e ℤ
 216   letI := Functor.isTriangulated_of_iso e
 217   let F := tensorLeft model.coefficients.object
 218   let U := F.mapTriangle.obj T
 219   have hU : U ∈ distTriang Spectrum := F.map_distinguished T hT
 220   have hu (n : ℤ) (a : HomotopyGroup n U.obj₃) : a≫U.mor₃=0 := by
 221     change a≫F.map T.mor₃≫_=0
 222     change a≫(model.coefficients.object ◁ T.mor₃)≫_=0
 223     have hh : a≫(model.coefficients.object ◁ T.mor₃)=0 := hz n a
 224     rw [←Category.assoc,hh,zero_comp]
 225   intro n
 226   refine ⟨?_,?_,?_⟩
 227   · intro a b hab
 228     have hab' : (a-b)≫U.mor₁=0 := by
 229       change a≫U.mor₁=b≫U.mor₁ at hab
 230       rw [Preadditive.sub_comp,hab,sub_self]
 231     let S := shiftFunctor Spectrum (1:ℤ)
 232     have hh : S.map (a-b)≫S.map U.mor₁=0 := by
 233       rw [←Functor.map_comp,hab',Functor.map_zero]
 234     obtain ⟨d,hd⟩ := U.coyoneda_exact₁ hU (S.map (a-b)) hh
 235     let eS : Sphere (C:=Spectrum) (n+1) ≅ S.obj (Sphere n) :=
 236       ((shiftFunctorAdd' Spectrum n 1 (n+1) rfl).app SphereSpectrum)
 237     have hz' : (eS.hom≫d)≫U.mor₃=0 := hu (n+1) _
 238     have hd0 : d≫U.mor₃=0 := (cancel_epi eS.hom).mp
 239       (by simpa only [Category.assoc,comp_zero] using hz')
 240     have hs0 : S.map (a-b)=0 := hd.trans hd0
 241     have hab0 : a-b=0 := S.map_injective
 242       (by simpa only [Functor.map_zero] using hs0)
 243     exact sub_eq_zero.mp hab0
 244   · intro a
 245     constructor
 246     · intro ha
 247       obtain ⟨b,hb⟩ := U.coyoneda_exact₂ hU a ha
 248       exact ⟨b,hb.symm⟩
 249     · rintro ⟨b,rfl⟩
 250       change (b≫U.mor₁)≫U.mor₂=0
 251       rw [Category.assoc,comp_distTriang_mor_zero₁₂ U hU,comp_zero]
 252   · intro a
 253     obtain ⟨b,hb⟩ := U.coyoneda_exact₃ hU a (hu n a)
 254     exact ⟨b,hb.symm⟩
