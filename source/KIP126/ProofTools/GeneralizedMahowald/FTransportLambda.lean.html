  18 private theorem shifted_finite_lambda_class_general
  19     (X : ClassicalObject) (δ : ℤ×ℤ) (q d : ℕ) (hd : 0<d) (hdq : d<q) (s t : ℤ)
  20     (εn : XModLambdaN ((SyntheticCategory.biShift δ).obj
  21         (model.nu.functor.obj (X.obj model.auxiliary))) (q-d) ≅
  22       (SyntheticCategory.biShift δ).obj
  23         (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) (q-d)))
  24     (εq : XModLambdaN ((SyntheticCategory.biShift δ).obj
  25         (model.nu.functor.obj (X.obj model.auxiliary))) q ≅
  26       (SyntheticCategory.biShift δ).obj
  27         (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) q))
  28     (hεn : XModLambdaN.incl _ (q-d)≫εn.hom=
  29       (SyntheticCategory.biShift δ).map (XModLambdaN.incl _ (q-d)))
  30     (hεq : XModLambdaN.incl _ q≫εq.hom=
  31       (SyntheticCategory.biShift δ).map (XModLambdaN.incl _ q))
  32     (b : BiHom (t-s) t (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) (q-d))) :
  33     homotopyRegrade (m':=t-s+δ.1) (w':=t+δ.2-d) rfl (by dsimp; omega)
  34         (shiftHomotopyClass δ (t-s,t-d) (finiteLambdaClass X q d hd hdq s t b)) ≫ εq.inv =
  35       homotopyRegrade (by simp) (by simp [sub_eq_add_neg])
  36           (shiftHomotopyClass (0,-(d:ℤ)) (t-s+δ.1,t+δ.2)
  37             (shiftHomotopyClass δ (t-s,t) b ≫ εn.inv)) ≫
  38         ((model.quotients.tower _).triangle hd hdq).lambdaMap := by
  39   apply (cancel_mono εq.hom).mp
  40   simp only [Category.assoc,Iso.inv_hom_id,Category.comp_id]
  41   unfold finiteLambdaClass
  42   rw [shift_hom_comp,regrade_comp,shift_hom_comp,regrade_comp]
  43   simp only [Category.assoc]
  44   rw [quotient_shift_lambda_on_spheres X δ q d hd hdq εn εq hεn hεq]
  45   rw [shift_regrade,shift_regrade,shift_regrade]
  46   simp only [shiftHomotopyClass]
  47   rw [←Category.assoc]
  48   apply congrArg (fun z => z ≫ (SyntheticCategory.biShift δ).map
  49     ((model.quotients.tower _).triangle hd hdq).lambdaMap)
  50   simpa [biShiftAddIso,Category.assoc] using
  51     (shift_hom_swap (t-s,t) δ (0,-(d:ℤ)) (t-s+δ.1,t+δ.2) (t-s,t-d)
  52       (t-s+δ.1,t+δ.2-d)
  53       rfl (by ext <;> simp [sub_eq_add_neg])
  54       (by ext <;> simp [sub_eq_add_neg]) (by ext <;> dsimp; omega) b).symm

  57 private theorem quotient_unshift_finite_lambda
  58     (X : ClassicalObject) (δ : ℤ×ℤ) (q d : ℕ) (hd : 0<d) (hdq : d<q) (s t : ℤ)
  59     (εn : XModLambdaN ((SyntheticCategory.biShift δ).obj
  60         (model.nu.functor.obj (X.obj model.auxiliary))) (q-d) ≅
  61       (SyntheticCategory.biShift δ).obj
  62         (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) (q-d)))
  63     (εq : XModLambdaN ((SyntheticCategory.biShift δ).obj
  64         (model.nu.functor.obj (X.obj model.auxiliary))) q ≅
  65       (SyntheticCategory.biShift δ).obj
  66         (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) q))
  67     (hεn : XModLambdaN.incl _ (q-d)≫εn.hom=
  68       (SyntheticCategory.biShift δ).map (XModLambdaN.incl _ (q-d)))
  69     (hεq : XModLambdaN.incl _ q≫εq.hom=
  70       (SyntheticCategory.biShift δ).map (XModLambdaN.incl _ q))
  71     (a : BiHom ((t-s,t)+δ).1 ((t-s,t)+δ).2
  72       (XModLambdaN ((SyntheticCategory.biShift δ).obj
  73         (model.nu.functor.obj (X.obj model.auxiliary))) (q-d))) :
  74     quotientUnshift X δ (t-s,t-d) q εq
  75       (homotopyRegrade (m':=((t-s,t-d)+δ).1) (w':=((t-s,t-d)+δ).2)
  76           (by simp) (by dsimp; omega)
  77           (shiftHomotopyClass (0,-(d:ℤ)) (t-s+δ.1,t+δ.2) a) ≫
  78         ((model.quotients.tower _).triangle hd hdq).lambdaMap) =
  79       finiteLambdaClass X q d hd hdq s t
  80         (quotientUnshift X δ (t-s,t) (q-d) εn a) := by
  81   let b := quotientUnshift X δ (t-s,t) (q-d) εn a
  82   have hb : shiftHomotopyClass δ (t-s,t) b≫εn.inv=a :=
  83     shift_quotient_unshift X δ (t-s,t) (q-d) εn a
  84   have hshift := shifted_finite_lambda_class_general X δ q d hd hdq s t εn εq hεn hεq b
  85   rw [hb] at hshift
  86   have hraw := congrArg (homotopyRegrade
  87     (show t-s+δ.1=((t-s,t-d)+δ).1 from rfl)
  88     (show t+δ.2-d=((t-s,t-d)+δ).2 from by dsimp; omega)) hshift
  89   rw [←regrade_comp,homotopy_regrade_inverse] at hraw
  90   have regrade_trans {B : SyntheticSpectrum} {m₁ m₂ m₃ w₁ w₂ w₃ : ℤ}
  91       (hm₁ : m₁=m₂) (hm₂ : m₂=m₃) (hw₁ : w₁=w₂) (hw₂ : w₂=w₃)
  92       (z : BiHom m₁ w₁ B) :
  93       homotopyRegrade hm₂ hw₂ (homotopyRegrade hm₁ hw₁ z) =
  94         homotopyRegrade (hm₁.trans hm₂) (hw₁.trans hw₂) z := by
  95     subst m₂; subst m₃; subst w₂; subst w₃; rfl
  96   rw [regrade_comp,regrade_trans] at hraw
  97   change quotientUnshift X δ (t-s,t-d) q εq _ = finiteLambdaClass X q d hd hdq s t b
  98   rw [←hraw,quotient_unshift_shift]

 101 private theorem quotient_unshift_finite_lambda_detects
 102     (X : ClassicalObject) (δ : ℤ×ℤ) (q d : ℕ) (hd : 0<d) (hdq : d<q) (s t : ℤ)
 103     (x : Ambient model X (s,t))
 104     (εn : XModLambdaN ((SyntheticCategory.biShift δ).obj
 105         (model.nu.functor.obj (X.obj model.auxiliary))) (q-d) ≅
 106       (SyntheticCategory.biShift δ).obj
 107         (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) (q-d)))
 108     (εq : XModLambdaN ((SyntheticCategory.biShift δ).obj
 109         (model.nu.functor.obj (X.obj model.auxiliary))) q ≅
 110       (SyntheticCategory.biShift δ).obj
 111         (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) q))
 112     (hεn : XModLambdaN.incl _ (q-d)≫εn.hom=
 113       (SyntheticCategory.biShift δ).map (XModLambdaN.incl _ (q-d)))
 114     (hεq : XModLambdaN.incl _ q≫εq.hom=
 115       (SyntheticCategory.biShift δ).map (XModLambdaN.incl _ q))
 116     (a : BiHom ((t-s,t)+δ).1 ((t-s,t)+δ).2
 117       (XModLambdaN ((SyntheticCategory.biShift δ).obj
 118         (model.nu.functor.obj (X.obj model.auxiliary))) (q-d)))
 119     (ha : quotientUnshift X δ (t-s,t) (q-d) εn a ≫
 120         (model.quotients.tower _).rho 1 (q-d) (by omega) =
 121       firstLabel model X s t x ≫ XModLambdaN.map (SyntheticCategory.biShift_zero.hom.app _) 1) :
 122     Detects (model.convergence (.quotient q (.nu X))) (s,t,t-d)
 123       (finiteTargetLabel model X q s t d x)
 124       (quotientUnshift X δ (t-s,t-d) q εq
 125         (homotopyRegrade (m':=((t-s,t-d)+δ).1) (w':=((t-s,t-d)+δ).2)
 126             (by simp) (by dsimp; omega)
 127             (shiftHomotopyClass (0,-(d:ℤ)) (t-s+δ.1,t+δ.2) a) ≫
 128           ((model.quotients.tower _).triangle hd hdq).lambdaMap)) := by
 129   rw [quotient_unshift_finite_lambda X δ q d hd hdq s t εn εq hεn hεq a]
 130   exact finite_quotient_lambda_detects X q d hd hdq s t x _ ha
