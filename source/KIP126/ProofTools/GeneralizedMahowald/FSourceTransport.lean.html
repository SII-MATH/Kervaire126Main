  17 private def fSourceRawFunctor (eg eh : ℤ) : SyntheticSpectrum ⥤ SyntheticSpectrum :=
  18   SyntheticCategory.biShift (1,1) ⋙ SyntheticCategory.biShift (0,-eh) ⋙
  19     SyntheticCategory.biShift (0,-eg)

  21 private def fSourceChi (ef eg eh : ℤ) (he : ef+eg+eh=1) :
  22     fSourceRawFunctor eg eh ≅ SyntheticCategory.biShift (1,ef) :=
  23   Functor.isoWhiskerRight (SyntheticCategory.biShift_comp (1,1) (0,-eh))
  24       (SyntheticCategory.biShift (0,-eg)) ≪≫
  25     SyntheticCategory.biShift_comp ((1,1)+(0,-eh)) (0,-eg) ≪≫
  26     eqToIso (congrArg SyntheticCategory.biShift
  27       (show ((1,1)+(0,-eh))+(0,-eg)=(1,ef) by ext; rfl; dsimp; omega))
  28 
  29 private def fSourceKappa (ef eg eh : ℤ) (he : ef+eg+eh=1) :
  30     fSourceRawFunctor eg eh ≅ SyntheticCategory.biShift (1,ef) :=
  31   Functor.isoWhiskerRight
  32       (biShiftAddIso (1,1) (0,-eh) (1,ef+eg)
  33         (by ext; rfl; dsimp; omega))
  34       (SyntheticCategory.biShift (0,-eg)) ≪≫
  35     biShiftAddIso (1,ef+eg) (0,-eg) (1,ef) (by ext <;> simp)
  36 
  37 private def fSourceDifference (ef eg eh : ℤ) (he : ef+eg+eh=1) :
  38     SyntheticCategory.biShift (Syn:=SyntheticSpectrum) (1,ef) ≅
  39       SyntheticCategory.biShift (1,ef) :=
  40   (fSourceKappa ef eg eh he).symm ≪≫ fSourceChi ef eg eh he

  42 private def negativeFunctorAutomorphism (F : SyntheticSpectrum ⥤ SyntheticSpectrum) : F ≅ F where
  43   hom := { app := fun X => -(𝟙 (F.obj X))
  44            naturality := by intro X Y f; simp }
  45   inv := { app := fun X => -(𝟙 (F.obj X))
  46            naturality := by intro X Y f; simp }
  47   hom_inv_id := by ext X; simp
  48   inv_hom_id := by ext X; simp

  50 private def fSourceSignedChi (ef eg eh : ℤ) (he : ef+eg+eh=1) :
  51     fSourceRawFunctor eg eh ≅ SyntheticCategory.biShift (1,ef) :=
