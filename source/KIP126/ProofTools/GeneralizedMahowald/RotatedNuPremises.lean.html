 100 private def zero_finite_extension_witness (X Y : ClassicalObject)
 101     (f : X.obj model.auxiliary ⟶ Y.obj model.auxiliary)
 102     (r : ℕ) (n s t : ℤ) (hr : 2≤r)
 103     (hl : (normalizedExponent model.classicalTower f : ℤ)≤n)
 104     (hu : n≤(r:ℤ)-2+normalizedExponent model.classicalTower f) :
 105     FiniteExtensionWitness model X Y f r n s t 0 0 := by
 106   refine {
 107     page_ge_two := hr
 108     length_lower := hl
 109     length_upper := hu
 110     source_cycle := ⟨0,RepresentsOnPage.zero (by exact_mod_cast hr)⟩
 111     target_cycle := ⟨0,RepresentsOnPage.zero (by omega)⟩
 112     source := 0
 113     target := 0
 114     source_rep := ?_
 115     target_rep := ?_
 116     equation := ?_ }
 117   · have hlabel : finiteSourceForExtension model X Y f (r-1) s t 0=0 := by
 118       simp only [finiteSourceForExtension,finiteNuLabel,map_zero,regradeE2_zero]
 119     rw [hlabel]
 120     exact HasInfinityRepresentative.zero _ 2 (by decide) _
 121   · have hlabel : finiteTargetForExtension model X Y f (r-1) n s t hl 0=0 := by
 122       simp only [finiteTargetForExtension,finiteTargetLabel,target_nu_label_zero,
 123         map_zero,regradeE2_zero]
 124     rw [hlabel]
 125     exact HasInfinityRepresentative.zero _ 2 (by decide) _
 126   · apply (tower_extension_iff _ _ _ _ _ _).mpr
 127     refine ⟨by omega,0,0,?_,?_,?_⟩
 128     · simp only [map_zero,zero_comp]
