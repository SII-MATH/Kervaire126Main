  13 private theorem finite_top_boundary_detects_zero (X : ClassicalObject) (q : ℕ) (hq : 0 < q)
  14     (s t : ℤ) (z : Ambient model X (s,t))
  15     (hz : z ∈ boundaries model X q (s,t)) :
  16     Detects (model.convergence (.quotient q (.nu X))) (s,t,t-(q-1:ℕ))
  17       (finiteTargetLabel model X q s t (q-1) z) 0 := by
  18   let y : cycles model X ((q:ℤ)-t+(t-(q-1:ℕ))) (s,t) :=
  19     ⟨z, boundaries_le_cycles model X (s,t) q _ hz⟩
  20   have he : HasInfinityRepresentative _ 2 _ (finiteTargetLabel model X q s t (q-1) z) 0 := by
  21     apply (finite_target_label_iff X q hq s t (q-1) (by omega) y 0).mpr
  22     rw [map_zero]
  23     symm
  24     exact quotient_map_class_eq_zero (le_refl _) (le_refl _) y (by
  25       change z ∈ boundaries model X (1+t-(t-(q-1:ℕ))) (s,t)
  26       convert hz using 1
  27       congr 1
  28       omega)
  29   refine ⟨0,he,0,?_,?_⟩
  30   · exact map_zero _
  31   · simp only [map_zero]
  32 end
