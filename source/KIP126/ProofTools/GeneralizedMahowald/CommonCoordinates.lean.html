  18 /- The cofiber coordinates use exactly the quotient comparison already
  19 chosen for cNu; their common desuspension therefore has an actual strict
  20 representative comparison with cNu. -/
  21 private def commonCofiberTargetIso
  22     {C V : SyntheticSpectrum} (e : ℤ) (e₀ : C ≅ (SyntheticCategory.biShift (1,e)).obj V)

  24     (eQ : XModLambdaN ((SyntheticCategory.biShift (1,e)).obj V) r ≅
  25       (SyntheticCategory.biShift (1,e)).obj (XModLambdaN V r)) :
  26     XModLambdaN C r ≅ (SyntheticCategory.biShift (1,-(i:ℤ))).obj
  27       ((SyntheticCategory.biShift (0,(i:ℤ)+e)).obj (XModLambdaN V r)) :=
  28   (model.quotients.functoriality.functor r).mapIso e₀ ≪≫ eQ ≪≫
  29     (biShiftAddIso (0,(i:ℤ)+e) (1,-(i:ℤ)) (1,e)
  30       (by ext <;> dsimp; all_goals omega)).symm.app _

  48 /- Choose all actual cofiber/common-shift coordinates from the fixed T.
  49 The only quotient comparison assumption is obtained internally from the
  50 existing exact quotient-shift construction. -/
  51 private theorem normalized_common_cofiber_coordinates_exists
  52     (T : TriangleData model.auxiliary)
  53     (he : (normalizedExponent model.classicalTower T.f : ℤ) +
  54       normalizedExponent model.classicalTower T.g + normalizedExponent model.classicalTower T.h = 1)
  55     (hT : NormalizedTriangleCompatible model T he) (i r : ℕ) :
  56     let e : ℤ := normalizedExponent model.classicalTower T.f + normalizedExponent model.classicalTower T.g
  57     let V := model.nu.functor.obj (T.X.obj model.auxiliary)
  58     ∃ eQ : XModLambdaN ((SyntheticCategory.biShift (1,e)).obj V) r ≅
  59         (SyntheticCategory.biShift (1,e)).obj (XModLambdaN V r),
  60       XModLambdaN.incl _ r ≫ eQ.hom =
  61         (SyntheticCategory.biShift (1,e)).map (XModLambdaN.incl V r) ∧
  62       Nonempty (XModLambdaN (HasFunctorialCofiber.cofib (model.normalizedMap T.Y T.Z T.g).map) r ≅
  63         (SyntheticCategory.biShift (1,-(i:ℤ))).obj
  64           ((SyntheticCategory.biShift (0,(i:ℤ)+e)).obj (XModLambdaN V r))) := by
  65   dsimp only
  66   let e : ℤ := normalizedExponent model.classicalTower T.f + normalizedExponent model.classicalTower T.g
  67   letI := model.exactness.shiftCommShift (1,e)
  68   letI := model.exactness.shiftExact (1,e)
  69   obtain ⟨eQ,heQ,_⟩ := quotient_shift_exists model.quotients.coherence (1,e) r
  70     (model.nu.functor.obj (T.X.obj model.auxiliary))
  71   exact ⟨eQ,heQ,⟨commonCofiberTargetIso e (normalized_g_cofiber_iso T he hT) i r eQ⟩⟩

  73 /- The common-shift lift is exactly the positive vertical shift of the
  74 native cNu lift. This is an equality of actual sphere representatives and
  75 does not claim a comparison of arbitrary nonpermanent shifted E₂ labels. -/
  76 private theorem common_cofiber_desuspend_eq_shift_cNu
  77     (X : ClassicalObject) {C : SyntheticSpectrum} (e : ℤ)
  78     (e₀ : C ≅ (SyntheticCategory.biShift (1,e)).obj
  79       (model.nu.functor.obj (X.obj model.auxiliary)))
  80     (i r : ℕ)
  81     (eQ : XModLambdaN ((SyntheticCategory.biShift (1,e)).obj
  82         (model.nu.functor.obj (X.obj model.auxiliary))) r ≅
  83       (SyntheticCategory.biShift (1,e)).obj
  84         (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) r))
  85     (m w : ℤ) (c : BiHom m w (XModLambdaN C r)) :
  86     let ff := SyntheticCategory.biShift_fullyFaithful (Syn:=SyntheticSpectrum) (1,-(i:ℤ))
  87     ff.preimage ((biShiftAddIso (m-1,w+i) (1,-(i:ℤ)) (m,w)
  88         (by ext <;> dsimp; all_goals omega)).hom.app S_0_0 ≫ c ≫
  89       (commonCofiberTargetIso e e₀ i r eQ).hom) =
  90     homotopyRegrade (by simp) (by dsimp; omega)
  91       (shiftHomotopyClass (0,(i:ℤ)+e) (m-1,w-e)
  92         (quotientShiftDesuspend X (1,e) r eQ m w (c ≫ XModLambdaN.map e₀.hom r))) := by
  93   dsimp only
  94   let ff := SyntheticCategory.biShift_fullyFaithful (Syn:=SyntheticSpectrum) (1,-(i:ℤ))
  95   apply ff.map_injective
  96   rw [shift_regrade]
  97   simp only [Functor.map_comp,Functor.FullyFaithful.map_preimage]
  98   apply (cancel_epi ((biShiftAddIso (m-1,w+i) (1,-(i:ℤ)) (m,w)
  99     (by ext <;> dsimp; all_goals omega)).inv.app (S_0_0 : SyntheticSpectrum))).mp
 100   simp only [Iso.inv_hom_id_app_assoc]
 101   have hassoc := shift_hom_associativity model.quotients.coherence
 102     (m-1,w-e) (0,(i:ℤ)+e) (1,-(i:ℤ)) (m-1,w+i) (1,e) (m,w)
 103     (by ext <;> dsimp; all_goals omega) (by ext <;> dsimp; all_goals omega)
 104     (by ext <;> dsimp; all_goals omega) (by ext <;> dsimp; all_goals omega)
 105     (quotientShiftDesuspend X (1,e) r eQ m w (c ≫ XModLambdaN.map e₀.hom r))
 106   rw [←Functor.map_comp,←hassoc]
 107   simp only [quotientShiftDesuspend,Functor.FullyFaithful.map_preimage,
 108     commonCofiberTargetIso,Iso.trans_hom,Iso.symm_hom,Iso.app_hom,Category.assoc,
 109     Iso.inv_hom_id_app_assoc,Functor.mapIso_hom,LambdaQuotientFunctoriality.functor]

 114 extension lemmas; this is an actual isomorphism between them. -/

 116     (eA : XModLambdaN ((SyntheticCategory.biShift (0,a)).obj V) r ≅

 121   (SyntheticCategory.biShift (0,(i:ℤ))).mapIso eA ≪≫

 135 /- The flattening carries the same actual K, Jedge or residual t₀ preimage;
 136 there is no independent choice of a second May representative. -/

 139     (esusp : XModLambdaN (((SyntheticCategory.biShift (0,a)).obj V)⟦(1:ℤ)⟧) (i+q) ≅
 140       (XModLambdaN ((SyntheticCategory.biShift (0,a)).obj V) (i+q))⟦(1:ℤ)⟧)
 141     (eA : XModLambdaN ((SyntheticCategory.biShift (0,a)).obj V) (i+q) ≅
 142       (SyntheticCategory.biShift (0,a)).obj (XModLambdaN V (i+q)))
 143     (p : (SyntheticCategory.biShift (1,-(i:ℤ))).obj U ⟶
 144       XModLambdaN (((SyntheticCategory.biShift (0,a)).obj V)⟦(1:ℤ)⟧) (i+q)) :

 147       ff.preimage (p ≫ (commonEdgeTargetIso ((SyntheticCategory.biShift (0,a)).obj V) i q esusp).hom) ≫
 148         (commonEdgeFlattenIso V a i (i+q) eA).hom := by
 149   dsimp only
 150   apply (SyntheticCategory.biShift_fullyFaithful (Syn:=SyntheticSpectrum) (1,-(i:ℤ))).map_injective
 151   simp only [Functor.map_comp,Functor.FullyFaithful.map_preimage,
