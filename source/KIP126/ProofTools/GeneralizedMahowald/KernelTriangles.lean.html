 100 theorem shift_minus_one_preserves_sphere_evaluation {A B : SyntheticSpectrum}
 101     (f g : A ⟶ B)
 102     (h : ∀ (m w : ℤ) (a : BiHom m w A), a ≫ f = a ≫ g)
 103     (m w : ℤ) (b : BiHom m w ((SyntheticCategory.biShift (0,-1)).obj A)) :
 104     b ≫ (SyntheticCategory.biShift (0,-1)).map f =
 105       b ≫ (SyntheticCategory.biShift (0,-1)).map g := by
 106   let F := SyntheticCategory.biShift (Syn:=SyntheticSpectrum) (0,-1)
 107   let e : F.obj (Smn m (w+1)) ≅ Smn m w :=
 108     (SyntheticCategory.biShift_comp (m,w+1) (0,-1)).app S_0_0 ≪≫
 109       eqToIso (congrArg (fun p => (SyntheticCategory.biShift p).obj (S_0_0 : SyntheticSpectrum))
 110         (show (m,w+1)+(0,-1)=(m,w) by ext <;> simp))
 111   let a : BiHom m (w+1) A :=
 112     (SyntheticCategory.biShift_fullyFaithful (0,-1)).preimage (e.hom ≫ b)
 113   have ha : F.map a = e.hom ≫ b :=
 114     (SyntheticCategory.biShift_fullyFaithful (0,-1)).map_preimage _
 115   apply (cancel_epi e.hom).mp
 116   change e.hom ≫ b ≫ F.map f = e.hom ≫ b ≫ F.map g
 117   rw [←Category.assoc,←ha,←Functor.map_comp,h,Functor.map_comp,ha,Category.assoc]
