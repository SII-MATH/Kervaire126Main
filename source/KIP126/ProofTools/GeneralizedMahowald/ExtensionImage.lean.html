   8 /-! Private E₂ consequences of the finite g-extension in AimPaper Theorem 6.12.
   9 At the bottom weight no differential enters, so the detected associated graded
  10 class determines the E₂ label. A positive-length extension has zero leading
  11 image; a length-zero extension has the specified target as its leading image.
  12 These lemmas do not identify the May comparison maps or shifted ν labels. -/

  25 /-- An E₂ label representing zero at infinity is zero when no differential enters. -/
  26 private theorem infinity_zero_no_incoming (A : SyntheticAdamsSS) (p : Tridegree)
  27     (hin : ∀ r : ℤ, 2 ≤ r → A.sequence.d r (p-A.sequence.diffDeg r)=0)
  28     (x : A.E₂ p) (hx : HasInfinityRepresentative A 2 p x 0) : x=0 := by
  29   obtain ⟨_,z,hz,hz0⟩ := hx
  30   let D := A.sequence.ssData p
  31   let k : WithTop ℕ := ↑(2-A.sequence.r₀).toNat
  32   have hB : D.B ⊤ = D.B k := boundaries_top_eq_of_d_eq_zero A.sequence 2
  33     (by rw [A.firstPage]) p hin
  34   have hb := (subobject_cokernel_π_eq_zero_iff (D.B ⊤) (D.Z ⊤) (D.B_le_Z _) z).mp hz0
  35   rw [hB] at hb
  36   rw [← hz]
  37   let z' := (Subobject.ofLE (D.Z ⊤) (D.Z k) (D.Z_anti le_top)) z
  38   change D.pageπ k z' = 0
  39   apply (subobject_cokernel_π_eq_zero_iff (D.B k) (D.Z k) (D.B_le_Z _) z').mpr
  40   change ((Subobject.ofLE (D.Z ⊤) (D.Z k) (D.Z_anti le_top)) ≫ (D.Z k).arrow) z ∈ _
  41   rwa [Subobject.ofLE_arrow]

  43 /-- A representative lying one filtration higher has zero associated graded class. -/
  44 private theorem detects_high_infinity_zero (A : SyntheticObject) (p : Tridegree)
  45     (x : (model.family.obj (A.obj model.nu model.auxiliary)).E₂ p)
  46     (a : BiHom (p.2.1-p.1) p.2.2 (A.obj model.nu model.auxiliary))
  47     (ha : Detects (model.convergence A) p x a)
  48     (hhigh : FiltrationAtLeast model.syntheticTower (p.1+1) a) :
  49     HasInfinityRepresentative _ 2 p x 0 := by
  50   obtain ⟨e,he,a',ha',hgr⟩ := ha
  51   have hgr0 : (model.syntheticTower.filtration _).toAssociatedGraded p.1
  52       (p.2.1-p.1,p.2.2) a' = 0 := by
  53     apply ((model.syntheticTower.filtration _).toAssociatedGraded_eq_zero_iff _ _ _).mpr
  54     rw [ha']
  55     simpa only [SyntheticTowerPresentation.filtration,OrderIso.apply_symm_apply,
  56       FiltrationAtLeast] using hhigh
  57   have he0 : e=0 := by
  58     apply ((model.convergence A).identification p).toLinearEquiv.injective
  59     change ((model.convergence A).identification p).hom e =
  60       ((model.convergence A).identification p).hom 0
  61     rw [map_zero,hgr,hgr0]
  62   simpa only [he0] using he

  64 /-- The bottom of a finite ν quotient has no incoming differentials. -/
  65 private theorem finite_bottom_high_label_zero (Y : ClassicalObject) (q : ℕ) (hq : 0<q)
  66     (s t w : ℤ) (hw : t ≤ w)
  67     (x : (model.family.obj (XModLambdaN (model.nu.functor.obj (Y.obj model.auxiliary)) q)).E₂ (s,t,w))
  68     (a : BiHom (t-s) w (XModLambdaN (model.nu.functor.obj (Y.obj model.auxiliary)) q))
  69     (ha : Detects (model.convergence (.quotient q (.nu Y))) (s,t,w) x a)
  70     (hhigh : FiltrationAtLeast model.syntheticTower (s+1) a) : x=0 := by
  71   apply infinity_zero_no_incoming _ (s,t,w) _ x
  72     (detects_high_infinity_zero (.quotient q (.nu Y)) (s,t,w) x a ha hhigh)
  73   intro r hr
  74   have hdeg : (s,t,w)-(model.family.obj
  75       (XModLambdaN (model.nu.functor.obj (Y.obj model.auxiliary)) q)).sequence.diffDeg r =
  76       (s-r,t-r+1,w) := by
  77     change (s,t,w)-(model.family.functor.obj _).diffDeg r = _
  78     rw [model.family.differentialDegree]
  79     simp only [syntheticAdamsRawShift,Prod.mk_sub_mk,sub_zero]
  80     congr 2
  81     omega
  82   have hzero := KIP126.Literature.Standard.finite_quotient_vanishing Y q hq r hr
  83     (s-r) (t-r+1) w (Or.inl (by omega))
  84   have hzero' : IsZero ((model.family.obj
  85       (XModLambdaN (model.nu.functor.obj (Y.obj model.auxiliary)) q)).Page r
  86       ((s,t,w)-(model.family.obj
  87         (XModLambdaN (model.nu.functor.obj (Y.obj model.auxiliary)) q)).sequence.diffDeg r)) := by
  88     rw [hdeg]
  89     exact hzero
  90   exact hzero'.eq_of_src _ _

  92 /-- Positive extension length forces the leading finite E₂ image to vanish. -/
  93 private theorem finite_positive_extension_map_e2_zero
  94     (X Y : ClassicalObject) (g : X.obj model.auxiliary ⟶ Y.obj model.auxiliary)
  95     (r : ℕ) (m s t : ℤ) (hm : 0<m)
  96     (x : Ambient model X (s,t)) (y : Ambient model Y (s+m,t+m))
  97     (W : FiniteExtensionWitness model X Y g r m s t x y) :
  98     familyPageMap model.family (finiteExtensionMap model X Y g (r-1)) 2
  99       (s,(t-s)+s,t+normalizedExponent model.classicalTower g)
 100       (finiteSourceForExtension model X Y g (r-1) s t x) = 0 := by
 101   let A := SyntheticObject.quotient (r-1) (extensionSourceObject model X Y g)
 102   let B := SyntheticObject.quotient (r-1) (.nu Y)
 103   let p := (t-s,t+normalizedExponent model.classicalTower g)
 104   obtain ⟨_,a,b,hab,hax,hby⟩ := (tower_extension_iff
 105     (model.convergentMap A B (finiteExtensionMap model X Y g (r-1)))
 106     p s m W.source W.target).mp W.equation
 107   have ha := source_frame_detects (model.convergence A) p s W.source
 108     _ W.source_rep a hax
 109   have hmap := detects_map A B (finiteExtensionMap model X Y g (r-1))
 110     (s,p.1+s,p.2) _ _ ha
 111   have hbfil : FiltrationAtLeast model.syntheticTower (s+m) (((model.syntheticTower.filtration
 112       (B.obj model.nu model.auxiliary)).F (s+m) p).arrow b) := by
 113     have hmem : (((model.syntheticTower.filtration (B.obj model.nu model.auxiliary)).F (s+m) p).arrow b)
 114         ∈ ModuleCat.subobjectModule _ ((model.syntheticTower.filtration
 115           (B.obj model.nu model.auxiliary)).F (s+m) p) := ⟨b,rfl⟩
 116     simpa only [SyntheticTowerPresentation.filtration,OrderIso.apply_symm_apply,
 117       FiltrationAtLeast] using hmem
 118   have hhigh := model.syntheticTower.filtrationSubmodule_antitone
 119     (B.obj model.nu model.auxiliary) p (show s+1≤s+m by omega) hbfil
 120   have hhigh' : FiltrationAtLeast model.syntheticTower (s+1)
 121       ((homotopyRegrade (show p.1=(p.1+s)-s by omega) rfl
 122         (((model.syntheticTower.filtration (A.obj model.nu model.auxiliary)).F s p).arrow a)) ≫
 123           finiteExtensionMap model X Y g (r-1)) := by
 124     have hcast (m m' w : ℤ) (he : m=m') (a : BiHom m w (A.obj model.nu model.auxiliary)) :
 125         homotopyRegrade he rfl a ≫ finiteExtensionMap model X Y g (r-1) =
 126           homotopyRegrade he rfl (a ≫ finiteExtensionMap model X Y g (r-1)) := by
 127       subst m'; rfl
 128     rw [hcast,hab]
 129     have hf (m m' w : ℤ) (he : m=m')
 130         (b : BiHom m w (B.obj model.nu model.auxiliary))
 131         (hb : FiltrationAtLeast model.syntheticTower (s+1) b) :
 132         FiltrationAtLeast model.syntheticTower (s+1) (homotopyRegrade he rfl b) := by
 133       subst m'; exact hb
 134     exact hf _ _ _ _ _ hhigh
 135   exact finite_bottom_high_label_zero Y (r-1) (by have := W.page_ge_two; omega)
 136     s (p.1+s) p.2 (by dsimp [p]; omega) _ _ hmap hhigh'

 139 /-- The same vanishing before quotienting, by the retained E₂ window. -/
 140 private theorem positive_extension_map_e2_zero
 141     (X Y : ClassicalObject) (g : X.obj model.auxiliary ⟶ Y.obj model.auxiliary)
 142     (r : ℕ) (m s t : ℤ) (hm : 0<m)
 143     (x : Ambient model X (s,t)) (y : Ambient model Y (s+m,t+m))
 144     (W : FiniteExtensionWitness model X Y g r m s t x y) :
 145     familyPageMap model.family (model.normalizedMap X Y g).map 2
 146       (s,(t-s)+s,t+normalizedExponent model.classicalTower g)
 147       (infiniteSourceForExtension model X Y g s t x) = 0 := by
 148   let A := (extensionSourceObject model X Y g).obj model.nu model.auxiliary
 149   let B := model.nu.functor.obj (Y.obj model.auxiliary)
 150   let p : Tridegree := (s,(t-s)+s,t+normalizedExponent model.classicalTower g)
 151   have hsource : finiteSourceForExtension model X Y g (r-1) s t x =
 152       familyPageMap model.family (XModLambdaN.incl A (r-1)) 2 p
 153         (infiniteSourceForExtension model X Y g s t x) := by
 154     unfold finiteSourceForExtension infiniteSourceForExtension finiteNuLabel
 155     rw [page_regrade]
 156     rfl
 157   have hfinite := finite_positive_extension_map_e2_zero X Y g r m s t hm x y W
 158   rw [hsource] at hfinite
 159   change familyPageMap model.family (XModLambdaN.map (model.normalizedMap X Y g).map (r-1)) 2 p
 160     (familyPageMap model.family (XModLambdaN.incl A (r-1)) 2 p
 161       (infiniteSourceForExtension model X Y g s t x))=0 at hfinite
 162   dsimp only [A,extensionSourceObject,SyntheticObject.obj] at hfinite
 163   rw [← ModuleCat.comp_apply,family_page_comp] at hfinite
 164   change familyPageMap model.family
 165     (XModLambdaN.incl A (r-1) ≫ XModLambdaN.map (model.normalizedMap X Y g).map (r-1)) 2 p
 166       (infiniteSourceForExtension model X Y g s t x)=0 at hfinite
 167   rw [←XModLambdaN.incl_naturality,←family_page_comp,ModuleCat.comp_apply] at hfinite
 168   by_cases he : normalizedExponent model.classicalTower g = 0
 169   · haveI : IsIso (familyPageMap model.family (XModLambdaN.incl B (r-1)) 2 p) := by
 170       exact KIP126.Literature.Standard.finite_quotient_retained Y (r-1)
 171         (by have := W.page_ge_two; omega) s ((t-s)+s) (t+normalizedExponent model.classicalTower g)
 172         (by have hr := W.page_ge_two; simp only [he,Nat.cast_zero,add_zero,sub_add_cancel,sub_self]; omega)
 173     apply (ModuleCat.mono_iff_injective
 174       (familyPageMap model.family (XModLambdaN.incl B (r-1)) 2 p)).mp inferInstance
 175     change familyPageMap model.family (XModLambdaN.incl B (r-1)) 2 p
 176       (familyPageMap model.family (model.normalizedMap X Y g).map 2 p
 177         (infiniteSourceForExtension model X Y g s t x)) =
 178       familyPageMap model.family (XModLambdaN.incl B (r-1)) 2 p 0
 179     rw [map_zero]
 180     exact hfinite
 181   · have hz := KIP126.Literature.Standard.above_diagonal Y 2 (by omega) s ((t-s)+s)
 182       (t+normalizedExponent model.classicalTower g) (by omega)
 183     exact ModuleCat.isZero_iff_subsingleton.mp hz |>.elim _ _

 188     (Y : ClassicalObject) (q : ℕ) (hq : 0<q) (δ p : ℤ × ℤ) (s : ℤ)

 191       model.nu model.auxiliary)).E₂ (s,(p+δ).1+s,(p+δ).2))
 192     (a : BiHom ((p+δ).1+s-s) (p+δ).2
 193       ((SyntheticObject.shift δ (.quotient q (.nu Y))).obj model.nu model.auxiliary))

 195       (s,(p+δ).1+s,(p+δ).2) x a)
 196     (hhigh : FiltrationAtLeast model.syntheticTower (s+1) a) : x=0 := by

 202       (model.family.obj (Z.obj model.nu model.auxiliary)).sequence.diffDeg r =

 210   let e := (essShift δ).page (.quotient q (.nu Y)) r hr (p.1+1,p.2) (s-r)

 214     rw [hdeg]
 215     exact hzero

 220     (X Y : ClassicalObject) (g : X.obj model.auxiliary ⟶ Y.obj model.auxiliary)
 221     (r : ℕ) (m s t : ℤ) (hm : 0<m) (δ : ℤ × ℤ)
 222     (x : Ambient model X (s,t)) (y : Ambient model Y (s+m,t+m))

 227       (finiteExtensionMap model X Y g (r-1))) 2 (s,(p+δ).1+s,(p+δ).2)
 228       (((essShift δ).page A 2 (by decide) p s).hom

 283       (model.nu.functor.obj (Y.obj model.auxiliary)) q))

 294 /-- A zero-length finite extension has the specified target as its leading image. -/
 295 private theorem finite_zero_extension_map_e2_target
 296     (X Y : ClassicalObject) (g : X.obj model.auxiliary ⟶ Y.obj model.auxiliary)
 297     (r : ℕ) (s t : ℤ)
 298     (x : Ambient model X (s,t)) (y : Ambient model Y (s+0,t+0))
 299     (W : FiniteExtensionWitness model X Y g r 0 s t x y) :
 300     familyPageMap model.family (finiteExtensionMap model X Y g (r-1)) 2
 301       (s,(t-s)+s,t+normalizedExponent model.classicalTower g)
 302       (finiteSourceForExtension model X Y g (r-1) s t x) =
 303       regradeE2 (by simp) (finiteTargetForExtension model X Y g (r-1) 0 s t W.length_lower y) := by
 304   let A := SyntheticObject.quotient (r-1) (extensionSourceObject model X Y g)
 305   let B := SyntheticObject.quotient (r-1) (.nu Y)
 306   let p := (t-s,t+normalizedExponent model.classicalTower g)
 307   obtain ⟨_,a,b,hab,hax,hby⟩ := (tower_extension_iff
 308     (model.convergentMap A B (finiteExtensionMap model X Y g (r-1)))
 309     p s 0 W.source W.target).mp W.equation
 310   have ha := source_frame_detects (model.convergence A) p s W.source
 311     _ W.source_rep a hax
 312   have hb := source_frame_detects (model.convergence B) p (s+0) W.target
 313     _ W.target_rep b hby
 314   have hmap := detects_map A B (finiteExtensionMap model X Y g (r-1))
 315     (s,p.1+s,p.2) _ _ ha
 316   have hcast (m m' w : ℤ) (he : m=m') (a : BiHom m w (A.obj model.nu model.auxiliary)) :
 317       homotopyRegrade he rfl a ≫ finiteExtensionMap model X Y g (r-1) =
 318         homotopyRegrade he rfl (a ≫ finiteExtensionMap model X Y g (r-1)) := by
 319     subst m'; rfl
 320   rw [hcast,hab] at hmap
 321   have hb' : Detects (model.convergence B) (s,p.1+s,p.2)
 322       (regradeE2 (by dsimp [p]; simp) (finiteTargetForExtension model X Y g (r-1) 0 s t W.length_lower y))
 323       (homotopyRegrade (show p.1=(p.1+s)-s by omega) rfl
 324         (((model.syntheticTower.filtration (B.obj model.nu model.auxiliary)).F (s+0) p).arrow b)) := by
 325     have transport {j k : Tridegree} (he : j=k)
 326         (z : (model.family.obj (B.obj model.nu model.auxiliary)).E₂ j)
 327         {u w : ℤ} (b : BiHom u w (B.obj model.nu model.auxiliary))
 328         (hu : u=j.2.1-j.1) (hw : w=j.2.2)
 329         (hu' : u=k.2.1-k.1) (hw' : w=k.2.2)
 330         (h : Detects (model.convergence B) j z (homotopyRegrade hu hw b)) :
 331         Detects (model.convergence B) k (regradeE2 he z) (homotopyRegrade hu' hw' b) := by
 332       subst k
 333       exact h
 334     exact transport (by dsimp [p]; simp) _ _ _ _ _ _ hb
 335   exact finite_bottom_detects_label_unique Y (r-1) (by have := W.page_ge_two; omega)
 336     s (p.1+s) p.2 (by dsimp [p]; omega) _ _ _ hmap hb'

 338 /-- Remove the quotient from the zero-length leading-image equation. -/
