  21 private theorem finite_target_rho_detects (X : ClassicalObject)
  22     (i j k : ℕ) (hki : k < i) (hij : i ≤ j) (s t : ℤ)
  23     (x : Ambient model X (s,t))
  24     (a : BiHom (t-s) (t-k) (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) i))
  25     (ha : Detects (model.convergence (.quotient i (.nu X))) (s,t,t-k)
  26       (finiteTargetLabel model X i s t k x) a)
  27     (b : BiHom (t-s) (t-k) (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) j))
  28     (hb : b≫(model.quotients.tower _).rho i j hij=a) :
  29     Detects (model.convergence (.quotient j (.nu X))) (s,t,t-k)
  30       (finiteTargetLabel model X j s t k x) b := by
  31   have hlabel : nuWeightLabel model X s t (t-k) (by omega) x =
  32       targetNuLabel model X s t k x := by
  33     have hk' : (t-(t-k)).toNat=k := by omega
  34     have transport (a b : ℕ) (hab : a=b)
  35         (h : (s,t,t-a)=(s,t,t-b)) :
  36         regradeE2 h (targetNuLabel model X s t a x) = targetNuLabel model X s t b x := by
  37       subst b
  38       rfl
  39     exact transport _ k hk' _
  40   have ha' : Detects (model.convergence (.quotient i (.nu X))) (s,t,t-k)
  41       ((model.family.quotientProjection (model.nu.functor.obj (X.obj model.auxiliary)) i).pageMap
  42         2 (s,t,t-k) (nuWeightLabel model X s t (t-k) (by omega) x)) a := by
  43     rw [hlabel]
  44     exact ha
  45   have hh := rho_lift_detects X i j (by omega) hij s t (t-k) (by omega) x a ha' b hb
  46   rw [hlabel] at hh
  47   exact hh

  49 private theorem finite_target_rho_bottom_lift (X : ClassicalObject)
  50     (i j k : ℕ) (hki : k < i) (hij : i ≤ j) (s t : ℤ)
  51     (x : Ambient model X (s,t))
  52     (a : BiHom (t-s) (t-k) (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) i))
  53     (ha : Detects (model.convergence (.quotient i (.nu X))) (s,t,t-k)
  54       (finiteTargetLabel model X i s t k x) a)
  55     (b : BiHom (t-s) (t-k) (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) j))
  56     (hb : b≫(model.quotients.tower _).rho i j hij=a) :
  57     ∃ c : BiHom (t-s) t (XModLambdaN (nuZero model X) (j-k)),
  58       c≫(model.quotients.tower _).rho 1 (j-k) (by omega) = firstLabel model X s t x := by
  59   obtain ⟨c,hc⟩ := finite_detected_target_bottom_lift X j k (by omega) s t x b
  60     (finite_target_rho_detects X i j k hki hij s t x a ha b hb)
  61   let f : model.nu.functor.obj (X.obj model.auxiliary) ⟶ nuZero model X :=
  62     SyntheticCategory.biShift_zero.inv.app _
  63   refine ⟨c≫XModLambdaN.map f (j-k),?_⟩
  64   rw [Category.assoc,←(model.quotients.natural f).rho_naturality (by omega),←Category.assoc,hc]
  65   simp only [f,Category.assoc,←model.quotients.functoriality.map_comp,
  66     Iso.hom_inv_id_app,model.quotients.functoriality.map_id,Category.comp_id]
  67 
  68 end
