  21 private theorem shifted_nu_to_shifted_finite_no_return (X Y : ClassicalObject)
  22     (r : ℕ) (hr : 0<r) (δ ε : ℤ×ℤ)
  23     (hgap : (r:ℤ)≤δ.1-δ.2-(ε.1-ε.2)) (m w : ℤ)
  24     (c : BiHom m w ((SyntheticCategory.biShift δ).obj
  25       (model.nu.functor.obj (X.obj model.auxiliary))))
  26     (z : (SyntheticCategory.biShift δ).obj
  27       (model.nu.functor.obj (X.obj model.auxiliary)) ⟶
  28       (SyntheticCategory.biShift ε).obj
  29         (XModLambdaN (model.nu.functor.obj (Y.obj model.auxiliary)) r)) : c≫z=0 := by
  30   let A : SyntheticObject := .shift δ (.nu X)
  31   let B : SyntheticObject := .shift ε (.quotient r (.nu Y))
  32   let L : ℤ := w-m+δ.1-δ.2
  33   have hzA (k : ℤ) (hk : k<L) :
  34       IsZero (((model.family.obj (A.obj model.nu model.auxiliary)).sequence.ssData (k,m+k,w)).eInfty) := by
  35     have hz := nu_infinity_zero_above X k (m-δ.1+k) (w-δ.2)
  36       (by dsimp [L] at hk; omega)
  37     let e := towerInfinityIso (model.convergence (.nu X)) (model.convergence A)
  38       ((essShift δ).filtered (.nu X) (m-δ.1,w-δ.2)) k
  39     have he := hz.of_iso e.symm
  40     simpa only [e,A,Prod.fst_add,Prod.snd_add,sub_add_cancel] using he
  41   have hc0 := filtration_at_zero model.syntheticTower _ m w c
  42   have hcL : FiltrationAtLeast model.syntheticTower L c := by
  43     by_cases hL : L≤0
  44     · exact model.syntheticTower.filtrationSubmodule_antitone _ (m,w) hL hc0
  45     · exact (filtration_iff_of_infinity_zero (model.convergence A) 0 L m w
  46         (by omega) (fun k _ hk => hzA k hk) c).mp hc0
  47   have hcz := map_preserves_filtration (model.convergentMap A B z) (m,w) L c hcL
  48   have hzB (k : ℤ) (hk : L≤k) :
  49       IsZero (((model.family.obj (B.obj model.nu model.auxiliary)).sequence.ssData (k,m+k,w)).eInfty) := by
  50     have hz := finite_infinity_zero_outside Y r hr k (m-ε.1+k) (w-ε.2)
  51       (by dsimp [L] at hk; omega)
  52     let e := towerInfinityIso (model.convergence (.quotient r (.nu Y))) (model.convergence B)
  53       ((essShift ε).filtered (.quotient r (.nu Y)) (m-ε.1,w-ε.2)) k
  54     have he := hz.of_iso e.symm
  55     simpa only [e,B,Prod.fst_add,Prod.snd_add,sub_add_cancel] using he
  56   apply shifted_quotient_homotopy_separated (.nu Y) r ε m w (c≫z)
  57   intro k
  58   by_cases hk : L≤(k:ℤ)
  59   · exact (filtration_iff_of_infinity_zero (model.convergence B) L k m w hk
  60       (fun j hj _ => hzB j hj) (c≫z)).mp hcz
  61   · exact model.syntheticTower.filtrationSubmodule_antitone _ (m,w) (by omega) hcz

  63 /- Into the high shifted final quotient, any two actual finite-source
  64 extensions agreeing before quotienting have identical action on all spheres. -/
  65 private theorem finite_source_high_target_extensions_equal
  66     (X Y : ClassicalObject) (q r : ℕ) (hr : 0<r) (a b : ℤ)
  67     (hgap : (r:ℤ)≤q+b-a+1)
  68     (u v : XModLambdaN ((SyntheticCategory.biShift (0,a)).obj
  69         (model.nu.functor.obj (X.obj model.auxiliary))) q ⟶
  70       (SyntheticCategory.biShift (0,b)).obj
  71         (XModLambdaN (model.nu.functor.obj (Y.obj model.auxiliary)) r))
  72     (hincl : XModLambdaN.incl _ q≫u=XModLambdaN.incl _ q≫v)
  73     (m w : ℤ)
  74     (c : BiHom m w (XModLambdaN ((SyntheticCategory.biShift (0,a)).obj
  75       (model.nu.functor.obj (X.obj model.auxiliary))) q)) : c≫u=c≫v := by
  76   apply quotient_maps_equal_on_spheres_of_no_return q u v hincl
  77   intro m' w' d z
  78   let V := model.nu.functor.obj (X.obj model.auxiliary)
  79   let ε : (((SyntheticCategory.biShift (0,-(q:ℤ))).obj
  80       ((SyntheticCategory.biShift (0,a)).obj V)))⟦(1:ℤ)⟧ ≅
  81       (SyntheticCategory.biShift (1,a-q)).obj V :=
  82     (biSuspensionIso (q:ℤ) ((SyntheticCategory.biShift (0,a)).obj V)).symm ≪≫
  83       (SyntheticCategory.biShift_comp (0,a) (1,-(q:ℤ))).app V ≪≫
  84       eqToIso (by simp [sub_eq_add_neg])
  85   have hz := shifted_nu_to_shifted_finite_no_return X Y r hr (1,a-q) (0,b)
  86     (by dsimp; omega) m' w' (d≫ε.hom) (ε.inv≫z)
  87   simpa only [Category.assoc,Iso.hom_inv_id_assoc] using hz

  90 private theorem sphere_nu_high_target_map_zero
  91     (Y X : ClassicalObject) (n w₀ a b : ℤ) (r : ℕ) (hr : 0<r)
  92     (hgap : (r:ℤ)≤b-a)
  93     (eY : model.nu.functor.obj (Y.obj model.auxiliary) ≅ Smn n w₀)
  94     (d : (SyntheticCategory.biShift (0,a)).obj
  95         (model.nu.functor.obj (Y.obj model.auxiliary)) ⟶
  96       (SyntheticCategory.biShift (0,b)).obj
  97         (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) r)) : d=0 := by
  98   let e : (SyntheticCategory.biShift (0,a)).obj
  99       (model.nu.functor.obj (Y.obj model.auxiliary)) ≅ Smn n (w₀+a) :=
 100     (SyntheticCategory.biShift (0,a)).mapIso eY ≪≫
 101       (SyntheticCategory.biShift_comp (n,w₀) (0,a)).app S_0_0 ≪≫
 102       eqToIso (by simp [Smn])
 103   apply (cancel_epi e.inv).mp
 104   simpa only [comp_zero] using
 105     shifted_nu_to_shifted_finite_no_return Y X r hr (0,a) (0,b)
 106       (by dsimp; omega) n (w₀+a) e.inv d

 108 private theorem sphere_nu_high_target_extension
 109     (Y X : ClassicalObject) (n w₀ a b : ℤ) (q r : ℕ) (hr : 0<r)
 110     (hgap : (r:ℤ)≤q+b-a)
 111     (eY : model.nu.functor.obj (Y.obj model.auxiliary) ≅ Smn n w₀)
 112     (d : (SyntheticCategory.biShift (0,a)).obj
 113         (model.nu.functor.obj (Y.obj model.auxiliary)) ⟶
 114       (SyntheticCategory.biShift (0,b)).obj
 115         (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) r)) :
 116     ∃ dQ : XModLambdaN ((SyntheticCategory.biShift (0,a)).obj
 117         (model.nu.functor.obj (Y.obj model.auxiliary))) q ⟶
 118       (SyntheticCategory.biShift (0,b)).obj
 119         (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) r),
 120       XModLambdaN.incl _ q≫dQ=d := by
 121   let V := model.nu.functor.obj (Y.obj model.auxiliary)
 122   let A := (SyntheticCategory.biShift (0,a)).obj V
 123   let ε : (SyntheticCategory.biShift (0,-(q:ℤ))).obj A ≅
 124       (SyntheticCategory.biShift (0,a-q)).obj V :=
 125     (SyntheticCategory.biShift_comp (0,a) (0,-(q:ℤ))).app V ≪≫
 126       eqToIso (by simp [sub_eq_add_neg])
 127   have hz := sphere_nu_high_target_map_zero Y X n w₀ (a-q) b r hr
 128     (by omega) eY (ε.inv≫lambdaPow q A≫d)
 129   have hzero : lambdaPow q A≫d=0 := by
 130     apply (cancel_epi ε.inv).mp
 131     simpa only [comp_zero] using hz
 132   obtain ⟨dQ,hdQ⟩ := (XModLambdaN.cofiberTriangle A q).yoneda_exact₂
 133     (XModLambdaN.cofiberTriangle_distinguished A q) d hzero
 134   exact ⟨dQ,hdQ.symm⟩

 136 private theorem detects_zero_of_detects_same_label_zero
 137     (D : SyntheticObject) (p : Tridegree)
 138     (x : (model.family.obj (D.obj model.nu model.auxiliary)).E₂ p)
 139     (a : BiHom (p.2.1-p.1) p.2.2 (D.obj model.nu model.auxiliary))
 140     (ha : Detects (model.convergence D) p x a)
 141     (hzero : Detects (model.convergence D) p x 0) :
 142     Detects (model.convergence D) p 0 a := by
 143   obtain ⟨e,he,a',ha',hga⟩ := ha
 144   obtain ⟨e0,he0,b,hb,hgb⟩ := hzero
 145   have hb0 : b=0 := (ModuleCat.mono_iff_injective
 146     ((model.syntheticTower.filtration (D.obj model.nu model.auxiliary)).F p.1
 147       (p.2.1-p.1,p.2.2)).arrow).mp inferInstance (hb.trans (map_zero _).symm)
 148   have he00 : e0=0 := ((model.convergence D).identification p).toLinearEquiv.injective (by
 149     change ((model.convergence D).identification p).hom e0 =
 150       ((model.convergence D).identification p).hom 0
 151     rw [hgb,hb0,map_zero,map_zero])
 152   have hee : e=0 := (he.unique he0).trans he00
 153   refine ⟨0,HasInfinityRepresentative.zero _ 2 (by omega) p,a',ha',?_⟩
 154   simpa only [hee] using hga

 157 /- A complete elementwise May correction for a genuine sphere source.
 158 Both defect summands are extended through Qq by actual triangle exactness.
 159 The correction is alpha dQ for the same alpha; no E2 map subtraction is used. -/
 160 private theorem sphere_source_actual_edge_correction
 161     (Y X Z : ClassicalObject) (n w₀ a b cWeight : ℤ)
 162     (q r i : ℕ) (hr : 0 < r) (hi : 0 < i)
 163     (hgapC : (r:ℤ)≤q+b-a) (hgapD : (r:ℤ)≤q+cWeight-a)
 164     (hkill : (i:ℤ)≤b-a)
 165     (eY : model.nu.functor.obj (Y.obj model.auxiliary) ≅ Smn n w₀)
 166     (A C D : SyntheticObject)
 167     (hA : A=.shift (0,a) (.nu Y))
 168     (hC : C=.shift (0,b) (.quotient r (.nu X)))
 169     (hD : D=.shift (0,cWeight) (.quotient r (.nu Z)))
 170     (d : A.obj model.nu model.auxiliary ⟶ C.obj model.nu model.auxiliary)
 171     (h : C.obj model.nu model.auxiliary ⟶ D.obj model.nu model.auxiliary)
 172     (R : A.obj model.nu model.auxiliary ⟶ D.obj model.nu model.auxiliary)
 173     (K J : (SyntheticObject.quotient q A).obj model.nu model.auxiliary ⟶ D.obj model.nu model.auxiliary)
 174     (rho : C.obj model.nu model.auxiliary ⟶ (SyntheticCategory.biShift (0,b)).obj
 175       (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) i))
 176     (hedge : XModLambdaN.incl (A.obj model.nu model.auxiliary) q≫K-
 177       XModLambdaN.incl (A.obj model.nu model.auxiliary) q≫J=d≫h-R)
 178     (p : Tridegree) (x : (model.family.obj (A.obj model.nu model.auxiliary)).E₂ p)
 179     (alpha : BiHom (p.2.1-p.1) p.2.2 ((SyntheticObject.quotient q A).obj model.nu model.auxiliary))
 180     (v : BiHom (p.2.1-p.1) p.2.2 (C.obj model.nu model.auxiliary))
 181     (halpha : Detects (model.convergence (SyntheticObject.quotient q A)) p
 182       (familyPageMap model.family (XModLambdaN.incl (A.obj model.nu model.auxiliary) q) 2 p x) alpha)
 183     (hv : v≫h=alpha≫K)
 184     (hRzero : Detects (model.convergence D) p (familyPageMap model.family R 2 p x) 0) :
 185     ∃ (dQ : (SyntheticObject.quotient q A).obj model.nu model.auxiliary ⟶ C.obj model.nu model.auxiliary)
 186       (v' : BiHom (p.2.1-p.1) p.2.2 (C.obj model.nu model.auxiliary)),
 187       XModLambdaN.incl (A.obj model.nu model.auxiliary) q≫dQ=d ∧ v'=v-alpha≫dQ ∧
 188       Detects (model.convergence C) p (familyPageMap model.family d 2 p x) (alpha≫dQ) ∧
 189       v'≫rho=v≫rho ∧
 190       Detects (model.convergence D) p
 191         (familyPageMap model.family J 2 p
 192           (familyPageMap model.family (XModLambdaN.incl (A.obj model.nu model.auxiliary) q) 2 p x)) (v'≫h) := by
 193   subst A
 194   subst C
 195   subst D
 196   dsimp only [SyntheticObject.obj] at d h R K J rho x alpha v halpha hv hedge hRzero ⊢
 197   obtain ⟨dQ,hdQ⟩ := sphere_nu_high_target_extension Y X n w₀ a b q r hr hgapC eY d
 198   obtain ⟨RQ,hRQ⟩ := sphere_nu_high_target_extension Y Z n w₀ a cWeight q r hr hgapD eY R
 199   have hdRho : d≫rho=0 := sphere_nu_high_target_map_zero Y X n w₀ a b i hi hkill eY (d≫rho)
 200   have hincRho : XModLambdaN.incl _ q≫(dQ≫rho)=XModLambdaN.incl _ q≫0 := by
 201     rw [←Category.assoc,hdQ,hdRho,comp_zero]
 202   have hcdRho : (alpha≫dQ)≫rho=0 := by
 203     have heq := finite_source_high_target_extensions_equal Y X q i hi a b
 204       (by omega) (dQ≫rho) 0 hincRho (p.2.1-p.1) p.2.2 alpha
 205     simpa only [Category.assoc,comp_zero] using heq
 206   have hinc : XModLambdaN.incl _ q≫(K-J) = XModLambdaN.incl _ q≫(dQ≫h-RQ) := by
 207     simp only [Preadditive.comp_sub,←Category.assoc,hdQ,hRQ]
 208     exact hedge
 209   have heq := finite_source_high_target_extensions_equal Y Z q r hr a cWeight
 210     (by omega) (K-J) (dQ≫h-RQ) hinc (p.2.1-p.1) p.2.2 alpha
 211   have heq' : alpha≫K-alpha≫J=(alpha≫dQ)≫h-alpha≫RQ := by
 212     simpa only [Preadditive.comp_sub,Category.assoc] using heq
 213   have hmath : (v-alpha≫dQ)≫h=alpha≫J-alpha≫RQ := by
 214     rw [Preadditive.sub_comp,hv]
 215     calc
 216       alpha≫K-(alpha≫dQ)≫h = (alpha≫K-alpha≫J)+alpha≫J-(alpha≫dQ)≫h := by abel
 217       _ = ((alpha≫dQ)≫h-alpha≫RQ)+alpha≫J-(alpha≫dQ)≫h := by rw [heq']
 218       _ = alpha≫J-alpha≫RQ := by abel
 219   have hdLabel : familyPageMap model.family dQ 2 p
 220       (familyPageMap model.family (XModLambdaN.incl _ q) 2 p x) =
 221       familyPageMap model.family d 2 p x := by
 222     rw [←ModuleCat.comp_apply,family_page_comp,hdQ]
 223   have hRLabel : familyPageMap model.family RQ 2 p
 224       (familyPageMap model.family (XModLambdaN.incl _ q) 2 p x) =
 225       familyPageMap model.family R 2 p x := by
 226     rw [←ModuleCat.comp_apply,family_page_comp,hRQ]
 227   have hcd := detects_map (.quotient q (.shift (0,a) (.nu Y)))
 228     (.shift (0,b) (.quotient r (.nu X))) dQ p _ alpha halpha
 229   have hcd := (congrArg (fun z => Detects
 230     (model.convergence (.shift (0,b) (.quotient r (.nu X)))) p z (alpha≫dQ)) hdLabel).mp hcd
 231   have hres := detects_map (.quotient q (.shift (0,a) (.nu Y)))
 232     (.shift (0,cWeight) (.quotient r (.nu Z))) RQ p _ alpha halpha
 233   have hres := (congrArg (fun z => Detects
 234     (model.convergence (.shift (0,cWeight) (.quotient r (.nu Z)))) p z (alpha≫RQ)) hRLabel).mp hres
 235   have hres0 := detects_zero_of_detects_same_label_zero
 236     (.shift (0,cWeight) (.quotient r (.nu Z))) p _ _ hres hRzero
 237   have hJ := detects_map (.quotient q (.shift (0,a) (.nu Y)))
 238     (.shift (0,cWeight) (.quotient r (.nu Z))) J p _ alpha halpha
 239   have htarget := detects_add (model.convergence (.shift (0,cWeight) (.quotient r (.nu Z))))
 240     p hJ (detects_neg hres0)
 241   refine ⟨dQ,v-alpha≫dQ,hdQ,rfl,hcd,?_,?_⟩
 242   · rw [Preadditive.sub_comp,hcdRho,sub_zero]
 243   · rw [hmath]
 244     simpa only [SyntheticObject.obj,neg_zero,add_zero,sub_eq_add_neg] using htarget
