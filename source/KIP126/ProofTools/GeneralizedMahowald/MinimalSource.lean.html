  18     (x : Ambient model X (s,t))
  19     (a : BiHom (t-s) (t+e) (XModLambdaN ((SyntheticCategory.biShift (0,e)).obj
  20       (model.nu.functor.obj (X.obj model.auxiliary))) q))
  21     (ha : Detects (model.convergence (.quotient q (.shift (0,e) (.nu X))))
  22       (s,t,t+e) (regradeE2 (by simp) (finiteNuLabel model X q e s t 0 x)) a) :
  23     Detects (model.convergence (.quotient q (.shift (0,e) (.nu X))))
  24       (s,(t-s)+s,t+e) (regradeE2 (by simp) (finiteNuLabel model X q e s t 0 x))
  25       (homotopyRegrade (by dsimp; omega) rfl a) := by

  33 private theorem extension_at_source_of_normalized_source
  34     (X Y : ClassicalObject) (f : X.obj model.auxiliary ⟶ Y.obj model.auxiliary)
  35     (r : ℕ) (n s t : ℤ) (x : Ambient model X (s,t)) (y : Ambient model Y (s+n,t+n))
  36     (W : FiniteExtensionWitness model X Y f r n s t x y)
  37     (aZ : BiHom (t-s) t (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) (r-1)))
  38     (ah : BiHom (t-s) (t+normalizedExponent model.classicalTower f)
  39       ((SyntheticObject.quotient (r-1) (extensionSourceObject model X Y f)).obj model.nu model.auxiliary))
  40     (hah : Detects (model.convergence (.quotient (r-1) (extensionSourceObject model X Y f)))
  41       (s,(t-s)+s,t+normalizedExponent model.classicalTower f)
  42       (finiteSourceForExtension model X Y f (r-1) s t x)
  43       (homotopyRegrade (by dsimp; omega) rfl ah))
  44     (hu : FiniteSourceUnshifts X Y f (r-1) s t aZ ah)
  45     (hnc : Leibniz.TowerNoCrossing (model.convergentMap
  46       (.quotient (r-1) (extensionSourceObject model X Y f)) (.quotient (r-1) (.nu Y))
  47       (finiteExtensionMap model X Y f (r-1)))
  48       (t-s,t+normalizedExponent model.classicalTower f) s n) :
  49     FiniteExtensionAtSource X Y f r n s t x y W aZ := by
  50   let p := (t-s,t+normalizedExponent model.classicalTower f)

  52     (model.convergence (.quotient (r-1) (extensionSourceObject model X Y f))) p s _ ah hah

  54   rw [hsame] at hgar
  55   obtain ⟨br,hab,hgbr⟩ := Leibniz.tower_target_of_any_source_frame
  56     (model.convergentMap (.quotient (r-1) (extensionSourceObject model X Y f))
  57       (.quotient (r-1) (.nu Y)) (finiteExtensionMap model X Y f (r-1)))
  58       p s n W.source W.target W.equation hnc ar hgar
  59   have ht := source_frame_detects (model.convergence (.quotient (r-1) (.nu Y)))
  60     p (s+n) W.target _ W.target_rep br hgbr
  61   rw [←hab,har] at ht
  62   exact ⟨ah,hah,ht,hu⟩

 167     (0,1) (t-s,t) (s+1) (aZ-a₀)).mpr hd
 168   have hshift' : FiltrationAtLeast model.syntheticTower (s+1)
 169       (homotopyRegrade (m' := t-s) (w' := t+1) (by simp) rfl
 170         (shiftHomotopyClass (0,1) (t-s,t) (aZ-a₀))) :=
 171     (Leibniz.regrade_filtration_iff _ _ _ _).mpr hshift

 174     (.quotient (r-1) (.shift (0,1) (.nu X))) ε.inv)
 175     (t-s,t+1) (s+1) _ hshift'

 182     (s,t,t+1) _ ah1 ahNew hsource1 hdiff

 290 /-- At the normalization length, every prescribed detecting source has the
 291 original finite extension's specified target. The normalized source is rebuilt
 292 from that same source and its exact inverse-shift relation is retained. -/
 293 private theorem finite_extension_at_source_minimal
 294     (X Y : ClassicalObject) (f : X.obj model.auxiliary ⟶ Y.obj model.auxiliary)
 295     (r : ℕ) (n s t : ℤ) (x : Ambient model X (s,t)) (y : Ambient model Y (s+n,t+n))
 296     (W : FiniteExtensionWitness model X Y f r n s t x y)
 297     (hn : n ≤ normalizedExponent model.classicalTower f)
 298     (aZ : BiHom (t-s) t (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) (r-1)))
 299     (haZ : Detects (model.convergence (.quotient (r-1) (.nu X))) (s,t,t)
 300       (regradeE2 (by simp) (finiteTargetLabel model X (r-1) s t 0 x)) aZ) :
