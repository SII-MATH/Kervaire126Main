  17 private def cancelBiShiftNatIso (β : ℤ×ℤ) (F G : SyntheticSpectrum ⥤ SyntheticSpectrum)
  18     (η : F ⋙ SyntheticCategory.biShift β ≅ G ⋙ SyntheticCategory.biShift β) : F ≅ G :=
  19   NatIso.ofComponents (fun A =>
  20     (SyntheticCategory.biShift_fullyFaithful β).preimageIso (η.app A)) (by
  21       intro A B f
  22       apply (SyntheticCategory.biShift_fullyFaithful β).map_injective
  23       simp only [Functor.map_comp,Functor.FullyFaithful.preimageIso_hom,
  24         Functor.FullyFaithful.map_preimage]
  25       exact η.hom.naturality f)
