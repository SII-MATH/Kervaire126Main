  19 private theorem quotient_restriction_maps_equal_on_spheres
  20     (X Y : ClassicalObject) (i q : ℕ) (hi : 0 < i) (hiq : i ≤ q)
  21     (f g : XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) q ⟶
  22       XModLambdaN (model.nu.functor.obj (Y.obj model.auxiliary)) i)
  23     (hfg : XModLambdaN.incl _ q ≫ f = XModLambdaN.incl _ q ≫ g) :
  24     ∀ (m w : ℤ) (c : BiHom m w
  25       (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) q)),
  26       c ≫ f = c ≫ g := by
  27   apply quotient_maps_equal_on_spheres_of_no_return q f g hfg
  28   intro m w c z
  29   let V := model.nu.functor.obj (X.obj model.auxiliary)
  30   let ε : ((SyntheticCategory.biShift (0,-(q:ℤ))).obj V)⟦(1:ℤ)⟧ ≅
  31       (SyntheticCategory.biShift (0,-(q:ℤ)-1)).obj
  32         (model.nu.functor.obj ((ClassicalObject.shift 1 X).obj model.auxiliary)) :=
  33     (SyntheticCategory.biShift_compat (Syn:=SyntheticSpectrum) 1).symm.app _ ≪≫
  34       (SyntheticCategory.biShift_comp (0,-(q:ℤ)) (1,0)).app V ≪≫
  35       eqToIso (congrArg (fun p => (SyntheticCategory.biShift p).obj V)
  36         (show (0,-(q:ℤ))+(1,0)=(1,1)+(0,-(q:ℤ)-1) by ext <;> simp)) ≪≫
  37       (SyntheticCategory.biShift_comp (1,1) (0,-(q:ℤ)-1)).symm.app V ≪≫
  38       (SyntheticCategory.biShift (0,-(q:ℤ)-1)).mapIso (model.nu.suspensionIso _).symm
  39   have h := nonreturn_at_window_end (.shift 1 X) Y i hi (-(q:ℤ)-1)
  40     (by omega) m w (c ≫ ε.hom) (ε.inv ≫ z)
  41   simpa only [Category.assoc,Iso.hom_inv_id_assoc] using h

  43 private theorem shift_preserves_sphere_congruence (A B : SyntheticSpectrum)
  44     (δ : ℤ × ℤ) (f g : A ⟶ B)
  45     (hfg : ∀ (m w : ℤ) (c : BiHom m w A), c ≫ f = c ≫ g)
  46     (m w : ℤ) (c : BiHom m w ((SyntheticCategory.biShift δ).obj A)) :
  47     c ≫ (SyntheticCategory.biShift δ).map f =
  48       c ≫ (SyntheticCategory.biShift δ).map g := by
  49   let F := SyntheticCategory.biShift (Syn:=SyntheticSpectrum) δ
  50   let ff := SyntheticCategory.biShift_fullyFaithful (Syn:=SyntheticSpectrum) δ
  51   let ε : F.obj (Smn (m-δ.1) (w-δ.2)) ≅ Smn m w :=
  52     (biShiftAddIso (m-δ.1,w-δ.2) δ (m,w) (by ext <;> simp)).app S_0_0
  53   let c₀ := ff.preimage (ε.hom ≫ c)
  54   have h := congrArg F.map (hfg (m-δ.1) (w-δ.2) c₀)
  55   have hc : F.map c₀ = ε.hom ≫ c := ff.map_preimage _
  56   rw [Functor.map_comp,Functor.map_comp,hc] at h
  57   apply (cancel_epi ε.hom).mp
  58   simpa only [Category.assoc] using h

  60 private theorem shifted_quotient_sphere_congruence
  61     (X Y : ClassicalObject) (δ : ℤ × ℤ)
  62     (i q : ℕ) (hi : 0 < i) (hiq : i ≤ q)
  63     (e : XModLambdaN ((SyntheticCategory.biShift δ).obj
  64         (model.nu.functor.obj (X.obj model.auxiliary))) q ≅
  65       (SyntheticCategory.biShift δ).obj
  66         (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) q))
  67     (he : XModLambdaN.incl _ q ≫ e.hom =
  68       (SyntheticCategory.biShift δ).map (XModLambdaN.incl _ q))
  69     (f g : XModLambdaN ((SyntheticCategory.biShift δ).obj
  70         (model.nu.functor.obj (X.obj model.auxiliary))) q ⟶
  71       (SyntheticCategory.biShift δ).obj
  72         (XModLambdaN (model.nu.functor.obj (Y.obj model.auxiliary)) i))
  73     (hfg : XModLambdaN.incl _ q ≫ f = XModLambdaN.incl _ q ≫ g)
  74     (m w : ℤ) (c : BiHom m w (XModLambdaN ((SyntheticCategory.biShift δ).obj
  75       (model.nu.functor.obj (X.obj model.auxiliary))) q)) : c ≫ f = c ≫ g := by
  76   let F := SyntheticCategory.biShift (Syn:=SyntheticSpectrum) δ
  77   let ff := SyntheticCategory.biShift_fullyFaithful (Syn:=SyntheticSpectrum) δ
  78   let f₀ := ff.preimage (e.inv ≫ f)
  79   let g₀ := ff.preimage (e.inv ≫ g)
  80   have hf : F.map f₀ = e.inv ≫ f := ff.map_preimage _
  81   have hg : F.map g₀ = e.inv ≫ g := ff.map_preimage _
  82   have h₀ : XModLambdaN.incl _ q ≫ f₀ = XModLambdaN.incl _ q ≫ g₀ := by
  83     apply ff.map_injective
  84     rw [Functor.map_comp,Functor.map_comp,hf,hg,←he]
  85     simpa only [Category.assoc,Iso.hom_inv_id_assoc] using hfg
  86   have hbase := quotient_restriction_maps_equal_on_spheres X Y i q hi hiq f₀ g₀ h₀
  87   have h := shift_preserves_sphere_congruence _ _ δ f₀ g₀ hbase m w (c ≫ e.hom)
  88   rw [hf,hg] at h
  89   simpa only [Category.assoc,Iso.hom_inv_id_assoc] using h

  91 private theorem quotient_shift_restriction_on_spheres
  92     (X : ClassicalObject) (δ : ℤ × ℤ)
  93     (i q : ℕ) (hi : 0 < i) (hiq : i ≤ q)
  94     (eᵢ : XModLambdaN ((SyntheticCategory.biShift δ).obj
  95         (model.nu.functor.obj (X.obj model.auxiliary))) i ≅
  96       (SyntheticCategory.biShift δ).obj
  97         (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) i))
  98     (e_q : XModLambdaN ((SyntheticCategory.biShift δ).obj
  99         (model.nu.functor.obj (X.obj model.auxiliary))) q ≅
 100       (SyntheticCategory.biShift δ).obj
 101         (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) q))
 102     (hᵢ : XModLambdaN.incl _ i ≫ eᵢ.hom =
 103       (SyntheticCategory.biShift δ).map (XModLambdaN.incl _ i))
 104     (h_q : XModLambdaN.incl _ q ≫ e_q.hom =
 105       (SyntheticCategory.biShift δ).map (XModLambdaN.incl _ q))
 106     (m w : ℤ) (c : BiHom m w (XModLambdaN ((SyntheticCategory.biShift δ).obj
 107       (model.nu.functor.obj (X.obj model.auxiliary))) q)) :
 108     c ≫ (model.quotients.tower _).rho i q hiq ≫ eᵢ.hom =
 109       c ≫ e_q.hom ≫ (SyntheticCategory.biShift δ).map
 110         ((model.quotients.tower _).rho i q hiq) := by
 111   apply shifted_quotient_sphere_congruence X X δ i q hi hiq e_q h_q
 112   rw [←Category.assoc,(model.quotients.tower _).rho_quotient hiq,hᵢ,
 113     ←Category.assoc,h_q,←Functor.map_comp,(model.quotients.tower _).rho_quotient hiq]

 115 private def quotientShiftDesuspend (X : ClassicalObject) (δ : ℤ × ℤ) (q : ℕ)
 116     (e : XModLambdaN ((SyntheticCategory.biShift δ).obj
 117         (model.nu.functor.obj (X.obj model.auxiliary))) q ≅
 118       (SyntheticCategory.biShift δ).obj
 119         (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) q))
 120     (m w : ℤ) (c : BiHom m w (XModLambdaN ((SyntheticCategory.biShift δ).obj
 121       (model.nu.functor.obj (X.obj model.auxiliary))) q)) :
 122     BiHom (m-δ.1) (w-δ.2)
 123       (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) q) :=
 124   (SyntheticCategory.biShift_fullyFaithful δ).preimage
 125     ((biShiftAddIso (m-δ.1,w-δ.2) δ (m,w) (by ext <;> simp)).hom.app S_0_0 ≫
 126       c ≫ e.hom)

 128 private theorem quotient_shift_desuspend_restriction
 129     (X : ClassicalObject) (δ : ℤ × ℤ)
 130     (i q : ℕ) (hi : 0 < i) (hiq : i ≤ q)
 131     (eᵢ : XModLambdaN ((SyntheticCategory.biShift δ).obj
 132         (model.nu.functor.obj (X.obj model.auxiliary))) i ≅
 133       (SyntheticCategory.biShift δ).obj
 134         (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) i))
 135     (e_q : XModLambdaN ((SyntheticCategory.biShift δ).obj
 136         (model.nu.functor.obj (X.obj model.auxiliary))) q ≅
 137       (SyntheticCategory.biShift δ).obj
 138         (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) q))
 139     (hᵢ : XModLambdaN.incl _ i ≫ eᵢ.hom =
 140       (SyntheticCategory.biShift δ).map (XModLambdaN.incl _ i))
 141     (h_q : XModLambdaN.incl _ q ≫ e_q.hom =
 142       (SyntheticCategory.biShift δ).map (XModLambdaN.incl _ q))
 143     (m w : ℤ)
 144     (a : BiHom m w (XModLambdaN ((SyntheticCategory.biShift δ).obj
 145       (model.nu.functor.obj (X.obj model.auxiliary))) i))
 146     (c : BiHom m w (XModLambdaN ((SyntheticCategory.biShift δ).obj
 147       (model.nu.functor.obj (X.obj model.auxiliary))) q))
 148     (hc : c ≫ (model.quotients.tower _).rho i q hiq = a) :
 149     quotientShiftDesuspend X δ q e_q m w c ≫ (model.quotients.tower _).rho i q hiq =
 150       quotientShiftDesuspend X δ i eᵢ m w a := by
 151   apply (SyntheticCategory.biShift_fullyFaithful δ).map_injective
 152   rw [Functor.map_comp]
 153   simp only [quotientShiftDesuspend,Functor.FullyFaithful.map_preimage,Category.assoc]
 154   have h := quotient_shift_restriction_on_spheres X δ i q hi hiq eᵢ e_q hᵢ h_q m w c
 155   rw [←Category.assoc,hc] at h
 156   rw [←h]
