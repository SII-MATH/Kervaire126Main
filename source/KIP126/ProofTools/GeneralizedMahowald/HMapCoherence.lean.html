  15 private theorem shift_negativeLift_cancel (k : ℕ) {A B : SyntheticSpectrum}
  16     (f : (SyntheticCategory.biShift (0,(k:ℤ))).obj A ⟶ B) :
  17     (SyntheticCategory.biShift (0,(k:ℤ))).map (negativeLift k f) ≫
  18       (biShiftAddIso (0,-(k:ℤ)) (0,(k:ℤ)) 0 (by ext <;> simp)).hom.app B ≫
  19         SyntheticCategory.biShift_zero.hom.app B = f := by
  20   let h : ℤ×ℤ := (0,(k:ℤ))
  21   let nh : ℤ×ℤ := (0,-(k:ℤ))
  22   let e : (SyntheticCategory.biShift nh).obj ((SyntheticCategory.biShift h).obj A) ≅ A :=
  23     (biShiftAddIso h nh 0 (by ext <;> simp [h,nh])).app A ≪≫
  24       SyntheticCategory.biShift_zero.app A
  25   have hs : e.hom ≫ negativeLift k f = (SyntheticCategory.biShift nh).map f := by
  26     simp only [e,h,nh,biShiftAddIso,Iso.trans_hom,Iso.app_hom,NatTrans.comp_app,
  27       eqToIso.hom,eqToHom_app,negativeLift,Category.assoc,
  28       Iso.hom_inv_id_app_assoc,eqToHom_trans_assoc]
  29     simp
  30   have hu : (SyntheticCategory.biShift h).map e.hom =
  31       (biShiftAddIso nh h 0 (by ext <;> simp [h,nh])).hom.app
  32         ((SyntheticCategory.biShift h).obj A) ≫
  33       SyntheticCategory.biShift_zero.hom.app ((SyntheticCategory.biShift h).obj A) := by
  34     have H := model.quotients.coherence.associativity h nh h 0 0 h
  35       (by ext <;> simp [h,nh]) (by ext <;> simp [h,nh])
  36       (zero_add h) (add_zero h) A
  37     rw [model.quotients.coherence.left_unit,model.quotients.coherence.right_unit] at H
  38     simpa only [e,Iso.trans_hom,Iso.app_hom,Functor.map_comp] using H
  39   apply (cancel_epi ((SyntheticCategory.biShift h).mapIso e).hom).mp
  40   change (SyntheticCategory.biShift h).map e.hom ≫ _ = _
  41   rw [←Category.assoc,←Functor.map_comp,hs]
  42   erw [(biShiftAddIso nh h 0 (by ext <;> simp [h,nh])).hom.naturality_assoc]
  43   erw [SyntheticCategory.biShift_zero.hom.naturality]
  44   change _ = (SyntheticCategory.biShift h).map e.hom ≫ f
  45   rw [hu]
  46   simp only [Category.assoc,Functor.id_map]
  47   rfl

  49 private theorem shift_negativeLift_target (k : ℕ) {A B X : SyntheticSpectrum}
  50     (σ δ : ℤ×ℤ) (hσ : σ+(0,-(k:ℤ))=δ) (hδ : δ+(0,(k:ℤ))=σ)
  51     (f : (SyntheticCategory.biShift (0,(k:ℤ))).obj A ⟶ B)
  52     (u : B ⟶ (SyntheticCategory.biShift σ).obj X) :
  53     (SyntheticCategory.biShift (0,(k:ℤ))).map
  54       (negativeLift k f ≫ (SyntheticCategory.biShift (0,-(k:ℤ))).map u ≫
  55         (biShiftAddIso σ (0,-(k:ℤ)) δ hσ).hom.app X) ≫
  56       (biShiftAddIso δ (0,(k:ℤ)) σ hδ).hom.app X = f≫u := by
  57   let h : ℤ×ℤ := (0,(k:ℤ))
  58   let nh : ℤ×ℤ := (0,-(k:ℤ))
  59   have H := model.quotients.coherence.associativity σ nh h δ 0 σ
  60     hσ (by ext <;> simp [h,nh]) hδ (add_zero σ) X
  61   rw [model.quotients.coherence.right_unit] at H
  62   simp only [Functor.map_comp,Category.assoc]
  63   rw [H]
  64   erw [(biShiftAddIso nh h 0 (by ext <;> simp [h,nh])).hom.naturality_assoc]
  65   erw [SyntheticCategory.biShift_zero.hom.naturality]
  66   simpa only [Category.assoc,Functor.id_map] using
  67     congrArg (fun z => z≫u) (shift_negativeLift_cancel k f)

  69 private theorem normalized_g_cofiber_inclusion_shifted (T : TriangleData model.auxiliary)
  70     (he : (normalizedExponent model.classicalTower T.f : ℤ) +
  71       normalizedExponent model.classicalTower T.g + normalizedExponent model.classicalTower T.h = 1)
  72     (hT : NormalizedTriangleCompatible model T he) :
  73     (SyntheticCategory.biShift (0,(normalizedExponent model.classicalTower T.h : ℤ))).map
  74       (HasFunctorialCofiber.cofibι (model.normalizedMap T.Y T.Z T.g).map ≫
  75         (normalized_g_cofiber_iso T he hT).hom) ≫
  76       (biShiftAddIso
  77         (1,(normalizedExponent model.classicalTower T.f : ℤ) + normalizedExponent model.classicalTower T.g)
  78         (0,(normalizedExponent model.classicalTower T.h : ℤ)) (1,1)
  79         (by ext <;> simp only [Prod.fst_add,Prod.snd_add] <;> omega)).hom.app _ =
  80       (model.normalizedMap T.Z (.shift 1 T.X) T.h).map ≫
  81         (model.nu.suspensionIso (T.X.obj model.auxiliary)).hom := by
  82   rw [normalized_g_cofiber_inclusion]
  83   exact shift_negativeLift_target _ _ _ _ _ _ _
