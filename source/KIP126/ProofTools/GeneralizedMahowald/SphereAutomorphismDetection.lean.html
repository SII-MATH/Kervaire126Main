  18 private theorem sphere_precomp_preserves_filtration
  19     {A : SyntheticSpectrum} (m w s : ℤ) (u : Smn (Syn:=SyntheticSpectrum) m w ⟶ Smn m w)
  20     (a : BiHom m w A) (ha : FiltrationAtLeast model.syntheticTower s a) :
  21     FiltrationAtLeast model.syntheticTower s (u≫a) := by
  22   obtain ⟨b,hb⟩ := ha
  23   refine ⟨u≫b,?_⟩
  24   change (u≫b)≫(model.syntheticTower.tower A).projection s=u≫a
  25   change b≫(model.syntheticTower.tower A).projection s=a at hb
  26   rw [Category.assoc,hb]

  28 private theorem sphere_automorphism_first_finite
  29     (X : ClassicalObject) (m w : ℤ) (u : Smn (Syn:=SyntheticSpectrum) m w ≅ Smn m w)
  30     (a : BiHom m w (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) 1)) :
  31     u.hom≫a=a := by
  32   let A := model.nu.functor.obj (X.obj model.auxiliary)
  33   let Q := XModLambdaN A 1
  34   have ha : FiltrationAtLeast model.syntheticTower (w-m) a := by
  35     have h := finite_bottom_filtration X 1 (by omega) (w-m) w
  36     rw [show w-(w-m)=m by omega] at h
  37     exact h a
  38   have hs : FiltrationAtLeast model.syntheticTower (w-m)
  39       (shiftHomotopyClass (0,-1) (m,w) a) :=
  40     (shift_filtration_iff (.quotient 1 (.nu X)) (0,-1) (m,w) (w-m) a).mpr ha
  41   have hl : FiltrationAtLeast model.syntheticTower (w-m)
  42       (shiftHomotopyClass (0,-1) (m,w) a ≫ lambdaPow 1 Q) :=
  43     map_preserves_filtration
  44     (model.convergentMap (.shift (0,-1) (.quotient 1 (.nu X))) (.quotient 1 (.nu X))
  45       (lambdaPow 1 Q)) ((m,w)+(0,-1)) (w-m) (shiftHomotopyClass (0,-1) (m,w) a) hs
  46   have hz : shiftHomotopyClass (0,-1) (m,w) a≫lambdaPow 1 Q=0 :=
  47     finite_high_filtration_zero X 1 (by omega) ((m,w)+(0,-1)).1 ((m,w)+(0,-1)).2 (w-m)
  48       (by dsimp; omega) _ hl
  49   have hzero : lambdaPow 1 (Smn m w)≫a=0 := by
  50     apply (cancel_epi ((SyntheticCategory.biShift_comp (m,w) (0,-1)).inv.app S_0_0)).mp
  51     rw [comp_zero]
  52     erw [←lambdaPow_naturality]
  53     change ((SyntheticCategory.biShift_comp (m,w) (0,-1)).inv.app S_0_0) ≫
  54       (SyntheticCategory.biShift (0,-1)).map a ≫ lambdaPow 1 Q=0
  55     simpa only [shiftHomotopyClass,Category.assoc] using hz
  56   obtain ⟨b,hb⟩ := Triangle.yoneda_exact₂ (XModLambdaN.cofiberTriangle (Smn m w) 1)
  57     (XModLambdaN.cofiberTriangle_distinguished (Smn m w) 1) a hzero
  58   change a=XModLambdaN.incl (Smn m w) 1≫b at hb
  59   rw [hb,←Category.assoc,bigraded_sphere_automorphism_first_quotient]

  61 private theorem finite_lambda_precomp_sphere
  62     (X : ClassicalObject) (q d : ℕ) (hd : 0<d) (hdq : d<q) (s t : ℤ)
  63     (u : Smn (Syn:=SyntheticSpectrum) (t-s) (t-d) ≅ Smn (t-s) (t-d))
  64     (b : BiHom (t-s) t (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) (q-d))) :
  65     ∃ v : Smn (Syn:=SyntheticSpectrum) (t-s) t ≅ Smn (t-s) t,
  66       u.hom ≫ finiteLambdaClass X q d hd hdq s t b =
  67         finiteLambdaClass X q d hd hdq s t (v.hom≫b) := by
  68   let C := (biShiftAddIso (t-s,t) (0,-(d:ℤ)) (t-s,t-d) (by ext <;> simp [sub_eq_add_neg])).app
  69     (S_0_0 : SyntheticSpectrum)
  70   let ff := SyntheticCategory.biShift_fullyFaithful (Syn:=SyntheticSpectrum) (0,-(d:ℤ))
  71   let v : Smn (t-s) t ≅ Smn (t-s) t := ff.preimageIso (C ≪≫ u ≪≫ C.symm)
  72   refine ⟨v,?_⟩
  73   unfold finiteLambdaClass
  74   rw [shift_regrade,shift_regrade]
  75   simp only [Category.assoc]
  76   change u.hom ≫ (C.inv ≫ (SyntheticCategory.biShift (0,-(d:ℤ))).map b ≫ _) =
  77     C.inv ≫ (SyntheticCategory.biShift (0,-(d:ℤ))).map (v.hom≫b) ≫ _
  78   have hv : (SyntheticCategory.biShift (0,-(d:ℤ))).map v.hom = C.hom≫u.hom≫C.inv :=
  79     ff.map_preimage _
  80   simp only [Functor.map_comp,hv,Category.assoc,Iso.inv_hom_id_assoc]

  82 private theorem sphere_automorphism_finite_positive_detects
  83     (X : ClassicalObject) (q k : ℕ) (hk : 0<k) (hkq : k<q) (s t : ℤ)
  84     (x : Ambient model X (s,t))
  85     (u : Smn (Syn:=SyntheticSpectrum) (t-s) (t-k) ≅ Smn (t-s) (t-k))
  86     (a : BiHom (t-s) (t-k) (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) q))
  87     (ha : Detects (model.convergence (.quotient q (.nu X))) (s,t,t-k)
  88       (finiteTargetLabel model X q s t k x) a) :
  89     Detects (model.convergence (.quotient q (.nu X))) (s,t,t-k)
  90       (finiteTargetLabel model X q s t k x) (u.hom≫a) := by
  91   obtain ⟨b,hb⟩ := finite_detected_target_bottom_lift X q k hkq s t x a ha
  92   obtain ⟨v,hv⟩ := finite_lambda_precomp_sphere X q k hk hkq s t u b
  93   have hb' : (v.hom≫b) ≫ (model.quotients.tower _).rho 1 (q-k) (by omega) =
  94       firstLabel model X s t x ≫ XModLambdaN.map (SyntheticCategory.biShift_zero.hom.app _) 1 := by
  95     rw [Category.assoc,sphere_automorphism_first_finite,hb]
  96   have hgood := finite_quotient_lambda_detects X q k hk hkq s t x (v.hom≫b) hb'
  97   rw [←hv] at hgood
  98   have hbase := finite_quotient_lambda_detects X q k hk hkq s t x b hb
  99   have hdiff := detects_sub_filtration (model.convergence (.quotient q (.nu X))) (s,t,t-k) ha hbase
 100   have hu := sphere_precomp_preserves_filtration (t-s) (t-k) (s+1) u.hom _ hdiff
 101   rw [Preadditive.comp_sub] at hu
 102   exact detects_of_high_difference (.quotient q (.nu X)) (s,t,t-k) _ _ _ hgood hu

 104 private theorem sphere_automorphism_finite_bottom_detects
 105     (X : ClassicalObject) (q : ℕ) (hq : 0<q) (s t : ℤ)
 106     (x : (model.family.obj ((SyntheticObject.quotient q (.nu X)).obj model.nu model.auxiliary)).E₂ (s,t,t))
 107     (u : Smn (Syn:=SyntheticSpectrum) (t-s) t ≅ Smn (t-s) t)
 108     (a : BiHom (t-s) t (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) q))
 109     (ha : Detects (model.convergence (.quotient q (.nu X))) (s,t,t) x a) :

 120 private theorem sphere_automorphism_finite_detects
 121     (X : ClassicalObject) (q k : ℕ) (hkq : k<q) (s t : ℤ)
 122     (x : Ambient model X (s,t))
 123     (u : Smn (Syn:=SyntheticSpectrum) (t-s) (t-k) ≅ Smn (t-s) (t-k))
 124     (a : BiHom (t-s) (t-k) (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) q))
 125     (ha : Detects (model.convergence (.quotient q (.nu X))) (s,t,t-k)
 126       (finiteTargetLabel model X q s t k x) a) :
 127     Detects (model.convergence (.quotient q (.nu X))) (s,t,t-k)
 128       (finiteTargetLabel model X q s t k x) (u.hom≫a) := by
 129   by_cases hk : k=0
 130   · subst k
 131     have bottom (w : ℤ) (hw : w=t)
 132         (z : (model.family.obj ((SyntheticObject.quotient q (.nu X)).obj model.nu model.auxiliary)).E₂ (s,t,w))
 133         (v : Smn (Syn:=SyntheticSpectrum) (t-s) w ≅ Smn (t-s) w)
 134         (b : BiHom (t-s) w (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) q))
 135         (hb : Detects (model.convergence (.quotient q (.nu X))) (s,t,w) z b) :
 136         Detects (model.convergence (.quotient q (.nu X))) (s,t,w) z (v.hom≫b) := by
 137       subst w
 138       exact sphere_automorphism_finite_bottom_detects X q hkq s t z v b hb
 139     exact bottom _ (by simp) _ u a ha
 140   · exact sphere_automorphism_finite_positive_detects X q k (by omega) hkq s t x u a ha

 142 private theorem quotient_unshift_natural_automorphism_detects
 143     (X : ClassicalObject) (δ : ℤ×ℤ) (q k : ℕ) (hkq : k<q) (s t : ℤ)
 144     (x : Ambient model X (s,t))
 145     (η : SyntheticCategory.biShift (Syn:=SyntheticSpectrum) δ ≅ SyntheticCategory.biShift δ)
 146     (ε : XModLambdaN ((SyntheticCategory.biShift δ).obj
 147         (model.nu.functor.obj (X.obj model.auxiliary))) q ≅
 148       (SyntheticCategory.biShift δ).obj (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) q))
 149     (hε : XModLambdaN.incl _ q≫ε.hom=(SyntheticCategory.biShift δ).map (XModLambdaN.incl _ q))
 150     (a : BiHom ((t-s,t-k)+δ).1 ((t-s,t-k)+δ).2
 151       (XModLambdaN ((SyntheticCategory.biShift δ).obj
 152         (model.nu.functor.obj (X.obj model.auxiliary))) q))
 153     (ha : Detects (model.convergence (.quotient q (.nu X))) (s,t,t-k)
 154       (finiteTargetLabel model X q s t k x) (quotientUnshift X δ (t-s,t-k) q ε a)) :
 155     Detects (model.convergence (.quotient q (.nu X))) (s,t,t-k)
 156       (finiteTargetLabel model X q s t k x)
 157       (quotientUnshift X δ (t-s,t-k) q ε (a≫XModLambdaN.map (η.hom.app _) q)) := by
 158   let A := model.nu.functor.obj (X.obj model.auxiliary)
 159   let Q := XModLambdaN A q
 160   let F := SyntheticCategory.biShift (Syn:=SyntheticSpectrum) δ
 161   let ff := SyntheticCategory.biShift_fullyFaithful (Syn:=SyntheticSpectrum) δ
 162   let M : Q⟶Q := ff.preimage (ε.inv≫XModLambdaN.map (η.hom.app A) q≫ε.hom)
 163   let N : Q⟶Q := ff.preimage (η.hom.app Q)
 164   have hM : F.map M=ε.inv≫XModLambdaN.map (η.hom.app A) q≫ε.hom := ff.map_preimage _
 165   have hN : F.map N=η.hom.app Q := ff.map_preimage _
 166   dsimp only [F] at hM hN
 167   have hbase : XModLambdaN.incl A q≫M=XModLambdaN.incl A q≫N := by
 168     apply ff.map_injective
 169     simp only [Functor.map_comp,hM,hN]
 170     rw [←hε,Category.assoc,Iso.hom_inv_id_assoc]
 171     rw [←Category.assoc,←XModLambdaN.incl_naturality,Category.assoc,hε]
 172     exact (η.hom.naturality (XModLambdaN.incl A q)).symm
 173   let b := quotientUnshift X δ (t-s,t-k) q ε a
 174   have he : quotientUnshift X δ (t-s,t-k) q ε (a≫XModLambdaN.map (η.hom.app A) q)=b≫M := by
 175     apply ff.map_injective
 176     simp only [quotientUnshift,b,Functor.map_comp,Functor.FullyFaithful.map_preimage,hM,Category.assoc,
 177       Iso.hom_inv_id_assoc]
 178   have hm := quotient_nu_maps_equal_on_spheres X X q (by omega) M N hbase (t-s) (t-k) b
 179   let u : Smn (Syn:=SyntheticSpectrum) (t-s) (t-k) ≅ Smn (t-s) (t-k) :=
 180     ff.preimageIso (η.app _)
 181   have hn : b≫N=u.hom≫b := by
 182     apply ff.map_injective
 183     simp only [Functor.map_comp,hN,u,Functor.FullyFaithful.preimageIso_hom,
 184       Functor.FullyFaithful.map_preimage]
 185     exact η.hom.naturality b
 186   rw [he,hm,hn]
 187   exact sphere_automorphism_finite_detects X q k hkq s t x u b ha
