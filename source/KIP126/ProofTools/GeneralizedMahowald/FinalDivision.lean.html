  47 private theorem finite_weighted_lambda_detects (X : ClassicalObject)
  48     (q k : ℕ) (hk : 0<k) (hkq : k<q) (s t w : ℤ)

  52     (hb : Detects (model.convergence (.quotient (q-k) (.nu X))) (s,t,w)
  53       ((model.family.quotientProjection (model.nu.functor.obj (X.obj model.auxiliary)) (q-k)).pageMap 2 (s,t,w)
  54         (nuWeightLabel model X s t w (by omega) y)) b) :
  55     Detects (model.convergence (.quotient q (.nu X))) (s,t,w-k)
  56       ((model.family.quotientProjection (model.nu.functor.obj (X.obj model.auxiliary)) q).pageMap 2 (s,t,w-k)

  62     ((model.quotients.tower A).triangle hk hkq).lambdaMap

  66     (0,-(k:ℤ)) (t-s,w) s L _ b hbStem

  68       (s,t,w-k) := by ext <;> simp [sub_eq_add_neg]

  70   obtain ⟨e,he,br,hbr,hge⟩ := hm
  71   obtain ⟨e0,he0,_⟩ := hb

  75   have heq := he.unique heMap
  76   have heq' : e = KIP126.Literature.Standard.infinityFiniteLambda X q k hkq (s,t) w e0 := by

  81       eqToIso.hom, KIP126.Literature.Standard.infinityShiftData,
  82       KIP126.Literature.Standard.infinityShift_comparison]
  83     rw [infinity_map_regrade L hp]
  84     rfl
  85   have hlabel : HasInfinityRepresentative _ 2 (s,t,w-k)
  86       ((model.family.quotientProjection (model.nu.functor.obj (X.obj model.auxiliary)) q).pageMap 2 (s,t,w-k)
  87         (nuWeightLabel model X s t (w-k) (by omega) y))
  88       (KIP126.Literature.Standard.infinityFiniteLambda X q k hkq (s,t) w e0) := by
  89     obtain ⟨hy,hye⟩ := (KIP126.Literature.Standard.eInfinity_finite_labels X (q-k) (by omega)
  90       s t w hw y e0).mp he0
  91     apply (KIP126.Literature.Standard.eInfinity_finite_labels X q (by omega)
  92       s t (w-k) (by omega) y _).mpr
  93     have hy' : y ∈ cycles model X (q-t+(w-k)) (s,t) := by
  94       convert hy using 1
  95       congr 1
  96       omega
  97     refine ⟨hy',?_⟩
  98     rw [KIP126.Literature.Standard.eInfinity_lambda_finite X q k hkq (s,t) w hw,hye]
  99     rfl
 100   refine ⟨e,?_,br,?_,hge⟩
 101   · rw [heq']
 102     exact hlabel
 103   · convert hbr using 1
 104     simp only [finiteWeightedLambdaClass,regrade_precomp,Category.assoc]

 184 /-- The final target and its precise boundary ambiguity, once an actual divided
 185 source with the required target filtration has been constructed. The two
 186 substantive inputs are kept separate: a strict finite f-image and detection of
 187 its actual long-quotient lambda image. No arbitrary May preimage is selected. -/
 188 private theorem final_f_division_target (X Y : ClassicalObject)
 189     (f : X.obj model.auxiliary ⟶ Y.obj model.auxiliary)
 190     (q k : ℕ) (hkq : k<q) (n s t : ℤ)
 191     (hn : (normalizedExponent model.classicalTower f : ℤ)≤n)
 192     (hw : n-normalizedExponent model.classicalTower f<(q-k:ℕ))
 193     (x : Ambient model X (s,t)) (y : Ambient model Y (s+n,t+n))
 194     (af : BiHom (t-s) (t+normalizedExponent model.classicalTower f)
 195       ((SyntheticObject.quotient (q-k) (extensionSourceObject model X Y f)).obj model.nu model.auxiliary))
 196     (haf : Detects (model.convergence (.quotient (q-k) (extensionSourceObject model X Y f)))
 197       (s,(t-s)+s,t+normalizedExponent model.classicalTower f)
 198       (finiteSourceForExtension model X Y f (q-k) s t x)
 199       (homotopyRegrade (by dsimp; omega) rfl af))
 200     (hfil : FiltrationAtLeast model.syntheticTower (s+n)
 201       (af≫finiteExtensionMap model X Y f (q-k)))
 202     (hMay : Detects (model.convergence (.quotient q (.nu Y)))
 203       (s+n,t+n,t+normalizedExponent model.classicalTower f-k)
 204       ((model.family.quotientProjection (model.nu.functor.obj (Y.obj model.auxiliary)) q).pageMap 2
 205         (s+n,t+n,t+normalizedExponent model.classicalTower f-k)
 206         (nuWeightLabel model Y (s+n) (t+n)
 207           (t+normalizedExponent model.classicalTower f-k) (by omega) y))
 208       (finiteWeightedLambdaDivisionClass Y q k hkq (s+n) (t+n)
 209         (t+normalizedExponent model.classicalTower f)
 210         (homotopyRegrade (by omega) rfl (af≫finiteExtensionMap model X Y f (q-k))))) :
 211     ∃ y' : Ambient model Y (s+n,t+n),
 212       y'-y∈boundaries model Y (1+n-normalizedExponent model.classicalTower f+k) (s+n,t+n) ∧
 213       ∃ (a : BiHom (t-s) (t+normalizedExponent model.classicalTower f)
 214           ((SyntheticObject.quotient (q-k) (extensionSourceObject model X Y f)).obj model.nu model.auxiliary))
 215         (b : BiHom (t-s) (t+normalizedExponent model.classicalTower f)
 216           (XModLambdaN (model.nu.functor.obj (Y.obj model.auxiliary)) (q-k))),
 217         Detects (model.convergence (.quotient (q-k) (extensionSourceObject model X Y f)))
 218           (s,(t-s)+s,t+normalizedExponent model.classicalTower f)
 219           (finiteSourceForExtension model X Y f (q-k) s t x)
 220           (homotopyRegrade (by dsimp; omega) rfl a) ∧
 221         Detects (model.convergence (.quotient (q-k) (.nu Y)))
 222           (s+n,(t-s)+(s+n),t+normalizedExponent model.classicalTower f)
 223           (finiteTargetForExtension model X Y f (q-k) n s t hn y')
 224           (homotopyRegrade (by dsimp; omega) rfl b) ∧
 225         a≫finiteExtensionMap model X Y f (q-k)=b := by
 226   let b := af≫finiteExtensionMap model X Y f (q-k)
 227   let b' := homotopyRegrade (by omega : t-s=(t+n)-(s+n)) rfl b
 228   have hfil' : FiltrationAtLeast model.syntheticTower (s+n) b' := by
 229     exact (regrade_filtration_iff _ _ _ _).mpr hfil
 230   obtain ⟨y',hy',hcong⟩ := finite_lambda_target_label_congruent_uniform Y q k hkq
 231     (s+n) (t+n) (t+normalizedExponent model.classicalTower f) (by omega)
 232     y b' hfil' hMay
 233   refine ⟨y',?_,af,b,haf,?_,rfl⟩
 234   · convert hcong using 1
 235     congr 1
 236     omega
 237   · have hp : (s+n,(t-s)+(s+n),t+normalizedExponent model.classicalTower f)=
 238         (s+n,t+n,t+normalizedExponent model.classicalTower f) := by
 239       apply Prod.ext
 240       · rfl
 241       · apply Prod.ext
 242         · dsimp
 243           omega
 244         · rfl
 245     have ht := detects_transport hp.symm _ _ hy'
 246     have hl := finite_extension_target_weight_label X Y f (q-k) n s t hn y'
 247     have hregrade {E : SyntheticAdamsSS} {p p' : Tridegree} (h : p=p') (z : E.E₂ p) :
 248         regradeE2 h.symm (regradeE2 h z)=z := by subst p'; rfl
 249     rw [←hl,hregrade] at ht
 250     convert ht using 1
 251     dsimp only [b']
 252     simp only [regrade_precomp,←Category.assoc,eqToHom_trans]

 306 private theorem finite_detected_source_division_exact (X : ClassicalObject)
 307     (q k : ℕ) (hk : 0<k) (hkq : k<q) (s t : ℤ)
 308     (x : Ambient model X (s,t))

 321   have hb0' : finiteLambdaClass X q k hk hkq s t b0=a :=

 343 private theorem finite_detected_source_division_exact_uniform (X : ClassicalObject)
 344     (q k : ℕ) (hkq : k<q) (s t : ℤ)
 345     (x : Ambient model X (s,t))
 346     (a : BiHom (t-s) (t-k) (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) q))
 347     (ha : Detects (model.convergence (.quotient q (.nu X))) (s,t,t-k)
 348       ((model.family.quotientProjection (model.nu.functor.obj (X.obj model.auxiliary)) q).pageMap 2 (s,t,t-k)
 349         (nuWeightLabel model X s t (t-k) (by omega) x)) a) :
 350     ∃ b : BiHom (t-s) t (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) (q-k)),
 351       b≫(model.quotients.tower _).rho 1 (q-k) (by omega)=
 352         firstLabel model X s t x≫XModLambdaN.map (SyntheticCategory.biShift_zero.hom.app _) 1 ∧
 353       Detects (model.convergence (.quotient (q-k) (.nu X))) (s,t,t)
 354         (regradeE2 (by simp) (finiteTargetLabel model X (q-k) s t 0 x)) b ∧
 355       finiteLambdaDivisionClass X q k hkq s t b=a := by
 356   by_cases hk : k=0
 357   · subst k
 358     let b := homotopyRegrade rfl (by simp : t-(0:ℕ)=t) a
 359     have hp : (s,t,t-(0:ℕ))=(s,t,t) := by simp
 360     have ht := detects_transport hp _ _ ha
 361     have hbdet : Detects (model.convergence (.quotient q (.nu X))) (s,t,t)
 362         ((model.family.quotientProjection (model.nu.functor.obj (X.obj model.auxiliary)) q).pageMap 2 (s,t,t)
 363           (nuWeightLabel model X s t t (by omega) x)) b := by
 364       convert ht using 1
 365       · have transport (v v' : ℤ) (hv : v=v') (hv₁ : v≤t) (hv₂ : v'≤t)
 366             (h : (s,t,v)=(s,t,v')) :
 367             regradeE2 h
 368               ((model.family.quotientProjection (model.nu.functor.obj (X.obj model.auxiliary)) q).pageMap 2 (s,t,v)
 369                 (nuWeightLabel model X s t v hv₁ x)) =
 370               ((model.family.quotientProjection (model.nu.functor.obj (X.obj model.auxiliary)) q).pageMap 2 (s,t,v')
 371                 (nuWeightLabel model X s t v' hv₂ x)) := by subst v'; rfl
 372         exact (transport _ t (by simp) _ _ hp).symm
 373     have hbfirst := finite_detected_bottom_first X q hkq s t x b hbdet
 374     refine ⟨b,hbfirst,first_label_detects_unshifted X q hkq s t x b hbfirst,?_⟩
 375     simp only [finiteLambdaDivisionClass,↓reduceDIte,b]
 376     have cancel (A : SyntheticSpectrum) (m v v' : ℤ) (hv : v=v') (hv' : v'=v)
 377         (c : BiHom m v A) :
 378         homotopyRegrade rfl hv' (homotopyRegrade rfl hv c)=c := by subst v'; rfl
 379     exact cancel _ _ _ _ _ _ a

 381     refine ⟨b,hb,hbd,?_⟩
 382     simpa only [finiteLambdaDivisionClass,dif_neg hk] using hbl
