  25 private theorem sphere_first_quotient_label  (s t : ℤ)
  26     (y : Ambient model .sphere (s,t)) :
  27     (sphereFirstQuotient model.nu model.quotients.functoriality model.firstQuotient) s t
  28       (firstLabel model .sphere s t y ≫ XModLambdaN.map
  29         ((SyntheticCategory.biShift_zero.hom.app _) ≫ model.nu.unitIso.hom) 1) = y := by
  30   let e := SyntheticCategory.biShift_zero.app (model.nu.functor.obj SphereSpectrum) ≪≫ model.nu.unitIso
  31   let Q := (model.quotients.functoriality.functor 1).mapIso e
  32   let c : BiHom (t-s) t (XModLambdaN (nuZero model .sphere) 1) ≃ₗ[ℤ] Ambient model .sphere (s,t) :=
  33     Eq.mp (congrArg (fun z : ℤ => BiHom (t-s) z
  34       (XModLambdaN (nuZero model .sphere) 1) ≃ₗ[ℤ] Ambient model .sphere (s,t))
  35       (Int.add_zero t)) (model.firstQuotient .sphere 0 s t)
  36   change c ((firstLabel model .sphere s t y ≫ Q.hom) ≫ Q.inv) = y
  37   rw [Category.assoc,Iso.hom_inv_id,Category.comp_id]
  38   have cast_symm (n m : ℤ) (h : n = m)
  39       (e : BiHom (t-s) n (XModLambdaN (nuZero model .sphere) 1) ≃ₗ[ℤ]
  40         Ambient model .sphere (s,t)) (z : Ambient model .sphere (s,t)) :
  41       (congrArg (fun w => BiHom (t-s) w (XModLambdaN (nuZero model .sphere) 1)) h).mp
  42           (e.symm z) =
  43         ((congrArg (fun w => BiHom (t-s) w (XModLambdaN (nuZero model .sphere) 1) ≃ₗ[ℤ]
  44           Ambient model .sphere (s,t)) h).mp e).symm z := by
  45     subst m
  46     rfl
  47   have hc : firstLabel model .sphere s t y = c.symm y := by
  48     exact cast_symm (t+0) t (Int.add_zero t) (model.firstQuotient .sphere 0 s t) y
  49   rw [hc,LinearEquiv.apply_symm_apply]

  51 private theorem sphere_finite_lambda_detects 
  52     (q : ℕ) (_hq : 0 < q) (s t : ℤ) (k : ℕ)
  53     (y : Ambient model .sphere (s,t))
  54     (α : BiHom (t-s) t (model.nu.functor.obj SphereSpectrum))
  55     (hα : α ≫ XModLambdaN.incl _ 1 =
  56       firstLabel model .sphere s t y ≫
  57         XModLambdaN.map (SyntheticCategory.biShift_zero.hom.app _) 1) :
  58     Detects (model.convergence (.quotient q (.nu .sphere))) (s,t,t-k)
  59       (finiteTargetLabel model .sphere q s t k y)
  60       (lambdaMultiply k α ≫ XModLambdaN.incl _ q) := by
  61   have family_page_id (X : SyntheticSpectrum) (r : ℤ) (i : Tridegree) :
  62       familyPageMap model.family (𝟙 X) r i = 𝟙 _ := by
  63     change (model.family.map (𝟙 X)).pageMap r i = 𝟙 _
  64     rw [SyntheticAdamsFamily.map_id,SpectralSequenceMorphism.pageMap_id]
  65   let a : BiHom (t-s) t (S_0_0 : SyntheticSpectrum) := α ≫ model.nu.unitIso.hom
  66   have hfirst : quotientClass 1 a =
  67       firstLabel model .sphere s t y ≫ XModLambdaN.map
  68         (SyntheticCategory.biShift_zero.hom.app _ ≫ model.nu.unitIso.hom) 1 := by
  69     dsimp only [quotientClass,a]
  70     rw [Category.assoc,XModLambdaN.incl_naturality,←Category.assoc,hα,
  71       Category.assoc,←model.quotients.functoriality.map_comp]
  72     rfl
  73   have hl : (sphereFirstQuotient model.nu model.quotients.functoriality model.firstQuotient) s t (quotientClass 1 a) = y := by
  74     rw [hfirst]
  75     exact sphere_first_quotient_label s t y
  76   have hs := model.sphereDetectionCompatible.homotopy_lambda s t k a
  77   rw [hl] at hs
  78   have hforw : familyPageMap model.family model.nu.unitIso.hom 2 (s,t,t-k)
  79       (targetNuLabel model .sphere s t k y) = model.comparison.sphereE2 s t k y := by
  80     let z : (model.family.obj (nuZero model .sphere)).E₂ (s,t,t-k) := by
  81       simpa [SyntheticAdamsSS.E₂] using model.comparison.nuE2 .sphere 0 s t k y
  82     exact (ConcreteCategory.congr_hom (family_page_comp
  83       (SyntheticCategory.biShift_zero.hom.app _) model.nu.unitIso.hom 2 (s,t,t-k)) z).trans
  84         (model.comparison.sphere_nu s t k y)
  85   have hback : familyPageMap model.family model.nu.unitIso.inv 2 (s,t,t-k)
  86       (model.comparison.sphereE2 s t k y) = targetNuLabel model .sphere s t k y := by
  87     rw [←hforw,←ModuleCat.comp_apply,family_page_comp,Iso.hom_inv_id,family_page_id]
  88     rfl
  89   have hc : lambdaMultiply k a ≫ model.nu.unitIso.inv = lambdaMultiply k α := by
  90     dsimp only [lambdaMultiply,a]
  91     simp only [Category.assoc,Iso.hom_inv_id,Category.comp_id]
  92   have hn := detects_map .sphere (.nu .sphere) model.nu.unitIso.inv
  93     (s,t,t-k) _ _ hs
  94   have hn' : Detects (model.convergence (.nu .sphere)) (s,t,t-k)
  95       (targetNuLabel model .sphere s t k y) (lambdaMultiply k α) := by
  96     exact (congrArg₂ (Detects (model.convergence (.nu .sphere)) (s,t,t-k)) hback hc).mp hn
  97   exact detects_map (.nu .sphere) (.quotient q (.nu .sphere))
  98     (XModLambdaN.incl _ q) (s,t,t-k) _ _ hn'

 104 private theorem shift_frame (A : SyntheticObject) (δ p : ℤ × ℤ) (s : ℤ)
 105     (e : ((model.family.obj (A.obj model.nu model.auxiliary)).sequence.ssData (s,p.1+s,p.2)).eInfty)
 106     (a : (Subobject.underlying.obj ((model.syntheticTower.filtration
 107       (A.obj model.nu model.auxiliary)).F s p) : ModuleCat ℤ))
 108     (ha : (model.syntheticTower.filtration (A.obj model.nu model.auxiliary)).toAssociatedGraded s p a =
 109       (infinityGradedIso (model.convergence A) p s).hom e) :
 110     ∃ b : (Subobject.underlying.obj ((model.syntheticTower.filtration
 111       ((SyntheticObject.shift δ A).obj model.nu model.auxiliary)).F s (p+δ)) : ModuleCat ℤ),
 112       ((model.syntheticTower.filtration ((SyntheticObject.shift δ A).obj model.nu model.auxiliary)).F
 113         s (p+δ)).arrow b = shiftHomotopyClass δ p
 114           (((model.syntheticTower.filtration (A.obj model.nu model.auxiliary)).F s p).arrow a) ∧
 115       (model.syntheticTower.filtration ((SyntheticObject.shift δ A).obj model.nu model.auxiliary)).toAssociatedGraded
 116         s (p+δ) b = (infinityGradedIso (model.convergence (.shift δ A)) (p+δ) s).hom
 117           ((towerInfinityIso (model.convergence A) (model.convergence (.shift δ A))
 118             ((essShift δ).filtered A p) s).hom e) := by
 119   let S := (essShift δ).filtered A p
 120   refine ⟨(S.level s).hom a,?_,?_⟩
 121   · exact (ConcreteCategory.congr_hom (S.level_arrow s) a).trans
 122       ((essShift δ).ambient_eq A p _)
 123   · have h := ConcreteCategory.congr_hom (S.projection_comm s) a
 124     simp only [ModuleCat.comp_apply] at h
 125     rw [ha] at h
 126     rw [←h]
 127     change (S.gradedIso s).hom ((towerGradedIso (model.convergence A) p s).hom e) =
 128       (towerGradedIso (model.convergence (.shift δ A)) (p+δ) s).hom
 129         ((towerInfinityIso (model.convergence A) (model.convergence (.shift δ A)) S s).hom e)
 130     simp only [towerInfinityIso, Iso.trans_hom, Iso.symm_hom, ←ModuleCat.comp_apply,
 131       Category.assoc, Iso.inv_hom_id, Category.comp_id]

 134     (x : (model.family.obj (A.obj model.nu model.auxiliary)).E₂ (s,p.1+s,p.2))

 136     (ha : Detects (model.convergence A) (s,p.1+s,p.2) x
 137       (homotopyRegrade (by dsimp; omega) rfl a)) :
 138     Detects (model.convergence (.shift δ A)) (s,(p+δ).1+s,(p+δ).2)

 144     ((essShift δ).representative_iff A 2 (by decide) p s x e |>.mp he) b
 145     hgb
 146   rw [hb,ha'] at hdet
 147   exact hdet

 235 /-- The actual λ-multiple of any permanent lift has the prescribed finite label.
 236 The shift, finite quotient map and E∞ window all come from the same fixed model. -/
 237 private theorem finite_lambda_detects (X : ClassicalObject) (q k : ℕ) (hk : 0 < k) (hkq : k < q)
 238     (s t : ℤ) (y : Ambient model X (s,t))
 239     (hy : y ∈ permanentCycles model X (s,t))
 240     (α : BiHom (t-s) t (model.nu.functor.obj (X.obj model.auxiliary)))
 241     (hα : α ≫ XModLambdaN.incl _ 1 = firstLabel model X s t y ≫
 242       XModLambdaN.map (SyntheticCategory.biShift_zero.hom.app _) 1) :
 243     Detects (model.convergence (.quotient q (.nu X))) (s,t,t-k)
 244       (finiteTargetLabel model X q s t k y)
 245       (lambdaMultiply k α ≫ XModLambdaN.incl _ q) := by
 246   let A := model.nu.functor.obj (X.obj model.auxiliary)
 247   let n := q-k
 248   have hn : 0 < n := by dsimp [n]; omega
 249   let a := α ≫ XModLambdaN.incl A n
 250   have ha := first_label_detects_unshifted X n hn s t y a (by
 251     dsimp only [a]
 252     rw [Category.assoc,(model.quotients.tower A).rho_quotient]
 253     exact hα)
 254   have hsource : (s,t,t)=(s,(t-s)+s,t) := by simp
 255   have hstem := detects_transport hsource _ _ ha
 256   let L : ((SyntheticObject.shift (0,-(k:ℤ)) (.quotient n (.nu X))).obj model.nu model.auxiliary) ⟶
 257       ((SyntheticObject.quotient q (.nu X)).obj model.nu model.auxiliary) :=
 258     ((model.quotients.tower A).triangle hk hkq).lambdaMap
 259   have hmap := shifted_map_detects (.quotient n (.nu X)) (.quotient q (.nu X))
 260     (0,-(k:ℤ)) (t-s,t) s L
 261     (regradeE2 hsource (regradeE2 (by simp) (finiteTargetLabel model X n s t 0 y))) a hstem
 262   have hindex : (s,((t-s,t)+(0,-(k:ℤ))).1+s,((t-s,t)+(0,-(k:ℤ))).2) =
 263       (s,t,t-k) := by ext <;> simp [sub_eq_add_neg]
 264   have hmap := detects_transport hindex _ _ hmap
 265   obtain ⟨e,he,b,hb,hgb⟩ := hmap
 266   obtain ⟨e0,he0,_⟩ := ha
 267   have heStem := representative_transport hsource _ _ he0
 268   have heMap := shifted_representative_map (.quotient n (.nu X)) (.quotient q (.nu X))
 269     (0,-(k:ℤ)) (t-s,t) s L _ _ heStem
 270   have heMap := representative_transport hindex _ _ heMap
 271   have heq := he.unique heMap
 272   have heq' : e = KIP126.Literature.Standard.infinityFiniteLambda X q k hkq (s,t) t e0 := by
 273     convert heq using 1
 274     simp only [KIP126.Literature.Standard.infinityFiniteLambda,
 275       InfinityShiftCoordinates.finiteLambdaMap, show k ≠ 0 by omega, ↓reduceDIte,
 276       InfinityShiftCoordinates.shift, infinityShift, Iso.trans_hom, ModuleCat.comp_apply,
 277       eqToIso.hom, KIP126.Literature.Standard.infinityShiftData,
 278       KIP126.Literature.Standard.infinityShift_comparison]
 279     rw [infinity_map_regrade L hindex]
 280     rfl
 281   have hlabel : HasInfinityRepresentative _ 2 (s,t,t-k)
 282       (finiteTargetLabel model X q s t k y)
 283       (KIP126.Literature.Standard.infinityFiniteLambda X q k hkq (s,t) t e0) := by
 284     apply (finite_target_label_iff X q (by omega) s t k hkq
 285       ⟨y,permanentCycles_le_cycles model X (s,t) _ hy⟩ _).mpr
 286     rw [KIP126.Literature.Standard.eInfinity_lambda_finite X q k hkq (s,t) t (by dsimp [n] at hn; omega)]
 287     have he0' := (KIP126.Literature.Standard.eInfinity_finite_labels X n hn s t t
 288       (by constructor <;> dsimp [n] at * <;> omega) y e0).mp
 289       ((congrArg (fun z => HasInfinityRepresentative _ 2 (s,t,t) z e0)
 290         (zero_weight_label X n s t y)).mpr he0)
 291     rw [he0'.2]
 292     rfl
 293   have heFinal : HasInfinityRepresentative _ 2 (s,t,t-k)
 294       (finiteTargetLabel model X q s t k y) e := by
 295     simpa only [heq'] using hlabel
 296   refine ⟨e,heFinal,b,?_,hgb⟩
 297   convert hb using 1
 298   simp only [regrade_comp]
 299   simp only [regrade_precomp, L, a, shiftHomotopyClass, lambdaMultiply,
 300     Functor.map_comp, Category.assoc]
 301   rw [((model.quotients.tower A).triangle hk hkq).lambda_quotient,
 302     ←Category.assoc ((SyntheticCategory.biShift (0,-(k:ℤ))).map α), lambdaPow_naturality]
 303   simp only [←Category.assoc,eqToHom_trans]
 304   rfl
