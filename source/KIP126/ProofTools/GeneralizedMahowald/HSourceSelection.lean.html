  17     (X Y : ClassicalObject) (f : X.obj model.auxiliary ⟶ Y.obj model.auxiliary)

 228 /- Records the actual inverse shift, separately at the two possible
 229 normalized exponents. This is a conclusion, never an added hypothesis. -/
 230 private def FiniteSourceUnshifts
 231     (X Y : ClassicalObject) (f : X.obj model.auxiliary ⟶ Y.obj model.auxiliary)
 232     (k : ℕ) (s t : ℤ)
 233     (a₀ : BiHom (t-s) t (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) k))
 234     (ah : BiHom (t-s) (t+normalizedExponent model.classicalTower f)
 235       ((SyntheticObject.quotient k (extensionSourceObject model X Y f)).obj model.nu model.auxiliary)) : Prop :=
 236     (∃ he : normalizedExponent model.classicalTower f = 0,
 237       homotopyRegrade rfl (show t=t+0 by simp) a₀≫
 238         XModLambdaN.map (SyntheticCategory.biShift_zero.inv.app _) k =
 239         ((congrArg (fun z : ℤ => BiHom (t-s) (t+z)
 240           (XModLambdaN ((SyntheticCategory.biShift (0,z)).obj
 241             (model.nu.functor.obj (X.obj model.auxiliary))) k))
 242           (show (normalizedExponent model.classicalTower f : ℤ)=0 by omega)).mp ah)) ∨
 243     (∃ he : normalizedExponent model.classicalTower f = 1,
 244       ∃ ε : XModLambdaN ((SyntheticCategory.biShift (0,1)).obj
 245           (model.nu.functor.obj (X.obj model.auxiliary))) k ≅
 246         (SyntheticCategory.biShift (0,1)).obj
 247           (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) k),
 248         XModLambdaN.incl _ k≫ε.hom =
 249           (SyntheticCategory.biShift (0,1)).map (XModLambdaN.incl _ k) ∧
 250         homotopyRegrade (show ((t-s,t)+(0,1)).1=t-s by simp)
 251           (show ((t-s,t)+(0,1)).2=t+1 by rfl)
 252           (shiftHomotopyClass (0,1) (t-s,t) a₀)≫ε.inv =
 253           ((congrArg (fun z : ℤ => BiHom (t-s) (t+z)
 254             (XModLambdaN ((SyntheticCategory.biShift (0,z)).obj
 255               (model.nu.functor.obj (X.obj model.auxiliary))) k))
 256             (show (normalizedExponent model.classicalTower f : ℤ)=1 by omega)).mp ah))

 276 private theorem finite_extension_source_sync
 277     (X Y : ClassicalObject) (f : X.obj model.auxiliary ⟶ Y.obj model.auxiliary)
 278     (r q : ℕ) (hq : 0<q) (n s t : ℤ)
 279     (x : Ambient model X (s,t)) (y : Ambient model Y (s+n,t+n))
 280     (W : FiniteExtensionWitness model X Y f r n s t x y)
 281     (hnc : NoCrossingOn model X ((r-1:ℕ)+q) r (s,t))
 282     (aZ : BiHom (t-s) t (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) (r-1)))
 283     (haZ : Detects (model.convergence (.quotient (r-1) (.nu X))) (s,t,t)
 284       (regradeE2 (by simp) (finiteTargetLabel model X (r-1) s t 0 x)) aZ) :
 285     ∃ a₀ : BiHom (t-s) t (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) (r-1)),
 286       Detects (model.convergence (.quotient (r-1) (.nu X))) (s,t,t)
 287         (regradeE2 (by simp) (finiteTargetLabel model X (r-1) s t 0 x)) a₀ ∧
 288       a₀ ≫ ((model.quotients.tower _).triangle
 289         (show 0<r-1 by have := W.page_ge_two; omega)
 290         (show r-1<r-1+q by omega)).delta =
 291         aZ ≫ ((model.quotients.tower _).triangle
 292           (show 0<r-1 by have := W.page_ge_two; omega)
 293           (show r-1<r-1+q by omega)).delta ∧
 294       FiniteExtensionAtSource X Y f r n s t x y W a₀ := by
 295   by_cases he : normalizedExponent model.classicalTower f = 0
 296   · obtain ⟨ah,hs,ht,a₀,ha₀,hδ,hback⟩ :=
 297       finite_extension_source_sync_zero X Y f r q hq n s t x y W he hnc aZ haZ
 298     exact ⟨a₀,ha₀,hδ,ah,hs,ht,Or.inl ⟨he,hback⟩⟩
 299   · have he1 : normalizedExponent model.classicalTower f = 1 := by
 300       have := normalizedExponent_le_one model.classicalTower f
