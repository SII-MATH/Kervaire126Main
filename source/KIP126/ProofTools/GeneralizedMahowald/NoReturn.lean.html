  26 window. This includes the positive source shift of a normalized map. -/
  27 private theorem nonreturn_at_window_end (X Y : ClassicalObject)
  28     (j : ℕ) (hj : 0 < j) (d : ℤ) (hd : d ≤ -(j:ℤ)) (m w : ℤ)
  29     (c : BiHom m w ((SyntheticCategory.biShift (0,d)).obj
  30       (model.nu.functor.obj (X.obj model.auxiliary))))
  31     (z : (SyntheticCategory.biShift (0,d)).obj
  32       (model.nu.functor.obj (X.obj model.auxiliary)) ⟶
  33         XModLambdaN (model.nu.functor.obj (Y.obj model.auxiliary)) j) : c ≫ z = 0 := by
  34   let A : SyntheticObject := .shift (0,d) (.nu X)
  35   let B : SyntheticObject := .quotient j (.nu Y)
  36   let L : ℤ := w-m-d
  37   have hzA (k : ℤ) (hk : k < L) :
  38       IsZero (((model.family.obj (A.obj model.nu model.auxiliary)).sequence.ssData (k,m+k,w)).eInfty) := by
  39     have hz := nu_infinity_zero_above X k (m+k) (w-d) (by dsimp [L] at hk; omega)
  40     let e := towerInfinityIso (model.convergence (.nu X)) (model.convergence A)
  41       ((essShift (0,d)).filtered (.nu X) (m,w-d)) k
  42     have he := hz.of_iso e.symm
  43     have hw : w - d + d = w := by omega
  44     simpa only [e, A, Prod.fst_add, Prod.snd_add, add_zero, hw] using he
  45   have hc0 := filtration_at_zero model.syntheticTower _ m w c
  46   have hcL : FiltrationAtLeast model.syntheticTower L c := by
  47     by_cases hL : L ≤ 0
  48     · exact model.syntheticTower.filtrationSubmodule_antitone _ (m,w) hL hc0
  49     · exact (filtration_iff_of_infinity_zero (model.convergence A) 0 L m w
  50         (by omega) (fun k _ hk => hzA k hk) c).mp hc0
  51   have hcz : FiltrationAtLeast model.syntheticTower L (c ≫ z) :=
  52     map_preserves_filtration (model.convergentMap A B z) (m,w) L c hcL
  53   exact finite_high_filtration_zero Y j hj m w L (by dsimp [L]; omega) (c ≫ z) hcz

  66 theorem quotient_nu_maps_equal_on_spheres
  67       (X Y : ClassicalObject)
  68       (j : ℕ) (hj : 0 < j)
  69       (a b : XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) j ⟶
  70         XModLambdaN (model.nu.functor.obj (Y.obj model.auxiliary)) j)
  71       (hab : XModLambdaN.incl _ j ≫ a = XModLambdaN.incl _ j ≫ b) :
  72       ∀ (m w : ℤ) (c : BiHom m w (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) j)),
  73         c ≫ a = c ≫ b := by
  74     apply quotient_maps_equal_on_spheres_of_no_return j a b hab
  75     intro m w c z
  76     let V := model.nu.functor.obj (X.obj model.auxiliary)
  77     let ε : ((SyntheticCategory.biShift (0,-(j:ℤ))).obj V)⟦(1:ℤ)⟧ ≅
  78         (SyntheticCategory.biShift (0,-(j:ℤ)-1)).obj
  79           (model.nu.functor.obj ((ClassicalObject.shift 1 X).obj model.auxiliary)) :=
  80       (SyntheticCategory.biShift_compat (Syn:=SyntheticSpectrum) 1).symm.app _ ≪≫
  81         (SyntheticCategory.biShift_comp (0,-(j:ℤ)) (1,0)).app V ≪≫
  82         eqToIso (congrArg (fun p => (SyntheticCategory.biShift p).obj V)
  83           (show (0,-(j:ℤ))+(1,0)=(1,1)+(0,-(j:ℤ)-1) by ext <;> simp)) ≪≫
  84         (SyntheticCategory.biShift_comp (1,1) (0,-(j:ℤ)-1)).symm.app V ≪≫
  85         (SyntheticCategory.biShift (0,-(j:ℤ)-1)).mapIso (model.nu.suspensionIso _).symm
  86     have h := negative_shift_nu_to_quotient_zero (.shift 1 X) Y j hj m w
  87       (c ≫ ε.hom) (ε.inv ≫ z)
  88     simpa only [Category.assoc,Iso.hom_inv_id_assoc] using h

  90 theorem shift_minus_one_quotient_maps_equal_on_spheres
  91       (X Y : ClassicalObject)
  92       (d : ℕ)
  93       (a b : (SyntheticCategory.biShift (0,-1)).obj
  94           (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) d) ⟶
  95         XModLambdaN (model.nu.functor.obj (Y.obj model.auxiliary)) (d+1))
  96       (hab : (SyntheticCategory.biShift (0,-1)).map (XModLambdaN.incl _ d) ≫ a =
  97         (SyntheticCategory.biShift (0,-1)).map (XModLambdaN.incl _ d) ≫ b) :
  98       ∀ (m w : ℤ) (c : BiHom m w ((SyntheticCategory.biShift (0,-1)).obj
  99           (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) d))), c ≫ a = c ≫ b := by
 100     letI := model.exactness.shiftCommShift (0,-1)
 101     letI := model.exactness.shiftExact (0,-1)
 102     let F := SyntheticCategory.biShift (Syn:=SyntheticSpectrum) (0,-1)
 103     let V := model.nu.functor.obj (X.obj model.auxiliary)
 104     let T := XModLambdaN.cofiberTriangle V d
 105     have hzero : (F.mapTriangle.obj T).mor₂ ≫ (a-b) = 0 := by
 106       change F.map (XModLambdaN.incl V d) ≫ (a-b) = 0
 107       rw [Preadditive.comp_sub,hab,sub_self]
 108     obtain ⟨z,hz⟩ := Triangle.yoneda_exact₃ (F.mapTriangle.obj T)
 109       (F.map_distinguished T (XModLambdaN.cofiberTriangle_distinguished V d)) (a-b) hzero
 110     intro m w c
 111     let ε : (F.obj ((SyntheticCategory.biShift (0,-(d:ℤ))).obj V))⟦(1:ℤ)⟧ ≅
 112         (SyntheticCategory.biShift (0,-((d+1:ℕ):ℤ)-1)).obj
 113           (model.nu.functor.obj ((ClassicalObject.shift 1 X).obj model.auxiliary)) :=
 114       (shiftFunctor SyntheticSpectrum (1:ℤ)).mapIso ((lambdaShiftAddIso d 1 (d+1) rfl).app V) ≪≫
 115         (SyntheticCategory.biShift_compat (Syn:=SyntheticSpectrum) 1).symm.app _ ≪≫
 116         (SyntheticCategory.biShift_comp (0,-((d+1:ℕ):ℤ)) (1,0)).app V ≪≫
 117         eqToIso (congrArg (fun p => (SyntheticCategory.biShift p).obj V)
 118           (show (0,-((d+1:ℕ):ℤ))+(1,0)=(1,1)+(0,-((d+1:ℕ):ℤ)-1) by ext <;> simp)) ≪≫
 119         (SyntheticCategory.biShift_comp (1,1) (0,-((d+1:ℕ):ℤ)-1)).symm.app V ≪≫
 120         (SyntheticCategory.biShift (0,-((d+1:ℕ):ℤ)-1)).mapIso (model.nu.suspensionIso _).symm
 121     have h := negative_shift_nu_to_quotient_zero (.shift 1 X) Y (d+1) (by omega) m w
 122       ((c ≫ (F.mapTriangle.obj T).mor₃) ≫ ε.hom) (ε.inv ≫ z)
 123     have hcz : (c ≫ (F.mapTriangle.obj T).mor₃) ≫ z = 0 := by
 124       simpa only [Category.assoc,Iso.hom_inv_id_assoc] using h
 125     apply sub_eq_zero.mp
 126     rw [←Preadditive.comp_sub,hz,←Category.assoc]
 127     exact hcz
 128 /-- Chosen cofiber fillings for a normalized source agree on sphere maps.
 129 The exponent-one case uses the endpoint of the finite no-return window. -/
 130 private theorem normalized_quotient_maps_equal_on_spheres
 131     (X Y : ClassicalObject) (k : ℕ) (hk : k ≤ 1) (j : ℕ) (hj : 0 < j)
 132     (a b : XModLambdaN ((SyntheticCategory.biShift (0,(k:ℤ))).obj
 133         (model.nu.functor.obj (X.obj model.auxiliary))) j ⟶
 134       XModLambdaN (model.nu.functor.obj (Y.obj model.auxiliary)) j)
 135     (hab : XModLambdaN.incl _ j ≫ a = XModLambdaN.incl _ j ≫ b) :
 136     ∀ (m w : ℤ) (c : BiHom m w (XModLambdaN ((SyntheticCategory.biShift (0,(k:ℤ))).obj
 137       (model.nu.functor.obj (X.obj model.auxiliary))) j)), c ≫ a = c ≫ b := by
 138   apply quotient_maps_equal_on_spheres_of_no_return j a b hab
 139   intro m w c z
 140   let V := model.nu.functor.obj (X.obj model.auxiliary)
 141   let ε : (((SyntheticCategory.biShift (0,-(j:ℤ))).obj
 142       ((SyntheticCategory.biShift (0,(k:ℤ))).obj V)))⟦(1:ℤ)⟧ ≅
 143       (SyntheticCategory.biShift (0,(k:ℤ)-(j:ℤ)-1)).obj
 144         (model.nu.functor.obj ((ClassicalObject.shift 1 X).obj model.auxiliary)) :=
 145     (shiftFunctor SyntheticSpectrum (1:ℤ)).mapIso
 146       ((SyntheticCategory.biShift_comp (0,(k:ℤ)) (0,-(j:ℤ))).app V) ≪≫
 147       (SyntheticCategory.biShift_compat (Syn:=SyntheticSpectrum) 1).symm.app _ ≪≫
 148       (SyntheticCategory.biShift_comp ((0,(k:ℤ))+(0,-(j:ℤ))) (1,0)).app V ≪≫
 149       eqToIso (congrArg (fun p => (SyntheticCategory.biShift p).obj V)
 150         (show ((0,(k:ℤ))+(0,-(j:ℤ)))+(1,0)=(1,1)+(0,(k:ℤ)-(j:ℤ)-1)
 151           by ext <;> simp; omega)) ≪≫
 152       (SyntheticCategory.biShift_comp (1,1) (0,(k:ℤ)-(j:ℤ)-1)).symm.app V ≪≫
 153       (SyntheticCategory.biShift (0,(k:ℤ)-(j:ℤ)-1)).mapIso (model.nu.suspensionIso _).symm
 154   have hzero (c' : BiHom m w ((SyntheticCategory.biShift (0,(k:ℤ)-(j:ℤ)-1)).obj
 155         (model.nu.functor.obj ((ClassicalObject.shift 1 X).obj model.auxiliary))))
 156       (z' : (SyntheticCategory.biShift (0,(k:ℤ)-(j:ℤ)-1)).obj
 157         (model.nu.functor.obj ((ClassicalObject.shift 1 X).obj model.auxiliary)) ⟶
 158           XModLambdaN (model.nu.functor.obj (Y.obj model.auxiliary)) j) : c' ≫ z' = 0 :=
 159     nonreturn_at_window_end (.shift 1 X) Y j hj ((k:ℤ)-(j:ℤ)-1) (by omega) m w c' z'
 160 
 161   have h := hzero (c ≫ ε.hom) (ε.inv ≫ z)
 162   simpa only [Category.assoc,Iso.hom_inv_id_assoc] using h
 163 end
