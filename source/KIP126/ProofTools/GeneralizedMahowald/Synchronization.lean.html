  21 private theorem relative_boundary_equal_of_no_crossing (X : ClassicalObject)
  22     (i q : ℕ) (hi : 0 < i) (hq : 0 < q) (s t : ℤ)
  23     (hnc : NoCrossingOn model X (i+q) (i+1) (s,t))
  24     (a a₀ : BiHom (t-s) t (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) i))
  25     (ha : FiltrationAtLeast model.syntheticTower (s+1) (a-a₀)) :
  26     a ≫ ((model.quotients.tower _).triangle hi (show i < i+q by omega)).delta =
  27       a₀ ≫ ((model.quotients.tower _).triangle hi (show i < i+q by omega)).delta := by
  28   let A := model.nu.functor.obj (X.obj model.auxiliary)
  29   let M := model.convergentMap (.quotient i (.nu X)) (.shift (1,-(i:ℤ)) (.nu X))
  30     (absoluteBoundary A i)
  31   have hnc' : TowerNoCrossing M (t-s,t) s (i+q) := by
  32     have H := classical_noCrossing_absolute_boundary X (i+1) (i+q) s t hnc
  33     exact H
  34   have h := tower_image_same_target_of_no_crossing M (t-s,t) s (i+q)
  35     (by omega) hnc' a a₀ ha
  36   change FiltrationAtLeast model.syntheticTower (s+(i+q)+1)
  37     (a ≫ absoluteBoundary A i - a₀ ≫ absoluteBoundary A i) at h
  38   rw [←Preadditive.sub_comp] at h
  39   have hα := (absolute_boundary_filtration_iff X i (t-s) t (s+(i+q)+1) (a-a₀)).mp h
  40   let α := biSuspensionHomEquiv (t-s) t i A ((a-a₀) ≫ XModLambdaN.proj A i)
  41   have hproj : FiltrationAtLeast model.syntheticTower (s+(i+q)+1)
  42       (α ≫ XModLambdaN.incl A ((i+q)-i)) :=
  43     map_preserves_filtration (model.convergentMap (.nu X) (.quotient ((i+q)-i) (.nu X))
  44       (XModLambdaN.incl A ((i+q)-i))) (t-s-1,t+i) (s+(i+q)+1) α hα
  45   have hz := finite_high_filtration_zero X ((i+q)-i) (by omega) (t-s-1) (t+i)
  46     (s+(i+q)+1) (by omega) _ hproj
  47   apply (biSuspensionHomEquiv (t-s) t i (XModLambdaN A ((i+q)-i))).injective
  48   rw [cofiber_boundary_projection X i (i+q) hi (by omega),
  49     cofiber_boundary_projection X i (i+q) hi (by omega)]
  50   apply sub_eq_zero.mp
  51   simpa only [α,Preadditive.sub_comp,map_sub] using hz
  52 
  53 end
