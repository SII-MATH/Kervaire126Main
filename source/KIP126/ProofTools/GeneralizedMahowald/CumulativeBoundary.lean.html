  16 private theorem cumulative_boundary_permanent (X : ClassicalObject) (b : ℤ)
  17     (p : Bidegree) (y : Ambient model X p)
  18     (hy : y ∈ boundaries model X b p) : ClassicalPermanentCycle model X p y := by
  19   obtain ⟨z,hz⟩ := hy
  20   let E := model.classical.obj (X.obj model.auxiliary)
  21   let P := E.ssData p
  22   let n₀ : WithTop ℕ := ↑(2-E.r₀).toNat
  23   have hn₀ : n₀=0 := by dsimp [n₀,E]; rw [model.classical.firstPage]; rfl
  24   let j := Subobject.ofLE (P.B (↑(b-1).toNat)) (P.Z ⊤)
  25     ((P.B_mono le_top).trans (P.B_le_Z ⊤))
  26   refine ⟨j z,?_⟩
  27   change (j ≫ Subobject.ofLE (P.Z ⊤) (P.Z n₀) (P.Z_anti le_top) ≫ P.pageπ n₀) z=y
  28   rw [normalizeInitialProjection P n₀ ⊤ hn₀ le_top]
  29   change ((j ≫ Subobject.ofLE (P.Z ⊤) (P.Z 0) (P.Z_anti bot_le)) ≫ P.pageπ 0 ≫ _) z=y
  30   rw [show j ≫ Subobject.ofLE (P.Z ⊤) (P.Z 0) (P.Z_anti bot_le) =
  31     Subobject.ofLE (P.B (↑(b-1).toNat)) (P.Z 0) _ by simp only [j,Subobject.ofLE_comp_ofLE]]
  32   exact hz

  34 private theorem cumulative_boundary_hit (X : ClassicalObject)
  35     (r : ℕ) (hr : 2 ≤ r) (p : Bidegree) (y : Ambient model X p)
  36     (hy : y ∈ boundaries model X r p)
  37     (hprev : y ∉ boundaries model X (r-1) p) :
  38     HitOnPage (model.classical.obj (X.obj model.auxiliary)) r p y := by
  39   let E := model.classical.obj (X.obj model.auxiliary)
  40   let P := E.ssData p
  41   let n : ℕ := ((r:ℤ)-E.r₀).toNat
  42   let n₀ : WithTop ℕ := ↑(2-E.r₀).toNat
  43   have hn₀ : n₀=0 := by dsimp [n₀,E]; rw [model.classical.firstPage]; rfl
  44   have hn' : ((r:ℤ)-1).toNat=n+1 := by dsimp [n,E]; rw [model.classical.firstPage]; omega
  45   change y ∈ LinearMap.range (boundaryLabel model X (↑((r:ℤ)-1).toNat) p).hom at hy
  46   rw [hn'] at hy
  47   obtain ⟨z,hz⟩ := hy
  48   let j := Subobject.ofLE (P.B ↑(n+1)) (P.Z ↑n)
  49     ((P.B_le_Z _).trans (P.Z_anti (by exact_mod_cast Nat.le_succ n)))
  50   let yr : E.Page r p := P.pageπ ↑n (j z)
  51   have hyr : RepresentsOnPage E r p y yr := by
  52     refine ⟨by omega,j z,?_,rfl⟩
  53     change (j ≫ Subobject.ofLE (P.Z ↑n) (P.Z n₀) _ ≫ P.pageπ n₀) z=y
  54     rw [normalizeInitialProjection P n₀ ↑n hn₀ (by rw [hn₀]; exact bot_le)]
  55     change ((j ≫ Subobject.ofLE (P.Z ↑n) (P.Z 0) _) ≫ P.pageπ 0 ≫ _) z=y
  56     rw [show j ≫ Subobject.ofLE (P.Z ↑n) (P.Z 0) _ =
  57       Subobject.ofLE (P.B ↑(n+1)) (P.Z 0) _ by simp only [j,Subobject.ofLE_comp_ofLE]]
  58     exact hz
  59   have hyrne : yr ≠ 0 := by
  60     intro hyrzero
  61     exact hprev (represents_zero_mem_boundary model X r p y (by simpa only [E,hyrzero] using hyr))
  62   let q := p-E.diffDeg r
  63   have hqp : q+E.diffDeg r=p := by dsimp [q]; abel
  64   have himage : yr ∈ LinearMap.range (E.d r q ≫ eqToHom (congrArg (E.Page r) hqp)).hom := by
  65     have h := congrArg (ModuleCat.subobjectModule (E.Page r (q+E.diffDeg r)))
  66       (E.B_succ r q (by dsimp [E]; rw [model.classical.firstPage]; omega))
  67     simp only [subobjectModule_image] at h
  68     have transport (p' : Bidegree) (hp : p'=p)
  69         (d : E.Page r q ⟶ E.Page r p')
  70         (hd : LinearMap.range d.hom = LinearMap.range
  71           (Subobject.ofLE ((E.ssData p').B ↑(n+1)) ((E.ssData p').Z ↑n)
  72             (((E.ssData p').B_le_Z _).trans ((E.ssData p').Z_anti
  73               (by exact_mod_cast Nat.le_succ n))) ≫
  74             (E.ssData p').pageπ ↑n).hom) :
  75         yr ∈ LinearMap.range (d ≫ eqToHom (congrArg (E.Page r) hp)).hom := by
  76       subst p'
  77       simp only [eqToHom_refl,Category.comp_id,hd]
  78       exact ⟨z,rfl⟩
  79     exact transport _ hqp (E.d r q) h
  80   obtain ⟨a,ha⟩ := himage
  81   exact ⟨q,hqp,a,yr,hyr,hyrne,ha⟩

  83 private theorem lambdaMultiply_zero_of_le {A : SyntheticSpectrum} (m w : ℤ)
  84     (i j : ℕ) (hij : i ≤ j) (a : BiHom m w A)
  85     (ha : lambdaMultiply i a=0) : lambdaMultiply j a=0 := by
  86   simp only [lambdaMultiply,comp_eq_zero_iff_of_epi] at ha ⊢
  87   rw [lambdaPow_add model.quotients.coherence (j-i) i j (Nat.sub_add_cancel hij) _,
  88     Category.assoc,Category.assoc,ha,comp_zero,comp_zero]

  90 private theorem cumulative_boundary_torsion_lift (X : ClassicalObject)
  91     (k : ℕ) (s t : ℤ) (y : Ambient model X (s,t))
  92     (hy : y ∈ boundaries model X (k+1) (s,t)) :
  93     ∃ a : BiHom (t-s) t (nuZero model X),
  94       quotientClass 1 a=firstLabel model X s t y ∧ lambdaMultiply k a=0 := by
  95   induction k with
  96   | zero =>
  97     have hyzero : y=0 := by
  98       simpa only [Nat.cast_zero,zero_add,boundaries_one,Submodule.mem_bot] using hy
  99     subst y
 100     refine ⟨0,?_,?_⟩
 101     · have cast_zero (m w w' : ℤ) (A : SyntheticSpectrum) (hw : w=w')
 102           (he : BiHom m w A = BiHom m w' A) : he.mp 0=0 := by subst w'; rfl
 103       simp only [quotientClass,zero_comp,firstLabel,map_zero]
 104       rw [cast_zero _ _ _ _ (Int.add_zero t)]
 105     · simp only [lambdaMultiply,comp_zero]
 106   | succ k ih =>
 107     by_cases hprev : y ∈ boundaries model X (k+1) (s,t)
 108     · obtain ⟨a,ha,hak⟩ := ih hprev
 109       exact ⟨a,ha,lambdaMultiply_zero_of_le (t-s) t k (k+1) (by omega) a hak⟩
 110     · have hy' : y ∈ boundaries model X (k+2) (s,t) := by
 111         simpa only [Nat.cast_succ,add_assoc,Int.reduceAdd] using hy
 112       have hhit := cumulative_boundary_hit X (k+2) (by omega) (s,t) y hy'
 113         (by simpa only [show ((k+2:ℕ):ℤ)-1=(k:ℤ)+1 by omega] using hprev)
 114       obtain ⟨a,ha,hak⟩ := KIP126.Literature.Standard.selected_realization_boundary_lift X s t
 115         (k+2) (by omega) y (cumulative_boundary_permanent X _ (s,t) y hy') hhit
 116       have hk : (k+2:ℕ)-1=k+1 := by omega
 117       rw [hk] at hak
 118       exact ⟨a,ha,hak⟩

 120 private theorem cumulative_boundary_torsion_lift_unshifted (X : ClassicalObject)
 121     (k : ℕ) (s t : ℤ) (y : Ambient model X (s,t))
 122     (hy : y ∈ boundaries model X (k+1) (s,t)) :
 123     ∃ a : BiHom (t-s) t (model.nu.functor.obj (X.obj model.auxiliary)),
 124       a ≫ XModLambdaN.incl _ 1 = firstLabel model X s t y ≫
 125         XModLambdaN.map (SyntheticCategory.biShift_zero.hom.app _) 1 ∧
 126       lambdaMultiply k a=0 := by
 127   obtain ⟨a,ha,hak⟩ := cumulative_boundary_torsion_lift X k s t y hy
 128   let A := model.nu.functor.obj (X.obj model.auxiliary)
 129   let z : nuZero model X ⟶ A := SyntheticCategory.biShift_zero.hom.app A
 130   refine ⟨a ≫ z,?_,?_⟩
 131   · rw [Category.assoc,XModLambdaN.incl_naturality,←Category.assoc]
 132     exact congrArg (fun f => f ≫ XModLambdaN.map z 1) ha
 133   · simpa only [lambdaMultiply,Category.assoc,zero_comp] using congrArg (fun f => f ≫ z) hak
