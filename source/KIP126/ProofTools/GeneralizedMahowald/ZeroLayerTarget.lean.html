  15 /- The corrected May target uses the inclusion of the infinite normalized
  16 source label. The finite-edge decoder uses finiteNuLabel. They are exactly
  17 the same label; this bridge changes no representative or quotient length. -/
  18 private theorem normalized_included_source_label
  19     (X Y : ClassicalObject) (g : X.obj model.auxiliary ⟶ Y.obj model.auxiliary)
  20     (q : ℕ) (s t : ℤ) (x : Ambient model X (s,t)) :
  21     familyPageMap model.family
  22       (XModLambdaN.incl ((extensionSourceObject model X Y g).obj model.nu model.auxiliary) q)
  23       2 (s,t,t+normalizedExponent model.classicalTower g)
  24       (regradeE2 (by simp) (infiniteSourceForExtension model X Y g s t x)) =
  25     regradeE2 (by simp)
  26       (finiteNuLabel model X q (normalizedExponent model.classicalTower g) s t 0 x) := by
  27   unfold infiniteSourceForExtension finiteNuLabel
  28   rw [page_regrade]
  29   have trans {E : SyntheticAdamsSS} {p q r : Tridegree}
  30       (hpq : p=q) (hqr : q=r) (a : E.E₂ p) :
  31       regradeE2 hqr (regradeE2 hpq a)=regradeE2 (hpq.trans hqr) a := by
  32     subst q; subst r; rfl
  33   rw [trans]
  34   rfl

  36 private theorem corrected_second_image_finite_label
  37     (X Y : ClassicalObject) (g : X.obj model.auxiliary ⟶ Y.obj model.auxiliary)
  38     (q : ℕ) (s t : ℤ) (x : Ambient model X (s,t))
  39     (D : SyntheticObject)
  40     (J : XModLambdaN ((extensionSourceObject model X Y g).obj model.nu model.auxiliary) q ⟶
  41       D.obj model.nu model.auxiliary)
  42     (u : BiHom (t-s) (t+normalizedExponent model.classicalTower g)
  43       (D.obj model.nu model.auxiliary))
  44     (hu : Detects (model.convergence D) (s,t,t+normalizedExponent model.classicalTower g)
  45       (familyPageMap model.family J 2 (s,t,t+normalizedExponent model.classicalTower g)
  46         (familyPageMap model.family
  47           (XModLambdaN.incl ((extensionSourceObject model X Y g).obj model.nu model.auxiliary) q)
  48           2 (s,t,t+normalizedExponent model.classicalTower g)
  49           (regradeE2 (by simp) (infiniteSourceForExtension model X Y g s t x)))) u) :
  50     Detects (model.convergence D) (s,t,t+normalizedExponent model.classicalTower g)
  51       (familyPageMap model.family J 2 (s,t,t+normalizedExponent model.classicalTower g)
  52         (regradeE2 (by simp)
  53           (finiteNuLabel model X q (normalizedExponent model.classicalTower g) s t 0 x))) u := by
  54   rw [normalized_included_source_label] at hu
  55   exact hu
