  19 private def commonCorrectedGeometry
  20     {C D : SyntheticSpectrum} (i : ℕ) (M W : ℤ)
  21     (eC : C ≅ (SyntheticCategory.biShift (1,-(i:ℤ))).obj D)
  22     (c : BiHom (M-1) (W+i) D) : BiHom M W C :=
  23   (biShiftAddIso (M-1,W+i) (1,-(i:ℤ)) (M,W)
  24     (by ext <;> dsimp <;> omega)).inv.app S_0_0 ≫
  25     (SyntheticCategory.biShift (1,-(i:ℤ))).map c ≫ eC.inv

  27 private theorem common_corrected_geometry_desuspend
  28     {C D : SyntheticSpectrum} (i : ℕ) (M W : ℤ)
  29     (eC : C ≅ (SyntheticCategory.biShift (1,-(i:ℤ))).obj D)
  30     (c : BiHom (M-1) (W+i) D) :
  31     (SyntheticCategory.biShift_fullyFaithful (Syn:=SyntheticSpectrum) (1,-(i:ℤ))).preimage
  32       ((biShiftAddIso (M-1,W+i) (1,-(i:ℤ)) (M,W)
  33         (by ext <;> dsimp <;> omega)).hom.app S_0_0 ≫
  34       commonCorrectedGeometry i M W eC c ≫ eC.hom) = c := by
  35   apply (SyntheticCategory.biShift_fullyFaithful (Syn:=SyntheticSpectrum) (1,-(i:ℤ))).map_injective
  36   simp only [Functor.FullyFaithful.map_preimage,commonCorrectedGeometry,
  37     Category.assoc,Iso.inv_hom_id,Category.comp_id,Iso.hom_inv_id_app_assoc]

  39 private theorem common_correction_preserves_native_rho
  40     (X : ClassicalObject) {C : SyntheticSpectrum} (e : ℤ)
  41     (e₀ : C ≅ (SyntheticCategory.biShift (1,e)).obj
  42       (model.nu.functor.obj (X.obj model.auxiliary)))
  43     (i r : ℕ) (hir : i ≤ r)
  44     (eQ : XModLambdaN ((SyntheticCategory.biShift (1,e)).obj
  45       (model.nu.functor.obj (X.obj model.auxiliary))) r ≅
  46       (SyntheticCategory.biShift (1,e)).obj
  47         (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) r))
  48     (M W : ℤ) (c : BiHom M W (XModLambdaN C r))
  49     (c' : BiHom (M-1) (W+i) ((SyntheticCategory.biShift (0,(i:ℤ)+e)).obj
  50       (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) r)))
  51     (hρ : c'≫(SyntheticCategory.biShift (0,(i:ℤ)+e)).map
  52         ((model.quotients.tower _).rho i r hir) =
  53       (SyntheticCategory.biShift_fullyFaithful (Syn:=SyntheticSpectrum) (1,-(i:ℤ))).preimage
  54         ((biShiftAddIso (M-1,W+i) (1,-(i:ℤ)) (M,W)
  55           (by ext <;> dsimp <;> omega)).hom.app S_0_0 ≫ c ≫
  56           (commonCofiberTargetIso e e₀ i r eQ).hom) ≫
  57         (SyntheticCategory.biShift (0,(i:ℤ)+e)).map ((model.quotients.tower _).rho i r hir)) :
  58     quotientShiftDesuspend X (1,e) r eQ M W
  59         (commonCorrectedGeometry i M W (commonCofiberTargetIso e e₀ i r eQ) c' ≫
  60           XModLambdaN.map e₀.hom r) ≫ (model.quotients.tower _).rho i r hir =
  61       quotientShiftDesuspend X (1,e) r eQ M W (c≫XModLambdaN.map e₀.hom r) ≫
  62         (model.quotients.tower _).rho i r hir := by
  63   have hleft := common_cofiber_desuspend_eq_shift_cNu X e e₀ i r eQ M W
  64     (commonCorrectedGeometry i M W (commonCofiberTargetIso e e₀ i r eQ) c')
  65   dsimp only at hleft
  66   rw [common_corrected_geometry_desuspend] at hleft
  67   have hright := common_cofiber_desuspend_eq_shift_cNu X e e₀ i r eQ M W c
  68   dsimp only at hright
  69   rw [hright, hleft] at hρ
  70   rw [shift_regrade,shift_regrade] at hρ
  71   let Cb := (biShiftAddIso (Syn:=SyntheticSpectrum) (M-1,W-e) (0,(i:ℤ)+e) (M-1,W+i)
  72     (by ext <;> dsimp <;> omega)).app S_0_0
  73   apply (SyntheticCategory.biShift_fullyFaithful (Syn:=SyntheticSpectrum) (0,(i:ℤ)+e)).map_injective
  74   apply (cancel_epi Cb.inv).mp
  75   simpa only [Cb,Iso.app_inv,Functor.map_comp,Category.assoc] using hρ

  78     (X : ClassicalObject) (i r : ℕ) (hi : 0 < i) (hir : i ≤ r) (s t : ℤ)
