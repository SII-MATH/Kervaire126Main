  30 /-- Once higher-source crossings are excluded, a fixed nonzero target grade
  31 makes every actual extension to that grade essential. -/
  32 private theorem tower_essential_iff_target_nonzero_of_no_crossing
  33     (M : ConvergentMap E g cx.toConvergence cy.toConvergence)
  34     (p : ℤ × ℤ) (s n : ℤ)
  35     (x : ((E.obj X).sequence.ssData (s,p.1+s,p.2)).eInfty)
  36     (y : ((E.obj Y).sequence.ssData (s+n,p.1+(s+n),p.2)).eInfty)
  37     (he : TowerExtensionEquation M p s n x y)
  38     (hnc : TowerNoCrossing M p s n) :
  39     TowerEssentialExtension M p s n x y ↔ y ≠ 0 := by
  40   constructor
  41   · intro hess hy
  42     exact ((tower_essential_iff_no_zero M p s n x y he).mp hess) (by rwa [←hy])
  43   · intro hy
  44     apply (tower_essential_iff_no_zero M p s n x y he).mpr
  45     intro he0
  46     obtain ⟨hn,ar,br,hab,hax,hby⟩ := (tower_extension_iff M p s n x y).mp he
  47     obtain ⟨br0,hab0,hb0⟩ := tower_target_of_any_source_frame M p s n x 0 he0 hnc ar hax
  48     have hbr : br = br0 :=
  49       (ModuleCat.mono_iff_injective ((P.filtration Y).F (s+n) p).arrow).mp inferInstance
  50         (hab.symm.trans hab0)
  51     apply hy
  52     apply (infinityGradedIso cy p (s+n)).toLinearEquiv.injective
  53     change (infinityGradedIso cy p (s+n)).hom y =
  54       (infinityGradedIso cy p (s+n)).hom 0
  55     rw [←hby,←hb0,hbr]
  56 
  57 end General

  85 /-- In the first nontrivial short position, the definition of a paper crossing
  86 has exactly displacement one and no target setback. In particular its required
  87 boundary exclusion is B₂, independently of the normalization exponent. -/
  88 private theorem short_finite_has_crossing_iff
  89     (X Y : ClassicalObject) (f : X.obj model.auxiliary ⟶ Y.obj model.auxiliary)
  90     (r : ℕ) (s t : ℤ) :
  91     FiniteHasCrossing model X Y f r
  92       ((normalizedExponent model.classicalTower f:ℤ)+1) s t ↔
  93     3 ≤ r ∧ ∃ (x : Ambient model X (s+1,t+1))
  94       (y : Ambient model Y
  95         (s+1+normalizedExponent model.classicalTower f,
  96           t+1+normalizedExponent model.classicalTower f)),
  97       ∃ W : FiniteExtensionWitness model X Y f (r-1)
  98         (normalizedExponent model.classicalTower f) (s+1) (t+1) x y,
  99         W.Essential ∧ y ∉ boundaries model Y 2
 100           (s+1+normalizedExponent model.classicalTower f,
 101             t+1+normalizedExponent model.classicalTower f) := by
 102   constructor
 103   · rintro ⟨a,b,ha,har,hb,H⟩
 104     have hlen : (normalizedExponent model.classicalTower f:ℤ) ≤
 105         (normalizedExponent model.classicalTower f:ℤ)+1-a-b := by
 106       obtain ⟨x,y,W,_,_⟩ := H
 107       exact W.length_lower
 108     have ha1 : a=1 := by omega
 109     have hb0 : b=0 := by omega
 110     subst a; subst b
 111     refine ⟨by omega,?_⟩
 112     have he : (normalizedExponent model.classicalTower f:ℤ)+1-(1:ℕ)-(0:ℕ) =
 113         normalizedExponent model.classicalTower f := by omega
 114     have hB : 1+((normalizedExponent model.classicalTower f:ℤ)+1)-(0:ℕ)-
 115         normalizedExponent model.classicalTower f=2 := by omega
 116     rw [he,hB] at H
 117     exact H
 118   · rintro ⟨hr,x,y,W,hess,hy⟩
 119     refine ⟨1,0,by omega,by omega,by omega,?_⟩
 120     have he : (normalizedExponent model.classicalTower f:ℤ)+1-(1:ℕ)-(0:ℕ) =
 121         normalizedExponent model.classicalTower f := by omega
 122     have hB : 1+((normalizedExponent model.classicalTower f:ℤ)+1)-(0:ℕ)-
 123         normalizedExponent model.classicalTower f=2 := by omega
 124     rw [he,hB]
 125     exact ⟨x,y,W,hess,hy⟩

 151     (XModLambdaN ((SyntheticCategory.biShift (0,e)).obj A) (q-1))) at b

 155   obtain ⟨ε,_,_⟩ := quotient_shift_exists model.quotients.coherence (0,e) (q-1) A

 167   have hbf := map_preserves_filtration (model.convergentMap

 170     (t-s,t+e) (s+1) (b≫I.hom) hbottom'

 173     simpa only [FiltrationAtLeast,Category.assoc,Iso.hom_inv_id,Category.comp_id] using hbf

 183     (X : ClassicalObject) (q : ℕ) (hq : 0 < q) (e s t : ℤ)
 184     (x : Ambient model X (s,t)) (hx : x ∈ cycles model X q (s,t))

 189       (model.firstQuotient X e s t).symm x) :
 190     Detects (model.convergence (.quotient q (.shift (0,e) (.nu X))))
 191       (s,t,t+e) (regradeE2 (by simp) (finiteNuLabel model X q e s t 0 x)) a := by
 192   obtain ⟨v,hv⟩ := shifted_finite_source_exists X q hq e s t x hx
 193   have hv := representative_transport (show (s,t,t+e-(0:ℕ))=(s,t,t+e) by simp) _ _ hv

 196   haveI : Epi (F.toAssociatedGraded s (t-s,t+e)) := inferInstanceAs (Epi (cokernel.π _))
 197   obtain ⟨ar,har⟩ := (ModuleCat.epi_iff_surjective (F.toAssociatedGraded s (t-s,t+e))).mp
 198     inferInstance (((model.convergence (.quotient q (.shift (0,e) (.nu X)))).identification
 199       (s,t,t+e)).hom _)
 200   let a0 : BiHom (t-s) (t+e) (XModLambdaN
 201       ((SyntheticCategory.biShift (0,e)).obj
 202         (model.nu.functor.obj (X.obj model.auxiliary))) q) :=
 203     (F.F s (t-s,t+e)).arrow ar

 206     exact ⟨_,hv,ar,rfl,har.symm⟩

 208   have hd := shifted_finite_first_kernel_filtration X q hq e s t (a-a0) (by

 210   exact detects_of_high_difference (.quotient q (.shift (0,e) (.nu X)))

 213 /-- Every bottom-weight representative in a positively normalized finite
 214 source has an actual classical cycle label, with no representative assumption. -/
 215 private theorem positive_finite_source_label_exists
 216     (X : ClassicalObject) (q : ℕ) (hq : 0 < q) (s t : ℤ)
 217     (a : BiHom (t-s) (t+1)
 218       (XModLambdaN ((SyntheticCategory.biShift (0,1)).obj
 219         (model.nu.functor.obj (X.obj model.auxiliary))) q)) :
 220     ∃ x : Ambient model X (s,t), x ∈ cycles model X q (s,t) ∧
 221       Detects (model.convergence (.quotient q (.shift (0,1) (.nu X))))
 222         (s,t,t+1) (regradeE2 (by simp) (finiteNuLabel model X q 1 s t 0 x)) a := by
 223   letI := model.exactness.shiftCommShift (0,1)
 224   letI := model.exactness.shiftExact (0,1)
 225   let A := model.nu.functor.obj (X.obj model.auxiliary)
 226   obtain ⟨ε₁,hε₁,_⟩ := quotient_shift_exists model.quotients.coherence (0,1) 1 A
 227   obtain ⟨εq,hεq,_⟩ := quotient_shift_exists model.quotients.coherence (0,1) q A
 228   let a0 := quotientUnshift X (0,1) (t-s,t) q εq (homotopyRegrade (by simp) rfl a)
 229   obtain ⟨x,hx⟩ := quotient_first_label_exists X q hq s t a0
 230   have hcycles := first_label_lift_cycles X q hq s t x a0 hx
 231   refine ⟨x,hcycles,shifted_finite_source_detects_of_first X q hq 1 s t x hcycles a ?_⟩
 232   have hunshift : quotientUnshift X (0,1) (t-s,t) 1 ε₁
 233       (homotopyRegrade (by simp) rfl (a≫(model.quotients.tower _).rho 1 q hq)) =
 234       quotientUnshift X (0,1) (t-s,t) 1 ε₁
 235         (homotopyRegrade (by simp) rfl ((model.firstQuotient X 1 s t).symm x)) := by
 236     rw [positive_first_desuspend X s t x ε₁ hε₁]
 237     rw [regrade_comp,←quotient_unshift_restriction X (0,1) (t-s,t) 1 q
 238       (by decide) hq ε₁ εq hε₁ hεq]
 239     exact hx
 240   have hh := congrArg (fun z => shiftHomotopyClass (0,1) (t-s,t) z≫ε₁.inv) hunshift
 241   simp only [shift_quotient_unshift] at hh
 242   have hinj : Function.Injective (homotopyRegrade
 243       (show t-s=((t-s,t)+(0,1)).1 by simp) (rfl : t+1=t+1)
 244       (X:=XModLambdaN ((SyntheticCategory.biShift (0,1)).obj A) 1)) := by
 245     intro a b hab
 246     have h := congrArg (homotopyRegrade (show ((t-s,t)+(0,1)).1=t-s by simp)
 247       (rfl : t+1=t+1)) hab
 248     simpa only [homotopy_regrade_inverse] using h
 249   exact hinj hh
