  23 /- A classical q-cycle need not lift to nu Y. Its image under this actual
  24 map nevertheless becomes permanent because the outgoing target groups
  25 vanish from page q+1 onward. This is the window needed for the octahedral
  26 defect d after conjugating the common suspension/weight shift. -/
  27 private theorem finite_cycle_image_infinity
  28     (Y X : ClassicalObject) (q r : ℕ) (hq : 0 < q) (hr : 0 < r)
  29     (a b s t : ℤ) (hgap : (r:ℤ) ≤ q+b-a)

  31         (model.nu.functor.obj (Y.obj model.auxiliary)) ⟶
  32       XModLambdaN ((SyntheticCategory.biShift (0,b)).obj
  33         (model.nu.functor.obj (X.obj model.auxiliary))) r)

  35     ∃ e, HasInfinityRepresentative
  36       (model.family.obj (XModLambdaN ((SyntheticCategory.biShift (0,b)).obj
  37         (model.nu.functor.obj (X.obj model.auxiliary))) r)) 2 (s,t,t+a-(0:ℕ))
  38       (familyPageMap model.family d 2 (s,t,t+a-(0:ℕ))
  39         (model.comparison.nuE2 Y a s t 0 y)) e := by
  40   let E := model.classical.obj (Y.obj model.auxiliary)
  41   have hpage : 2 ≤ (q+1:ℕ) := by omega

  43     apply (isCycle_iff_reaches model Y (q+1) (by omega) (s,t) y).mp
  44     refine ⟨by omega,?_⟩
  45     simpa using hy
  46   obtain ⟨yr,hyr⟩ := hy'
  47   have hdeg : (s,t)+E.diffDeg (q+1:ℕ) = (s+(q+1:ℕ),t+(q+1:ℕ)-1) := by
  48     rw [model.classical.differentialDegree]
  49     ext <;> first | rfl | (dsimp; omega)
  50   let zr := (E.d (q+1:ℕ) (s,t) ≫ eqToHom (congrArg (E.Page (q+1:ℕ)) hdeg)) yr
  51   obtain ⟨z,hz⟩ := FinitePageCalculus.page_has_initial_representative (by omega : (2:ℤ)≤(q+1:ℕ)) zr
  52   have hc : HasDifferential E (q+1:ℕ) (s,t) (s+(q+1:ℕ),t+(q+1:ℕ)-1) y z :=
  53     ⟨hdeg,yr,zr,hyr,hz,rfl⟩
  54   obtain ⟨_,ys,zs,hys,hzs,hds⟩ :=
  55     (KIP126.Literature.Standard.bhs_differentials Y a s t (q+1) 0 hpage y z).mp hc
  56   have hm := syntheticRep_map (model.family.map d) hys
  57   apply syntheticRep_infty _ (q+1:ℕ) (s,t,t+a-(0:ℕ)) _ _ hm
  58   intro n hn
  59   have hdeg' : (s,t,t+a-(0:ℕ))+(model.family.obj (XModLambdaN
  60       ((SyntheticCategory.biShift (0,b)).obj
  61         (model.nu.functor.obj (X.obj model.auxiliary))) r)).sequence.diffDeg n =
  62       (s+n,t+n-1,t+a-(0:ℕ)) := by
  63     change (s,t,t+a-(0:ℕ)) + (model.family.functor.obj _).diffDeg n = _
  64     rw [model.family.differentialDegree]
  65     simp only [syntheticAdamsRawShift,Prod.mk_add_mk,add_zero]
  66     ext <;> first | rfl | (dsimp; omega)
  67   have hs := shiftedFinitePageVanishing X r hr b n (s+n) (t+n-1) (t+a-(0:ℕ))
  68     (by omega) (Or.inr (by omega))
  69   have hsub : Subsingleton ((model.family.obj (XModLambdaN
  70       ((SyntheticCategory.biShift (0,b)).obj
  71         (model.nu.functor.obj (X.obj model.auxiliary))) r)).Page n
  72       ((s,t,t+a-(0:ℕ))+(model.family.obj (XModLambdaN
  73         ((SyntheticCategory.biShift (0,b)).obj
  74           (model.nu.functor.obj (X.obj model.auxiliary))) r)).sequence.diffDeg n)) := by
  75     rw [hdeg']
  76     exact hs
  77   ext v
  78   exact @Subsingleton.elim _ hsub _ _

 414 /- Preserve the actual raw d and t choices behind the common edge. In
 415 particular the very same t₀ in the edge equation has the lambda_i transpose
 416 factor needed in the m=0 argument. No arbitrary choice of λ-preimage is used. -/
 417 private theorem quotient_common_desuspended_boundary_defect_with_transpose
 418     {A B C₀ D₀ E : SyntheticSpectrum} (G : A ⟶ B)
 419     (i q : ℕ) (hi : 0 < i) (hq : 0 < q)
 420     (eC : XModLambdaN (HasFunctorialCofiber.cofib G) (i+q) ≅
 421       (SyntheticCategory.biShift (1,-(i:ℤ))).obj C₀)
 422     (eD : XModLambdaN (A⟦(1:ℤ)⟧) (i+q) ≅
 423       (SyntheticCategory.biShift (1,-(i:ℤ))).obj D₀)
 424     (epsilon : (SyntheticCategory.biShift (0,-(i:ℤ))).obj D₀ ≅ E) :
 425     let F := SyntheticCategory.biShift (Syn:=SyntheticSpectrum) (0,-(i:ℤ))
 426     let ff := SyntheticCategory.biShift_fullyFaithful (Syn:=SyntheticSpectrum) (1,-(i:ℤ))
 427     let L := (model.quotients.tower B).triangle hi (show i < i+q by omega)
 428     let u := F.map (XModLambdaN.map G (i+q-i))
 429     let f := u≫L.lambdaMap
 430     let h := HasFunctorialCofiber.cofibδ G
 431     ∀ (φ : HasFunctorialCofiber.cofib f ⟶ XModLambdaN (HasFunctorialCofiber.cofib G) (i+q))
 432       (K : (F.obj (XModLambdaN A (i+q-i)))⟦(1:ℤ)⟧ ⟶ XModLambdaN (A⟦(1:ℤ)⟧) (i+q)),
 433       HasFunctorialCofiber.cofibι f≫φ=XModLambdaN.map (HasFunctorialCofiber.cofibι G) (i+q) →
 434       HasFunctorialCofiber.cofibδ f≫K=φ≫XModLambdaN.map h (i+q) →
 435       let K₀ := ff.preimage ((biSuspensionIso (i:ℤ) (XModLambdaN A (i+q-i))).hom ≫ K ≫ eD.hom)
 436       let J₀ := ff.preimage ((biSuspensionIso (i:ℤ) A).hom ≫
 437         (shiftFunctor SyntheticSpectrum (1:ℤ)).map (lambdaPow i A) ≫
 438           XModLambdaN.incl (A⟦(1:ℤ)⟧) (i+q) ≫ eD.hom)
 439       let h₀ := ff.preimage (eC.inv ≫ XModLambdaN.map h (i+q) ≫ eD.hom)
 440       ∃ (d : (F.obj A)⟦(1:ℤ)⟧ ⟶ XModLambdaN (HasFunctorialCofiber.cofib G) (i+q))
 441         (raw : B⟦(1:ℤ)⟧ ⟶ XModLambdaN (A⟦(1:ℤ)⟧) (i+q)),
 442         let d₀ : A ⟶ C₀ := ff.preimage ((biSuspensionIso (i:ℤ) A).hom ≫ d ≫ eC.hom)
 443         let t₀ : B ⟶ D₀ := ff.preimage ((biSuspensionIso (i:ℤ) B).hom ≫
 444           (shiftFunctor SyntheticSpectrum (1:ℤ)).map (lambdaPow i B) ≫ raw ≫ eD.hom)
 445         (XModLambdaN.incl A (i+q-i) ≫ K₀ - J₀ = d₀≫h₀ - G≫t₀) ∧
 446         ∃ z : B ⟶ E, (SyntheticCategory.biShift (0,-(i:ℤ))).map t₀ ≫ epsilon.hom =
 447           lambdaPow i B ≫ z := by
 448   dsimp only
 449   intro φ K hφ hK
 450   obtain ⟨d,raw,hedge⟩ := quotient_unquotiented_boundary_defect G i q hi hq φ K hφ hK
 451   refine ⟨d,raw,?_,?_⟩
 452   · let ff := SyntheticCategory.biShift_fullyFaithful (Syn:=SyntheticSpectrum) (1,-(i:ℤ))
 453     apply ff.map_injective
 454     simp only [Functor.map_sub,Functor.map_comp,Functor.FullyFaithful.map_preimage]
 455     rw [←Category.assoc _ (biSuspensionIso (i:ℤ) (XModLambdaN A (i+q-i))).hom,
 456       biSuspensionIso_natural]
 457     simp only [Category.assoc,Iso.hom_inv_id_assoc]
 458     have hh := congrArg (fun z => (biSuspensionIso (i:ℤ) A).hom ≫ z ≫ eD.hom) hedge
 459     simp only [Preadditive.comp_sub,Preadditive.sub_comp,Category.assoc] at hh
 460     rw [common_shift_residual_factor G i (raw≫eD.hom)] at hh
 461     exact hh
 462   · exact common_desuspended_residual_transpose i eD epsilon raw
