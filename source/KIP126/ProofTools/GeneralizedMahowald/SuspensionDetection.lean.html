  25 private theorem quotient_unshift_preserves_filtration
  26     (X : ClassicalObject) (δ p : ℤ × ℤ) (q : ℕ)
  27     (ε : XModLambdaN ((SyntheticCategory.biShift δ).obj
  28         (model.nu.functor.obj (X.obj model.auxiliary))) q ≅
  29       (SyntheticCategory.biShift δ).obj
  30         (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) q))
  31     (s : ℤ)
  32     (a : BiHom (p+δ).1 (p+δ).2
  33       (XModLambdaN ((SyntheticCategory.biShift δ).obj
  34         (model.nu.functor.obj (X.obj model.auxiliary))) q))
  35     (ha : FiltrationAtLeast model.syntheticTower s a) :
  36     FiltrationAtLeast model.syntheticTower s (quotientUnshift X δ p q ε a) := by
  37   let E := (essShift δ).filtered (.quotient q (.nu X)) p
  38   have hv := map_preserves_filtration
  39     (model.convergentMap (.quotient q (.shift δ (.nu X)))
  40       (.shift δ (.quotient q (.nu X))) ε.hom) (p+δ) s a ha
  41   obtain ⟨b,hb⟩ := filtration_representative_exists model.syntheticTower _ (p+δ) s
  42     (a ≫ ε.hom) hv
  43   have he : E.ambient.hom (quotientUnshift X δ p q ε a) = a ≫ ε.hom := by
  44     rw [(essShift δ).ambient_eq]
  45     have h := congrArg (fun z => z ≫ ε.hom) (shift_quotient_unshift X δ p q ε a)
  46     simpa only [Category.assoc,Iso.inv_hom_id,Category.comp_id] using h
  47   have hi : E.ambient.inv (a ≫ ε.hom) = quotientUnshift X δ p q ε a := by
  48     rw [←he]
  49     exact E.ambient.toLinearEquiv.symm_apply_apply _
  50   have hc := ConcreteCategory.congr_hom (E.symm.level_arrow s) b
  51   change ((model.syntheticTower.filtration _).F s p).arrow ((E.level s).inv b) =
  52     E.ambient.inv (((model.syntheticTower.filtration _).F s (p+δ)).arrow b) at hc
  53   erw [hb,hi] at hc
  54   have hm : quotientUnshift X δ p q ε a ∈ ModuleCat.subobjectModule _
  55       ((model.syntheticTower.filtration _).F s p) := ⟨(E.level s).inv b,hc⟩
  56   simpa only [SyntheticTowerPresentation.filtration,OrderIso.apply_symm_apply,
  57     FiltrationAtLeast] using hm

  59 private theorem detects_of_high_difference
  60     (A : SyntheticObject) (p : Tridegree)
  61     (x : (model.family.obj (A.obj model.nu model.auxiliary)).E₂ p)
  62     (a b : BiHom (p.2.1-p.1) p.2.2 (A.obj model.nu model.auxiliary))
  63     (ha : Detects (model.convergence A) p x a)
  64     (hd : FiltrationAtLeast model.syntheticTower (p.1+1) (b-a)) :
  65     Detects (model.convergence A) p x b := by
  66   have hd' := model.syntheticTower.filtrationSubmodule_antitone
  67     (A.obj model.nu model.auxiliary) (p.2.1-p.1,p.2.2) (show p.1 ≤ p.1+1 by omega) hd
  68   obtain ⟨d,hd⟩ := filtration_representative_exists model.syntheticTower _
  69     (p.2.1-p.1,p.2.2) p.1 (b-a) hd'
  70   have hz : Detects (model.convergence A) p 0 (b-a) := by
  71     refine ⟨0,HasInfinityRepresentative.zero _ 2 (by omega) p,d,hd,?_⟩
  72     rw [map_zero]
  73     symm
  74     apply ((model.syntheticTower.filtration _).toAssociatedGraded_eq_zero_iff _ _ _).mpr
  75     rw [hd]
  76     simpa only [SyntheticTowerPresentation.filtration,OrderIso.apply_symm_apply,
  77       FiltrationAtLeast] using ‹FiltrationAtLeast model.syntheticTower (p.1+1) (b-a)›
  78   simpa only [add_zero,add_sub_cancel] using detects_add (model.convergence A) p ha hz

  95 private theorem finite_detected_target_bottom_lift
  96     (X : ClassicalObject) (q k : ℕ) (hkq : k<q) (s t : ℤ)
  97     (x : Ambient model X (s,t))
  98     (a : BiHom (t-s) (t-k) (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) q))
  99     (ha : Detects (model.convergence (.quotient q (.nu X))) (s,t,t-k)
 100       (finiteTargetLabel model X q s t k x) a) :
 101     ∃ b : BiHom (t-s) t (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) (q-k)),
 102       b ≫ (model.quotients.tower _).rho 1 (q-k) (by omega) =
 103         firstLabel model X s t x ≫ XModLambdaN.map (SyntheticCategory.biShift_zero.hom.app _) 1 := by
 104   obtain ⟨e,he,_⟩ := ha
 105   have hlabel : nuWeightLabel model X s t (t-k) (by omega) x =
 106       targetNuLabel model X s t k x := by
 107     have hk' : (t-(t-k)).toNat = k := by omega
 108     have transport (a b : ℕ) (hab : a=b)
 109         (h : (s,t,t-a)=(s,t,t-b)) :
 110         regradeE2 h (targetNuLabel model X s t a x) = targetNuLabel model X s t b x := by
 111       subst b
 112       rfl
 113     exact transport _ k hk' _
 114   have hh := KIP126.Literature.Standard.eInfinity_finite_labels X q (by omega)
 115     s t (t-k) (by omega) x e
 116   rw [hlabel] at hh
 117   have hx : x ∈ cycles model X (q-k:ℕ) (s,t) := by
 118     have hx := (hh.mp he).1
 119     convert hx using 1
 120     congr 1
 121     omega
 122   obtain ⟨b,hb⟩ := (KIP126.Literature.Standard.bhs_finite_lift X (q-k) (by omega) s t x).mp
 123     ((isCycle_iff_reaches model X ((q-k:ℕ)+1) (by omega) (s,t) x).mp
 124       ⟨by omega,by simpa using hx⟩)
 125   let f : nuZero model X ⟶ model.nu.functor.obj (X.obj model.auxiliary) :=
 126     SyntheticCategory.biShift_zero.hom.app _
 127   refine ⟨b ≫ XModLambdaN.map f (q-k),?_⟩
 128   rw [Category.assoc,←(model.quotients.natural f).rho_naturality (by omega),←Category.assoc,hb]

 132     {A : SyntheticSpectrum} (a b : BiHom m w A) :

 142 private theorem suspension_unshift_high_difference
 143     (X : ClassicalObject) (q k : ℕ) (s t : ℤ)
 144     (εq : XModLambdaN ((SyntheticCategory.biShift (1,1)).obj
 145         (model.nu.functor.obj (X.obj model.auxiliary))) q ≅
 146       (SyntheticCategory.biShift (1,1)).obj
 147         (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) q))
 148     (a b : BiHom ((t+1)-s) (t+1-k)
 149       (XModLambdaN (model.nu.functor.obj ((ClassicalObject.shift 1 X).obj model.auxiliary)) q))
 150     (hdiff : FiltrationAtLeast model.syntheticTower (s+1) (a-b)) :
 151     FiltrationAtLeast model.syntheticTower (s+1)
 152       (quotientUnshift X (1,1) (t-s,t-k) q εq
 153         (homotopyRegrade (by dsimp; omega) (by dsimp; omega)
 154           (a ≫ XModLambdaN.map (model.nu.suspensionIso (X.obj model.auxiliary)).hom q)) -
 155        quotientUnshift X (1,1) (t-s,t-k) q εq
 156         (homotopyRegrade (by dsimp; omega) (by dsimp; omega)
 157           (b ≫ XModLambdaN.map (model.nu.suspensionIso (X.obj model.auxiliary)).hom q))) := by
 158   let f := XModLambdaN.map (model.nu.suspensionIso (X.obj model.auxiliary)).hom q
 159   have hm := map_preserves_filtration
 160     (model.convergentMap (.quotient q (.nu (.shift 1 X))) (.quotient q (.shift (1,1) (.nu X))) f)
 161     ((t+1)-s,t+1-k) (s+1) _ hdiff
 162   have hr := homotopy_regrade_filtration
 163     (show (t+1)-s=((t-s,t-k)+(1,1)).1 by dsimp; omega)
 164     (show t+1-k=((t-s,t-k)+(1,1)).2 by dsimp; omega) (s+1) _ hm
 165   have hu := quotient_unshift_preserves_filtration X (1,1) (t-s,t-k) q εq (s+1) _ hr
 166   simp only [Preadditive.sub_comp,homotopy_regrade_sub] at hu
 167   erw [quotient_unshift_sub] at hu
 168   exact hu

 170 private theorem suspension_finite_bottom_detects
 171     (X : ClassicalObject) (q : ℕ) (hq : 0<q) (s t : ℤ)
 172     (x : Ambient model X (s,t)) (sx : Ambient model (.shift 1 X) (s,t+1))
 173     (ε₁ : XModLambdaN ((SyntheticCategory.biShift (1,1)).obj
 174         (model.nu.functor.obj (X.obj model.auxiliary))) 1 ≅
 175       (SyntheticCategory.biShift (1,1)).obj
 176         (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) 1))
 177     (εq : XModLambdaN ((SyntheticCategory.biShift (1,1)).obj
 178         (model.nu.functor.obj (X.obj model.auxiliary))) q ≅
 179       (SyntheticCategory.biShift (1,1)).obj
 180         (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) q))
 181     (hε₁ : XModLambdaN.incl _ 1 ≫ ε₁.hom =
 182       (SyntheticCategory.biShift (1,1)).map (XModLambdaN.incl _ 1))
 183     (hεq : XModLambdaN.incl _ q ≫ εq.hom =
 184       (SyntheticCategory.biShift (1,1)).map (XModLambdaN.incl _ q))
 185     (hsx : (model.classicalSuspension X).DesuspendsClass s (t+1) sx (by simpa using x))
 186     (a : BiHom ((t+1)-s) (t+1)
 187       (XModLambdaN (model.nu.functor.obj ((ClassicalObject.shift 1 X).obj model.auxiliary)) q))
 188     (ha : Detects (model.convergence (.quotient q (.nu (.shift 1 X)))) (s,t+1,t+1)
 189       (regradeE2 (by simp) (finiteTargetLabel model (.shift 1 X) q s (t+1) 0 sx)) a) :
 190     Detects (model.convergence (.quotient q (.nu X))) (s,t,t)
 191       (regradeE2 (by simp) (finiteTargetLabel model X q s t 0 x))
 192       (quotientUnshift X (1,1) (t-s,t) q εq
 193         (homotopyRegrade (by dsimp; omega) rfl
 194           (a ≫ XModLambdaN.map (model.nu.suspensionIso (X.obj model.auxiliary)).hom q))) := by
 195   have ha' : Detects (model.convergence (.quotient q (.nu (.shift 1 X)))) (s,t+1,t+1-(0:ℕ))
 196       (finiteTargetLabel model (.shift 1 X) q s (t+1) 0 sx)
 197       (homotopyRegrade rfl (by simp) a) := by
 198     convert detects_transport (show (s,t+1,t+1)=(s,t+1,t+1-(0:ℕ)) by simp) _ _ ha using 1
 199     have inv {A : SyntheticAdamsSS} {i j : Tridegree} (h : i=j) (y : A.E₂ i) :
 200         regradeE2 h.symm (regradeE2 h y) = y := by subst j; rfl
 201     exact (inv _ _).symm
 202   obtain ⟨b,hb⟩ := finite_detected_target_bottom_lift (.shift 1 X) q 0 hq s (t+1) sx _ ha'
 203   simp only [Nat.sub_zero] at b hb
 204   have hbX := suspension_finite_target_desuspend X q hq s t x sx ε₁ εq hε₁ hεq hsx b hb
 205   have hbS := first_label_detects_unshifted (.shift 1 X) q hq s (t+1) sx b hb
 206   have hd := detects_sub_filtration (model.convergence (.quotient q (.nu (.shift 1 X))))
 207     (s,t+1,t+1) ha hbS
 208   let f := XModLambdaN.map (model.nu.suspensionIso (X.obj model.auxiliary)).hom q
 209   have hm := map_preserves_filtration
 210     (model.convergentMap (.quotient q (.nu (.shift 1 X))) (.quotient q (.shift (1,1) (.nu X))) f)
 211     ((t+1)-s,t+1) (s+1) _ hd
 212   have hr := homotopy_regrade_filtration
 213     (show (t+1)-s=((t-s,t)+(1,1)).1 by dsimp; omega) rfl (s+1) _ hm
 214   have hu := quotient_unshift_preserves_filtration X (1,1) (t-s,t) q εq (s+1) _ hr
 215   simp only [Preadditive.sub_comp,homotopy_regrade_sub] at hu
 216   erw [quotient_unshift_sub] at hu
 217   exact detects_of_high_difference (.quotient q (.nu X)) (s,t,t) _ _ _ hbX.2 hu

 220 private theorem suspension_finite_positive_detects
 221     (X : ClassicalObject) (q k : ℕ) (hk : 0<k) (hkq : k<q) (s t : ℤ)
 222     (x : Ambient model X (s,t)) (sx : Ambient model (.shift 1 X) (s,t+1))
 223     (ε₁ : XModLambdaN ((SyntheticCategory.biShift (1,1)).obj
 224         (model.nu.functor.obj (X.obj model.auxiliary))) 1 ≅
 225       (SyntheticCategory.biShift (1,1)).obj
 226         (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) 1))
 227     (εb : XModLambdaN ((SyntheticCategory.biShift (1,1)).obj
 228         (model.nu.functor.obj (X.obj model.auxiliary))) (q-k) ≅
 229       (SyntheticCategory.biShift (1,1)).obj
 230         (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) (q-k)))
 231     (εq : XModLambdaN ((SyntheticCategory.biShift (1,1)).obj
 232         (model.nu.functor.obj (X.obj model.auxiliary))) q ≅
 233       (SyntheticCategory.biShift (1,1)).obj
 234         (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) q))
 235     (hε₁ : XModLambdaN.incl _ 1 ≫ ε₁.hom =
 236       (SyntheticCategory.biShift (1,1)).map (XModLambdaN.incl _ 1))
 237     (hεb : XModLambdaN.incl _ (q-k) ≫ εb.hom =
 238       (SyntheticCategory.biShift (1,1)).map (XModLambdaN.incl _ (q-k)))
 239     (hεq : XModLambdaN.incl _ q ≫ εq.hom =
 240       (SyntheticCategory.biShift (1,1)).map (XModLambdaN.incl _ q))
 241     (hsx : (model.classicalSuspension X).DesuspendsClass s (t+1) sx (by simpa using x))
 242     (a : BiHom ((t+1)-s) (t+1-k)
 243       (XModLambdaN (model.nu.functor.obj ((ClassicalObject.shift 1 X).obj model.auxiliary)) q))
 244     (ha : Detects (model.convergence (.quotient q (.nu (.shift 1 X)))) (s,t+1,t+1-k)
 245       (finiteTargetLabel model (.shift 1 X) q s (t+1) k sx) a) :
 246     Detects (model.convergence (.quotient q (.nu X))) (s,t,t-k)
 247       (finiteTargetLabel model X q s t k x)
 248       (quotientUnshift X (1,1) (t-s,t-k) q εq

 251   obtain ⟨b,hb⟩ := finite_detected_target_bottom_lift (.shift 1 X) q k hkq s (t+1) sx a ha
 252   have hbX := suspension_finite_target_desuspend X (q-k) (by omega) s t x sx

 256   have hdiff := detects_sub_filtration (model.convergence (.quotient q (.nu (.shift 1 X))))
 257     (s,t+1,t+1-k) ha hlS
 258   have hu := suspension_unshift_high_difference X q k s t εq _ _ hdiff
 259   rw [suspension_finite_lambda_unshift X q k hk hkq s t εb εq hεb hεq b] at hu
 260   exact detects_of_high_difference (.quotient q (.nu X)) (s,t,t-k) _ _ _ hlX hu

 262 /-- Actual suspension detection in every retained finite lambda layer.
 263 Only the comparison used by the caller is supplied; all auxiliary quotient
 264 comparisons are selected internally. -/
 265 private theorem suspension_finite_detects
 266     (X : ClassicalObject) (q k : ℕ) (hkq : k<q) (s t : ℤ)
 267     (x : Ambient model X (s,t)) (sx : Ambient model (.shift 1 X) (s,t+1))
 268     (εq : XModLambdaN ((SyntheticCategory.biShift (1,1)).obj
 269         (model.nu.functor.obj (X.obj model.auxiliary))) q ≅
 270       (SyntheticCategory.biShift (1,1)).obj
 271         (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) q))
 272     (hεq : XModLambdaN.incl _ q ≫ εq.hom =
 273       (SyntheticCategory.biShift (1,1)).map (XModLambdaN.incl _ q))
 274     (hsx : (model.classicalSuspension X).DesuspendsClass s (t+1) sx (by simpa using x))
 275     (a : BiHom ((t+1)-s) (t+1-k)
 276       (XModLambdaN (model.nu.functor.obj ((ClassicalObject.shift 1 X).obj model.auxiliary)) q))
 277     (ha : Detects (model.convergence (.quotient q (.nu (.shift 1 X)))) (s,t+1,t+1-k)
 278       (finiteTargetLabel model (.shift 1 X) q s (t+1) k sx) a) :
 279     Detects (model.convergence (.quotient q (.nu X))) (s,t,t-k)
 280       (finiteTargetLabel model X q s t k x)
 281       (quotientUnshift X (1,1) (t-s,t-k) q εq
 282         (homotopyRegrade (by dsimp; omega) (by dsimp; omega)
 283           (a ≫ XModLambdaN.map (model.nu.suspensionIso (X.obj model.auxiliary)).hom q))) := by
 284   letI := model.exactness.shiftCommShift (1,1)
 285   letI := model.exactness.shiftExact (1,1)
 286   obtain ⟨ε₁,hε₁,_⟩ := quotient_shift_exists model.quotients.coherence (1,1) 1
 287     (model.nu.functor.obj (X.obj model.auxiliary))
 288   by_cases hk : k=0
 289   · subst k
 290     have ha0 := detects_transport (show (s,t+1,t+1-(0:ℕ))=(s,t+1,t+1) by simp) _ _ ha
 291     have h0 := suspension_finite_bottom_detects X q hkq s t x sx ε₁ εq hε₁ hεq hsx _ ha0
 292     have hback := detects_transport (show (s,t,t)=(s,t,t-(0:ℕ)) by simp) _ _ h0
 293     convert hback using 1
 294     · have inv {A : SyntheticAdamsSS} {i j : Tridegree} (h : i=j) (y : A.E₂ i) :
 295           regradeE2 h.symm (regradeE2 h y) = y := by subst j; rfl
 296       exact (inv _ _).symm
 297     · have unshift_weight (m w w' : ℤ) (hw : w=w')
 298           (b : BiHom ((m,w)+(1,1)).1 ((m,w)+(1,1)).2
 299             (XModLambdaN ((SyntheticCategory.biShift (1,1)).obj
 300               (model.nu.functor.obj (X.obj model.auxiliary))) q)) :
 301           homotopyRegrade rfl hw (quotientUnshift X (1,1) (m,w) q εq b) =
 302             quotientUnshift X (1,1) (m,w') q εq
 303               (homotopyRegrade rfl (congrArg (fun z => z+1) hw) b) := by
 304         subst w'; rfl
 305       have twice {A : SyntheticSpectrum} {m m' m'' w w' w'' : ℤ}
 306           (hm : m=m') (hm' : m'=m'') (hw : w=w') (hw' : w'=w'')
 307           (b : BiHom m w A) :
 308           homotopyRegrade hm' hw' (homotopyRegrade hm hw b) =
 309             homotopyRegrade (hm.trans hm') (hw.trans hw') b := by
 310         subst m'; subst m''; subst w'; subst w''; rfl
 311       rw [unshift_weight]
 312       simp only [←regrade_comp,twice]
 313 
 314 
 315   · obtain ⟨εb,hεb,_⟩ := quotient_shift_exists model.quotients.coherence (1,1) (q-k)
 316       (model.nu.functor.obj (X.obj model.auxiliary))
 317     exact suspension_finite_positive_detects X q k (by omega) hkq s t x sx ε₁ εb εq
 318       hε₁ hεb hεq hsx a ha
