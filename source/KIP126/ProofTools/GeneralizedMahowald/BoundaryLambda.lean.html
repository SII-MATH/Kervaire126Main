  26 theorem finite_target_label_iff (X : ClassicalObject) (q : ℕ) (hq : 0 < q)
  27     (s t : ℤ) (k : ℕ) (hk : k < q)
  28     (y : cycles model X ((q:ℤ)-t+(t-k)) (s,t))
  29     (e : ((model.family.quotient (model.nu.functor.obj (X.obj model.auxiliary)) q).sequence.ssData
  30       (s,t,t-k)).eInfty) :
  31     HasInfinityRepresentative _ 2 (s,t,t-k) (finiteTargetLabel model X q s t k y.val) e ↔
  32       KIP126.Literature.Standard.infinityFiniteWindow X q hq (s,t) (t-k) (by omega) e =
  33         subquotientClass (cycles model X ((q:ℤ)-t+(t-k)) (s,t))
  34           (boundaries model X (1+t-(t-k)) (s,t)) y := by
  35   have hlabel : nuWeightLabel model X s t (t-k) (by omega) y.val =
  36       targetNuLabel model X s t k y.val := by
  37     have hk' : (t-(t-k)).toNat = k := by omega
  38     have transport (a b : ℕ) (hab : a=b)
  39         (h : (s,t,t-a)=(s,t,t-b)) :
  40         regradeE2 h (targetNuLabel model X s t a y.val) = targetNuLabel model X s t b y.val := by
  41       subst b
  42       rfl
  43     exact transport _ k hk' _
  44   have h := KIP126.Literature.Standard.eInfinity_finite_labels X q hq s t (t-k)
  45     (by omega) y.val e
  46   rw [hlabel] at h
  47   change HasInfinityRepresentative _ 2 (s,t,t-k) (finiteTargetLabel model X q s t k y.val) e ↔ _ at h
  48   exact h.trans ⟨fun ⟨_,he⟩ => he, fun he => ⟨y.property,he⟩⟩

  51     (A : SyntheticObject) (X : ClassicalObject) (r : ℕ) (hr : 0 < r)

  54       (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) r))

  56     (a : BiHom (t-s) w (A.obj model.nu model.auxiliary))
  57     (ha : Detects (model.convergence A) (s,t,w) x a)
  58     (e : ((model.family.obj (A.obj model.nu model.auxiliary)).sequence.ssData (s,t,w)).eInfty)
  59     (he : HasInfinityRepresentative _ 2 (s,t,w) x e)

  66   exact shifted_finite_high_filtration_zero X r hr i (t-s) w (s+1) (by omega) (a ≫ h) hf

  77     (model.family.map g).eInftyMap (p.1,p.2,w-i)

  98 theorem finite_target_boundary_lambda_zero (X : ClassicalObject)
  99     (r i : ℕ) (hi : 0 < i) (hir : i < r) (s t : ℤ)
 100     (y : cycles model X ((r-i:ℕ)-t+(t-(r-i-1:ℕ))) (s,t))
 101     (hy : y.val ∈ boundaries model X r (s,t))
 102     (b : BiHom (t-s) (t-(r-i-1:ℕ))
 103       (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) (r-i)))
 104     (hb : Detects (model.convergence (.quotient (r-i) (.nu X)))
 105       (s,t,t-(r-i-1:ℕ)) (finiteTargetLabel model X (r-i) s t (r-i-1) y.val) b) :
 106     (SyntheticCategory.biShift (0,-(i:ℤ))).map b ≫
 107       ((model.quotients.tower (model.nu.functor.obj (X.obj model.auxiliary))).triangle hi hir).lambdaMap = 0 := by
 108   let Q := XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) (r-i)
 109   let Y := XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) r
 110   let T := (model.quotients.tower (model.nu.functor.obj (X.obj model.auxiliary))).triangle hi hir
 111   obtain ⟨e,he,ha⟩ := hb
 112   have hwin : 0 ≤ t-(t-(r-i-1:ℕ)) ∧ t-(t-(r-i-1:ℕ)) < (r-i:ℕ) := by omega
 113   have hlabel := (finite_target_label_iff X (r-i) (by omega) s t (r-i-1) (by omega) y e).mp he
 114   have hk := KIP126.Literature.Standard.eInfinity_lambda_finite X r i hir (s,t)
 115     (t-(r-i-1:ℕ)) hwin e
 116   rw [hlabel] at hk
 117   have hk0 : KIP126.Literature.Standard.infinityFiniteLambda X r i hir (s,t)
 118       (t-(r-i-1:ℕ)) e = 0 := by
 119     apply (KIP126.Literature.Standard.infinityFiniteWindow X r (by omega)
 120       (s,t) ((t-(r-i-1:ℕ))-i) (by constructor <;> omega)).injective
 121     rw [map_zero,hk]
 122     unfold quotientMap
 123     refine quotient_map_class_eq_zero
 124       (cycles_antitone model X (s,t) (by omega))
 125       (boundaries_monotone model X (s,t) (by omega)) y ?_
 126     change y.val ∈ boundaries model X (1+t-((t-(r-i-1:ℕ))-i)) (s,t)
 127     convert hy using 1 <;> congr 1 <;> omega
 128   let ε : (SyntheticCategory.biShift (0,-(i:ℤ))).obj
 129       ((SyntheticCategory.biShift (0,(i:ℤ))).obj Y) ≅ Y :=
 130     (SyntheticCategory.biShift_comp (0,(i:ℤ)) (0,-(i:ℤ))).app Y ≪≫
 131       eqToIso (congrArg (fun k => (SyntheticCategory.biShift k).obj Y)
 132         (show (0,(i:ℤ))+(0,-(i:ℤ))=(0,0) by ext <;> simp)) ≪≫
 133       SyntheticCategory.biShift_zero.app Y
 134   let h : Q ⟶ (SyntheticCategory.biShift (0,(i:ℤ))).obj Y :=
 135     (SyntheticCategory.biShift_fullyFaithful (0,-(i:ℤ))).preimage (T.lambdaMap ≫ ε.inv)
 136   have hh : (SyntheticCategory.biShift (0,-(i:ℤ))).map h ≫ ε.hom = T.lambdaMap := by
 137     rw [show (SyntheticCategory.biShift (0,-(i:ℤ))).map h = T.lambdaMap ≫ ε.inv from
 138       (SyntheticCategory.biShift_fullyFaithful (0,-(i:ℤ))).map_preimage _,
 139       Category.assoc,ε.inv_hom_id,Category.comp_id]
 140   have hzero : (model.family.map h).eInftyMap (s,t,t-(r-i-1:ℕ)) e = 0 := by
 141     apply (weight_shift_lower_kernel_via_transpose (.quotient (r-i) (.nu X))
 142       (.shift (0,(i:ℤ)) (.quotient r (.nu X))) (.quotient r (.nu X))
 143       i (s,t) (t-(r-i-1:ℕ)) h ε T.lambdaMap hh e).mp
 144     simpa only [KIP126.Literature.Standard.infinityFiniteLambda,
 145       InfinityShiftCoordinates.finiteLambdaMap, dif_neg (Nat.ne_of_gt hi),
 146       KIP126.Literature.Standard.infinityShiftData, InfinityShiftCoordinates.shift,
 147       ModuleCat.comp_apply,T,SyntheticObject.obj,SyntheticAdamsFamily.quotient] using hk0
 148   have hb0 : b ≫ h = 0 := detected_map_zero_above_shifted_finite_top
 149     (.quotient (r-i) (.nu X)) X r (by omega) i s t (t-(r-i-1:ℕ)) (by omega)
 150     h (finiteTargetLabel model X (r-i) s t (r-i-1) y.val) b ⟨e,he,ha⟩ e he hzero
 151   exact (shifted_composite_zero_of_conjugate (i:ℤ) b h T.lambdaMap ε hh).mpr hb0
