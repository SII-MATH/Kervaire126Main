  17 private def quotientShiftAlongIso (β : ℤ×ℤ) (q : ℕ)
  18     {C A : SyntheticSpectrum} (e : C ≅ A)
  19     (ε : XModLambdaN ((SyntheticCategory.biShift β).obj A) q ≅
  20       (SyntheticCategory.biShift β).obj (XModLambdaN A q)) :
  21     XModLambdaN ((SyntheticCategory.biShift β).obj C) q ≅
  22       (SyntheticCategory.biShift β).obj (XModLambdaN C q) :=
  23   (model.quotients.functoriality.functor q).mapIso
  24       ((SyntheticCategory.biShift β).mapIso e) ≪≫ ε ≪≫
  25     (SyntheticCategory.biShift β).mapIso
  26       ((model.quotients.functoriality.functor q).mapIso e.symm)

  28 private theorem quotientShiftAlongIso_inclusion (β : ℤ×ℤ) (q : ℕ)
  29     {C A : SyntheticSpectrum} (e : C ≅ A)
  30     (ε : XModLambdaN ((SyntheticCategory.biShift β).obj A) q ≅
  31       (SyntheticCategory.biShift β).obj (XModLambdaN A q))
  32     (hε : XModLambdaN.incl _ q≫ε.hom =
  33       (SyntheticCategory.biShift β).map (XModLambdaN.incl _ q)) :
  34     XModLambdaN.incl _ q≫(quotientShiftAlongIso β q e ε).hom =
  35       (SyntheticCategory.biShift β).map (XModLambdaN.incl _ q) := by
  36   simp only [quotientShiftAlongIso,Iso.trans_hom,Functor.mapIso_hom,Iso.symm_hom,
  37     LambdaQuotientFunctoriality.functor]
  38   rw [←Category.assoc,←XModLambdaN.incl_naturality,Category.assoc,
  39     ←Category.assoc (XModLambdaN.incl _ q),hε]
  40   rw [←Functor.map_comp,←XModLambdaN.incl_naturality,Functor.map_comp,
  41     ←Category.assoc,←Functor.map_comp,Iso.hom_inv_id,CategoryTheory.Functor.map_id,Category.id_comp]
