  25     (eD : U ≅ (SyntheticCategory.biShift (1,-(i:ℤ))).obj V)

  27     (SyntheticCategory.biShift_fullyFaithful (Syn:=SyntheticSpectrum) (1,-(i:ℤ))).preimage
  28       ((biShiftAddIso (M-1,W+i) (1,-(i:ℤ)) (M,W)
  29         (by ext <;> dsimp; all_goals omega)).hom.app S_0_0 ≫
  30       commonCorrectedGeometry i M W eC c ≫ h ≫ eD.hom) =
  31     c ≫ (SyntheticCategory.biShift_fullyFaithful (Syn:=SyntheticSpectrum) (1,-(i:ℤ))).preimage

  33   apply (SyntheticCategory.biShift_fullyFaithful (Syn:=SyntheticSpectrum) (1,-(i:ℤ))).map_injective
  34   simp only [Functor.FullyFaithful.map_preimage,Functor.map_comp,commonCorrectedGeometry,
  35     Category.assoc,Iso.hom_inv_id_app_assoc]
