  29 /- Common desuspension preserves the second image of the very same
  30 geometric v. This consumes the actual K square and the original hvSecond. -/
  31 private theorem common_desuspended_geometric_second_image
  32     {A C₀ D₀ V Cr Dr : SyntheticSpectrum} (i : ℕ) (m w : ℤ)
  33     (eC : Cr ≅ (SyntheticCategory.biShift (1,-(i:ℤ))).obj C₀)
  34     (eD : Dr ≅ (SyntheticCategory.biShift (1,-(i:ℤ))).obj D₀)
  35     (φ : V ⟶ Cr) (δ : V ⟶ (((SyntheticCategory.biShift (0,-(i:ℤ))).obj A)⟦(1:ℤ)⟧))
  36     (K : (((SyntheticCategory.biShift (0,-(i:ℤ))).obj A)⟦(1:ℤ)⟧) ⟶ Dr)
  37     (h : Cr ⟶ Dr) (hK : φ≫h=δ≫K)
  38     (v : BiHom m w V) (ag : BiHom (m-1) (w+i) A)
  39     (hvSecond : biSuspensionHomEquiv m w i A (v≫δ)=ag) :
  40     let ff := SyntheticCategory.biShift_fullyFaithful (Syn:=SyntheticSpectrum) (1,-(i:ℤ))

  42       (by ext <;> dsimp; all_goals omega)).hom.app S_0_0 ≫ v ≫ φ ≫ eC.hom)
  43     let h₀ := ff.preimage (eC.inv≫h≫eD.hom)
  44     let K₀ := ff.preimage ((biSuspensionIso (i:ℤ) A).hom≫K≫eD.hom)
  45     c₀≫h₀=ag≫K₀ := by
  46   dsimp only
  47   let ff := SyntheticCategory.biShift_fullyFaithful (Syn:=SyntheticSpectrum) (1,-(i:ℤ))
  48   have hv : v≫δ=(biSuspensionHomEquiv m w i A).symm ag := by
  49     rw [←hvSecond]
  50     exact (AddEquiv.symm_apply_apply _ _).symm
  51   apply ff.map_injective
  52   simp only [Functor.map_comp,Functor.FullyFaithful.map_preimage,Category.assoc,
  53     Iso.hom_inv_id_assoc]
  54   rw [←Category.assoc φ h,hK]
  55   simp only [Category.assoc]
  56   rw [←Category.assoc v δ,hv]
  57   rw [←Category.assoc _ ((biSuspensionHomEquiv m w i A).symm ag),common_bi_susp_inverse]
  58   simp only [Category.assoc]

  66     (f : (SyntheticCategory.biShift (0,-(i:ℤ))).obj (XModLambdaN A j) ⟶ B)

  70     (hK : HasFunctorialCofiber.cofibδ f≫K=φ≫h)
  71     (v : BiHom M W (HasFunctorialCofiber.cofib f))
  72     (ag : BiHom (M-1) (W+i) (XModLambdaN A q))
  73     (hvSecond : biSuspensionHomEquiv M W i (XModLambdaN A q)
  74       (by simpa only [hj] using v≫HasFunctorialCofiber.cofibδ f)=ag) :
  75     let ff := SyntheticCategory.biShift_fullyFaithful (Syn:=SyntheticSpectrum) (1,-(i:ℤ))
  76     let c₀ := ff.preimage ((biShiftAddIso (M-1,W+i) (1,-(i:ℤ)) (M,W)
  77       (by ext <;> dsimp; all_goals omega)).hom.app S_0_0≫v≫φ≫eC.hom)

  80       (by simpa only [hj] using K)≫eD.hom)
