  22 private theorem zero_normalized_leading_nonzero (X Y : ClassicalObject)
  23     (f : X.obj model.auxiliary ⟶ Y.obj model.auxiliary)
  24     (he : normalizedExponent model.classicalTower f=0)
  25     (x : Ambient model X (3,17))
  26     (hnu : familyPageMap model.family (model.nu.functor.map f) 2 (3,17,17)
  27       (nuWeightLabel model X 3 17 17 (by omega) x)≠0) :
  28     familyPageMap model.family (model.normalizedMap X Y f).map 2
  29       (3,(17-3)+3,17+normalizedExponent model.classicalTower f)
  30       (infiniteSourceForExtension model X Y f 3 17 x)≠0 := by
  31   have core (k : ℕ) (hk : k=0)
  32       (M : (SyntheticCategory.biShift (0,(k:ℤ))).obj
  33         (model.nu.functor.obj (X.obj model.auxiliary)) ⟶
  34           model.nu.functor.obj (Y.obj model.auxiliary))
  35       (hM : lambdaToPositivePow k _≫M=model.nu.functor.map f) :
  36       familyPageMap model.family M 2 (3,(17-3)+3,17+k)
  37         (regradeE2 (by simp) (model.comparison.nuE2 X k 3 17 0 x))≠0 := by
  38     subst k
  39     rw [normalized_zero_map_form rfl M _ hM]
  40     simp only [eqToHom_refl,Category.id_comp]
  41     rw [←family_page_comp,ModuleCat.comp_apply]
  42     exact hnu
  43   exact core _ he _ (model.normalizedMap X Y f).factorization

  53 /-- All geometric inputs and original hypotheses are now supplied by the
  54 proved rotated-triangle modules. Only leading-image nonvanishing is explicit. -/
  55 private theorem mh2_candidate_refutation_of_bottom_nonzero
  56     (hbottom : familyPageMap model.family
  57       (model.normalizedMap .sphere .nuCofiber rotatedNuRouteTriangle.f).map 2
  58       (3,(17-3)+3,17+normalizedExponent model.classicalTower rotatedNuRouteTriangle.f)
  59       (infiniteSourceForExtension model .sphere .nuCofiber rotatedNuRouteTriangle.f 3 17
  60         mh2BoundaryClass) ≠ 0) :
  61     GeneralizedMahowaldLaw model rotatedNuRouteTriangle → False := by
  62   intro hLaw
  63   have he := rotated_nu_exponent_sum
  64   have htriangle : NormalizedTriangleCompatible model rotatedNuRouteTriangle he :=
  65     rotated_nu_normalized_distinguished
  66   obtain ⟨sx,hsusp,hh,hcross,hg⟩ := rotated_nu_remaining_premises
  67   have hef := rotated_nu_f_exponent
  68   have heg := rotated_nu_g_exponent
  69   have heh : normalizedExponent model.classicalTower rotatedNuRouteTriangle.h=1 := by
  70     rw [hef,heg] at he
  71     omega
  72   have hx : IsCycle model .sphere 1 (3,17) mh2BoundaryClass :=
  73     ⟨by decide,boundaries_le_cycles model .sphere (3,17) 2 1 mh2BoundaryClass_boundary⟩
  74   have hy : IsCycle model .nuCofiber 1 (4,18) 0 := ⟨by decide,Submodule.zero_mem _⟩
  75   have hxBar : IsCycle model rotatedNuRouteTriangle.Z 2 (1,16) 0 :=
  76     ⟨by decide,Submodule.zero_mem _⟩
  77   have hyBar : IsPermanent model rotatedNuRouteTriangle.Z (4,18) 0 := Submodule.zero_mem _
  78   have H := hLaw he htriangle 1 0 2 1 16
  79   dsimp only at H
  80   norm_num only [hef,heg,heh] at H
  81   obtain ⟨_,y',hy',_⟩ := H mh2BoundaryClass 0 0 0 sx
  82     (by decide) (by decide) (by decide) hx hy hxBar hyBar hsusp hh
  83     candidate_zero_differential (Or.inr hcross) hg
  84   exact nonzero_leading_image_no_positive_extension .sphere .nuCofiber
  85     rotatedNuRouteTriangle.f 3 1 3 17 (by decide) mh2BoundaryClass hbottom ⟨y',hy'⟩
