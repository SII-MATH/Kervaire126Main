  37       (model.convergence (.quotient 1 (.shift (0,e) (.nu X))))

  39       (fun k hk _ => shifted_quotient_eInfty_zero X 1 (by decide) e k (t-s+k) (t+e)

  43 private theorem shifted_finite_source_first
  44     (X : ClassicalObject) (q : ℕ) (hq : 0 < q) (e s t : ℤ)
  45     (x : Ambient model X (s,t))
  46     (a : BiHom (t-s) (t+e) (XModLambdaN
  47       ((SyntheticCategory.biShift (0,e)).obj (model.nu.functor.obj (X.obj model.auxiliary))) q))
  48     (ha : Detects (model.convergence (.quotient q (.shift (0,e) (.nu X)))) (s,t,t+e)
  49       (regradeE2 (by simp) (finiteNuLabel model X q e s t 0 x)) a) :
  50     a ≫ (model.quotients.tower _).rho 1 q hq = (model.firstQuotient X e s t).symm x := by
  51   have hrho := detects_map (.quotient q (.shift (0,e) (.nu X)))
  52     (.quotient 1 (.shift (0,e) (.nu X))) ((model.quotients.tower _).rho 1 q hq)
  53     (s,t,t+e) _ _ ha
  54   have hlabel : familyPageMap model.family ((model.quotients.tower
  55       ((SyntheticCategory.biShift (0,e)).obj (model.nu.functor.obj (X.obj model.auxiliary)))).rho 1 q hq)
  56       2 (s,t,t+e) (regradeE2 (by simp) (finiteNuLabel model X q e s t 0 x)) =
  57       regradeE2 (by simp) (finiteNuLabel model X 1 e s t 0 x) := by
  58     rw [←page_regrade]
  59     congr 1
  60     unfold finiteNuLabel
  61     have hc := congrArg (fun g => familyPageMap model.family g 2
  62       (s,t,t+e-(0:ℕ)))
  63       ((model.quotients.tower ((SyntheticCategory.biShift (0,e)).obj
  64         (model.nu.functor.obj (X.obj model.auxiliary)))).rho_quotient hq)
  65     rw [←family_page_comp] at hc
  66     exact ConcreteCategory.congr_hom hc ((model.comparison.nuE2 X e s t 0) x)
  67   have hrho' : Detects (model.convergence (.quotient 1 (.shift (0,e) (.nu X)))) (s,t,t+e)
  68       (regradeE2 (by simp) (finiteNuLabel model X 1 e s t 0 x))
  69       (a ≫ (model.quotients.tower _).rho 1 q hq) :=
  70     (congrArg (fun z => Detects (model.convergence (.quotient 1 (.shift (0,e) (.nu X))))
  71       (s,t,t+e) z (a ≫ (model.quotients.tower _).rho 1 q hq)) hlabel).mp hrho
  72   apply shifted_first_detection_unique X e s t _ _ _ hrho'
  73   have hh := model.comparisonCompatible.first_quotient X e s t x
  74   unfold finiteNuLabel
  75   rw [page_regrade]
  76   have regrade_cast {A : SyntheticAdamsSS} {i j : Tridegree} (h : i=j) (z : A.E₂ i) :
  77       regradeE2 h z = (congrArg (fun k => (A.E₂ k : Type)) h).mp z := by
  78     subst j
  79     rfl
  80   rw [regrade_cast]
  81   exact hh

  83 private def quotientUnshift (X : ClassicalObject) (δ p : ℤ × ℤ) (q : ℕ)
  84     (ε : XModLambdaN ((SyntheticCategory.biShift δ).obj
  85         (model.nu.functor.obj (X.obj model.auxiliary))) q ≅
  86       (SyntheticCategory.biShift δ).obj
  87         (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) q))
  88     (a : BiHom (p+δ).1 (p+δ).2 (XModLambdaN ((SyntheticCategory.biShift δ).obj
  89       (model.nu.functor.obj (X.obj model.auxiliary))) q)) :
  90     BiHom p.1 p.2 (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) q) :=
  91   (SyntheticCategory.biShift_fullyFaithful δ).preimage
  92     ((SyntheticCategory.biShift_comp p δ).hom.app S_0_0 ≫ a ≫ ε.hom)

 109 private theorem positive_first_desuspend
 110     (X : ClassicalObject) (s t : ℤ) (x : Ambient model X (s,t))
 111     (ε : XModLambdaN ((SyntheticCategory.biShift (0,1)).obj
 112         (model.nu.functor.obj (X.obj model.auxiliary))) 1 ≅
 113       (SyntheticCategory.biShift (0,1)).obj
 114         (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) 1))
 115     (hε : XModLambdaN.incl _ 1 ≫ ε.hom =
 116       (SyntheticCategory.biShift (0,1)).map (XModLambdaN.incl _ 1)) :
 117     quotientUnshift X (0,1) (t-s,t) 1 ε
 118       (homotopyRegrade (by simp) rfl ((model.firstQuotient X 1 s t).symm x)) =
 119       firstLabel model X s t x ≫ XModLambdaN.map (SyntheticCategory.biShift_zero.hom.app _) 1 := by
 120   have hf := firstQuotient_positive_shift X s t x ε hε

 124       homotopyRegrade rfl h a = (congrArg (fun v => BiHom m v A) h).mp a := by
 125     subst w'; rfl
 126   exact cast _ _ _ _ _ _

 152 private theorem positive_finite_source_desuspend
 153     (X : ClassicalObject) (q : ℕ) (hq : 0 < q) (s t : ℤ)
 154     (x : Ambient model X (s,t))
 155     (ε₁ : XModLambdaN ((SyntheticCategory.biShift (0,1)).obj
 156         (model.nu.functor.obj (X.obj model.auxiliary))) 1 ≅
 157       (SyntheticCategory.biShift (0,1)).obj
 158         (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) 1))
 159     (εq : XModLambdaN ((SyntheticCategory.biShift (0,1)).obj
 160         (model.nu.functor.obj (X.obj model.auxiliary))) q ≅
 161       (SyntheticCategory.biShift (0,1)).obj
 162         (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) q))
 163     (hε₁ : XModLambdaN.incl _ 1 ≫ ε₁.hom =
 164       (SyntheticCategory.biShift (0,1)).map (XModLambdaN.incl _ 1))
 165     (hεq : XModLambdaN.incl _ q ≫ εq.hom =
 166       (SyntheticCategory.biShift (0,1)).map (XModLambdaN.incl _ q))
 167     (a : BiHom (t-s) (t+1) (XModLambdaN
 168       ((SyntheticCategory.biShift (0,1)).obj (model.nu.functor.obj (X.obj model.auxiliary))) q))
 169     (ha : Detects (model.convergence (.quotient q (.shift (0,1) (.nu X)))) (s,t,t+1)
 170       (regradeE2 (by simp) (finiteNuLabel model X q 1 s t 0 x)) a) :
 171     let a₀ := quotientUnshift X (0,1) (t-s,t) q εq (homotopyRegrade (by simp) rfl a)
 172     a₀ ≫ (model.quotients.tower _).rho 1 q hq =
 173       firstLabel model X s t x ≫ XModLambdaN.map (SyntheticCategory.biShift_zero.hom.app _) 1 ∧
 174     Detects (model.convergence (.quotient q (.nu X))) (s,t,t)
 175       (regradeE2 (by simp) (finiteTargetLabel model X q s t 0 x)) a₀ := by
 176   dsimp only
 177   have hfirst := shifted_finite_source_first X q hq 1 s t x a ha
 178   have h : quotientUnshift X (0,1) (t-s,t) q εq (homotopyRegrade (by simp) rfl a) ≫
 179       (model.quotients.tower _).rho 1 q hq =
 180       firstLabel model X s t x ≫ XModLambdaN.map (SyntheticCategory.biShift_zero.hom.app _) 1 := by
 181     rw [quotient_unshift_restriction X (0,1) (t-s,t) 1 q (by decide) hq ε₁ εq hε₁ hεq,
 182       ←regrade_comp,hfirst,positive_first_desuspend X s t x ε₁ hε₁]
 183   exact ⟨h,first_label_detects_unshifted X q hq s t x _ h⟩

 223 private theorem positive_finite_source_unshifted
 224     (X : ClassicalObject) (q : ℕ) (hq : 0 < q) (s t : ℤ)
 225     (x : Ambient model X (s,t))
 226     (a : BiHom (t-s) (t+1) (XModLambdaN
 227       ((SyntheticCategory.biShift (0,1)).obj (model.nu.functor.obj (X.obj model.auxiliary))) q))
 228     (ha : Detects (model.convergence (.quotient q (.shift (0,1) (.nu X)))) (s,t,t+1)
 229       (regradeE2 (by simp) (finiteNuLabel model X q 1 s t 0 x)) a) :
 230     ∃ (εq : XModLambdaN ((SyntheticCategory.biShift (0,1)).obj
 231           (model.nu.functor.obj (X.obj model.auxiliary))) q ≅
 232         (SyntheticCategory.biShift (0,1)).obj
 233           (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) q)),
 234       XModLambdaN.incl _ q ≫ εq.hom =
 235         (SyntheticCategory.biShift (0,1)).map (XModLambdaN.incl _ q) ∧
 236       ∃ a₀ : BiHom (t-s) t (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) q),
 237         a₀ ≫ (model.quotients.tower _).rho 1 q hq =
 238           firstLabel model X s t x ≫ XModLambdaN.map (SyntheticCategory.biShift_zero.hom.app _) 1 ∧
 239         Detects (model.convergence (.quotient q (.nu X))) (s,t,t)
 240           (regradeE2 (by simp) (finiteTargetLabel model X q s t 0 x)) a₀ ∧
 241         homotopyRegrade (by simp) rfl (shiftHomotopyClass (0,1) (t-s,t) a₀) ≫ εq.inv = a := by
 242   letI := model.exactness.shiftCommShift (0,1)
 243   letI := model.exactness.shiftExact (0,1)
 244   obtain ⟨ε₁,hε₁,_⟩ := quotient_shift_exists model.quotients.coherence (0,1) 1
 245     (model.nu.functor.obj (X.obj model.auxiliary))
 246   obtain ⟨εq,hεq,_⟩ := quotient_shift_exists model.quotients.coherence (0,1) q
 247     (model.nu.functor.obj (X.obj model.auxiliary))
 248   have h := positive_finite_source_desuspend X q hq s t x ε₁ εq hε₁ hεq a ha
 249   refine ⟨εq,hεq,_,h.1,h.2,?_⟩
 250   rw [←regrade_comp,shift_quotient_unshift,homotopy_regrade_inverse]
