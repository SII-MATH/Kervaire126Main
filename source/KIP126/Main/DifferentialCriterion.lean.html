  43 /-- LWX Proposition 7.8: both the exclusive dichotomy and the full criterion. -/
  44 public theorem h6_square_differential_dichotomy_and_criterion :
  45     ((H6SquarePermanent ∧ ¬ H6SquareSupportsD12) ∨ (H6SquareSupportsD12 ∧ ¬ H6SquarePermanent)) ∧
  46       (H6SquareSupportsD12 ↔ Stem126D6Vanishes ∧ Theta5SquareNonzeroDetection ∧ EtaExtensionDetection) := by
  47   constructor
  48   · rcases permanence_or_d12 with hp | hd
  49     · exact Or.inl ⟨hp, fun hd => (d12_necessary hd).1 hp⟩
  50     · exact Or.inr ⟨hd, (d12_necessary hd).1⟩
  51   · constructor
  52     · intro hd
  53       exact ⟨(d12_necessary hd).2, d12_implies_square_and_eta hd⟩
  54     · rintro ⟨h3, h4, h5⟩
  55       exact conditions_imply_d12 h3 h4 h5
