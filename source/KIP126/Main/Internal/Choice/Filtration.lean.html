  15 private theorem theta_square_difference
  16     (θ θ' : BiHom 62 64 (S_0_0 : SyntheticSpectrum)) (h2 : θ'+θ'=0) :
  17     sphereProduct θ θ-sphereProduct θ' θ'=sphereProduct (θ-θ') (θ-θ') := by
  18   classical
  19   letI := model.exactness.shiftAdditive (62,64)
  20   let d := θ-θ'
  21   have ht : θ=θ'+d := by dsimp only [d]; abel
  22   have hc : sphereProduct d θ'=sphereProduct θ' d := by
  23     have h := model.sphereProductCommutative 62 64 62 64 d θ'
  24     norm_num [homotopyRegrade] at h
  25     exact h
  26   have hz : sphereProduct θ' d+sphereProduct d θ'=0 := by
  27     rw [hc]
  28     dsimp only [sphereProduct]
  29     rw [←Preadditive.comp_add,←Preadditive.add_comp,←Functor.map_add,h2,Functor.map_zero,
  30       zero_comp,comp_zero]
  31   have hs : sphereProduct (θ'+d) (θ'+d)=sphereProduct θ' θ'+sphereProduct d d := by
  32     have he : sphereProduct (θ'+d) (θ'+d)=
  33         sphereProduct θ' θ'+sphereProduct θ' d+(sphereProduct d θ'+sphereProduct d d) := by
  34       simp only [sphereProduct,Functor.map_add,Preadditive.add_comp,Preadditive.comp_add]
  35       abel
  36     rw [he]
  37     calc
  38       _ = sphereProduct θ' θ'+(sphereProduct θ' d+sphereProduct d θ')+sphereProduct d d := by abel
  39       _ = _ := by rw [hz,add_zero]
  40   change sphereProduct θ θ-sphereProduct θ' θ'=sphereProduct d d
  41   rw [ht,hs,add_sub_cancel_left]

  45     {H' : SyntheticSpectrum} {unit : S_0_0 ⟶ H'}
  46     {F : SyntheticAdamsFamily SyntheticSpectrum} {X : SyntheticSpectrum}
  47     {T : SyntheticTowerPresentation unit F}
  48     (c : SyntheticTowerConvergence T X) (i : Tridegree)

  50     (ha : Detects c i x a) (hb : FiltrationAtLeast T (i.1+1) (b-a)) :
  51     Detects c i x b := by

  53   obtain ⟨e,he,a',ha',hga⟩ := ha

  55   let p := (i.2.1-i.1,i.2.2)
  56   have hmem : b-a∈(ModuleCat.subobjectModule (syntheticHomotopy X p)) (P.F (i.1+1) p) := by
  57     simpa only [P,p,FiltrationAtLeast,SyntheticTowerPresentation.filtration,OrderIso.apply_symm_apply] using hb
  58   obtain ⟨z,hz⟩ := hmem
  59   let j := Subobject.ofLE (P.F (i.1+1) p) (P.F i.1 p) (P.mono i.1 p)
  60   refine ⟨e,he,a'+j z,?_,?_⟩
  61   · rw [map_add]
  62     have hj : (P.F i.1 p).arrow (j z)=(P.F (i.1+1) p).arrow z :=
  63       ConcreteCategory.congr_hom (Subobject.ofLE_arrow (P.mono i.1 p)) z

  68       ConcreteCategory.congr_hom (cokernel.condition j) z

  71 private theorem filtration_step_of_infinity_zero
  72     {H' : SyntheticSpectrum} {unit : S_0_0 ⟶ H'}
  73     {F : SyntheticAdamsFamily SyntheticSpectrum} {X : SyntheticSpectrum}
  74     {T : SyntheticTowerPresentation unit F} (c : SyntheticTowerConvergence T X)
  75     (s m w : ℤ) (hzero : IsZero ((F.obj X).sequence.ssData (s,m+s,w)).eInfty)
  76     (a : BiHom m w X) (ha : FiltrationAtLeast T s a) :
  77     FiltrationAtLeast T (s+1) a := by
  78   let P := T.filtration X
  79   have hz : IsZero (P.associatedGraded s (m,w)) := by
  80     simpa only [add_sub_cancel_right] using
  81       (c.identification (s,m+s,w)).isZero_iff.mp hzero

  83     simpa only [P, FiltrationAtLeast, SyntheticTowerPresentation.filtration,
  84       OrderIso.apply_symm_apply] using ha

  86   have hgrade : P.toAssociatedGraded s (m,w) a' = 0 :=
  87     (ModuleCat.subsingleton_of_isZero hz).elim _ _

  89   rw [ha'] at hnext
  90   simpa only [P, FiltrationAtLeast, SyntheticTowerPresentation.filtration,
  91     OrderIso.apply_symm_apply] using hnext

 102 private theorem permanentQuotient_zero_of_page_zero
 103     (X : ClassicalObject) (r b : ℤ) (p : ℤ × ℤ) (hr : 2 ≤ r) (hb : r-1 ≤ b)
