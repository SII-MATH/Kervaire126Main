  23 private def lambdaAction (m n : ℤ) (X : SyntheticSpectrum) :
  24     BiHom m n X → BiHom m (n - 1) X := by
  25   intro f
  26   have heq : (m, n) + (0, -1) = (m, n - 1) := by
  27     ext <;> simp [sub_eq_add_neg]
  28   refine eqToHom ?_ ≫
  29     (SyntheticCategory.biShift_comp (m, n) (0, -1)).inv.app S_0_0 ≫
  30       SyntheticCategory.lam.app (Smn m n) ≫ f
  31   simp only [Smn]
  32   rw [heq]

  39 private theorem lambda_powers_injective_of_lower_weights
  40     (coh : BiShiftCoherence SyntheticSpectrum) (m wt : ℤ) (X : SyntheticSpectrum)
  41     (h : ∀ wt' : ℤ, wt' ≤ wt → LambdaInjectiveAt m wt' X) :
  42     LambdaPowersInjectiveAt m wt X := by
  43   have one (k : ℕ) : Function.Injective (fun a :
  44       (SyntheticCategory.biShift (0,-(k : ℤ))).obj (Smn (Syn := SyntheticSpectrum) m wt) ⟶ X =>
  45       SyntheticCategory.lam.app _ ≫ a) := by
  46     let e : Smn (Syn := SyntheticSpectrum) m (wt-k) ≅
  47         (SyntheticCategory.biShift (0,-(k : ℤ))).obj (Smn m wt) :=
  48       eqToIso (by dsimp only [Smn]; congr 1; simp [sub_eq_add_neg]) ≪≫
  49         ((SyntheticCategory.biShift_comp (m,wt) (0,-(k : ℤ))).app S_0_0).symm
  50     intro a b hab
  51     apply (cancel_epi e.hom).mp
  52     apply h (wt-k) (by omega)
  53     dsimp only [lambdaAction]
  54     have hc : SyntheticCategory.lam.app (Smn (Syn := SyntheticSpectrum) m (wt-k)) ≫ (e.hom ≫ a) =
  55         SyntheticCategory.lam.app (Smn (Syn := SyntheticSpectrum) m (wt-k)) ≫ (e.hom ≫ b) := by
  56       have hn := SyntheticCategory.lam.naturality e.hom
  57       dsimp only [Functor.id_map] at hn
  58       rw [← Category.assoc, ← Category.assoc, ← hn]
  59       simpa only [Category.assoc] using
  60         congrArg (fun z => (SyntheticCategory.biShift (0,-1)).map e.hom ≫ z) hab
  61     exact congrArg (fun z => _ ≫ _ ≫ z) hc
  62   have powers (k : ℕ) : Function.Injective (fun a : BiHom m wt X =>
  63       lambdaPow k (Smn m wt) ≫ a) := by
  64     induction k with
  65     | zero =>
  66       intro a b hab
  67       exact (cancel_epi (SyntheticCategory.biShift_zero.hom.app (Smn m wt))).mp hab
  68     | succ k ih =>
  69       intro a b hab
  70       apply ih
  71       apply one k
  72       rw [lambdaPow_add coh k 1 (k+1) rfl, lambdaPow_one coh] at hab
  73       have hc := (cancel_epi ((lambdaShiftAddIso k 1 (k+1) rfl).inv.app (Smn (Syn := SyntheticSpectrum) m wt))).mp
  74         (by simpa only [Category.assoc] using hab)
  75       have hn := SyntheticCategory.lam.naturality (lambdaPow k (Smn (Syn := SyntheticSpectrum) m wt))
  76       dsimp only [Functor.id_map] at hn
  77       change (SyntheticCategory.biShift (0,-1)).map (lambdaPow k (Smn m wt)) ≫
  78         (SyntheticCategory.lam.app (Smn m wt) ≫ a) =
  79         (SyntheticCategory.biShift (0,-1)).map (lambdaPow k (Smn m wt)) ≫
  80         (SyntheticCategory.lam.app (Smn m wt) ≫ b) at hc
  81       rw [← Category.assoc, ← Category.assoc] at hc
  82       rw [hn] at hc
  83       simpa only [Category.assoc] using hc
  84   intro k a b hab
  85   apply powers k
  86   dsimp only [lambdaMultiply] at hab
  87   have hc := (cancel_epi (eqToHom (by congr 1; simp))).mp hab
  88   exact (cancel_epi ((SyntheticCategory.biShift_comp (m,wt) (0,-(k : ℤ))).inv.app S_0_0)).mp
  89     (by simpa only [Category.assoc] using hc)

 227 private theorem lambda_quotient_maps_equal_on_spheres
 228     (X : ClassicalObject) (q : ℕ) (hq : 0<q)
 229     (f g : XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) q ⟶
 230       XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) q)
 231     (hfg : XModLambdaN.incl _ q ≫ f = XModLambdaN.incl _ q ≫ g)
 232     (m w : ℤ) (c : BiHom m w (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) q)) :
 233     c ≫ f = c ≫ g := by
 234   let Y := model.nu.functor.obj (X.obj model.auxiliary)
 235   have hzero : XModLambdaN.incl Y q ≫ (f-g) = 0 := by
 236     rw [Preadditive.comp_sub,hfg,sub_self]
 237   obtain ⟨z,hz⟩ := (XModLambdaN.cofiberTriangle Y q).yoneda_exact₃
 238     (XModLambdaN.cofiberTriangle_distinguished Y q) (f-g) hzero
 239   let e := (KIP126.ProofTools.Operations.biSuspensionIso (q:ℤ) Y).symm
 240   have hn := lambda_boundary_no_return X q hq m w
 241     ((c ≫ XModLambdaN.proj Y q) ≫ e.hom) (e.inv ≫ z)
 242   have hn' : (c ≫ XModLambdaN.proj Y q) ≫ z = 0 := by
 243     simpa only [Category.assoc,Iso.hom_inv_id_assoc] using hn
 244   apply sub_eq_zero.mp
 245   rw [←Preadditive.comp_sub,hz,←Category.assoc]
 246   exact hn'
