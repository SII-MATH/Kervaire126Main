  15 theorem nonzero_differential_not_permanent {E : KIP126.Core.SpectralSequence (ModuleCat.{v} ℤ) (ℤ × ℤ)} {r : ℤ} {p : ℤ × ℤ} {q : ℤ × ℤ} {x : E.Page 2 p} {y : E.Page 2 q} (hr0 : E.r₀ ≤ r)
  16     (h : HasNonzeroDifferential E r p q x y) : ¬ NonzeroSurvival E p x := by
  17   classical
  18   rintro ⟨z, hz, _⟩
  19   let A := E.ssData p
  20   let n : WithTop ℕ := ↑(r + 1 - E.r₀).toNat
  21   let z' := (Subobject.ofLE (A.Z ⊤) (A.Z n) (A.Z_anti le_top)) z
  22   obtain ⟨hdeg, xr, yr, hx, hy, hd, hne⟩ := h
  23   have hr2 := hx.1
  24   have hlater : ReachesPage E (r + 1) p x := by
  25     refine ⟨A.pageπ n z', by omega, z', ?_, rfl⟩
  26     dsimp only [z']
  27     rw [← CategoryTheory.comp_apply, ← Category.assoc, Subobject.ofLE_comp_ofLE]
  28     exact hz
  29   have hd0 := represents_d_zero_of_later hr0 (by omega : r < r + 1) hx hlater
  30   rw [ModuleCat.comp_apply, hd0, map_zero] at hd
  31   exact hne hd.symm

  34 theorem earlier_zero_of_target_survives {E : KIP126.Core.SpectralSequence (ModuleCat.{v} ℤ) (ℤ × ℤ)} {r : ℤ} {t : ℤ} {p : ℤ × ℤ} {x : E.Page 2 p}
  35     (hr0 : E.r₀ ≤ r) (hrt : r < t)
  36     {y : E.Page 2 (p + E.diffDeg r)}
  37     (hsource : SurvivesTo E r p x)
  38     (htargets : DifferentialTargets E r p x y)
  39     (htarget : SurvivesTo E t (p + E.diffDeg r) y) :
  40     HasDifferential E r p (p + E.diffDeg r) x 0 := by
  41   classical
  42   obtain ⟨xr, hxr, _⟩ := hsource
  43   have hd0 : E.d r p xr = 0 := by
  44     rcases htargets xr hxr with hz | ⟨yr, hyr, hdr⟩

  46     · have hd : HasDifferential E r p (p + E.diffDeg r) x y :=
  47         ⟨rfl, xr, yr, hxr, hyr, by simpa only [eqToHom_refl, Category.comp_id] using hdr⟩

  55 theorem next_nonzero_of_not_hit {E : KIP126.Core.SpectralSequence (ModuleCat.{v} ℤ) (ℤ × ℤ)} {r : ℤ} {p : ℤ × ℤ} {x : E.Page 2 p}
  56     (hr0 : E.r₀ ≤ r) (hr : 2 ≤ r)
  57     {a : E.Page (r+1) p} {b : E.Page r p}
  58     (ha : RepresentsOnPage E (r+1) p x a) (hb : RepresentsOnPage E r p x b)
  59     (hne : b ≠ 0) (hnot : ¬ HitOnPage E r p x) : a ≠ 0 := by
  60   classical
  61   generalize hq : p-E.diffDeg r=q
  62   have hp : p=q+E.diffDeg r := by rw [←hq]; abel
  63   subst p
  64   obtain ⟨_,z,hz,hza⟩ := ha
  65   let A := E.ssData (q+E.diffDeg r)
  66   have hi : (r+1-E.r₀).toNat=(r-E.r₀).toNat+1 := by omega
  67   have hnm : (↑(r-E.r₀).toNat : WithTop ℕ) ≤ ↑(r+1-E.r₀).toNat := by
  68     exact_mod_cast (show (r-E.r₀).toNat≤(r+1-E.r₀).toNat by omega)
  69   let br := (Subobject.ofLE _ _ (A.Z_anti hnm) ≫ A.pageπ ↑(r-E.r₀).toNat) z
  70   have hbr : RepresentsOnPage E r (q+E.diffDeg r) x br := by
  71     refine ⟨hr, (Subobject.ofLE _ _ (A.Z_anti hnm)) z, ?_, rfl⟩
  72     rw [← CategoryTheory.comp_apply, ← Category.assoc, Subobject.ofLE_comp_ofLE]
  73     exact hz
  74   have heb : br=b := represents_unique hbr hb
  75   intro ha0
  76   have hv := next_projection_zero_iff_incoming E r hr0 q
  77   have hv' : ∀ zz : (Subobject.underlying.obj (A.Z ↑(r+1-E.r₀).toNat) : ModuleCat ℤ),
  78       A.pageπ ↑(r+1-E.r₀).toNat zz=0 ↔
  79         (Subobject.ofLE _ _ (A.Z_anti hnm) ≫ A.pageπ ↑(r-E.r₀).toNat) zz∈
  80           LinearMap.range (E.d r q).hom := by
  81     have transport (m : ℕ) (hm : m=(r-E.r₀).toNat+1)
  82         (hge : (↑(r-E.r₀).toNat : WithTop ℕ)≤↑m) :
  83         ∀ zz : (Subobject.underlying.obj (A.Z ↑m) : ModuleCat ℤ),
  84           A.pageπ ↑m zz=0 ↔
  85             (Subobject.ofLE _ _ (A.Z_anti hge) ≫ A.pageπ ↑(r-E.r₀).toNat) zz∈
  86               LinearMap.range (E.d r q).hom := by
  87       subst m
  88       exact hv
  89     exact transport _ hi hnm
  90   have hmem : br∈LinearMap.range (E.d r q).hom := by
  91     exact (hv' z).mp (hza.trans ha0)
  92   obtain ⟨c,hc⟩ := hmem
  93   apply hnot
  94   refine ⟨q, rfl, c,b,hb,hne,?_⟩
  95   simpa only [eqToHom_refl, Category.comp_id] using hc.trans heb

  98 theorem permanent_survives_of_no_early_hit {E : KIP126.Core.SpectralSequence (ModuleCat.{v} ℤ) (ℤ × ℤ)} {p : ℤ × ℤ} {x : E.Page 2 p}
  99     (hstart : E.r₀≤2) (N : ℕ) (hN : 2≤N)
 100     (hperm : IsPermanentCycle E p x) (hne : x≠0)
 101     (hnot : ∀ r : ℤ, 2≤r → r<N → ¬HitOnPage E r p x) :
 102     SurvivesTo E N p x := by
 103   classical
 104   obtain ⟨z,hz⟩ := hperm
 105   let A := E.ssData p
 106   have hrep (r : ℤ) (hr : 2≤r) : ReachesPage E r p x := by
 107     let zr := (Subobject.ofLE (A.Z ⊤) (A.Z ↑(r-E.r₀).toNat) (A.Z_anti le_top)) z
 108     refine ⟨A.pageπ _ zr,hr,zr,?_,rfl⟩
 109     dsimp only [zr]
 110     rw [← CategoryTheory.comp_apply, ← Category.assoc, Subobject.ofLE_comp_ofLE]
 111     exact hz
 112   have hall (n : ℕ) : n+2≤N → SurvivesTo E ((n:ℤ)+2) p x := by
 113     induction n with
 114     | zero =>
 115       intro hn
 116       obtain ⟨a,ha⟩ := hrep 2 (by omega)
 117       change SurvivesTo E 2 p x
 118       exact ⟨a,ha,by rw [←ha.eq_on_page_two]; exact hne⟩
 119     | succ n ih =>
 120       intro hn
 121       obtain ⟨b,hb,hbn⟩ := ih (by omega)
 122       obtain ⟨a,ha⟩ := hrep ((n:ℤ)+2+1) (by omega)
 123       have han := next_nonzero_of_not_hit (by omega) (by omega) ha hb hbn
 124         (hnot _ (by omega) (by omega))
 125       have he : ((n+1:ℕ):ℤ)+2=(n:ℤ)+2+1 := by omega
 126       rw [he]
 127       exact ⟨a,ha,han⟩
 128   have h := hall (N-2) (by omega)
 129   have he : ((N-2:ℕ):ℤ)+2=N := by omega
 130   rwa [he] at h

 133 theorem nonzero_differential_not_zero {E : KIP126.Core.SpectralSequence (ModuleCat.{v} ℤ) (ℤ × ℤ)} {r : ℤ} {p : ℤ × ℤ} {q : ℤ × ℤ} {x : E.Page 2 p} {y : E.Page 2 q}
 134     (h : HasNonzeroDifferential E r p q x y)
 135     (hzero : HasDifferential E r p q x 0) : False := by
 136   classical
 137   obtain ⟨he,a,b,ha,hb,hd,hbn⟩ := h
 138   obtain ⟨he',a',b',ha',hb',hd'⟩ := hzero
 139   have haa : a=a' := represents_unique ha ha'
 140   have hbb : b'=0 := represents_unique hb' (RepresentsOnPage.zero hb'.1)
 141   apply hbn
 142   calc b = (E.d r p ≫ eqToHom (congrArg (E.Page r) he)) a := hd.symm
 143        _ = b' := by simpa only [haa] using hd'
 144        _ = 0 := hbb

 147 theorem cycle_successor_lift
 148     (E : KIP126.Core.SpectralSequence (ModuleCat.{v} ℤ) (ℤ × ℤ))
 149     (r : ℤ) (hr : E.r₀ ≤ r) (p : ℤ × ℤ)
 150     (z : (Subobject.underlying.obj ((E.ssData p).Z ↑(r-E.r₀).toNat) : ModuleCat.{v} ℤ))
 151     (hz : E.d r p ((E.ssData p).pageπ ↑(r-E.r₀).toNat z) = 0) :
 152     ∃ zn : (Subobject.underlying.obj ((E.ssData p).Z ↑((r-E.r₀).toNat+1)) : ModuleCat.{v} ℤ),
 153       (Subobject.ofLE _ _ ((E.ssData p).Z_anti (by exact_mod_cast Nat.le_succ (r-E.r₀).toNat))) zn = z := by
 154   classical
 155   let D := E.ssData p
 156   let n := (r-E.r₀).toNat
 157   have hsucc : (↑n : WithTop ℕ) ≤ ↑(n+1) := by exact_mod_cast Nat.le_succ n
 158   let i := Subobject.ofLE (D.Z ↑(n+1)) (D.Z ↑n) (D.Z_anti hsucc)
 159   have hker : D.pageπ ↑n z ∈ LinearMap.ker (E.d r p).hom := hz
 160   rw [←subobjectModule_kernel,E.Z_succ r p hr,subobjectModule_image] at hker
 161   obtain ⟨z',hz'⟩ := hker
 162   change D.pageπ ↑n (i z') = D.pageπ ↑n z at hz'
 163   have hzero : D.pageπ ↑n (z-i z')=0 := by rw [map_sub,hz',sub_self]
 164   have hb : ∃ b : (Subobject.underlying.obj (D.B ↑n) : ModuleCat.{v} ℤ),
 165       (Subobject.ofLE (D.B ↑n) (D.Z ↑n) (D.B_le_Z _)) b = z-i z' :=
 166     (cokernel_π_eq_zero_iff_mem_range _ _).mp hzero
 167   obtain ⟨b,hb⟩ := hb
 168   have hBZ : D.B ↑n ≤ D.Z ↑(n+1) := (D.B_mono hsucc).trans (D.B_le_Z _)
 169   let zn : (Subobject.underlying.obj (D.Z ↑(n+1)) : ModuleCat.{v} ℤ) :=
 170     z' + (Subobject.ofLE (D.B ↑n) (D.Z ↑(n+1)) hBZ) b
 171   have hnz : i zn=z := by
 172     dsimp only [zn]
 173     rw [map_add]
 174     change i z' + (Subobject.ofLE (D.B ↑n) (D.Z ↑(n+1)) hBZ ≫ i) b = z
 175     dsimp only [i]
 176     rw [Subobject.ofLE_comp_ofLE,hb]
 177     abel
 178   exact ⟨zn,hnz⟩

 195 theorem later_page_frame_of_earlier {E : KIP126.Core.SpectralSequence (ModuleCat.{v} ℤ) (ℤ × ℤ)} {r : ℤ} {t : ℤ} {p : ℤ × ℤ} {x : E.Page 2 p}
 196     (hr : 2≤r) (hrt : r≤t)
 197     (hframe : ∃ a : E.Page r p, RepresentsOnPage E r p x a ∧ ∀ y:E.Page r p,y=0∨y=a)
 198     (ht : ReachesPage E t p x) :
 199     ∃ a : E.Page t p, RepresentsOnPage E t p x a ∧ ∀ y:E.Page t p,y=0∨y=a := by
 200   classical
 201   obtain ⟨ar,har,hall⟩ := hframe
 202   obtain ⟨at',htrep⟩ := ht
 203   have htrep' := htrep
 204   obtain ⟨ht2,z,hzt,hat⟩ := htrep'
 205   let A := E.ssData p
 206   let nr : WithTop ℕ := ↑(r-E.r₀).toNat
 207   let nt : WithTop ℕ := ↑(t-E.r₀).toNat
 208   have hrt' : nr≤nt := by
 209     dsimp only [nr,nt]
 210     exact_mod_cast (show (r-E.r₀).toNat≤(t-E.r₀).toNat by omega)
 211   let i := Subobject.ofLE (A.Z nt) (A.Z nr) (A.Z_anti hrt')
 212   have har' : RepresentsOnPage E r p x (A.pageπ nr (i z)) := by
 213     refine ⟨hr,i z,?_,rfl⟩
 214     dsimp only [i]
 215     rw [←CategoryTheory.comp_apply,←Category.assoc,Subobject.ofLE_comp_ofLE]
 216     exact hzt
 217   have hzr : A.pageπ nr (i z)=ar := represents_unique har' har
 218   have hzero (w : (Subobject.underlying.obj (A.Z nt) : ModuleCat ℤ))
 219       (hw : A.pageπ nr (i w)=0) : A.pageπ nt w=0 := by
 220     apply (subobject_cokernel_π_eq_zero_iff (A.B nt) (A.Z nt) (A.B_le_Z _) _).mpr
 221     have hh := (subobject_cokernel_π_eq_zero_iff (A.B nr) (A.Z nr) (A.B_le_Z _) _).mp hw
 222     have he : (A.Z nr).arrow (i w)=(A.Z nt).arrow w :=
 223       ConcreteCategory.congr_hom (Subobject.ofLE_arrow (A.Z_anti hrt')) w
 224     rw [he] at hh
 225     exact (ModuleCat.subobjectModule A.V).monotone (A.B_mono hrt') hh
 226   refine ⟨at',htrep,?_⟩
 227   intro y
 228   haveI : Epi (A.pageπ nt) := inferInstanceAs (Epi (cokernel.π _))
 229   obtain ⟨w,hw⟩ := (ModuleCat.epi_iff_surjective (A.pageπ nt)).mp inferInstance y
 230   rcases hall (A.pageπ nr (i w)) with h0|ha
 231   · left
 232     rw [←hw]
 233     exact hzero w h0
 234   · right
 235     have hh : A.pageπ nr (i (w-z))=0 := by rw [map_sub,map_sub,ha,hzr,sub_self]
 236     have hh' := hzero (w-z) hh
 237     rw [map_sub,hw,hat,sub_eq_zero] at hh'
 238     exact hh'
