  50     ∀ j : ℕ, FiltrationAtLeast model.syntheticTower j z := by
  51   have hz (j:ℤ) (hj:0≤j) :
  52       IsZero (((model.family.obj (S_0_0:SyntheticSpectrum)).sequence.ssData (j,1+j,3)).eInfty) := by
  53     by_cases hw:3≤1+j

  55         by_cases he:j=2

  57         · exact KIP126.Literature.Standard.sphereVanishing _ _ (by omega) (by omega)
  58       exact sphere_infinity_zero_of_permanent_quotient j (1+j) 3 (by omega)
  59         (permanent_quotient_zero_of_page_zero .sphere 2 _ (by decide) (by omega) _ hp)
  60     · haveI : Subsingleton (nuEInftyModel model .sphere (j,1+j) 3) := by
  61         simp only [nuEInftyModel,hw,ite_false]; infer_instance

  63         (KIP126.Literature.Standard.eInfinity_nu .sphere (j,1+j) 3).injective.subsingleton

  65       let f := (model.family.map model.nu.unitIso.hom).eInftyMap (j,1+j,3)
  66       let g := (model.family.map model.nu.unitIso.inv).eInftyMap (j,1+j,3)
  67       have hgf:g≫f=𝟙 _ := by
  68         dsimp only [f,g]
  69         rw [←SpectralSequenceMorphism.eInftyMap_comp,←SyntheticAdamsFamily.map_comp,
  70           Iso.inv_hom_id,SyntheticAdamsFamily.map_id,SpectralSequenceMorphism.eInftyMap_id]
  71       apply (IsZero.iff_id_eq_zero _).mpr
  72       exact hgf.symm.trans (by
  73         have hf:f=0 := hn.eq_of_src f 0
  74         rw [hf,comp_zero])

  76     refine ⟨z≫(model.syntheticTower.tower _).zeroIso.inv,?_⟩

  82     (by omega) (fun k hk _ => hz k hk) z).mp h0

  93 theorem eta_finite_quotient_eq_of_first_quotient_eq
  94     (n : ℕ) (x y : BiHom 1 2 (S_0_0 : SyntheticSpectrum))
  95     (h : quotientClass 1 x = quotientClass 1 y) : quotientClass n x = quotientClass n y := by
  96   have hz : quotientClass 1 (x-y) = 0 := by
  97     simp only [quotientClass, Preadditive.sub_comp] at h ⊢
  98     rw [h, sub_self]
  99   obtain ⟨b,hb⟩ := AlphaOne.lambda_one_division_of_quotient_zero 1 3 (x-y) hz
 100   have hh := congrArg (fun u => quotientClass n u) hb
 101   have hbn := sphere_one_three_finite_quotient_zero b n
 102   have he : quotientClass n (lambdaMultiply 1 b) = 0 := by
 103     change b ≫ XModLambdaN.incl _ n = 0 at hbn
 104     simp only [quotientClass, lambdaMultiply, Category.assoc]
 105     rw [hbn]
 106     simp only [comp_zero]
 107   rw [he] at hh
 108   change 0 = (x-y) ≫ XModLambdaN.incl _ n at hh
 109   rw [Preadditive.sub_comp] at hh
 110   exact sub_eq_zero.mp hh.symm

 112 /-- Source automorphisms act trivially on the image of η in each finite λ quotient. -/
 113 theorem eta_source_iso_fixed_in_finite_quotient
 114     (n : ℕ) (eta : BiHom 1 2 (S_0_0 : SyntheticSpectrum))
 115     (a : Smn (Syn := SyntheticSpectrum) 1 2 ≅ Smn 1 2) :
 116     quotientClass n (a.hom ≫ eta) = quotientClass n eta := by
 117   apply eta_finite_quotient_eq_of_first_quotient_eq
 118   let e := KIP126.Literature.sphereFirstQuotient model 1 2
 119   by_cases h : quotientClass 1 eta = 0
 120   · change (a.hom ≫ eta) ≫ XModLambdaN.incl _ 1 = eta ≫ XModLambdaN.incl _ 1
 121     change eta ≫ XModLambdaN.incl _ 1 = 0 at h
 122     rw [Category.assoc, h, comp_zero]
 123   · apply e.injective
 124     apply low_eta_page_unique
 125     · intro he
 126       have hz : quotientClass 1 (a.hom ≫ eta) = 0 := e.injective (he.trans (map_zero e).symm)
 127       apply h
 128       have hh := congrArg (fun u => a.inv ≫ u) hz
 129       simpa only [quotientClass, Category.assoc, Iso.inv_hom_id_assoc, comp_zero] using hh
 130     · exact fun he => h (e.injective (he.trans (map_zero e).symm))

 132 theorem eta_action_source_iso_fixed_in_finite_quotient
 133     (n : ℕ) (hn : 0 < n) (eta : BiHom 1 2 (S_0_0 : SyntheticSpectrum))
 134     {k l : ℤ} (x : BiHom k l (XModLambdaN (S_0_0 : SyntheticSpectrum) n))
 135     (a : Smn (Syn := SyntheticSpectrum) (1+k) (2+l) ≅ Smn (1+k) (2+l)) :
 136     a.hom ≫ sphereAction eta x = sphereAction eta x := by
 137   let F := SyntheticCategory.biShift (Syn := SyntheticSpectrum) (k,l)
 138   let ff := SyntheticCategory.biShift_fullyFaithful (Syn := SyntheticSpectrum) (k,l)
 139   let d : Smn (Syn := SyntheticSpectrum) (1+k) (2+l) ≅ F.obj (Smn 1 2) :=
 140     ((SyntheticCategory.biShift_comp (1,2) (k,l)).app (S_0_0 : SyntheticSpectrum)).symm
 141   let b := ff.preimageIso (d.symm ≪≫ a ≪≫ d)
 142   have hb : F.map b.hom = d.inv ≫ a.hom ≫ d.hom := ff.map_preimage _
 143   have hh : sphereAction (b.hom ≫ eta) x = sphereAction eta x := by
 144     rw [← KIP126.Literature.Standard.quotientAlgebra_sphere_action n hn 1 2 k l (b.hom ≫ eta) x,
 145       ← KIP126.Literature.Standard.quotientAlgebra_sphere_action n hn 1 2 k l eta x,
 146       eta_source_iso_fixed_in_finite_quotient n eta b]
 147   change d.hom ≫ F.map (b.hom ≫ eta) ≫ x = d.hom ≫ F.map eta ≫ x at hh
 148   rw [Functor.map_comp, hb] at hh
 149   change a.hom ≫ (d.hom ≫ F.map eta ≫ x) = d.hom ≫ F.map eta ≫ x
 150   simpa only [Category.assoc, d.hom_inv_id_assoc] using hh
