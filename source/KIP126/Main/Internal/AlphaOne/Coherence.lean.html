  18     (SyntheticCategory.biShift b).obj ((SyntheticCategory.biShift a).obj A) ≅

  24 private theorem shiftSwap_add (coh : BiShiftCoherence SyntheticSpectrum)
  25     (a b c : ℤ × ℤ) (A : SyntheticSpectrum) :
  26     (shiftSwap a (b+c) A).hom ≫
  27       (SyntheticCategory.biShift a).map ((biShiftAddIso b c (b+c) rfl).inv.app A) =
  28     (biShiftAddIso b c (b+c) rfl).inv.app ((SyntheticCategory.biShift a).obj A) ≫
  29       (SyntheticCategory.biShift c).map (shiftSwap a b A).hom ≫
  30       (shiftSwap a c ((SyntheticCategory.biShift b).obj A)).hom := by
  31   classical
  32   let habc : (a+b)+c=a+(b+c) := add_assoc a b c
  33   let hbac : b+(a+c)=a+(b+c) := by abel
  34   have h₁ := coh.associativity a b c (a+b) (b+c) (a+(b+c)) rfl rfl habc rfl A
  35   have h₂ := coh.associativity b a c (a+b) (a+c) (a+(b+c))
  36     (add_comm _ _) rfl habc hbac A
  37   have h₃ := coh.associativity b c a (b+c) (a+c) (a+(b+c))
  38     rfl (add_comm _ _) (add_comm _ _) hbac A
  39   simp only [shiftSwap,Iso.trans_hom,Iso.symm_hom,Iso.app_hom,Iso.app_inv]
  40   rw [Functor.map_comp]
  41   apply (cancel_epi ((biShiftAddIso b c (b+c) rfl).hom.app
  42     ((SyntheticCategory.biShift a).obj A))).mp
  43   simp only [Category.assoc,Iso.hom_inv_id_app_assoc]
  44   rw [← Category.assoc,← h₁]
  45   simp only [Category.assoc]
  46   apply (cancel_epi ((SyntheticCategory.biShift c).map
  47     ((biShiftAddIso a b (a+b) rfl).hom.app A))).mpr
  48   -- Cancel the common left factor introduced by the first associativity square.
  49   apply (cancel_epi ((SyntheticCategory.biShift c).map
  50     ((biShiftAddIso b a (a+b) (add_comm _ _)).hom.app A))).mp
  51   simp only [← Functor.map_comp_assoc, Iso.hom_inv_id_app,CategoryTheory.Functor.map_id,Category.id_comp]
  52   rw [← Category.assoc,h₂]
  53   simp only [Category.assoc]
  54   apply (cancel_epi ((biShiftAddIso a c (a+c) rfl).hom.app
  55     ((SyntheticCategory.biShift b).obj A))).mpr
  56   apply (cancel_epi ((biShiftAddIso c a (a+c) (add_comm _ _)).hom.app
  57     ((SyntheticCategory.biShift b).obj A))).mp
  58   rw [← Category.assoc,← Category.assoc,← h₃]
  59   simp only [Category.assoc,
  60     Iso.hom_inv_id_app]
  61   erw [CategoryTheory.Category.id_comp,←Functor.map_comp,Iso.hom_inv_id_app,CategoryTheory.Functor.map_id]
  62   rfl

  96 private theorem lambda_pow_shift (coh : BiShiftCoherence SyntheticSpectrum)
  97     (n : ℕ) (a : ℤ × ℤ) (A : SyntheticSpectrum) :
  98     lambdaPow n ((SyntheticCategory.biShift a).obj A) =
  99       (shiftSwap a (KIP126.Synthetic.Context.lambdaDegree n) A).hom ≫
 100         (SyntheticCategory.biShift a).map (lambdaPow n A) := by
 101   classical
 102   induction n generalizing A with
 103   | zero =>
 104     change SyntheticCategory.biShift_zero.hom.app ((SyntheticCategory.biShift a).obj A) =
 105       (shiftSwap a 0 A).hom ≫
 106         (SyntheticCategory.biShift a).map (SyntheticCategory.biShift_zero.hom.app A)
 107     have h : SyntheticCategory.biShift_zero.hom.app ((SyntheticCategory.biShift a).obj A) =
 108         (biShiftAddIso a 0 a (add_zero a)).hom.app A ≫
 109           (biShiftAddIso 0 a a (zero_add a)).inv.app A ≫
 110             (SyntheticCategory.biShift a).map (SyntheticCategory.biShift_zero.hom.app A) := by
 111       rw [←coh.left_unit a A,Iso.inv_hom_id_app,Category.comp_id]
 112       exact (coh.right_unit a A).symm
 113     rw [shiftSwap_hom_eq a 0 a (add_zero a) (zero_add a),Category.assoc]
 114     exact h
 115   | succ n ih =>
 116     have h1 : lambdaPow 1 ((SyntheticCategory.biShift a).obj A) =
 117         (shiftSwap a (KIP126.Synthetic.Context.lambdaDegree 1) A).hom ≫
 118           (SyntheticCategory.biShift a).map (lambdaPow 1 A) := by
 119       rw [lambdaPow_one coh,lambdaPow_one coh]
 120       change SyntheticCategory.lam.app ((SyntheticCategory.biShift a).obj A) =
 121         (shiftSwap a (0,-1) A).hom ≫ (SyntheticCategory.biShift a).map (SyntheticCategory.lam.app A)
 122       rw [shiftSwap_hom_eq a (0,-1) (a+(0,-1)) rfl (add_comm _ _),Category.assoc]
 123       exact coh.lambda_comm a A
 124     have hn := shiftSwap_natural a (KIP126.Synthetic.Context.lambdaDegree n) (lambdaPow 1 A)
 125     have hh := shiftSwap_add_eq coh a (KIP126.Synthetic.Context.lambdaDegree 1) (KIP126.Synthetic.Context.lambdaDegree n)
 126       (KIP126.Synthetic.Context.lambdaDegree (n+1)) (by ext <;> simp [KIP126.Synthetic.Context.lambdaDegree]) A
 127     change (shiftSwap a (KIP126.Synthetic.Context.lambdaDegree (n+1)) A).hom ≫
 128         (SyntheticCategory.biShift a).map ((lambdaShiftAddIso 1 n (n+1) (by omega)).inv.app A) =
 129       (lambdaShiftAddIso 1 n (n+1) (by omega)).inv.app ((SyntheticCategory.biShift a).obj A) ≫
 130         (SyntheticCategory.biShift (KIP126.Synthetic.Context.lambdaDegree n)).map (shiftSwap a (KIP126.Synthetic.Context.lambdaDegree 1) A).hom ≫
 131         (shiftSwap a (KIP126.Synthetic.Context.lambdaDegree n) ((SyntheticCategory.biShift (KIP126.Synthetic.Context.lambdaDegree 1)).obj A)).hom at hh
 132     rw [lambdaPow_add coh 1 n (n+1) (by omega) ((SyntheticCategory.biShift a).obj A),
 133       lambdaPow_add coh 1 n (n+1) (by omega) A,h1,ih]
 134     simp only [Functor.map_comp,Category.assoc]
 135     rw [reassoc_of% hn]
 136     rw [reassoc_of% hh]
 137     rfl

 140 private theorem shift_assoc_inv (coh : BiShiftCoherence SyntheticSpectrum)
 141     (a b c ab bc total : ℤ × ℤ) (hab : a+b=ab) (hbc : b+c=bc)
 142     (habc : ab+c=total) (habc' : a+bc=total) (A : SyntheticSpectrum) :
 143     (biShiftAddIso ab c total habc).inv.app A ≫
 144       (SyntheticCategory.biShift c).map ((biShiftAddIso a b ab hab).inv.app A) =
 145     (biShiftAddIso a bc total habc').inv.app A ≫
 146       (biShiftAddIso b c bc hbc).inv.app ((SyntheticCategory.biShift a).obj A) := by
 147   classical
 148   have he : ((SyntheticCategory.biShift c).mapIso ((biShiftAddIso a b ab hab).app A) ≪≫
 149       (biShiftAddIso ab c total habc).app A) =
 150     ((biShiftAddIso b c bc hbc).app ((SyntheticCategory.biShift a).obj A) ≪≫
 151       (biShiftAddIso a bc total habc').app A) :=
 152     Iso.ext (coh.associativity a b c ab bc total hab hbc habc habc' A)
 153   have hi := congrArg Iso.inv he
 154   exact hi
