  24 private theorem normalize_permanent (p : ℤ×ℤ) (x : model.classical.sphere.Page 2 p)
  25     (hx : IsPermanentCycle model.classical.sphere p x) :
  26     IsPermanentCycle Normalized.sphere p (Normalized.label p x) := by
  27   have tr (E E' : KIP126.Core.SpectralSequence (ModuleCat ℤ) (ℤ×ℤ)) (h : E=E')
  28       (x : E.Page 2 p) (hx : IsPermanentCycle E p x) :
  29       IsPermanentCycle E' p ((eqToHom (congrArg (fun A => A.Page 2 p) h)) x) := by
  30     subst E'
  31     exact hx
  32   exact tr _ _ Normalized.sphere_eq.symm x hx
