  19 private theorem permanent_cycle_of_reaches1000 (s t : ℕ)
  20     (hn : 1 < (t:ℤ)-s) (hn' : (t:ℤ)-s≤127)
  21     (x : model.classical.sphere.Page 2 (s,t))
  22     (hx : ReachesPage model.classical.sphere 1000 (s,t) x) :
  23     IsPermanentCycle model.classical.sphere (s,t) x := by
  24   apply isPermanentCycle_of_reachesPage model.classical.sphere 1000
  25     (by rw [model.classical.firstPage]; omega) (s,t) x hx
  26   intro r hr
  27   have hp : Subsingleton (model.classical.sphere.Page r ((s:ℤ)+r,(t:ℤ)+r-1)) :=
  28     page_subsingleton_of_le _ 2 r _ (le_of_eq (model.classical.firstPage _)) (by omega)
  29       (KIP126.Literature.Standard.sphereVanishing _ _ (by omega) (by omega))
  30   haveI : Subsingleton (model.classical.sphere.Page r
  31       (((s:ℤ),(t:ℤ))+model.classical.sphere.diffDeg r)) := by
  32     rw [model.classical.differentialDegree]
  33     simpa only [Prod.mk_add_mk,add_sub_assoc] using hp
  34   ext y
  35   exact Subsingleton.elim _ _
