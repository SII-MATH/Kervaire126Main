  34     (r s t w : ℤ) (hr : 2≤r) (hwindow : t<w ∨ (q:ℤ)≤t-w) :
  35     Subsingleton ((model.family.quotient (S_0_0 : SyntheticSpectrum) q).Page r (s,t,w)) := by

  37   let e := model.family.functor.mapIso (Q.mapIso model.nu.unitIso)
  38   have hz := KIP126.Literature.Standard.finite_quotient_vanishing .sphere q hq r hr s t w (by omega)
  39   let i := e.inv.pageMap r (s,t,w)
  40   let j := e.hom.pageMap r (s,t,w)
  41   have hij : i ≫ j=𝟙 _ := by
  42     dsimp only [i,j]
  43     rw [←SpectralSequenceMorphism.pageMap_comp,e.inv_hom_id,SpectralSequenceMorphism.pageMap_id]
  44   have hi : Function.Injective i := by
  45     intro x y hxy
  46     have h := congrArg j hxy
  47     simpa only [←CategoryTheory.comp_apply,hij,ModuleCat.id_apply] using h
  48   have hs : Subsingleton (((model.family.functor.obj
  49       (XModLambdaN (model.nu.functor.obj SphereSpectrum) q)).Page r (s,t,w)) : ModuleCat ℤ) := by
  50     have hh := ModuleCat.isZero_iff_subsingleton.mp hz
  51     simpa only [Page,model.family.firstPage,SyntheticAdamsFamily.quotient,
  52       SyntheticAdamsFamily.obj,SyntheticAdamsSS.Page,ClassicalObject.obj] using hh
  53   have ht : Subsingleton (((model.family.functor.obj
  54       (XModLambdaN (S_0_0 : SyntheticSpectrum) q)).Page r (s,t,w)) : ModuleCat ℤ) :=
  55     ⟨fun x y => hi (hs.elim _ _)⟩
