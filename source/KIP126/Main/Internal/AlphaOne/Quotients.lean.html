  16 private def cycleMapFor (E : KIP126.Core.SpectralSequence (ModuleCat ℤ) (ℤ×ℤ))
  17     (hr : E.r₀=2) (n : WithTop ℕ) (p : ℤ×ℤ) :
  18     Subobject.underlying.obj ((E.ssData p).Z n) ⟶ E.Page 2 p :=
  19   Subobject.ofLE ((E.ssData p).Z n) ((E.ssData p).Z 0) ((E.ssData p).Z_anti bot_le) ≫
  20     (E.ssData p).pageπ 0 ≫ eqToHom (by simp only [Page,hr,sub_self,Int.toNat_zero,Nat.cast_zero])
  21 private def boundaryMapFor (E : KIP126.Core.SpectralSequence (ModuleCat ℤ) (ℤ×ℤ))
  22     (hr : E.r₀=2) (n : WithTop ℕ) (p : ℤ×ℤ) :
  23     Subobject.underlying.obj ((E.ssData p).B n) ⟶ E.Page 2 p :=
  24   Subobject.ofLE ((E.ssData p).B n) ((E.ssData p).Z 0)
  25     (((E.ssData p).B_le_Z n).trans ((E.ssData p).Z_anti bot_le)) ≫
  26     (E.ssData p).pageπ 0 ≫ eqToHom (by simp only [Page,hr,sub_self,Int.toNat_zero,Nat.cast_zero])
  27 private abbrev cycleMap := cycleMapFor Normalized.sphere rfl
  28 private abbrev boundaryMap := boundaryMapFor Normalized.sphere rfl
  29 private def cycles (c : ℤ) (p : ℤ×ℤ) := LinearMap.range (cycleMap (↑(c-1).toNat) p).hom
  30 private def boundaries (b : ℤ) (p : ℤ×ℤ) := LinearMap.range (boundaryMap (↑(b-1).toNat) p).hom
  31 private abbrev CycleQuotient (c b : ℤ) (p : ℤ×ℤ) := E2Subquotient (cycles c p) (boundaries b p)

  33 private theorem quotient_vanishing_actual (c b : ℤ) (p : ℤ×ℤ)
  34     (hz : Subsingleton (CycleQuotient c b p)) :
  35     Subsingleton (KIP126.Literature.CycleQuotient model .sphere c b p) := by
  36   let Q (E : KIP126.Core.SpectralSequence (ModuleCat ℤ) (ℤ×ℤ)) (hr : E.r₀=2) :=
  37     E2Subquotient (LinearMap.range (cycleMapFor E hr (↑(c-1).toNat) p).hom)
  38       (LinearMap.range (boundaryMapFor E hr (↑(b-1).toNat) p).hom)
  39   have transport (E E' : KIP126.Core.SpectralSequence (ModuleCat ℤ) (ℤ×ℤ))
  40       (he : E=E') (hr : E.r₀=2) (hr' : E'.r₀=2) (h : Subsingleton (Q E hr)) :
  41       Subsingleton (Q E' hr') := by
  42     subst E'
  43     exact h
  44   exact transport _ _ (Normalized.sequence_eq .sphere) rfl (model.classical.firstPage _) hz

  46 private theorem boundaries_monotone (p : ℤ×ℤ) : Monotone (fun b => boundaries b p) := by
  47   intro b b' h x hx
  48   obtain ⟨z,rfl⟩ := hx
  49   let E := Normalized.sphere.ssData p
  50   have hn : (↑(b-1).toNat : WithTop ℕ)≤↑(b'-1).toNat := by
  51     exact_mod_cast Int.toNat_le_toNat (by omega : b-1≤b'-1)
  52   refine ⟨(Subobject.ofLE _ _ (E.B_mono hn)) z,?_⟩
  53   change (Subobject.ofLE _ _ (E.B_mono hn) ≫ boundaryMap (↑(b'-1).toNat) p) z=_
  54   simp only [boundaryMap,boundaryMapFor,E,←Category.assoc,Subobject.ofLE_comp_ofLE]

  56 private theorem cycles_reaches (c : ℤ) (p : ℤ×ℤ) (hc : 1≤c)
  57     (x : Normalized.sphere.Page 2 p) :
  58     x∈cycles c p ↔ ReachesPage Normalized.sphere (c+1) p x := by
  59   have hn : c+1-2=c-1 := by omega
  60   change (∃ z, cycleMap (↑(c-1).toNat) p z=x) ↔ _
  61   change (∃ z, cycleMap (↑(c-1).toNat) p z=x) ↔
  62     ∃ y : (Normalized.sphere.ssData p).page ↑(c+1-2).toNat,
  63       ∃ _ : 2≤c+1,
  64         ∃ z : (Subobject.underlying.obj ((Normalized.sphere.ssData p).Z ↑(c+1-2).toNat) : ModuleCat ℤ),
  65           (Subobject.ofLE ((Normalized.sphere.ssData p).Z ↑(c+1-2).toNat)
  66             ((Normalized.sphere.ssData p).Z 0) ((Normalized.sphere.ssData p).Z_anti bot_le) ≫
  67             (Normalized.sphere.ssData p).pageπ 0) z=x ∧
  68             (Normalized.sphere.ssData p).pageπ ↑(c+1-2).toNat z=y
  69   rw [hn]
  70   constructor
  71   · rintro ⟨z,hz⟩
  72     refine ⟨_,by omega,z,?_,rfl⟩
  73     change ((Subobject.ofLE ((Normalized.sphere.ssData p).Z ↑(c-1).toNat)
  74       ((Normalized.sphere.ssData p).Z 0) _) ≫ (Normalized.sphere.ssData p).pageπ 0 ≫ 𝟙 _) z=x at hz
  75     erw [Category.comp_id] at hz
  76     exact hz
  77   · rintro ⟨_,hr,z,hz,_⟩
  78     refine ⟨z,?_⟩
  79     change ((Subobject.ofLE ((Normalized.sphere.ssData p).Z ↑(c-1).toNat)
  80       ((Normalized.sphere.ssData p).Z 0) _) ≫ (Normalized.sphere.ssData p).pageπ 0 ≫ 𝟙 _) z=x
  81     erw [Category.comp_id]
  82     exact hz

  93   simp only [cycleMap,cycleMapFor,E,←Category.assoc,Subobject.ofLE_comp_ofLE]

  95 private theorem boundaries_le_cycles (p : ℤ×ℤ) (b c : ℤ) : boundaries b p≤cycles c p := by
  96   rintro x ⟨z,rfl⟩
  97   let E := Normalized.sphere.ssData p
  98   have h : E.B (↑(b-1).toNat)≤E.Z (↑(c-1).toNat) :=
  99     (E.B_mono le_top).trans ((E.B_le_Z ⊤).trans (E.Z_anti le_top))
 100   refine ⟨(Subobject.ofLE _ _ h) z,?_⟩
 101   change (Subobject.ofLE _ _ h ≫ cycleMap (↑(c-1).toNat) p) z=_
 102   simp only [cycleMap,cycleMapFor,boundaryMap,boundaryMapFor,E,
 103     ←Category.assoc,Subobject.ofLE_comp_ofLE]
