  26 private def RawRepresentsOnPage {R : Type u} [Ring R]
  27     (E : SpectralSequence (ModuleCat.{v} R) Tridegree)
  28     (r : ℤ) (p : Tridegree) (x : E.Page 2 p) (y : E.Page r p) : Prop :=
  29   ∃ hr : 2 ≤ r,
  30     let D := E.ssData p
  31     let n : WithTop ℕ := ↑(2 - E.r₀).toNat
  32     let m : WithTop ℕ := ↑(r - E.r₀).toNat
  33     ∃ z : (Subobject.underlying.obj (D.Z m) : ModuleCat R),
  34       (Subobject.ofLE (D.Z m) (D.Z n) (D.Z_anti (by
  35         change (↑(2 - E.r₀).toNat : WithTop ℕ) ≤ ↑(r - E.r₀).toNat
  36         exact_mod_cast (show (2 - E.r₀).toNat ≤ (r - E.r₀).toNat by omega))) ≫
  37         D.pageπ n) z = x ∧ D.pageπ m z = y

  39 private def RawHasDifferential {R : Type u} [Ring R]
  40     (E : SpectralSequence (ModuleCat.{v} R) Tridegree)
  41     (r : ℤ) (p q : Tridegree) (x : E.Page 2 p) (y : E.Page 2 q) : Prop :=
  42   ∃ h : p + E.diffDeg r = q,
  43     ∃ (xr : E.Page r p) (yr : E.Page r q),
  44       RawRepresentsOnPage E r p x xr ∧ RawRepresentsOnPage E r q y yr ∧
  45         (E.d r p ≫ eqToHom (congrArg (E.Page r) h)) xr = yr

  47 private theorem SSDataMorphism.rawRepresentsOnPage_reindexed
  48     {E E' : SpectralSequence (ModuleCat.{v} R) Tridegree}
  49     {τ : Tridegree → Tridegree}
  50     (f : SSDataMorphism Tridegree E.ssData (fun p => E'.ssData (τ p)))
  51     (hstart : E.r₀ = E'.r₀) {r : ℤ} {p : Tridegree}
  52     {x : E.Page 2 p} {y : E.Page r p} (h : RawRepresentsOnPage E r p x y) :
  53     RawRepresentsOnPage E' r (τ p)
  54       ((f.pageMap p (↑(2 - E.r₀).toNat : WithTop ℕ) ≫ eqToHom
  55         (congrArg (fun a : ℤ => (E'.ssData (τ p)).page
  56           (↑(2 - a).toNat : WithTop ℕ)) hstart)) x)
  57       ((f.pageMap p (↑(r - E.r₀).toNat : WithTop ℕ) ≫ eqToHom
  58         (congrArg (fun a : ℤ => (E'.ssData (τ p)).page
  59           (↑(r - a).toNat : WithTop ℕ)) hstart)) y) := by
  60   rcases E with ⟨⟨r₀, D, δ, d⟩, dd, Zs, Bs⟩
  61   rcases E' with ⟨⟨r₀', D', δ', d'⟩, dd', Zs', Bs'⟩
  62   dsimp at hstart
  63   subst r₀'
  64   obtain ⟨hr, z, hx, hy⟩ := h
  65   let n : WithTop ℕ := ↑(2 - r₀).toNat
  66   let m : WithTop ℕ := ↑(r - r₀).toNat
  67   have hnm : n ≤ m := by
  68     dsimp [n, m]
  69     exact_mod_cast (show (2 - r₀).toNat ≤ (r - r₀).toNat by omega)
  70   refine ⟨hr, f.cycleMap p m z, ?_, ?_⟩
  71   · have hs := f.cycleMap_ofLE_assoc p hnm ((D' (τ p)).pageπ n)
  72     have hp := f.pageπ_pageMap p n
  73     have heq :
  74         (Subobject.ofLE ((D p).Z m) ((D p).Z n) ((D p).Z_anti hnm) ≫
  75           (D p).pageπ n) ≫ f.pageMap p n =
  76         f.cycleMap p m ≫
  77           Subobject.ofLE ((D' (τ p)).Z m) ((D' (τ p)).Z n)
  78             ((D' (τ p)).Z_anti hnm) ≫ (D' (τ p)).pageπ n := by
  79       rw [Category.assoc, hp]
  80       exact hs
  81     have hz := congrArg (fun a => a z) heq
  82     simpa only [ModuleCat.comp_apply, hx, eqToHom_refl, Category.comp_id, n, m]
  83       using hz.symm
  84   · have hz := congrArg (fun a => a z) (f.pageπ_pageMap p m)
  85     simpa only [ModuleCat.comp_apply, hy, eqToHom_refl, Category.comp_id, n, m]
  86       using hz.symm

  88 /-- Differential equations transfer along a reindexed ambient morphism whose
  89 induced quotient maps commute with the actual differentials. The reindexing
  90 need only respect each differential translation; it need not be additive.
  91 The commutation premise is a morphism law to be proved for any actual model. -/
  92 private theorem SSDataMorphism.rawHasDifferential_reindexed
  93     {E E' : SpectralSequence (ModuleCat.{v} R) Tridegree}
  94     {τ : Tridegree → Tridegree}
  95     (f : SSDataMorphism Tridegree E.ssData (fun p => E'.ssData (τ p)))
  96     (hstart : E.r₀ = E'.r₀)
  97     (hdegree : ∀ (r : ℤ) (p : Tridegree),
  98       τ (p + E.diffDeg r) = τ p + E'.diffDeg r)
  99     (hcomm :
 100       let Φ : ∀ (r : ℤ) (p : Tridegree), E.Page r p ⟶ E'.Page r (τ p) :=
 101         fun r p => f.pageMap p (↑(r - E.r₀).toNat : WithTop ℕ) ≫ eqToHom
 102           (congrArg (fun a : ℤ => (E'.ssData (τ p)).page
 103             (↑(r - a).toNat : WithTop ℕ)) hstart)
 104       ∀ (r : ℤ) (p : Tridegree),
 105         Φ r p ≫ E'.d r (τ p) =
 106           E.d r p ≫ Φ r (p + E.diffDeg r) ≫
 107             eqToHom (congrArg (E'.Page r) (hdegree r p)))
 108     {r : ℤ} {p q : Tridegree} {x : E.Page 2 p} {y : E.Page 2 q}
 109     (h : RawHasDifferential E r p q x y) :
 110     RawHasDifferential E' r (τ p) (τ q)
 111       ((f.pageMap p (↑(2 - E.r₀).toNat : WithTop ℕ) ≫ eqToHom
 112         (congrArg (fun a : ℤ => (E'.ssData (τ p)).page
 113           (↑(2 - a).toNat : WithTop ℕ)) hstart)) x)
 114       ((f.pageMap q (↑(2 - E.r₀).toNat : WithTop ℕ) ≫ eqToHom
 115         (congrArg (fun a : ℤ => (E'.ssData (τ q)).page
 116           (↑(2 - a).toNat : WithTop ℕ)) hstart)) y) := by
 117   let Φ : ∀ (r : ℤ) (p : Tridegree), E.Page r p ⟶ E'.Page r (τ p) :=
 118     fun r p => f.pageMap p (↑(r - E.r₀).toNat : WithTop ℕ) ≫ eqToHom
 119       (congrArg (fun a : ℤ => (E'.ssData (τ p)).page
 120         (↑(r - a).toNat : WithTop ℕ)) hstart)
 121   change ∀ (r : ℤ) (p : Tridegree),
 122     Φ r p ≫ E'.d r (τ p) = E.d r p ≫ Φ r (p + E.diffDeg r) ≫
 123       eqToHom (congrArg (E'.Page r) (hdegree r p)) at hcomm
 124   change RawHasDifferential E' r (τ p) (τ q) (Φ 2 p x) (Φ 2 q y)
 125   obtain ⟨hdeg, xr, yr, hx, hy, hd⟩ := h
 126   have hdeg' : τ p + E'.diffDeg r = τ q :=
 127     (hdegree r p).symm.trans (congrArg τ hdeg)
 128   refine ⟨hdeg', Φ r p xr, Φ r q yr, ?_, ?_, ?_⟩
 129   · exact SSDataMorphism.rawRepresentsOnPage_reindexed f hstart hx
 130   · exact SSDataMorphism.rawRepresentsOnPage_reindexed f hstart hy
 131   · have heq : Φ r p ≫ E'.d r (τ p) ≫ eqToHom (congrArg (E'.Page r) hdeg') =
 132         (E.d r p ≫ eqToHom (congrArg (E.Page r) hdeg)) ≫ Φ r q := by
 133       rw [← Category.assoc, hcomm, Category.assoc]
 134       subst q
 135       simp
 136     have hz := congrArg (fun a => a xr) heq
 137     simpa only [ModuleCat.comp_apply, hd] using hz

 149 The image equation can have zero source or target. -/
 150 private theorem SpectralSequenceMorphism.rawHasDifferential
 151     {E E' : SpectralSequence (ModuleCat.{v} R) Tridegree}
 152     (f : SpectralSequenceMorphism E E') {r : ℤ} {p q : Tridegree}
 153     {x : E.Page 2 p} {y : E.Page 2 q} (h : RawHasDifferential E r p q x y) :
 154     RawHasDifferential E' r p q (f.pageMap 2 p x) (f.pageMap 2 q y) := by
 155   have hdegree : ∀ (r : ℤ) (p : Tridegree),
 156       id (p + E.diffDeg r) = id p + E'.diffDeg r := by
 157     intro r p

 161     (τ := id) f.r₀_eq hdegree (fun r p => by
 162       convert f.pageMap_comm_d r p using 1 <;> rfl) h
 163   convert hf using 1 <;> rfl

 166 private theorem finite_last_nonzero_last_cycle_infinity_lift
 167     (E : KIP126.Core.SpectralSequence (ModuleCat.{v} ℤ) Tridegree)
 168     (r : ℤ) (hr : E.r₀ ≤ r) (p : Tridegree)
 169     (z : (Subobject.underlying.obj ((E.ssData p).Z ↑(r-E.r₀).toNat) : ModuleCat.{v} ℤ))
 170     (hz : E.d r p ((E.ssData p).pageπ ↑(r-E.r₀).toNat z) = 0)
 171     (hout : ∀ j : ℤ, r+1 ≤ j → E.d j p = 0) :
 172     ∃ zi : (Subobject.underlying.obj ((E.ssData p).Z ⊤) : ModuleCat.{v} ℤ),
 173       (Subobject.ofLE _ _ ((E.ssData p).Z_anti le_top)) zi = z := by
 174   classical
 175   let D := E.ssData p
 176   let n := (r-E.r₀).toNat
 177   have hsucc : (↑n : WithTop ℕ) ≤ ↑(n+1) := by exact_mod_cast Nat.le_succ n
 178   let i := Subobject.ofLE (D.Z ↑(n+1)) (D.Z ↑n) (D.Z_anti hsucc)
 179   have hker : D.pageπ ↑n z ∈ LinearMap.ker (E.d r p).hom := hz
 180   rw [←subobjectModule_kernel,E.Z_succ r p hr,subobjectModule_image] at hker
 181   obtain ⟨z',hz'⟩ := hker
 182   change D.pageπ ↑n (i z') = D.pageπ ↑n z at hz'
 183   have hzero : D.pageπ ↑n (z-i z')=0 := by rw [map_sub,hz',sub_self]
 184   have hb : ∃ b : (Subobject.underlying.obj (D.B ↑n) : ModuleCat.{v} ℤ),
 185       (Subobject.ofLE (D.B ↑n) (D.Z ↑n) (D.B_le_Z _)) b = z-i z' :=
 186     (cokernel_π_eq_zero_iff_mem_range _ _).mp hzero
 187   obtain ⟨b,hb⟩ := hb
 188   have hBZ : D.B ↑n ≤ D.Z ↑(n+1) := (D.B_mono hsucc).trans (D.B_le_Z _)
 189   let zn : (Subobject.underlying.obj (D.Z ↑(n+1)) : ModuleCat.{v} ℤ) :=
 190     z' + (Subobject.ofLE (D.B ↑n) (D.Z ↑(n+1)) hBZ) b
 191   have hnz : i zn=z := by
 192     dsimp only [zn]
 193     rw [map_add]
 194     change i z' + (Subobject.ofLE (D.B ↑n) (D.Z ↑(n+1)) hBZ ≫ i) b = z
 195     dsimp only [i]
 196     rw [Subobject.ofLE_comp_ofLE,hb]
 197     abel
 198   have hn1 : (r+1-E.r₀).toNat=n+1 := by dsimp only [n]; omega
 199   have hZ : D.Z ⊤ = D.Z ↑(n+1) := by
 200     have hh := cycles_top_eq_of_d_eq_zero E (r+1) (by omega) p hout
 201     simpa only [hn1] using hh
 202   let zi := (Subobject.ofLE (D.Z ↑(n+1)) (D.Z ⊤) hZ.ge) zn
 203   refine ⟨zi,?_⟩
 204   dsimp only [zi]
 205   rw [←CategoryTheory.comp_apply,Subobject.ofLE_comp_ofLE]
 206   exact hnz
