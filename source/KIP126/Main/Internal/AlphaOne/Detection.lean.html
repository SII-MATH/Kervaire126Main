  36 private theorem finite_sphere_detection_exists
  37     (q : ℕ) (hq : 0<q) (s t : ℤ)
  38     (x : model.classical.sphere.Page 2 (s,t))
  39     (hx : x∈KIP126.Literature.cycles model .sphere q (s,t)) (hne : x≠0) :
  40     ∃ a : BiHom (t-s) (t-0) (XModLambdaN (S_0_0 : SyntheticSpectrum) q), FiniteDetected q hq s t 0 x a := by
  41   classical
  42   let P := KIP126.Literature.Standard.infinityFiniteWindow .sphere q hq (s,t) (t-0)
  43     (by constructor <;> omega)
  44   let y : KIP126.Literature.cycles model .sphere ((q:ℤ)-t+(t-0)) (s,t) :=
  45     ⟨x,by simpa only [sub_zero,sub_add_cancel] using hx⟩
  46   let e := P.symm (KIP126.Literature.subquotientClass _ _ y)
  47   have he : HasInfinityRepresentative (model.family.quotient (model.nu.functor.obj SphereSpectrum) q) 2 (s,t,t-0)
  48       (finiteTargetLabel model .sphere q s t 0 x) e := by
  49     apply (finite_target_label_iff .sphere q hq s t 0 hq y e).mpr
  50     exact P.apply_symm_apply _
  51   have en : e≠0 := by
  52     intro hz
  53     have hp := congrArg P hz
  54     change P (P.symm _)=P 0 at hp
  55     rw [P.apply_symm_apply,map_zero] at hp
  56     have hm := (subquotientClass_eq_zero_iff _ _ y).mp hp
  57     have hb : (1:ℤ)+t-(t-0)=1 := by omega
  58     change x∈KIP126.Literature.boundaries model .sphere (1+t-(t-0)) (s,t) at hm
  59     rw [hb,boundaries_one model .sphere,Submodule.mem_bot] at hm
  60     exact hne hm
  61   let f := XModLambdaN.map model.nu.unitIso.hom q
  62   let g := XModLambdaN.map model.nu.unitIso.inv q
  63   have hgf : g≫f=𝟙 _ := by
  64     dsimp only [f,g]
  65     rw [←model.quotients.functoriality.map_comp,Iso.inv_hom_id,model.quotients.functoriality.map_id]
  66   have hfg : f≫g=𝟙 _ := by
  67     dsimp only [f,g]
  68     rw [←model.quotients.functoriality.map_comp,Iso.hom_inv_id,model.quotients.functoriality.map_id]
  69   let e' := (model.family.functor.map f).eInftyMap (s,t,t-0) e
  70   have hel : familyPageMap model.family f 2 (s,t,t-0) (finiteTargetLabel model .sphere q s t 0 x)=quotientLabel q s t 0 x :=
  71     sphere_quotient_label_from_nu q 0 s t x
  72 
  73   have he' : HasInfinityRepresentative (model.family.quotient (S_0_0 : SyntheticSpectrum) q) 2 (s,t,t-0)
  74       (quotientLabel q s t 0 x) e' := by
  75     have hh := family_infinity_representative_map f (s,t,t-0) _ _ he
  76     rw [hel] at hh
  77     exact hh
  78   have en' : e'≠0 := by
  79     intro hz
  80     apply en
  81     have hh := congrArg (fun a=>(model.family.functor.map g).eInftyMap (s,t,t-0) a) hz
  82     change ((model.family.functor.map f).eInftyMap (s,t,t-0) ≫ (model.family.functor.map g).eInftyMap (s,t,t-0)) e=_ at hh
  83     rw [←SpectralSequenceMorphism.eInftyMap_comp,←Functor.map_comp,hfg,
  84       CategoryTheory.Functor.map_id,SpectralSequenceMorphism.eInftyMap_id] at hh
  85     simpa only [ConcreteCategory.id_apply,map_zero] using hh
  86   exact detection_exists_of_infinity (model.convergence (.quotient q .sphere)) (s,t,t-0) _ e' he' en'

  94     (he : HasInfinityRepresentative (model.family.quotient (S_0_0 : SyntheticSpectrum) q) 2 (s,t,t-0)

  99   have hl := sphere_quotient_label_to_nu q 0 s t x

 114 private theorem finite_sphere_detected_rho
 115     (i j : ℕ) (hi : 0 < i) (hj : 0 < j) (hij : i ≤ j) (s t : ℤ)
 116     (x : model.classical.sphere.Page 2 (s,t))
 117     (hx : x∈KIP126.Literature.cycles model .sphere i (s,t)) (hne : x≠0)
 118     (a : BiHom (t-s) (t-0) (XModLambdaN (S_0_0 : SyntheticSpectrum) j))
 119     (ha : FiniteDetected j hj s t 0 x a) :
 120     FiniteDetected i hi s t 0 x (a≫(model.quotients.tower S_0_0).rho i j hij) := by
 121   classical
 122   let rho := (model.quotients.tower S_0_0).rho i j hij
 123   have hl : familyPageMap model.family rho 2 (s,t,t-0) (quotientLabel j s t 0 x)=quotientLabel i s t 0 x := by
 124     change familyPageMap model.family rho 2 (s,t,t-0)
 125       (familyPageMap model.family (XModLambdaN.incl (S_0_0 : SyntheticSpectrum) j) 2 (s,t,t-0) (model.comparison.sphereE2 s t 0 x))=_
 126     calc
 127       _ = familyPageMap model.family (XModLambdaN.incl (S_0_0 : SyntheticSpectrum) j≫rho) 2 (s,t,t-0) (model.comparison.sphereE2 s t 0 x) :=
 128         ConcreteCategory.congr_hom (family_page_comp (XModLambdaN.incl (S_0_0 : SyntheticSpectrum) j) rho 2 (s,t,t-0)) _
 129       _ = quotientLabel i s t 0 x := by
 130         dsimp only [rho]

 132         rfl
 133   have hd := detects_map (.quotient j .sphere) (.quotient i .sphere) rho (s,t,t-0) _ a ha.1
 134   have hd' : Detects (model.convergence (.quotient i .sphere)) (s,t,t-0)
 135       (quotientLabel i s t 0 x) (a≫rho) := hl ▸ hd
 136   refine ⟨hd',?_⟩
