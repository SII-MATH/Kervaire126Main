  13 private theorem page_subsingleton_of_le
  14     (E : KIP126.Core.SpectralSequence (ModuleCat.{v} ℤ) (ℤ×ℤ))
  15     (r t : ℤ) (p : ℤ×ℤ) (_hr : E.r₀ ≤ r) (hrt : r≤t)
  16     (hzero : Subsingleton (E.Page r p)) : Subsingleton (E.Page t p) := by
  17   let A := E.ssData p
  18   let n : WithTop ℕ := ↑(r-E.r₀).toNat
  19   let m : WithTop ℕ := ↑(t-E.r₀).toNat
  20   have hnm : n≤m := by
  21     dsimp only [n,m]
  22     exact_mod_cast Int.toNat_le_toNat (by omega : r-E.r₀≤t-E.r₀)
  23   have hz (z : (Subobject.underlying.obj (A.Z m) : ModuleCat ℤ)) : A.pageπ m z=0 := by
  24     let i := Subobject.ofLE (A.Z m) (A.Z n) (A.Z_anti hnm)
  25     have hzn : A.pageπ n (i z)=0 := hzero.elim _ _
  26     have hb := (subobject_cokernel_π_eq_zero_iff (A.B n) (A.Z n) (A.B_le_Z n) (i z)).mp hzn
  27     change (i ≫ (A.Z n).arrow) z ∈ _ at hb
  28     simp only [i,Subobject.ofLE_arrow] at hb
  29     apply (subobject_cokernel_π_eq_zero_iff (A.B m) (A.Z m) (A.B_le_Z m) z).mpr
  30     exact (ModuleCat.subobjectModule A.V).monotone (A.B_mono hnm) hb
  31   haveI : Epi (A.pageπ m) := inferInstanceAs (Epi (cokernel.π _))
  32   have hall (x : E.Page t p) : x=0 := by
  33     obtain ⟨z,rfl⟩ := (ModuleCat.epi_iff_surjective (A.pageπ m)).mp inferInstance x
  34     exact hz z
  35   exact ⟨fun x y => (hall x).trans (hall y).symm⟩
  36 
  37 end
