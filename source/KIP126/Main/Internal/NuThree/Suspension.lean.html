  19 private theorem desuspend_bijective (X : C) (n : ℤ) :
  20     Function.Bijective (AbstractAdamsTower.desuspend X n) := by
  21   let e := (shiftFunctorAdd' C n (-1) (n - 1) (by omega)).app SphereSpectrum
  22   let d := (shiftFunctorCompIsoId C 1 (-1) (by omega)).app X
  23   constructor
  24   · intro a b h
  25     change e.hom ≫ (shiftFunctor C (-1 : ℤ)).map a ≫ d.hom =
  26       e.hom ≫ (shiftFunctor C (-1 : ℤ)).map b ≫ d.hom at h
  27     apply (shiftFunctor C (-1 : ℤ)).map_injective
  28     apply (cancel_epi e.hom).mp
  29     apply (cancel_mono d.hom).mp
  30     simpa only [Category.assoc] using h
  31   · intro b
  32     refine ⟨(shiftFunctor C (-1 : ℤ)).preimage (e.inv ≫ b ≫ d.inv), ?_⟩
  33     change e.hom ≫ (shiftFunctor C (-1 : ℤ)).map
  34       ((shiftFunctor C (-1 : ℤ)).preimage _) ≫ d.hom = b
  35     rw [Functor.map_preimage]
  36     simp only [Category.assoc, Iso.inv_hom_id, Category.comp_id, Iso.hom_inv_id_assoc]

  39 private theorem desuspend_postcompose {X Y : C} (f : X ⟶ Y) (n : ℤ)
  40     (a : HomotopyGroup n (X⟦(1 : ℤ)⟧)) :
  41     AbstractAdamsTower.desuspend Y n (a ≫ f⟦(1 : ℤ)⟧') = AbstractAdamsTower.desuspend X n a ≫ f := by
  42   change _ ≫ (shiftFunctor C (-1 : ℤ)).map (a ≫ f⟦(1 : ℤ)⟧') ≫ _ = _
  43   rw [Functor.map_comp]
  44   have h := (shiftFunctorCompIsoId C 1 (-1) (by omega)).hom.naturality f
  45   change (shiftFunctor C (-1 : ℤ)).map (f⟦(1 : ℤ)⟧') ≫ _ = _ ≫ f at h
  46   simp only [Category.assoc]
  47   rw [h]
  48   simp only [AbstractAdamsTower.desuspend, AddMonoidHom.coe_mk, ZeroHom.coe_mk, Category.assoc]

  50 private theorem raw_cycles_iff [HasFunctorialCofiber (C:=C)] {H : C} {unit : SphereSpectrum ⟶ H}
  51     {X : C} (T : AbstractAdamsTower unit X) (r : ℕ) (hr : 1≤r) (s t : ℤ)
  52     (a : T.FirstPage (s,t)) :
  53     a∈T.cycles r hr (s,t) ↔
  54     ∃ c : HomotopyGroup (t-s) ((T.stageAt (s+r))⟦(1:ℤ)⟧),
  55       c≫(T.mapAt (show s+1≤s+r by omega))⟦(1:ℤ)⟧' =
  56       a≫HasFunctorialCofiber.cofibδ (T.mapAt (show s≤s+1 by omega)) := by
  57   constructor
  58   · rintro ⟨b,hb⟩
  59     obtain ⟨c,hc⟩ := (desuspend_bijective (T.stageAt (s+r)) (t-s)).2 b
  60     refine ⟨c,?_⟩
  61     apply (desuspend_bijective (T.stageAt (s+1)) (t-s)).1
  62     rw [desuspend_postcompose,hc]
  63     exact hb
  64   · rintro ⟨c,hc⟩
  65     refine ⟨AbstractAdamsTower.desuspend (T.stageAt (s+r)) (t-s) c,?_⟩
  66     change _≫T.mapAt (show s+1≤s+r by omega)=
  67       AbstractAdamsTower.desuspend (T.stageAt (s+1)) (t-s)
  68         (a≫HasFunctorialCofiber.cofibδ (T.mapAt (show s≤s+1 by omega)))
  69     rw [←desuspend_postcompose,hc]

  71 private theorem desuspend_mem_raw_cycles (X : ClassicalObject)
  72     (r : ℕ) (hr : 1≤r) (s t : ℤ)
  73     (a : (model.classicalTower.tower ((X.obj model.auxiliary)⟦(1:ℤ)⟧)).FirstPage (s,t))
  74     (ha : a∈(model.classicalTower.tower ((X.obj model.auxiliary)⟦(1:ℤ)⟧)).cycles r hr (s,t)) :
  75     (model.classicalSuspension X).desuspendFirstPage s t a ∈
  76       (model.classicalTower.tower (X.obj model.auxiliary)).cycles r hr (s,t-1) := by
  77   let S := model.classicalSuspension X
  78   let T := model.classicalTower.tower (X.obj model.auxiliary)
  79   obtain ⟨c,hc⟩ := (raw_cycles_iff _ r hr s t a).mp ha
  80   let R := shiftFunctorComm Spectrum (1:ℤ) (1:ℤ)
  81   let e := c≫(S.tower (s+r)).hom⟦(1:ℤ)⟧'≫R.inv.app (T.stageAt (s+r))
  82   have he : e≫(T.mapAt (show s+1≤s+r by omega))⟦(1:ℤ)⟧'⟦(1:ℤ)⟧' =
  83       -(a≫(S.layer s).hom≫(HasFunctorialCofiber.cofibδ
  84         (T.mapAt (show s≤s+1 by omega)))⟦(1:ℤ)⟧') := by
  85     dsimp only [e]
  86     simp only [Category.assoc]
  87     erw [←R.inv.naturality]
  88     simp only [Functor.comp_map]
  89     rw [←Functor.map_comp_assoc,S.tower_comm,Functor.map_comp]
  90     simp only [←Category.assoc]
  91     erw [hc]
  92     simp only [Category.assoc]
  93     have hs := S.layer_connecting s
  94     have hs' := congrArg (fun f=>f≫R.inv.app (T.stageAt (s+1))) hs
  95     dsimp only [T,AbstractAdamsTower.layer] at hs'
  96     simp only [R,Category.assoc,Iso.hom_inv_id_app,Preadditive.neg_comp] at hs'
  97     erw [Category.comp_id] at hs'
  98     erw [hs']
  99     simp only [Preadditive.comp_neg,neg_neg]
 100     rfl
 101   apply (raw_cycles_iff T r hr s (t-1) _).mpr
 102   let j := eqToHom (congrArg (Sphere (C:=Spectrum)) (show (t-1)-s=(t-s)-1 by omega))
 103   refine ⟨-(j≫AbstractAdamsTower.desuspend ((T.stageAt (s+r))⟦(1:ℤ)⟧) (t-s) e),?_⟩
 104   rw [Preadditive.neg_comp,Category.assoc,←desuspend_postcompose,he,map_neg]
 105   simp only [Preadditive.comp_neg,neg_neg]
 106   rw [←Category.assoc a]
 107   erw [desuspend_postcompose]
 108   exact (Category.assoc _ _ _).symm

 110 private theorem initial_class_cycle_iff (D : SSData (ModuleCat.{0} ℤ))
 111     (n : WithTop ℕ) (z : (Subobject.underlying.obj (D.Z 0) : ModuleCat ℤ)) :
 112     (∃ v : (Subobject.underlying.obj (D.Z n) : ModuleCat ℤ),
 113       (Subobject.ofLE (D.Z n) (D.Z 0) (D.Z_anti bot_le) ≫ D.pageπ 0) v=D.pageπ 0 z) ↔
 114       (D.Z 0).arrow z∈(ModuleCat.subobjectModule D.V) (D.Z n) := by
 115   let i := Subobject.ofLE (D.Z n) (D.Z 0) (D.Z_anti bot_le)
 116   constructor
 117   · rintro ⟨v,hv⟩
 118     have hz : D.pageπ 0 (z-i v)=0 := by
 119       rw [map_sub]
 120       exact sub_eq_zero.mpr hv.symm
 121     have hb := (subobject_cokernel_π_eq_zero_iff (D.B 0) (D.Z 0) (D.B_le_Z 0) _).mp hz
 122     have hb' : (D.Z 0).arrow (z-i v)∈(ModuleCat.subobjectModule D.V) (D.Z n) := (ModuleCat.subobjectModule D.V).monotone
 123       ((D.B_mono le_top).trans ((D.B_le_Z ⊤).trans (D.Z_anti le_top))) hb
 124     have hv' : (D.Z 0).arrow (i v)∈(ModuleCat.subobjectModule D.V) (D.Z n) := by
 125       change (i≫(D.Z 0).arrow) v∈_
 126       rw [Subobject.ofLE_arrow]
 127       exact ⟨v,rfl⟩
 128     have h := ((ModuleCat.subobjectModule D.V) (D.Z n)).add_mem hb' hv'
 129     rw [map_sub,sub_add_cancel] at h
 130     exact h
 131   · rintro ⟨v,hv⟩
 132     refine ⟨v,?_⟩
 133     have he : i v=z := by
 134       apply (ModuleCat.mono_iff_injective (D.Z 0).arrow).mp inferInstance
 135       change (i≫(D.Z 0).arrow) v=(D.Z 0).arrow z
 136       rw [Subobject.ofLE_arrow]
 137       exact hv
 138     change D.pageπ 0 (i v)=D.pageπ 0 z
 139     rw [he]

 141 private theorem class_cycle_iff (X : ClassicalObject)
 142     (c : ℤ) (p : ℤ×ℤ)
 143     (z : (Subobject.underlying.obj (((model.classical.obj (X.obj model.auxiliary)).ssData p).Z 0) : ModuleCat ℤ)) :
 144     cycleLabel model X 0 p z∈cycles model X c p ↔
 145       ((((model.classicalTower.sequence (X.obj model.auxiliary)).ambient p).hom
 146         ((((model.classical.obj (X.obj model.auxiliary)).ssData p).Z 0).arrow z)).val)∈
 147           (model.classicalTower.tower (X.obj model.auxiliary)).cycles ((c-1).toNat+2) (by omega) p := by
 148   let D := (model.classical.obj (X.obj model.auxiliary)).ssData p
 149   rw [←(model.classicalTower.sequence (X.obj model.auxiliary)).cycles p (c-1).toNat]
 150   rw [←initial_class_cycle_iff D (↑(c-1).toNat) z]
 151   change (∃v, cycleLabel model X (↑(c-1).toNat) p v=cycleLabel model X 0 p z) ↔ _
 152   simp only [cycleLabel,Subobject.ofLE_refl,Category.id_comp]
 153   constructor
 154   · rintro ⟨v,hv⟩
 155     refine ⟨v,?_⟩
 156     apply (ModuleCat.mono_iff_injective (eqToHom _)).mp inferInstance
 157     exact hv
 158   · rintro ⟨v,hv⟩
 159     refine ⟨v,?_⟩
 160     exact congrArg (fun a=>(eqToHom _ : D.page 0 ⟶ (model.classical.obj (X.obj model.auxiliary)).Page 2 p) a) hv

 162 private theorem desuspendsClass_isCycle (X : ClassicalObject)
 163     (level s t : ℤ)
 164     (sx : (model.classical.obj ((X.obj model.auxiliary)⟦(1:ℤ)⟧)).Page 2 (s,t))
 165     (x : (model.classical.obj (X.obj model.auxiliary)).Page 2 (s,t-1))
 166     (h : (model.classicalSuspension X).DesuspendsClass s t sx x)
 167     (hc : IsCycle model (.shift 1 X) level (s,t) sx) :
 168     IsCycle model X level (s,t-1) x := by
 169   obtain ⟨a,b,ha,hb,hab⟩ := h
 170   refine ⟨hc.1,?_⟩
 171   have ha' : cycleLabel model (.shift 1 X) 0 (s,t) a=sx := by
 172     simpa only [cycleLabel,Subobject.ofLE_refl,Category.id_comp,ClassicalObject.obj] using ha
 173   have hb' : cycleLabel model X 0 (s,t-1) b=x := by
 174     simpa only [cycleLabel,Subobject.ofLE_refl,Category.id_comp] using hb
 175   rw [←hb']
 176   apply (class_cycle_iff X level (s,t-1) b).mpr
 177   rw [←hab]
 178   exact desuspend_mem_raw_cycles X ((level-1).toNat+2) (by omega) s t _
 179     ((class_cycle_iff (.shift 1 X) level (s,t) a).mp (ha'.symm ▸ hc.2))
