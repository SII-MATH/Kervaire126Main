  18 private theorem classical_page_reaches (X Y : ClassicalObject)
  19     (f : X.obj model.auxiliary ⟶ Y.obj model.auxiliary) (r : ℤ) (s t : ℤ)
  20     (x : (model.classical.obj (X.obj model.auxiliary)).Page 2 (s,t))
  21     (hx : ReachesPage (model.classical.obj (X.obj model.auxiliary)) r (s,t) x) :
  22     ReachesPage (model.classical.obj (Y.obj model.auxiliary)) r (s,t)
  23       (model.classical.pageMap f 2 (s,t) x) := by
  24   obtain ⟨xr,hxr⟩ := hx
  25   exact ⟨(model.classical.map f).pageMap r (s,t) xr,(model.classical.map f).representsOnPage hxr⟩

  27 private theorem finite_zero_extension_8
  28     (X Y : ClassicalObject) (f : X.obj model.auxiliary ⟶ Y.obj model.auxiliary)
  29     (he : normalizedExponent model.classicalTower f=0)
  30     (x : (model.classical.obj (X.obj model.auxiliary)).Page 2 (8,134))
  31     (hx : x∈cycles model X 2 (8,134)) :
  32     FiniteExtension model X Y f 3 0 8 134 x (model.classical.pageMap f 2 (8,134) x) := by
  33   let y := model.classical.pageMap f 2 (8,134) x
  34   have hlower : (normalizedExponent model.classicalTower f:ℤ)≤0 := by omega
  35   obtain ⟨es,hes⟩ := normalized_zero_source_infinity_8 X Y f he 2 (by decide) x hx
  36   let et := (model.family.map (finiteExtensionMap model X Y f 2)).eInftyMap
  37     (8,134,134+normalizedExponent model.classicalTower f) es
  38   have hm := normalized_zero_map_label_8 X Y f he 2 x
  39   have ht : HEq (finiteTargetForExtension model X Y f 2 0 8 134 hlower y)
  40       (finiteTargetLabel model Y 2 8 134 0 y) :=
  41     zero_exponent_target_label_8 Y 2 _ he y
  42   have heq := eq_of_heq (hm.trans ht.symm)
  43   have hmap := family_infinity_representative_map (finiteExtensionMap model X Y f 2)
  44     (8,134,134+normalizedExponent model.classicalTower f) _ es hes
  45   have hrep : HasInfinityRepresentative (model.family.obj ((SyntheticObject.quotient 2 (.nu Y)).obj model.nu model.auxiliary)) 2
  46       (8,134,134+normalizedExponent model.classicalTower f)
  47       (finiteTargetForExtension model X Y f 2 0 8 134 hlower y) et := heq ▸ hmap
  48   have hxR : ReachesPage (model.classical.obj (X.obj model.auxiliary)) 3 (8,134) x :=
  49     (isCycle_iff_reaches model X 3 (by decide) (8,134) x).mp ⟨by decide,hx⟩
  50   have hyR := classical_page_reaches X Y f 3 8 134 x hxR
  51   refine ⟨{
  52     page_ge_two := by decide
  53     length_lower := hlower
  54     length_upper := by omega
  55     source_cycle := hxR
  56     target_cycle := ?_
  57     source := es
  58     target := et
  59     source_rep := hes
  60     target_rep := hrep
  61     equation := ?_ }⟩
  62   · change ReachesPage (model.classical.obj (Y.obj model.auxiliary))
  63       ((3:ℤ)-0+normalizedExponent model.classicalTower f) (8,134) y
  64     rw [he]
  65     exact hyR
  66   · exact towerExtensionEquation_zero
  67       (model.convergentMap (.quotient 2 (extensionSourceObject model X Y f))
  68         (.quotient 2 (.nu Y)) (finiteExtensionMap model X Y f 2))
  69       (126,134+normalizedExponent model.classicalTower f) 8 es
