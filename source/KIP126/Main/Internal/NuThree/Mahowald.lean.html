  22 private theorem boundary_permanent (X : ClassicalObject) (p : ℤ×ℤ) (c : ℤ)
  23     (x : Ambient model X p) (hx : x∈boundaries model X c p) : IsPermanent model X p x := by
  24   obtain ⟨b,rfl⟩ := hx
  25   let D := (model.classical.obj (X.obj model.auxiliary)).ssData p
  26   have h : D.B (↑(c-1).toNat)≤D.Z ⊤ := (D.B_mono le_top).trans (D.B_le_Z ⊤)
  27   refine ⟨(Subobject.ofLE _ _ h) b,?_⟩
  28   change (Subobject.ofLE _ _ h ≫ cycleLabel model X ⊤ p) b=boundaryLabel model X _ p b
  29   simp only [cycleLabel,boundaryLabel,D,←Category.assoc,Subobject.ofLE_comp_ofLE]
