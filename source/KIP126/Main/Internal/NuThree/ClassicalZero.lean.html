  15 private theorem classical_zero_page (p : ℤ × ℤ) :
  16     Subsingleton ((model.classical.obj (0 : Spectrum)).Page 2 p) := by
  17   let B := KIP126.AdamsSynthetic.Standard.closedBackground
  18   letI := B.symmetric
  19   let H := model.coefficients.object
  20   letI := B.tensorShift H
  21   letI := B.tensorExact H
  22   let T := model.classicalTower.tower (0 : Spectrum)
  23   have hz (n : ℕ) : IsZero (T.stage n) := by
  24     induction n with
  25     | zero => exact (isZero_zero Spectrum).of_iso T.zeroIso
  26     | succ n ih =>
  27       have hh : IsZero (H ⊗ T.stage n) :=
  28         ((tensorRight H).map_isZero ih).of_iso (β_ H (T.stage n))
  29       have hc : IsZero (HasFunctorialCofiber.cofib (T.map (Nat.le_succ n))) :=
  30         hh.of_iso (T.layerIso n)
  31       exact Triangle.isZero₁_of_isZero₂₃ _
  32         (HasFunctorialCofiber.cofib_distinguished (T.map (Nat.le_succ n))) ih hc
  33   have hl : IsZero (T.layer p.1) := by
  34     apply Triangle.isZero₃_of_isZero₁₂ _
  35       (HasFunctorialCofiber.cofib_distinguished (T.mapAt (show p.1 ≤ p.1+1 by omega)))
  36     · exact hz (p.1+1).toNat
  37     · exact hz p.1.toNat
  38   haveI : Subsingleton (T.FirstPage p) := ⟨fun a b => hl.eq_of_tgt a b⟩
  39   haveI : Subsingleton (T.cycles 2 (by decide) p) := inferInstance
  40   let D := (model.classical.obj (0 : Spectrum)).ssData p
  41   let e := (model.classicalTower.sequence (0 : Spectrum)).ambient p
  42   haveI : Subsingleton D.V := Function.Injective.subsingleton e.toLinearEquiv.injective
  43   haveI : Subsingleton (Subobject.underlying.obj (D.Z 0) : ModuleCat ℤ) :=
  44     Function.Injective.subsingleton ((ModuleCat.mono_iff_injective (D.Z 0).arrow).mp inferInstance)
  45   letI : Epi (D.pageπ 0) := coequalizer.π_epi
  46   haveI : Subsingleton (D.page 0) :=
  47     Function.Surjective.subsingleton ((ModuleCat.epi_iff_surjective (D.pageπ 0)).mp inferInstance)
  48   simpa only [KIP126.Core.SpectralSequence.Page,model.classical.firstPage,sub_self,
  49     Int.toNat_zero,Nat.cast_zero] using (inferInstance : Subsingleton (D.page 0))

  51 private theorem classical_page_zero {X Y : Spectrum} (p : ℤ × ℤ)
  52     (x : (model.classical.obj X).Page 2 p) : model.classical.pageMap (0 : X ⟶ Y) 2 p x=0 := by
  53   have hf : (0 : X ⟶ Y) = (0 : X ⟶ (0 : Spectrum)) ≫ (0 : (0 : Spectrum) ⟶ Y) := by simp
  54   rw [hf]
  55   change ((model.classical.map ((0 : X ⟶ (0 : Spectrum)) ≫ (0 : (0 : Spectrum) ⟶ Y))).pageMap 2 p) x=0
  56   rw [model.classical.map_comp]
  57   rw [SpectralSequenceMorphism.pageMap_comp,ConcreteCategory.comp_apply]
  58   haveI := classical_zero_page p
  59   rw [Subsingleton.elim (((model.classical.map (0 : X ⟶ (0 : Spectrum))).pageMap 2 p) x) 0,map_zero]
