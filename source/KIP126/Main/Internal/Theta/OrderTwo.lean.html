  22 private theorem classical_infinity_double (p : ℤ × ℤ)
  23     (x : (model.classical.sphere.ssData p).eInfty) : x+x=0 := by
  24   let D := model.classical.sphere.ssData p
  25   let n : WithTop ℕ := ↑(2-model.classical.sphere.r₀).toNat
  26   have hpage (y : D.page n) : y+y=0 := by
  27     change model.classical.sphere.Page 2 p at y
  28     letI := model.sphereAlgebra.scalar p
  29     change y+y=0
  30     calc
  31       y+y = (1 : ZMod 2) • y + (1 : ZMod 2) • y := by simp only [one_smul]
  32       _ = ((1 : ZMod 2)+1) • y := (add_smul _ _ _).symm
  33       _ = 0 := by rw [show (1 : ZMod 2)+1=0 from rfl,zero_smul]
  34   haveI : Epi (D.pageπ ⊤) := inferInstanceAs (Epi (cokernel.π _))
  35   obtain ⟨z,rfl⟩ := (ModuleCat.epi_iff_surjective (D.pageπ ⊤)).mp inferInstance x
  36   have h := D.infinity_projection_eq_of_page_projection_eq n (z+z) 0 (by
  37     simp only [map_add, map_zero]
  38     exact hpage _)
  39   simpa only [map_add,map_zero] using h

  41 private theorem classical_doubling_filtration (s : ℤ)
  42     (a : HomotopyGroup (C := Spectrum) 62 SphereSpectrum)
  43     (ha : a ∈ (model.classicalTower.tower SphereSpectrum).filtrationSubmodule s 62) :
  44     a+a ∈ (model.classicalTower.tower SphereSpectrum).filtrationSubmodule (s+1) 62 := by
  45   let F := (model.classicalTower.tower SphereSpectrum).filtration
  46   let e : (model.classical.sphere.ssData (s,62+s)).eInfty ≅ F.associatedGraded s 62 := by
  47     simpa only [add_sub_cancel_right,F,ClassicalObject.obj,ClassicalAdamsFamily.sphere]
  48       using (model.classicalConvergence .sphere).identification (s,62+s)
  49   have hdouble (x : F.associatedGraded s 62) : x+x=0 := by
  50     apply (ModuleCat.mono_iff_injective e.inv).mp inferInstance
  51     simpa only [map_add,map_zero] using classical_infinity_double (s,62+s) (e.inv x)
  52   have hm : a ∈ ModuleCat.subobjectModule _ (F.F s 62) := by
  53     simpa only [F,AbstractAdamsTower.filtration,OrderIso.apply_symm_apply] using ha
  54   obtain ⟨b,hb⟩ := hm
  55   have hg : F.toAssociatedGraded s 62 (b+b)=0 := by
  56     rw [map_add]
  57     exact hdouble _
  58   have hn := (F.toAssociatedGraded_eq_zero_iff s 62 (b+b)).mp hg
  59   rw [map_add,hb] at hn
  60   simpa only [F,AbstractAdamsTower.filtration,OrderIso.apply_symm_apply] using hn

  62 private theorem classical_stem62_tail_step (s : ℤ) (hs : 34≤s)
  63     (a : HomotopyGroup (C := Spectrum) 62 SphereSpectrum)
  64     (ha : a ∈ (model.classicalTower.tower SphereSpectrum).filtrationSubmodule s 62) :
  65     a ∈ (model.classicalTower.tower SphereSpectrum).filtrationSubmodule (s+1) 62 := by
  66   let D := model.classical.sphere.ssData (s,62+s)
  67   let n : WithTop ℕ := ↑(2-model.classical.sphere.r₀).toNat
  68   have hp : Subsingleton (D.page n) :=
  69     KIP126.Literature.Standard.sphereVanishing s (62+s) (by omega) (by omega)
  70   have hinf : Subsingleton D.eInfty := by
  71     have hall (e : D.eInfty) : e=0 := by
  72       haveI : Epi (D.pageπ ⊤) := inferInstanceAs (Epi (cokernel.π _))
  73       obtain ⟨z,rfl⟩ := (ModuleCat.epi_iff_surjective (D.pageπ ⊤)).mp inferInstance e
  74       have h := D.infinity_projection_eq_of_page_projection_eq n z 0 (hp.elim _ _)
  75       simpa only [map_zero] using h
  76     exact ⟨fun x y => (hall x).trans (hall y).symm⟩
  77   let F := (model.classicalTower.tower SphereSpectrum).filtration
  78   let e : D.eInfty ≅ F.associatedGraded s 62 := by
  79     simpa only [D,add_sub_cancel_right,F,ClassicalObject.obj,ClassicalAdamsFamily.sphere]
  80       using (model.classicalConvergence .sphere).identification (s,62+s)
  81   letI : Subsingleton (F.associatedGraded s 62) :=
  82     (ModuleCat.mono_iff_injective e.inv).mp inferInstance |>.subsingleton
  83   have hm : a ∈ ModuleCat.subobjectModule _ (F.F s 62) := by
  84     simpa only [F,AbstractAdamsTower.filtration,OrderIso.apply_symm_apply] using ha
  85   obtain ⟨b,hb⟩ := hm
  86   have hg : F.toAssociatedGraded s 62 b=0 := Subsingleton.elim _ _
  87   have hn := (F.toAssociatedGraded_eq_zero_iff s 62 b).mp hg
  88   rw [hb] at hn
  89   simpa only [F,AbstractAdamsTower.filtration,OrderIso.apply_symm_apply] using hn

  91 /-- Remove the odd-primary part by an odd multiple, preserving every mod-two
  92 label. Only a finite filtration argument is used, not integral separation. -/
  93 private theorem classical_stem62_odd_multiple_primary
  94     (a : HomotopyGroup (C := Spectrum) 62 SphereSpectrum) :
  95     ∃ n : ℕ, Odd n ∧ ∃ k : ℕ, (2^k : ℕ) • (n • a)=0 := by
  96   have hpower (k : ℕ) : (2^k : ℕ) • a ∈
  97       (model.classicalTower.tower SphereSpectrum).filtrationSubmodule k 62 := by
  98     induction k with
  99     | zero =>
 100       simp only [pow_zero,one_nsmul,Nat.cast_zero]
 101       refine ⟨a ≫ (model.classicalTower.tower SphereSpectrum).zeroIso.inv,?_⟩
 102       change (a ≫ (model.classicalTower.tower SphereSpectrum).zeroIso.inv) ≫
 103         (model.classicalTower.tower SphereSpectrum).projection 0=a
 104       simp only [AbstractAdamsTower.projection,Int.toNat_zero,
 105         (model.classicalTower.tower SphereSpectrum).map_self]
 106       erw [Category.assoc,Category.id_comp,Iso.inv_hom_id,Category.comp_id]
 107     | succ k ih =>
 108       have h := classical_doubling_filtration k ((2^k : ℕ) • a) ih
 109       simpa only [pow_succ,mul_nsmul,two_nsmul,Nat.cast_add,Nat.cast_one] using h
 110   let b := (2^34 : ℕ) • a
 111   have hall (s : ℕ) (hs : 34≤s) :
 112       b ∈ (model.classicalTower.tower SphereSpectrum).filtrationSubmodule s 62 := by
 113     induction s,hs using Nat.le_induction with
 114     | base => exact hpower 34
 115     | succ s hs ih =>
 116       simpa only [Nat.cast_add,Nat.cast_one] using classical_stem62_tail_step s (by omega) b ih
 117   obtain ⟨n,hn,_,hz⟩ := KIP126.AdamsSynthetic.Standard.classical_sphere_separated 62 b (by
 118     intro s
 119     by_cases hs : 34≤s
 120     · exact hall s hs
 121     · exact (model.classicalTower.tower SphereSpectrum).filtrationSubmodule_antitone 62
 122         (by omega : (s:ℤ)≤34) (hpower 34))
 123   refine ⟨n,hn,34,?_⟩
 124   simpa only [b,smul_comm n (2^34 : ℕ) a] using hz

 192 /-- Select an order-two representative of the same first-quotient class.
 193 An odd multiple removes the integral odd-primary component. -/
 194 private theorem first_quotient_order_two_lift
 195     (x : model.classical.sphere.Page 2 (8,70))
 196     (b₀ : BiHom 62 70 (S_0_0 : SyntheticSpectrum))
 197     (hb₀ : KIP126.Literature.sphereFirstQuotient model 8 70 (quotientClass 1 b₀)=x) :
 198     ∃ b : BiHom 62 70 (S_0_0 : SyntheticSpectrum),
 199       KIP126.Literature.sphereFirstQuotient model 8 70 (quotientClass 1 b)=x ∧ b+b=0 := by
 200   obtain ⟨n,hn,hprimary⟩ := classical_stem62_odd_multiple_primary
 201     (KIP126.Literature.Standard.realizeSphere 62 70 b₀)
 202   refine ⟨n • b₀,?_,?_⟩
 203   · have hx : x+x=0 := by
 204       letI := model.sphereAlgebra.scalar (8,70)
 205       calc
 206         x+x = (1 : ZMod 2) • x + (1 : ZMod 2) • x := by simp only [one_smul]
 207         _ = ((1 : ZMod 2)+1) • x := (add_smul _ _ _).symm
 208         _ = 0 := by rw [show (1 : ZMod 2)+1=0 from rfl,zero_smul]
 209     have hq : quotientClass 1 (n • b₀)=n • quotientClass 1 b₀ := by
 210       simp only [quotientClass,Preadditive.nsmul_comp]
 211     rw [hq,map_nsmul,hb₀]
 212     obtain ⟨k,rfl⟩ := hn
 213     rw [add_nsmul,one_nsmul,mul_nsmul,two_nsmul,hx,nsmul_zero,zero_add]
 214   · apply two_torsion_62_70_of_realization_primary
 215     simpa only [map_nsmul] using hprimary
