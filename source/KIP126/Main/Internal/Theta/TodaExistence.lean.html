  33 private theorem synthetic_theta_b_bracket_nonempty
  34     (coh : BiShiftCoherence SyntheticSpectrum)
  35     (theta : BiHom 62 64 (S_0_0 : SyntheticSpectrum)) (b : BiHom 62 70 (S_0_0 : SyntheticSpectrum))
  36     (htheta : theta+theta=0) (hb : b+b=0) :
  37     ∃ z : BiHom 125 134 (S_0_0 : SyntheticSpectrum), TripleToda theta syntheticTwo b z := by
  38   let F := SyntheticCategory.biShift (Syn:=SyntheticSpectrum) (62,70)
  39   let f := F.map theta
  40   let g := (SyntheticCategory.biShift_comp (0,0) (62,70)).inv.app (S_0_0:SyntheticSpectrum) ≫ F.map syntheticTwo
  41   have hg : g=2 • 𝟙 (Smn 62 70 : SyntheticSpectrum) := by
  42     have hc := coh.left_unit (62,70) (S_0_0:SyntheticSpectrum)
  43     change (biShiftAddIso (0,0) (62,70) (62,70) (by decide)).hom.app _=_ at hc
  44     have he : (SyntheticCategory.biShift_comp (0,0) (62,70)).hom.app (S_0_0:SyntheticSpectrum)=
  45         F.map (SyntheticCategory.biShift_zero.hom.app (S_0_0:SyntheticSpectrum)) := by
  46       simpa only [biShiftAddIso,Iso.trans_hom,NatTrans.comp_app,eqToIso.hom,
  47         eqToHom_app,eqToHom_refl,Category.comp_id,F] using hc
  48     dsimp only [g,syntheticTwo]
  49     rw [Functor.map_comp,←he,←Category.assoc,Iso.inv_hom_id_app,Category.id_comp]
  50     simp only [Functor.map_nsmul]
  51     change 2 • F.map (𝟙 (S_0_0 : SyntheticSpectrum)) = 2 • 𝟙 (F.obj (S_0_0 : SyntheticSpectrum))
  52     rw [CategoryTheory.Functor.map_id]
  53   have hfg : f≫g=0 := by
  54     rw [hg]
  55     simp only [two_smul,Preadditive.comp_add]
  56     erw [Category.comp_id]
  57     change F.map theta+F.map theta=0
  58     rw [←Functor.map_add,htheta,Functor.map_zero]
  59   have hgb : g≫b=0 := by
  60     rw [hg]
  61     simp only [two_smul,Preadditive.add_comp]
  62     erw [Category.id_comp]
  63     exact hb
  64   obtain ⟨v,hv⟩ := Toda.exists_relation f g b hfg hgb
  65   refine ⟨(tripleTodaSource 62 64 0 0 62 70).hom≫v,?_⟩
  66   unfold TripleToda
  67   rw [Iso.inv_hom_id_assoc]
  68   exact hv
