  25 private def SphereMossCrossing (r : ℤ) (p : ℤ × ℤ) : Prop :=
  26   ∃ m q : ℤ, r < m ∧ 0 ≤ q ∧ q < p.1 - (r - 1) ∧ p.1 < q + m ∧
  27     Normalized.sphere.d m (q, q + (p.2 - p.1) + 1) ≠ 0
