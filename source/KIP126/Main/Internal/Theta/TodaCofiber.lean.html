  31 private theorem relation_from_middle_cofiber
  32     {X Y Z W Q : SyntheticSpectrum} {f : X ⟶ Y} {g : Y ⟶ Z} {h : Z ⟶ W}
  33     {i : Z ⟶ Q} {p : Q ⟶ Y⟦(1:ℤ)⟧}
  34     (hT : Triangle.mk g i p ∈ distTriang SyntheticSpectrum)
  35     (hfg : f≫g=0)
  36     (a : X⟦(1:ℤ)⟧ ⟶ Q) (ha : a≫p = -f⟦(1:ℤ)⟧')
  37     (b : Q ⟶ W) (hb : i≫b=h) :
  38     Relation (a≫b) f g h := by
  39   obtain ⟨Qf,j,q,hTf⟩ := distinguished_cocone_triangle f
  40   let Tf := Triangle.mk f j q
  41   let Tg := Triangle.mk g i p
  42   obtain ⟨gbar,hgbar⟩ := Tf.yoneda_exact₂ hTf g hfg
  43   dsimp only [Tf,Triangle.mk] at gbar hgbar
  44   have hTfr : Tf.rotate ∈ distTriang SyntheticSpectrum := rot_of_distTriang Tf hTf
  45   obtain ⟨a0,ha0i,ha0p⟩ := complete_distinguished_triangle_morphism
  46     Tf.rotate Tg hTfr hT (𝟙 Y) gbar (by
  47       change j≫gbar=𝟙 Y≫g
  48       simpa only [Category.id_comp] using hgbar.symm)
  49   dsimp only [Tf,Tg,Triangle.mk,Triangle.rotate] at a0 ha0i ha0p
  50   have hai : q≫a0=gbar≫i := ha0i
  51   have hap : a0≫p = -f⟦(1:ℤ)⟧' := by
  52     simpa only [CategoryTheory.Functor.map_id,Category.comp_id] using ha0p.symm
  53   have hdiff : (a-a0)≫p=0 := by
  54     rw [Preadditive.sub_comp,ha,hap,sub_self]
  55   obtain ⟨c,hc⟩ := Tg.coyoneda_exact₃ hT (a-a0) hdiff
  56   dsimp only [Tg,Triangle.mk] at c hc
  57   have h0 : Relation (a0≫b) f g h := by
  58     refine ⟨Qf,j,q,hTf,gbar,hgbar.symm,?_⟩
  59     rw [←Category.assoc,hai,Category.assoc,hb]
  60   have hsum : a≫b=a0≫b+c≫h := by
  61     have hh := congrArg (fun z => z≫b) hc
  62     rw [Preadditive.sub_comp,Category.assoc,hb] at hh
  63     exact (sub_eq_iff_eq_add'.mp hh)
  64   rw [hsum]
  65   apply (relation_iff_indeterminacy h0).mpr
  66   refine ⟨0,c,?_⟩
  67   simp only [comp_zero,zero_add,add_sub_cancel_left]
