  73 theorem finite_sphere_infinity_cycle_label_exists
  74     (q k : ℕ) (hq : 0 < q) (hkq : k < q) (s t : ℤ)
  75     (e : ((model.family.quotient (S_0_0 : SyntheticSpectrum) q).sequence.ssData (s,t,t-k)).eInfty) :
  76     ∃ x : model.classical.sphere.Page 2 (s,t),
  77       x ∈ cycles model .sphere ((q:ℤ)-k) (s,t) ∧
  78       HasInfinityRepresentative (model.family.quotient (S_0_0 : SyntheticSpectrum) q) 2 (s,t,t-k)
  79         (quotientLabel q s t k x) e := by
  80   let P := KIP126.Literature.Standard.infinityFiniteWindow .sphere q hq (s,t) (t-k)
  81     (by constructor <;> omega)
  82   let f := XModLambdaN.map model.nu.unitIso.hom q
  83   let g := XModLambdaN.map model.nu.unitIso.inv q
  84   have hgf : g ≫ f = 𝟙 _ := by
  85     dsimp only [f,g]
  86     rw [←model.quotients.functoriality.map_comp,Iso.inv_hom_id,model.quotients.functoriality.map_id]
  87   have heq : (model.family.map f).eInftyMap (s,t,t-k)
  88       ((model.family.map g).eInftyMap (s,t,t-k) e) = e := by
  89     change ((model.family.map g).eInftyMap (s,t,t-k) ≫
  90       (model.family.map f).eInftyMap (s,t,t-k)) e = e
  91     rw [←SpectralSequenceMorphism.eInftyMap_comp,←SyntheticAdamsFamily.map_comp,hgf,
  92       SyntheticAdamsFamily.map_id,SpectralSequenceMorphism.eInftyMap_id]
  93     rfl
  94   obtain ⟨c,hc⟩ := Quotient.exists_rep (P ((model.family.map g).eInftyMap (s,t,t-k) e))
  95   have hn := (finite_target_label_iff .sphere q hq s t k hkq c _).mpr hc.symm
  96   have h := family_infinity_representative_map f (s,t,t-k) _ _ hn
  97   rw [sphere_quotient_label_from_nu,heq] at h
  98   refine ⟨c.val,?_,h⟩
  99   have hc' := c.property
 100   change c.val ∈ cycles model .sphere ((q:ℤ)-t+(t-k)) (s,t) at hc'
 101   exact (congrArg (fun j : ℤ => c.val ∈ cycles model .sphere j (s,t))
 102     (show (q:ℤ)-t+(t-k)=(q:ℤ)-k by omega)).mp hc'

 104 theorem finite_sphere_filtration_has_cycle_label
 105     (q k : ℕ) (hq : 0 < q) (hkq : k < q) (s t : ℤ)
 106     (a : BiHom (t-s) (t-k) (XModLambdaN (S_0_0 : SyntheticSpectrum) q))
 107     (ha : FiltrationAtLeast model.syntheticTower s a) :
 108     ∃ x : model.classical.sphere.Page 2 (s,t),
 109       x ∈ cycles model .sphere ((q:ℤ)-k) (s,t) ∧
 110       Detects (model.convergence (.quotient q .sphere)) (s,t,t-k) (quotientLabel q s t k x) a := by
 111   let F := model.syntheticTower.filtration (XModLambdaN (S_0_0 : SyntheticSpectrum) q)
 112   obtain ⟨a',ha'⟩ := filtration_representative_exists model.syntheticTower _ (t-s,t-k) s a ha
 113   let gr := F.toAssociatedGraded s (t-s,t-k) a'
 114   let e := ((model.convergence (.quotient q .sphere)).identification (s,t,t-k)).inv gr
 115   have he : ((model.convergence (.quotient q .sphere)).identification (s,t,t-k)).hom e = gr :=
 116     ConcreteCategory.congr_hom ((model.convergence (.quotient q .sphere)).identification (s,t,t-k)).inv_hom_id gr
 117   obtain ⟨x,hc,hx⟩ := finite_sphere_infinity_cycle_label_exists q k hq hkq s t e
 118   exact ⟨x,hc,e,hx,a',ha',he⟩

 120 theorem permanent_cycle_iff_mem (X : ClassicalObject) (p : ℤ × ℤ)
 121     (x : (model.classical.obj (X.obj model.auxiliary)).Page 2 p) :
 122     IsPermanentCycle (model.classical.obj (X.obj model.auxiliary)) p x ↔
 123       x ∈ permanentCycles model X p := by
 124   let E := model.classical.obj (X.obj model.auxiliary)
 125   let P := E.ssData p
 126   have normalize (P : SSData (ModuleCat ℤ)) (n : WithTop ℕ) (hn : n=0) :
 127       Subobject.ofLE (P.Z ⊤) (P.Z n) (P.Z_anti le_top) ≫ P.pageπ n =
 128         Subobject.ofLE (P.Z ⊤) (P.Z 0) (P.Z_anti le_top) ≫ P.pageπ 0 ≫
 129           eqToHom (congrArg P.page hn.symm) := by
 130     subst n
 131     simp only [eqToHom_refl,Category.comp_id]
 132   have h := normalize P (↑(2-E.r₀).toNat) (by dsimp only [E]; rw [model.classical.firstPage]; rfl)
 133   constructor
 134   · rintro ⟨z,hz⟩
 135     refine ⟨z,?_⟩
 136     exact (congrArg (fun f => f z) h).symm.trans hz
 137   · rintro ⟨z,hz⟩
 138     refine ⟨z,?_⟩
 139     exact (congrArg (fun f => f z) h).trans hz

 141 theorem sphere_permanent_first_quotient_lift (s t : ℤ)
 142     (x : model.classical.sphere.Page 2 (s,t))
 143     (hx : IsPermanentCycle model.classical.sphere (s,t) x) :
 144     ∃ a : BiHom (t-s) t (S_0_0 : SyntheticSpectrum),

 157 theorem sphere_quotient_high_filtration_zero (q : ℕ) (hq : 0 < q)
 158     (m w s : ℤ) (hs : (q:ℤ) ≤ m+s-w)
 159     (a : BiHom m w (XModLambdaN (S_0_0 : SyntheticSpectrum) q))
 160     (ha : FiltrationAtLeast model.syntheticTower s a) : a = 0 := by
 161   let g : XModLambdaN (S_0_0 : SyntheticSpectrum) q ⟶ XModLambdaN (model.nu.functor.obj SphereSpectrum) q :=
 162     XModLambdaN.map model.nu.unitIso.inv q
 163   let f : XModLambdaN (model.nu.functor.obj SphereSpectrum) q ⟶ XModLambdaN (S_0_0 : SyntheticSpectrum) q :=
 164     XModLambdaN.map model.nu.unitIso.hom q
 165   have hm := map_preserves_filtration
 166     (model.convergentMap (.quotient q .sphere) (.quotient q (.nu .sphere)) g)
 167     (m,w) s a ha
 168   have hz := finite_high_filtration_zero .sphere q hq m w s hs (a ≫ g) hm
 169   have hgf : g ≫ f = 𝟙 _ := by
 170     dsimp only [f,g]
 171     rw [←model.quotients.functoriality.map_comp,Iso.inv_hom_id,model.quotients.functoriality.map_id]
 172     rfl
 173   have he := congrArg (fun z => z ≫ f) hz
 174   change (a ≫ g) ≫ f = 0 ≫ f at he
 175   rw [Category.assoc,hgf,Category.comp_id,zero_comp] at he
 176   exact he

 178 theorem finite_sphere_frame (q k : ℕ) (hq : 0 < q) (hkq : k < q) (s t : ℤ)
 179     (c : cycles model .sphere ((q:ℤ)-t+(t-k)) (s,t))
 180     (hframe : ∀ y : CycleQuotient model .sphere ((q:ℤ)-t+(t-k)) (1+t-(t-k)) (s,t),
 181       y = 0 ∨ y = subquotientClass _ _ c) :
 182     ∀ e : ((model.family.quotient (S_0_0 : SyntheticSpectrum) q).sequence.ssData (s,t,t-k)).eInfty,
 183       e = 0 ∨ HasInfinityRepresentative (model.family.quotient (S_0_0 : SyntheticSpectrum) q)
 184         2 (s,t,t-k) (quotientLabel q s t k c.val) e := by
 185   intro e
 186   let P := KIP126.Literature.Standard.infinityFiniteWindow .sphere q hq (s,t) (t-k)
 187     (by constructor <;> omega)
 188   let f := XModLambdaN.map model.nu.unitIso.hom q
 189   let g := XModLambdaN.map model.nu.unitIso.inv q
 190   have hgf : g ≫ f = 𝟙 _ := by
 191     dsimp only [f,g]
 192     rw [←model.quotients.functoriality.map_comp,Iso.inv_hom_id,model.quotients.functoriality.map_id]
 193   have heq : (model.family.map f).eInftyMap (s,t,t-k)
 194       ((model.family.map g).eInftyMap (s,t,t-k) e) = e := by
 195     change ((model.family.map g).eInftyMap (s,t,t-k) ≫
 196       (model.family.map f).eInftyMap (s,t,t-k)) e = e
 197     rw [←SpectralSequenceMorphism.eInftyMap_comp,←SyntheticAdamsFamily.map_comp,hgf,
 198       SyntheticAdamsFamily.map_id,SpectralSequenceMorphism.eInftyMap_id]
 199     rfl
 200   rcases hframe (P ((model.family.map g).eInftyMap (s,t,t-k) e)) with hz | hz
 201   · left
 202     have hzero : (model.family.map g).eInftyMap (s,t,t-k) e = 0 :=
 203       P.injective (hz.trans (map_zero P).symm)
 204     rw [hzero,map_zero] at heq
 205     exact heq.symm
 206   · right
 207     have hn := (finite_target_label_iff .sphere q hq s t k hkq c _).mpr hz
 208     have h := family_infinity_representative_map f (s,t,t-k) _ _ hn
 209     rw [sphere_quotient_label_from_nu,heq] at h
 210     exact h

 212 theorem finite_detection_peel (q : ℕ) (i : Tridegree)
 213     (x : (model.family.quotient (S_0_0 : SyntheticSpectrum) q).E₂ i)
 214     (hframe : ∀ e : ((model.family.quotient (S_0_0 : SyntheticSpectrum) q).sequence.ssData i).eInfty,
 215       e = 0 ∨ HasInfinityRepresentative (model.family.quotient (S_0_0 : SyntheticSpectrum) q) 2 i x e)
 216     (c : BiHom (i.2.1-i.1) i.2.2 (XModLambdaN (S_0_0 : SyntheticSpectrum) q))
 217     (hc : Detects (model.convergence (.quotient q .sphere)) i x c)
 218     (a : BiHom (i.2.1-i.1) i.2.2 (XModLambdaN (S_0_0 : SyntheticSpectrum) q))
 219     (ha : FiltrationAtLeast model.syntheticTower i.1 a) :
 220     ∃ b : Bool, FiltrationAtLeast model.syntheticTower (i.1+1) (a-(if b then c else 0)) := by
 221   let F := model.syntheticTower.filtration (XModLambdaN (S_0_0 : SyntheticSpectrum) q)
 222   obtain ⟨a',ha'⟩ := filtration_representative_exists model.syntheticTower _
 223     (i.2.1-i.1,i.2.2) i.1 a ha
 224   let gr := F.toAssociatedGraded i.1 (i.2.1-i.1,i.2.2) a'
 225   let e := ((model.convergence (.quotient q .sphere)).identification i).inv gr
 226   have he : ((model.convergence (.quotient q .sphere)).identification i).hom e = gr :=
 227     ConcreteCategory.congr_hom ((model.convergence (.quotient q .sphere)).identification i).inv_hom_id gr
 228   rcases hframe e with hz | hz
 229   · refine ⟨false,?_⟩
 230     have hdet : Detects (model.convergence (.quotient q .sphere)) i 0 a :=
 231       ⟨0,HasInfinityRepresentative.zero _ 2 (by decide) i,a',ha',by rw [←hz]; exact he⟩
 232     simpa only [Bool.false_eq_true, ↓reduceIte,sub_zero] using
 233       detects_zero_filtration (model.convergence (.quotient q .sphere)) i hdet
 234   · refine ⟨true,?_⟩
 235     have hdet : Detects (model.convergence (.quotient q .sphere)) i x a := ⟨e,hz,a',ha',he⟩
 236     simpa only [↓reduceIte] using detects_sub_filtration
 237       (model.convergence (.quotient q .sphere)) i hdet hc
