  19 private theorem page_subsingleton_of_le (E : KIP126.Core.SpectralSequence (ModuleCat.{v} ℤ) (ℤ×ℤ))
  20     (r t s u : ℤ) (hr : 2 ≤ r) (hrt : r ≤ t)
  21     (hp : Subsingleton (E.Page r (s,u))) : Subsingleton (E.Page t (s,u)) := by
  22   have hall (a : E.Page t (s,u)) : a=0 := by
  23     obtain ⟨x,hx⟩ := page_has_initial_representative (by omega) a
  24     obtain ⟨xr,hxr⟩ := KIP126.Main.Internal.reaches_before hr hrt ⟨a,hx⟩
  25     have hz : xr=0 := hp.elim _ _
  26     rw [hz] at hxr
  27     exact represents_later_zero hrt hxr hx
  28   exact ⟨fun a b => (hall a).trans (hall b).symm⟩
