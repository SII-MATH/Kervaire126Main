 334 private theorem lambda_multiply_compose {m w : ℤ} {X : SyntheticSpectrum}
 335     (k n : ℕ) (a : BiHom m w X) :
 336     HEq (lambdaMultiply n (lambdaMultiply k a)) (lambdaMultiply (k+n) a) := by
 337   let v := w-(k:ℤ)
 338   let u := v-(n:ℤ)
 339   let hk : (m,w)+(0,-(k:ℤ))=(m,v) := by ext <;> dsimp [v] <;> omega
 340   let hn : (m,v)+(0,-(n:ℤ))=(m,u) := by ext <;> dsimp [u] <;> omega
 341   let hkn : (m,w)+(0,-((k+n:ℕ):ℤ))=(m,u) := by ext <;> dsimp [u,v] <;> omega

 343     (m,v) (0,-((k+n:ℕ):ℤ)) (m,u)
