  23 private theorem detector_page_two_zero (s : ℤ) (hs : s ≤ 0) :
  24     Subsingleton ((model.classical.obj model.auxiliary.detector).Page 2 (s,63+s)) := by
  25   let f := (model.classical.map Standard.tmfIso.hom).pageMap 2 (s,63+s)
  26   let g := (model.classical.map Standard.tmfIso.inv).pageMap 2 (s,63+s)
  27   have hgf : g ≫ f = 𝟙 _ := by
  28     dsimp only [f,g]
  29     rw [←SpectralSequenceMorphism.pageMap_comp,←ClassicalAdamsFamily.map_comp,
  30       Iso.inv_hom_id,ClassicalAdamsFamily.map_id,SpectralSequenceMorphism.pageMap_id]
  31   have hf (a : (model.classical.obj model.auxiliary.detector).Page 2 (s,63+s)) : f (g a)=a :=
  32     ConcreteCategory.congr_hom hgf a
  33   letI := Standard.tmf_low_filtration63 s hs
  34   exact ⟨fun a b => (hf a).symm.trans ((congrArg f (Subsingleton.elim (g a) (g b))).trans (hf b))⟩

  49     (theta : BiHom 62 64 (S_0_0 : SyntheticSpectrum)) :
  50     theta ≫ detectorMap = 0 := by

  52   let a := theta ≫ detectorMap

  54     (Standard.realization_sphere 62 64).hom ≫ model.recovery.realization.map a ≫
  55       (model.recovery.nuRealizationIso.app model.auxiliary.detector).hom ≫ Standard.tmfIso.inv
  56   have hz := Standard.tmf_vanishing62 alpha

  59   rw [Functor.map_zero]
  60   apply (cancel_epi (Standard.realization_sphere 62 64).hom).mp
  61   apply (cancel_mono ((model.recovery.nuRealizationIso.app model.auxiliary.detector).hom ≫
  62     Standard.tmfIso.inv)).mp
