  24 private theorem nu_target_label_iff (s t : ℤ) (k : ℕ)
  25     (x : model.classical.sphere.Page 2 (s,t))
  26     (e : ((model.family.nu model.nu SphereSpectrum).sequence.ssData (s,t,t-k)).eInfty) :
  27     HasInfinityRepresentative _ 2 (s,t,t-k) (targetNuLabel model .sphere s t k x) e ↔
  28       ∃ hx : x ∈ permanentCycles model .sphere (s,t),
  29         KIP126.Literature.Standard.infinityNuWindow .sphere (s,t) (t-k) (by omega) e =
  30           subquotientClass (permanentCycles model .sphere (s,t))
  31             (boundaries model .sphere (1+t-(t-k)) (s,t)) ⟨x,hx⟩ := by
  32   have hlabel : nuWeightLabel model .sphere s t (t-k) (by omega) x =
  33       targetNuLabel model .sphere s t k x := by
  34     have hk' : (t-(t-k)).toNat = k := by omega
  35     have transport (a b : ℕ) (hab : a=b)
  36         (h : (s,t,t-a)=(s,t,t-b)) :
  37         regradeE2 h (targetNuLabel model .sphere s t a x) = targetNuLabel model .sphere s t b x := by
  38       subst b
  39       rfl
  40     exact transport _ k hk' _
  41   have h := KIP126.Literature.Standard.eInfinity_nu_labels .sphere s t (t-k) (by omega) x e
  42   rw [hlabel] at h
  43   exact h
