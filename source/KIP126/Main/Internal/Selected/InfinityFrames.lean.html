  31 private theorem permanent_quotient_zero_of_page_zero
  32     (X : ClassicalObject) (r b : ℤ) (hr : 2≤r) (hb : r-1≤b) (p : ℤ × ℤ)
  33     (hp : Subsingleton ((model.classical.obj (X.obj model.auxiliary)).Page r p)) :
  34     Subsingleton (PermanentQuotient model X b p) := by
  35   have hall (a : PermanentQuotient model X b p) : a=0 := by
  36     obtain ⟨x,rfl⟩ := Quotient.exists_rep a
  37     obtain ⟨yr,hy⟩ := permanent_reaches X r hr p x.val x.property
  38     have hz : yr=0 := hp.elim _ _
  39     rw [hz] at hy
  40     have hm := represents_zero_mem_boundary model X r p x.val hy
  41     exact (subquotientClass_eq_zero_iff _ _ _).mpr (boundaries_monotone model X p hb hm)
  42   exact ⟨fun a b => (hall a).trans (hall b).symm⟩

  44 private theorem permanent_quotient_frame_of_page
  45     (r b : ℤ) (hr : 2≤r) (hb : r-1≤b) (s t : ℤ)
  46     (x : model.classical.sphere.Page 2 (s,t))
  47     (hx : IsPermanentCycle model.classical.sphere (s,t) x)
  48     (a : model.classical.sphere.Page r (s,t))
  49     (ha : RepresentsOnPage model.classical.sphere r (s,t) x a)
  50     (hall : ∀ y : model.classical.sphere.Page r (s,t), y=0 ∨ y=a) :
  51     ∃ x' : permanentCycles model .sphere (s,t), x'.val=x ∧
  52       ∀ q : PermanentQuotient model .sphere b (s,t),
  53         q=0 ∨ q=subquotientClass _ _ x' := by
  54   let x' : permanentCycles model .sphere (s,t) :=
  55     ⟨x,(permanent_cycle_iff_mem .sphere (s,t) x).mp hx⟩
  56   refine ⟨x',rfl,?_⟩
  57   intro q
  58   obtain ⟨y,rfl⟩ := Quotient.exists_rep q
  59   obtain ⟨yr,hy⟩ := permanent_reaches .sphere r hr (s,t) y.val y.property
  60   rcases hall yr with h0|h1
  61   · left
  62     rw [h0] at hy
  63     exact (subquotientClass_eq_zero_iff _ _ _).mpr
  64       (boundaries_monotone model .sphere (s,t) hb
  65         (represents_zero_mem_boundary model .sphere r (s,t) y.val hy))
  66   · right
  67     apply sub_eq_zero.mp
  68     change subquotientClass _ _ y-subquotientClass _ _ x'=0
  69     rw [←map_sub]
  70     apply (subquotientClass_eq_zero_iff _ _ _).mpr
  71     have hsub := FinitePageCalculus.represents_sub hy ha
  72     rw [h1,sub_self] at hsub
  73     exact boundaries_monotone model .sphere (s,t) hb
  74       (represents_zero_mem_boundary model .sphere r (s,t) _ hsub)

  76 private theorem sphere_frame_of_permanent_quotient
  77     (s t : ℤ) (k : ℕ) (x : model.classical.sphere.Page 2 (s,t))
  78     (hq : ∃ x' : permanentCycles model .sphere (s,t), x'.val=x ∧
  79       ∀ q : PermanentQuotient model .sphere (1+t-(t-k)) (s,t),
  80         q=0 ∨ q=subquotientClass _ _ x') :
  81     ∀ e : (model.family.sphere.sequence.ssData (s,t,t-k)).eInfty,
  82       e=0 ∨ HasInfinityRepresentative model.family.sphere 2 (s,t,t-k)
  83         (model.comparison.sphereE2 s t k x) e := by
  84   obtain ⟨x',hx',hframe⟩ := hq
  85   intro e
  86   let i : Tridegree := (s,t,t-k)
  87   let f := (model.family.map model.nu.unitIso.hom).eInftyMap i
  88   let g := (model.family.map model.nu.unitIso.inv).eInftyMap i
  89   have hfg : g ≫ f = 𝟙 _ := by
  90     dsimp only [f,g]
  91     rw [←SpectralSequenceMorphism.eInftyMap_comp,←SyntheticAdamsFamily.map_comp,Iso.inv_hom_id,
  92       SyntheticAdamsFamily.map_id,SpectralSequenceMorphism.eInftyMap_id]
  93   have hback : f (g e)=e := ConcreteCategory.congr_hom hfg e
  94   let P := KIP126.Literature.Standard.infinityNuWindow .sphere (s,t) (t-k) (by omega)
  95   rcases hframe (P (g e)) with h0|h1
  96   · left
  97     have hg0 : g e=0 := P.injective (h0.trans (map_zero P).symm)
  98     rw [←hback,hg0,map_zero]
  99   · right
 100     have hn := (nu_target_label_iff s t k x'.val (g e)).mpr ⟨x'.property,h1⟩
 101     have hh := family_infinity_representative_map model.nu.unitIso.hom i _ _ hn
 102     change HasInfinityRepresentative model.family.sphere 2 i _ (f (g e)) at hh
 103     rw [hback,sphere_label_from_nu,hx'] at hh
 104     exact hh
