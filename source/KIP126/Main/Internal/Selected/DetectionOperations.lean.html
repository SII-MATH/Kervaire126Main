  19 theorem detects_infinity_equal (X : SyntheticObject) (i : Tridegree)
  20     (a : BiHom (i.2.1-i.1) i.2.2 (X.obj model.nu model.auxiliary))
  21     (x y : (model.family.obj (X.obj model.nu model.auxiliary)).E₂ i)
  22     (ha : Detects (model.convergence X) i x a)
  23     (hb : Detects (model.convergence X) i y a) :
  24     ∃ e, HasInfinityRepresentative (model.family.obj (X.obj model.nu model.auxiliary)) 2 i x e ∧
  25       HasInfinityRepresentative (model.family.obj (X.obj model.nu model.auxiliary)) 2 i y e := by
  26   obtain ⟨e,he,a',ha',hga⟩ := ha
  27   obtain ⟨f,hf,b',hb',hgb⟩ := hb
  28   have hab : a'=b' := (ModuleCat.mono_iff_injective _).mp inferInstance (ha'.trans hb'.symm)
  29   have hef : e=f := (ModuleCat.mono_iff_injective _).mp inferInstance (by rw [hga,hgb,hab])
  30   exact ⟨e,he,hef.symm ▸ hf⟩

  32 theorem theta_detection_step
  33     (i : Tridegree) (x : model.family.sphere.E₂ i)
  34     (hframe : ∀ e : (model.family.sphere.sequence.ssData i).eInfty,
  35       e=0 ∨ HasInfinityRepresentative model.family.sphere 2 i x e)
  36     (a : BiHom (i.2.1-i.1) i.2.2 (S_0_0 : SyntheticSpectrum))
  37     (ha : FiltrationAtLeast model.syntheticTower i.1 a) :
  38     DetectsNonzero (model.convergence .sphere) i x a ∨
  39       FiltrationAtLeast model.syntheticTower (i.1+1) a := by
  40   classical
  41   let F := model.syntheticTower.filtration (S_0_0 : SyntheticSpectrum)
  42   have ham : a ∈ (ModuleCat.subobjectModule
  43       (syntheticHomotopy (S_0_0 : SyntheticSpectrum) (i.2.1-i.1,i.2.2)))
  44       (F.F i.1 (i.2.1-i.1,i.2.2)) := by
  45     simpa only [F,SyntheticTowerPresentation.filtration,OrderIso.apply_symm_apply,
  46       FiltrationAtLeast] using ha
  47   obtain ⟨a',ha'⟩ := ham
  48   let q := F.toAssociatedGraded i.1 (i.2.1-i.1,i.2.2) a'
  49   let e := ((model.convergence .sphere).identification i).inv q
  50   have he : ((model.convergence .sphere).identification i).hom e=q :=
  51     ConcreteCategory.congr_hom ((model.convergence .sphere).identification i).inv_hom_id q
  52   by_cases he0 : e=0
  53   · right
  54     have hq : F.toAssociatedGraded i.1 (i.2.1-i.1,i.2.2) a'=0 := by
  55       rw [he0,map_zero] at he
  56       exact he.symm
  57     have hmem := (F.toAssociatedGraded_eq_zero_iff i.1 (i.2.1-i.1,i.2.2) a').mp hq
  58     rw [ha'] at hmem
  59     simpa only [F,SyntheticTowerPresentation.filtration,OrderIso.apply_symm_apply,
  60       FiltrationAtLeast] using hmem
  61   · left
  62     have hr := (hframe e).resolve_left he0
  63     exact ⟨⟨e,hr,a',ha',he⟩,e,hr,he0⟩
