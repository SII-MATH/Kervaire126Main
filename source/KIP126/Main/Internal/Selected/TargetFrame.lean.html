  15 private theorem later_page_frame_of_earlier
  16     {E : KIP126.Core.SpectralSequence (ModuleCat.{v} ℤ) (ℤ×ℤ)}
  17     {r t : ℤ} {p : ℤ×ℤ} {x : E.Page 2 p} (hr : 2≤r) (hrt : r≤t)
  18     (hframe : ∃ a : E.Page r p, RepresentsOnPage E r p x a ∧ ∀ y:E.Page r p,y=0∨y=a)
  19     (ht : ReachesPage E t p x) :
  20     ∃ a : E.Page t p, RepresentsOnPage E t p x a ∧ ∀ y:E.Page t p,y=0∨y=a := by
  21   obtain ⟨a,ha,hall⟩ := hframe
  22   obtain ⟨b,hb⟩ := ht
  23   refine ⟨b,hb,?_⟩
  24   intro y
  25   obtain ⟨x',hy⟩ := page_has_initial_representative (by omega : 2≤t) y
  26   obtain ⟨a',ha'⟩ := reaches_before hr hrt ⟨y,hy⟩
  27   rcases hall a' with h0|h1
  28   · left
  29     rw [h0] at ha'
  30     exact represents_later_zero hrt ha' hy
  31   · right
  32     have hs := represents_sub ha' ha
  33     rw [h1,sub_self] at hs
  34     exact sub_eq_zero.mp (represents_later_zero hrt hs (represents_sub hy hb))
