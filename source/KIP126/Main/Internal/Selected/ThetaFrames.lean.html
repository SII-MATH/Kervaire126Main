 122 private theorem theta_quotient_frame (s t : ℤ) (k : ℕ) (hst : t=s+124) (hsk : s-(k+2)≤4)
 123     (x : model.classical.sphere.Page 2 (s,t)) (hx : InfinityGenerated model.classical.sphere (s,t) x) :
 124     ∃ x' : permanentCycles model .sphere (s,t), x'.val=x ∧
 125       ∀ q : PermanentQuotient model .sphere (1+(k:ℤ)) (s,t), q=0 ∨ q=subquotientClass _ _ x' := by
 126   let E := Normalized.sphere
 127   let A := E.ssData (s,t)
 128   have hB : A.B ⊤=A.B (k:WithTop ℕ) := by
 129     have hin (r : ℤ) (hr : (k:ℤ)+2≤r) : E.d r ((s,t)-E.diffDeg r)=0 := by

 135     change A.B ⊤=A.B ↑((k:ℤ)+2-2).toNat at h
 136     simpa only [add_sub_cancel_right,Int.toNat_natCast] using h
 137   obtain ⟨z,hz,_,hall⟩ := hx
 138   have hz' : cycleLabel model .sphere ⊤ (s,t) z=x := by
 139     have norm (P : SSData (ModuleCat ℤ)) (n : WithTop ℕ) (hn : n=0) :
 140         Subobject.ofLE (P.Z ⊤) (P.Z n) (P.Z_anti le_top) ≫ P.pageπ n =
 141           Subobject.ofLE (P.Z ⊤) (P.Z 0) (P.Z_anti le_top) ≫ P.pageπ 0 ≫
 142             eqToHom (congrArg P.page hn.symm) := by
 143       subst n
 144       simp only [eqToHom_refl,Category.comp_id]
 145     have hn := norm (model.classical.sphere.ssData (s,t)) (↑(2-model.classical.sphere.r₀).toNat)
 146       (by rw [model.classical.firstPage]; rfl)
 147     exact (congrArg (fun f => f z) hn).symm.trans hz
 148   let x' : permanentCycles model .sphere (s,t) := ⟨x,⟨z,hz'⟩⟩
 149   refine ⟨x',rfl,?_⟩
 150   intro q
 151   obtain ⟨y,rfl⟩ := Quotient.exists_rep q
 152   obtain ⟨z',hz''⟩ := y.property
 153   have zero_mem (v : (Subobject.underlying.obj (A.Z ⊤) : ModuleCat ℤ)) (hv : A.pageπ ⊤ v=0) :

 156     have hb := hB
 157     change (Normalized.sphere.ssData (s,t)).B ⊤=(Normalized.sphere.ssData (s,t)).B ↑k at hb
 158     rw [Normalized.sphere_eq] at hb
 159     simpa only [show 1+(k:ℤ)-1=(k:ℤ) by omega,Int.toNat_natCast] using hb
 160   rcases hall (A.pageπ ⊤ z') with h0|h1
 161   · left
 162     apply (subquotientClass_eq_zero_iff _ _ _).mpr
 163     rw [←hz'']
 164     exact zero_mem z' h0
 165   · right
 166     apply sub_eq_zero.mp
 167     change subquotientClass _ _ y-subquotientClass _ _ x'=0
 168     rw [←map_sub]
 169     apply (subquotientClass_eq_zero_iff _ _ _).mpr
 170     change y.val-x'.val∈_
 171     have hzX : cycleLabel model .sphere ⊤ (s,t) z=x'.val := hz'
 172     rw [←hz'',←hzX,←map_sub]
 173     let z0 : (Subobject.underlying.obj (((model.classical.obj (ClassicalObject.sphere.obj model.auxiliary)).ssData (s,t)).Z ⊤) : ModuleCat ℤ) := z
 174     exact zero_mem (z'-z0) (by rw [map_sub,h1]; exact sub_self _)
