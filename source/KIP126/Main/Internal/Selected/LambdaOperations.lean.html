  14 private theorem shift_assoc_inv (coh : BiShiftCoherence SyntheticSpectrum)
  15     (a b c ab bc total : ℤ × ℤ) (hab : a+b=ab) (hbc : b+c=bc)
  16     (habc : ab+c=total) (habc' : a+bc=total) (A : SyntheticSpectrum) :
  17     (biShiftAddIso ab c total habc).inv.app A ≫
  18       (SyntheticCategory.biShift c).map ((biShiftAddIso a b ab hab).inv.app A) =
  19     (biShiftAddIso a bc total habc').inv.app A ≫
  20       (biShiftAddIso b c bc hbc).inv.app ((SyntheticCategory.biShift a).obj A) := by
  21   classical
  22   have he : ((SyntheticCategory.biShift c).mapIso ((biShiftAddIso a b ab hab).app A) ≪≫
  23       (biShiftAddIso ab c total habc).app A) =
  24     ((biShiftAddIso b c bc hbc).app ((SyntheticCategory.biShift a).obj A) ≪≫
  25       (biShiftAddIso a bc total habc').app A) :=
  26     Iso.ext (coh.associativity a b c ab bc total hab hbc habc habc' A)
  27   have hi := congrArg Iso.inv he
  28   exact hi

 184 private theorem vanishesModLambda_iff_factors {m n : ℤ} {X : SyntheticSpectrum} (r : ℕ)
 185     (x : BiHom m n X) :
 186     VanishesModLambda r x ↔
 187       ∃ y : Smn m n ⟶ (SyntheticCategory.biShift (0, -(r : ℤ))).obj X,
 188         y ≫ lambdaPow r X = x := by
 189   let T := XModLambdaN.cofiberTriangle X r
 190   have hT : T ∈ distTriang SyntheticSpectrum := HasFunctorialCofiber.cofib_distinguished _
 191   constructor
 192   · intro hx
 193     obtain ⟨y, hy⟩ := T.coyoneda_exact₂ hT x hx
 194     exact ⟨y, hy.symm⟩
 195   · rintro ⟨y, rfl⟩
 196     change (y ≫ lambdaPow r X) ≫ XModLambdaN.incl X r = 0
 197     rw [Category.assoc,XModLambdaN.lambdaPow_comp_incl,Limits.comp_zero]

 199 private theorem vanishes_mod_lambda_iff_multiple
 200     (n : ℕ) {X : SyntheticSpectrum} (m w : ℤ) (z : BiHom m (w-(n:ℤ)) X) :
 201     VanishesModLambda n z ↔ ∃ b : BiHom m w X, lambdaMultiply n b=z := by
 202   classical
 203   let F := SyntheticCategory.biShift (Syn:=SyntheticSpectrum) (0,-(n:ℤ))
 204   let e : F.obj (Smn m w : SyntheticSpectrum) ≅ Smn m (w-(n:ℤ)) :=
 205     (SyntheticCategory.biShift_comp (m,w) (0,-(n:ℤ))).app S_0_0 ≪≫
 206       eqToIso (by congr 1; simp [sub_eq_add_neg])
 207   have he (b : BiHom m w X) : lambdaMultiply n b=e.inv≫F.map b≫lambdaPow n X := by
 208     have he' : lambdaMultiply n b=e.inv≫lambdaPow n (Smn m w)≫b := by
 209       unfold lambdaMultiply
 210       change eqToHom (by congr 1; simp) ≫
 211         (SyntheticCategory.biShift_comp (m,w) (0,-(n:ℤ))).inv.app S_0_0 ≫
 212         lambdaPow n (Smn m w) ≫ b =
 213         (eqToHom (by congr 1; simp) ≫
 214           (SyntheticCategory.biShift_comp (m,w) (0,-(n:ℤ))).inv.app S_0_0) ≫
 215         lambdaPow n (Smn m w) ≫ b
 216       exact (Category.assoc _ _ _).symm
 217     rw [he',←lambdaPow_naturality n b]
 218   constructor
 219   · intro hz
 220     obtain ⟨a,ha⟩ := (vanishesModLambda_iff_factors n z).mp hz
 221     let b := (SyntheticCategory.biShift_fullyFaithful (Syn:=SyntheticSpectrum) (0,-(n:ℤ))).preimage (e.hom≫a)
 222     have hb : F.map b=e.hom≫a :=
 223       (SyntheticCategory.biShift_fullyFaithful (Syn:=SyntheticSpectrum) (0,-(n:ℤ))).map_preimage _
 224     refine ⟨b,?_⟩
 225     rw [he,hb]
 226     simp only [Category.assoc,e.inv_hom_id_assoc]
 227     exact ha
 228   · rintro ⟨b,rfl⟩

 230     exact ⟨e.inv≫F.map b,by rw [Category.assoc,←he]⟩

 233 private theorem map_filtration (A B : SyntheticObject)
 234     (f : A.obj model.nu model.auxiliary ⟶ B.obj model.nu model.auxiliary)
 235     {m w s : ℤ} (a : BiHom m w (A.obj model.nu model.auxiliary))
 236     (ha : FiltrationAtLeast model.syntheticTower s a) :
 237     FiltrationAtLeast model.syntheticTower s (a ≫ f) := by
 238   let M := model.convergentMap A B f
 239   let FA := model.syntheticTower.filtration (A.obj model.nu model.auxiliary)
 240   let FB := model.syntheticTower.filtration (B.obj model.nu model.auxiliary)
 241   have ham : a ∈ ModuleCat.subobjectModule (syntheticHomotopy
 242       (A.obj model.nu model.auxiliary) (m,w)) (FA.F s (m,w)) := by
 243     simpa only [FA,SyntheticTowerPresentation.filtration,OrderIso.apply_symm_apply,FiltrationAtLeast] using ha
 244   obtain ⟨a',ha'⟩ := ham
 245   have hb : a ≫ f ∈ ModuleCat.subobjectModule (syntheticHomotopy
 246       (B.obj model.nu model.auxiliary) (m,w)) (FB.F s (m,w)) := by
 247     refine ⟨(M.filteredMap.compat s (m,w)).choose a',?_⟩
 248     have h := ConcreteCategory.congr_hom (M.filteredMap.compat s (m,w)).choose_spec a'
 249     have hf := ConcreteCategory.congr_hom (M.filteredMap_map (m,w)) ((FA.F s (m,w)).arrow a')
 250     change M.filteredMap.map (m,w) ((FA.F s (m,w)).arrow a') = (FA.F s (m,w)).arrow a' ≫ f at hf
 251     change (FB.F s (m,w)).arrow ((M.filteredMap.compat s (m,w)).choose a') =
 252       M.filteredMap.map (m,w) ((FA.F s (m,w)).arrow a') at h
 253     exact h.trans (hf.trans (congrArg (fun z => z ≫ f) ha'))
 254   simpa only [FB,SyntheticTowerPresentation.filtration,OrderIso.apply_symm_apply,FiltrationAtLeast] using hb
