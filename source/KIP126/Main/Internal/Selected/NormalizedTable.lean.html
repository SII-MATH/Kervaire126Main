  16 private def pre (E : KIP126.Core.SpectralSequence (ModuleCat.{v} ℤ) (ℤ×ℤ))
  17     (hr : E.r₀=2) (hd : E.diffDeg=fun r => (r,r-1)) :
  18     PreSS (ModuleCat.{v} ℤ) (ℤ×ℤ) where
  19   r₀ := 2
  20   ssData := E.ssData
  21   diffDeg := fun r => (r,r-1)
  22   d r p := by simpa only [hr,hd] using E.d r p

  24 private theorem pre_eq (E : KIP126.Core.SpectralSequence (ModuleCat.{v} ℤ) (ℤ×ℤ))
  25     (hr : E.r₀=2) (hd : E.diffDeg=fun r => (r,r-1)) : pre E hr hd=E.toPreSS := by
  26   rcases E with ⟨⟨r,data,degree,d⟩,hdd,hZ,hB⟩
  27   dsimp only at hr hd
  28   subst r
  29   subst degree
  30   rfl

  32 private def normalize (E : KIP126.Core.SpectralSequence (ModuleCat.{v} ℤ) (ℤ×ℤ))
  33     (hr : E.r₀=2) (hd : E.diffDeg=fun r => (r,r-1)) :
  34     KIP126.Core.SpectralSequence (ModuleCat.{v} ℤ) (ℤ×ℤ) where
  35   toPreSS := pre E hr hd
  36   d_comp_d := by rw [pre_eq]; exact E.d_comp_d
  37   Z_succ := by rw [pre_eq]; exact E.Z_succ
  38   B_succ := by rw [pre_eq]; exact E.B_succ

  58 private def basis (o : Raw.Spectrum) (s t i : ℕ) : Page o s t :=

  62 private def decode (o : Raw.Spectrum) (s t : ℕ) (indices : List ℕ) : Option (Page o s t) :=
  63   if Raw.coordinatesValid Raw.degrees o s t indices then some ((indices.map (basis o s t)).sum) else none

  65 private def BasisCorrect (d : Raw.Degree) : Prop :=
  66   ∃ e : Page d.spectrum d.s d.t ≃ₗ[ℤ] (Fin d.monomials.length →₀ ZMod 2),
  67     ∀ i, e.symm (Finsupp.single i 1)=basis d.spectrum d.s d.t i.val
  68 
  69 private theorem basis_correct (d : Raw.Degree) (hd : d ∈ Raw.degrees) : BasisCorrect d := by
  70   have transport (E E' : KIP126.Core.SpectralSequence (ModuleCat ℤ) (ℤ×ℤ)) (h : E=E')
  71       (b : ℕ → E.Page 2 (d.s,d.t))
  72       (hb : ∃ e : E.Page 2 (d.s,d.t) ≃ₗ[ℤ] (Fin d.monomials.length →₀ ZMod 2),
  73         ∀ i, e.symm (Finsupp.single i 1)=b i.val) :
  74       ∃ e : E'.Page 2 (d.s,d.t) ≃ₗ[ℤ] (Fin d.monomials.length →₀ ZMod 2),
  75         ∀ i, e.symm (Finsupp.single i 1)=
  76           (eqToHom (congrArg (fun A => A.Page 2 (d.s,d.t)) h)) (b i.val) := by
  77     subst E'
  78     simpa only [eqToHom_refl,ModuleCat.id_apply] using hb
  79   exact transport _ _ (sequence_eq d.spectrum).symm
  80     (KIP126.Computation.Table.basis d.spectrum d.s d.t)
  81     (KIP126.Computation.Table.basis_correct d hd)

  83 private def Statement (c : Raw.Claim) : Prop :=
  84   ∃ x, decode c.spectrum c.s c.t c.x=some x ∧
  85     match c.kind with
  86     | .reaches => ReachesPage (sequence c.spectrum) c.r (c.s,c.t) x
  87     | .boundaryBy => IsBoundaryBy (sequence c.spectrum) c.r (c.s,c.t) x
  88     | .equation => ∃ y, decode c.spectrum c.ts c.tt c.y=some y ∧
  89         HasDifferential (sequence c.spectrum) c.r (c.s,c.t) (c.ts,c.tt) x y
  90     | .refutation => ∃ y, decode c.spectrum c.ts c.tt c.y=some y ∧
  91         ¬ HasDifferential (sequence c.spectrum) c.r (c.s,c.t) (c.ts,c.tt) x y
  92 
  93 private theorem result (c : Raw.Claim) (hc : c ∈ Raw.claims) : Statement c := by
  94   let stmt (E : KIP126.Core.SpectralSequence (ModuleCat ℤ) (ℤ×ℤ))
  95       (b : (s t i : ℕ) → E.Page 2 (s,t)) : Prop :=
  96     let dec (s t : ℕ) (indices : List ℕ) : Option (E.Page 2 (s,t)) :=
  97       if Raw.coordinatesValid Raw.degrees c.spectrum s t indices then some ((indices.map (b s t)).sum) else none
  98     ∃ x, dec c.s c.t c.x=some x ∧
  99       match c.kind with
 100       | .reaches => ReachesPage E c.r (c.s,c.t) x
 101       | .boundaryBy => IsBoundaryBy E c.r (c.s,c.t) x
 102       | .equation => ∃ y, dec c.ts c.tt c.y=some y ∧ HasDifferential E c.r (c.s,c.t) (c.ts,c.tt) x y
 103       | .refutation => ∃ y, dec c.ts c.tt c.y=some y ∧ ¬HasDifferential E c.r (c.s,c.t) (c.ts,c.tt) x y
 104   have transport (E E' : KIP126.Core.SpectralSequence (ModuleCat ℤ) (ℤ×ℤ)) (h : E=E')
 105       (b : (s t i : ℕ) → E.Page 2 (s,t)) (hb : stmt E b) :
 106       stmt E' (fun s t i => (eqToHom (congrArg (fun A => A.Page 2 (s,t)) h)) (b s t i)) := by
 107     subst E'
 108     simpa only [eqToHom_refl,ModuleCat.id_apply] using hb
 109   exact transport _ _ (sequence_eq c.spectrum).symm
 110     (KIP126.Computation.Table.basis c.spectrum) (KIP126.Computation.Table.result c hc)
