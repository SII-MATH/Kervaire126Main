 419       if let some value := body then pending := pending ++ value.getUsedConstants
 420   -- Reject stale boundary entries as well as newly introduced dependencies.
 421   unless boundary.toArray.qsort Name.quickLt == reviewedProofToolsBoundary.qsort Name.quickLt do
