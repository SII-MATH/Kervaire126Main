  18 private theorem sphere_boundary_hit_or_previous (r : ℤ) (hr : 2≤r) (p : ℤ×ℤ)
  19     (x : model.classical.sphere.Page 2 p) (hx : x∈boundaries model .sphere r p) :
  20     x∈boundaries model .sphere (r-1) p ∨ HitOnPage model.classical.sphere r p x := by
  21   let E := model.classical.sphere
  22   have hrep : RepresentsOnPage E (r+1) p x 0 :=
  23     KIP126.ProofTools.representsOnPage_of_congruent model .sphere (by omega)
  24       (RepresentsOnPage.zero (by omega)) ⟨by omega,by simpa only [add_sub_cancel_right,sub_zero] using hx⟩
  25   obtain ⟨a,ha⟩ := represents_before hr (by omega : r≤r+1) hrep
  26   by_cases haz : a=0
  27   · left
  28     rw [haz] at ha
  29     exact represents_zero_mem_boundary model .sphere r p x ha
  30   · right
  31     let q := p-E.diffDeg r
  32     have hq : q+E.diffDeg r=p := sub_add_cancel _ _
  33     have incoming : ∃ u : E.Page r q,
  34         (E.d r q ≫ eqToHom (congrArg (E.Page r) hq)) u = a := by
  35       by_contra h
  36       have hn := represents_next_nonzero_of_not_incoming_at hq
  37         (by dsimp only [E]; rw [model.classical.firstPage]; exact hr) hr hrep ha
  38         (fun u hu => h ⟨u,hu⟩)
  39       exact hn rfl
  40     obtain ⟨u,hu⟩ := incoming
  41     exact ⟨q,hq,u,a,ha,haz,hu⟩
