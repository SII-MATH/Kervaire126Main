  18 /-- One common infinite representative generates the entire nonzero
  19 classical E∞ component. The specified E₂ class is retained. -/
  20 private def ClassicalInfinityGenerated (s t : ℤ)
  21     (x : sphereSequence.Page 2 (s,t)) : Prop :=
  22   let P := sphereSequence.ssData (s,t)
  23   let n : WithTop ℕ := ↑(2-sphereSequence.r₀).toNat
  24   ∃ z : (Subobject.underlying.obj (P.Z ⊤) : ModuleCat ℤ),
  25     (Subobject.ofLE (P.Z ⊤) (P.Z n) (P.Z_anti le_top) ≫ P.pageπ n) z = x ∧
  26     P.pageπ ⊤ z ≠ 0 ∧ ∀ y : P.eInfty, y = 0 ∨ y = P.pageπ ⊤ z

  28 /-- An actual finite-quotient differential with common representatives and
  29 nonzero target on the stated page, not just nonzero E₂ coordinates. -/
  30 private def FiniteNonzeroDifferential (q : ℕ) (r s t s' t' : ℤ) (k k' : ℕ)
  31     (x : sphereSequence.Page 2 (s,t)) (y : sphereSequence.Page 2 (s',t')) : Prop :=
  32   let A := family.quotient syntheticSphere q
  33   ∃ h : (s,t,t-k) + A.sequence.diffDeg r = (s',t',t'-k'),
  34     ∃ (xr : A.Page r (s,t,t-k)) (yr : A.Page r (s',t',t'-k')),
  35       SyntheticRepresentsOnPage A r (s,t,t-k)
  36         ((family.quotientProjection syntheticSphere q).pageMap 2 (s,t,t-k)
  37           (comparison.sphereE2 s t k x)) xr ∧
  38       SyntheticRepresentsOnPage A r (s',t',t'-k')
  39         ((family.quotientProjection syntheticSphere q).pageMap 2 (s',t',t'-k')
  40           (comparison.sphereE2 s' t' k' y)) yr ∧
  41       (A.sequence.d r (s,t,t-k) ≫ eqToHom (congrArg (A.Page r) h)) xr = yr ∧ yr ≠ 0

  43 /-- One Q11 representative and its actual Q9 image; correction choices may
  44 vary with the untruncated U representative. The final relation is exact. -/
  45 private def AlphaOneProperties
  46     (a11 : BiHom 123 132 (XModLambdaN (S_0_0 : SyntheticSpectrum) 11)) : Prop :=
  47   let a9 := a11 ≫ (model.quotients.tower S_0_0).rho 9 11 (by decide)
  48   FiniteDetected 11 (by decide) 9 132 0 KIP126.Computation.Standard.V a11 ∧
  49   FiniteDetected 9 (by decide) 9 132 0 KIP126.Computation.Standard.V a9 ∧
  50   (∀ u : BiHom 124 134 (S_0_0 : SyntheticSpectrum),
  51     SphereDetected 10 134 0 KIP126.Computation.Standard.U u →
  52     ∃ (a2 : BiHom 124 137 (XModLambdaN (S_0_0 : SyntheticSpectrum) 9))
  53       (a3 : BiHom 125 140 (XModLambdaN (S_0_0 : SyntheticSpectrum) 9)),
