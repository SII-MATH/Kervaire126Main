  28 private def negativeZeroIso (k : ℕ) (hk : k = 0) (X : SyntheticSpectrum) :
  29     (SyntheticCategory.biShift (0,-(k:ℤ))).obj X ≅ X := by
  30   subst k
  31   exact SyntheticCategory.biShift_zero.app X

  33 private theorem negative_lift_zero_factorization {X Y : SyntheticSpectrum}
  34     (k : ℕ) (hk : k=0)
  35     (f : (SyntheticCategory.biShift (0,(k:ℤ))).obj X ⟶ Y)
  36     (g : X ⟶ Y) (h : lambdaToPositivePow k X ≫ f = g) :
  37     negativeLift k f ≫ (negativeZeroIso k hk Y).hom = g := by
  38   subst k
  39   rw [← h]
  40   dsimp only [negativeLift, negativeZeroIso, lambdaToPositivePow, lambdaPow]
  41   simp only [eqToHom_refl, Category.id_comp, Iso.app_hom, Category.assoc]
  42   erw [SyntheticCategory.biShift_zero.hom.naturality]
  43   rfl

  45 private theorem normalized_nu_bottom_zero : hopfNu ≫ nuBottom = 0 := by
  46   have hz := comp_distTriang_mor_zero₁₂ _
  47     (Standard.nuCofiber_triangle Standard.nuCofiber_exponent_sum)
  48   change (model.normalizedMap (.shift 3 .sphere) .sphere model.auxiliary.nuMap).map ≫
  49     negativeLift (normalizedExponent model.classicalTower model.auxiliary.nuRouteTriangle.g)
  50       (model.normalizedMap .sphere .nuCofiber model.auxiliary.nuRouteTriangle.g).map = 0 at hz
  51   have hb := negative_lift_zero_factorization
  52     (normalizedExponent model.classicalTower model.auxiliary.nuRouteTriangle.g)
  53     Standard.nuCofiber_bottom_exponent
  54     (model.normalizedMap .sphere .nuCofiber model.auxiliary.nuRouteTriangle.g).map
  55     (model.nu.functor.map model.auxiliary.nuRouteTriangle.g)
  56     (model.normalizedMap .sphere .nuCofiber model.auxiliary.nuRouteTriangle.g).factorization
  57   have hzero : (model.normalizedMap (.shift 3 .sphere) .sphere model.auxiliary.nuMap).map ≫
  58       model.nu.functor.map (HasFunctorialCofiber.cofibι model.auxiliary.nuMap) = 0 := by
  59     erw [← hb, ← Category.assoc, hz, zero_comp]
  60   dsimp only [hopfNu, normalizedNu, nuBottom]
  61   simp only [Category.assoc, Iso.hom_inv_id_assoc]
  62   rw [hzero, comp_zero]
