  20 /-- Nonzero detection of lambda^k x in the specified finite quotient.
  21 The E2 label is induced by the actual sphere quotient map, not chosen anew. -/
  22 private def FiniteDetected (n : ℕ) (_hn : 0 < n) (s t : ℤ) (k : ℕ)
  23     (x : sphereSequence.Page 2 (s,t))
  24     (a : BiHom (t-s) (t-k) (XModLambdaN (S_0_0 : SyntheticSpectrum) n)) : Prop :=
  25   DetectsNonzero (model.convergence (.quotient n .sphere)) (s,t,t-k)
  26     ((family.quotientProjection syntheticSphere n).pageMap 2 (s,t,t-k) (comparison.sphereE2 s t k x)) a

  34 /-- The right-module bracket <x,y,z>, for x in pi(X) and sphere classes
  35 y,z. The actual arrow chain is z, then y, then x; only its LAST arrow
  36 lands in X. This is essential: putting x first would instead allow
  37 arbitrary endomorphisms of X in the indeterminacy.
  38 
  39 For <lambda^3 alpha1,h0,eta>, the two indeterminacy images are
  40 lambda^3 alpha1 * pi_(2,3)(sphere) and pi_(124,130)(Q9) * eta.
  41 The source has degree (124,132), and its Toda suspension (1,0) gives
  42 the required (125,132). All cofiber and extension choices are retained. -/
  43 private def ModuleTripleToda {X : SyntheticSpectrum} {a aw b bw c cw : ℤ}
  44     (x : BiHom a aw X) (y : BiHom b bw (S_0_0 : SyntheticSpectrum))
  45     (z : BiHom c cw (S_0_0 : SyntheticSpectrum))
  46     (value : BiHom (a+(b+c)+1) (aw+(bw+cw)) X) : Prop :=
  47   Toda.Relation ((tripleTodaSource c cw b bw a aw).inv ≫
  48       homotopyRegrade (by omega) (by omega) value)
  49     ((SyntheticCategory.biShift (b+a,bw+aw)).map z)
  50     ((SyntheticCategory.biShift_comp (b,bw) (a,aw)).inv.app S_0_0 ≫
  51       (SyntheticCategory.biShift (a,aw)).map y) x

  72 /-- The exact quotient relation in Lemma `lem:nuext125`, with existential

  74 private def NuExtension9 (X : sphereSequence.Page 2 (8,130))
  75     (Y : sphereSequence.Page 2 (11,136)) (h2 : BiHom 3 4 (S_0_0 : SyntheticSpectrum)) : Prop :=
  76   ∃ (x : BiHom 122 126 (XModLambdaN (S_0_0 : SyntheticSpectrum) 9))
  77     (y : BiHom 125 131 (XModLambdaN (S_0_0 : SyntheticSpectrum) 9)),
  78     FiniteDetected 9 (by decide) 8 130 4 X x ∧
  79     FiniteDetected 9 (by decide) 11 136 5 Y y ∧
