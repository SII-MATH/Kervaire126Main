  53 theorem anyy_boundary_filtration_peel
  54     (s t : ℤ) (k : ℕ) (hk : k<9)
  55     (hb : cycles model .sphere (9-k) (s,t) ≤ boundaries model .sphere 4 (s,t))
  56     (a : BiHom (t-s) (t-k) (XModLambdaN (S_0_0 : SyntheticSpectrum) 9))
  57     (ha : FiltrationAtLeast model.syntheticTower s a) :
  58     ∃ a' : BiHom (t-s) (t-k) (XModLambdaN (S_0_0 : SyntheticSpectrum) 9),
  59       FiltrationAtLeast model.syntheticTower (s+1) a' ∧
  60       lambdaMultiply 6 a'=lambdaMultiply 6 a := by
  61   obtain ⟨x,hxc,hxa⟩ := finite_sphere_filtration_has_cycle_label 9 k (by decide) hk s t a ha
  62   obtain ⟨b,hbq,hbt⟩ := sphere_boundary_submodule_torsion_lift s t 4 (by decide) x (hb hxc)
  63   let a0 : BiHom (t-s) (t-k) (XModLambdaN (S_0_0 : SyntheticSpectrum) 9) := quotientClass 9 (lambdaMultiply k b)
  64   have ha0 : Detects (model.convergence (.quotient 9 .sphere)) (s,t,t-k) (quotientLabel 9 s t k x) a0 := by
  65     have h := model.sphereDetectionCompatible.homotopy_lambda s t k b
  66     change Detects (model.convergence .sphere) (s,t,t-k)
  67       (sphereLabel s t k (KIP126.Literature.sphereFirstQuotient model s t (quotientClass 1 b))) (lambdaMultiply k b) at h
  68     rw [hbq] at h
  69     exact detects_map .sphere (.quotient 9 .sphere) (XModLambdaN.incl _ 9) (s,t,t-k) _ _ h
  70   have ha0zero : lambdaMultiply 6 a0=0 := by
  71     have hb6 := lambda_multiply_zero_mono 3 6 (by decide) b hbt
  72     have hb6k := lambda_multiply_zero_iterated k 6 b hb6
  73     simpa only [a0,quotientClass,lambdaMultiply,Category.assoc,Limits.zero_comp] using
  74       congrArg (fun f=>f≫XModLambdaN.incl (S_0_0 : SyntheticSpectrum) 9) hb6k
  75   refine ⟨a-a0,detects_sub_filtration (model.convergence (.quotient 9 .sphere)) (s,t,t-k) hxa ha0,?_⟩
  76   have hdist : lambdaMultiply 6 (a-a0)=lambdaMultiply 6 a-lambdaMultiply 6 a0 := by
  77     simp only [lambdaMultiply,Preadditive.comp_sub]
  78   rw [hdist,ha0zero,sub_zero]
