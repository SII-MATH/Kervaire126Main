  17 theorem shiftSwap_add (coh : BiShiftCoherence SyntheticSpectrum)
  18     (a b c : ℤ × ℤ) (A : SyntheticSpectrum) :
  19     (shiftSwap a (b+c) A).hom ≫
  20       (SyntheticCategory.biShift a).map ((biShiftAddIso b c (b+c) rfl).inv.app A) =
  21     (biShiftAddIso b c (b+c) rfl).inv.app ((SyntheticCategory.biShift a).obj A) ≫
  22       (SyntheticCategory.biShift c).map (shiftSwap a b A).hom ≫
  23       (shiftSwap a c ((SyntheticCategory.biShift b).obj A)).hom := by
  24   let habc : (a+b)+c=a+(b+c) := add_assoc a b c
  25   let hbac : b+(a+c)=a+(b+c) := by abel
  26   have h₁ := coh.associativity a b c (a+b) (b+c) (a+(b+c)) rfl rfl habc rfl A
  27   have h₂ := coh.associativity b a c (a+b) (a+c) (a+(b+c))
  28     (add_comm _ _) rfl habc hbac A
  29   have h₃ := coh.associativity b c a (b+c) (a+c) (a+(b+c))
  30     rfl (add_comm _ _) (add_comm _ _) hbac A
  31   simp only [shiftSwap,Iso.trans_hom,Iso.symm_hom,Iso.app_hom,Iso.app_inv]
  32   rw [Functor.map_comp]
  33   apply (cancel_epi ((biShiftAddIso b c (b+c) rfl).hom.app
  34     ((SyntheticCategory.biShift a).obj A))).mp
  35   simp only [Category.assoc,Iso.hom_inv_id_app_assoc]
  36   rw [← Category.assoc,← h₁]
  37   simp only [Category.assoc]
  38   apply (cancel_epi ((SyntheticCategory.biShift c).map
  39     ((biShiftAddIso a b (a+b) rfl).hom.app A))).mpr
  40   -- Cancel the common left factor introduced by the first associativity square.
  41   apply (cancel_epi ((SyntheticCategory.biShift c).map
  42     ((biShiftAddIso b a (a+b) (add_comm _ _)).hom.app A))).mp
  43   simp only [← Functor.map_comp_assoc, Iso.hom_inv_id_app,CategoryTheory.Functor.map_id,Category.id_comp]
  44   rw [← Category.assoc,h₂]
  45   simp only [Category.assoc]
  46   apply (cancel_epi ((biShiftAddIso a c (a+c) rfl).hom.app
  47     ((SyntheticCategory.biShift b).obj A))).mpr
  48   apply (cancel_epi ((biShiftAddIso c a (a+c) (add_comm _ _)).hom.app
  49     ((SyntheticCategory.biShift b).obj A))).mp
  50   rw [← Category.assoc,← Category.assoc,← h₃]
  51   simp only [Category.assoc,
  52     Iso.hom_inv_id_app]
  53   erw [CategoryTheory.Category.id_comp,←Functor.map_comp,Iso.hom_inv_id_app,CategoryTheory.Functor.map_id]
  54   rfl

  86 theorem lambda_pow_shift (coh : BiShiftCoherence SyntheticSpectrum)
  87     (n : ℕ) (a : ℤ × ℤ) (A : SyntheticSpectrum) :
  88     lambdaPow n ((SyntheticCategory.biShift a).obj A) =
  89       (shiftSwap a (KIP126.Synthetic.Context.lambdaDegree n) A).hom ≫
  90         (SyntheticCategory.biShift a).map (lambdaPow n A) := by
  91   induction n generalizing A with
  92   | zero =>
  93     change SyntheticCategory.biShift_zero.hom.app ((SyntheticCategory.biShift a).obj A) =
  94       (shiftSwap a 0 A).hom ≫
  95         (SyntheticCategory.biShift a).map (SyntheticCategory.biShift_zero.hom.app A)
  96     have h : SyntheticCategory.biShift_zero.hom.app ((SyntheticCategory.biShift a).obj A) =
  97         (biShiftAddIso a 0 a (add_zero a)).hom.app A ≫
  98           (biShiftAddIso 0 a a (zero_add a)).inv.app A ≫
  99             (SyntheticCategory.biShift a).map (SyntheticCategory.biShift_zero.hom.app A) := by
 100       rw [←coh.left_unit a A,Iso.inv_hom_id_app,Category.comp_id]
 101       exact (coh.right_unit a A).symm
 102     rw [shiftSwap_hom_eq a 0 a (add_zero a) (zero_add a),Category.assoc]
 103     exact h
 104   | succ n ih =>
 105     have h1 : lambdaPow 1 ((SyntheticCategory.biShift a).obj A) =
 106         (shiftSwap a (KIP126.Synthetic.Context.lambdaDegree 1) A).hom ≫
 107           (SyntheticCategory.biShift a).map (lambdaPow 1 A) := by
 108       rw [lambdaPow_one coh,lambdaPow_one coh]
 109       change SyntheticCategory.lam.app ((SyntheticCategory.biShift a).obj A) =
 110         (shiftSwap a (0,-1) A).hom ≫ (SyntheticCategory.biShift a).map (SyntheticCategory.lam.app A)
 111       rw [shiftSwap_hom_eq a (0,-1) (a+(0,-1)) rfl (add_comm _ _),Category.assoc]
 112       exact coh.lambda_comm a A
 113     have hn := shiftSwap_natural a (KIP126.Synthetic.Context.lambdaDegree n) (lambdaPow 1 A)
 114     have hh := shiftSwap_add_eq coh a (KIP126.Synthetic.Context.lambdaDegree 1) (KIP126.Synthetic.Context.lambdaDegree n)
 115       (KIP126.Synthetic.Context.lambdaDegree (n+1)) (by ext <;> simp [KIP126.Synthetic.Context.lambdaDegree]) A
 116     change (shiftSwap a (KIP126.Synthetic.Context.lambdaDegree (n+1)) A).hom ≫
 117         (SyntheticCategory.biShift a).map ((lambdaShiftAddIso 1 n (n+1) (by omega)).inv.app A) =
 118       (lambdaShiftAddIso 1 n (n+1) (by omega)).inv.app ((SyntheticCategory.biShift a).obj A) ≫
 119         (SyntheticCategory.biShift (KIP126.Synthetic.Context.lambdaDegree n)).map (shiftSwap a (KIP126.Synthetic.Context.lambdaDegree 1) A).hom ≫
 120         (shiftSwap a (KIP126.Synthetic.Context.lambdaDegree n) ((SyntheticCategory.biShift (KIP126.Synthetic.Context.lambdaDegree 1)).obj A)).hom at hh
 121     rw [lambdaPow_add coh 1 n (n+1) (by omega) ((SyntheticCategory.biShift a).obj A),
 122       lambdaPow_add coh 1 n (n+1) (by omega) A,h1,ih]
 123     simp only [Functor.map_comp,Category.assoc]
 124     rw [reassoc_of% hn]
 125     rw [reassoc_of% hh]
 126     rfl
