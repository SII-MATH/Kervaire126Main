  12 private theorem binary_expansion {N : ℕ} (f : Fin N →₀ ZMod 2) :
  13     f = ∑ i : Fin N, if f i = 0 then 0 else Finsupp.single i 1 := by
  14   classical
  15   ext i
  16   simp only [Finsupp.finsetSum_apply]
  17   rw [Finset.sum_eq_single i]
  18   · generalize f i = b
  19     fin_cases b
  20     · change (0 : ZMod 2) = (if (0 : ZMod 2)=0 then 0 else Finsupp.single i 1) i
  21       simp
  22     · change (1 : ZMod 2) = (if (1 : ZMod 2)=0 then 0 else Finsupp.single i 1) i
  23       simp
  24   · intro j hj hji
  25     by_cases hf : f j = 0
  26     · simp [hf]
  27     · simp [hf,hji]
  28   · simp

  30 private theorem two_coordinate_cosets {V : Type*} [AddCommGroup V] [Module ℤ V]
  31     (e : V ≃ₗ[ℤ] (Fin 2 →₀ ZMod 2)) (B : Submodule ℤ V)
  32     (hB : e.symm (Finsupp.single 0 1) + e.symm (Finsupp.single 1 1) ∈ B)
  33     (x : V) : x ∈ B ∨ x - e.symm (Finsupp.single 1 1) ∈ B := by
  34   classical
  35   have hn : -e.symm (Finsupp.single (1:Fin 2) 1) = e.symm (Finsupp.single (1:Fin 2) 1) := by
  36     rw [←map_neg]
  37     congr 1
  38     ext i
  39     simp
  40   let c : Fin 2 → ZMod 2 := e x
  41   have hx := congrArg e.symm (binary_expansion (e x))
  42   simp only [LinearEquiv.symm_apply_apply,map_sum,map_add,Fin.sum_univ_succ,apply_ite,map_zero] at hx
  43   change x = (if c 0=0 then 0 else e.symm (Finsupp.single 0 1)) +
  44     ((if c 1=0 then 0 else e.symm (Finsupp.single 1 1))+0) at hx
  45   simp only [add_zero] at hx
  46   by_cases h0 : c 0=0 <;> by_cases h1 : c 1=0
  47   · left
  48     rw [hx]
  49     simpa only [h0,h1,if_true,zero_add] using B.zero_mem
  50   · right
  51     rw [hx]
  52     simp only [h0,h1,if_false,if_true,zero_add,sub_self]
  53     exact B.zero_mem
  54   · right
  55     rw [hx]
  56     simpa only [h0,h1,if_false,if_true,add_zero,sub_eq_add_neg,hn] using hB
  57   · left
  58     rw [hx]
  59     simpa only [h0,h1,if_false] using hB

  61 private theorem three_coordinate_cosets {V : Type*} [AddCommGroup V] [Module ℤ V]
  62     (e : V ≃ₗ[ℤ] (Fin 3 →₀ ZMod 2)) (B : Submodule ℤ V)
  63     (h1 : e.symm (Finsupp.single 1 1) ∈ B)
  64     (h2 : e.symm (Finsupp.single 2 1) ∈ B) (x : V) :
  65     x ∈ B ∨ x-e.symm (Finsupp.single 0 1) ∈ B := by
  66   classical
  67   have hif (c : ZMod 2) {a : V} (h : a ∈ B) : (if c=0 then 0 else a) ∈ B := by
  68     split
  69     · exact B.zero_mem
  70     · exact h
  71   let c : Fin 3 → ZMod 2 := e x
  72   have hx := congrArg e.symm (binary_expansion (e x))
  73   simp only [LinearEquiv.symm_apply_apply,map_sum,map_add,Fin.sum_univ_succ,apply_ite,map_zero] at hx
  74   change x = (if c 0=0 then 0 else e.symm (Finsupp.single 0 1)) +
  75     ((if c 1=0 then 0 else e.symm (Finsupp.single 1 1)) +
  76     ((if c 2=0 then 0 else e.symm (Finsupp.single 2 1))+0)) at hx
  77   simp only [add_zero] at hx
  78   have hm := B.add_mem (hif (c 1) h1) (hif (c 2) h2)
  79   by_cases h0 : c 0=0
  80   · left
  81     rw [hx]
  82     simpa only [h0,if_true,zero_add] using hm
  83   · right
  84     rw [hx]
  85     simp only [h0,if_false]
  86     convert hm using 1; first | rfl | abel
  87 
  88 end
