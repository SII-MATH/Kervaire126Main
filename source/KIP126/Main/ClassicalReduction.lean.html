  20 private theorem earlier_zero_at
  21     {E : KIP126.Core.SpectralSequence (ModuleCat.{v} ℤ) (ℤ × ℤ)}
  22     {r t : ℤ} {p q : ℤ × ℤ} {x : E.Page 2 p} {y : E.Page 2 q}
  23     (hdeg : p + E.diffDeg r = q) (hr0 : E.r₀ ≤ r) (hrt : r < t)
  24     (hx : SurvivesTo E r p x)
  25     (htargets : DifferentialTargets E r p x
  26       (Eq.mp (congrArg (fun d => ↥(E.Page 2 d)) hdeg.symm) y))
  27     (hy : SurvivesTo E t q y) : HasDifferential E r p q x 0 := by
  28   subst q
  29   exact earlier_zero_of_target_survives hr0 hrt hx htargets hy
