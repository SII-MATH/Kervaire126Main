  19 /-- The list in Corollary 1.5, with equality in the actual E₂ module.
  20 The degree equation prevents heterogeneous identifications between stems.
  21 Indices are natural numbers, so the square branch includes h₀². -/
  22 public def AdamsTwoLineListed (t : ℤ) (x : model.classical.sphere.Page 2 (2,t)) : Prop :=
  23   ∃ i j : ℕ,
  24     ((i = 0 ∧ j = 2) ∨ (i = 0 ∧ j = 3) ∨ (i = 2 ∧ j = 4) ∨
  25       (i = 1 ∧ 3 ≤ j) ∨ (i = j ∧ j ≤ 6)) ∧
  26     ∃ degree : hiDegree i + hiDegree j = (2,t),
  27       x = (eqToHom (congrArg (ActualE2At model.classical) degree))
  28         (model.sphereAlgebra.product (hiDegree i) (hiDegree j)
  29           (model.standardClasses.standardHi i) (model.standardClasses.standardHi j))

  31 /-- Exact exhaustive classification target for AimPaper Corollary 1.5. Survival
  32 retains a common Z∞ representative and a nonzero E∞ image. -/
  33 public def AdamsTwoLineClassification : Prop :=
  34   ∀ (t : ℤ) (x : model.classical.sphere.Page 2 (2,t)),
  35     NonzeroSurvival model.classical.sphere (2,t) x ↔ AdamsTwoLineListed t x

  82 private theorem square_survival_iff (j : ℕ) :
  83     NonzeroSurvival model.classical.sphere (hiDegree j + hiDegree j)
  84       (KIP126.Literature.Standard.hiProduct j j) ↔
  85     NonzeroSurvival model.classical.sphere (2,((2^(j+1) : ℕ) : ℤ))
  86       (model.standardClasses.standardHiSquare j) := by
  87   have hd : hiDegree j + hiDegree j = (2,((2^(j+1) : ℕ) : ℤ)) := by
  88     ext <;> simp [hiDegree,pow_succ,mul_two]

  90       (eqToHom (congrArg (ActualE2At model.classical) hd))
  91         (KIP126.Literature.Standard.hiProduct j j) := rfl
  92   rw [hs]
  93   exact (survival_regrade hd _).symm

 106 private theorem survival_reaches_three {p : ℤ × ℤ}
 107     {x : model.classical.sphere.Page 2 p}
 108     (hx : NonzeroSurvival model.classical.sphere p x) :
 109     ReachesPage model.classical.sphere 3 p x := by
 110   obtain ⟨z,hz,_⟩ := hx
 111   let D := model.classical.sphere.ssData p
 112   let zr := (Subobject.ofLE (D.Z ⊤) (D.Z ↑(3-model.classical.sphere.r₀).toNat)
 113     (D.Z_anti le_top)) z
 114   refine ⟨D.pageπ _ zr,by decide,zr,?_,rfl⟩
 115   dsimp only [zr]
 116   rw [← CategoryTheory.comp_apply,← Category.assoc,Subobject.ofLE_comp_ofLE]
 117   exact hz

 119 private theorem listed_pair_survives (i j : ℕ)
 120     (h : (i = 0 ∧ j = 2) ∨ (i = 0 ∧ j = 3) ∨ (i = 2 ∧ j = 4) ∨
 121       (i = 1 ∧ 3 ≤ j) ∨ (i = j ∧ j ≤ 6)) :
 122     NonzeroSurvival model.classical.sphere (hiDegree i + hiDegree j)
 123       (KIP126.Literature.Standard.hiProduct i j) := by
 124   rcases h with h02 | h03 | h24 | ⟨rfl,hj⟩ | ⟨heq,hj⟩
 125   · exact KIP126.Literature.Standard.may_two_line_products_permanent i j (Or.inl h02)
 126   · exact KIP126.Literature.Standard.may_two_line_products_permanent i j (Or.inr (Or.inl h03))
 127   · exact KIP126.Literature.Standard.may_two_line_products_permanent i j (Or.inr (Or.inr h24))
 128   · exact KIP126.Literature.Standard.mahowald_eta_family_permanent j hj

 130     apply (square_survival_iff j).mpr
 131     by_cases hlow : j ≤ 3
 132     · exact KIP126.Literature.Standard.may_low_squares_permanent j hlow
 133     · have hcases : j = 4 ∨ j = 5 ∨ j = 6 := by omega

 139 private theorem surviving_pair_listed (i j : ℕ) (hij : i ≤ j)
 140     (hx : NonzeroSurvival model.classical.sphere (hiDegree i + hiDegree j)
 141       (KIP126.Literature.Standard.hiProduct i j)) :
 142     (i = 0 ∧ j = 2) ∨ (i = 0 ∧ j = 3) ∨ (i = 2 ∧ j = 4) ∨
 143       (i = 1 ∧ 3 ≤ j) ∨ (i = j ∧ j ≤ 6) := by
 144   have hc := KIP126.AdamsCalculations.two_line_d2_candidates i j hij
 145     (survival_nonzero hx) (survival_reaches_three hx)
 146   rcases hc with h02 | h03 | h24 | heta | hsq | ⟨rfl,rfl⟩ | ⟨rfl,rfl⟩ | ⟨rfl,rfl⟩ |
 147     ⟨rfl,rfl⟩
 148   · exact Or.inl h02
 149   · exact Or.inr (Or.inl h03)
 150   · exact Or.inr (Or.inr (Or.inl h24))
 151   · exact Or.inr (Or.inr (Or.inr (Or.inl heta)))
 152   · subst i
 153     have hj : j ≤ 6 := by
 154       by_contra hn
 155       exact KIP126.Literature.Standard.hhr_high_squares_not_permanent j (by omega)
 156         ((square_survival_iff j).mp hx)
 157     exact Or.inr (Or.inr (Or.inr (Or.inr ⟨rfl,hj⟩)))
 158   · obtain ⟨y,hy⟩ := KIP126.Literature.Standard.mahowald_tangora_h2_h5_d3
 159     exact False.elim (Internal.nonzero_differential_not_permanent
 160       (by rw [model.classical.firstPage]; decide) hy hx)
 161   · obtain ⟨y,hy⟩ := KIP126.Literature.Standard.mahowald_tangora_h3_h5_d4
 162     exact False.elim (Internal.nonzero_differential_not_permanent
 163       (by rw [model.classical.firstPage]; decide) hy hx)
 164   · obtain ⟨y,hy⟩ := KIP126.Literature.Standard.iwx_h3_h6_d3

 168     exact False.elim (Internal.nonzero_differential_not_permanent

 171 /-- AimPaper Corollary 1.5: the full second-line classification on the fixed
 172 sphere. All external input obligations remain in their upstream modules. -/
 173 public theorem adams_two_line_classification : AdamsTwoLineClassification := by
 174   intro t x
 175   constructor
 176   · intro hx
 177     obtain ⟨i,j,hij,degree,heq⟩ := KIP126.E2Computation.two_line_exhaustive t x
 178       (survival_nonzero hx)
 179     have hp := (survival_regrade degree (KIP126.Literature.Standard.hiProduct i j)).mp
 180       (heq ▸ hx)
 181     exact ⟨i,j,surviving_pair_listed i j hij hp,degree,heq⟩
 182   · rintro ⟨i,j,hij,degree,heq⟩
 183     rw [heq]
 184     exact (survival_regrade degree (KIP126.Literature.Standard.hiProduct i j)).mpr
 185       (listed_pair_survives i j hij)
