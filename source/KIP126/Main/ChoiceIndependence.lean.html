 104 /-- The source-to-arbitrary-choice step belongs to the present paper.
 105 The square difference has high filtration; the Section 7 eta-error and tmf
 106 arguments annihilate it. Only the lambda-eta-square expression is asserted
 107 independent, as in Remark 7.5. -/
 108 public theorem lambda_eta_theta_square_independent
 109     (theta : BiHom 62 64 (S_0_0 : SyntheticSpectrum)) (htheta : ThetaChoice theta) :
 110     KIP126.Literature.Standard.lambdaEtaThetaSquare theta =
 111       KIP126.Literature.Standard.lambdaEtaThetaSquare
 112         KIP126.Literature.Standard.bxTheta5 := by
 113   classical
 114   let theta' := KIP126.Literature.Standard.bxTheta5
 115   have hgap := Internal.Choice.theta_choice_filtration_gap_of_pages
 116     Internal.Choice.theta_choice_finite_pages theta theta' htheta
 117     KIP126.Literature.Standard.bx_theta_label
 118   have hsquare : FiltrationAtLeast model.syntheticTower 12
 119       (sphereProduct theta theta - sphereProduct theta' theta') := by
 120     rw [Internal.Choice.theta_square_difference theta theta'
 121       KIP126.Literature.Standard.bx_order_two]
 122     exact KIP126.Literature.Standard.action_filtration .sphere 62 64 62 64 6 6
 123       (theta-theta') (theta-theta') hgap hgap
 124   have herror := Internal.Selected.eta_weight128_branches_filtration_fifteen _
 125     (Internal.Selected.stem124_weight128_filtration11_branches
 126       (Internal.Selected.theta_sphere_frame 13 137 9 (by decide) (by decide) _
 127         Internal.Selected.correction_infinity_generated) _
 128       (model.syntheticTower.filtrationSubmodule_antitone _ (124,128)
 129         (by decide : (11:ℤ) ≤ 12) hsquare))
 130   have hlinear : sphereProduct KIP126.Literature.Standard.eta
 131       (sphereProduct theta theta - sphereProduct theta' theta') =
 132       sphereProduct KIP126.Literature.Standard.eta (sphereProduct theta theta) -
 133         sphereProduct KIP126.Literature.Standard.eta (sphereProduct theta' theta') := by
 134     simp only [sphereProduct, Int.reduceAdd, CategoryTheory.Preadditive.comp_sub]
 135   rw [hlinear] at herror
 136   have heq := Internal.High125.detector_injective_filtration15 _ herror
 137     (Internal.High125.eta_product_difference_order_two 124 128
 138       (sphereProduct theta theta) (sphereProduct theta' theta')) (by
 139     rw [CategoryTheory.Preadditive.sub_comp,
 140       Internal.High125.synthetic_eta_theta_square_detector_zero theta,
 141       Internal.High125.synthetic_eta_theta_square_detector_zero theta', sub_self])
 142   have h := congrArg (lambdaMultiply 1) (sub_eq_zero.mp heq)
 143   exact h

 145 /-- The complete normalized criterion at any admissible theta choice, for
 146 every natural r. The r=0 case is the independent E3/first-quotient endpoint;
 147 positive r uses the original fixed-choice BX criterion and choice transport. -/
 148 public theorem any_choice_criterion
 149     (theta : BiHom 62 64 (S_0_0 : SyntheticSpectrum)) (htheta : ThetaChoice theta) :
 150     (∀ r : ℕ,
 151       (SurvivesTo model.classical.sphere ((r : ℤ)+3) (2,128)
 152         model.standardClasses.standardH6Square ↔
 153         quotientClass (r+1) (KIP126.Literature.Standard.lambdaEtaThetaSquare theta) = 0)) ∧
 154     (H6SquarePermanent ↔ KIP126.Literature.Standard.lambdaEtaThetaSquare theta = 0) := by
 155   have he := lambda_eta_theta_square_independent theta htheta
 156   constructor
 157   · intro r
 158     cases r with
 159     | zero =>
 160       exact ⟨fun _ => lambda_eta_theta_square_mod_lambda_zero theta,
 161         fun _ => h6_square_survives_three⟩
 162     | succ r =>
 163       rw [he]
 164       exact normalized_bx_finite (r+1) (by omega)
 165   · rw [he]
 166     exact KIP126.Literature.Standard.bx_untruncated
