  39 /-- AimPaper Lemma 7.14 (`lem:x1239`). One Q11 representative supplies
  40 the compatible Q9 alpha-one class, its two correction identities for every
  41 untruncated U representative, and its exact h0 annihilation relation. -/
  42 public theorem lemma_x1239 :
  43     ∃ a11 : BiHom 123 132 (XModLambdaN (S_0_0 : SyntheticSpectrum) 11),
  44       let a9 := a11 ≫ (model.quotients.tower S_0_0).rho 9 11 (by decide)
  45       FiniteDetected 11 (by decide) 9 132 0 KIP126.Computation.Standard.V a11 ∧
  46       FiniteDetected 9 (by decide) 9 132 0 KIP126.Computation.Standard.V a9 ∧
  47       (∀ u : BiHom 124 134 (S_0_0 : SyntheticSpectrum),
  48         SphereDetected 10 134 0 KIP126.Computation.Standard.U u →
  49         ∃ (a2 : BiHom 124 137 (XModLambdaN (S_0_0 : SyntheticSpectrum) 9))
  50           (a3 : BiHom 125 140 (XModLambdaN (S_0_0 : SyntheticSpectrum) 9)),
  51           lambdaMultiply 3 (sphereAction eta a9) =
  52             (show BiHom 124 131 (XModLambdaN (S_0_0 : SyntheticSpectrum) 9) from
  53               lambdaMultiply 3 (quotientClass 9 u)) +
  54             (show BiHom 124 131 (XModLambdaN (S_0_0 : SyntheticSpectrum) 9) from
  55               lambdaMultiply 6 a2) ∧
  56           sphereAction eta a2 = lambdaMultiply 1 a3) ∧
  57       lambdaMultiply 3 (sphereAction syntheticH0 a9) = 0 := by
  58   exact KIP126.Main.Internal.AlphaOne.alpha_one_exists
