  28 private theorem square_regrade_comp {m m' w w' : ℤ}
  29     {X Y : SyntheticSpectrum} (hm : m=m') (hw : w=w')
  30     (a : BiHom m w X) (f : X ⟶ Y) :
  31     homotopyRegrade hm hw (a ≫ f)=(homotopyRegrade hm hw a) ≫ f := by
  32   subst m'

  50 private theorem sphere_filtration_lambda (m w s : ℤ) (h : w-m ≤ s)
  51     (a : BiHom m w (S_0_0 : SyntheticSpectrum)) :
  52     FiltrationAtLeast model.syntheticTower s a ↔
  53       ∃ b : BiHom m (m+s) (S_0_0 : SyntheticSpectrum),
  54         homotopyRegrade rfl (by rw [Int.toNat_of_nonneg (by omega)]; omega)
  55           (lambdaMultiply (m+s-w).toNat b)=a := by
  56   have hnu := KIP126.Literature.Standard.selected_filtration_lambda .sphere m w s h
  57     (a ≫ model.nu.unitIso.inv)
  58   constructor
  59   · intro ha
  60     obtain ⟨b,hb⟩ := hnu.mp (map_filtration .sphere (.nu .sphere) model.nu.unitIso.inv a ha)
  61     refine ⟨b ≫ model.nu.unitIso.hom,?_⟩
  62     simpa only [square_lambda_comp,square_regrade_comp,Category.assoc,Iso.inv_hom_id,Category.comp_id] using
  63       congrArg (fun z => z ≫ model.nu.unitIso.hom) hb
  64   · rintro ⟨b,hb⟩
  65     have hn := hnu.mpr ⟨b ≫ model.nu.unitIso.inv,by
  66       simpa only [square_lambda_comp,square_regrade_comp,Category.assoc] using
  67         congrArg (fun z => z ≫ model.nu.unitIso.inv) hb⟩
  68     have hh := map_filtration (.nu .sphere) .sphere model.nu.unitIso.hom
  69       (a ≫ model.nu.unitIso.inv) hn
  70     simpa only [Category.assoc,Iso.inv_hom_id,Category.comp_id] using hh

 156 private theorem theta5_square_nonzero_detection_exact_representative
 157     (theta : BiHom 62 64 (S_0_0 : SyntheticSpectrum)) (hc4 : Theta5SquareNonzeroDetectionAt theta) :
 158     ∃ u : BiHom 124 134 (S_0_0 : SyntheticSpectrum), UChoice u ∧

 190 private theorem correction_exact_representative
