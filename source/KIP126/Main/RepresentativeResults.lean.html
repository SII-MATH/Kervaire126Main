  36 /-- Actual lambda-six divisibility, retaining the choice of lift. A specified
  37 T detection of that lift is not a premise or a conclusion of this equation. -/
  38 private theorem lambda_three_eta_u_eq_lambda_six
  39     (u : BiHom 124 134 (S_0_0 : SyntheticSpectrum)) (hu : UChoice u) :
  40     ∃ y : BiHom 125 139 (S_0_0 : SyntheticSpectrum),
  41       lambdaMultiply 3 (sphereProduct KIP126.Literature.Standard.eta u) =
  42         lambdaMultiply 6 y := by
  43   let a : BiHom 125 133 (S_0_0 : SyntheticSpectrum) :=
  44     lambdaMultiply 3 (sphereProduct KIP126.Literature.Standard.eta u)
  45   have ha : FiltrationAtLeast model.syntheticTower 14 a :=
  46     lambda_three_eta_u_filtration_fourteen u hu
  47   have h := KIP126.Literature.Standard.selected_filtration_lambda
  48     .sphere 125 133 14 (by decide) (a ≫ model.nu.unitIso.inv)
  49   change FiltrationAtLeast model.syntheticTower 14 (a ≫ model.nu.unitIso.inv) ↔
  50     ∃ b : BiHom 125 139 (model.nu.functor.obj SphereSpectrum),
  51       lambdaMultiply 6 b = a ≫ model.nu.unitIso.inv at h
  52   obtain ⟨b, hb⟩ := h.mp
  53     (Internal.Selected.map_filtration .sphere (.nu .sphere) model.nu.unitIso.inv a ha)
  54   refine ⟨b ≫ model.nu.unitIso.hom, ?_⟩
  55   symm
  56   simpa only [a, lambdaMultiply, Category.assoc, Iso.inv_hom_id, Category.comp_id] using
  57     congrArg (fun z => z ≫ model.nu.unitIso.hom) hb

  59 /-- The difference of two lambda-three eta products lies in filtration 14
  60 at degree (125, 133). The stronger filtration-15 bound remains a separate goal. -/
  61 private theorem lambda_three_eta_u_difference_filtration_fourteen
  62     (u u' : BiHom 124 134 (S_0_0 : SyntheticSpectrum))
  63     (hu : UChoice u) (hu' : UChoice u') :
  64     FiltrationAtLeast model.syntheticTower 14
  65       ((show BiHom 125 133 (S_0_0 : SyntheticSpectrum) from
  66           lambdaMultiply 3 (sphereProduct KIP126.Literature.Standard.eta u)) -
  67         (show BiHom 125 133 (S_0_0 : SyntheticSpectrum) from
  68           lambdaMultiply 3 (sphereProduct KIP126.Literature.Standard.eta u'))) := by
  69   exact (model.syntheticTower.filtrationSubmodule (S_0_0 : SyntheticSpectrum)
  70     14 (125, 133)).sub_mem
  71       (lambda_three_eta_u_filtration_fourteen u hu)
