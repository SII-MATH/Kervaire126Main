  25 private theorem stem124_low_finite_quotient_zero
  26     (s r : ℤ) (hs : 6 ≤ s ∧ s ≤ 9) (hr : 2 ≤ r) (hrc : r ≤ 16-s)
  27     (hz : Subsingleton (Normalized.sphere.Page r (s,s+124))) :
  28     Subsingleton (KIP126.Literature.CycleQuotient model .sphere (15-s) (s-5) (s,s+124)) := by
  29   apply AlphaOne.quotient_vanishing_actual
  30   apply AlphaOne.quotient_subsingleton_of_le
  31   intro x hx
  32   obtain ⟨_,hxc⟩ := (AlphaOne.cycles_reaches (15-s) (s,s+124) (by omega) x).mp hx
  33   obtain ⟨xr,hxr⟩ := represents_before hr (show r ≤ 15-s+1 by omega) hxc
  34   rw [hz.elim xr 0] at hxr
  35   have hb := AlphaOne.stem123_weight130_finite_quotients_boundary_of_zero_representative hxr
  36   let E := Normalized.sphere
  37   let A := E.ssData (s,s+124)
  38   have hin (m : ℤ) (hm : s-4 ≤ m) : E.d m ((s,s+124)-E.diffDeg m)=0 := by
  39     have hi : (s,s+124)-E.diffDeg m=(s-m,(s-m)+125) := by
  40       change (s,s+124)-(m,m-1)=_
  41       ext <;> dsimp <;> omega
  42     rw [hi]
  43     ext z
  44     exact no_outgoing_stem125_low (s-m) (by omega) m (by omega) z
  45   have hB := boundaries_top_eq_of_d_eq_zero E (s-4) (by change (2:ℤ)≤s-4; omega) (s,s+124) hin
  46   change A.B ⊤=A.B ↑(s-4-2).toNat at hB
  47   have hle : A.B ↑(r-1-1).toNat ≤ A.B ↑(s-5-1).toNat := by
  48     have hi : s-4-2=s-5-1 := by omega
  49     rw [hi] at hB
  50     exact (A.B_mono le_top).trans hB.le
  51   obtain ⟨y,hy⟩ := hb
  52   refine ⟨(Subobject.ofLE _ _ hle) y,?_⟩
  53   change (Subobject.ofLE _ _ hle ≫ AlphaOne.boundaryMap (↑(s-5-1).toNat) (s,s+124)) y=x
  54   simpa only [AlphaOne.boundaryMap,AlphaOne.boundaryMapFor,A,E,
  55     ←Category.assoc,Subobject.ofLE_comp_ofLE] using hy

  57 private theorem ninth_stem124_weight130_filtration_ten
  58     (a : BiHom 124 130 (XModLambdaN (S_0_0 : SyntheticSpectrum) 9)) :
  59     FiltrationAtLeast model.syntheticTower 10 a := by
  60   have h6 := finite_sphere_bottom_filtration 9 (by decide) 6 130 a
  61   apply (filtration_iff_of_infinity_zero (model.convergence (.quotient 9 .sphere))
  62     6 10 124 130 (by decide) ?_ a).mp h6
  63   intro s hs hs'
  64   have hss : 6 ≤ s ∧ s ≤ 9 := by omega
  65   have hz : Subsingleton (KIP126.Literature.CycleQuotient model .sphere (15-s) (s-5) (s,s+124)) := by
  66     obtain ⟨h6,h7,h8,h9⟩ := stem124_finite_pages
  67     interval_cases s
  68     · exact stem124_low_finite_quotient_zero 6 6 (by decide) (by decide) (by decide) h6
  69     · exact stem124_low_finite_quotient_zero 7 6 (by decide) (by decide) (by decide) h7
  70     · exact stem124_low_finite_quotient_zero 8 3 (by decide) (by decide) (by decide) h8
  71     · exact stem124_low_finite_quotient_zero 9 4 (by decide) (by decide) (by decide) h9
  72   rw [show 124+s=s+124 by omega]
  73   apply finite_sphere_zero_of_classical_quotient 9 (by decide) s (s+124) 130 (by constructor <;> omega)
  74   convert hz using 1 <;> congr 1 <;> norm_num <;> omega
