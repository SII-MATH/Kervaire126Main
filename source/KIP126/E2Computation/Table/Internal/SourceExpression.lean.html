  18 private def sourceValue : (e : Expression) →
  19     Standard.sourceData.component e.degree.1 e.degree.2
  20   | .one => Standard.sourceData.unit
  21   | .named a => Standard.sourceData.atom a
  22   | .generator id s t =>
  23       if _ht : t ≤ 261 then
  24         if hi : id < Standard.archive.generators.count then
  25           if hd : Standard.archive.generators.degree ⟨id,hi⟩ = (s,t) then
  26             (eqToHom (congrArg (fun p : ℕ × ℕ => Standard.sourceData.component p.1 p.2) hd))
  27               (Standard.sourceGenerator ⟨id,hi⟩)
  28           else 0
  29         else 0
  30       else 0
  31   | .mul a b => Standard.sourceData.product a.degree.1 a.degree.2
  32       b.degree.1 b.degree.2 (sourceValue a) (sourceValue b)
  33 
  34 private theorem value_comparison (e : Expression) (ht : e.degree.2 ≤ 261) :
  35     e.value = Standard.comparison e.degree.1 e.degree.2 ht (sourceValue e) := by
  36   induction e with
  37   | one => exact Standard.comparison_unit.symm
  38   | named a => rfl
  39   | generator id s t =>
  40     change t ≤ 261 at ht
  41     change generatorImage id s t = Standard.comparison s t ht
  42       (sourceValue (.generator id s t))
  43     dsimp only [generatorImage, sourceValue]
  44     split_ifs <;> first | rfl | exact (map_zero _).symm
  45   | mul a b iha ihb =>
  46     have ha : a.degree.2 ≤ 261 := by change a.degree.2 + b.degree.2 ≤ 261 at ht; omega
  47     have hb : b.degree.2 ≤ 261 := by change a.degree.2 + b.degree.2 ≤ 261 at ht; omega
  48     have h := (Standard.comparison_product a.degree.1 a.degree.2 b.degree.1 b.degree.2
  49       ht (sourceValue a) (sourceValue b)).symm
  50     change Standard.product a.value b.value _ = _
  51     rw [iha ha, ihb hb]
  52     delta Standard.product
  53     simpa [Expression.degree, sourceValue, Nat.cast_add] using h
  54 
  55 end
