  12 private theorem actual_page_zero (s t : ℕ) (ht : t ≤ 261)
  13     (hz : basisDimension s t = 0) (x : model.classical.sphere.Page 2 (s,t)) : x = 0 := by
  14   let e := Standard.coordinates s t ht
  15   have hn : Standard.archive.basisCount s t = 0 := (archive_dimension s t ht).trans hz
  16   haveI : Subsingleton (Fin (Standard.archive.basisCount s t) →₀ ZMod 2) := by
  17     rw [hn]
  18     infer_instance
  19   apply e.injective
  20   exact Subsingleton.elim _ _
  21 
  22 end
