  29 /-- Coordinate validation is independent of any mathematical realization.
  30 Missing degree, duplicate/out-of-order indices, and out-of-range indices fail. -/
  31 public def strictlyIncreasing : List Nat → Bool
  32   | [] => true
  33   | [_] => true
  34   | a :: b :: rest => a < b && strictlyIncreasing (b :: rest)
  35 
  36 public def coordinatesValid (degrees : List Degree) (o : Spectrum)
  37     (s t : Nat) (indices : List Nat) : Bool :=
  38   match degrees.find? (fun d => d.spectrum == o && d.s == s && d.t == t) with
  39   | none => false
  40   | some d => indices.all (· < d.monomials.length) && strictlyIncreasing indices
