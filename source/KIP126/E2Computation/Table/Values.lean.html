  16 public inductive Expression where
  17   | one
  18   | named (a : Atom)
  19   | generator (id s t : ℕ)
  20   | mul (a b : Expression)
  21   deriving DecidableEq
  22 
  23 public def Expression.degree : Expression → ℕ × ℕ
  24   | .one => (0,0)
  25   | .named a => a.degree
  26   | .generator _ s t => (s,t)
  27   | .mul a b => (a.degree.1+b.degree.1,a.degree.2+b.degree.2)
  28 
  29 public def generatorImage (id s t : ℕ) : Page .sphere s t :=
  30   if ht : t ≤ 261 then
  31     if hi : id < Standard.archive.generators.count then
  32       if hd : Standard.archive.generators.degree ⟨id,hi⟩ = (s,t) then
  33         Standard.comparison s t ht
  34           ((eqToHom (congrArg (fun p : ℕ × ℕ => Standard.sourceData.component p.1 p.2) hd))
  35             (Standard.sourceGenerator ⟨id,hi⟩))
  36       else 0
  37     else 0
  38   else 0
  39 
  40 public def Expression.value : (e : Expression) → Page .sphere e.degree.1 e.degree.2
  41   | .one => model.sphereAlgebra.unit
  42   | .named a => Standard.atom a
  43   | .generator id s t => generatorImage id s t
  44   | .mul a b => Standard.product a.value b.value
  45       (by simp only [Expression.degree, Prod.mk_add_mk, Nat.cast_add])
