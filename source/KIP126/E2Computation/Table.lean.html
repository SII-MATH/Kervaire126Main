  24 public abbrev sequence (o : Raw.Spectrum) := model.classical.obj (object o)
  25 public abbrev Page (o : Raw.Spectrum) (s t : ℕ) := (sequence o).Page 2 (s,t)

  27 public def sphereBasis (s t i : ℕ) : model.classical.sphere.Page 2 (s,t) :=
  28   if ht : t ≤ 261 then
  29     if hi : i < Standard.archive.basisCount s t then
  30       (Standard.coordinates s t ht).symm (Finsupp.single ⟨i,hi⟩ 1)
  31     else 0
  32   else 0

  34 /-- The actual Cν basis; the following original table bindings constrain it. -/
  35 public def cofiberBasis (s t i : ℕ) : Page .nuCofiber s t := by
  36   sorry

  38 public def basis : (o : Raw.Spectrum) → (s t i : ℕ) → Page o s t
  39   | .sphere, s, t, i => sphereBasis s t i
  40   | .nuCofiber, s, t, i => cofiberBasis s t i

  42 public def decode (o : Raw.Spectrum) (s t : ℕ) (indices : List ℕ) :
  43     Option (Page o s t) :=
  44   if Raw.coordinatesValid Raw.degrees o s t indices then
  45     some ((indices.map (basis o s t)).sum)
  46   else none

  48 public def BasisCorrect (d : Raw.Degree) : Prop :=
  49   ∃ e : Page d.spectrum d.s d.t ≃ₗ[ℤ] (Fin d.monomials.length →₀ ZMod 2),
  50     ∀ i, e.symm (Finsupp.single i 1) = basis d.spectrum d.s d.t i.val

  52 /-- Exact original finite catalogue lengths, on the existing sphere archive. -/
  53 public theorem basis_count (d : Raw.Degree) (hd : d ∈ Raw.degrees)
  54     (hs : d.spectrum = .sphere) :
  55     Standard.archive.basisCount d.s d.t = d.monomials.length := by
  56   have hbound : ∀ d ∈ Raw.degrees, d.t ≤ 261 := by decide
  57   rw [archive_dimension d.s d.t (hbound d hd)]
  58   have hlength : ∀ d ∈ Raw.degrees, d.spectrum = .sphere →
  59       basisDimension d.s d.t = d.monomials.length := by decide
  60   exact hlength d hd hs

  67 private theorem table_degree_bound (d : Raw.Degree) (hd : d ∈ Raw.degrees) : d.t ≤ 261 := by
  68   have all : ∀ d ∈ Raw.degrees, d.t ≤ 261 := by decide

  71 public theorem basis_correct (d : Raw.Degree) (hd : d ∈ Raw.degrees) : BasisCorrect d := by
  72   rcases d with ⟨o,s,t,ms⟩
  73   cases o with
  74   | sphere =>
  75     have hc := basis_count ⟨.sphere,s,t,ms⟩ hd rfl
  76     have ht := table_degree_bound ⟨.sphere,s,t,ms⟩ hd
  77     dsimp only at hc ht
  78     change ∃ e : model.classical.sphere.Page 2 (s,t) ≃ₗ[ℤ] (Fin ms.length →₀ ZMod 2),
  79       ∀ i, e.symm (Finsupp.single i 1) = sphereBasis s t i.val
  80     rw [← hc]
  81     refine ⟨Standard.coordinates s t ht, ?_⟩
  82     intro i
  83     simp only [sphereBasis, dif_pos ht, dif_pos i.isLt]
  84   | nuCofiber => exact cofiber_basis_correct ⟨.nuCofiber,s,t,ms⟩ hd rfl
  85 
  86 end
