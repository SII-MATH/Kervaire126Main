  82 /-- One bounded comparison for one data algebra and the same actual sphere E₂.
  83 The bound applies to comparison, and to the total degree of every product. -/
  84 public structure SpherePresentation (D : Model C Syn) (P : ProgramData.{v}) where
  85   comparison : ∀ s t : ℕ, t ≤ 261 →
  86     P.component s t ≃ₗ[ℤ] D.classical.sphere.Page 2 ((s : ℤ),(t : ℤ))
  87   comparison_product : ∀ s t s' t' : ℕ, ∀ h : t+t' ≤ 261,
  88     ∀ (x : P.component s t) (y : P.component s' t'),
  89     comparison (s+s') (t+t') h (P.product s t s' t' x y) =
  90       D.sphereAlgebra.product (s,t) (s',t')
  91         (comparison s t (by omega) x) (comparison s' t' (by omega) y)
  92   comparison_unit : comparison 0 0 (by decide) P.unit = D.sphereAlgebra.unit
  93   h0 : comparison 1 1 (by decide) (P.atom .h0) = D.standardClasses.standardHi 0
  94   h1 : comparison 1 2 (by decide) (P.atom .h1) = D.standardClasses.standardHi 1
  95   h2 : comparison 1 4 (by decide) (P.atom .h2) = D.standardClasses.standardHi 2
  96   h4 : comparison 1 16 (by decide) (P.atom .h4) = D.standardClasses.standardHi 4
  97   h5 : comparison 1 32 (by decide) (P.atom .h5) = D.standardClasses.standardHi 5
  98   h6 : comparison 1 64 (by decide) (P.atom .h6) = D.standardClasses.standardH6

 103 /-- F₂-linearity follows from the existing additive structure; no second scalar
 104 structure is chosen on the actual page. -/
 105 public noncomputable def comparisonF2 (e : SpherePresentation D P) (s t : ℕ)
 106     (ht : t ≤ 261) :
 107     letI := D.sphereAlgebra.scalar (s,t)
 108     P.component s t ≃ₗ[ZMod 2] D.classical.sphere.Page 2 (s,t) := by
 109   letI := D.sphereAlgebra.scalar (s,t)
 110   exact { (e.comparison s t ht).toAddEquiv with
 111     map_smul' := fun a x => ZMod.map_smul (e.comparison s t ht).toAddMonoidHom a x }
 112 
 113 public theorem comparison_mul (e : SpherePresentation D P) (s t s' t' : ℕ)
 114     (h : t+t' ≤ 261) (x : P.component s t) (y : P.component s' t') :
 115     e.comparison (s+s') (t+t') h (P.product s t s' t' x y) =
 116       D.sphereAlgebra.product (s,t) (s',t')
 117         (e.comparison s t (by omega) x) (e.comparison s' t' (by omega) y) :=
 118   e.comparison_product s t s' t' h x y

 120 /-- The data square becomes the actual product of the designated h₆. -/
 121 public theorem h6_square (e : SpherePresentation D P) :
 122     e.comparison 2 128 (by decide)
 123       (P.product 1 64 1 64 (P.atom .h6) (P.atom .h6)) =
 124       D.standardClasses.standardH6Square := by
 125   have h := e.comparison_mul 1 64 1 64 (by decide) (P.atom .h6) (P.atom .h6)
 126   change e.comparison 2 128 _ (P.product 1 64 1 64 (P.atom .h6) (P.atom .h6)) =
 127     D.sphereAlgebra.product (1,64) (1,64)
 128       (e.comparison 1 64 _ (P.atom .h6)) (e.comparison 1 64 _ (P.atom .h6)) at h
 129   rw [e.h6] at h
 130   exact h
