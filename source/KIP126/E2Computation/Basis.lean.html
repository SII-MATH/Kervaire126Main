  18 /-- Complete generator catalogue of an archived sphere calculation.
  19 The fixed project list is exposed by Standard.generators; these are source
  20 coordinates rather than a second set of actual page classes. -/
  21 public structure GeneratorCatalogue where
  22   count : ℕ
  23   record : Fin count → String × ℕ × ℕ
  24 
  25 public abbrev GeneratorCatalogue.degree (G : GeneratorCatalogue) (i : Fin G.count) : ℕ × ℕ :=
  26   (G.record i).2
  27 
  28 /-- A degree-checked source monomial. Both multiplication order and every
  29 factor refer to the same complete generator catalogue. -/
  30 public inductive SourceMonomial (G : GeneratorCatalogue) : ℕ → ℕ → Type
  31   | one : SourceMonomial G 0 0
  32   | generator (i : Fin G.count) : SourceMonomial G (G.degree i).1 (G.degree i).2
  33   | mul {s t s' t' : ℕ} : SourceMonomial G s t → SourceMonomial G s' t' →
  34       SourceMonomial G (s+s') (t+t')
  35 
  36 /-- One mathematical description of the archived data. A producer must supply
  37 its complete lists from the pinned source. All selected names retain the exact
  38 archived id, name and degree, rather than merely an arbitrary name string. -/
  39 public structure SphereArchive where
  40   generators : GeneratorCatalogue
  41   atomIndex : Atom → Fin generators.count
  42   atom_record : ∀ a : Atom, ((atomIndex a).val, generators.record (atomIndex a)) = a.record
  43   basisCount : ℕ → ℕ → ℕ
  44   basisMonomial : ∀ s t : ℕ, Fin (basisCount s t) → SourceMonomial generators s t
  45 
  46 public theorem SphereArchive.atom_degree (A : SphereArchive) (a : Atom) :
  47     A.generators.degree (A.atomIndex a) = a.degree :=
  48   congrArg (fun r : ℕ × String × ℕ × ℕ => r.2.2) (A.atom_record a)
  49 
  50 /-- Evaluate each displayed source monomial using this very source product
  51 and unit. No independent evaluation function or actual-page value is chosen. -/
  52 public noncomputable def SourceMonomial.value {G : GeneratorCatalogue}
  53     (P : ProgramData.{v}) (generator : ∀ i : Fin G.count,
  54       P.component (G.degree i).1 (G.degree i).2) :
  55     ∀ {s t : ℕ}, SourceMonomial G s t → P.component s t
  56   | _, _, .one => P.unit
  57   | _, _, .generator i => generator i
  58   | _, _, .mul (s := s) (t := t) (s' := s') (t' := t') a b =>
  59       P.product s t s' t' (a.value P generator) (b.value P generator)
  60 
  61 /-- Full bounded basis correctness on one data presentation and one archive.
  62 The equivalence proves spanning and independence of all coordinate vectors;
  63 `basis_values` fixes every unit vector to the archived source monomial. -/
  64 public structure SourceBasisSpecification (P : ProgramData.{v}) (A : SphereArchive) where
  65   generator : ∀ i : Fin A.generators.count,
  66     P.component (A.generators.degree i).1 (A.generators.degree i).2
  67   atom_agrees : ∀ a : Atom,
  68     (eqToHom (congrArg (fun d : ℕ × ℕ => P.component d.1 d.2) (A.atom_degree a)))
  69       (generator (A.atomIndex a)) = P.atom a
  70   coordinates : ∀ s t : ℕ, t ≤ 261 →
  71     P.component s t ≃ₗ[ℤ] (Fin (A.basisCount s t) →₀ ZMod 2)
  72   basis_values : ∀ (s t : ℕ) (ht : t ≤ 261) (i : Fin (A.basisCount s t)),
  73     (coordinates s t ht).symm (Finsupp.single i 1) =
  74       (A.basisMonomial s t i).value P generator

  84 /-- Complete coordinates on the same actual sphere E₂, derived by composing
  85 the one source basis and the one comparison. This is the original cm1 contract. -/
  86 public noncomputable def coordinates (e : SpherePresentation D P)
  87     (S : SourceBasisSpecification P A) (s t : ℕ) (ht : t ≤ 261) :
  88     D.classical.sphere.Page 2 (s,t) ≃ₗ[ℤ] (Fin (A.basisCount s t) →₀ ZMod 2) :=
  89   (e.comparison s t ht).symm.trans (S.coordinates s t ht)
  90 
  91 /-- Every actual coordinate unit vector is the image of the same archived
  92 monomial, not a freshly selected basis of the same dimension. -/
  93 public theorem coordinates_basis (e : SpherePresentation D P)
  94     (S : SourceBasisSpecification P A) (s t : ℕ) (ht : t ≤ 261)
  95     (i : Fin (A.basisCount s t)) :
  96     (e.coordinates S s t ht).symm (Finsupp.single i 1) =
  97       e.comparison s t ht ((A.basisMonomial s t i).value P S.generator) := by
  98   change e.comparison s t ht ((S.coordinates s t ht).symm (Finsupp.single i 1)) = _
  99   rw [S.basis_values]
 100 
 101 /-- Original cm1's pullback formulation, derived without another input. -/
 102 public theorem comparison_symm_coordinates_basis (e : SpherePresentation D P)
 103     (S : SourceBasisSpecification P A) (s t : ℕ) (ht : t ≤ 261)
 104     (i : Fin (A.basisCount s t)) :
 105     (e.comparison s t ht).symm ((e.coordinates S s t ht).symm (Finsupp.single i 1)) =
 106       (A.basisMonomial s t i).value P S.generator := by
 107   rw [e.coordinates_basis S s t ht i]
 108   exact (e.comparison s t ht).symm_apply_apply _
 109 end SpherePresentation
