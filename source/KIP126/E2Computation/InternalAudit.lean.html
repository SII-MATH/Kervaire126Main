  21 /-! Trust audit for fixed upstream declarations. Only the exact individually
  22 registered proof/construction holes are permitted; project axioms and synthetic
  23 (elaboration-error) holes are always rejected. SpectralSequence stays sorry-free. -/
  24 open Lean Elab Command
  25 set_option maxHeartbeats 0
  26 set_option maxRecDepth 8192
  27 run_cmd do
  28   let ledger ← Lean.ofExcept <| Json.parse proofObligationLedger
  29   let entries ← Lean.ofExcept <| ledger.getObjValAs? (Array Json) "obligations"
  30   let env ← getEnv
  31   let expected : Array Json ← entries.filterM fun (row : Json) => do
  32     let name ← Lean.ofExcept <| row.getObjValAs? String "name"
  33     let some idx := env.getModuleIdxFor? name.toName | return false
  34     let mod := env.header.moduleNames[idx.toNat]!
  35     pure ((`KIP126.E2Computation).isPrefixOf mod)
  36   let expectedNames : Array String ← expected.mapM fun (row : Json) => Lean.ofExcept <| row.getObjValAs? String "name"
  37   let mut checked : Nat := 0
  38   let mut holes : Array String := #[]
  39   for (name, info) in env.constants.toList do
  40     let some idx := env.getModuleIdxFor? name | continue
  41     let mod := env.header.moduleNames[idx.toNat]!
  42     unless (`KIP126.E2Computation).isPrefixOf mod do continue
  43     if info.isAxiom then throwError "Project axiom declaration: {name}"
  44     let body := info.value? (allowOpaque := true)
  45     if info.type.hasSyntheticSorry || body.any (·.hasSyntheticSorry) then
  46       throwError "Synthetic sorry from failed elaboration: {name}"
  47     if info.type.hasSorry || body.any (·.hasSorry) then
  48       holes := holes.push name.toString
  49       let some entry := expected.find? (fun e => e.getObjValAs? String "name" == .ok name.toString)
  50         | throwError "Unregistered proof/construction obligation: {name}"
  51       let expectedType ← Lean.ofExcept <| entry.getObjValAs? String "type_fingerprint"
  52       let expectedBody ← Lean.ofExcept <| entry.getObjValAs? String "body_fingerprint"
  53       unless expectedType == toString (hash (toString info.type)) &&
  54           expectedBody == toString (hash (toString body)) do
  55         throwError "Registered obligation changed; review its exact type/body: {name}"
  56     let axioms ← collectAxioms name
  57     let unexpected := axioms.filter fun n =>
  58       n != `propext && n != `Classical.choice && n != `Quot.sound && n != `sorryAx
  59     unless unexpected.isEmpty do
  60       throwError "Unexpected axiom dependency in {name}: {unexpected}"
  61     checked := checked + 1
  62   unless holes.qsort (· < ·) == expectedNames.qsort (· < ·) do
  63     throwError "Proof obligation registry has stale or missing declarations: {holes}"
  64   logInfo m!"E2Computation trust audit: {checked} declarations; {holes.size} individually registered pending proofs/constructions; no project axioms"
