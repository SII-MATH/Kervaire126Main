  27 public abbrev nuZero (X : ClassicalObject) :=
  28   (SyntheticCategory.biShift (0,0)).obj (D.nu.functor.obj (X.obj D.auxiliary))

  30 public def firstLabel (X : ClassicalObject) (s t : ℤ)
  31     (x : (D.classical.obj (X.obj D.auxiliary)).Page 2 (s,t)) :
  32     BiHom (t-s) t (XModLambdaN (nuZero D X) 1) := by
  33   simpa only [add_zero] using (D.firstQuotient X 0 s t).symm x

  35 /-- A single common infinite-cycle representative; no nonzero assertion. -/
  36 public def ClassicalPermanentCycle (X : ClassicalObject) (p : ℤ × ℤ)
  37     (x : (D.classical.obj (X.obj D.auxiliary)).Page 2 p) : Prop :=
  38   let E := D.classical.obj (X.obj D.auxiliary)
  39   ∃ z : (Subobject.underlying.obj ((E.ssData p).Z ⊤) : ModuleCat ℤ),
  40     (Subobject.ofLE _ _ ((E.ssData p).Z_anti le_top) ≫
  41       (E.ssData p).pageπ (↑(2-E.r₀).toNat : WithTop ℕ)) z = x

  43 public def bocksteinArrow (X : ClassicalObject) (q : ℕ) (hq : 0 < q) :
  44     XModLambdaN (nuZero D X) q ⟶
  45       (SyntheticCategory.biShift (1,-(q : ℤ))).obj (XModLambdaN (nuZero D X) 1) :=
  46   ((D.quotients.tower (nuZero D X)).triangle hq (Nat.lt_succ_self q)).delta ≫
  47     (SyntheticCategory.biShift_compat (Syn := Syn) 1).inv.app _ ≫
  48     (SyntheticCategory.biShift_comp (0,-(q : ℤ)) (1,0)).hom.app _ ≫
  49     eqToHom (by congr 1 <;> simp)

  51 public def bocksteinLabel (X : ClassicalObject) (q : ℕ) (hq : 0 < q) (s t : ℤ)
  52     (a : BiHom (t-s) t (XModLambdaN (nuZero D X) q)) :
  53     (D.classical.obj (X.obj D.auxiliary)).Page 2 (s+q+1,t+q) := by
  54   let b : BiHom (t-s-1) (t+q) (XModLambdaN (nuZero D X) 1) :=
  55     (susp_invariance (t-s-1) (t+q) 1 (-(q : ℤ)) _).symm
  56       (homotopyRegrade (by omega) (by omega) (a ≫ bocksteinArrow D X q hq))
  57   exact D.firstQuotient X 0 (s+q+1) (t+q)
  58     (homotopyRegrade (by omega) (by omega) b)

  60 /-- BHS A.1(1): reaching E_(q+1), with zero permitted, is exactly the
  61 existence of a lift through this actual λ^q quotient and actual restriction. -/
  62 public def FiniteLiftCriterion : Prop :=
  63   ∀ (X : ClassicalObject) (q : ℕ) (hq : 0 < q) (s t : ℤ)
  64     (x : (D.classical.obj (X.obj D.auxiliary)).Page 2 (s,t)),
  65     (ReachesPage (D.classical.obj (X.obj D.auxiliary)) (q+1) (s,t) x ↔
  66       ∃ a : BiHom (t-s) t (XModLambdaN (nuZero D X) q),
  67         a ≫ (D.quotients.tower (nuZero D X)).rho 1 q hq = firstLabel D X s t x)
  68 
  69 public def BocksteinDifferential : Prop :=
  70   ∀ (X : ClassicalObject) (q : ℕ) (hq : 0 < q) (s t : ℤ)
  71     (x : (D.classical.obj (X.obj D.auxiliary)).Page 2 (s,t)),
  72     ReachesPage (D.classical.obj (X.obj D.auxiliary)) (q+1) (s,t) x →
  73     ∃ a : BiHom (t-s) t (XModLambdaN (nuZero D X) q),
  74       a ≫ (D.quotients.tower (nuZero D X)).rho 1 q hq = firstLabel D X s t x ∧
  75       HasDifferential (D.classical.obj (X.obj D.auxiliary))
  76         (q+1) (s,t) (s+q+1,t+q) x (bocksteinLabel D X q hq s t a)
  77 
  78 public def PermanentLiftAt (X : ClassicalObject) : Prop :=
  79   ∀ (s t : ℤ) (x : (D.classical.obj (X.obj D.auxiliary)).Page 2 (s,t)),
  80     (ClassicalPermanentCycle D X (s,t) x ↔
  81       ∃ a : BiHom (t-s) t (nuZero D X), quotientClass 1 a = firstLabel D X s t x)
  82 
  83 public def LambdaDivisible (X : ClassicalObject) (m w s : ℤ) (h : w-m ≤ s)
  84     (a : BiHom m w (D.nu.functor.obj (X.obj D.auxiliary))) : Prop :=
  85   ∃ b : BiHom m (m+s) (D.nu.functor.obj (X.obj D.auxiliary)),
  86     homotopyRegrade rfl (by rw [Int.toNat_of_nonneg (by omega)]; omega)
  87       (lambdaMultiply (m+s-w).toNat b) = a
  88 
  89 public def FiltrationLambdaAt (X : ClassicalObject) : Prop :=
  90   ∀ (m w s : ℤ) (h : w-m ≤ s)
  91     (a : BiHom m w (D.nu.functor.obj (X.obj D.auxiliary))),
  92     FiltrationAtLeast D.syntheticTower s a ↔ LambdaDivisible D X m w s h a
  93 
  94 /-- Actual labeled differential equivalence, with the λ exponent on the
  95 target increased by r-1 so both sides have exactly the same weight. -/
  96 public def DifferentialRigidity : Prop :=
  97   ∀ (X : ClassicalObject) (a s t : ℤ) (r k : ℕ), 2 ≤ r →
  98     ∀ (x : (D.classical.obj (X.obj D.auxiliary)).Page 2 (s,t))
  99       (y : (D.classical.obj (X.obj D.auxiliary)).Page 2 (s+r,t+r-1)),
 100     (HasDifferential (D.classical.obj (X.obj D.auxiliary)) r
 101         (s,t) (s+r,t+r-1) x y ↔
 102       SyntheticHasDifferential
 103         (D.family.obj ((SyntheticCategory.biShift (0,a)).obj
 104           (D.nu.functor.obj (X.obj D.auxiliary)))) r
 105         (s,t,t+a-k) (s+r,t+r-1,(t+r-1)+a-(k+(r-1) : ℕ))
 106         (D.comparison.nuE2 X a s t k x)
 107         (D.comparison.nuE2 X a (s+r) (t+r-1) (k+(r-1)) y))

 109 /-- λ-localization kernel on the existing object closure. A producer must
 110 transport the full source category's compactness argument to this exact
 111 realization; compactness of the hypercomplete sphere is not assumed. -/
 112 public def RealizationKernel : Prop :=
 113   ∀ (X : SyntheticObject) (m w : ℤ) (a : BiHom m w (X.obj D.nu D.auxiliary)),
 114     (D.recovery.realization.map a = 0 ↔ ∃ k : ℕ, lambdaMultiply k a = 0)

 133 /-- Canonical realization on νX with the zero shift removed. -/
 134 public def realizeNuZero (R : RealizationCoordinates D) (X : ClassicalObject)
 135     {m w : ℤ} (a : BiHom m w (nuZero D X)) : HomotopyGroup m (X.obj D.auxiliary) :=
 136   (R.sphere m w).hom ≫ D.recovery.realization.map a ≫
 137     D.recovery.realization.map (SyntheticCategory.biShift_zero.hom.app _) ≫
 138     D.recovery.nuRealizationIso.hom.app _

 140 /-- BHS realization results with every source applicability condition. The
 141 boundary lift is existential: it never asserts a formula for an arbitrary lift. -/
 142 private structure BHSRealizationResults (R : RealizationCoordinates D) : Prop where
 143   lifetime : ∀ (X : ClassicalObject) (products : HasProductsOfShape ℕ C),
 144     BHSObjectApplicability D products (X.obj D.auxiliary) →
 145     ∀ (s t : ℤ) (r : ℕ), 1 ≤ r →
 146     ∀ (x : (D.classical.obj (X.obj D.auxiliary)).Page 2 (s,t))
 147       (a : BiHom (t-s) t (nuZero D X)),
 148     ClassicalPermanentCycle D X (s,t) x →
 149     SurvivesTo (D.classical.obj (X.obj D.auxiliary)) (r+1) (s,t) x →
 150     quotientClass 1 a = firstLabel D X s t x → lambdaMultiply (r-1) a ≠ 0
 151   detection : ∀ (X : ClassicalObject) (products : HasProductsOfShape ℕ C),
 152     BHSObjectApplicability D products (X.obj D.auxiliary) →
 153     ∀ (s t : ℤ) (x : (D.classical.obj (X.obj D.auxiliary)).Page 2 (s,t))
 154       (a : BiHom (t-s) t (nuZero D X)),
 155     NonzeroSurvival (D.classical.obj (X.obj D.auxiliary)) (s,t) x →
 156     quotientClass 1 a = firstLabel D X s t x →
 157     ClassicalDetects (D.classicalConvergence X) (s,t) x (realizeNuZero D R X a)
 158   prescribed_lift : ∀ (X : ClassicalObject) (products : HasProductsOfShape ℕ C),
 159     BHSObjectApplicability D products (X.obj D.auxiliary) →
 160     ∀ (s t : ℤ) (x : (D.classical.obj (X.obj D.auxiliary)).Page 2 (s,t))
 161       (a : HomotopyGroup (t-s) (X.obj D.auxiliary)),
 162     NonzeroSurvival (D.classical.obj (X.obj D.auxiliary)) (s,t) x →
 163     ClassicalDetects (D.classicalConvergence X) (s,t) x a →
 164     ∃ b : BiHom (t-s) t (nuZero D X), quotientClass 1 b = firstLabel D X s t x ∧
 165       realizeNuZero D R X b = a
 166   boundary_lift : ∀ (X : ClassicalObject) (products : HasProductsOfShape ℕ C),
 167     BHSObjectApplicability D products (X.obj D.auxiliary) →
 168     ∀ (s t : ℤ) (r : ℕ), 2 ≤ r →
 169     ∀ y : (D.classical.obj (X.obj D.auxiliary)).Page 2 (s,t),
 170     ClassicalPermanentCycle D X (s,t) y →
 171     HitOnPage (D.classical.obj (X.obj D.auxiliary)) r (s,t) y →
 172     ∃ a : BiHom (t-s) t (nuZero D X), quotientClass 1 a = firstLabel D X s t y ∧
 173       lambdaMultiply (r-1) a = 0

 175 end
