  26     Subobject.underlying.obj (((D.classical.obj (X.obj D.auxiliary)).ssData p).Z n) ⟶

  63     Antitone (fun c => cycles D X c p) := by
  64   intro a b hab x hx
  65   obtain ⟨z, rfl⟩ := hx

  68   refine ⟨(Subobject.ofLE _ _ (((D.classical.obj (X.obj D.auxiliary)).ssData p).Z_anti h)) z, ?_⟩
  69   exact congrArg (fun f => f z) (cycleLabel_factor D X p _ _ h)

  89 /-- The actual submodule quotient, uniformly over its scalar ring. -/
  90 public abbrev E2Subquotient {R : Type a} [Ring R] {A : Type b}
  91     [AddCommGroup A] [Module R A] (Z B : Submodule R A) := Z ⧸ B.comap Z.subtype
  92 
  93 public def subquotientClass {R : Type a} [Ring R] {A : Type b}
  94     [AddCommGroup A] [Module R A] (Z B : Submodule R A) : Z →ₗ[R] E2Subquotient Z B :=
  95   (B.comap Z.subtype).mkQ
  96 
  97 public def subquotientMap {R : Type a} [Ring R] {A : Type b}
  98     [AddCommGroup A] [Module R A] {Z Z' B B' : Submodule R A}
  99     (hZ : Z ≤ Z') (hB : B ≤ B') : E2Subquotient Z B →ₗ[R] E2Subquotient Z' B' :=
 100   (B.comap Z.subtype).mapQ (B'.comap Z'.subtype)
 101     (Submodule.inclusion hZ) (fun _ hx => hB hx)
 102 
 103 public theorem subquotientMap_surjective {R : Type a} [Ring R] {A : Type b}
 104     [AddCommGroup A] [Module R A] {Z B B' : Submodule R A} (hB : B ≤ B') :
 105     Function.Surjective (subquotientMap (le_rfl : Z ≤ Z) hB) := by
 106   intro y
 107   obtain ⟨x,rfl⟩ := Submodule.Quotient.mk_surjective _ y
 108   exact ⟨Submodule.Quotient.mk x, rfl⟩

 110 public abbrev CycleQuotient (X : ClassicalObject) (c b : ℤ) (p : Bidegree) :=
 111   E2Subquotient (cycles D X c p) (boundaries D X b p)
 112 public abbrev PermanentQuotient (X : ClassicalObject) (b : ℤ) (p : Bidegree) :=
 113   E2Subquotient (permanentCycles D X p) (boundaries D X b p)

 115 public def quotientMap (X : ClassicalObject) {c c' b b' : ℤ}
 116     (hc : c' ≤ c) (hb : b ≤ b') (p : Bidegree) :
 117     CycleQuotient D X c b p →ₗ[ℤ] CycleQuotient D X c' b' p :=
 118   subquotientMap (cycles_antitone D X p hc) (boundaries_monotone D X p hb)

 120 public def permanentQuotientMap (X : ClassicalObject) {b b' : ℤ}
 121     (hb : b ≤ b') (p : Bidegree) :
 122     PermanentQuotient D X b p →ₗ[ℤ] PermanentQuotient D X b' p :=
 123   subquotientMap (le_refl (permanentCycles D X p)) (boundaries_monotone D X p hb)

 125 public def permanentToFinite (X : ClassicalObject) (c b : ℤ) (p : Bidegree) :
 126     PermanentQuotient D X b p →ₗ[ℤ] CycleQuotient D X c b p :=
 127   subquotientMap (permanentCycles_le_cycles D X p c) (le_refl (boundaries D X b p))

 175 /-- The E∞ shift is determined by actual filtered homotopy transport and
 176 the model's given convergence, using the established AdamsSynthetic interface. -/
 177 public def infinityShift (K : ∀ k : ℕ, SyntheticESSShiftComparison D (0,-(k:ℤ)))
 178     (X : SyntheticObject) (k : ℕ) (p : Bidegree) (w : ℤ) :
 179     ((D.family.obj (X.obj D.nu D.auxiliary)).sequence.ssData (p.1,p.2,w)).eInfty ≅
 180       ((D.family.obj ((SyntheticObject.shift (0,-(k:ℤ)) X).obj D.nu D.auxiliary)).sequence.ssData
 181         (p.1,p.2,w-k)).eInfty :=
 182   eqToIso (by simp) ≪≫
 183     towerInfinityIso (D.convergence X) (D.convergence (.shift (0,-(k:ℤ)) X))
 184       ((K k).filtered X (p.2-p.1,w)) p.1 ≪≫ eqToIso (by simp [sub_eq_add_neg])

 186 /-- Naturality concerns the shift fixed by the actual homotopy maps, so no
 187 unrelated infinity-page automorphism can be selected. -/
 188 public structure InfinityShiftCoordinates where
 189   comparison : ∀ k : ℕ, SyntheticESSShiftComparison D (0,-(k:ℤ))
 190   natural : ∀ (X Y : SyntheticObject) (f : X.obj D.nu D.auxiliary ⟶ Y.obj D.nu D.auxiliary)
 191     (k : ℕ) (p : Bidegree) (w : ℤ),
 192     (D.family.map f).eInftyMap (p.1,p.2,w) ≫ (infinityShift D comparison Y k p w).hom =
 193       (infinityShift D comparison X k p w).hom ≫
 194         (D.family.map ((SyntheticCategory.biShift (0,-(k:ℤ))).map f)).eInftyMap (p.1,p.2,w-k)

 196 public def InfinityShiftCoordinates.shift (S : InfinityShiftCoordinates D) :=

 239 /-- Weight comparison lowers from w₂ to w₁ by the actual λ power.
 240 The old KIPBase weight-raising direction reverses the boundary inclusion. -/
 241 public def InfinityShiftCoordinates.weightMap (S : InfinityShiftCoordinates D)
 242     (X : ClassicalObject) (s t w₁ w₂ : ℤ) (hw : w₁ ≤ w₂) :
 243     ((D.family.nu D.nu (X.obj D.auxiliary)).sequence.ssData (s,t,w₂)).eInfty ⟶
 244       ((D.family.nu D.nu (X.obj D.auxiliary)).sequence.ssData (s,t,w₁)).eInfty :=

 250 private structure EInfinityCompatibility (P : EInfinityPresentation D) (S : InfinityShiftCoordinates D) : Prop where
 251   lambda_nu : ∀ (X : ClassicalObject) (k : ℕ) (p : Bidegree) (w : ℤ) (hw : w ≤ p.2)
 252       (x : ((D.family.nu D.nu (X.obj D.auxiliary)).sequence.ssData (p.1,p.2,w)).eInfty),
 253     P.nuWindow D X p (w-k) (by omega) (S.lambdaMap D (.nu X) k p w x) =
 254       permanentQuotientMap D X (by omega) p (P.nuWindow D X p w hw x)
 255   lambda_finite : ∀ (X : ClassicalObject) (q k : ℕ) (hkq : k < q)
 256       (p : Bidegree) (w : ℤ) (hw : 0 ≤ p.2-w ∧ p.2-w < (q-k : ℕ))
 257       (x : ((D.family.quotient (D.nu.functor.obj (X.obj D.auxiliary)) (q-k)).sequence.ssData (p.1,p.2,w)).eInfty),
 258     P.finiteWindow D X q (by omega) p (w-k) (by constructor <;> omega)
 259         (S.finiteLambdaMap D X q k hkq p w x) =
 260       quotientMap D X (by omega) (by omega) p
 261         (P.finiteWindow D X (q-k) (by omega) p w hw x)
 262   rho_finite : ∀ (X : ClassicalObject) (i j : ℕ) (hi : 0 < i) (hij : i ≤ j)
 263       (p : Bidegree) (w : ℤ) (hw : 0 ≤ p.2-w ∧ p.2-w < i)
 264       (x : ((D.family.quotient (D.nu.functor.obj (X.obj D.auxiliary)) j).sequence.ssData (p.1,p.2,w)).eInfty),
 265     P.finiteWindow D X i hi p w hw
 266         ((D.family.map ((D.quotients.tower (D.nu.functor.obj (X.obj D.auxiliary))).rho i j hij)).eInftyMap (p.1,p.2,w) x) =
 267       quotientMap D X (by omega) le_rfl p
 268         (P.finiteWindow D X j (by omega) p w (by constructor <;> omega) x)
 269   rho_nu : ∀ (X : ClassicalObject) (q : ℕ) (hq : 0 < q)
 270       (p : Bidegree) (w : ℤ) (hw : 0 ≤ p.2-w ∧ p.2-w < q)
 271       (x : ((D.family.nu D.nu (X.obj D.auxiliary)).sequence.ssData (p.1,p.2,w)).eInfty),
 272     P.finiteWindow D X q hq p w hw
 273         ((D.family.quotientProjection (D.nu.functor.obj (X.obj D.auxiliary)) q).eInftyMap (p.1,p.2,w) x) =
 274       permanentToFinite D X (q-p.2+w) (1+p.2-w) p
 275         (P.nuWindow D X p w (by omega) x)
 276   nu_labels : ∀ (X : ClassicalObject) (s t w : ℤ) (hw : w ≤ t)
 277       (x : (D.classical.obj (X.obj D.auxiliary)).Page 2 (s,t))
 278       (z : ((D.family.nu D.nu (X.obj D.auxiliary)).sequence.ssData (s,t,w)).eInfty),
 279     HasInfinityRepresentative _ 2 (s,t,w) (nuWeightLabel D X s t w hw x) z ↔
 280       ∃ hx : x ∈ permanentCycles D X (s,t),
 281         P.nuWindow D X (s,t) w hw z = subquotientClass (permanentCycles D X (s,t)) (boundaries D X (1+t-w) (s,t)) ⟨x,hx⟩
 282   finite_labels : ∀ (X : ClassicalObject) (q : ℕ) (hq : 0 < q)
 283       (s t w : ℤ) (hw : 0 ≤ t-w ∧ t-w < q)
 284       (x : (D.classical.obj (X.obj D.auxiliary)).Page 2 (s,t))
 285       (z : ((D.family.quotient (D.nu.functor.obj (X.obj D.auxiliary)) q).sequence.ssData (s,t,w)).eInfty),
 286     HasInfinityRepresentative _ 2 (s,t,w)
 287       ((D.family.quotientProjection (D.nu.functor.obj (X.obj D.auxiliary)) q).pageMap 2 (s,t,w)
 288         (nuWeightLabel D X s t w (by omega) x)) z ↔
 289       ∃ hx : x ∈ cycles D X (q-t+w) (s,t),
 290         P.finiteWindow D X q hq (s,t) w hw z = subquotientClass (cycles D X (q-t+w) (s,t)) (boundaries D X (1+t-w) (s,t)) ⟨x,hx⟩

 292 /-- Surjectivity is derived from the actual quotient formula and λ equation. -/
 293 private theorem EInfinityCompatibility.lambda_epi {P : EInfinityPresentation D}
 294     {S : InfinityShiftCoordinates D} (R : EInfinityCompatibility D P S)
 295     (X : ClassicalObject) (k : ℕ) (p : Bidegree) (w : ℤ) (hw : w ≤ p.2) :
 296     Epi (S.lambdaMap D (.nu X) k p w) := by
 297   apply (ModuleCat.epi_iff_surjective _).mpr
 298   intro y
 299   obtain ⟨z,hz⟩ := permanentQuotientMap_surjective D X
 300     (show 1+p.2-w ≤ 1+p.2-(w-k) by omega) p
 301     (P.nuWindow D X p (w-k) (by omega) y)
 302   refine ⟨(P.nuWindow D X p w hw).symm z, ?_⟩
 303   apply (P.nuWindow D X p (w-k) (by omega)).injective
 304   rw [R.lambda_nu X k p w hw]
 305   simpa only [AddEquiv.apply_symm_apply] using hz
