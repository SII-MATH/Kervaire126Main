  28 /-! Trust audit for fixed upstream declarations. Only the exact individually
  29 registered proof/construction holes are permitted; project axioms and synthetic
  30 (elaboration-error) holes are always rejected. SpectralSequence stays sorry-free. -/
  31 open Lean Elab Command
  32 set_option maxHeartbeats 0
  33 set_option maxRecDepth 8192
  34 run_cmd do
  35   let ledger ← Lean.ofExcept <| Json.parse proofObligationLedger
  36   let entries ← Lean.ofExcept <| ledger.getObjValAs? (Array Json) "obligations"
  37   let expected : Array Json ← entries.filterM fun (row : Json) => do
  38     let mod ← Lean.ofExcept <| row.getObjValAs? String "module"
  39     pure ((`KIP126.Literature).isPrefixOf mod.toName)
  40   let expectedNames : Array String ← expected.mapM fun (row : Json) => Lean.ofExcept <| row.getObjValAs? String "name"
  41   let env ← getEnv
  42   let mut checked : Nat := 0
  43   let mut holes : Array String := #[]
  44   for (name, info) in env.constants.toList do
  45     let some idx := env.getModuleIdxFor? name | continue
  46     let mod := env.header.moduleNames[idx.toNat]!
  47     unless (`KIP126.Literature).isPrefixOf mod do continue
  48     if info.isAxiom then throwError "Project axiom declaration: {name}"
  49     let body := info.value? (allowOpaque := true)
  50     if info.type.hasSyntheticSorry || body.any (·.hasSyntheticSorry) then
  51       throwError "Synthetic sorry from failed elaboration: {name}"
  52     if info.type.hasSorry || body.any (·.hasSorry) then
  53       holes := holes.push name.toString
  54       let some entry := expected.find? (fun e => e.getObjValAs? String "name" == .ok name.toString)
  55         | throwError "Unregistered proof/construction obligation: {name}"
  56       let expectedType ← Lean.ofExcept <| entry.getObjValAs? String "type_fingerprint"
  57       let expectedBody ← Lean.ofExcept <| entry.getObjValAs? String "body_fingerprint"
  58       unless expectedType == toString (hash (toString info.type)) &&
  59           expectedBody == toString (hash (toString body)) do
  60         throwError "Registered obligation changed; review its exact type/body: {name}"
  61     let axioms ← collectAxioms name
  62     let unexpected := axioms.filter fun n =>
  63       n != `propext && n != `Classical.choice && n != `Quot.sound && n != `sorryAx
  64     unless unexpected.isEmpty do
  65       throwError "Unexpected axiom dependency in {name}: {unexpected}"
  66     checked := checked + 1
  67   unless holes.qsort (· < ·) == expectedNames.qsort (· < ·) do
  68     throwError "Proof obligation registry has stale or missing declarations: {holes}"
  69   logInfo m!"Literature trust audit: {checked} declarations; {holes.size} individually registered pending proofs/constructions; no project axioms"
