  18 /-- Apply this exact functor with its specified suspension comparison. -/
  19 public def HoCofiberSequence.map (T : HoCofiberSequence (C := C)) (F : C ⥤ C)
  20     [F.CommShift ℤ] [F.IsTriangulated] : HoCofiberSequence (C := C) where
  21   X := F.obj T.X
  22   Y := F.obj T.Y
  23   Z := F.obj T.Z
  24   f := F.map T.f
  25   g := F.map T.g
  26   h := F.map T.h ≫ (F.commShiftIso (1 : ℤ)).hom.app T.X
  27   distinguished := F.map_distinguished _ T.distinguished
  28 
  29 /-- The actual boundary map, obtained by desuspending the triangle's last
  30 map; its source and target are the represented homotopy groups. -/
  31 public noncomputable def connectingHomomorphism
  32     (T : HoCofiberSequence (C := C)) (n : ℤ) :
  33     HomotopyGroup n T.Z →+ HomotopyGroup (n-1) T.X where
  34   toFun z := (shiftFunctorAdd' C n (-1) (n-1) (by omega)).hom.app SphereSpectrum ≫
  35     (shiftFunctor C (-1)).map (z ≫ T.h) ≫
  36     (shiftFunctorCompIsoId C 1 (-1) (by omega)).hom.app T.X
  37   map_zero' := by simp only [Functor.map_zero, zero_comp, comp_zero]
  38   map_add' := by
  39     intros
  40     simp only [Preadditive.add_comp, Functor.map_add, Preadditive.comp_add]
  41 
  42 end KIP126.StableHomotopy

  50 /-- Exact tensor functors with fixed shift choices. Realizing TC3 for
  51 these choices remains an applicability obligation of the source model. -/
  52 public structure MayContext where
  53   leftShift : ∀ X : Syn, (tensorLeft X).CommShift ℤ
  54   rightShift : ∀ X : Syn, (tensorRight X).CommShift ℤ
  55   leftExact : ∀ X : Syn, letI := leftShift X; (tensorLeft X).IsTriangulated
  56   rightExact : ∀ X : Syn, letI := rightShift X; (tensorRight X).IsTriangulated
  57 
  58 /-- The part of TC3 and its pushpull square used by the route. The two
  59 boundaries have opposite signs in the same actual suspension target. -/
  60 public structure MayPushpullData (B : MayContext Syn)
  61     (T U : HoCofiberSequence (C := Syn)) where
  62   vertex : Syn
  63   j1 : vertex ⟶ T.X ⊗ U.Z
  64   j2 : vertex ⟶ T.Y ⊗ U.Y
  65   j3 : vertex ⟶ T.Z ⊗ U.X
  66   square : j1 ≫ (T.f ▷ U.Z) = j2 ≫ (T.Y ◁ U.g)
  67   other_square : j2 ≫ (T.g ▷ U.Y) = j3 ≫ (T.Z ◁ U.f)
  68   boundary : letI := B.leftShift; letI := B.rightShift
  69     letI := B.leftExact; letI := B.rightExact
  70     j1 ≫ (U.map (tensorLeft T.X)).h = -(j3 ≫ (T.map (tensorRight U.X)).h)
  71   lift : ∀ (n : ℤ) (a : HomotopyGroup n (T.X ⊗ U.Z))
  72       (b : HomotopyGroup n (T.Y ⊗ U.Y)),
  73     a ≫ (T.f ▷ U.Z) = b ≫ (T.Y ◁ U.g) →
  74     ∃ v : HomotopyGroup n vertex, v ≫ j1 = a ∧ v ≫ j2 = b
  75 
  76 /-- Source theorem input, not a construction of a TC3 model. -/
  77 private def MaySourceResults (B : MayContext Syn) : Prop :=
  78   ∀ T U, Nonempty (MayPushpullData Syn B T U)
  79 
  80 public def MayContext.SignedBoundary (B : MayContext Syn) : Prop :=
  81   letI := B.leftShift; letI := B.rightShift
  82   letI := B.leftExact; letI := B.rightExact
  83   ∀ (T U : HoCofiberSequence (C := Syn)) (n : ℤ)
  84     (a : HomotopyGroup n (T.X ⊗ U.Z)) (b : HomotopyGroup n (T.Y ⊗ U.Y)),
  85     inducedMap (T.f ▷ U.Z) n a = inducedMap (T.Y ◁ U.g) n b →
  86     ∃ c : HomotopyGroup n (T.Z ⊗ U.X),
  87       inducedMap (T.g ▷ U.Y) n b = inducedMap (T.Z ◁ U.f) n c ∧
  88       connectingHomomorphism (U.map (tensorLeft T.X)) n a =
  89         -(connectingHomomorphism (T.map (tensorRight U.X)) n c)

  91 variable {Syn}
  92 set_option backward.isDefEq.respectTransparency false in
  93 /-- A common pushpull lift proves the signed formula. -/
  94 private theorem may_signed_boundary_of_source (B : MayContext Syn)
  95     (hB : MaySourceResults Syn B) : B.SignedBoundary := by
  96   letI := B.leftShift
  97   letI := B.rightShift
  98   letI := B.leftExact
  99   letI := B.rightExact
 100   intro T U n a b hab
 101   rcases hB T U with ⟨P⟩
 102   rcases P.lift n a b hab with ⟨z, hza, hzb⟩
 103   refine ⟨z ≫ P.j3, ?_, ?_⟩
 104   · change b ≫ (T.g ▷ U.Y) = (z ≫ P.j3) ≫ (T.Z ◁ U.f)
 105     rw [← hzb, Category.assoc, P.other_square, Category.assoc]
 106   · have hs : a ≫ (U.map (tensorLeft T.X)).h =
 107         -((z ≫ P.j3) ≫ (T.map (tensorRight U.X)).h) := by
 108       rw [← hza, Category.assoc, P.boundary]
 109       simp only [Preadditive.comp_neg, Category.assoc]
 110     simp only [connectingHomomorphism, AddMonoidHom.coe_mk, ZeroHom.coe_mk,
 111       hs, Functor.map_neg, Preadditive.comp_neg, Preadditive.neg_comp]
 112     rfl
 113 
 114 set_option backward.isDefEq.respectTransparency false in
 115 /-- Sign removal requires exponent two in the actual projected group.
 116 Mod-two Adams coordinates alone do not imply this hypothesis. -/
 117 private theorem may_boundary_projected_of_exponent_two (B : MayContext Syn)
 118     (hB : MaySourceResults Syn B) (T U : HoCofiberSequence (C := Syn))
 119     (n : ℤ) (a : HomotopyGroup n (T.X ⊗ U.Z))
 120     (b : HomotopyGroup n (T.Y ⊗ U.Y))
 121     (hab : inducedMap (T.f ▷ U.Z) n a = inducedMap (T.Y ◁ U.g) n b)
 122     {G : Type w} [AddCommGroup G]
 123     (q : HomotopyGroup (n-1) (T.X ⊗ U.X) →+ G)
 124     (htwo : ∀ x, q x + q x = 0) :
 125     letI := B.leftShift; letI := B.rightShift
 126     letI := B.leftExact; letI := B.rightExact
 127     ∃ c : HomotopyGroup n (T.Z ⊗ U.X),
 128       inducedMap (T.g ▷ U.Y) n b = inducedMap (T.Z ◁ U.f) n c ∧
 129       q (connectingHomomorphism (U.map (tensorLeft T.X)) n a) =
 130         q (connectingHomomorphism (T.map (tensorRight U.X)) n c) := by
 131   letI := B.leftShift
 132   letI := B.rightShift
 133   letI := B.leftExact
 134   letI := B.rightExact
 135   rcases may_signed_boundary_of_source B hB T U n a b hab with ⟨c, hc, hboundary⟩
 136   refine ⟨c, hc, ?_⟩
 137   rw [hboundary, map_neg]
 138   exact neg_eq_of_add_eq_zero_right (htwo _)
