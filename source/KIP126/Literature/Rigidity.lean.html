  24 public def classicalDifferential (X : ClassicalObject) (r s t : ℤ) :
  25     (D.classical.obj (X.obj D.auxiliary)).Page r (s,t) ⟶
  26       (D.classical.obj (X.obj D.auxiliary)).Page r (s+r,t+r-1) :=
  27   (D.classical.obj (X.obj D.auxiliary)).d r (s,t) ≫ eqToHom
  28     (congrArg ((D.classical.obj (X.obj D.auxiliary)).Page r) (by
  29     rw [D.classical.differentialDegree]

  39 /-- Suspension coordinates for the actual ν-family; λ below is the actual
  40 functorial map of the already supplied deformation transformation. -/
  41 public structure NuPageCoordinates where
  42   weightShift : ∀ (X : ClassicalObject) (r s t w : ℤ),
  43     (D.family.nu D.nu (X.obj D.auxiliary)).Page r (s,t,w) ≅
  44       (D.family.obj ((SyntheticCategory.biShift (0,-1)).obj
  45         (D.nu.functor.obj (X.obj D.auxiliary)))).Page r (s,t,w-1)

  48     HEq ((weightShift X 2 s t (t-k)).hom (targetNuLabel D X s t k x))
  49       (D.comparison.nuE2 X (-1) s t k x)

  59 /-- The free component identifies directly with the actual classical page.
  60 This eliminates an independently selected generator object. -/
  61 public structure FreeLambdaPages (P : NuPageCoordinates D) where
  62   component : ∀ (X : ClassicalObject) (r : ℤ) (_hr : 2 ≤ r)
  63     (s t w : ℤ) (_h : r-2 ≤ t-w),
  64     (D.family.nu D.nu (X.obj D.auxiliary)).Page r (s,t,w) ≅
  65       (D.classical.obj (X.obj D.auxiliary)).Page r (s,t)
  66   e2 : ∀ (X : ClassicalObject) (s t w : ℤ) (hw : w ≤ t)
  67       (x : (D.classical.obj (X.obj D.auxiliary)).Page 2 (s,t)),
  68     (component X 2 (by omega) s t w (by omega)).hom (nuWeightLabel D X s t w hw x) = x
  69   lambda : ∀ (X : ClassicalObject) (r : ℤ) (hr : 2 ≤ r)
  70       (s t w : ℤ) (h : r-2 ≤ t-w),
  71     P.lambda D X r s t w ≫ (component X r hr s t (w-1) (by omega)).hom =
  72       (component X r hr s t w h).hom
  73   differential : ∀ (X : ClassicalObject) (r : ℤ) (hr : 2 ≤ r)
  74       (s t w : ℤ) (h : r-2 ≤ t-w),
  75     syntheticDifferential D X r s t w ≫
  76         (component X r hr (s+r) (t+r-1) w (by omega)).hom =
  77       (component X r hr s t w h).hom ≫ classicalDifferential D X r s t

  79 /-- A specified classical page class has a canonical lift in the free range. -/
  80 public def FreeLambdaPages.classicalLift {P : NuPageCoordinates D} (F : FreeLambdaPages D P)
  81     (X : ClassicalObject) (r : ℤ) (hr : 2 ≤ r) (s t w : ℤ) (h : r-2 ≤ t-w)
  82     (x : (D.classical.obj (X.obj D.auxiliary)).Page r (s,t)) :
  83     (D.family.nu D.nu (X.obj D.auxiliary)).Page r (s,t,w) :=
  84   (F.component X r hr s t w h).inv x

  86 /-- Differential translation follows from the supplied comparison equation. -/
  87 public theorem FreeLambdaPages.classicalLift_differential {P : NuPageCoordinates D}
  88     (F : FreeLambdaPages D P) (X : ClassicalObject) (r : ℤ) (hr : 2 ≤ r)
  89     (s t w : ℤ) (h : r-2 ≤ t-w)
  90     (x : (D.classical.obj (X.obj D.auxiliary)).Page r (s,t)) :
  91     syntheticDifferential D X r s t w (F.classicalLift D X r hr s t w h x) =
  92       F.classicalLift D X r hr (s+r) (t+r-1) w (by omega)
  93         (classicalDifferential D X r s t x) := by
  94   have eqn := congrArg (fun f => f ((F.component X r hr s t w h).inv x))
  95     (F.differential X r hr s t w h)
  96   have lifted := congrArg (fun z => (F.component X r hr (s+r) (t+r-1) w (by omega)).inv z) eqn
  97   simpa [classicalLift] using lifted

  99 public theorem FreeLambdaPages.classicalLift_lambda {P : NuPageCoordinates D}
 100     (F : FreeLambdaPages D P) (X : ClassicalObject) (r : ℤ) (hr : 2 ≤ r)
 101     (s t w : ℤ) (h : r-2 ≤ t-w)
 102     (x : (D.classical.obj (X.obj D.auxiliary)).Page r (s,t)) :
 103     P.lambda D X r s t w (F.classicalLift D X r hr s t w h x) =
 104       F.classicalLift D X r hr s t (w-1) (by omega) x := by
 105   have eqn := congrArg (fun f => f ((F.component X r hr s t w h).inv x))
 106     (F.lambda X r hr s t w h)
 107   have lifted := congrArg (fun z => (F.component X r hr s t (w-1) (by omega)).inv z) eqn
 108   simpa [classicalLift] using lifted

 110 /-- The map induced by a nested-data comparison, using the fixed first page. -/
 111 public def comparisonPageMap (X : ClassicalObject)
 112     (F : SSDataMorphism Tridegree
 113       (D.family.nu D.nu (X.obj D.auxiliary)).sequence.ssData
 114       (fun i => (D.classical.obj (X.obj D.auxiliary)).ssData (i.1,i.2.1)))
 115     (r s t w : ℤ) :
 116     (D.family.nu D.nu (X.obj D.auxiliary)).Page r (s,t,w) ⟶
 117       (D.classical.obj (X.obj D.auxiliary)).Page r (s,t) :=
 118   eqToHom (by simp [SyntheticAdamsSS.Page, Page, SyntheticAdamsFamily.nu,
 119     SyntheticAdamsFamily.obj, D.family.firstPage]) ≫
 120     F.pageMap (s,t,w) (↑(r-2).toNat) ≫
 121     eqToHom (by simp [Page, D.classical.firstPage])

 123 /-- A full filtered-data comparison. Preservation of every Z and B gives
 124 common-representative compatibility at all pages, not just unrelated maps. -/
 125 public structure LambdaInversionComparison (P : NuPageCoordinates D) where
 126   data : ∀ X : ClassicalObject,
 127     SSDataMorphism Tridegree
 128       (D.family.nu D.nu (X.obj D.auxiliary)).sequence.ssData
 129       (fun i => (D.classical.obj (X.obj D.auxiliary)).ssData (i.1,i.2.1))
 130   e2 : ∀ (X : ClassicalObject) (s t w : ℤ) (hw : w ≤ t)
 131       (x : (D.classical.obj (X.obj D.auxiliary)).Page 2 (s,t)),
 132     comparisonPageMap D X (data X) 2 s t w (nuWeightLabel D X s t w hw x) = x
 133   differential : ∀ (X : ClassicalObject) (r : ℤ) (hr : 2 ≤ r) (s t w : ℤ),
 134     let map := fun s t => comparisonPageMap D X (data X) r s t w
 135     syntheticDifferential D X r s t w ≫ map (s+r) (t+r-1) =
 136       map s t ≫ classicalDifferential D X r s t
 137   lambda : ∀ (X : ClassicalObject) (r : ℤ) (hr : 2 ≤ r) (s t w : ℤ),
 138     let map := fun w => comparisonPageMap D X (data X) r s t w
 139     P.lambda D X r s t w ≫ map (w-1) = map w

 141 /-- All truncated E₂ components, including the zero regions. -/
 142 public def quotientE2Component (X : ClassicalObject) (n : ℕ) (s t w : ℤ) : ModuleCat.{v} ℤ :=
 143   if 0 ≤ t-w ∧ t-w < n then (D.classical.obj (X.obj D.auxiliary)).Page 2 (s,t)

 146 /-- The actual quotient map is an isomorphism only in the retained strip
 147 or the stated later-page safe range. -/
 148 public structure NuModLambdaPageData (X : ClassicalObject) (n : ℕ) where
 149   component : ∀ s t w : ℤ,
 150     (D.family.quotient (D.nu.functor.obj (X.obj D.auxiliary)) n).Page 2 (s,t,w) ≅
 151       quotientE2Component D X n s t w
 152   component_retained : ∀ (s t w : ℤ) (h : 0 ≤ t-w ∧ t-w < n)
 153       (x : (D.classical.obj (X.obj D.auxiliary)).Page 2 (s,t)),
 154     HEq ((component s t w).hom
 155       ((D.family.quotientProjection (D.nu.functor.obj (X.obj D.auxiliary)) n).pageMap
 156         2 (s,t,w) (nuWeightLabel D X s t w (by omega) x))) x
 157   retained : ∀ (s t w : ℤ), 0 ≤ t-w ∧ t-w < n →
 158     IsIso ((D.family.quotientProjection (D.nu.functor.obj (X.obj D.auxiliary)) n).pageMap 2 (s,t,w))
 159   safe : ∀ (r s t w : ℤ), 2 ≤ r → 0 ≤ t-w → t-w+r-2 < n →
 160     IsIso ((D.family.quotientProjection (D.nu.functor.obj (X.obj D.auxiliary)) n).pageMap r (s,t,w))
