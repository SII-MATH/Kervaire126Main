  24 /-- The first two arrows are precisely the ν images of this triangle's
  25 arrows; only the synthetic connecting map is existential. -/
  26 public def NuImageIsCofiber (T : Triangle C) : Prop :=
  27   ∃ δ : D.nu.functor.obj T.obj₃ ⟶ (D.nu.functor.obj T.obj₁)⟦(1 : ℤ)⟧,
  28     Triangle.mk (D.nu.functor.map T.mor₁) (D.nu.functor.map T.mor₂) δ ∈ distTriang Syn
  29 
  30 /-- Pstrągowski's full bidirectional criterion, without added finiteness,
  31 connectivity or completeness assumptions. -/
  32 public def NuCofiberCriterion : Prop :=
  33   ∀ T : Triangle C, T ∈ distTriang C →
  34     (NuImageIsCofiber D T ↔ HomologyShortExact D.coefficients T)
  35 
  36 /-- The forward preservation input is obtained from the same full source
  37 criterion, with exactly the shared short-exactness predicate. -/
  38 public def NuCofiberCriterion.toNuExactTriangleInput (h : NuCofiberCriterion D) :
  39     NuExactTriangleInput D.coefficients D.nu where
  40   preserves T hT hH := (h T hT).mpr hH

  42 /-- An actual λᵏ factorization of νf in the negative-target convention. -/
  43 public structure SyntheticFullLift {X Y : C} (f : X ⟶ Y) (k : ℕ) where
  44   map : D.nu.functor.obj X ⟶
  45     (SyntheticCategory.biShift (0,-(k : ℤ))).obj (D.nu.functor.obj Y)
  46   factorization : map ≫ lambdaPow k (D.nu.functor.obj Y) = D.nu.functor.map f
  47 
  48 /-- BHS full lift for every pair of classical objects and every filtration
  49 bound. The premise is factorization through D's actual target tower. -/
  50 public def SyntheticLiftComparison : Prop :=
  51   ∀ {X Y : C} (f : X ⟶ Y) (k : ℕ),
  52     MapFiltrationAtLeast D.classicalTower f k → Nonempty (SyntheticFullLift D f k)
  53 
  54 /-- One triangle lift binds factorization and distinguishedness to the same
  55 connecting arrow, using the given ν suspension comparison. -/
  56 public structure SyntheticTriangleLift (T : Triangle C) where
  57   connecting : D.nu.functor.obj T.obj₃ ⟶
  58     (SyntheticCategory.biShift (0,-1)).obj (D.nu.functor.obj (T.obj₁⟦(1 : ℤ)⟧))
  59   factorization : connecting ≫ SyntheticCategory.lam.app
  60     (D.nu.functor.obj (T.obj₁⟦(1 : ℤ)⟧)) = D.nu.functor.map T.mor₃
  61   distinguished : Triangle.mk (D.nu.functor.map T.mor₁) (D.nu.functor.map T.mor₂)
  62     (connecting ≫ (D.nu.boundaryLandingIso T.obj₁).hom) ∈ distTriang Syn
  63 
  64 /-- Actual λ torsion in a Hom group, not vanishing of one page coordinate. -/
  65 public def LambdaTorsion {X Y : Syn} (f : X ⟶ Y) : Prop :=
  66   ∃ n : ℕ, lambdaPow n X ≫ f = 0
  67 
  68 /-- Reduce a full lift to one λ division. This operation does not assert
  69 that the result is a distinguished triangle's connecting arrow. -/
  70 public def SyntheticFullLift.normalizedMap {X Y : C} {f : X ⟶ Y} {k : ℕ}
  71     (L : SyntheticFullLift D f k) (hk : 1 ≤ k) :
  72     D.nu.functor.obj X ⟶ (SyntheticCategory.biShift (0,-1)).obj (D.nu.functor.obj Y) := by
  73   have hdegree : ((0 : ℤ), -(k : ℤ)) =
  74       ((0 : ℤ), -1) + ((0 : ℤ), -((k-1 : ℕ) : ℤ)) := by
  75     ext <;> simp
  76     omega
  77   exact L.map ≫ eqToHom (congrArg
  78     (fun p : ℤ × ℤ => (SyntheticCategory.biShift p).obj (D.nu.functor.obj Y)) hdegree) ≫
  79     (SyntheticCategory.biShift_comp (0,-1) (0,-((k-1 : ℕ) : ℤ))).inv.app
  80       (D.nu.functor.obj Y) ≫
  81     lambdaPow (k-1) ((SyntheticCategory.biShift (0,-1)).obj (D.nu.functor.obj Y))
  82 
  83 /-- Arbitrary full lifts agree with the selected triangle component only
  84 modulo λ torsion. No strict equality or universal distinguishedness is added. -/
  85 public def SyntheticTriangleLift.FullLiftComparison {T : Triangle C}
  86     (L : SyntheticTriangleLift D T) : Prop :=
  87   ∀ (k : ℕ) (hk : 1 ≤ k) (F : SyntheticFullLift D T.mor₃ k),
  88     LambdaTorsion (L.connecting - F.normalizedMap D hk)
  89 
  90 /-- BHS triangle lifting, retaining positive filtration and all short-exact
  91 conditions. All objects range over the full classical category. -/
  92 public def SyntheticTriangleLiftComparison : Prop :=
  93   ∀ T : Triangle C, T ∈ distTriang C →
  94     MapFiltrationAtLeast D.classicalTower T.mor₃ 1 → HomologyShortExact D.coefficients T →
  95       ∃ L : SyntheticTriangleLift D T, L.FullLiftComparison D
