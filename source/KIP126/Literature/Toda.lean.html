  24 /-- A shifted Toda representative obtained from a distinguished cofiber
  25 triangle for `f`. The extension `gbar` of `g` composes with `h` through the
  26 triangle's connecting map to give `x`. -/
  27 public def Relation {X Y Z W : C} (x : X⟦(1 : ℤ)⟧ ⟶ W)
  28     (f : X ⟶ Y) (g : Y ⟶ Z) (h : Z ⟶ W) : Prop :=
  29   ∃ (Q : C) (i : Y ⟶ Q) (p : Q ⟶ X⟦(1 : ℤ)⟧),
  30     Triangle.mk f i p ∈ distTriang C ∧
  31       ∃ gbar : Q ⟶ Z, i ≫ gbar = g ∧ p ≫ x = gbar ≫ h
  32 
  33 end KIP126.StableHomotopy.Toda

  45 /-- The actual suspension comparison used for a synthetic triple bracket. -/
  46 public def tripleTodaSource (a aw b bw c cw : ℤ) :
  47     Smn (Syn := Syn) (a + (b + c) + 1) (aw + (bw + cw)) ≅
  48       ((SyntheticCategory.biShift (b+c,bw+cw)).obj (Smn a aw))⟦(1 : ℤ)⟧ := by
  49   simpa only [Prod.mk_add_mk, add_zero, Smn, Functor.comp_obj] using
  50     ((SyntheticCategory.biShift_comp (a+(b+c),aw+(bw+cw)) (1,0)).app
  51       (S_0_0 : Syn)).symm ≪≫
  52     (SyntheticCategory.biShift (1,0)).mapIso
  53       ((SyntheticCategory.biShift_comp (a,aw) (b+c,bw+cw)).app S_0_0).symm ≪≫
  54     (SyntheticCategory.biShift_compat 1).app _

  56 /-- Membership in the actual triangulated Toda bracket, including its
  57 full indeterminacy; this is not a freely chosen ternary relation. -/
  58 public def TripleToda {a aw b bw c cw : ℤ}
  59     (x : BiHom a aw (S_0_0 : Syn)) (y : BiHom b bw (S_0_0 : Syn))
  60     (z : BiHom c cw (S_0_0 : Syn))
  61     (value : BiHom (a+(b+c)+1) (aw+(bw+cw)) (S_0_0 : Syn)) : Prop :=
  62   KIP126.StableHomotopy.Toda.Relation
  63     ((tripleTodaSource a aw b bw c cw).inv ≫ value)
  64     ((SyntheticCategory.biShift (b+c,bw+cw)).map x)
  65     ((SyntheticCategory.biShift_comp (b,bw) (c,cw)).inv.app S_0_0 ≫
  66       (SyntheticCategory.biShift (c,cw)).map y) z

 114 /-- A Toda relation requires the two adjacent composites to vanish. -/
 115 public theorem composable {X Y Z W : C} {x : X⟦(1 : ℤ)⟧ ⟶ W}
 116     {f : X ⟶ Y} {g : Y ⟶ Z} {h : Z ⟶ W}
 117     (hx : Relation x f g h) : f ≫ g = 0 ∧ g ≫ h = 0 := by
 118   rcases hx with ⟨Q, i, p, hT, gbar, hg, hx⟩
 119   let T := Triangle.mk f i p
 120   have hfi : f ≫ i = 0 := by
 121     have h := comp_distTriang_mor_zero₁₂ T hT
 122     dsimp only [T, Triangle.mk] at h
 123     exact h
 124   have hip : i ≫ p = 0 := by
 125     have h := comp_distTriang_mor_zero₂₃ T hT
 126     dsimp only [T, Triangle.mk] at h
 127     exact h
 128   constructor
 129   · calc
 130       f ≫ g = (f ≫ i) ≫ gbar := by rw [← hg, Category.assoc]
 131       _ = 0 := by rw [hfi, zero_comp]
 132   · calc
 133       g ≫ h = i ≫ (gbar ≫ h) := by rw [← hg, Category.assoc]
 134       _ = (i ≫ p) ≫ x := by rw [← hx, Category.assoc]
 135       _ = 0 := by rw [hip, zero_comp]
 136 
 137 /-- Every pair of vanishing adjacent composites admits a Toda representative. -/
 138 public theorem exists_relation {X Y Z W : C} (f : X ⟶ Y) (g : Y ⟶ Z)
 139     (h : Z ⟶ W) (hfg : f ≫ g = 0) (hgh : g ≫ h = 0) :
 140     ∃ x : X⟦(1 : ℤ)⟧ ⟶ W, Relation x f g h := by
 141   obtain ⟨Q, i, p, hT⟩ := distinguished_cocone_triangle f
 142   let T := Triangle.mk f i p
 143   obtain ⟨gbar, hgbar⟩ := T.yoneda_exact₂ hT g hfg
 144   dsimp only [T, Triangle.mk] at hgbar
 145   have hzero : i ≫ (gbar ≫ h) = 0 := by
 146     calc
 147       i ≫ (gbar ≫ h) = (i ≫ gbar) ≫ h := by simp only [Category.assoc]
 148       _ = g ≫ h := by rw [hgbar]
 149       _ = 0 := hgh
 150   obtain ⟨x, hx⟩ := T.yoneda_exact₃ hT (gbar ≫ h) hzero
 151   exact ⟨x, Q, i, p, hT, gbar, hgbar.symm, hx.symm⟩
 152 
 153 /-- Adding a left-indeterminacy term preserves the Toda relation. -/
 154 private theorem indeterminacy_left {X Y Z W : C} {x : X⟦(1 : ℤ)⟧ ⟶ W}
 155     {f : X ⟶ Y} {g : Y ⟶ Z} {h : Z ⟶ W}
 156     (hx : Relation x f g h) (y : Y⟦(1 : ℤ)⟧ ⟶ W) :
 157     Relation (x + f⟦(1 : ℤ)⟧' ≫ y) f g h := by
 158   rcases hx with ⟨Q, i, p, hT, gbar, hg, hx⟩
 159   refine ⟨Q, i, p, hT, gbar, hg, ?_⟩
 160   have hpf : p ≫ f⟦(1 : ℤ)⟧' = 0 :=
 161     comp_distTriang_mor_zero₃₁ (Triangle.mk f i p) hT
 162   calc
 163     p ≫ (x + f⟦(1 : ℤ)⟧' ≫ y) = p ≫ x + p ≫ (f⟦(1 : ℤ)⟧' ≫ y) := by
 164       simp only [Preadditive.comp_add]
 165     _ = p ≫ x + (p ≫ f⟦(1 : ℤ)⟧') ≫ y := by rw [Category.assoc]
 166     _ = p ≫ x := by rw [hpf, zero_comp, add_zero]
 167     _ = gbar ≫ h := hx
 168 
 169 /-- Adding a right-indeterminacy term preserves the Toda relation. -/
 170 private theorem indeterminacy_right {X Y Z W : C} {x : X⟦(1 : ℤ)⟧ ⟶ W}
 171     {f : X ⟶ Y} {g : Y ⟶ Z} {h : Z ⟶ W}
 172     (hx : Relation x f g h) (z : X⟦(1 : ℤ)⟧ ⟶ Z) :
 173     Relation (x + z ≫ h) f g h := by
 174   rcases hx with ⟨Q, i, p, hT, gbar, hg, hx⟩
 175   refine ⟨Q, i, p, hT, gbar + p ≫ z, ?_, ?_⟩
 176   · have hip : i ≫ p = 0 :=
 177       comp_distTriang_mor_zero₂₃ (Triangle.mk f i p) hT
 178     calc
 179       i ≫ (gbar + p ≫ z) = i ≫ gbar + i ≫ (p ≫ z) := by
 180         simp only [Preadditive.comp_add]
 181       _ = g + (i ≫ p) ≫ z := by rw [hg, Category.assoc]
 182       _ = g := by rw [hip, zero_comp, add_zero]
 183   · calc
 184       p ≫ (x + z ≫ h) = p ≫ x + p ≫ (z ≫ h) := by
 185         simp only [Preadditive.comp_add]
 186       _ = gbar ≫ h + (p ≫ z) ≫ h := by rw [hx, Category.assoc]
 187       _ = (gbar + p ≫ z) ≫ h := by simp only [Preadditive.add_comp]

 203 /-- Every difference of representatives is a sum of the two actual indeterminacy terms. -/
 204 private theorem indeterminacy_complete {X Y Z W : C}
 205     {x₁ x₂ : X⟦(1 : ℤ)⟧ ⟶ W} {f : X ⟶ Y} {g : Y ⟶ Z} {h : Z ⟶ W}
 206     (hx₁ : Relation x₁ f g h) (hx₂ : Relation x₂ f g h) :
 207     ∃ (y : Y⟦(1 : ℤ)⟧ ⟶ W) (z : X⟦(1 : ℤ)⟧ ⟶ Z),
 208       x₁ - x₂ = f⟦(1 : ℤ)⟧' ≫ y + z ≫ h := by
 209   rcases hx₁ with ⟨Q₁, i₁, p₁, hT₁, g₁, hg₁, hx₁⟩
 210   rcases hx₂ with ⟨Q₂, i₂, p₂, hT₂, g₂, hg₂, hx₂⟩
 211   obtain ⟨c, hc₁, hc₂⟩ := complete_distinguished_triangle_morphism
 212     (Triangle.mk f i₁ p₁) (Triangle.mk f i₂ p₂) hT₁ hT₂
 213     (𝟙 X) (𝟙 Y) (by
 214       change f ≫ 𝟙 Y = 𝟙 X ≫ f
 215       simp only [Category.comp_id, Category.id_comp])
 216   dsimp only [Triangle.mk] at c hc₁ hc₂
 217   have hi : i₁ ≫ c = i₂ := by simpa using hc₁
 218   have hp : p₁ = c ≫ p₂ := by simpa using hc₂
 219   let δ : Q₁ ⟶ Z := g₁ - c ≫ g₂
 220   have hδ : i₁ ≫ δ = 0 := by
 221     dsimp [δ]
 222     rw [Preadditive.comp_sub, hg₁, ← Category.assoc, hi, hg₂, sub_self]
 223   obtain ⟨z, hz⟩ := (Triangle.mk f i₁ p₁).yoneda_exact₃ hT₁ δ hδ
 224   dsimp only [Triangle.mk] at z hz
 225   have hp₂x₂ : p₁ ≫ x₂ = (c ≫ g₂) ≫ h := by
 226     rw [hp, Category.assoc, hx₂, ← Category.assoc]
 227   have hpzh : p₁ ≫ (z ≫ h) = δ ≫ h := by
 228     rw [← Category.assoc, ← hz]
 229   let e : X⟦(1 : ℤ)⟧ ⟶ W := x₁ - x₂ - z ≫ h
 230   have he : p₁ ≫ e = 0 := by
 231     dsimp [e]
 232     rw [Preadditive.comp_sub, Preadditive.comp_sub, hx₁, hp₂x₂, hpzh]
 233     dsimp [δ]
 234     rw [Preadditive.sub_comp]
 235     abel
 236   have hrot : (Triangle.mk f i₁ p₁).rotate ∈ distTriang C :=
 237     rot_of_distTriang (Triangle.mk f i₁ p₁) hT₁
 238   obtain ⟨y₀, hy₀⟩ := (Triangle.mk f i₁ p₁).rotate.yoneda_exact₃ hrot e he
 239   dsimp only [Triangle.mk, Triangle.rotate] at y₀ hy₀
 240   refine ⟨-y₀, z, ?_⟩
 241   have hy₀' : e = f⟦(1 : ℤ)⟧' ≫ (-y₀) := by
 242     simpa [Triangle.rotate] using hy₀
 243   calc
 244     x₁ - x₂ = e + z ≫ h := by dsimp [e]; abel
 245     _ = f⟦(1 : ℤ)⟧' ≫ (-y₀) + z ≫ h := by rw [hy₀']
 246 
 247 /-- With any chosen representative as origin, the Toda bracket is exactly
 248 its coset by the full left and right indeterminacy. No uniqueness or vanishing
 249 of either indeterminacy term is assumed. -/
 250 public theorem relation_iff_indeterminacy {X Y Z W : C}
 251     {x₀ x : X⟦(1 : ℤ)⟧ ⟶ W} {f : X ⟶ Y} {g : Y ⟶ Z} {h : Z ⟶ W}
 252     (hx₀ : Relation x₀ f g h) :
 253     Relation x f g h ↔
 254       ∃ (y : Y⟦(1 : ℤ)⟧ ⟶ W) (z : X⟦(1 : ℤ)⟧ ⟶ Z),
 255         x - x₀ = f⟦(1 : ℤ)⟧' ≫ y + z ≫ h := by
 256   constructor
 257   · exact fun hx => indeterminacy_complete hx hx₀
 258   · rintro ⟨y, z, hx⟩
 259     have heq : x = (x₀ + f⟦(1 : ℤ)⟧' ≫ y) + z ≫ h := by
 260       calc
 261         x = (x - x₀) + x₀ := (sub_add_cancel x x₀).symm
 262         _ = (f⟦(1 : ℤ)⟧' ≫ y + z ≫ h) + x₀ := by rw [hx]
 263         _ = (x₀ + f⟦(1 : ℤ)⟧' ≫ y) + z ≫ h := by abel
 264     rw [heq]
 265     exact indeterminacy_right (indeterminacy_left hx₀ y) z
 266 
 267 end

 283 /-- Toda juggling with its shifted negative sign; this proof requires the octahedral axiom. -/
 284 public theorem juggling [IsTriangulated C]
 285     {X Y Z W V : C} {x : X⟦(1 : ℤ)⟧ ⟶ W}
 286     {f : X ⟶ Y} {g : Y ⟶ Z} {h : Z ⟶ W} {d : W ⟶ V}
 287     (hx : Relation x f g h) (hhd : h ≫ d = 0) :
 288     ∃ y : Y⟦(1 : ℤ)⟧ ⟶ V,
 289       Relation y g h d ∧ x ≫ d = (-f⟦(1 : ℤ)⟧') ≫ y := by
 290   rcases hx with ⟨Q, i, p, hTf, gbar, hg, hx⟩
 291   let Tf := Triangle.mk f i p
 292   have hTfi : Tf.rotate ∈ distTriang C := rot_of_distTriang Tf hTf
 293   obtain ⟨K, v, w, hTbar⟩ := distinguished_cocone_triangle gbar
 294   obtain ⟨Qg, j, q, hTg⟩ := distinguished_cocone_triangle g
 295   let Tbar := Triangle.mk gbar v w
 296   let Tg := Triangle.mk g j q
 297   let O := CategoryTheory.Triangulated.someOctahedron hg hTfi hTbar hTg
 298   dsimp only [Tf, Tbar, Tg, Triangle.mk, Triangle.rotate] at O
 299   have hpm : p ≫ O.m₁ = gbar ≫ j := by
 300     simpa [Tf, Tbar, Tg, O] using O.comm₁
 301   have hmq : O.m₁ ≫ q = -f⟦(1 : ℤ)⟧' := by
 302     exact O.comm₂
 303   have hgh : g ≫ h = 0 := (composable ⟨Q, i, p, hTf, gbar, hg, hx⟩).2
 304   obtain ⟨hbar₀, hhbar₀⟩ := Tg.yoneda_exact₂ hTg h hgh
 305   dsimp only [Tg, Triangle.mk] at hbar₀ hhbar₀
 306   let δ : X⟦(1 : ℤ)⟧ ⟶ W := x - O.m₁ ≫ hbar₀
 307   have hpδ : p ≫ δ = 0 := by
 308     dsimp [δ]
 309     rw [Preadditive.comp_sub, hx, ← Category.assoc, hpm, Category.assoc,
 310       ← hhbar₀, sub_self]
 311   obtain ⟨a, ha⟩ := Tf.rotate.yoneda_exact₃ hTfi δ hpδ
 312   dsimp only [Tf, Triangle.mk, Triangle.rotate] at a ha
 313   let hbar : Qg ⟶ W := hbar₀ + q ≫ a
 314   have hjhbar : j ≫ hbar = h := by
 315     have hjq : j ≫ q = 0 := by
 316       have hzero := comp_distTriang_mor_zero₂₃ Tg hTg
 317       dsimp only [Tg, Triangle.mk] at hzero
 318       exact hzero
 319     dsimp [hbar]
 320     rw [Preadditive.comp_add, ← hhbar₀, ← Category.assoc, hjq, zero_comp, add_zero]
 321   have hmx : O.m₁ ≫ hbar = x := by
 322     dsimp [hbar]
 323     rw [Preadditive.comp_add, ← Category.assoc, hmq]
 324     rw [← ha]
 325     dsimp [δ]
 326     abel
 327   have hjhd : j ≫ (hbar ≫ d) = 0 := by
 328     rw [← Category.assoc, hjhbar, hhd]
 329   obtain ⟨y, hy⟩ := Tg.yoneda_exact₃ hTg (hbar ≫ d) hjhd
 330   dsimp only [Tg, Triangle.mk] at y hy
 331   refine ⟨y, ?_, ?_⟩
 332   · exact ⟨Qg, j, q, hTg, hbar, hjhbar, hy.symm⟩
 333   · calc
 334       x ≫ d = (O.m₁ ≫ hbar) ≫ d := by rw [hmx]
 335       _ = O.m₁ ≫ (hbar ≫ d) := Category.assoc _ _ _
 336       _ = O.m₁ ≫ (q ≫ y) := by rw [← hy]
 337       _ = (O.m₁ ≫ q) ≫ y := by rw [Category.assoc]
 338       _ = (-f⟦(1 : ℤ)⟧') ≫ y := by rw [hmq]
 339

 357 /-- Precomposition gives a containment, with the actual suspended map
 358 on the Toda representative. -/
 359 public theorem precompose {A X Y Z W : C} {x : X⟦(1 : ℤ)⟧ ⟶ W}
 360     {f : X ⟶ Y} {g : Y ⟶ Z} {h : Z ⟶ W}
 361     (hx : Relation x f g h) (a : A ⟶ X) :
 362     Relation (a⟦(1 : ℤ)⟧' ≫ x) (a ≫ f) g h := by
 363   obtain ⟨Q, i, p, hT, gbar, hg, hx⟩ := hx
 364   obtain ⟨Q', i', p', hT'⟩ := distinguished_cocone_triangle (a ≫ f)
 365   obtain ⟨c, hi, hp⟩ := complete_distinguished_triangle_morphism
 366     (Triangle.mk (a ≫ f) i' p') (Triangle.mk f i p) hT' hT a (𝟙 Y) (by
 367       change (a ≫ f) ≫ 𝟙 Y = a ≫ f
 368       exact Category.comp_id _)
 369   dsimp only [Triangle.mk] at hi hp
 370   refine ⟨Q', i', p', hT', c ≫ gbar, ?_, ?_⟩
 371   · rw [← Category.assoc, hi]
 372     simpa using hg
 373   · rw [← Category.assoc, hp, Category.assoc, hx, Category.assoc]

 375 /-- Postcomposition gives a containment for the actual target map. -/
 376 public theorem postcompose {X Y Z W V : C} {x : X⟦(1 : ℤ)⟧ ⟶ W}
 377     {f : X ⟶ Y} {g : Y ⟶ Z} {h : Z ⟶ W}
 378     (hx : Relation x f g h) (d : W ⟶ V) :
 379     Relation (x ≫ d) f g (h ≫ d) := by
 380   obtain ⟨Q, i, p, hT, gbar, hg, hx⟩ := hx
 381   refine ⟨Q, i, p, hT, gbar, hg, ?_⟩
 382   rw [← Category.assoc, hx, Category.assoc]

 384 /-- Moving a factor from the first input into the middle input gives
 385 this containment; its reverse is not asserted. -/
 386 public theorem absorb_first {X Y Z W V : C} {x : X⟦(1 : ℤ)⟧ ⟶ V}
 387     {f : X ⟶ Y} {g : Y ⟶ Z} {h : Z ⟶ W} {d : W ⟶ V}
 388     (hx : Relation x (f ≫ g) h d) :
 389     Relation x f (g ≫ h) d := by
 390   obtain ⟨Q, i, p, hT, hbar, hh, hx⟩ := hx
 391   obtain ⟨Q', i', p', hT'⟩ := distinguished_cocone_triangle f
 392   obtain ⟨c, hi, hp⟩ := complete_distinguished_triangle_morphism
 393     (Triangle.mk f i' p') (Triangle.mk (f ≫ g) i p) hT' hT (𝟙 X) g (by
 394       change f ≫ g = 𝟙 X ≫ (f ≫ g)
 395       exact (Category.id_comp _).symm)
 396   dsimp only [Triangle.mk] at hi hp
 397   have hp' : p' = c ≫ p := by simpa using hp
 398   refine ⟨Q', i', p', hT', c ≫ hbar, ?_, ?_⟩
 399   · rw [← Category.assoc, hi, Category.assoc, hh]
 400   · rw [hp', Category.assoc, hx, Category.assoc]

 413 /-- The four-map shuffle identifies the two sets of products, with the
 414 shifted negative sign. It does not equate arbitrary representatives of
 415 the two Toda brackets. The octahedral hypothesis is the one required
 416 by the underlying juggling law. -/
 417 public theorem shuffle_iff [IsTriangulated C] {X Y Z W V : C}
 418     (f : X ⟶ Y) (g : Y ⟶ Z) (h : Z ⟶ W) (d : W ⟶ V)
 419     (hfg : f ≫ g = 0) (hgh : g ≫ h = 0) (hhd : h ≫ d = 0)
 420     (z : X⟦(1 : ℤ)⟧ ⟶ V) :
 421     (∃ x : X⟦(1 : ℤ)⟧ ⟶ W, Relation x f g h ∧ x ≫ d = z) ↔
 422       ∃ y : Y⟦(1 : ℤ)⟧ ⟶ V,
 423         Relation y g h d ∧ (-f⟦(1 : ℤ)⟧') ≫ y = z := by
 424   constructor
 425   · rintro ⟨x, hx, rfl⟩
 426     obtain ⟨y, hy, he⟩ := juggling hx hhd
 427     exact ⟨y, hy, he.symm⟩
 428   · rintro ⟨y, hy, rfl⟩
 429     obtain ⟨x₀, hx₀⟩ := exists_relation f g h hfg hgh
 430     obtain ⟨y₀, hy₀, hxy₀⟩ := juggling hx₀ hhd
 431     obtain ⟨u, v, huv⟩ := indeterminacy_complete hy hy₀
 432     refine ⟨x₀ + f⟦(1 : ℤ)⟧' ≫ (-v), indeterminacy_left hx₀ (-v), ?_⟩
 433     have hzero : f⟦(1 : ℤ)⟧' ≫ g⟦(1 : ℤ)⟧' = 0 := by
 434       rw [← Functor.map_comp, hfg, Functor.map_zero]
 435     have he := congrArg (fun k => f⟦(1 : ℤ)⟧' ≫ k) huv
 436     simp only [Preadditive.comp_sub, Preadditive.comp_add, ← Category.assoc,
 437       hzero, zero_comp, zero_add] at he
 438     simp only [Preadditive.add_comp, Preadditive.comp_neg, Preadditive.neg_comp]
 439     rw [hxy₀]
 440     simp only [Preadditive.neg_comp]
 441     rw [← he]
 442     abel

 444 private theorem shifted_triangle_distinguished (n : ℤ) {X Y Q : C}
 445     (f : X ⟶ Y) (i : Y ⟶ Q) (p : Q ⟶ X⟦(1 : ℤ)⟧)
 446     (hT : Triangle.mk f i p ∈ distTriang C) :
 447     Triangle.mk (f⟦n⟧') (i⟦n⟧')
 448       (n.negOnePow • p⟦n⟧' ≫ (shiftFunctorComm C (1 : ℤ) n).hom.app X) ∈ distTriang C := by
 449   let e : Y⟦n⟧ ≅ Y⟦n⟧ :=
 450     { hom := n.negOnePow • 𝟙 _
 451       inv := n.negOnePow • 𝟙 _
 452       hom_inv_id := by simp [Linear.comp_units_smul, smul_smul]
 453       inv_hom_id := by simp [Linear.comp_units_smul, smul_smul] }
 454   apply isomorphic_distinguished _ (Triangle.shift_distinguished _ hT n)
 455   refine Triangle.isoMk _ _ (Iso.refl _) e (Iso.refl _) ?_ ?_ ?_
 456   · simp [Triangle.shiftFunctor, Triangle.mk, Category.id_comp, Category.comp_id, e, Linear.comp_units_smul]
 457   · simp [Triangle.shiftFunctor, Triangle.mk, Category.id_comp, Category.comp_id, e, Linear.units_smul_comp, Linear.comp_units_smul, smul_smul]

 470 /-- Suspension transports the entire Toda relation with its forced
 471 `(-1)^n` sign and the actual comparison of the two iterated shifts. -/
 472 public theorem suspension_iff (n : ℤ) {X Y Z W : C}
 473     {x : X⟦(1 : ℤ)⟧ ⟶ W} {f : X ⟶ Y} {g : Y ⟶ Z} {h : Z ⟶ W} :
 474     Relation
 475       (n.negOnePow • ((shiftFunctorComm C (1 : ℤ) n).inv.app X ≫ x⟦n⟧'))
 476       (f⟦n⟧') (g⟦n⟧') (h⟦n⟧') ↔ Relation x f g h := by
 477   constructor
 478   · rintro ⟨Q', i', p', hT', gbar', hg', hx'⟩
 479     obtain ⟨Q, i, p, hT⟩ := distinguished_cocone_triangle f
 480     let δ := n.negOnePow • p⟦n⟧' ≫ (shiftFunctorComm C (1 : ℤ) n).hom.app X
 481     obtain ⟨c, hi, hp⟩ := complete_distinguished_triangle_morphism
 482       (Triangle.mk (f⟦n⟧') (i⟦n⟧') δ) (Triangle.mk (f⟦n⟧') i' p')
 483       (shifted_triangle_distinguished n f i p hT) hT' (𝟙 _) (𝟙 _) (by
 484         change f⟦n⟧' ≫ 𝟙 _ = 𝟙 _ ≫ f⟦n⟧'
 485         simp only [Category.comp_id, Category.id_comp])
 486     dsimp only [Triangle.mk] at hi hp
 487     have hi' : i⟦n⟧' ≫ c = i' := by simpa only [Category.id_comp] using hi
 488     have hp' : δ = c ≫ p' := by
 489       erw [(shiftFunctor C (1 : ℤ)).map_id, Category.comp_id] at hp
 490       exact hp
 491     let gbar := (shiftFunctor C n).preimage (c ≫ gbar')
 492     have hgbar : gbar⟦n⟧' = c ≫ gbar' := Functor.map_preimage _ _
 493     refine ⟨Q, i, p, hT, gbar, ?_, ?_⟩
 494     · apply (shiftFunctor C n).map_injective
 495       rw [Functor.map_comp, hgbar, ← Category.assoc, hi', hg']
 496     · apply (shiftFunctor C n).map_injective
 497       rw [← shifted_connecting_comp n p x]
 498       change δ ≫ _ = _
 499       rw [hp', Category.assoc, hx', Functor.map_comp, hgbar, Category.assoc]
 500   · rintro ⟨Q, i, p, hT, gbar, hg, hx⟩
 501     refine ⟨Q⟦n⟧, i⟦n⟧', n.negOnePow • p⟦n⟧' ≫
 502       (shiftFunctorComm C (1 : ℤ) n).hom.app X,
 503       shifted_triangle_distinguished n f i p hT, gbar⟦n⟧', ?_, ?_⟩
 504     · rw [← Functor.map_comp, hg]
 505     · rw [shifted_connecting_comp, hx, Functor.map_comp]
