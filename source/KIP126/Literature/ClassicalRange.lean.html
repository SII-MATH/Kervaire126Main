  24 public def SphereVanishingLine : Prop :=
  25   ∀ s t : ℤ, 0 < t-s → t-s < 2*s-3 → Subsingleton (D.classical.sphere.Page 2 (s,t))
  26 
  27 public def classicalTwo : HomotopyGroup (C := C) 0 SphereSpectrum :=
  28   (shiftFunctorZero C ℤ).hom.app SphereSpectrum ≫ (2 • 𝟙 _)
  29 
  30 /-- This target is the actual h₀ times the standard square. -/
  31 public def oneLineTarget (j : ℕ) : D.classical.sphere.Page 2
  32     (3, 1+((2^(j-1+1) : ℕ) : ℤ)) :=
  33   D.sphereAlgebra.product (1,1) (2,((2^(j-1+1) : ℕ) : ℤ))
  34     (D.standardClasses.standardHi 0) (D.standardClasses.standardHiSquare (j-1))

  36 private structure ClassicalRangeResults : Prop where
  37   sphereVanishing : SphereVanishingLine D
  38   two_detection : ClassicalDetects (D.classicalConvergence .sphere) (1,1)
  39     (D.standardClasses.standardHi 0) (classicalTwo (C := C))
  40   adamsOneLine_d2 : ∀ (j : ℕ), 4 ≤ j →

  42       (1,((2^j : ℕ) : ℤ)) (3,1+((2^(j-1+1) : ℕ) : ℤ))
  43       (D.standardClasses.standardHi j) (oneLineTarget D j)

  45 public def classicalSphereProduct {m n : ℤ}
  46     (x : HomotopyGroup (C := C) m SphereSpectrum)
  47     (y : HomotopyGroup (C := C) n SphereSpectrum) :
  48     HomotopyGroup (C := C) (m+n) SphereSpectrum :=
  49   (shiftFunctorAdd C m n).hom.app SphereSpectrum ≫ (shiftFunctor C n).map x ≫ y

  69   connective : ∀ n : ℤ, n < 0 → Subsingleton (HomotopyGroup n B.source)

  74   delta_unique : ∀ x : D.classical.sphere.Page 2 (9,54), x ≠ 0 → x = B.deltaH1g

  76   low_filtration63 : ∀ s : ℤ, s ≤ 0 →
  77     Subsingleton ((D.classical.obj B.source).Page 2 (s,63+s))
