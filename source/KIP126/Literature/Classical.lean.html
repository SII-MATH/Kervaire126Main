  25 /-- Connectivity uses the actual sphere shifts of this category. -/
  26 public def IsConnected (X : C) (n : ℤ) : Prop :=
  27   ∀ i : ℤ, i ≤ n → Subsingleton (HomotopyGroup i X)
  28 
  29 /-- Cohomology has the legacy sign convention: maps from `ΣⁿX` to HF₂. -/
  30 public abbrev ModTwoCohomology (n : ℤ) (X : C) :=
  31   (shiftFunctor C n).obj X ⟶ D.coefficients.object

  33 /-- The ordinary stable-connectivity and HF₂ universal-coefficient results.
  34 Their applicability to this chosen background is a producer obligation. -/
  35 private structure ClassicalFoundationResults where
  36   sphere_connected : IsConnected (SphereSpectrum (C := C)) (-1)
  37   connected_homology_vanishing : ∀ (X : C) (n : ℤ), IsConnected X n →
  38     ∀ i : ℤ, i ≤ n → Subsingleton (ModTwoHomology D.coefficients i X)
  39   universal_coefficient : ∀ (n : ℤ) (X : C),
  40     ModTwoCohomology D n X ≃+ (ModTwoHomology D.coefficients (-n) X →+ ZMod 2)

  64 /-- Realization coordinates use the existing λ-localization, unit and λ powers. -/
  65 public structure RealizationCoordinates where
  66   additive : D.recovery.realization.Additive
  67   sphere : ∀ m w : ℤ, Sphere (C := C) m ≅ D.recovery.realization.obj (Smn (Syn := Syn) m w)
  68   unit : (sphere 0 0).hom ≫ D.recovery.realization.map
  69       (SyntheticCategory.biShift_zero.hom.app (S_0_0 : Syn)) =
  70     (shiftFunctorZero C ℤ).hom.app SphereSpectrum ≫
  71       D.recovery.nuRealizationIso.inv.app SphereSpectrum ≫
  72       D.recovery.realization.map D.nu.unitIso.hom
  73   lambda : ∀ (m w : ℤ) (k : ℕ),
  74     (sphere m (w-k)).hom ≫
  75       D.recovery.realization.map (lambdaMultiply k (𝟙 (Smn (Syn := Syn) m w))) =
  76         (sphere m w).hom

  78 public def RealizationCoordinates.sphereMap (R : RealizationCoordinates D) (m w : ℤ) :
  79     BiHom m w (S_0_0 : Syn) →+ HomotopyGroup (C := C) m SphereSpectrum := by
  80   letI := R.additive
  81   exact
  82     { toFun := fun a => (R.sphere m w).hom ≫ D.recovery.realization.map a ≫
  83         D.recovery.realization.map D.nu.unitIso.inv ≫
  84         D.recovery.nuRealizationIso.hom.app SphereSpectrum
  85       map_zero' := by simp
  86       map_add' := by intros; simp [Preadditive.comp_add, Preadditive.add_comp] }

 118 /-- Xu Corollary 1.3, IWX's 62-stem, and standard Hopf detections.
 119 The stronger arbitrary-choice synthetic consequences are derived separately. -/
 120 private structure ClassicalResults (B : ClassicalBindings D) : Prop where
 121   h5Square_permanent : NonzeroSurvival D.classical.sphere (2,64)
 122     (D.standardClasses.standardHiSquare 5)
 123   theta5_detection : ClassicalTheta5Detected D B.theta5
 124   stem62_exponent_two : ∀ a : HomotopyGroup (C := C) 62 SphereSpectrum, a+a=0
 125   theta5_filtration_gap : ∀ a b, ClassicalTheta5Detected D a → ClassicalTheta5Detected D b →
 126     a-b ∈ (D.classicalTower.tower SphereSpectrum).filtrationSubmodule 6 62
 127   eta_detection : ClassicalDetects (D.classicalConvergence .sphere) (1,2)
 128     (D.standardClasses.standardHi 1) D.auxiliary.etaMap
 129   nu_detection : ClassicalDetects (D.classicalConvergence .sphere) (1,4)
 130     (D.standardClasses.standardHi 2) D.auxiliary.nuMap
 131   eta_permanent : NonzeroSurvival D.classical.sphere (1,2) (D.standardClasses.standardHi 1)
 132   nu_permanent : NonzeroSurvival D.classical.sphere (1,4) (D.standardClasses.standardHi 2)
 133   eta_label : sphereFirstQuotient D 1 2 (quotientClass 1 B.eta) = D.standardClasses.standardHi 1
 134   eta_realization : B.realization.sphereMap D 1 2 B.eta = D.auxiliary.etaMap
 135   theta5_lift : ∃ a : BiHom 62 64 (S_0_0 : Syn),
 136     Theta5Detected D a ∧ B.realization.sphereMap D 62 64 a = B.theta5

 138 /-- The chosen classical θ₅ has order dividing two by the 62-stem result. -/
 139 private theorem ClassicalResults.theta5_order_two {B : ClassicalBindings D}
 140     (R : ClassicalResults D B) : B.theta5 + B.theta5 = 0 :=
 141   R.stem62_exponent_two B.theta5

 143 /-- The precise additional comparison needed for arbitrary synthetic choices.
 144 These transport obligations are explicit parameters, not silently upgraded
 145 classical source facts or new unconditional Literature results. -/
 146 private structure Theta5Transport (B : ClassicalBindings D) : Prop where
 147   detection : ∀ a, Theta5Detected D a →
 148     ClassicalTheta5Detected D (B.realization.sphereMap D 62 64 a)
 149   doubling_kernel : ∀ a, Theta5Detected D a →
 150     B.realization.sphereMap D 62 64 (a+a) = 0 → a+a=0
 151   filtration_reflection : ∀ a b, Theta5Detected D a → Theta5Detected D b →
 152     B.realization.sphereMap D 62 64 (a-b) ∈
 153       (D.classicalTower.tower SphereSpectrum).filtrationSubmodule 6 62 →
 154     FiltrationAtLeast D.syntheticTower 6 (a-b)
 155 
 156 private theorem theta5_order_two {B : ClassicalBindings D} (R : ClassicalResults D B)
 157     (T : Theta5Transport D B) (a : BiHom 62 64 (S_0_0 : Syn)) (ha : Theta5Detected D a) : a+a=0 :=
 158   T.doubling_kernel a ha (by rw [map_add]; exact R.stem62_exponent_two _)
 159 
 160 private theorem theta5_indeterminacy_AF_ge_6 {B : ClassicalBindings D} (R : ClassicalResults D B)
 161     (T : Theta5Transport D B) (a b : BiHom 62 64 (S_0_0 : Syn))
 162     (ha : Theta5Detected D a) (hb : Theta5Detected D b) : FiltrationAtLeast D.syntheticTower 6 (a-b) :=
 163   T.filtration_reflection a b ha hb (by
 164     rw [map_sub]
 165     exact R.theta5_filtration_gap _ _ (T.detection a ha) (T.detection b hb))
 166 
 167 private theorem theta5_detected_exists {B : ClassicalBindings D} (R : ClassicalResults D B) :
 168     ∃ a : BiHom 62 64 (S_0_0 : Syn), Theta5Detected D a := by
 169   obtain ⟨a, ha, _⟩ := R.theta5_lift
 170   exact ⟨a, ha⟩
