  36 /-- Reindexing by equality, with no independently chosen page map. -/
  37 public def pageCast (X Y : MappingObject scope) {r r' : ℤ} {i i' : ℤ × ℤ}
  38     (hr : r = r') (hi : i = i') (x : (sequence S X Y).Page r i) :
  39     (sequence S X Y).Page r' i' := by
  40   subst r'; subst i'; exact x

  46 public def definingDifferential (X Y : MappingObject scope) (r : ℤ) (i j : ℤ × ℤ)
  47     (y : (sequence S X Y).Page r (i+j-(r,r-1))) :
  48     (sequence S X Y).Page r (i+j) :=
  49   pageCast S X Y rfl (by
  50     change i+j-(r,r-1)+(A.obj (mappingSpectrum B X.val Y.val)).diffDeg r = i+j
  51     rw [A.differentialDegree]

  54 public def precedingDifferential (X Y : MappingObject scope) (r : ℤ) (i j : ℤ × ℤ)
  55     (y : (sequence S X Y).Page (r-1) (i+j-(r-1,r-2))) :
  56     (sequence S X Y).Page (r-1) (i+j) :=
  57   definingDifferential S X Y (r-1) i j
  58     (pageCast S X Y rfl (by congr 2; omega) y)

  60 /-- One common next-cycle representative, in the actual successor quotient. -/

  68 /-- The signed defining value `yc - (-1)^(stem a) az` on E_(r-1). -/
  69 public def definingValue (W X Y Z : MappingObject scope) (r : ℤ) (i j k : ℤ × ℤ)
  70     (a : (sequence S W X).Page (r-1) i) (c : (sequence S Y Z).Page (r-1) k)
  71     (y : (sequence S W Y).Page (r-1) (i+j-(r-1,r-2)))
  72     (z : (sequence S X Z).Page (r-1) (j+k-(r-1,r-2))) :
  73     (sequence S W Z).Page (r-1) (degree r i j k) :=
  74   let left := pageCast S W Z rfl (by unfold degree; abel)
  75     ((M.pairing W Y Z).ssPairing.pair (r-1) (i+j-(r-1,r-2)) k y c)
  76   let right := pageCast S W Z rfl (by unfold degree; abel)
  77     ((M.pairing W X Z).ssPairing.pair (r-1) i (j+k-(r-1,r-2)) a z)
  78   if Even (stem i) then left-right else left+right

  80 /-- A complete preceding-page defining system. This does not substitute an
  81 arbitrary ternary relation, or define the separate E₂ cobar Massey product. -/
  82 public def Relation (r : ℤ) (_hr : 3 ≤ r) {W X Y Z : MappingObject scope}
  83     {i j k : ℤ × ℤ} (x : (sequence S W Z).Page r (degree r i j k))
  84     (a : (sequence S W X).Page r i) (b : (sequence S X Y).Page r j)
  85     (c : (sequence S Y Z).Page r k) : Prop :=
  86   ∃ (a' : (sequence S W X).Page (r-1) i)
  87     (b' : (sequence S X Y).Page (r-1) j) (c' : (sequence S Y Z).Page (r-1) k)
  88     (y : (sequence S W Y).Page (r-1) (i+j-(r-1,r-2)))
  89     (z : (sequence S X Z).Page (r-1) (j+k-(r-1,r-2))),
  90     Passage S W X r i a' a ∧ Passage S X Y r j b' b ∧ Passage S Y Z r k c' c ∧
  91     precedingDifferential S W Y r i j y = (M.pairing W X Y).ssPairing.pair (r-1) i j a' b' ∧
  92     precedingDifferential S X Z r j k z = (M.pairing X Y Z).ssPairing.pair (r-1) j k b' c' ∧
  93     Passage S W Z r (degree r i j k) (definingValue S M W X Y Z r i j k a' c' y z) x

  95 /-- Full additive indeterminacy, with both free summands and their degrees. -/
  96 public def Indeterminacy (r : ℤ) {W X Y Z : MappingObject scope} {i j k : ℤ × ℤ}
  97     (a : (sequence S W X).Page r i) (c : (sequence S Y Z).Page r k)
  98     (z : (sequence S W Z).Page r (degree r i j k)) : Prop :=
  99   ∃ (u : (sequence S W Y).Page r (i+j-(r-1,r-2)))
 100     (v : (sequence S X Z).Page r (j+k-(r-1,r-2))),
 101     z = pageCast S W Z rfl (by unfold degree; abel)
 102       ((M.pairing W Y Z).ssPairing.pair r (i+j-(r-1,r-2)) k u c) +
 103       pageCast S W Z rfl (by unfold degree; abel)
 104       ((M.pairing W X Z).ssPairing.pair r i (j+k-(r-1,r-2)) a v)
 105 end PageMassey

 107 /-- Residual injectivity in the actual mapping Adams tower. An associated
 108 graded convergence isomorphism alone does not imply this condition. -/
 109 public def WeakConvergence (X Y : MappingObject scope) : Prop :=
 110   let T := P.tower (mappingSpectrum B X.val Y.val)
 111   ∀ (s : ℕ) (n : ℤ) (f : HomotopyGroup n (T.stage (s+1))),
 112     (∀ (t : ℕ) (hst : s+1 ≤ t), ∃ g : HomotopyGroup n (T.stage t), g ≫ T.map hst = f) →
 113     f ≫ T.map (Nat.le_succ s) = 0 → f = 0

 115 public def crossingSourceDegree (k : ℤ × ℤ) (q : ℤ) : ℤ × ℤ :=
 116   (q,q+stem k+1)

 118 /-- Moss-direction crossing, distinct from extension-sequence crossings. -/
 119 public def HasMossCrossing (X Y : MappingObject scope) (r : ℤ) (k : ℤ × ℤ) : Prop :=
 120   ∃ m q : ℤ, r < m ∧ 0 ≤ q ∧ q < k.1-(r-1) ∧ k.1 < q+m ∧
 121     (sequence S X Y).d m (crossingSourceDegree k q) ≠ 0

 123 public def NoMossCrossingForProducts (W X Y Z : MappingObject scope)
 124     (r : ℤ) (i j k : ℤ × ℤ) : Prop :=
 125   ¬ HasMossCrossing S W Y r (i+j) ∧ ¬ HasMossCrossing S X Z r (j+k)

 127 /-- A finite and infinite class from one actual Z∞ representative. -/
 128 public def HasEInftyClass (X Y : MappingObject scope) (r : ℤ) (k : ℤ × ℤ)
 129     (x : (sequence S X Y).Page r k) (e : ((sequence S X Y).ssData k).eInfty) : Prop :=
 130   ∃ z : InfiniteCycles (sequence S X Y) k,
 131     infinityToPage (sequence S X Y) r k z = x ∧ ((sequence S X Y).ssData k).pageπ ⊤ z = e

 133 /-- Permanence here allows zero, as in the source Moss statement. -/
 134 public def PermanentCycle (X Y : MappingObject scope) (r : ℤ) (k : ℤ × ℤ)
 135     (x : (sequence S X Y).Page r k) : Prop := ∃ e, HasEInftyClass S X Y r k x e

 137 /-- Detection through the same tower's filtration and convergence. -/
 138 public def DetectsAbutment (X Y : MappingObject scope) (r : ℤ) (k : ℤ × ℤ)
 139     (x : (sequence S X Y).Page r k) (α : abutment (B := B) X Y (stem k)) : Prop :=
 140   ∃ e, HasEInftyClass S X Y r k x e ∧
 141     ∃ a : (Subobject.underlying.obj
 142       ((P.tower (mappingSpectrum B X.val Y.val)).filtration.F k.1 (stem k)) : ModuleCat ℤ),
 143       ((P.tower (mappingSpectrum B X.val Y.val)).filtration.F k.1 (stem k)).arrow a = α ∧
 144       ((S.convergence X.val Y.val X.property Y.property).identification k).hom e =
 145         (P.tower (mappingSpectrum B X.val Y.val)).filtration.toAssociatedGraded k.1 (stem k) a

 147 public def abutmentMap (X Y : MappingObject scope) (k : ℤ × ℤ)
 148     (a : abutment (B := B) X Y (stem k)) : X.val⟦stem k⟧ ⟶ Y.val :=
 149   mappingHomotopyEquiv B (stem k) X.val Y.val a

 151 public def shiftMap (X Y : MappingObject scope) (k : ℤ × ℤ) (n : ℤ)
 152     (a : abutment (B := B) X Y (stem k)) : X.val⟦stem k+n⟧ ⟶ Y.val⟦n⟧ :=
 153   (shiftFunctorAdd C (stem k) n).hom.app X.val ≫
 154     (shiftFunctor C n).map (abutmentMap X Y k a)

 156 /-- The actual stable composite, transported through the fixed adjunction. -/
 157 public def abutmentComposition (X Y Z : MappingObject scope) (i j : ℤ × ℤ)
 158     (a : abutment (B := B) X Y (stem i)) (b : abutment (B := B) Y Z (stem j)) :
 159     abutment (B := B) X Z (stem (i+j)) :=
 160   (mappingHomotopyEquiv B (stem (i+j)) X.val Z.val).symm
 161     (eqToHom (congrArg (fun n : ℤ => X.val⟦n⟧) (by
 162       dsimp [stem]; omega : stem (i+j) = stem i+stem j)) ≫
 163       shiftMap X Y i (stem j) a ≫ abutmentMap Y Z j b)

 165 public def todaMap (r : ℤ) (W Z : MappingObject scope) (i j k : ℤ × ℤ)
 166     (a : abutment (B := B) W Z (stem (PageMassey.degree r i j k))) :
 167     (W.val⟦stem i+(stem j+stem k)⟧)⟦(1 : ℤ)⟧ ⟶ Z.val :=
 168   (shiftFunctorAdd C (stem i+(stem j+stem k)) 1).inv.app W.val ≫
 169     eqToHom (congrArg (fun n : ℤ => W.val⟦n⟧) (by
 170       dsimp [stem, PageMassey.degree]; omega : stem i+(stem j+stem k)+1 =
 171         stem (PageMassey.degree r i j k))) ≫
 172       abutmentMap W Z (PageMassey.degree r i j k) a

 176 /-- The finite-page detection compatibility appearing in the original Moss
 177 hypotheses. All products and targets are already fixed by S and M. -/
 178 public def DetectionCompatible : Prop :=
 179   ∀ (X Y Z : MappingObject scope) (r : ℤ) (_hr : 2 ≤ r) (i j : ℤ × ℤ)
 180     (a : (sequence S X Y).Page r i) (b : (sequence S Y Z).Page r j)
 181     (α : abutment (B := B) X Y (stem i)) (β : abutment (B := B) Y Z (stem j)),
 182     DetectsAbutment S X Y r i a α → DetectsAbutment S Y Z r j b β →
 183     DetectsAbutment S X Z r (i+j) ((M.pairing X Y Z).ssPairing.pair r i j a b)
 184       (abutmentComposition X Y Z i j α β)

 186 /-- General Moss implication. Both product-tower weak-convergence premises,
 187 both noncrossing ranges, actual zero composites and full defining systems
 188 are retained. Only existence of some surviving Massey member is asserted. -/
 189 public def StatementAt (W X Y Z : MappingObject scope) : Prop :=
 190   ∀ (r : ℤ) (hr : 3 ≤ r) (i j k : ℤ × ℤ),
 191     WeakConvergence (P := P) (B := B) W Y → WeakConvergence (P := P) (B := B) X Z →
 192     ∀ (a : (sequence S W X).Page r i) (b : (sequence S X Y).Page r j)
 193       (c : (sequence S Y Z).Page r k)
 194       (α : abutment (B := B) W X (stem i)) (β : abutment (B := B) X Y (stem j))
 195       (γ : abutment (B := B) Y Z (stem k)),
 196     let f := shiftMap W X i (stem j+stem k) α
 197     let g := shiftMap X Y j (stem k) β
 198     let h := abutmentMap Y Z k γ
 199     DetectsAbutment S W X r i a α → DetectsAbutment S X Y r j b β →
 200     DetectsAbutment S Y Z r k c γ →
 201     (M.pairing W X Y).ssPairing.pair r i j a b = 0 →
 202     (M.pairing X Y Z).ssPairing.pair r j k b c = 0 →
 203     f ≫ g = 0 → g ≫ h = 0 → NoMossCrossingForProducts S W X Y Z r i j k →
 204     ∃ (x : (sequence S W Z).Page r (PageMassey.degree r i j k))
 205       (ξ : abutment (B := B) W Z (stem (PageMassey.degree r i j k))),
 206       PageMassey.Relation S M r hr x a b c ∧
 207       PermanentCycle S W Z r (PageMassey.degree r i j k) x ∧
 208       DetectsAbutment S W Z r (PageMassey.degree r i j k) x ξ ∧
 209       KIP126.StableHomotopy.Toda.Relation (todaMap r W Z i j k ξ) f g h

 211 /-- The literature delivery contract for this object scope. Scope identity
 212 (for example finite spectra) and comparison with the source model are producer
 213 obligations. This definition neither supplies an inhabitant nor removes them. -/
 214 public def Statement : Prop :=
 215   DetectionCompatible S M → ∀ W X Y Z, StatementAt S M W X Y Z
 216 
 217 /-- Model-side delivery for Moss. The complete coset law is an obligation
 218 on the defined Massey relation; it does not replace that relation with a
 219 chosen operation or discard either indeterminacy summand. The generic
 220 pairing already contains signed Leibniz, successor compatibility and
 221 associativity, so no duplicate coherence structure is selected here. -/
 222 private structure Context where
 223   composition : MappingComposition S
 224   weak_convergence : ∀ X Y : MappingObject scope, WeakConvergence (P := P) (B := B) X Y
 225   detection : DetectionCompatible S composition
 226   indeterminacy : ∀ (r : ℤ) (hr : 3 ≤ r) (W X Y Z : MappingObject scope) (i j k : ℤ × ℤ)
 227     (a : (sequence S W X).Page r i) (b : (sequence S X Y).Page r j)
 228     (c : (sequence S Y Z).Page r k)
 229     (x₀ x : (sequence S W Z).Page r (PageMassey.degree r i j k)),
 230     PageMassey.Relation S composition r hr x₀ a b c →
 231       (PageMassey.Relation S composition r hr x a b c ↔
 232         PageMassey.Indeterminacy S composition r (j := j) a c (x-x₀))
 233 
 234 /-- The external theorem is separate from the internal mapping context.
 235 Its source-to-model comparison is required of a producer supplying it. -/
 236 private def SourceResults (Q : Context S) : Prop := Statement S Q.composition
 237 
 238 /-- Consume the source theorem on exactly the context's fixed data. -/
 239 private theorem SourceResults.at {Q : Context S} (h : SourceResults S Q)
 240     (W X Y Z : MappingObject scope) : StatementAt S Q.composition W X Y Z :=
 241   h Q.detection W X Y Z

 243 end
