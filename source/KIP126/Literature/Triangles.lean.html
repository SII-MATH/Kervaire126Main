  22 /-- Third-vertex comparison for the actual normalized first arrow. -/
  23 public structure NormalizedCofiberComparison (T : TriangleData D.auxiliary) where
  24   cofiber : HasFunctorialCofiber.cofib (D.normalizedMap T.X T.Y T.f).map ≅
  25     (SyntheticCategory.biShift (0,-(normalizedExponent D.classicalTower T.g : ℤ))).obj

  28 /-- The triangle's arrows are obtained from this comparison, so the cofiber
  29 and connecting squares commute by construction. -/
  30 public def NormalizedCofiberComparison.triangle {T : TriangleData D.auxiliary}
  31     (P : NormalizedCofiberComparison D T) : Triangle Syn :=
  32   Triangle.mk (D.normalizedMap T.X T.Y T.f).map
  33     (HasFunctorialCofiber.cofibι _ ≫ P.cofiber.hom)
  34     (P.cofiber.inv ≫ HasFunctorialCofiber.cofibδ _)

  36 public theorem NormalizedCofiberComparison.distinguished {T : TriangleData D.auxiliary}
  37     (P : NormalizedCofiberComparison D T) : P.triangle D ∈ distTriang Syn := by
  38   refine isomorphic_distinguished _
  39     (HasFunctorialCofiber.cofib_distinguished (D.normalizedMap T.X T.Y T.f).map) _ ?_
  40   exact Triangle.isoMk _ _ (Iso.refl _) (Iso.refl _) P.cofiber.symm
  41     (by simp [triangle, Triangle.mk]) (by simp [triangle, Triangle.mk])
  42     (by simp [triangle, Triangle.mk])

  57 /-- Reexpress a positive-source shift as a negative-target shift, using the
  58 same synthetic shift comparisons. Source: Route/Triangles/Data at bbac1fc05843. -/
  59 public def negativeLift {X Y : Syn} (k : ℕ)
  60     (f : (SyntheticCategory.biShift (0,(k : ℤ))).obj X ⟶ Y) :
  61     X ⟶ (SyntheticCategory.biShift (0,-(k : ℤ))).obj Y :=
  62   SyntheticCategory.biShift_zero.inv.app X ≫
  63     eqToHom (congrArg (fun p => (SyntheticCategory.biShift p).obj X)
  64       (show ((0 : ℤ),(0 : ℤ)) = (0,(k : ℤ))+(0,-(k : ℤ)) from by ext <;> simp)) ≫
  65     (SyntheticCategory.biShift_comp (0,(k : ℤ)) (0,-(k : ℤ))).inv.app X ≫
  66     (SyntheticCategory.biShift (0,-(k : ℤ))).map f

  68 /-- The connecting arrow uses the model's selected top lift, including its
  69 suspension and weight transport. -/
  70 public def normalizedConnecting (T : TriangleData D.auxiliary)
  71     (he : (normalizedExponent D.classicalTower T.f : ℤ) +
  72       normalizedExponent D.classicalTower T.g + normalizedExponent D.classicalTower T.h = 1) :
  73     (SyntheticCategory.biShift (0,-(normalizedExponent D.classicalTower T.g : ℤ))).obj
  74       (D.nu.functor.obj (T.Z.obj D.auxiliary)) ⟶
  75     ((SyntheticCategory.biShift (0,(normalizedExponent D.classicalTower T.f : ℤ))).obj
  76       (D.nu.functor.obj (T.X.obj D.auxiliary)))⟦(1 : ℤ)⟧ :=
  77   let eg : ℤ := normalizedExponent D.classicalTower T.g
  78   let eh : ℤ := normalizedExponent D.classicalTower T.h
  79   let X := D.nu.functor.obj (T.X.obj D.auxiliary)
  80   (SyntheticCategory.biShift (0,-eg)).map
  81       (negativeLift (normalizedExponent D.classicalTower T.h)
  82         (D.normalizedMap T.Z (.shift 1 T.X) T.h).map) ≫
  83     (SyntheticCategory.biShift (0,-eg)).map
  84       ((SyntheticCategory.biShift (0,-eh)).map (D.nu.suspensionIso (T.X.obj D.auxiliary)).hom) ≫
  85     (SyntheticCategory.biShift (0,-eg)).map
  86       ((SyntheticCategory.biShift_comp (1,1) (0,-eh)).hom.app X) ≫
  87     (SyntheticCategory.biShift_comp ((1,1)+(0,-eh)) (0,-eg)).hom.app X ≫
  88     eqToHom (congrArg (fun p => (SyntheticCategory.biShift p).obj X)
  89       (show ((1,1)+(0,-eh))+(0,-eg) =
  90           (0,(normalizedExponent D.classicalTower T.f : ℤ))+(1,0) from by
  91         dsimp [eg, eh]; ext <;> simp only []; omega)) ≫
  92     (SyntheticCategory.biShift_comp (0,(normalizedExponent D.classicalTower T.f : ℤ)) (1,0)).inv.app X ≫
  93     (SyntheticCategory.biShift_compat (Syn := Syn) 1).hom.app _

  95 /-- The three arrows are derived from the three normalized maps already
  96 selected by D; distinguishedness is a separate comparison obligation. -/
  97 public def normalizedTriangle (T : TriangleData D.auxiliary)
  98     (he : (normalizedExponent D.classicalTower T.f : ℤ) +
  99       normalizedExponent D.classicalTower T.g + normalizedExponent D.classicalTower T.h = 1) :
 100     Triangle Syn :=
 101   Triangle.mk (D.normalizedMap T.X T.Y T.f).map
 102     (negativeLift (normalizedExponent D.classicalTower T.g) (D.normalizedMap T.Y T.Z T.g).map)
 103     (normalizedConnecting D T he)

 154 /-- The needed exponent equation follows from the three supplied values. -/
 155 private theorem NuCofiberApplicability.exponent_sum (A : NuCofiberApplicability D) :
 156     (normalizedExponent D.classicalTower D.auxiliary.nuRouteTriangle.f : ℤ) +
 157       normalizedExponent D.classicalTower D.auxiliary.nuRouteTriangle.g +
 158       normalizedExponent D.classicalTower D.auxiliary.nuRouteTriangle.h = 1 := by
 159   change (normalizedExponent D.classicalTower D.auxiliary.nuMap : ℤ) + _ + _ = 1
 160   rw [A.nu_exponent, A.bottom_exponent, A.top_exponent]
 161   norm_num
