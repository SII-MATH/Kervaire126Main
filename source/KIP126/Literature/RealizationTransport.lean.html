  44 /-- The same actual lambda action is natural under postcomposition. -/
  45 private theorem bhsLambdaMultiplyPostcompose {X Y : Syn} (g : X ⟶ Y)
  46     {m w : ℤ} (a : BiHom m w X) (k : ℕ) :
  47     lambdaMultiply k (a ≫ g) = lambdaMultiply k a ≫ g := by
  48   dsimp only [lambdaMultiply]
  49   simp (config := { congrConsts := false }) only [Category.assoc]

  51 private theorem bhsHomotopyRegradePostcompose {X Y : Syn} (g : X ⟶ Y)
  52     {m m' w w' : ℤ} (hm : m=m') (hw : w=w') (a : BiHom m w X) :
  53     homotopyRegrade hm hw (a ≫ g) = homotopyRegrade hm hw a ≫ g := by
  54   cases hm
  55   cases hw
  56   rfl

  58 /-- The realization square is forced by the existing nu recovery and zero
  59 shift naturality, for the same actual classical map. -/
  60 private theorem bhsRealizeNuZeroNatural (R : RealizationCoordinates D)
  61     (X Y : ClassicalObject) (f : X.obj D.auxiliary ⟶ Y.obj D.auxiliary)
  62     {m w : ℤ} (a : BiHom m w (nuZero D X)) :
  63     realizeNuZero D R Y
  64         (a ≫ (SyntheticCategory.biShift (0,0)).map (D.nu.functor.map f)) =
  65       realizeNuZero D R X a ≫ f := by
  66   have hz := SyntheticCategory.biShift_zero.hom.naturality (D.nu.functor.map f)
  67   have hr := D.recovery.nuRealizationIso.hom.naturality f
  68   dsimp only [Functor.comp_map, Functor.id_map] at hz hr
  69   have hmap := congrArg
  70     (fun z => D.recovery.realization.map z ≫
  71       D.recovery.nuRealizationIso.hom.app (Y.obj D.auxiliary)) hz
  72   simp (config := { congrConsts := false }) only [Functor.map_comp, Category.assoc] at hmap
  73   rw [hr] at hmap
  74   unfold realizeNuZero
  75   rw [Functor.map_comp]
  76   simpa (config := { congrConsts := false }) only [Category.assoc] using congrArg
  77     (fun z => (R.sphere m w).hom ≫ D.recovery.realization.map a ≫ z) hmap

  79 /-- No compactness is needed for the torsion-to-zero direction.  It uses
  80 the prescribed realization coordinates for the same lambda powers. -/
  81 private theorem bhsRealizationLambdaAction (R : RealizationCoordinates D)
  82     {X : Syn} (m w : ℤ) (k : ℕ) (a : BiHom m w X) :
  83     (R.sphere m (w-k)).hom ≫ D.recovery.realization.map (lambdaMultiply k a) =
  84       (R.sphere m w).hom ≫ D.recovery.realization.map a := by
  85   have hf : lambdaMultiply k a =
  86       lambdaMultiply k (𝟙 (Smn (Syn := Syn) m w)) ≫ a := by
  87     dsimp only [lambdaMultiply]
  88     simp (config := { congrConsts := false }) only [Category.assoc, Category.comp_id]
  89   rw [hf, Functor.map_comp, ← Category.assoc, R.lambda]
  90 
  91 private theorem bhsRealizationZeroOfLambdaTorsion (R : RealizationCoordinates D)
  92     {X : Syn} {m w : ℤ} (a : BiHom m w X)
  93     (ha : ∃ k : ℕ, lambdaMultiply k a = 0) :
  94     D.recovery.realization.map a = 0 := by
  95   letI := R.additive
  96   obtain ⟨k,hk⟩ := ha
  97   have h := bhsRealizationLambdaAction D R m w k a
  98   rw [hk, Functor.map_zero, comp_zero] at h
  99   apply (cancel_epi (R.sphere m w).hom).mp
 100   simpa (config := { congrConsts := false }) only [comp_zero] using h.symm

 102 /-- An actual lambda-power nonzero image reflects nonvanishing without
 103 assuming injectivity of the completion map or replacing its source. -/
 104 private theorem bhsLambdaNonzeroOfPostcompose {X Y : Syn} (g : X ⟶ Y)
 105     {m w : ℤ} (a : BiHom m w X) (k : ℕ)
 106     (ha : lambdaMultiply k (a ≫ g) ≠ 0) : lambdaMultiply k a ≠ 0 := by
 107   intro hz
 108   apply ha
 109   rw [bhsLambdaMultiplyPostcompose, hz, zero_comp]

 111 /-- Tower maps transport actual filtration witnesses.  No completeness or
 112 separation is inferred from convergence. -/
 113 private theorem bhsFiltrationPostcompose {X Y : Syn} (g : X ⟶ Y)
 114     {m w : ℤ} (s : ℤ) (a : BiHom m w X)
 115     (ha : FiltrationAtLeast D.syntheticTower s a) :
 116     FiltrationAtLeast D.syntheticTower s (a ≫ g) := by
 117   obtain ⟨b,hb⟩ := ha
 118   change b ≫ (D.syntheticTower.tower X).projection s = a at hb
 119   refine ⟨b ≫ (D.syntheticTower.map g).stage s.toNat, ?_⟩
 120   change (b ≫ (D.syntheticTower.map g).stage s.toNat) ≫
 121       (D.syntheticTower.tower Y).projection s = a ≫ g
 122   rw [Category.assoc, TowerMap.projection, ← Category.assoc, hb]
 123 
 124 /-- The actual same nu map carries a lambda-division witness to its image. -/
 125 private theorem bhsLambdaDivisiblePostcompose
 126     (X Y : ClassicalObject) (f : X.obj D.auxiliary ⟶ Y.obj D.auxiliary)
 127     (m w s : ℤ) (h : w-m ≤ s)
 128     (a : BiHom m w (D.nu.functor.obj (X.obj D.auxiliary)))
 129     (ha : LambdaDivisible D X m w s h a) :
 130     LambdaDivisible D Y m w s h (a ≫ D.nu.functor.map f) := by
 131   obtain ⟨b,hb⟩ := ha
 132   refine ⟨b ≫ D.nu.functor.map f, ?_⟩
 133   have hs : m+s - ((m+s-w).toNat : ℤ) = w := by
 134     rw [Int.toNat_of_nonneg (by omega)]
 135     omega
 136   change homotopyRegrade rfl hs
 137       (lambdaMultiply (m+s-w).toNat (b ≫ D.nu.functor.map f)) = a ≫ D.nu.functor.map f
 138   rw [bhsLambdaMultiplyPostcompose]
 139   rw [bhsHomotopyRegradePostcompose, hb]

 141 /-- Pointwise version of a48's permanent-lift transport: it records every
 142 required actual comparison instead of assuming the selected conclusion. -/
 143 private theorem bhsPermanentLiftTransport
 144     (X Y : ClassicalObject) (f : X.obj D.auxiliary ⟶ Y.obj D.auxiliary)
 145     (s t : ℤ) (x : (D.classical.obj (X.obj D.auxiliary)).Page 2 (s,t))
 146     (source : PermanentLiftAt D Y)
 147     (cycles : ClassicalPermanentCycle D X (s,t) x ↔
 148       ClassicalPermanentCycle D Y (s,t) ((D.classical.map f).pageMap 2 (s,t) x))
 149     (lifts : (∃ b : BiHom (t-s) t (nuZero D Y),
 150         quotientClass 1 b = firstLabel D Y s t ((D.classical.map f).pageMap 2 (s,t) x)) ↔
 151       ∃ a : BiHom (t-s) t (nuZero D X), quotientClass 1 a = firstLabel D X s t x) :
 152     ClassicalPermanentCycle D X (s,t) x ↔
 153       ∃ a : BiHom (t-s) t (nuZero D X), quotientClass 1 a = firstLabel D X s t x :=
 154   cycles.trans ((source s t _).trans lifts)

 156 /-- The source BHS filtration formula transports only with BOTH actual
 157 filtration comparison and lambda-divisibility reflection. -/
 158 private theorem bhsFiltrationLambdaTransport
 159     (X Y : ClassicalObject) (f : X.obj D.auxiliary ⟶ Y.obj D.auxiliary)
 160     (m w s : ℤ) (h : w-m ≤ s)
 161     (a : BiHom m w (D.nu.functor.obj (X.obj D.auxiliary)))
 162     (source : FiltrationLambdaAt D Y)
 163     (filtration : FiltrationAtLeast D.syntheticTower s a ↔
 164       FiltrationAtLeast D.syntheticTower s (a ≫ D.nu.functor.map f))
 165     (divisibility : LambdaDivisible D Y m w s h (a ≫ D.nu.functor.map f) ↔
 166       LambdaDivisible D X m w s h a) :
 167     FiltrationAtLeast D.syntheticTower s a ↔ LambdaDivisible D X m w s h a :=
 168   filtration.trans ((source m w s h _).trans divisibility)

 173 /-- The transported BHS conclusions for one given realization and the same
 174 selected towers, classes and first-quotient labels. The complete-source
 175 comparison is a binding obligation; no arbitrary θ₅ choice theorem is a field. -/
 176 private structure BHSSelectedRealizationComparisons (R : RealizationCoordinates D) : Prop where
 177   lifetime : ∀ (X : ClassicalObject) (s t : ℤ) (r : ℕ), 1 ≤ r →
 178     ∀ (x : (D.classical.obj (X.obj D.auxiliary)).Page 2 (s,t))
 179       (a : BiHom (t-s) t (nuZero D X)),
 180     ClassicalPermanentCycle D X (s,t) x →
 181     SurvivesTo (D.classical.obj (X.obj D.auxiliary)) (r+1) (s,t) x →
 182     quotientClass 1 a = firstLabel D X s t x → lambdaMultiply (r-1) a ≠ 0
 183   detection : ∀ (X : ClassicalObject) (s t : ℤ)
 184     (x : (D.classical.obj (X.obj D.auxiliary)).Page 2 (s,t))
 185     (a : BiHom (t-s) t (nuZero D X)),
 186     NonzeroSurvival (D.classical.obj (X.obj D.auxiliary)) (s,t) x →
 187     quotientClass 1 a = firstLabel D X s t x →
 188     ClassicalDetects (D.classicalConvergence X) (s,t) x (realizeNuZero D R X a)
 189   prescribed_lift : ∀ (X : ClassicalObject) (s t : ℤ)
 190     (x : (D.classical.obj (X.obj D.auxiliary)).Page 2 (s,t))
 191     (a : HomotopyGroup (t-s) (X.obj D.auxiliary)),
 192     NonzeroSurvival (D.classical.obj (X.obj D.auxiliary)) (s,t) x →
 193     ClassicalDetects (D.classicalConvergence X) (s,t) x a →
 194     ∃ b : BiHom (t-s) t (nuZero D X), quotientClass 1 b = firstLabel D X s t x ∧
 195       realizeNuZero D R X b = a
 196   boundary_lift : ∀ (X : ClassicalObject) (s t : ℤ) (r : ℕ), 2 ≤ r →
 197     ∀ y : (D.classical.obj (X.obj D.auxiliary)).Page 2 (s,t),
 198     ClassicalPermanentCycle D X (s,t) y →
 199     HitOnPage (D.classical.obj (X.obj D.auxiliary)) r (s,t) y →
 200     ∃ a : BiHom (t-s) t (nuZero D X), quotientClass 1 a = firstLabel D X s t y ∧
 201       lambdaMultiply (r-1) a = 0
 202 
 203 end
