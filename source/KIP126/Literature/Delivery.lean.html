  39 /-- Sources, their fixed coordinates, and comparison witnesses. -/
  40 private structure LiteratureBindings extends ClassicalBindings D where
  41   rigidity : RigidityBindings D
  42   eInfinity : EInfinityPresentation D
  43   infinityShift : InfinityShiftCoordinates D
  44   weights_lambda : ∀ (X : ClassicalObject) (s t w₁ w₂ : ℤ)
  45     (hw : w₁ ≤ w₂) (ht : w₂ ≤ t),
  46     rigidity.weights.map X s t w₁ w₂ hw ht = infinityShift.weightMap D X s t w₁ w₂ hw
  47   tmfSource : TmfSourceData D tmf
  48   secondary : SecondaryBindings D
  49   nuCofiber : NuCofiberApplicability D
  50   realizationTransport : BHSSelectedRealizationComparisons D realization
  51 
  52 /-- Mathematical results use exactly the selected sources and background.
  53 Internal arbitrary-choice θ₅ transport remains a separate theorem parameter. -/
  54 private structure LiteratureResults (B : LiteratureBindings D) where
  55   foundation : ClassicalFoundationResults D
  56   classical : ClassicalResults D B.toClassicalBindings
  57   classicalRange : ClassicalRangeResults D
  58   rigidity : RigidityResults D B.rigidity
  59   triangles : NormalizedTriangleResults D
  60   lifts : LiftSourceResults D
  61   eInfinity : EInfinityCompatibility D B.eInfinity B.infinityShift
  62   bhs : BHSResults D
  63   selectedComparisons : BHSSelectedComparisons D
  64   realization : BHSRealizationResults D B.realization
  65   tmf : TmfSourceResults D B.tmfSource
  66   bx : BXDistinguishedInput D B.eta
  67   secondary : SecondaryResults D B.toClassicalBindings B.secondary
  68 
  69 /-- A consumer supplies one correlated delivery on its given model. -/
  70 private structure LiteratureInterface where
  71   bindings : LiteratureBindings D
  72   results : LiteratureResults D bindings
  73 
  74 /-- The corrected weight map is epi by the same actual λ/quotient formula. -/
  75 private theorem LiteratureInterface.weight_epi (L : LiteratureInterface D)
  76     (X : ClassicalObject) (s t w₁ w₂ : ℤ) (hw : w₁ ≤ w₂) (ht : w₂ ≤ t) :
  77     CategoryTheory.Epi (L.bindings.rigidity.weights.map X s t w₁ w₂ hw ht) := by
  78   rw [L.bindings.weights_lambda]
  79   exact L.results.eInfinity.weight_epi D X s t w₁ w₂ hw ht
