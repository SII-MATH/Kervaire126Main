  12 /-! Correlated secondary-operation input for the one supplied model. The
  13 general Moss contract remains available, while this delivery fixes its route
  14 scope to the actual sphere and its mapping composition to the already chosen
  15 sphere E₂ algebra. No second η or freely selected object-scope predicate is
  16 introduced. -/

  26   {Syn : Type w} [SyntheticCategory.{w,v} Syn]

  29 /-- Model-side mapping, tensor and Toda data. The sphere mapping spectrum
  30 uses D's actual tower; its E₂ product and unit agree with D's existing algebra. -/
  31 private structure SecondaryBindings where
  32   closed : ClosedSymmetricTensorTriangulated C
  33   mapping : MappingSpectrumSystem closed D.classicalTower
  34     (fun X : C => X = SphereSpectrum)
  35   moss : Moss.Context mapping
  36   sphereComposition : MappingSphereCompatibility moss.composition D.sphereAlgebra rfl
  37   may : Route.MayContext Syn
  38   toda : Route.TodaSourceData (Syn := Syn)
  39 
  40 /-- Source results are separate from construction of the bound model data.
  41 The Toda statements use precisely the η in the existing classical binding. -/
  42 private structure SecondaryResults (B : ClassicalBindings D) (Q : SecondaryBindings D) : Prop where
  43   moss : Moss.SourceResults Q.mapping Q.moss
  44   may : Route.MaySourceResults Syn Q.may
  45   toda : Route.TodaSourceResults D B.eta Q.toda
