  15 /-- The actual product of the two specified standard one-line classes. -/
  16 public noncomputable def hiProduct (i j : ℕ) :
  17     ActualE2At model.classical (hiDegree i + hiDegree j) :=
  18   model.sphereAlgebra.product (hiDegree i) (hiDegree j)
  19     (model.standardClasses.standardHi i) (model.standardClasses.standardHi j)
