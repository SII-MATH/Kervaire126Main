  21 /-- Two infinite-cycle representatives with the same class on one page
  22 have the same class at infinity: their difference is already a boundary. -/
  23 public theorem SSData.infinity_projection_eq_of_page_projection_eq
  24     (D : SSData (ModuleCat.{v} R)) (n : WithTop ℕ)
  25     (z z' : (Subobject.underlying.obj (D.Z ⊤) : ModuleCat R))
  26     (h : (Subobject.ofLE _ _ (D.Z_anti le_top) ≫ D.pageπ n) z =
  27       (Subobject.ofLE _ _ (D.Z_anti le_top) ≫ D.pageπ n) z') :
  28     D.pageπ ⊤ z = D.pageπ ⊤ z' := by
  29   apply sub_eq_zero.mp
  30   rw [← map_sub]
  31   apply (subobject_cokernel_π_eq_zero_iff (D.B ⊤) (D.Z ⊤) (D.B_le_Z ⊤) _).mpr
  32   have hz : D.pageπ n ((Subobject.ofLE _ _ (D.Z_anti le_top)) (z - z')) = 0 := by
  33     change (Subobject.ofLE _ _ (D.Z_anti le_top) ≫ D.pageπ n) (z - z') = 0
  34     rw [map_sub, h, sub_self]
  35   have hb := (subobject_cokernel_π_eq_zero_iff (D.B n) (D.Z n) (D.B_le_Z n) _).mp hz
  36   have heq : (D.Z n).arrow ((Subobject.ofLE _ _ (D.Z_anti le_top)) (z - z')) =
  37       (D.Z ⊤).arrow (z - z') :=
  38     ConcreteCategory.congr_hom (Subobject.ofLE_arrow (D.Z_anti le_top)) _
  39   rw [heq] at hb
  40   exact (ModuleCat.subobjectModule D.V).monotone (D.B_mono le_top) hb

  42 /-- A common representative on a finite page is permanent when the later
  43 outgoing differentials vanish at its grading. It may still be a boundary. -/
  44 public theorem isPermanentCycle_of_reachesPage
  45     (E : SpectralSequence (ModuleCat.{v} R) (ℤ × ℤ))
  46     (N : ℤ) (hN : E.r₀ ≤ N) (p : ℤ × ℤ) (x : E.Page 2 p)
  47     (hx : ReachesPage E N p x)
  48     (hout : ∀ r : ℤ, N ≤ r → E.d r p = 0) :
  49     IsPermanentCycle E p x := by
  50   obtain ⟨xr, hN2, z, hz, _⟩ := hx
  51   let D := E.ssData p
  52   let n : WithTop ℕ := ↑(N - E.r₀).toNat
  53   have hZ : D.Z ⊤ = D.Z n := cycles_top_eq_of_d_eq_zero E N hN p hout
  54   let zInf := (Subobject.ofLE (D.Z n) (D.Z ⊤) hZ.ge) z
  55   refine ⟨zInf, ?_⟩
  56   change (Subobject.ofLE _ _ (D.Z_anti le_top) ≫ D.pageπ _)
  57     ((Subobject.ofLE (D.Z n) (D.Z ⊤) hZ.ge) z) = x
  58   rw [← CategoryTheory.comp_apply, ← Category.assoc, Subobject.ofLE_comp_ofLE]
  59   exact hz

  61 /-- Nonzero survival on a finite page persists at infinity when both later
  62 outgoing and incoming differentials vanish at this grading. -/
  63 public theorem nonzeroSurvival_of_survivesTo
  64     (E : SpectralSequence (ModuleCat.{v} R) (ℤ × ℤ))
  65     (N : ℤ) (hN : E.r₀ ≤ N) (p : ℤ × ℤ) (x : E.Page 2 p)
  66     (hx : SurvivesTo E N p x)
  67     (hout : ∀ r : ℤ, N ≤ r → E.d r p = 0)
  68     (hin : ∀ r : ℤ, N ≤ r → E.d r (p - E.diffDeg r) = 0) :
  69     NonzeroSurvival E p x := by
  70   obtain ⟨xr, ⟨hN2, z, hz, hzr⟩, hne⟩ := hx
  71   let D := E.ssData p
  72   let n : WithTop ℕ := ↑(N - E.r₀).toNat
  73   have hZ : D.Z ⊤ = D.Z n := cycles_top_eq_of_d_eq_zero E N hN p hout
  74   have hB : D.B ⊤ = D.B n := boundaries_top_eq_of_d_eq_zero E N hN p hin
  75   let zInf := (Subobject.ofLE (D.Z n) (D.Z ⊤) hZ.ge) z
  76   refine ⟨zInf, ?_, ?_⟩
  77   · change (Subobject.ofLE _ _ (D.Z_anti le_top) ≫ D.pageπ _)
  78       ((Subobject.ofLE (D.Z n) (D.Z ⊤) hZ.ge) z) = x
  79     rw [← CategoryTheory.comp_apply, ← Category.assoc, Subobject.ofLE_comp_ofLE]
  80     exact hz
  81   · intro hzero
  82     apply hne
  83     rw [← hzr]
  84     apply (subobject_cokernel_π_eq_zero_iff (D.B n) (D.Z n) (D.B_le_Z n) z).mpr
  85     have hb := (subobject_cokernel_π_eq_zero_iff (D.B ⊤) (D.Z ⊤)
  86       (D.B_le_Z ⊤) zInf).mp hzero
  87     have heq : (D.Z ⊤).arrow zInf = (D.Z n).arrow z :=
  88       ConcreteCategory.congr_hom (Subobject.ofLE_arrow hZ.ge) z
  89     rwa [heq, hB] at hb
