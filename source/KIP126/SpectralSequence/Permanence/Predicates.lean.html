  15 /-- Successor-page representatives in the internal nested-subobject model.
  16 This uses one next-cycle representative for both quotient classes. -/
  17 public def NextPageRelation {R : Type u} [Ring R]
  18     (E : SpectralSequence (ModuleCat.{v} R) (ℤ × ℤ)) (r : ℤ)
  19     (p : ℤ × ℤ) (x : E.Page r p) (y : E.Page (r + 1) p) : Prop :=
  20   let D := E.ssData p
  21   let n : WithTop ℕ := ↑(r - E.r₀).toNat
  22   let n' : WithTop ℕ := ↑(r + 1 - E.r₀).toNat
  23   ∃ z : (Subobject.underlying.obj (D.Z n') : ModuleCat R),
  24     (Subobject.ofLE (D.Z n') (D.Z n)
  25       (D.Z_anti (by
  26         change (↑(r - E.r₀).toNat : WithTop ℕ) ≤ ↑(r + 1 - E.r₀).toNat
  27         exact_mod_cast (show (r - E.r₀).toNat ≤ (r + 1 - E.r₀).toNat by omega))) ≫
  28       D.pageπ n) z = x ∧
  29       D.pageπ n' z = y

  31 /-- A specified page class has a common Z∞ representative whose E∞ image is
  32 nonzero. This is internal SSData reasoning, with no Mathlib SS dependency. -/
  33 public def NonzeroSurvival {R : Type u} [Ring R]
  34     (E : SpectralSequence (ModuleCat.{v} R) (ℤ × ℤ))
  35     (p : ℤ × ℤ) (x : E.Page 2 p) : Prop :=
  36   let D := E.ssData p
  37   let n : WithTop ℕ := ↑(2 - E.r₀).toNat
  38   ∃ z : (Subobject.underlying.obj (D.Z ⊤) : ModuleCat R),
  39     (Subobject.ofLE (D.Z ⊤) (D.Z n) (D.Z_anti le_top) ≫ D.pageπ n) z = x ∧
  40       D.pageπ ⊤ z ≠ 0
