   7 /-!
   8 # Quotient pages of a filtered complex
   9 
  10 The cycle and boundary subobjects determine these objects directly. This
  11 module adds no alternative spectral-sequence structure.
  12 -/
  13 
  14 namespace KIP126.Core.SpectralSequence.FilteredComplex
  15 
  16 open CategoryTheory CategoryTheory.Limits
  17 
  18 universe u v
  19 
  20 variable {C : Type u} [Category.{v} C] [Abelian C]
  21 
  22 /-- The quotient of page cycles by page boundaries, including the infinity page. -/
  23 noncomputable def pageObj (FC : FilteredComplex C) (s k : ℤ) (r : WithTop ℕ) : C :=
  24   cokernel (Subobject.ofLE (FC.boundarySubobject s k r)
  25     (FC.cycleSubobject s k r) (FC.B_le_Z_aux s k r))
  26 
  27 /-- Projection from the cycle subobject to its quotient page. -/
  28 noncomputable def pageπ (FC : FilteredComplex C) (s k : ℤ) (r : WithTop ℕ) :
  29     Subobject.underlying.obj (FC.cycleSubobject s k r) ⟶ FC.pageObj s k r :=
  30   cokernel.π (Subobject.ofLE (FC.boundarySubobject s k r)
  31     (FC.cycleSubobject s k r) (FC.B_le_Z_aux s k r))
  32 
  33 end KIP126.Core.SpectralSequence.FilteredComplex
