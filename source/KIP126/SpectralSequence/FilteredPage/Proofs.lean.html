  26 /-- The Z_anti law for the canonical page subobjects. -/
  27 theorem cycleSubobject_antitone (FC : FilteredComplex C) (s k : ℤ) :
  28     Antitone (FC.cycleSubobject s k) := by
  29   intro r₁ r₂ hr₁₂
  30   suffices key : ∀ {X Y : C} {K₁ K₂ : Subobject X} (g : X ⟶ Y)
  31       (hle : K₂ ≤ K₁),
  32       imageSubobject (K₂.arrow ≫ g) ≤ imageSubobject (K₁.arrow ≫ g) by
  33     rcases r₁ with _ | n₁ <;> rcases r₂ with _ | n₂
  34     · -- (⊤, ⊤)
  35       exact le_refl _
  36     · -- (⊤, ↑n₂): impossible
  37       exact absurd hr₁₂ (WithTop.not_top_le_coe n₂)
  38     · -- (↑n₁, ⊤): ker(arrow ≫ d) ≤ ker(arrow ≫ d ≫ cokernel.π)
  39       change imageSubobject
  40           ((kernelSubobject ((FC.filtration.F s k).arrow ≫ FC.complex.d k (k - 1))).arrow ≫ _) ≤
  41         imageSubobject
  42           ((kernelSubobject ((FC.filtration.F s k).arrow ≫ FC.complex.d k (k - 1) ≫
  43             cokernel.π ((FC.filtration.F (s + ↑n₁) (k - 1)).arrow))).arrow ≫ _)
  44       apply key
  45       apply le_kernelSubobject
  46       have h1 : (kernelSubobject ((FC.filtration.F s k).arrow ≫ FC.complex.d k (k - 1))).arrow ≫
  47           ((FC.filtration.F s k).arrow ≫ FC.complex.d k (k - 1)) = 0 := kernelSubobject_arrow_comp _
  48       calc (kernelSubobject ((FC.filtration.F s k).arrow ≫ FC.complex.d k (k - 1))).arrow ≫
  49               ((FC.filtration.F s k).arrow ≫ (FC.complex.d k (k - 1) ≫
  50                 cokernel.π ((FC.filtration.F (s + ↑n₁) (k - 1)).arrow)))
  51           = ((kernelSubobject ((FC.filtration.F s k).arrow ≫ FC.complex.d k (k - 1))).arrow ≫
  52               (FC.filtration.F s k).arrow ≫ FC.complex.d k (k - 1)) ≫
  53               cokernel.π ((FC.filtration.F (s + ↑n₁) (k - 1)).arrow) := by
  54               simp only [Category.assoc]
  55         _ = 0 ≫ cokernel.π ((FC.filtration.F (s + ↑n₁) (k - 1)).arrow) := by rw [h1]
  56         _ = 0 := zero_comp
  57     · -- (↑n₁, ↑n₂): n₁ ≤ n₂
  58       change imageSubobject
  59           ((kernelSubobject ((FC.filtration.F s k).arrow ≫ FC.complex.d k (k - 1) ≫
  60             cokernel.π ((FC.filtration.F (s + ↑n₂) (k - 1)).arrow))).arrow ≫ _) ≤
  61         imageSubobject
  62           ((kernelSubobject ((FC.filtration.F s k).arrow ≫ FC.complex.d k (k - 1) ≫
  63             cokernel.π ((FC.filtration.F (s + ↑n₁) (k - 1)).arrow))).arrow ≫ _)
  64       have hn : n₁ ≤ n₂ := WithTop.coe_le_coe.mp hr₁₂
  65       have hfil : FC.filtration.F (s + ↑n₂) (k - 1) ≤ FC.filtration.F (s + ↑n₁) (k - 1) :=
  66         FC.filtration.le_of_le (by omega) (k - 1)
  67       have hcomp : (FC.filtration.F (s + ↑n₂) (k - 1)).arrow ≫
  68           cokernel.π ((FC.filtration.F (s + ↑n₁) (k - 1)).arrow) = 0 := by
  69         rw [show (FC.filtration.F (s + ↑n₂) (k - 1)).arrow =
  70           Subobject.ofLE _ _ hfil ≫ (FC.filtration.F (s + ↑n₁) (k - 1)).arrow
  71           from (Subobject.ofLE_arrow hfil).symm]
  72         rw [Category.assoc, cokernel.condition, comp_zero]
  73       set f_n₂ := (FC.filtration.F s k).arrow ≫ FC.complex.d k (k - 1) ≫
  74         cokernel.π ((FC.filtration.F (s + ↑n₂) (k - 1)).arrow) with hf_n₂_def
  75       set f_n₁ := (FC.filtration.F s k).arrow ≫ FC.complex.d k (k - 1) ≫
  76         cokernel.π ((FC.filtration.F (s + ↑n₁) (k - 1)).arrow) with hf_n₁_def
  77       have hker : kernelSubobject f_n₂ ≤ kernelSubobject f_n₁ := by
  78         apply le_kernelSubobject
  79         set desc := cokernel.desc ((FC.filtration.F (s + ↑n₂) (k - 1)).arrow)
  80           (cokernel.π ((FC.filtration.F (s + ↑n₁) (k - 1)).arrow)) hcomp
  81         have hfactor : f_n₁ = f_n₂ ≫ desc := by
  82           simp only [f_n₂, f_n₁, desc, Category.assoc, cokernel.π_desc]
  83         rw [hfactor, ← Category.assoc, kernelSubobject_arrow_comp, zero_comp]
  84       exact key _ hker
  85   intro X Y K₁ K₂ g hle
  86   rw [show K₂.arrow ≫ g = Subobject.ofLE K₂ K₁ hle ≫ K₁.arrow ≫ g by
  87     rw [← Category.assoc, Subobject.ofLE_arrow]]
  88   exact imageSubobject_comp_le _ _

  90 /-- The B_mono law for the canonical page subobjects. -/
  91 theorem boundarySubobject_monotone (FC : FilteredComplex C) (s k : ℤ) :
  92     Monotone (FC.boundarySubobject s k) := by
  93   intro r₁ r₂ hr₁₂
  94   suffices img_mono : ∀ {X Y : C} {K₁ K₂ : Subobject X} (g : X ⟶ Y)
  95       (hle : K₁ ≤ K₂),
  96       imageSubobject (K₁.arrow ≫ g) ≤ imageSubobject (K₂.arrow ≫ g) by
  97     have ofLE_mono : ∀ {X Y : C} {I₁ I₂ Q : Subobject X}
  98         (h₁ : I₁ ≤ Q) (h₂ : I₂ ≤ Q) (hle : I₁ ≤ I₂)
  99         (g : Subobject.underlying.obj Q ⟶ Y),
 100         imageSubobject (Subobject.ofLE I₁ Q h₁ ≫ g) ≤
 101         imageSubobject (Subobject.ofLE I₂ Q h₂ ≫ g) := by
 102       intro X Y I₁ I₂ Q h₁ h₂ hle g
 103       have factored : Subobject.ofLE I₁ Q h₁ =
 104           Subobject.ofLE I₁ I₂ hle ≫ Subobject.ofLE I₂ Q h₂ := by
 105         apply (cancel_mono Q.arrow).mp
 106         simp only [Category.assoc, Subobject.ofLE_arrow]
 107       rw [factored, Category.assoc]
 108       exact imageSubobject_comp_le _ _
 109     simp only [FilteredComplex.boundarySubobject]
 110     rcases r₁ with _ | n₁ <;> rcases r₂ with _ | n₂
 111     · exact le_refl _
 112     · exact absurd hr₁₂ (WithTop.not_top_le_coe n₂)
 113     · -- (↑n₁, ⊤): image from F^{s-n₁} ≤ image from all of A^{k+1}
 114       apply ofLE_mono inf_le_right inf_le_right
 115       apply inf_le_inf_right
 116       exact imageSubobject_comp_le _ _
 117     · -- (↑n₁, ↑n₂): F^{s-n₁+1} ≤ F^{s-n₂+1} (since n₁ ≤ n₂, s-n₂+1 ≤ s-n₁+1)
 118       have hn : n₁ ≤ n₂ := WithTop.coe_le_coe.mp hr₁₂
 119       have hfil : FC.filtration.F (s - ↑n₁ + 1) (k + 1) ≤ FC.filtration.F (s - ↑n₂ + 1) (k + 1) :=
 120         FC.filtration.le_of_le (by omega) (k + 1)
 121       apply ofLE_mono inf_le_right inf_le_right
 122       apply inf_le_inf_right
 123       exact img_mono _ hfil
 124   intro X Y K₁ K₂ g hle
 125   rw [show K₁.arrow ≫ g = Subobject.ofLE K₁ K₂ hle ≫ K₂.arrow ≫ g by
 126     rw [← Category.assoc, Subobject.ofLE_arrow]]
 127   exact imageSubobject_comp_le _ _

 129 /-- The Z_zero law for the canonical page subobjects. -/
 130 theorem cycleSubobject_zero (FC : FilteredComplex C) (s k : ℤ) :
 131     FC.cycleSubobject s k 0 = ⊤ := by
 132   simp only [FilteredComplex.cycleSubobject]
 133   have hf0 : (FC.filtration.F s k).arrow ≫ FC.complex.d k (k - 1) ≫
 134       cokernel.π ((FC.filtration.F (s + ↑(0 : ℕ)) (k - 1)).arrow) = 0 := by
 135     obtain ⟨φ, hφ⟩ := FC.differential_preserves s k
 136     simp only [← Category.assoc]
 137     rw [← hφ]
 138     simp only [Category.assoc]
 139     have hle : FC.filtration.F s (k - 1) ≤ FC.filtration.F (s + ↑(0 : ℕ)) (k - 1) :=
 140       le_of_eq (by congr 1; omega)
 141     rw [show (FC.filtration.F s (k - 1)).arrow =
 142       Subobject.ofLE (FC.filtration.F s (k - 1)) (FC.filtration.F (s + ↑(0 : ℕ)) (k - 1)) hle ≫
 143       (FC.filtration.F (s + ↑(0 : ℕ)) (k - 1)).arrow from (Subobject.ofLE_arrow hle).symm]
 144     simp only [Category.assoc, cokernel.condition, comp_zero]
 145   haveI : IsIso (kernelSubobject ((FC.filtration.F s k).arrow ≫ FC.complex.d k (k - 1) ≫
 146       cokernel.π ((FC.filtration.F (s + ↑(0 : ℕ)) (k - 1)).arrow))).arrow := by
 147     rw [Subobject.isIso_arrow_iff_eq_top]
 148     simp only [hf0, kernelSubobject_zero]
 149   rw [imageSubobject_iso_comp]
 150   let π := cokernel.π (Subobject.ofLE (FC.filtration.F (s + 1) k)
 151     (FC.filtration.F s k) (FC.filtration.mono s k))
 152   haveI : Epi (image.ι π) := epi_image_of_epi _
 153   haveI : IsIso (image.ι π) := isIso_of_mono_of_epi _
 154   haveI : IsIso (imageSubobject π).arrow := by
 155     have h : IsIso ((imageSubobjectIso π).hom ≫ image.ι π) := inferInstance
 156     rw [imageSubobject_arrow] at h
 157     exact h
 158   apply (Subobject.isIso_arrow_iff_eq_top _).mp
 159   change IsIso (imageSubobject π).arrow
 160   infer_instance

 162 /-- The Z_top_greatest endpoint law under the corresponding filtration bound. -/
 163 theorem cycleSubobject_top_greatest (FC : FilteredComplex C)
 164     (bnd : FC.filtration.IsBoundedAbove) (s k : ℤ)
 165     (X : Subobject (FC.filtration.associatedGraded s k))
 166     (hX : ∀ i : ℕ, X ≤ FC.cycleSubobject s k ↑i) :
 167     X ≤ FC.cycleSubobject s k ⊤ := by
 168   obtain ⟨N, hN⟩ : ∃ N : ℕ, FC.filtration.F (s + ↑N) (k - 1) = ⊥ := by
 169     use (bnd.hi (k - 1) - s).toNat
 170     apply bnd.boundedAbove; omega
 171   have h_zero : (FC.filtration.F (s + ↑N) (k - 1)).arrow = 0 := by
 172     rw [hN, Subobject.bot_arrow]
 173   haveI : IsIso (cokernel.π ((FC.filtration.F (s + ↑N) (k - 1)).arrow)) := by
 174     rw [h_zero]; exact cokernel.π_zero_isIso
 175   have hle : kernelSubobject ((FC.filtration.F s k).arrow ≫ FC.complex.d k (k - 1) ≫
 176       cokernel.π ((FC.filtration.F (s + ↑N) (k - 1)).arrow)) ≤
 177     kernelSubobject ((FC.filtration.F s k).arrow ≫ FC.complex.d k (k - 1)) := by
 178     apply le_kernelSubobject
 179     have h1 := kernelSubobject_arrow_comp ((FC.filtration.F s k).arrow ≫ FC.complex.d k (k - 1) ≫
 180         cokernel.π ((FC.filtration.F (s + ↑N) (k - 1)).arrow))
 181     rw [← cancel_mono (cokernel.π ((FC.filtration.F (s + ↑N) (k - 1)).arrow))]
 182     simp only [Category.assoc, zero_comp]
 183     exact h1
 184   have hle' : kernelSubobject ((FC.filtration.F s k).arrow ≫ FC.complex.d k (k - 1) ≫
 185       cokernel.π ((FC.filtration.F (s + ↑(↑N : ℕ)) (k - 1)).arrow)) ≤
 186     kernelSubobject ((FC.filtration.F s k).arrow ≫ FC.complex.d k (k - 1)) := hle
 187   suffices hsuff : FC.cycleSubobject s k ↑N ≤ FC.cycleSubobject s k ⊤ from
 188     le_trans (hX N) hsuff
 189   change imageSubobject
 190       ((kernelSubobject ((FC.filtration.F s k).arrow ≫ FC.complex.d k (k - 1) ≫
 191         cokernel.π ((FC.filtration.F (s + ↑N) (k - 1)).arrow))).arrow ≫
 192         cokernel.π (Subobject.ofLE (FC.filtration.F (s + 1) k) (FC.filtration.F s k) (FC.filtration.mono s k))) ≤
 193     imageSubobject
 194       ((kernelSubobject ((FC.filtration.F s k).arrow ≫ FC.complex.d k (k - 1))).arrow ≫
 195         cokernel.π (Subobject.ofLE (FC.filtration.F (s + 1) k) (FC.filtration.F s k) (FC.filtration.mono s k)))
 196   have heq : (kernelSubobject ((FC.filtration.F s k).arrow ≫ FC.complex.d k (k - 1) ≫
 197       cokernel.π ((FC.filtration.F (s + ↑N) (k - 1)).arrow))).arrow ≫
 198       cokernel.π (Subobject.ofLE (FC.filtration.F (s + 1) k) (FC.filtration.F s k) (FC.filtration.mono s k)) =
 199     Subobject.ofLE _ _ hle ≫ (kernelSubobject ((FC.filtration.F s k).arrow ≫ FC.complex.d k (k - 1))).arrow ≫
 200       cokernel.π (Subobject.ofLE (FC.filtration.F (s + 1) k) (FC.filtration.F s k) (FC.filtration.mono s k)) := by
 201     conv_rhs => rw [← Category.assoc]
 202     congr 1
 203     exact (Subobject.ofLE_arrow hle).symm
 204   rw [heq]
 205   exact imageSubobject_comp_le _ _

 207 /-- The B_top_least endpoint law under the corresponding filtration bound. -/
 208 theorem boundarySubobject_top_least (FC : FilteredComplex C)
 209     (bnd : FC.filtration.IsBoundedBelow) (s k : ℤ)
 210     (X : Subobject (FC.filtration.associatedGraded s k))
 211     (hX : ∀ i : ℕ, FC.boundarySubobject s k ↑i ≤ X) :
 212     FC.boundarySubobject s k ⊤ ≤ X := by
 213   obtain ⟨N, hN⟩ : ∃ N : ℕ, FC.filtration.F (s - ↑N + 1) (k + 1) = ⊤ := by
 214     use (s + 1 - bnd.lo (k + 1)).toNat
 215     apply bnd.boundedBelow; omega
 216   haveI : IsIso (FC.filtration.F (s - ↑N + 1) (k + 1)).arrow := by
 217     rw [Subobject.isIso_arrow_iff_eq_top]; exact hN
 218   have himgD : imageSubobject ((FC.filtration.F (s - ↑N + 1) (k + 1)).arrow ≫ FC.dToK k) =
 219       imageSubobject (FC.dToK k) :=
 220     imageSubobject_iso_comp _ _
 221   suffices hsuff : FC.boundarySubobject s k ⊤ ≤ FC.boundarySubobject s k ↑N from
 222     le_trans hsuff (hX N)
 223   change imageSubobject (Subobject.ofLE (imageSubobject (FC.dToK k) ⊓ FC.filtration.F s k)
 224         (FC.filtration.F s k) inf_le_right ≫
 225         cokernel.π (Subobject.ofLE (FC.filtration.F (s + 1) k) (FC.filtration.F s k) (FC.filtration.mono s k))) ≤
 226     imageSubobject (Subobject.ofLE
 227         (imageSubobject ((FC.filtration.F (s - ↑N + 1) (k + 1)).arrow ≫ FC.dToK k) ⊓ FC.filtration.F s k)
 228         (FC.filtration.F s k) inf_le_right ≫
 229         cokernel.π (Subobject.ofLE (FC.filtration.F (s + 1) k) (FC.filtration.F s k) (FC.filtration.mono s k)))
 230   rw [himgD]

 232 /-! ### Basic quotient-page consequences
 233 
 234 These are the canonical-page counterparts of the elementary `SSData` laws in
 235 the historical spectral-sequence API.  They use the existing cycle/boundary
 236 subobjects and `pageObj`; no second page-data structure is introduced. -/
 237 
 238 /-- Every boundary at the bottom page is contained in every cycle subobject. -/
 239 theorem boundarySubobject_bot_le_cycle (FC : FilteredComplex C) (s k : ℤ)
 240     (r : WithTop ℕ) :
 241     FC.boundarySubobject s k ⊥ ≤ FC.cycleSubobject s k r := by
 242   exact le_trans (boundarySubobject_monotone FC s k bot_le)
 243     (FC.B_le_Z_aux s k r)

 245 /-- If the boundary and cycle subobjects coincide, the quotient page vanishes. -/
 246 theorem pageObj_isZero_of_eq (FC : FilteredComplex C) (s k : ℤ)
 247     (r : WithTop ℕ) (h : FC.boundarySubobject s k r = FC.cycleSubobject s k r) :
 248     IsZero (FC.pageObj s k r) := by
 249   unfold pageObj
 250   have hi : IsIso (Subobject.ofLE (FC.boundarySubobject s k r)
 251       (FC.cycleSubobject s k r) (FC.B_le_Z_aux s k r)) := by
 252     rw [← Subobject.isoOfEq_hom _ _ h]
 253     infer_instance
 254   exact isZero_cokernel_of_epi _
 255 
 256 /-- A zero quotient page forces its boundary and cycle subobjects to coincide. -/
 257 theorem eq_of_pageObj_isZero (FC : FilteredComplex C) (s k : ℤ)
 258     (r : WithTop ℕ) (h : IsZero (FC.pageObj s k r)) :
 259     FC.boundarySubobject s k r = FC.cycleSubobject s k r := by
 260   unfold pageObj at h
 261   have hepi : Epi (Subobject.ofLE (FC.boundarySubobject s k r)
 262       (FC.cycleSubobject s k r) (FC.B_le_Z_aux s k r)) := by
 263     rwa [Preadditive.epi_iff_isZero_cokernel]
 264   haveI : IsIso (Subobject.ofLE (FC.boundarySubobject s k r)
 265       (FC.cycleSubobject s k r) (FC.B_le_Z_aux s k r)) :=
 266     isIso_of_mono_of_epi _
 267   apply le_antisymm (FC.B_le_Z_aux s k r)
 268   exact Subobject.le_of_comm
 269     (inv (Subobject.ofLE (FC.boundarySubobject s k r)
 270       (FC.cycleSubobject s k r) (FC.B_le_Z_aux s k r)))
 271     (by simp [Subobject.ofLE_arrow])
 272 
 273 /-- The canonical page is zero exactly when its boundary and cycle subobjects agree. -/
 274 theorem pageObj_isZero_iff (FC : FilteredComplex C) (s k : ℤ) (r : WithTop ℕ) :
 275     IsZero (FC.pageObj s k r) ↔
 276       FC.boundarySubobject s k r = FC.cycleSubobject s k r := by
 277   constructor
 278   · exact eq_of_pageObj_isZero FC s k r
 279   · exact pageObj_isZero_of_eq FC s k r

 281 /-! ### Quotient maps for nested subobjects
 282 
 283 These constructions are the categorical quotient tools used by the historical
 284 page-homology argument. They are stated independently of any spectral-sequence
 285 record and therefore remain reusable with the canonical `pageObj` quotient. -/
 286 
 287 /-- The quotient map induced by `P ≤ Q ≤ R`: `R/P ⟶ R/Q`. -/
 288 noncomputable def Subobject.cokernelDescOfLE {V : C}
 289     (P Q R : Subobject V) (hPQ : P ≤ Q) (hQR : Q ≤ R)
 290     (hPR : P ≤ R := le_trans hPQ hQR) :
 291     cokernel (Subobject.ofLE P R hPR) ⟶ cokernel (Subobject.ofLE Q R hQR) :=
 292   Subobject.cokernelDesc_ofLE P Q R hPQ hQR hPR
 293 
 294 /-- The map induced by `P ≤ Q ≤ R`: `Q/P ⟶ R/P`. -/
 295 noncomputable def Subobject.cokernelMapOfLE {V : C}
 296     (P Q R : Subobject V) (hPQ : P ≤ Q) (hQR : Q ≤ R)
 297     (hPR : P ≤ R := le_trans hPQ hQR) :
 298     cokernel (Subobject.ofLE P Q hPQ) ⟶ cokernel (Subobject.ofLE P R hPR) :=
 299   Subobject.cokernelMap_ofLE P Q R hPQ hQR hPR
 300 
 301 /-- The third-isomorphism identification `(R/P)/(Q/P) ≅ R/Q`. -/
 302 noncomputable def Subobject.thirdQuotientIso {V : C}
 303     (P Q R : Subobject V) (hPQ : P ≤ Q) (hQR : Q ≤ R)
 304     (hPR : P ≤ R := le_trans hPQ hQR) :
 305     cokernel (Subobject.cokernelMapOfLE P Q R hPQ hQR hPR) ≅
 306       cokernel (Subobject.ofLE Q R hQR) :=
 307   Subobject.thirdIso P Q R hPQ hQR hPR

 309 /-- A generalized element in a finite page boundary has a filtered source
 310 representative whose differential lands at the target filtration level. -/
 311 theorem lift_boundary (FC : FilteredComplex C) (s k : ℤ) (n : ℕ)
 312     {T : C} [Projective T]
 313     {v : T ⟶ FC.filtration.associatedGraded s k}
 314     (hv : (FC.boundarySubobject s k (↑n)).Factors v) :
 315     ∃ (a : T ⟶ Subobject.underlying.obj (FC.filtration.F (s - ↑n + 1) (k + 1)))
 316       (b : T ⟶ Subobject.underlying.obj (FC.filtration.F s k)),
 317       a ≫ (FC.filtration.F (s - ↑n + 1) (k + 1)).arrow ≫ FC.dToK k =
 318         b ≫ (FC.filtration.F s k).arrow ∧
 319       b ≫ FC.filtration.toAssociatedGraded s k = v := by
 320   let h := (FC.filtration.F (s - ↑n + 1) (k + 1)).arrow ≫ FC.dToK k
 321   let J := imageSubobject h
 322   let I := J ⊓ FC.filtration.F s k
 323   let i := Subobject.ofLE I (FC.filtration.F s k) inf_le_right
 324   let j := Subobject.ofLE I J inf_le_left
 325   let g := i ≫ FC.filtration.toAssociatedGraded s k
 326   have hfac : (imageSubobject g).Factors v := by
 327     simpa [FilteredComplex.boundarySubobject, h, J, I, i, g] using hv
 328   let vB := Subobject.factorThru (imageSubobject g) v hfac
 329   let bI := Projective.factorThru vB (factorThruImageSubobject g)
 330   let bJ := bI ≫ j
 331   let a := Projective.factorThru bJ (factorThruImageSubobject h)
 332   let b := bI ≫ i
 333   refine ⟨a, b, ?_, ?_⟩
 334   · calc
 335       a ≫ (FC.filtration.F (s - ↑n + 1) (k + 1)).arrow ≫ FC.dToK k =
 336           a ≫ h := by rfl
 337       _ = (a ≫ factorThruImageSubobject h) ≫ J.arrow := by
 338         rw [Category.assoc, imageSubobject_arrow_comp]
 339       _ = bJ ≫ J.arrow := by rw [Projective.factorThru_comp]
 340       _ = bI ≫ I.arrow := by
 341         simp only [bJ, j, Category.assoc, Subobject.ofLE_arrow]
 342       _ = b ≫ (FC.filtration.F s k).arrow := by
 343         simp only [b, i, Category.assoc, Subobject.ofLE_arrow]
 344   · calc
 345       b ≫ FC.filtration.toAssociatedGraded s k = bI ≫ g := by
 346         simp only [b, g, Category.assoc]
 347       _ = (bI ≫ factorThruImageSubobject g) ≫ (imageSubobject g).arrow := by
 348         rw [Category.assoc, imageSubobject_arrow_comp]
 349       _ = vB ≫ (imageSubobject g).arrow := by
 350         rw [Projective.factorThru_comp]
 351       _ = v := Subobject.factorThru_arrow _ _ _

 353 /-- A boundary on the differential's target page can be represented by a
 354 source in filtration `s + 1` whose differential lies in filtration `s + n`. -/
 355 theorem lift_boundary_at_differential (FC : FilteredComplex C)
