  11 namespace KIP126.Core.SpectralSequence.BoundedExtension

  13 open CategoryTheory CategoryTheory.Limits
  14 
  15 universe u v
  16 
  17 variable {C : Type u} [CategoryTheory.Category.{v} C] [Abelian C]
  18 
  19 @[simp]
  20 lemma twoTermObj_one (X₁ X₂ : C) : twoTermObj X₁ X₂ 1 = X₁ := by
  21   simp [twoTermObj]
  22 
  23 @[simp]
  24 lemma twoTermObj_zero (X₁ X₂ : C) : twoTermObj X₁ X₂ 0 = X₂ := by
  25   simp [twoTermObj]
  26 
  27 lemma twoTermObj_other (X₁ X₂ : C) (k : ℤ) (h₁ : k ≠ 1) (h₀ : k ≠ 0) :
  28     twoTermObj X₁ X₂ k = ⊥_ C := by
  29   simp [twoTermObj, h₁, h₀]
  30 
  31 @[simp]
  32 lemma twoTermDiff_sq (X₁ X₂ : C) (f : X₁ ⟶ X₂) (k : ℤ) :
  33     twoTermDiff X₁ X₂ f k ≫ twoTermDiff X₁ X₂ f (k - 1) = 0 := by
  34   simp only [twoTermDiff]
  35   by_cases h : k = 1
  36   · subst k
  37     change _ ≫ (0 : twoTermObj X₁ X₂ 0 ⟶ twoTermObj X₁ X₂ (-1)) = 0
  38     simp only [comp_zero]
  39   · simp [h]
  40 
  41 @[simp]
  42 lemma twoTermFil_one {X₁ X₂ : C}
  43     (fil₁ : ℤ → Subobject X₁) (fil₂ : ℤ → Subobject X₂) (s : ℤ) :
  44     twoTermFil fil₁ fil₂ s 1 = fil₁ s := by
  45   simp [twoTermFil]
  46 
  47 @[simp]
  48 lemma twoTermFil_zero {X₁ X₂ : C}
  49     (fil₁ : ℤ → Subobject X₁) (fil₂ : ℤ → Subobject X₂) (s : ℤ) :
  50     twoTermFil fil₁ fil₂ s 0 = fil₂ s := by
  51   simp [twoTermFil]
  52 
  53 lemma twoTermDiff_sq_transport (X₁ X₂ : C) (f : X₁ ⟶ X₂)
  54     (a b c : ℤ) (hab : a - 1 = b) (hbc : b - 1 = c) :
  55     (twoTermDiff X₁ X₂ f a ≫ eqToHom (congrArg (twoTermObj X₁ X₂) hab)) ≫
  56         twoTermDiff X₁ X₂ f b ≫ eqToHom (congrArg (twoTermObj X₁ X₂) hbc) = 0 := by
  57   subst b
  58   subst c
  59   simpa only [eqToHom_refl, Category.comp_id, Category.assoc] using
  60     twoTermDiff_sq X₁ X₂ f a
