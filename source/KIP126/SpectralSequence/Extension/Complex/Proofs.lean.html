  13 open CategoryTheory CategoryTheory.Limits
  14 
  15 universe u v
  16 
  17 variable {C : Type u} [CategoryTheory.Category.{v} C] [Abelian C]
  18 
  19 @[simp]
  20 lemma twoTermComplex_X (X₁ X₂ : C) (f : X₁ ⟶ X₂) (k : ℤ) :
  21     (twoTermComplex X₁ X₂ f).X k = twoTermObj X₁ X₂ k := by
  22   simp [twoTermComplex]
  23 
  24 @[simp]
  25 lemma twoTermComplex_d (X₁ X₂ : C) (f : X₁ ⟶ X₂) (k : ℤ) :
  26     (twoTermComplex X₁ X₂ f).d k (k - 1) = twoTermDiff X₁ X₂ f k := by
  27   unfold twoTermComplex
  28   change ChainComplex.of.d (twoTermObj X₁ X₂)
  29       (fun j => twoTermDiff X₁ X₂ f (j + 1) ≫
  30         eqToHom (congrArg (twoTermObj X₁ X₂) (by omega : (j + 1) - 1 = j)))
  31       k (k - 1) = twoTermDiff X₁ X₂ f k
  32   unfold ChainComplex.of.d
  33   rw [dif_pos (show k = k - 1 + 1 by omega)]
  34   simp
  35 
  36 @[simp]
  37 lemma twoTermComplex_d_one (X₁ X₂ : C) (f : X₁ ⟶ X₂) :
  38     (twoTermComplex X₁ X₂ f).d 1 0 = f := by
  39   simpa [twoTermDiff, twoTermObj] using twoTermComplex_d X₁ X₂ f 1
