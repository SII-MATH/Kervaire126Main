   8 /-!
   9 # Representatives for filtered-complex differentials
  10 
  11 This file connects the associated-graded differential of a filtered complex
  12 with filtration-level generalized-element representatives. The representative
  13 equations are adapted from
  14 `KIPInfra/SpectralSequence/FilteredComplex.lean` at commit `65de864`.
  15 Unlike that source, this module uses the project's existing filtered and
  16 associated-graded functors and does not introduce a spectral-sequence engine.
  17 
  18 The leading-term projection used by these representative calculations is the
  19 existing `Filtration.toAssociatedGraded`; no second spectral-sequence
  20 or associated-graded interface is introduced here.
  21 -/

  31 /-- The associated-graded differential has value `b` on a generalized
  32 element `a` exactly when `a` and `b` admit filtration-level representatives
  33 related by the original differential. Projectivity is the categorical
  34 replacement for choosing a lift through the leading-term epimorphism. -/
  35 theorem FilteredComplex.associatedGradedDifferential_iff_representative
  36     (FC : FilteredComplex C) (s k : ℤ) {T : C} [Projective T]
  37     (a : T ⟶ FC.filtration.associatedGraded s k)
  38     (b : T ⟶ FC.filtration.associatedGraded s (k - 1)) :
  39     a ≫ FC.associatedGradedDifferential s k = b ↔
  40       ∃ x : T ⟶ Subobject.underlying.obj (FC.filtration.F s k),
  41         x ≫ FC.filtration.toAssociatedGraded s k = a ∧
  42         ∃ dx : T ⟶ Subobject.underlying.obj (FC.filtration.F s (k - 1)),
  43           dx ≫ (FC.filtration.F s (k - 1)).arrow =
  44             x ≫ (FC.filtration.F s k).arrow ≫ FC.complex.d k (k - 1) ∧
  45           dx ≫ FC.filtration.toAssociatedGraded s (k - 1) = b := by
  46   constructor
  47   · intro hab
  48     haveI : Epi (FC.filtration.toAssociatedGraded s k) := by
  49       unfold Filtration.toAssociatedGraded Filtration.associatedGraded
  50       infer_instance
  51     let x := Projective.factorThru a (FC.filtration.toAssociatedGraded s k)
  52     have hx : x ≫ FC.filtration.toAssociatedGraded s k = a :=
  53       Projective.factorThru_comp _ _
  54     let dx := x ≫ (FC.differential_preserves s k).choose
  55     refine ⟨x, hx, dx, ?_, ?_⟩
  56     · rw [Category.assoc, (FC.differential_preserves s k).choose_spec]
  57     · calc
  58         dx ≫ FC.filtration.toAssociatedGraded s (k - 1) =
  59             x ≫ ((FC.differential_preserves s k).choose ≫
  60               FC.filtration.toAssociatedGraded s (k - 1)) := by
  61                 simp only [dx, Category.assoc]
  62         _ = x ≫ (FC.filtration.toAssociatedGraded s k ≫
  63               FC.associatedGradedDifferential s k) := by
  64                 rw [FC.toAssociatedGraded_comp_associatedGradedDifferential]
  65         _ = a ≫ FC.associatedGradedDifferential s k := by
  66                 simpa only [Category.assoc] using
  67                   congrArg (fun q => q ≫ FC.associatedGradedDifferential s k) hx
  68         _ = b := hab
  69   · rintro ⟨x, hx, dx, hdx, hdxLead⟩
  70     have hdx_eq : dx = x ≫ (FC.differential_preserves s k).choose := by
  71       apply (cancel_mono (FC.filtration.F s (k - 1)).arrow).mp
  72       rw [hdx, Category.assoc, (FC.differential_preserves s k).choose_spec]
  73     calc
  74       a ≫ FC.associatedGradedDifferential s k =
  75           (x ≫ FC.filtration.toAssociatedGraded s k) ≫
  76             FC.associatedGradedDifferential s k := by rw [hx]
  77       _ = x ≫ ((FC.differential_preserves s k).choose ≫
  78           FC.filtration.toAssociatedGraded s (k - 1)) := by
  79             rw [Category.assoc,
  80               FC.toAssociatedGraded_comp_associatedGradedDifferential]
  81       _ = dx ≫ FC.filtration.toAssociatedGraded s (k - 1) := by
  82             simpa only [Category.assoc] using
  83               congrArg (fun q => q ≫ FC.filtration.toAssociatedGraded s (k - 1))
  84                 hdx_eq.symm
  85       _ = b := hdxLead

  87 /-- An associated-graded class is a cycle exactly when it has a
  88 filtration-level representative whose differential lifts to the next
  89 filtration level. -/
  90 theorem FilteredComplex.associatedGradedDifferential_eq_zero_iff_representative
  91     (FC : FilteredComplex C) (s k : ℤ) {T : C} [Projective T]
  92     (a : T ⟶ FC.filtration.associatedGraded s k) :
  93     a ≫ FC.associatedGradedDifferential s k = 0 ↔
  94       ∃ x : T ⟶ Subobject.underlying.obj (FC.filtration.F s k),
  95         x ≫ FC.filtration.toAssociatedGraded s k = a ∧
  96         ∃ dx : T ⟶ Subobject.underlying.obj (FC.filtration.F (s + 1) (k - 1)),
  97           dx ≫ (FC.filtration.F (s + 1) (k - 1)).arrow =
  98             x ≫ (FC.filtration.F s k).arrow ≫ FC.complex.d k (k - 1) := by
  99   constructor
 100   · intro ha
 101     obtain ⟨x, hx, dx, hdx, hdxLead⟩ :=
 102       (FC.associatedGradedDifferential_iff_representative s k a 0).mp ha
 103     obtain ⟨dx', hdx'⟩ :=
 104       (FC.filtration.comp_toAssociatedGraded_eq_zero_iff_lifts s (k - 1) dx).mp hdxLead
 105     refine ⟨x, hx, dx', ?_⟩
 106     calc
 107       dx' ≫ (FC.filtration.F (s + 1) (k - 1)).arrow =
 108           (dx' ≫ Subobject.ofLE (FC.filtration.F (s + 1) (k - 1))
 109             (FC.filtration.F s (k - 1)) (FC.filtration.mono s (k - 1))) ≫
 110             (FC.filtration.F s (k - 1)).arrow := by
 111               rw [Category.assoc, Subobject.ofLE_arrow]
 112       _ = dx ≫ (FC.filtration.F s (k - 1)).arrow := by rw [hdx']
 113       _ = x ≫ (FC.filtration.F s k).arrow ≫ FC.complex.d k (k - 1) := hdx
 114   · rintro ⟨x, hx, dx, hdx⟩
 115     apply (FC.associatedGradedDifferential_iff_representative s k a 0).mpr
 116     let inclusion := Subobject.ofLE (FC.filtration.F (s + 1) (k - 1))
 117       (FC.filtration.F s (k - 1)) (FC.filtration.mono s (k - 1))
 118     refine ⟨x, hx, dx ≫ inclusion, ?_, ?_⟩
 119     · rw [Category.assoc, Subobject.ofLE_arrow, hdx]
 120     · have hinclusion :
 121           inclusion ≫ FC.filtration.toAssociatedGraded s (k - 1) = 0 := by
 122         dsimp only [inclusion, Filtration.toAssociatedGraded]
 123         exact cokernel.condition _
 124       rw [Category.assoc, hinclusion, comp_zero]

 126 /-- A filtered chain map carries a representative of a differential from any
 127 source filtration level to any target filtration level. Specializing
 128 `targetLevel = sourceLevel + r` gives the cycle lift; specializing
 129 `sourceLevel = targetLevel - r + 1` gives the boundary lift. -/
 130 theorem FilteredComplex.Morphism.differentialRepresentative_naturality
 131     {FC GD : FilteredComplex C} (f : FC ⟶ GD)
 132     (sourceLevel targetLevel k : ℤ) {T : C}
 133     (x : T ⟶ Subobject.underlying.obj (FC.filtration.F sourceLevel k))
 134     (dx : T ⟶ Subobject.underlying.obj
 135       (FC.filtration.F targetLevel (k - 1)))
 136     (hdx : dx ≫ (FC.filtration.F targetLevel (k - 1)).arrow =
 137       x ≫ (FC.filtration.F sourceLevel k).arrow ≫ FC.complex.d k (k - 1)) :
 138     (dx ≫ (f.preserves targetLevel (k - 1)).choose) ≫
 139         (GD.filtration.F targetLevel (k - 1)).arrow =
 140       (x ≫ (f.preserves sourceLevel k).choose) ≫
 141         (GD.filtration.F sourceLevel k).arrow ≫ GD.complex.d k (k - 1) := by
 142   calc
 143     (dx ≫ (f.preserves targetLevel (k - 1)).choose) ≫
 144           (GD.filtration.F targetLevel (k - 1)).arrow =
 145         dx ≫ ((f.preserves targetLevel (k - 1)).choose ≫
 146           (GD.filtration.F targetLevel (k - 1)).arrow) := by
 147       simp only [Category.assoc]
 148     _ = dx ≫ (FC.filtration.F targetLevel (k - 1)).arrow ≫
 149         f.map.f (k - 1) := by
 150       rw [(f.preserves targetLevel (k - 1)).choose_spec]
 151     _ = x ≫ (FC.filtration.F sourceLevel k).arrow ≫
 152         FC.complex.d k (k - 1) ≫ f.map.f (k - 1) := by
 153       simpa only [Category.assoc] using
 154         congrArg (fun q => q ≫ f.map.f (k - 1)) hdx
 155     _ = x ≫ (FC.filtration.F sourceLevel k).arrow ≫
 156         (f.map.f k ≫ GD.complex.d k (k - 1)) := by
 157       simpa only [Category.assoc] using
 158         congrArg (fun q => x ≫ (FC.filtration.F sourceLevel k).arrow ≫ q)
 159           (f.map.comm k (k - 1)).symm
 160     _ = x ≫ ((f.preserves sourceLevel k).choose ≫
 161         (GD.filtration.F sourceLevel k).arrow) ≫
 162           GD.complex.d k (k - 1) := by
 163       simpa only [Category.assoc] using
 164         congrArg (fun q => x ≫ q ≫ GD.complex.d k (k - 1))
 165           (f.preserves sourceLevel k).choose_spec.symm
 166     _ = (x ≫ (f.preserves sourceLevel k).choose) ≫
 167         (GD.filtration.F sourceLevel k).arrow ≫
 168           GD.complex.d k (k - 1) := by simp only [Category.assoc]
