  10 /-!
  11 # Degreewise completion of a decreasing filtration
  12 
  13 For a decreasing filtration `F` of a graded object, this file constructs the
  14 canonical inverse system of quotient objects
  15 `Aᵢ / Fˢ Aᵢ`.  The universal-property predicate `CompletionWitness` records
  16 exactly when the canonical cone from `Aᵢ` is limiting. Completeness is an
  17 explicit witness required from the caller; this file does not assert it for
  18 an arbitrary filtration.
  19 -/
  20 
  21 namespace KIP126.Core.SpectralSequence
  22 
  23 open CategoryTheory CategoryTheory.Limits
  24 
  25 universe u v w
  26 
  27 variable {C : Type u} [Category.{v} C] [Abelian C]
  28 variable {ι : Type w} {A : CategoryTheory.GradedObject ι C}
  29 
  30 namespace Filtration
  31 
  32 /-- The quotient `Aᵢ / Fˢ Aᵢ` at one filtration and grading degree. -/
  33 public noncomputable def quotientAt (F : Filtration A) (s : ℤ) (i : ι) : C :=
  34   cokernel (F.F s i).arrow
  35 
  36 /-- The canonical projection `Aᵢ ⟶ Aᵢ / Fˢ Aᵢ`. -/
  37 public noncomputable def quotientProjection (F : Filtration A) (s : ℤ) (i : ι) :
  38     A i ⟶ F.quotientAt s i :=
  39   cokernel.π (F.F s i).arrow
  40 
  41 /-- If `t ≤ s`, the quotient by `Fˢ` maps to the quotient by `Fᵗ`. -/
  42 public noncomputable def quotientTransition (F : Filtration A) {t s : ℤ} (h : t ≤ s) (i : ι) :
  43     F.quotientAt s i ⟶ F.quotientAt t i :=
  44   cokernel.desc (F.F s i).arrow (F.quotientProjection t i) (by
  45     change (F.F s i).arrow ≫ cokernel.π (F.F t i).arrow = 0
  46     rw [← F.inclusion_arrow h i, Category.assoc, cokernel.condition, comp_zero])
  47 
  48 /-- The quotient transition commutes with the quotient projections. -/
  49 @[reassoc]
  50 public lemma quotientProjection_transition (F : Filtration A) {t s : ℤ} (h : t ≤ s) (i : ι) :
  51     F.quotientProjection s i ≫ F.quotientTransition h i = F.quotientProjection t i := by
  52   exact cokernel.π_desc _ _ _
  53 
  54 /-- Quotient transitions compose in the evident order. -/
  55 @[reassoc]
  56 public lemma quotientTransition_comp (F : Filtration A) {r s t : ℤ}
  57     (hrs : r ≤ s) (hst : s ≤ t) (i : ι) :
  58     F.quotientTransition hst i ≫ F.quotientTransition hrs i =
  59       F.quotientTransition (hrs.trans hst) i := by
  60   letI : Epi (F.quotientProjection t i) := by
  61     change Epi (cokernel.π (F.F t i).arrow)
  62     infer_instance
  63   apply (cancel_epi (F.quotientProjection t i)).mp
  64   calc
  65     F.quotientProjection t i ≫ F.quotientTransition hst i ≫
  66         F.quotientTransition hrs i =
  67       (F.quotientProjection t i ≫ F.quotientTransition hst i) ≫
  68         F.quotientTransition hrs i := (Category.assoc _ _ _).symm
  69     _ = F.quotientProjection s i ≫ F.quotientTransition hrs i := by
  70         rw [F.quotientProjection_transition]
  71     _ = F.quotientProjection r i := F.quotientProjection_transition hrs i
  72     _ = F.quotientProjection t i ≫ F.quotientTransition (hrs.trans hst) i :=
  73       (F.quotientProjection_transition (hrs.trans hst) i).symm
  74 
  75 /-- The identity quotient transition is the identity. -/
  76 @[simp]
  77 public lemma quotientTransition_id (F : Filtration A) (s : ℤ) (i : ι) :
  78     F.quotientTransition (le_rfl : s ≤ s) i = 𝟙 _ := by
  79   letI : Epi (F.quotientProjection s i) := by
  80     change Epi (cokernel.π (F.F s i).arrow)
  81     infer_instance
  82   apply (cancel_epi (F.quotientProjection s i)).mp
  83   rw [F.quotientProjection_transition, Category.comp_id]
  84 
  85 /-- The canonical inverse system `s ↦ Aᵢ / Fˢ Aᵢ`.
  86 
  87 It is indexed by `OrderDual ℤ`: a morphism `s ⟶ t` has `t ≤ s`, and hence
  88 is precisely the quotient transition from level `s` to level `t`. -/
  89 public noncomputable def quotientTower (F : Filtration A) (i : ι) :
  90     (OrderDual ℤ) ⥤ C where
  91   obj s := F.quotientAt s i
  92   map f := F.quotientTransition f.le i
  93   map_id s := F.quotientTransition_id s i
  94   map_comp f g := by
  95     simpa only [Functor.comp_map] using (F.quotientTransition_comp g.le f.le i).symm
  96 
  97 /-- The canonical cone from `Aᵢ` to its tower of filtration quotients. -/
  98 public noncomputable def quotientTowerCone (F : Filtration A) (i : ι) :
  99     Cone (F.quotientTower i) where
 100   pt := A i
 101   π :=
 102     { app := fun s => F.quotientProjection s i
 103       naturality := by
 104         intro s t f
 105         dsimp [quotientTower]
 106         calc
 107           𝟙 (A i) ≫ F.quotientProjection t i = F.quotientProjection t i := Category.id_comp _
 108           _ = F.quotientProjection s i ≫ F.quotientTransition f.le i :=
 109             (F.quotientProjection_transition f.le i).symm }
 110 
 111 /-- A degreewise completeness witness: the canonical cone from `Aᵢ` to the
 112 quotient tower is a limit cone.  This is the categorical statement
 113 `Aᵢ ≅ lim_s Aᵢ/FˢAᵢ`, retaining the canonical comparison maps. -/
 114 public structure CompletionWitness (F : Filtration A) (i : ι) where
 115   isLimit : IsLimit (F.quotientTowerCone i)
 116 
 117 end Filtration
