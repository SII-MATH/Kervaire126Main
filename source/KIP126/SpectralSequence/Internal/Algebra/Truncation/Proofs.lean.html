  20 /-- A truncation is bounded above by its quotient level. -/
  21 def quotientFiltration_boundedAbove (fil : Filtration A) (q : ℤ) :
  22     (fil.quotientFiltration q).IsBoundedAbove := by
  23   refine { hi := fun _ => q, boundedAbove := ?_ }
  24   intro i s hs
  25   change imageSubobject ((fil.F s i).arrow ≫
  26     cokernel.π (fil.F (q) i).arrow) = ⊥
  27   have hmono : fil.F s i ≤ fil.F (q) i := by
  28     exact fil.le_of_le (by omega) i
  29   have hzero : (fil.F s i).arrow ≫ cokernel.π (fil.F (q) i).arrow = 0 := by
  30     rw [show (fil.F s i).arrow =
  31       Subobject.ofLE _ _ hmono ≫ (fil.F (q) i).arrow
  32       from by rw [Subobject.ofLE_arrow]]
  33     rw [Category.assoc, cokernel.condition, comp_zero]
  34   simp only [hzero, imageSubobject_zero]
  35 
  36 /-- If the source filtration is bounded below, its truncation is bounded. -/
  37 noncomputable def quotientFiltration_isBounded
  38     {fil : Filtration A} (hbb : fil.IsBoundedBelow) (q : ℤ) :
  39     (fil.quotientFiltration q).IsBounded := by
  40   refine
  41     { lo := fun i => min (hbb.lo i) (q)
  42       hi := fun _ => q
  43       lo_le_hi := fun _ => min_le_right _ _
  44       boundedBelow := ?_
  45       boundedAbove := ?_ }
  46   · intro i s hs
  47     have hs' : s ≤ hbb.lo i := le_trans hs (min_le_left _ _)
  48     have htop : fil.F s i = ⊤ := hbb.boundedBelow i s hs'
  49     have hiso : IsIso (fil.F s i).arrow :=
  50       (Subobject.isIso_arrow_iff_eq_top _).mpr htop
  51     change imageSubobject ((fil.F s i).arrow ≫
  52       cokernel.π (fil.F (q) i).arrow) = ⊤
  53     rw [imageSubobject_iso_comp]
  54     have h1 : Epi (image.ι (cokernel.π (fil.F (q) i).arrow)) :=
  55       epi_of_epi_fac (image.fac _)
  56     have h2 : IsIso (image.ι (cokernel.π (fil.F (q) i).arrow)) :=
  57       isIso_of_mono_of_epi _
  58     rw [← Subobject.isIso_arrow_iff_eq_top]
  59     have harr : (imageSubobject (cokernel.π (fil.F (q) i).arrow)).arrow =
  60         (imageSubobjectIso _).hom ≫ image.ι _ := by
  61       simp [imageSubobject_arrow]
  62     rw [harr]
  63     infer_instance
  64   · exact (fil.quotientFiltration_boundedAbove q).boundedAbove
