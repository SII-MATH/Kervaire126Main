   8 /-!
   9 # Filtrations underlying spectral sequences
  10 
  11 The public filtration record and associated-graded operations are defined once
  12 at this foundation layer. Filtered complexes, convergence, completion and
  13 truncation use this same record. Other algebra constructions remain private.
  14 Filtrations require only a category; quotients additionally require an abelian
  15 category.
  16 -/
  17

  26 /-- A decreasing filtration with the historical `mono` field name. -/
  27 public structure Filtration {ω : Type w} (A : ω → C) where
  28   /-- Filtration subobject at level `s` and grading `k`. -/
  29   F : ℤ → (k : ω) → Subobject (A k)
  30   /-- Successive levels form a decreasing family. -/
  31   mono : ∀ (s : ℤ) (k : ω), F (s + 1) k ≤ F s k
  32 
  33 section
  34 
  35 variable [Abelian C]
  36 
  37 /-- Associated graded piece `F^s A^k / F^(s+1) A^k`. -/
  38 public noncomputable def Filtration.associatedGraded
  39     {ω : Type w} {A : ω → C} (F : Filtration A) (s : ℤ) (k : ω) : C :=
  40   cokernel (Subobject.ofLE (F.F (s + 1) k) (F.F s k) (F.mono s k))
  41 
  42 /-- Projection from a filtration level to its associated graded piece. -/
  43 public noncomputable def Filtration.toAssociatedGraded
  44     {ω : Type w} {A : ω → C} (F : Filtration A) (s : ℤ) (k : ω) :
  45     Subobject.underlying.obj (F.F s k) ⟶ F.associatedGraded s k :=
  46   cokernel.π (Subobject.ofLE (F.F (s + 1) k) (F.F s k) (F.mono s k))
  47 
  48 /-- Transport associated graded pieces along equality of their indices. -/
  49 public noncomputable def Filtration.transportGraded
  50     {ω' : Type w} {A : ω' → C} (F : Filtration A)
  51     {r₁ r₂ : ℤ × ω'} (h : r₁ = r₂) :
  52     F.associatedGraded r₁.1 r₁.2 ⟶ F.associatedGraded r₂.1 r₂.2 :=
  53   eqToHom (by rw [h])
  54 
  55 end
  56 
  57 /-- A filtration decreases between any two ordered integer levels. -/
  58 public theorem Filtration.le_of_le {ω : Type w} {A : ω → C}
  59     (F : Filtration A) {s t : ℤ} (h : s ≤ t) (k : ω) : F.F t k ≤ F.F s k := by
  60   induction t, h using Int.leInduction with
  61   | base => exact le_rfl
  62   | succ t _ ih => exact (F.mono t k).trans ih

  68 /-! ### Maps between arbitrary filtration levels
  69 
  70 The successor inequality in `Filtration.mono` is the compact datum from
  71 which all of the maps in the filtration diagram are obtained.  Keeping this
  72 construction here (rather than re-proving it in the filtered-complex layer)
  73 also fixes the variance convention once and for all: for `t ≤ s`, the
  74 decreasing filtration has an inclusion `F s ↪ F t`.
  75 -/
  76 
  77 /-- The canonical inclusion `Fˢ Aᵢ ⟶ Fᵗ Aᵢ` when `t ≤ s`. -/
  78 noncomputable def inclusion (F : Filtration A) {t s : ℤ} (h : t ≤ s) (i : ι) :
  79     Subobject.underlying.obj (F.F s i) ⟶ Subobject.underlying.obj (F.F t i) :=
  80   Subobject.ofLE (F.F s i) (F.F t i) (F.le_of_le h i)
  81 
  82 @[simp]
  83 lemma inclusion_arrow (F : Filtration A) {t s : ℤ} (h : t ≤ s) (i : ι) :
  84     F.inclusion h i ≫ (F.F t i).arrow = (F.F s i).arrow :=
  85   Subobject.ofLE_arrow _
  86 
  87 @[simp]
  88 lemma inclusion_refl (F : Filtration A) (s : ℤ) (i : ι) :
  89     F.inclusion (le_rfl : s ≤ s) i = 𝟙 _ := by
  90   simp [inclusion]
  91 
  92 @[reassoc (attr := simp)]
  93 lemma inclusion_comp (F : Filtration A) {r s t : ℤ}
  94     (hrs : r ≤ s) (hst : s ≤ t) (i : ι) :
  95     F.inclusion hst i ≫ F.inclusion hrs i = F.inclusion (hrs.trans hst) i := by
  96   simp [inclusion]
  97 
  98 /-- Naturality of a filtration subobject arrow under an equality of grading
  99 indices.  This is the transport lemma needed whenever a complex shape writes
 100 the predecessor as an expression such as `(k + 1) - 1`. -/
 101 lemma transport_arrow (F : Filtration A) (s : ℤ) {i j : ι} (h : i = j) :
 102     eqToHom (congrArg (fun k => Subobject.underlying.obj (F.F s k)) h) ≫
 103         (F.F s j).arrow =
 104       (F.F s i).arrow ≫ eqToHom (congrArg A h) := by
 105   subst h
 106   simp
 107 
 108 /-- The level inclusions are natural with respect to index transports. -/
 109 lemma inclusion_naturality (F : Filtration A) {t s : ℤ} (h : t ≤ s)
 110     {i j : ι} (hij : i = j) :
 111     F.inclusion h i ≫
 112         eqToHom (congrArg (fun k => Subobject.underlying.obj (F.F t k)) hij) =
 113       eqToHom (congrArg (fun k => Subobject.underlying.obj (F.F s k)) hij) ≫
 114         F.inclusion h j := by
 115   subst hij
 116   simp

 119 /-- All associated graded pieces, graded by filtration degree and the original
 120 grading index. -/
 121 noncomputable def associatedGradedObject [Abelian C] (F : Filtration A) :
 122     CategoryTheory.GradedObject (ℤ × ι) C :=
 123   fun si => F.associatedGraded si.1 si.2

 127 /-- A morphism of filtered graded objects is a graded map whose restriction to
 128 each filtration subobject factors through the corresponding target
 129 subobject. -/
 130 structure InternalFilteredMorphism {ι : Type w}
 131     {A B : CategoryTheory.GradedObject ι C} (F : Filtration A) (G : Filtration B) where
 132   /-- The underlying degree-preserving map. -/
 133   map : A ⟶ B
 134   /-- Preservation of every filtration level. -/
 135   preserves : ∀ (s : ℤ) (i : ι),
 136     ∃ φ : Subobject.underlying.obj (F.F s i) ⟶ Subobject.underlying.obj (G.F s i),
 137       φ ≫ (G.F s i).arrow = (F.F s i).arrow ≫ map i

 145 /-- The identity map of a filtered graded object. -/
 146 def id (F : Filtration A) : InternalFilteredMorphism F F where
 147   map := 𝟙 A
 148   preserves := fun s i => ⟨𝟙 _, by simp⟩
 149 
 150 /-- Composition of filtration-preserving graded maps. -/
 151 def comp (f : InternalFilteredMorphism F G) (g : InternalFilteredMorphism G H) :
 152     InternalFilteredMorphism F H where
 153   map := f.map ≫ g.map
 154   preserves := fun s i => by
 155     refine ⟨(f.preserves s i).choose ≫ (g.preserves s i).choose, ?_⟩
 156     change ((f.preserves s i).choose ≫ (g.preserves s i).choose) ≫
 157       (H.F s i).arrow = (F.F s i).arrow ≫ (f.map i ≫ g.map i)
 158     rw [Category.assoc, (g.preserves s i).choose_spec, ← Category.assoc,
 159       (f.preserves s i).choose_spec]
 160     simp only [Category.assoc]
 161 
 162 lemma id_preserves_eq (F : Filtration A) (s : ℤ) (i : ι) :
 163     ((InternalFilteredMorphism.id F).preserves s i).choose = 𝟙 _ := by
 164   apply (cancel_mono (F.F s i).arrow).mp
 165   rw [(InternalFilteredMorphism.id F).preserves s i |>.choose_spec]
 166   dsimp [InternalFilteredMorphism.id]
 167   simp
 168 
 169 lemma comp_preserves_eq (f : InternalFilteredMorphism F G) (g : InternalFilteredMorphism G H)
 170     (s : ℤ) (i : ι) :
 171     ((InternalFilteredMorphism.comp f g).preserves s i).choose =
 172       (f.preserves s i).choose ≫ (g.preserves s i).choose := by
 173   apply (cancel_mono (H.F s i).arrow).mp
 174   calc
 175     ((InternalFilteredMorphism.comp f g).preserves s i).choose ≫
 176           (H.F s i).arrow =
 177         (F.F s i).arrow ≫ (InternalFilteredMorphism.comp f g).map i :=
 178       (InternalFilteredMorphism.comp f g).preserves s i |>.choose_spec
 179     _ = ((F.F s i).arrow ≫ f.map i) ≫ g.map i := by
 180       simp only [InternalFilteredMorphism.comp,
 181         GradedObject.categoryOfGradedObjects_comp]
 182       simp only [Category.assoc]
 183     _ = ((f.preserves s i).choose ≫ (G.F s i).arrow) ≫ g.map i := by
 184       rw [(f.preserves s i).choose_spec]
 185     _ = (f.preserves s i).choose ≫
 186           ((G.F s i).arrow ≫ g.map i) := by simp only [Category.assoc]
 187     _ = (f.preserves s i).choose ≫
 188           ((g.preserves s i).choose ≫ (H.F s i).arrow) := by
 189       rw [(g.preserves s i).choose_spec]
 190     _ = ((f.preserves s i).choose ≫ (g.preserves s i).choose) ≫
 191           (H.F s i).arrow := by simp only [Category.assoc]
 192 
 193 /-- The map on associated graded pieces induced by a filtration-preserving
 194 morphism. -/
 195 noncomputable def associatedGradedMap [Abelian C] (f : InternalFilteredMorphism F G) (s : ℤ) (i : ι) :
 196     F.associatedGraded s i ⟶ G.associatedGraded s i :=
 197   cokernel.map
 198     (Subobject.ofLE (F.F (s + 1) i) (F.F s i) (F.mono s i))
 199     (Subobject.ofLE (G.F (s + 1) i) (G.F s i) (G.mono s i))
 200     (f.preserves (s + 1) i).choose
 201     (f.preserves s i).choose
 202     (by
 203       apply (cancel_mono ((G.F s i).arrow)).mp
 204       simp only [Category.assoc, Subobject.ofLE_arrow]
 205       rw [(f.preserves s i).choose_spec, (f.preserves (s + 1) i).choose_spec,
 206         ← Category.assoc, Subobject.ofLE_arrow])
 207 
 208 end InternalFilteredMorphism
