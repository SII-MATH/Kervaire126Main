  18 variable {C : Type u} [Category.{v} C] [Abelian C]
  19 variable {ω : Type w} [AddCommGroup ω] [DecidableEq ω]
  20 variable {ω' : Type w}
  21 variable {E₁ E₂ : SpectralSequence C ω}
  22 variable {A₁ A₂ : ω' → C} {F₁ : Filtration A₁} {F₂ : Filtration A₂}
  23 variable {conv₁ : Convergence E₁ A₁ F₁} {conv₂ : Convergence E₂ A₂ F₂}

  25 /-- The two-term filtered complex of the convergence morphism after quotient
  26 truncating both target filtrations at `s₀`. -/
  27 noncomputable def truncatedUnderlyingComplex
  28     (cm : ConvergenceMorphism conv₁ conv₂) (s₀ : ℤ) (t : ω') :
  29     FilteredComplex C :=
  30   underlyingComplex
  31     (fun k' => cm.truncatedAMap s₀ k')
  32     (fun s k' => cm.truncatedFiltrationCompat s₀ s k') t
  33
