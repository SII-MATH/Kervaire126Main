  21 /-! ### Completion of filtered objects -/
  22 
  23 noncomputable instance Filtration.hasLimitCompletionFunctor {ω : Type w} {A : ω → C}
  24     (fil : Filtration A) (hbb : fil.IsBoundedBelow) (k : ω) :
  25     HasLimit (fil.completionFunctor hbb k) := inferInstance
  26 
  27 lemma Filtration.completionProj'_factor {ω : Type w} {A : ω → C}
  28     (fil : Filtration A) (hbb : fil.IsBoundedBelow) (k : ω) (s : ℤ) :
  29     fil.completionProj' hbb k (s + 1) ≫ fil.truncationTransition (show s ≤ s + 1 by omega) k =
  30     fil.completionProj' hbb k s := by
  31   simp only [Filtration.completionProj', Filtration.completion,
  32     Filtration.completionFunctor]
  33   rw [Category.assoc, fil.truncationTransition_comp]
  34   have hw := limit.w (fil.completionFunctor hbb k)
  35     (homOfLE (show (s - hbb.lo k).toNat ≤ (s + 1 - hbb.lo k).toNat by omega)).op
  36   erw [← hw, Category.assoc, fil.truncationTransition_comp]
  37   all_goals simp only [Opposite.unop_op]
  38   · rfl
  39   · omega

  41 /-! ### Mittag-Leffler condition -/
  42 
  43 -- Keep the original instance parameters of this theorem.
  44 set_option linter.unusedSectionVars false in
  45 /-- Bounded-above filtrations satisfy Mittag-Leffler. -/
  46 theorem Filtration.IsBoundedAbove.toIsMittagLeffler {ω : Type w} {A : ω → C}
  47     {fil : Filtration A} (hba : fil.IsBoundedAbove) : fil.IsMittagLeffler :=
  48   hba.isMittagLeffler
  49 
  50 /-- Bounded filtrations satisfy Mittag-Leffler. -/
  51 theorem Filtration.IsBounded.toIsMittagLeffler {ω : Type w} {A : ω → C}
  52     {fil : Filtration A} (hb : fil.IsBounded) : fil.IsMittagLeffler :=
  53   hb.toIsBoundedAbove.toIsMittagLeffler
