  24 noncomputable def Filtration.completionLift {ω : Type w} {A : ω → C}
  25     (fil : Filtration A) (hbb : fil.IsBoundedBelow) (k : ω) (T : C)
  26     (f : ∀ s₀ : ℤ, T ⟶ (fil.completionFiltration hbb).truncatedObj s₀ k)
  27     (hcompat : ∀ (s₀ s₁ : ℤ) (h : s₀ ≤ s₁),
  28       f s₁ ≫ (fil.completionFiltration hbb).truncationTransition h k = f s₀) :
  29     T ⟶ fil.completion hbb k :=
  30   limit.lift (fil.completionFunctor hbb k) {
  31     pt := T
  32     π := {
  33       app := fun n => fil.completionConeLeg hbb k T f n
  34       naturality := by
  35         intro m n φ
  36         simp only [Functor.const_obj_obj, Functor.const_obj_map, Category.id_comp]
  37         exact fil.completionConeLeg_naturality hbb k T f hcompat m n φ
  38     }
  39   }
