   9 /-!
  10 # Successor compatibility of pagewise spectral-sequence comparisons
  11 
  12 The successor square uses the canonical page-to-homology isomorphisms of
  13 the nested-subobject model and the homology map already induced by the same
  14 page comparison.  This is stronger than independent isomorphisms on each
  15 page, while leaving the two ambient objects unconstrained.
  16 -/
  17 
  18 namespace KIP126.Core.SpectralSequence.PagewiseIso
  19 
  20 open CategoryTheory
  21 
  22 universe u v w
  23 
  24 variable {C : Type u} [Category.{v} C] [Abelian C]
  25   {ι : Type w} [AddCommGroup ι] [DecidableEq ι]
  26   {E F : KIP126.Core.SpectralSequence C ι}
  27 
  28 /-- The next-page isomorphism agrees with the isomorphism induced on the
  29 homology of the preceding page's actual differential. -/
  30 def RespectsSuccessor (Q : PagewiseIso E F) : Prop :=
  31   ∀ (r : ℤ) (hr : E.r₀ ≤ r) (k : ι),
  32     (pageHomologyIso E r k hr).hom ≫ (Q.homologyIso r hr k).hom =
  33       (Q.pageIso (r + 1) (by omega) k).hom ≫
  34         (pageHomologyIso F r k (by rw [← Q.firstPage_eq]; exact hr)).hom
  35 
  36 end KIP126.Core.SpectralSequence.PagewiseIso
