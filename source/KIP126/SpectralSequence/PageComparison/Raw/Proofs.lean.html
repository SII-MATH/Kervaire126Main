   7 /-! Commuting squares for the actual centered page comparisons. -/
   8 
   9 namespace KIP126.Core.SpectralSequence.PagewiseIso

  11 open CategoryTheory
  12 
  13 universe u v w
  14 
  15 variable {C : Type u} [Category.{v} C] [Abelian C]
  16   {ι : Type w} [AddCommGroup ι] [DecidableEq ι]
  17   {E F : KIP126.Core.SpectralSequence C ι} (Q : PagewiseIso E F)
  18 
  19 /-- The left square is the specified differential square after transporting
  20 the left and middle indices by the common differential degree. -/
  21 theorem left_middle_comm (r : ℤ) (hr : E.r₀ ≤ r) (k : ι) :
  22     (Q.leftIso r hr k).hom ≫ (F.pageShortComplex r (k - F.diffDeg r)).f =
  23       (E.pageShortComplex r (k - E.diffDeg r)).f ≫ (Q.middleIso r hr k).hom := by
  24   simp only [leftIso, middleIso, KIP126.Core.SpectralSequence.pageShortComplex,
  25     Iso.trans_hom, eqToIso.hom]
  26   have hleft : k - E.diffDeg r = k - F.diffDeg r := by
  27     rw [Q.diffDeg_eq]
  28   rw [Category.assoc,
  29     ← eqToHom_naturality (fun j => F.d r j) hleft,
  30     ← Category.assoc, Q.comm_d r hr (k - E.diffDeg r)]
  31   simp [Category.assoc, Q.diffDeg_eq]
  32 
  33 /-- The right square is the same differential compatibility at the middle
  34 index, with the specified degree transports. -/
  35 theorem middle_right_comm (r : ℤ) (hr : E.r₀ ≤ r) (k : ι) :
  36     (Q.middleIso r hr k).hom ≫ (F.pageShortComplex r (k - F.diffDeg r)).g =
  37       (E.pageShortComplex r (k - E.diffDeg r)).g ≫ (Q.rightIso r hr k).hom := by
  38   simp only [middleIso, rightIso, KIP126.Core.SpectralSequence.pageShortComplex,
  39     Iso.trans_hom, eqToIso.hom]
  40   have hmiddle : k - E.diffDeg r + E.diffDeg r =
  41       k - F.diffDeg r + F.diffDeg r := by simp
  42   rw [Category.assoc,
  43     ← eqToHom_naturality (fun j => F.d r j) hmiddle,
  44     ← Category.assoc, Q.comm_d r hr (k - E.diffDeg r + E.diffDeg r)]
  45   simp [Category.assoc, Q.diffDeg_eq]
  46 
  47 end KIP126.Core.SpectralSequence.PagewiseIso
