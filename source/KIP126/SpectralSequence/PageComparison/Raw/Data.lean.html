   8 /-!
   9 # Page isomorphisms compatible with the actual differentials
  10 
  11 Two nested-subobject models can have different ambient objects and still
  12 describe isomorphic admissible pages.  This comparison therefore asks only
  13 for page isomorphisms, a common displayed page convention, and commuting
  14 differential squares.  It does not assert an ambient or cycle-subobject
  15 isomorphism.
  16 -/
  17 
  18 namespace KIP126.Core.SpectralSequence
  19 
  20 open CategoryTheory
  21 
  22 universe u v w
  23 
  24 variable {C : Type u} [Category.{v} C] [Abelian C]
  25   {ι : Type w} [AddCommGroup ι] [DecidableEq ι]
  26 
  27 /-- An isomorphism on each admissible page that commutes with the specified
  28 differentials.  Successor-page compatibility is stated separately using the
  29 homology map induced by these same isomorphisms. -/
  30 structure PagewiseIso (E F : KIP126.Core.SpectralSequence C ι) where
  31   firstPage_eq : E.r₀ = F.r₀
  32   diffDeg_eq : E.diffDeg = F.diffDeg
  33   pageIso : ∀ (r : ℤ), E.r₀ ≤ r → ∀ k : ι, E.Page r k ≅ F.Page r k
  34   comm_d : ∀ (r : ℤ) (hr : E.r₀ ≤ r) (k : ι),
  35     (pageIso r hr k).hom ≫ F.d r k =
  36       E.d r k ≫ (pageIso r hr (k + E.diffDeg r)).hom ≫
  37         eqToHom (by rw [diffDeg_eq])
  38 
  39 namespace PagewiseIso
  40 
  41 variable {E F : KIP126.Core.SpectralSequence C ι} (Q : PagewiseIso E F)
  42 
  43 /-- The actual comparison at the left vertex of the short complex centered
  44 at `k`, including the change of displayed differential degree. -/
  45 noncomputable def leftIso (r : ℤ) (hr : E.r₀ ≤ r) (k : ι) :
  46     (E.pageShortComplex r (k - E.diffDeg r)).X₁ ≅
  47       (F.pageShortComplex r (k - F.diffDeg r)).X₁ :=
  48   Q.pageIso r hr (k - E.diffDeg r) ≪≫ eqToIso (by
  49     change F.Page r (k - E.diffDeg r) = F.Page r (k - F.diffDeg r)
  50     rw [Q.diffDeg_eq])
  51 
  52 /-- The actual comparison at the middle vertex of the centered short complex.
  53 The displayed indices retain the same explicit transports as `pageShortComplex`. -/
  54 noncomputable def middleIso (r : ℤ) (hr : E.r₀ ≤ r) (k : ι) :
  55     (E.pageShortComplex r (k - E.diffDeg r)).X₂ ≅
  56       (F.pageShortComplex r (k - F.diffDeg r)).X₂ :=
  57   Q.pageIso r hr (k - E.diffDeg r + E.diffDeg r) ≪≫
  58     eqToIso (by
  59       change F.Page r (k - E.diffDeg r + E.diffDeg r) =
  60         F.Page r (k - F.diffDeg r + F.diffDeg r)
  61       rw [Q.diffDeg_eq])
  62 
  63 /-- The actual comparison at the right vertex of the centered short complex. -/
  64 noncomputable def rightIso (r : ℤ) (hr : E.r₀ ≤ r) (k : ι) :
  65     (E.pageShortComplex r (k - E.diffDeg r)).X₃ ≅
  66       (F.pageShortComplex r (k - F.diffDeg r)).X₃ :=
  67   Q.pageIso r hr (k - E.diffDeg r + E.diffDeg r + E.diffDeg r) ≪≫
  68     eqToIso (by
  69       change F.Page r (k - E.diffDeg r + E.diffDeg r + E.diffDeg r) =
  70         F.Page r (k - F.diffDeg r + F.diffDeg r + F.diffDeg r)
  71       rw [Q.diffDeg_eq])
  72 
  73 end PagewiseIso
  74 end KIP126.Core.SpectralSequence
