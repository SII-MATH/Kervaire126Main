  21 /-- The differential at `(r,k)` is essential when it is nonzero. -/
  22 def IsEssentialAt (E : SpectralSequence C ι) (r : ℤ) (k : ι) : Prop :=
  23   E.d r k ≠ 0
  24 
  25 /-- Relation on ambient representatives induced by `d_r`. -/
  26 public def DifferentialRelation
  27     (E : SpectralSequence C ι) (r : ℤ) (k : ι)
  28     {T : C} (x : T ⟶ (E.ssData k).V)
  29     (y : T ⟶ (E.ssData (k + E.diffDeg r)).V) : Prop :=
  30   let n : WithTop ℕ := ↑(r - E.r₀).toNat
  31   ∃ (xZ : T ⟶ Subobject.underlying.obj ((E.ssData k).Z n)),
  32     xZ ≫ ((E.ssData k).Z n).arrow = x ∧
  33       ∃ (yZ : T ⟶ Subobject.underlying.obj
  34         ((E.ssData (k + E.diffDeg r)).Z n)),
  35         yZ ≫ ((E.ssData (k + E.diffDeg r)).Z n).arrow = y ∧
  36           xZ ≫ (E.ssData k).pageπ n ≫ E.d r k =
  37             yZ ≫ (E.ssData (k + E.diffDeg r)).pageπ n
  38 
  39 /-- A differential relation whose target representative is not a boundary. -/
  40 public def EssentialDifferentialRelation
  41     (E : SpectralSequence C ι) (r : ℤ) (k : ι)
  42     {T : C} (x : T ⟶ (E.ssData k).V)
  43     (y : T ⟶ (E.ssData (k + E.diffDeg r)).V) : Prop :=
  44   DifferentialRelation E r k x y ∧
  45     ¬ Subobject.Factors
  46       ((E.ssData (k + E.diffDeg r)).B
  47         (↑(r - E.r₀).toNat : WithTop ℕ)) y
  48 
  49 /-- An ambient representative and the page class it represents. -/
  50 public def ElementPageRel
  51     (E : SpectralSequence C ι) (r : ℤ) (k : ι)
  52     {T : C} (x : T ⟶ (E.ssData k).V)
  53     (a : T ⟶ (E.ssData k).page
  54       (↑(r - E.r₀).toNat : WithTop ℕ)) : Prop :=
  55   ∃ (xZ : T ⟶ Subobject.underlying.obj
  56       ((E.ssData k).Z (↑(r - E.r₀).toNat : WithTop ℕ))),
  57     xZ ≫ ((E.ssData k).Z
  58       (↑(r - E.r₀).toNat : WithTop ℕ)).arrow = x ∧
  59       xZ ≫ (E.ssData k).pageπ
  60         (↑(r - E.r₀).toNat : WithTop ℕ) = a
  61 
  62 /-- A relation crossed by an essential relation from higher filtration. -/
  63 def RelationCrossedBy
  64     (E : SpectralSequence C ι) (filtDeg : ι → ℤ) (r : ℤ) (k : ι)
  65     {T : C} (x : T ⟶ (E.ssData k).V)
  66     (y : T ⟶ (E.ssData (k + E.diffDeg r)).V)
  67     (_h : DifferentialRelation E r k x y) : Prop :=
  68   ∃ (a : ℤ) (_ : 0 < a) (m : ℤ) (k' : ι)
  69     (x' : T ⟶ (E.ssData k').V)
  70     (y' : T ⟶ (E.ssData (k' + E.diffDeg m)).V),
  71     filtDeg k' = filtDeg k + a ∧
  72       EssentialDifferentialRelation E m k' x' y' ∧
  73       filtDeg (k' + E.diffDeg m) ≤ filtDeg k + r
  74 
  75 /-- A crossing whose target filtration is exactly that of the original differential. -/
  76 def RelationCrossedByAt
  77     (E : SpectralSequence C ι) (filtDeg : ι → ℤ) (r : ℤ) (k : ι)
  78     {T : C} (x : T ⟶ (E.ssData k).V)
  79     (y : T ⟶ (E.ssData (k + E.diffDeg r)).V)
  80     (_h : DifferentialRelation E r k x y) : Prop :=
  81   ∃ (a : ℤ) (_ : 0 < a) (m : ℤ) (k' : ι)
  82     (x' : T ⟶ (E.ssData k').V)
  83     (y' : T ⟶ (E.ssData (k' + E.diffDeg m)).V),
  84     filtDeg k' = filtDeg k + a ∧
  85       EssentialDifferentialRelation E m k' x' y' ∧
  86       filtDeg (k' + E.diffDeg m) = filtDeg k + r
  87 
  88 /-- A crossing whose target filtration is `p`. -/
  89 def HasCrossingAt (dd : DifferentialDatum C ι) (p : ℤ) : Prop :=
  90   let s := dd.filtDeg dd.k
  91   ∃ (a : ℤ) (_ : 0 < a) (m : ℤ) (k' : ι),
  92     dd.filtDeg k' = s + a ∧
  93       IsEssentialAt dd.E m k' ∧
  94       p = s + a + m ∧
  95       s + a + m ≤ s + dd.r
  96 
  97 /-- No essential crossing lands in the indicated filtration range. -/
  98 def NoCrossingRange (dd : DifferentialDatum C ι) (p : ℤ) : Prop :=
  99   ¬ ∃ (a : ℤ) (_ : 0 < a) (m : ℤ) (k' : ι),
 100     dd.filtDeg k' = dd.filtDeg dd.k + a ∧
 101       IsEssentialAt dd.E m k' ∧
 102       p ≤ dd.filtDeg dd.k + a + m ∧
 103       dd.filtDeg dd.k + a + m ≤ dd.filtDeg dd.k + dd.r
 104 
 105 /-- The selected differential has no crossing. -/
 106 def NoCrossing (dd : DifferentialDatum C ι) : Prop :=
 107   NoCrossingRange dd (dd.filtDeg dd.k + 1)
 108 
 109 /-- Crossing predicate relative to supplied convergence data. -/
 110 def HasCrossingAt_conv
 111     {ω : Type w} [AddCommGroup ω] [DecidableEq ω]
 112     {E : SpectralSequence C ω} {ω' : Type w} {A : ω' → C} {F : Filtration A}
 113     (_conv : Convergence E A F) (dd : DifferentialDatum C ω) (p : ℤ) : Prop :=
 114   HasCrossingAt dd p
