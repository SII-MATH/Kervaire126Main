   8 /-! Data carried by a differential in a nested-subobject spectral sequence. -/
   9 
  10 namespace KIP126.Core.SpectralSequence
  11 
  12 open CategoryTheory
  13 
  14 universe u v w
  15 
  16 variable {C : Type u} [Category.{v} C] [Abelian C]
  17 variable {ι : Type w} [AddCommGroup ι] [DecidableEq ι]
  18 
  19 /-- A differential together with the filtration degrees of its gradings. -/
  20 structure DifferentialDatum
  21     (C : Type u) [Category.{v} C] [Abelian C]
  22     (ι : Type w) [AddCommGroup ι] [DecidableEq ι] where
  23   /-- Underlying nested-subobject spectral sequence. -/
  24   E : SpectralSequence C ι
  25   /-- Page number. -/
  26   r : ℤ
  27   /-- Source grading. -/
  28   k : ι
  29   /-- Filtration degree at every grading. -/
  30   filtDeg : ι → ℤ
  31   /-- The selected differential is nonzero. -/
  32   is_essential : E.d r k ≠ 0
  33 
  34 /-- Construct differential data from a spectral sequence and an essentiality proof. -/
  35 def DifferentialDatum.ofSpectralSequence
  36     (E : SpectralSequence C ι) (r : ℤ) (k : ι)
  37     (filtDeg : ι → ℤ) (hess : E.d r k ≠ 0) :
  38     DifferentialDatum C ι :=
  39   { E := E
  40     r := r
  41     k := k
  42     filtDeg := filtDeg
  43     is_essential := hess }
  44 
  45 end KIP126.Core.SpectralSequence
