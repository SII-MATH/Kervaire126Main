  22 /-- The ambient differential relation is equivalent to compatible page representatives. -/
  23 public theorem dr_apply_iff_rel
  24     (E : SpectralSequence C ι) (r : ℤ) (k : ι)
  25     {T : C} (x : T ⟶ (E.ssData k).V)
  26     (y : T ⟶ (E.ssData (k + E.diffDeg r)).V) :
  27     (∃ (x₀ : T ⟶ (E.ssData k).page
  28           (↑(r - E.r₀).toNat : WithTop ℕ))
  29         (y₀ : T ⟶ (E.ssData (k + E.diffDeg r)).page
  30           (↑(r - E.r₀).toNat : WithTop ℕ)),
  31       x₀ ≫ E.d r k = y₀ ∧
  32         ElementPageRel E r k x x₀ ∧
  33         ElementPageRel E r (k + E.diffDeg r) y y₀) ↔
  34       DifferentialRelation E r k x y := by
  35   constructor
  36   · rintro ⟨x₀, y₀, hxy, ⟨xZ, hxZ, hx₀⟩, ⟨yZ, hyZ, hy₀⟩⟩
  37     subst hx₀
  38     subst hy₀
  39     exact ⟨xZ, hxZ, yZ, hyZ, by rw [← Category.assoc]; exact hxy⟩
  40   · rintro ⟨xZ, hxZ, yZ, hyZ, hrel⟩
  41     exact ⟨
  42       xZ ≫ (E.ssData k).pageπ (↑(r - E.r₀).toNat : WithTop ℕ),
  43       yZ ≫ (E.ssData (k + E.diffDeg r)).pageπ
  44         (↑(r - E.r₀).toNat : WithTop ℕ),
  45       by rwa [Category.assoc],
  46       ⟨xZ, hxZ, rfl⟩,
  47       ⟨yZ, hyZ, rfl⟩⟩

  49 /-- Targets of two differential relations with the same source differ by a page boundary. -/
  50 public theorem DifferentialRelation.targets_sub_factors_boundary
  51     (E : SpectralSequence C ι) (r : ℤ) (k : ι) {T : C}
  52     {x : T ⟶ (E.ssData k).V}
  53     {y₁ y₂ : T ⟶ (E.ssData (k + E.diffDeg r)).V}
  54     (h₁ : DifferentialRelation E r k x y₁)
  55     (h₂ : DifferentialRelation E r k x y₂) :
  56     Subobject.Factors
  57       ((E.ssData (k + E.diffDeg r)).B (↑(r - E.r₀).toNat : WithTop ℕ))
  58       (y₁ - y₂) := by
  59   rcases h₁ with ⟨xZ₁, hx₁, yZ₁, hy₁, hrel₁⟩
  60   rcases h₂ with ⟨xZ₂, hx₂, yZ₂, hy₂, hrel₂⟩
  61   have heq : xZ₁ = xZ₂ := by
  62     apply (cancel_mono ((E.ssData k).Z (↑(r - E.r₀).toNat : WithTop ℕ)).arrow).mp
  63     exact hx₁.trans hx₂.symm
  64   subst xZ₂
  65   have hpage : yZ₁ ≫
  66       (E.ssData (k + E.diffDeg r)).pageπ (↑(r - E.r₀).toNat : WithTop ℕ) =
  67       yZ₂ ≫
  68       (E.ssData (k + E.diffDeg r)).pageπ (↑(r - E.r₀).toNat : WithTop ℕ) :=
  69     hrel₁.symm.trans hrel₂
  70   simpa only [hy₁, hy₂] using
  71     (E.ssData (k + E.diffDeg r)).sub_factors_boundary_of_page_eq
  72       (↑(r - E.r₀).toNat : WithTop ℕ) yZ₁ yZ₂ hpage

  74 /-- The complete boundary coset of an ambient differential target. -/
  75 public def DifferentialTargetCoset
  76     (E : SpectralSequence C ι) (r : ℤ) (k : ι) {T : C}
  77     (y : T ⟶ (E.ssData (k + E.diffDeg r)).V) :
  78     Set (T ⟶ (E.ssData (k + E.diffDeg r)).V) :=
  79   {z | Subobject.Factors
  80     ((E.ssData (k + E.diffDeg r)).B (↑(r - E.r₀).toNat : WithTop ℕ)) (y - z)}

  82 /-- This coset is exactly the other targets of the same source relation. -/
  83 public theorem DifferentialRelation.mem_targetCoset_iff
  84     (E : SpectralSequence C ι) (r : ℤ) (k : ι) {T : C}
  85     {x : T ⟶ (E.ssData k).V}
  86     {y z : T ⟶ (E.ssData (k + E.diffDeg r)).V}
  87     (h : DifferentialRelation E r k x y) :
  88     z ∈ DifferentialTargetCoset E r k y ↔ DifferentialRelation E r k x z := by
  89   constructor
  90   · intro hz
  91     rcases h with ⟨xZ, hx, yZ, hy, hrel⟩
  92     let D := E.ssData (k + E.diffDeg r)
  93     let q : WithTop ℕ := ↑(r - E.r₀).toNat
  94     let i := Subobject.ofLE (D.B q) (D.Z q) (D.B_le_Z q)
  95     let b := (D.B q).factorThru (y - z) hz
  96     have hb : b ≫ (D.B q).arrow = y - z := Subobject.factorThru_arrow _ _ hz
  97     refine ⟨xZ, hx, yZ - b ≫ i, ?_, ?_⟩
  98     · rw [Preadditive.sub_comp, Category.assoc, Subobject.ofLE_arrow, hy, hb]
  99       abel
 100     · change xZ ≫ (E.ssData k).pageπ q ≫ E.d r k =
 101         (yZ - b ≫ i) ≫ cokernel.π i
 102       rw [Preadditive.sub_comp, Category.assoc, cokernel.condition, comp_zero, sub_zero]
 103       exact hrel
 104   · exact DifferentialRelation.targets_sub_factors_boundary E r k h

 106 /-- Essentiality excludes zero from the entire actual boundary coset. -/
 107 public theorem essentialDifferentialRelation_iff_zero_not_mem
 108     (E : SpectralSequence C ι) (r : ℤ) (k : ι) {T : C}
 109     {x : T ⟶ (E.ssData k).V}
 110     {y : T ⟶ (E.ssData (k + E.diffDeg r)).V}
 111     (h : DifferentialRelation E r k x y) :
 112     EssentialDifferentialRelation E r k x y ↔
 113       0 ∉ DifferentialTargetCoset E r k y := by
 114   simp only [EssentialDifferentialRelation, DifferentialTargetCoset,
 115     Set.mem_setOf_eq, sub_zero, h, true_and]

 117 /-- An exact-target crossing is also a crossing with bounded target filtration. -/
 118 theorem RelationCrossedByAt.toCrossed (E : SpectralSequence C ι)
 119     (filtDeg : ι → ℤ) (r : ℤ) (k : ι)
 120     {T : C} {x : T ⟶ (E.ssData k).V}
 121     {y : T ⟶ (E.ssData (k + E.diffDeg r)).V}
 122     {h : DifferentialRelation E r k x y}
 123     (hc : RelationCrossedByAt E filtDeg r k x y h) :
 124     RelationCrossedBy E filtDeg r k x y h := by
 125   rcases hc with ⟨a, ha, m, k', x', y', hs, he, ht⟩
 126   exact ⟨a, ha, m, k', x', y', hs, he, le_of_eq ht⟩
