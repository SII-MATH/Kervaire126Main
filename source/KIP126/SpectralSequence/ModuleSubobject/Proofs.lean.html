  16 /-- The categorical inclusion of subobjects is the usual submodule inclusion
  17 under the specified representative isomorphisms. -/
  18 theorem submodule_ofLE_underlyingIso (B Z : Submodule R M) (h : B ≤ Z) :
  19     Subobject.ofLE ((ModuleCat.subobjectModule M).symm B)
  20       ((ModuleCat.subobjectModule M).symm Z)
  21       ((ModuleCat.subobjectModule _).symm.monotone h) ≫
  22         (Subobject.underlyingIso (ModuleCat.ofHom Z.subtype)).hom =
  23       (Subobject.underlyingIso (ModuleCat.ofHom B.subtype)).hom ≫
  24         ModuleCat.ofHom (Submodule.inclusion h) := by
  25   change Subobject.ofLE (Subobject.mk (ModuleCat.ofHom B.subtype))
  26     (Subobject.mk (ModuleCat.ofHom Z.subtype)) _ ≫ _ = _
  27   rw [Subobject.ofLE_mk_le_mk_of_comm
  28     (ModuleCat.ofHom (Submodule.inclusion h)) (by rfl)]
  29   simp

  31 /-- The categorical image has exactly the ordinary linear-map range. -/
  32 public theorem subobjectModule_image {L : ModuleCat.{v} R} (f : L ⟶ M) :
  33     (ModuleCat.subobjectModule M) (imageSubobject f) = LinearMap.range f.hom := by
  34   have h : imageSubobject f =
  35       (ModuleCat.subobjectModule M).symm (LinearMap.range f.hom) :=
  36     Subobject.mk_eq_mk_of_comm _ _ (ModuleCat.imageIsoRange f)
  37       (ModuleCat.imageIsoRange_hom_subtype f)
  38   rw [h, OrderIso.apply_symm_apply]
  39 
  40 /-- The categorical kernel has exactly the ordinary linear-map kernel. -/
  41 public theorem subobjectModule_kernel {N : ModuleCat.{v} R} (f : M ⟶ N) :
  42     (ModuleCat.subobjectModule M) (kernelSubobject f) = LinearMap.ker f.hom := by
  43   have h : kernelSubobject f =
  44       (ModuleCat.subobjectModule M).symm (LinearMap.ker f.hom) :=
  45     Subobject.mk_eq_mk_of_comm _ _ (ModuleCat.kernelIsoKer f) (by simp)
  46   rw [h, OrderIso.apply_symm_apply]

  48 /-- A categorical cokernel class is zero exactly on the ordinary range. -/
  49 public theorem cokernel_π_eq_zero_iff_mem_range {L N : ModuleCat.{v} R}
  50     (f : L ⟶ N) (x : N) :
  51     cokernel.π f x = 0 ↔ x ∈ LinearMap.range f.hom := by
  52   rw [← (ModuleCat.cokernelIsoRangeQuotient f).toLinearEquiv.map_eq_zero_iff]
  53   change ((cokernel.π f ≫ (ModuleCat.cokernelIsoRangeQuotient f).hom) x = 0) ↔ _
  54   rw [ModuleCat.cokernel_π_cokernelIsoRangeQuotient_hom]
  55   exact Submodule.Quotient.mk_eq_zero _
  56 
  57 /-- The class of a cycle is zero exactly when its ambient representative
  58 belongs to the boundary subobject. -/
  59 public theorem subobject_cokernel_π_eq_zero_iff (B Z : Subobject M) (h : B ≤ Z)
  60     (z : (Subobject.underlying.obj Z : ModuleCat R)) :
  61     cokernel.π (Subobject.ofLE B Z h) z = 0 ↔
  62       Z.arrow z ∈ (ModuleCat.subobjectModule M) B := by
  63   rw [cokernel_π_eq_zero_iff_mem_range]
  64   change (∃ b, (Subobject.ofLE B Z h).hom b = z) ↔
  65     ∃ b, B.arrow.hom b = Z.arrow z
  66   constructor
  67   · rintro ⟨b, rfl⟩
  68     exact ⟨b, (ConcreteCategory.congr_hom (Subobject.ofLE_arrow h) b).symm⟩
  69   · rintro ⟨b, hb⟩
  70     refine ⟨b, (ModuleCat.mono_iff_injective Z.arrow).mp inferInstance ?_⟩
  71     change (Subobject.ofLE B Z h ≫ Z.arrow) b = Z.arrow z
  72     rw [Subobject.ofLE_arrow]
  73     exact hb
