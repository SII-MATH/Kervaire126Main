   8 /-! Converging spectral sequences with their full finite-page morphisms.
   9 The older convergence-only category forgets finite pages; it cannot serve as
  10 the ambient category for an enrichment which represents all-page pairings. -/
  11 namespace KIP126.Core.SpectralSequence
  12 open CategoryTheory
  13 universe u v
  14 variable {R : Type u} [CommRing R]
  15 noncomputable section
  16 
  17 /-- A wrapper selecting the full, rather than convergence-only, morphisms. -/
  18 public structure ConvergingSequence (R : Type u) [CommRing R] where
  19   underlying : ConvergingSS (ModuleCat.{v} R) (ℤ × ℤ) ℤ
  20 
  21 /-- A finite-page morphism and a filtered target map, with the same E∞ map. -/
  22 public structure ConvergingSequenceMorphism (X Y : ConvergingSequence.{u,v} R) where
  23   sequence : X.underlying.E ⟶ Y.underlying.E
  24   convergence : ConvergenceMorphism X.underlying.conv Y.underlying.conv
  25   eInfty_compatible : ∀ p, convergence.eMap p = sequence.eInftyMap p
  26 
  27 namespace ConvergingSequenceMorphism
  28 variable {X Y Z : ConvergingSequence.{u,v} R}
  29 
  30 @[ext] public theorem ext {f g : ConvergingSequenceMorphism X Y}
  31     (hs : f.sequence = g.sequence) (hc : f.convergence = g.convergence) : f = g := by
  32   cases f
  33   cases g
  34   cases hs
  35   cases hc
  36   rfl
  37 
  38 public def id (X : ConvergingSequence.{u,v} R) : ConvergingSequenceMorphism X X where
  39   sequence := 𝟙 _
  40   convergence := (𝟙 X.underlying : X.underlying ⟶ X.underlying)
  41   eInfty_compatible p := (SpectralSequenceMorphism.eInftyMap_id X.underlying.E p).symm
  42 
  43 public def comp (f : ConvergingSequenceMorphism X Y) (g : ConvergingSequenceMorphism Y Z) :
  44     ConvergingSequenceMorphism X Z where
  45   sequence := f.sequence ≫ g.sequence
  46   convergence := @CategoryStruct.comp (ConvergingSS (ModuleCat.{v} R) (ℤ × ℤ) ℤ)
  47     _ X.underlying Y.underlying Z.underlying f.convergence g.convergence
  48   eInfty_compatible p := by
  49     change f.convergence.eMap p ≫ g.convergence.eMap p =
  50       (f.sequence ≫ g.sequence).eInftyMap p
  51     rw [f.eInfty_compatible, g.eInfty_compatible, SpectralSequenceMorphism.eInftyMap_comp]
  52 end ConvergingSequenceMorphism
  53 
  54 public instance : Category (ConvergingSequence.{u,v} R) where
  55   Hom := ConvergingSequenceMorphism
  56   id := ConvergingSequenceMorphism.id
  57   comp := ConvergingSequenceMorphism.comp
  58   id_comp _f := ConvergingSequenceMorphism.ext (Category.id_comp _) (Category.id_comp _)
  59   comp_id _f := ConvergingSequenceMorphism.ext (Category.comp_id _) (Category.comp_id _)
  60   assoc {W X Y Z} f g h := ConvergingSequenceMorphism.ext
  61     (Category.assoc f.sequence g.sequence h.sequence)
  62     (@Category.assoc (ConvergingSS (ModuleCat.{v} R) (ℤ × ℤ) ℤ)
  63       _ W.underlying X.underlying Y.underlying Z.underlying
  64       f.convergence g.convergence h.convergence)
  65 
  66 end
  67 end KIP126.Core.SpectralSequence
