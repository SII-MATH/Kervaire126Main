  11 namespace KIP126.Core.SpectralSequence.FilteredComplex
  12 
  13 open CategoryTheory
  14 
  15 universe u v
  16 variable {C : Type u} [Category.{v} C] [Abelian C]
  17 
  18 namespace Solutions
  19 
  20 /-- The two actual filtered representatives, before imposing their equation. -/
  21 abbrev Representatives (FC : FilteredComplex C) (T : C) (r s k : ℤ) :=
  22   (T ⟶ Subobject.underlying.obj (FC.fil s k)) ×
  23     (T ⟶ Subobject.underlying.obj (FC.fil (s + r) (k - 1)))
  24 
  25 /-- Source label, target label, and the differential discrepancy. -/
  26 abbrev Equations (FC : FilteredComplex C) (T : C) (r s k : ℤ) :=
  27   (T ⟶ FC.assocGraded s k) × (T ⟶ FC.assocGraded (s + r) (k - 1)) ×
  28     (T ⟶ FC.A (k - 1))
  29 
  30 /-- All three equations use the original filtered complex and quotient maps.
  31 The ambient differential equation avoids a choice of a shorter-image group:
  32 changing the source lift already allows every higher-filtration correction. -/
  33 noncomputable def equation (FC : FilteredComplex C) (T : C) (r s k : ℤ) :
  34     Representatives FC T r s k →+ Equations FC T r s k where
  35   toFun z := (z.1 ≫ FC.filToAssocGraded s k,
  36     z.2 ≫ FC.filToAssocGraded (s + r) (k - 1),
  37     z.1 ≫ (FC.fil s k).arrow ≫ FC.d k -
  38       z.2 ≫ (FC.fil (s + r) (k - 1)).arrow)
  39   map_zero' := by simp
  40   map_add' a b := by
  41     simp only [Prod.fst_add, Prod.snd_add, Preadditive.add_comp, Prod.mk_add_mk]
  42     congr 2
  43     abel
  44 
  45 /-- The actual representative-solution fiber over the prescribed labels. -/
  46 def Fiber (FC : FilteredComplex C) {T : C} (r s k : ℤ)
  47     (x : T ⟶ FC.assocGraded s k)
  48     (y : T ⟶ FC.assocGraded (s + r) (k - 1)) :=
  49   { z : Representatives FC T r s k // equation FC T r s k z = (x, y, 0) }
  50 
  51 /-- Homogeneous solutions are precisely the kernel of the same equation map. -/
  52 noncomputable def differences (FC : FilteredComplex C) (T : C) (r s k : ℤ) :
  53     AddSubgroup (Representatives FC T r s k) :=
  54   (equation FC T r s k).ker
  55 
  56 end Solutions
