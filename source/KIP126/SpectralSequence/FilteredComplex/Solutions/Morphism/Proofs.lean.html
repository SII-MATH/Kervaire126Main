  15 /-- The actual representative equations commute with every filtered chain map. -/
  16 theorem equation_natural (f : Morphism FC GD) (T : C) (r s k : ℤ)
  17     (z : Representatives FC T r s k) :
  18     equation GD T r s k (representativesMap f T r s k z) =
  19       equationsMap f T r s k (equation FC T r s k z) := by
  20   have hs := InternalFilteredMorphism.toAssociatedGraded_comp_associatedGradedMap
  21     f.toFilteredMorphism s k
  22   have ht := InternalFilteredMorphism.toAssociatedGraded_comp_associatedGradedMap
  23     f.toFilteredMorphism (s + r) (k - 1)
  24   apply Prod.ext
  25   · change (z.1 ≫ (f.preserves s k).choose) ≫ GD.filToAssocGraded s k =
  26       (z.1 ≫ FC.filToAssocGraded s k) ≫ f.associatedGradedMap s k
  27     simpa only [Category.assoc, filToAssocGraded, Morphism.associatedGradedMap,
  28       Morphism.toFilteredMorphism] using congrArg (fun a => z.1 ≫ a) hs.symm
  29   apply Prod.ext
  30   · change (z.2 ≫ (f.preserves (s + r) (k - 1)).choose) ≫
  31         GD.filToAssocGraded (s + r) (k - 1) =
  32       (z.2 ≫ FC.filToAssocGraded (s + r) (k - 1)) ≫
  33         f.associatedGradedMap (s + r) (k - 1)
  34     simpa only [Category.assoc, filToAssocGraded, Morphism.associatedGradedMap,
  35       Morphism.toFilteredMorphism] using congrArg (fun a => z.2 ≫ a) ht.symm
  36   · change (z.1 ≫ (f.preserves s k).choose) ≫ (GD.fil s k).arrow ≫ GD.d k -
  37         (z.2 ≫ (f.preserves (s + r) (k - 1)).choose) ≫
  38           (GD.fil (s + r) (k - 1)).arrow =
  39       (z.1 ≫ (FC.fil s k).arrow ≫ FC.d k -
  40         z.2 ≫ (FC.fil (s + r) (k - 1)).arrow) ≫ f.map.f (k - 1)
  41     have hs' := congrArg (fun a => a ≫ GD.d k) (f.preserves s k).choose_spec
  42     simp only [Category.assoc] at hs'
  43     simp only [Category.assoc, Preadditive.sub_comp]
  44     rw [hs', (f.preserves (s + r) (k - 1)).choose_spec]
  45     simp only [d]
  46     rw [f.map.comm]
  47 
  48 theorem representativesMap_id (T : C) (r s k : ℤ)
  49     (z : Representatives FC T r s k) :
  50     representativesMap (Morphism.id FC) T r s k z = z := by
  51   simp [representativesMap, Morphism.id_preserves_eq]
  52 
  53 theorem representativesMap_comp (f : Morphism FC GD) (g : Morphism GD GE)
  54     (T : C) (r s k : ℤ) (z : Representatives FC T r s k) :
  55     representativesMap (Morphism.comp f g) T r s k z =
  56       representativesMap g T r s k (representativesMap f T r s k z) := by
  57   simp [representativesMap, Morphism.comp_preserves_eq, Category.assoc]
