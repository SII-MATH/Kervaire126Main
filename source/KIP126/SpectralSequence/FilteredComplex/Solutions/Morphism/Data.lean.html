   9 open CategoryTheory
  10 universe u v
  11 variable {C : Type u} [Category.{v} C] [Abelian C]
  12   {FC GD : FilteredComplex C}
  13 
  14 /-- Restrict both representatives using the same filtered chain map. -/
  15 noncomputable def representativesMap (f : Morphism FC GD) (T : C) (r s k : ℤ) :
  16     Representatives FC T r s k →+ Representatives GD T r s k where
  17   toFun z := (z.1 ≫ (f.preserves s k).choose,
  18     z.2 ≫ (f.preserves (s + r) (k - 1)).choose)
  19   map_zero' := by simp
  20   map_add' _ _ := by simp [Preadditive.add_comp]
  21 
  22 /-- Restriction on the labels and the actual ambient discrepancy. -/
  23 noncomputable def equationsMap (f : Morphism FC GD) (T : C) (r s k : ℤ) :
  24     Equations FC T r s k →+ Equations GD T r s k where
  25   toFun z := (z.1 ≫ f.associatedGradedMap s k,
  26     z.2.1 ≫ f.associatedGradedMap (s + r) (k - 1), z.2.2 ≫ f.map.f (k - 1))
  27   map_zero' := by simp
  28   map_add' _ _ := by simp [Preadditive.add_comp]
  29
