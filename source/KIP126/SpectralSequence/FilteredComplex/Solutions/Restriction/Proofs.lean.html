   9 open CategoryTheory
  10 universe u v
  11 variable {C : Type u} [Category.{v} C] [Abelian C]
  12   {FC GD GE : FilteredComplex C} {T : C} {r s k : ℤ}
  13   {x : T ⟶ FC.assocGraded s k} {y : T ⟶ FC.assocGraded (s + r) (k - 1)}
  14 
  15 /-- The action on each nonempty fiber is respected by restriction. -/
  16 theorem restrict_translate (f : Morphism FC GD) (g : differences FC T r s k)
  17     (a : Fiber FC r s k x y) :
  18     restrict f (translate g a) =
  19       translate (restrictDifferences f T r s k g) (restrict f a) := by
  20   apply Subtype.ext
  21   exact map_add (representativesMap f T r s k) g.val a.val
  22 
  23 theorem restrictDifferences_id (g : differences FC T r s k) :
  24     restrictDifferences (Morphism.id FC) T r s k g = g := by
  25   apply Subtype.ext
  26   exact representativesMap_id T r s k g.val
  27 
  28 theorem restrictDifferences_comp (f : Morphism FC GD) (g : Morphism GD GE)
  29     (a : differences FC T r s k) :
  30     restrictDifferences (Morphism.comp f g) T r s k a =
  31       restrictDifferences g T r s k (restrictDifferences f T r s k a) := by
  32   apply Subtype.ext
  33   exact representativesMap_comp f g T r s k a.val
  34 
  35 end KIP126.Core.SpectralSequence.FilteredComplex.Solutions
