  10 open CategoryTheory
  11 universe u v
  12 variable {C : Type u} [Category.{v} C] [Abelian C]
  13   {FC GD : FilteredComplex C} {T : C} {r s k : ℤ}
  14 
  15 /-- Restriction of a solution is induced by the actual filtered chain map. -/
  16 noncomputable def restrict (f : Morphism FC GD)
  17     {x : T ⟶ FC.assocGraded s k} {y : T ⟶ FC.assocGraded (s + r) (k - 1)}
  18     (a : Fiber FC r s k x y) :
  19     Fiber GD r s k (x ≫ f.associatedGradedMap s k)
  20       (y ≫ f.associatedGradedMap (s + r) (k - 1)) :=
  21   ⟨representativesMap f T r s k a.val, by
  22     rw [equation_natural, a.property]
  23     simp [equationsMap]⟩
  24 
  25 /-- The induced homomorphism on the actual groups of differences. -/
  26 noncomputable def restrictDifferences (f : Morphism FC GD) (T : C) (r s k : ℤ) :
  27     differences FC T r s k →+ differences GD T r s k where
  28   toFun g := ⟨representativesMap f T r s k g.val, by
  29     change equation GD T r s k _ = 0
  30     rw [equation_natural, show equation FC T r s k g.val = 0 from g.property,
  31       map_zero]⟩
  32   map_zero' := by apply Subtype.ext; exact map_zero _
  33   map_add' a b := by apply Subtype.ext; exact map_add _ a.val b.val
  34 
  35 end KIP126.Core.SpectralSequence.FilteredComplex.Solutions
