  11 open CategoryTheory
  12 universe u v
  13 variable {C : Type u} [Category.{v} C] [Abelian C]
  14 
  15 namespace Solutions
  16 variable {FC : FilteredComplex C} {T : C} {r s k : ℤ}
  17   {x : T ⟶ FC.assocGraded s k} {y : T ⟶ FC.assocGraded (s + r) (k - 1)}
  18 
  19 theorem mem_fiber_iff (z : Representatives FC T r s k) :
  20     equation FC T r s k z = (x, y, 0) ↔
  21       FC.IsLift s k z.1 x ∧ FC.IsLift (s + r) (k - 1) z.2 y ∧
  22         z.1 ≫ (FC.fil s k).arrow ≫ FC.d k =
  23           z.2 ≫ (FC.fil (s + r) (k - 1)).arrow := by
  24   simp only [equation, AddMonoidHom.coe_mk, ZeroHom.coe_mk,
  25     Prod.mk.injEq, sub_eq_zero, IsLift]
  26 
  27 /-- This fiber is exactly the existing strict representative relation. -/
  28 theorem nonempty_fiber_iff (hr : 0 ≤ r) :
  29     Nonempty (Fiber FC r s k x y) ↔ FC.RepresentativeRelation r hr s k x y := by
  30   constructor
  31   · rintro ⟨⟨z, hz⟩⟩
  32     obtain ⟨hx, hy, hd⟩ := (mem_fiber_iff z).mp hz
  33     refine ⟨z.1, z.2, hx, hy, ?_⟩
  34     apply (cancel_mono (FC.fil s (k - 1)).arrow).mp
  35     simpa only [Category.assoc, filDiff, fil, d,
  36       (FC.differential_preserves s k).choose_spec,
  37       Filtration.inclusion_arrow] using hd
  38   · rintro ⟨xl, yl, hx, hy, hd⟩
  39     refine ⟨⟨(xl, yl), (mem_fiber_iff _).mpr ⟨hx, hy, ?_⟩⟩⟩
  40     have h := congrArg (fun a => a ≫ (FC.fil s (k - 1)).arrow) hd
  41     simpa only [Category.assoc, filDiff, fil, d,
  42       (FC.differential_preserves s k).choose_spec,
  43       Filtration.inclusion_arrow] using h
  44 
  45 /-- Differences between solutions are actual homogeneous representative equations. -/
  46 theorem sub_mem_differences (a b : Fiber FC r s k x y) :
  47     a.val - b.val ∈ differences FC T r s k := by
  48   change equation FC T r s k (a.val - b.val) = 0
  49   rw [map_sub, a.property, b.property, sub_self]
  50 
  51 theorem add_mem_fiber (g : differences FC T r s k) (a : Fiber FC r s k x y) :
  52     equation FC T r s k (g.val + a.val) = (x, y, 0) := by
  53   rw [map_add, show equation FC T r s k g.val = 0 from g.property,
  54     zero_add, a.property]
