   7 /-! A strict solution is determined by its source representative. The target
   8 representative is fixed by the differential and its subobject inclusion. -/
   9 
  10 namespace KIP126.Core.SpectralSequence.FilteredComplex.Solutions
  11 
  12 open CategoryTheory
  13 universe u v
  14 variable {C : Type u} [Category.{v} C] [Abelian C]
  15   {FC : FilteredComplex C} {T : C} {r s k : ℤ}
  16   {x : T ⟶ FC.assocGraded s k} {y : T ⟶ FC.assocGraded (s + r) (k - 1)}
  17 
  18 theorem source_ambient_injective :
  19     Function.Injective (fun a : Fiber FC r s k x y =>
  20       a.val.1 ≫ (FC.fil s k).arrow) := by
  21   intro a b hab
  22   have hfirst : a.val.1 = b.val.1 := (cancel_mono _).mp hab
  23   have ha := ((mem_fiber_iff a.val).mp a.property).2.2
  24   have hb := ((mem_fiber_iff b.val).mp b.property).2.2
  25   have hsecond : a.val.2 = b.val.2 := by
  26     apply (cancel_mono (FC.fil (s + r) (k - 1)).arrow).mp
  27     rw [← ha, ← hb, hfirst]
  28   exact Subtype.ext (Prod.ext hfirst hsecond)
  29 
  30 /-- Only the source ambient Hom needs to be finite. No target finiteness or
  31 nonemptiness of the solution fiber is required. -/
  32 theorem finite_of_finite_source_hom [Finite (T ⟶ FC.A k)] :
  33     Finite (Fiber FC r s k x y) :=
  34   Finite.of_injective _ source_ambient_injective
  35 
  36 end KIP126.Core.SpectralSequence.FilteredComplex.Solutions
