  10 open CategoryTheory
  11 universe u v
  12 variable {C : Type u} [Category.{v} C] [Abelian C]
  13   {FC GD : FilteredComplex C} {T : C} {r s k : ℤ}
  14   {x : T ⟶ FC.assocGraded s k} {y : T ⟶ FC.assocGraded (s + r) (k - 1)}
  15 
  16 /-- Restriction carries the displacement of two actual solutions to the
  17 displacement of their restrictions. -/
  18 theorem restrict_displacement (f : Morphism FC GD) (b a : Fiber FC r s k x y) :
  19     restrictDifferences f T r s k (displacement b a) =
  20       displacement (restrict f b) (restrict f a) := by
  21   apply Subtype.ext
  22   exact map_sub (representativesMap f T r s k) b.val a.val
  23 
  24 /-- Once a later solution is specified, lifting every earlier strict solution
  25 is equivalent to lifting every homogeneous displacement. Existence of the
  26 basepoint alone does not imply either surjectivity assertion. -/
  27 theorem restrict_surjective_iff_differences_surjective (f : Morphism FC GD)
  28     (a : Fiber FC r s k x y) :
  29     Function.Surjective (restrict f : Fiber FC r s k x y → _) ↔
  30       Function.Surjective (restrictDifferences f T r s k) := by
  31   constructor
  32   · intro h g
  33     obtain ⟨b, hb⟩ := h (translate g (restrict f a))
  34     refine ⟨displacement b a, ?_⟩
  35     rw [restrict_displacement, hb]
  36     exact (coordinateEquiv (restrict f a)).left_inv g
  37   · intro h b
  38     obtain ⟨g, hg⟩ := h (displacement b (restrict f a))
  39     refine ⟨translate g a, ?_⟩
  40     rw [restrict_translate, hg]
  41     exact (coordinateEquiv (restrict f a)).apply_symm_apply b
  42 
  43 /-- Vanishing of the specified strict lifting obstruction for every earlier
  44 solution is exactly surjectivity on the actual solution fibers. -/
  45 theorem forall_obstruction_eq_zero_iff_surjective (f : Morphism FC GD) :
  46     (∀ a : Fiber GD r s k (x ≫ f.associatedGradedMap s k)
  47       (y ≫ f.associatedGradedMap (s + r) (k - 1)), obstruction f x y a = 0) ↔
  48       Function.Surjective (restrict f : Fiber FC r s k x y → _) := by
  49   simp only [obstruction_eq_zero_iff, Function.Surjective]
  50 
  51 /-- With a later basepoint, all strict lifting obstructions vanish precisely
  52 when the actual homomorphism of homogeneous solution groups is surjective. -/
  53 theorem forall_obstruction_eq_zero_iff_differences_surjective (f : Morphism FC GD)
  54     (a : Fiber FC r s k x y) :
  55     (∀ b : Fiber GD r s k (x ≫ f.associatedGradedMap s k)
  56       (y ≫ f.associatedGradedMap (s + r) (k - 1)), obstruction f x y b = 0) ↔
  57       Function.Surjective (restrictDifferences f T r s k) :=
  58   (forall_obstruction_eq_zero_iff_surjective f).trans
  59     (restrict_surjective_iff_differences_surjective f a)
  60 
  61 end KIP126.Core.SpectralSequence.FilteredComplex.Solutions
