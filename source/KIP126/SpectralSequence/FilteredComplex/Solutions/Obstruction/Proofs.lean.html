   9 open CategoryTheory
  10 universe u v
  11 variable {C : Type u} [Category.{v} C] [Abelian C]
  12   {FC GD : FilteredComplex C} {T : C} {r s k : ℤ}
  13   {x : T ⟶ FC.assocGraded s k} {y : T ⟶ FC.assocGraded (s + r) (k - 1)}
  14 
  15 /-- Exactness at the actual earlier solution fiber, with no nonemptiness
  16 assumption on the later fiber. This is not yet a crossing/loss certificate. -/
  17 theorem obstruction_eq_zero_iff (f : Morphism FC GD)
  18     (a : Fiber GD r s k (x ≫ f.associatedGradedMap s k)
  19       (y ≫ f.associatedGradedMap (s + r) (k - 1))) :
  20     obstruction f x y a = 0 ↔ ∃ b : Fiber FC r s k x y, restrict f b = a := by
  21   change QuotientAddGroup.mk ((x, y, 0), a.val) = 0 ↔ _
  22   erw [QuotientAddGroup.eq_zero_iff]
  23   constructor
  24   · rintro ⟨z, hz⟩
  25     have he : equation FC T r s k z = (x, y, 0) := congrArg Prod.fst hz
  26     have hm : representativesMap f T r s k z = a.val := congrArg Prod.snd hz
  27     exact ⟨⟨z, he⟩, Subtype.ext hm⟩
  28   · rintro ⟨b, hb⟩
  29     exact ⟨b.val, Prod.ext b.property (congrArg Subtype.val hb)⟩
  30 
  31 end KIP126.Core.SpectralSequence.FilteredComplex.Solutions
