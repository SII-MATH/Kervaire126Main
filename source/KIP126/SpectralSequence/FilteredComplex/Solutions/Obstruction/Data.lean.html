  10 open CategoryTheory
  11 universe u v
  12 variable {C : Type u} [Category.{v} C] [Abelian C]
  13   {FC GD : FilteredComplex C}
  14 
  15 /-- Simultaneously impose the later equation and its specified earlier lift. -/
  16 noncomputable def liftingMap (f : Morphism FC GD) (T : C) (r s k : ℤ) :
  17     Representatives FC T r s k →+
  18       Equations FC T r s k × Representatives GD T r s k :=
  19   (equation FC T r s k).prod (representativesMap f T r s k)
  20 
  21 /-- A concrete cokernel obstruction; no solution is chosen in either fiber. -/
  22 noncomputable def LiftingObstruction (f : Morphism FC GD) (T : C) (r s k : ℤ) :=
  23   (Equations FC T r s k × Representatives GD T r s k) ⧸
  24     (liftingMap f T r s k).range
  25 
  26 noncomputable instance (f : Morphism FC GD) (T : C) (r s k : ℤ) :
  27     AddCommGroup (LiftingObstruction f T r s k) :=
  28   inferInstanceAs (AddCommGroup (_ ⧸ (liftingMap f T r s k).range))
  29 
  30 /-- The obstruction of the actual earlier representative equation. -/
  31 noncomputable def obstruction (f : Morphism FC GD) {T : C} {r s k : ℤ}
  32     (x : T ⟶ FC.assocGraded s k) (y : T ⟶ FC.assocGraded (s + r) (k - 1))
  33     (a : Fiber GD r s k (x ≫ f.associatedGradedMap s k)
  34       (y ≫ f.associatedGradedMap (s + r) (k - 1))) :
  35     LiftingObstruction f T r s k :=
  36   QuotientAddGroup.mk ((x, y, 0), a.val)
  37 
  38 end KIP126.Core.SpectralSequence.FilteredComplex.Solutions
