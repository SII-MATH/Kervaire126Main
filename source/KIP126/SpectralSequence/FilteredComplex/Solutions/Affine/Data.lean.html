   9 open CategoryTheory
  10 universe u v
  11 variable {C : Type u} [Category.{v} C] [Abelian C]
  12   {FC : FilteredComplex C} {T : C} {r s k : ℤ}
  13   {x : T ⟶ FC.assocGraded s k} {y : T ⟶ FC.assocGraded (s + r) (k - 1)}
  14 
  15 /-- Translation by a homogeneous solution uses addition of the actual lifts. -/
  16 noncomputable def translate (g : differences FC T r s k) (a : Fiber FC r s k x y) :
  17     Fiber FC r s k x y :=
  18   ⟨g.val + a.val, add_mem_fiber g a⟩
  19 
  20 /-- The unique homogeneous displacement between two actual solutions. -/
  21 noncomputable def displacement (b a : Fiber FC r s k x y) :
  22     differences FC T r s k :=
  23   ⟨b.val - a.val, sub_mem_differences b a⟩
  24 
  25 /-- A chosen actual solution identifies its fiber with the actual kernel.
  26 This does not assert that an empty fiber has a point. -/
  27 noncomputable def coordinateEquiv (a : Fiber FC r s k x y) :
  28     differences FC T r s k ≃ Fiber FC r s k x y where
  29   toFun g := translate g a
  30   invFun b := displacement b a
  31   left_inv g := by apply Subtype.ext; simp [displacement, translate]
  32   right_inv b := by apply Subtype.ext; simp [displacement, translate]
  33 
  34 end KIP126.Core.SpectralSequence.FilteredComplex.Solutions
