  21 /-- Projectivity lifts an actual cycle through the image of its cycle kernel. -/
  22 theorem exists_strict_lifts_of_cycle (FC : FilteredComplex C) (s k : ℤ) (n : ℕ)
  23     {T : C} [Projective T]
  24     (z : T ⟶ Subobject.underlying.obj (FC.cycleSubobject s k ↑n)) :
  25     ∃ (xl : T ⟶ Subobject.underlying.obj (FC.fil s k))
  26       (yl : T ⟶ Subobject.underlying.obj (FC.fil (s + ↑n) (k - 1))),
  27       xl ≫ FC.filToAssocGraded s k = z ≫ (FC.cycleSubobject s k ↑n).arrow ∧
  28       xl ≫ (FC.fil s k).arrow ≫ FC.d k =
  29         yl ≫ (FC.fil (s + ↑n) (k - 1)).arrow := by
  30   let f := (FC.fil s k).arrow ≫ FC.d k ≫
  31     cokernel.π (FC.fil (s + ↑n) (k - 1)).arrow
  32   let K := kernelSubobject f
  33   let p := factorThruImageSubobject (K.arrow ≫ FC.filToAssocGraded s k)
  34   let u := Projective.factorThru z p
  35   let xl := u ≫ K.arrow
  36   have hzero : (xl ≫ (FC.fil s k).arrow ≫ FC.d k) ≫
  37       cokernel.π (FC.fil (s + ↑n) (k - 1)).arrow = 0 := by
  38     simpa only [xl, Category.assoc, comp_zero] using
  39       congrArg (fun q => u ≫ q) (kernelSubobject_arrow_comp f)
  40   let yl := Abelian.monoLift (FC.fil (s + ↑n) (k - 1)).arrow
  41     (xl ≫ (FC.fil s k).arrow ≫ FC.d k) hzero
  42   refine ⟨xl, yl, ?_, (Abelian.monoLift_comp _ _ _).symm⟩
  43   calc
  44     xl ≫ FC.filToAssocGraded s k =
  45         u ≫ p ≫ (FC.cycleSubobject s k ↑n).arrow := by
  46           change (u ≫ K.arrow) ≫ FC.filToAssocGraded s k =
  47             u ≫ factorThruImageSubobject (K.arrow ≫ FC.filToAssocGraded s k) ≫
  48               (imageSubobject (K.arrow ≫ FC.filToAssocGraded s k)).arrow
  49           simp only [Category.assoc, imageSubobject_arrow_comp]
  50     _ = z ≫ (FC.cycleSubobject s k ↑n).arrow := by
  51       rw [← Category.assoc, Projective.factorThru_comp]

  53 /-- Lift a boundary first through its intersection image, then through the
  54 actual differential image. The two lifts obey an equality in the ambient complex. -/
  55 theorem exists_lifts_of_boundary (FC : FilteredComplex C) (s k : ℤ) (n : ℕ)
  56     {T : C} [Projective T]
  57     (b : T ⟶ Subobject.underlying.obj (FC.boundarySubobject s k ↑n)) :
  58     ∃ (a : T ⟶ Subobject.underlying.obj (FC.fil (s - ↑n + 1) (k + 1)))
  59       (c : T ⟶ Subobject.underlying.obj (FC.fil s k)),
  60       a ≫ (FC.fil (s - ↑n + 1) (k + 1)).arrow ≫ FC.complex.d (k + 1) k =
  61         c ≫ (FC.fil s k).arrow ∧
  62       c ≫ FC.filToAssocGraded s k = b ≫ (FC.boundarySubobject s k ↑n).arrow := by
  63   let d := (FC.fil (s - ↑n + 1) (k + 1)).arrow ≫ FC.dToK k
  64   let I := imageSubobject d ⊓ FC.fil s k
  65   let oI := Subobject.ofLE I (FC.fil s k) inf_le_right
  66   let p := factorThruImageSubobject (oI ≫ FC.filToAssocGraded s k)
  67   let u := Projective.factorThru b p
  68   let c := u ≫ oI
  69   let a := Projective.factorThru
  70     (u ≫ Subobject.ofLE I (imageSubobject d) inf_le_left)
  71     (factorThruImageSubobject d)
  72   refine ⟨a, c, ?_, ?_⟩
  73   · calc
  74       a ≫ (FC.fil (s - ↑n + 1) (k + 1)).arrow ≫ FC.complex.d (k + 1) k = a ≫ d := rfl
  75       _ = (a ≫ factorThruImageSubobject d) ≫ (imageSubobject d).arrow := by
  76         rw [Category.assoc, imageSubobject_arrow_comp]
  77       _ = (u ≫ Subobject.ofLE I (imageSubobject d) inf_le_left) ≫
  78           (imageSubobject d).arrow := by rw [Projective.factorThru_comp]
  79       _ = c ≫ (FC.fil s k).arrow := by
  80         simp only [c, oI, Category.assoc, Subobject.ofLE_arrow]
  81   · calc
  82       c ≫ FC.filToAssocGraded s k =
  83           u ≫ p ≫ (FC.boundarySubobject s k ↑n).arrow := by
  84             change (u ≫ oI) ≫ FC.filToAssocGraded s k =
  85               u ≫ factorThruImageSubobject (oI ≫ FC.filToAssocGraded s k) ≫
  86                 (imageSubobject (oI ≫ FC.filToAssocGraded s k)).arrow
  87             simp only [Category.assoc, imageSubobject_arrow_comp]
  88       _ = b ≫ (FC.boundarySubobject s k ↑n).arrow := by
  89         rw [← Category.assoc, Projective.factorThru_comp]

  91 /-- Equal target page classes differ by an actual boundary on a projective
  92 test object, so a higher-filtration source correction makes the equation strict. -/
  93 theorem nonempty_fiber_of_differentialRelation
  94     (FC : FilteredComplex C) (bnd : FC.IsBounded) (n : ℕ) (s k : ℤ)
  95     {T : C} [Projective T]
  96     {x : T ⟶ FC.assocGraded s k}
  97     {y : T ⟶ FC.assocGraded (s + ↑n) (k - 1)}
  98     (h : DifferentialRelation (FC.toSpectralSequence bnd) n (s, k) x y) :
  99     Nonempty (Fiber FC n s k x y) := by
 100   obtain ⟨zx, hx, zy, hy, hd⟩ := h
 101   change T ⟶ Subobject.underlying.obj (FC.cycleSubobject s k ↑n) at zx
 102   change T ⟶ Subobject.underlying.obj (FC.cycleSubobject (s + ↑n) (k - 1) ↑n) at zy
 103   change zx ≫ (FC.cycleSubobject s k ↑n).arrow = x at hx
 104   change zy ≫ (FC.cycleSubobject (s + ↑n) (k - 1) ↑n).arrow = y at hy
 105   obtain ⟨xl, yl, hxl, hxy⟩ := exists_strict_lifts_of_cycle FC s k n zx
 106   have hstrict : xl ≫ (FC.differential_preserves s k).choose =
 107       yl ≫ FC.filtration.inclusion (by omega) (k - 1) := by
 108     apply (cancel_mono (FC.fil s (k - 1)).arrow).mp
 109     simpa only [Category.assoc, fil, d,
 110       (FC.differential_preserves s k).choose_spec,
 111       Filtration.inclusion_arrow] using hxy
 112   obtain ⟨zx', zy', hx', hy', hd'⟩ :=
 113     FC.pageDifferential_of_strict_lifts s k n xl yl hstrict
 114   have hzx : zx' = zx := by
 115     apply (cancel_mono (FC.cycleSubobject s k ↑n).arrow).mp
 116     exact hx'.trans hxl
 117   subst zx'
 118   have heq : (zy' - zy) ≫ FC.pageπ (s + ↑n) (k - 1) ↑n = 0 := by
 119     rw [Preadditive.sub_comp, sub_eq_zero, ← hd']
 120     change zx ≫ _ ≫ (FC.toPreSS bnd).d (n : ℤ) (s, k) = _ at hd
 121     rw [FC.toPreSS_d_nat bnd] at hd
 122     exact hd
 123   let B := FC.boundarySubobject (s + ↑n) (k - 1) ↑n
 124   let Z := FC.cycleSubobject (s + ↑n) (k - 1) ↑n
 125   let inclusion := Subobject.ofLE B Z (FC.B_le_Z_aux (s + ↑n) (k - 1) ↑n)
 126   let b := Abelian.monoLift inclusion (zy' - zy) heq
 127   have hb : b ≫ B.arrow =
 128       yl ≫ FC.filToAssocGraded (s + ↑n) (k - 1) - y := by
 129     calc
 130       b ≫ B.arrow = (b ≫ inclusion) ≫ Z.arrow := by
 131         rw [Category.assoc, Subobject.ofLE_arrow]
 132       _ = (zy' - zy) ≫ Z.arrow := by rw [Abelian.monoLift_comp]
 133       _ = yl ≫ FC.filToAssocGraded (s + ↑n) (k - 1) - y := by
 134         rw [Preadditive.sub_comp, hy', hy]
 135   have hboundary :
 136       ∃ (a : T ⟶ Subobject.underlying.obj (FC.fil (s + 1) k))
 137         (c : T ⟶ Subobject.underlying.obj (FC.fil (s + ↑n) (k - 1))),
 138         a ≫ (FC.fil (s + 1) k).arrow ≫ FC.d k =
 139           c ≫ (FC.fil (s + ↑n) (k - 1)).arrow ∧
 140         c ≫ FC.filToAssocGraded (s + ↑n) (k - 1) = b ≫ B.arrow := by
 141     have transport : ∀ (t l : ℤ), t = s + ↑n - ↑n + 1 → l = k - 1 + 1 →
 142         ∃ (a : T ⟶ Subobject.underlying.obj (FC.fil t l))
 143           (c : T ⟶ Subobject.underlying.obj (FC.fil (s + ↑n) (k - 1))),
 144           a ≫ (FC.fil t l).arrow ≫ FC.complex.d l (k - 1) =
 145             c ≫ (FC.fil (s + ↑n) (k - 1)).arrow ∧
 146           c ≫ FC.filToAssocGraded (s + ↑n) (k - 1) = b ≫ B.arrow := by
 147       intro t l ht hl
 148       subst t
 149       subst l
 150       exact exists_lifts_of_boundary FC (s + ↑n) (k - 1) n b
 151     exact transport (s + 1) k (by omega) (by omega)
 152   obtain ⟨a, c, hac, hc⟩ := hboundary
 153   let inc := Subobject.ofLE (FC.fil (s + 1) k) (FC.fil s k) (FC.fil_anti s k)
 154   have hac' : (a ≫ inc) ≫ (FC.fil s k).arrow ≫ FC.d k =
 155       c ≫ (FC.fil (s + ↑n) (k - 1)).arrow := by
 156     calc
 157       (a ≫ inc) ≫ (FC.fil s k).arrow ≫ FC.d k =
 158           a ≫ (FC.fil (s + 1) k).arrow ≫ FC.d k := by
 159             simpa only [Category.assoc] using
 160               congrArg (fun q => a ≫ q ≫ FC.d k)
 161                 (Subobject.ofLE_arrow (FC.fil_anti s k))
 162       _ = c ≫ (FC.fil (s + ↑n) (k - 1)).arrow := hac
 163   refine ⟨⟨(xl - a ≫ inc, yl - c), (mem_fiber_iff _).mpr ⟨?_, ?_, ?_⟩⟩⟩
 164   · change (xl - a ≫ inc) ≫ FC.filToAssocGraded s k = x
 165     rw [Preadditive.sub_comp, Category.assoc]
 166     have hinc : inc ≫ FC.filToAssocGraded s k = 0 := cokernel.condition inc
 167     rw [hinc, comp_zero, sub_zero, hxl]
 168     exact hx
 169   · change (yl - c) ≫ FC.filToAssocGraded (s + ↑n) (k - 1) = y
 170     rw [Preadditive.sub_comp, hc, hb]
 171     abel
 172   · simp only [Preadditive.sub_comp]
 173     rw [hxy, hac']

 175 /-- The strict solution fiber and the canonical page relation agree for every
 176 projective test object. In ModuleCat this includes the free rank-one ULift ℤ. -/
 177 theorem differentialRelation_iff_nonempty_fiber
 178     (FC : FilteredComplex C) (bnd : FC.IsBounded) (n : ℕ) (s k : ℤ)
 179     {T : C} [Projective T]
 180     (x : T ⟶ FC.assocGraded s k)
 181     (y : T ⟶ FC.assocGraded (s + ↑n) (k - 1)) :
 182     DifferentialRelation (FC.toSpectralSequence bnd) n (s, k) x y ↔
 183       Nonempty (Fiber FC n s k x y) := by
 184   constructor
 185   · exact nonempty_fiber_of_differentialRelation FC bnd n s k
 186   · rintro ⟨a⟩
 187     exact differentialRelation_of_fiber FC bnd n s k a

 189 /-- 在投射测试对象上，有限页的商页微分关系与严格过滤代表元关系等价。 -/
 190 theorem differentialRelation_iff_representativeRelation
 191     (FC : FilteredComplex C) (bnd : FC.IsBounded) (n : ℕ) (s k : ℤ)
 192     {T : C} [Projective T]
 193     (x : T ⟶ FC.assocGraded s k)
 194     (y : T ⟶ FC.assocGraded (s + ↑n) (k - 1)) :
 195     DifferentialRelation (FC.toSpectralSequence bnd) n (s, k) x y ↔
 196       FC.RepresentativeRelation n (Int.natCast_nonneg n) s k x y := by
 197   rw [differentialRelation_iff_nonempty_fiber,
 198     nonempty_fiber_iff (Int.natCast_nonneg n)]
