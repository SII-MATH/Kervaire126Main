  16 /-- An actual solution gives the canonical internal page differential. -/
  17 theorem differentialRelation_of_fiber (FC : FilteredComplex C) (bnd : FC.IsBounded)
  18     (n : ℕ) (s k : ℤ) {T : C}
  19     {x : T ⟶ FC.assocGraded s k}
  20     {y : T ⟶ FC.assocGraded (s + (n : ℤ)) (k - 1)}
  21     (a : Fiber FC n s k x y) :
  22     DifferentialRelation (FC.toSpectralSequence bnd) n (s, k) x y := by
  23   obtain ⟨xl, yl, hx, hy, hd⟩ :=
  24     (nonempty_fiber_iff (by omega : (0 : ℤ) ≤ n)).mp ⟨a⟩
  25   obtain ⟨zx, zy, hzx, hzy, hdiff⟩ :=
  26     FC.pageDifferential_of_strict_lifts s k n xl yl hd
  27   refine ⟨zx, hzx.trans hx, zy, hzy.trans hy, ?_⟩
  28   change zx ≫ _ ≫ (FC.toPreSS bnd).d (n : ℤ) (s, k) = _
  29   rw [FC.toPreSS_d_nat bnd]
  30   exact hdiff
