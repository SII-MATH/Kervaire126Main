   9 /-!
  10 # Filtered chain complexes
  11 
  12 A filtered chain complex is a Mathlib `ChainComplex` equipped with a decreasing
  13 filtration preserved by the differential.  Unlike the historical KIP core, this
  14 file does not define a second spectral-sequence record or a nested `Z/B`
  15 presentation.  It supplies the associated graded differential needed by a
  16 future construction through Mathlib's spectral-object machinery.
  17 -/
  18 
  19 namespace KIP126.Core.SpectralSequence

  21 open CategoryTheory CategoryTheory.Limits

  23 universe u v
  24 
  25 variable {C : Type u} [Category.{v} C] [Abelian C]

  27 /-- A chain complex together with a decreasing filtration preserved by its
  28 differential. -/
  29 structure FilteredComplex (C : Type u) [Category.{v} C] [Abelian C] where
  30   /-- The underlying Mathlib chain complex. -/
  31   complex : ChainComplex C ℤ
  32   /-- A decreasing filtration of the underlying graded object. -/
  33   filtration : Filtration complex.X
  34   /-- The differential maps every filtration level into the same level in the
  35 next chain degree. -/
  36   differential_preserves : ∀ (s k : ℤ),
  37     ∃ φ : Subobject.underlying.obj (filtration.F s k) ⟶
  38         Subobject.underlying.obj (filtration.F s (k - 1)),
  39       φ ≫ (filtration.F s (k - 1)).arrow =
  40         (filtration.F s k).arrow ≫ complex.d k (k - 1)

  44 /-! ### Homology of the underlying complex
  45 
  46 The following construction keeps the chain-degree convention explicit.  A
  47 homology object at degree `k` is the homology of the short complex
  48 `X (k + 1) ⟶ X k ⟶ X (k - 1)`. -/
  49 
  50 noncomputable def homologyShortComplex (FC : FilteredComplex C) (k : ℤ) :
  51     ShortComplex C :=
  52   ShortComplex.mk
  53     (FC.complex.d (k + 1) k)
  54     (FC.complex.d k (k - 1))
  55     (FC.complex.d_comp_d (k + 1) k (k - 1))
  56 
  57 /-- The homology graded object of the underlying filtered chain complex. -/
  58 noncomputable def homologyObj (FC : FilteredComplex C) : ℤ → C :=
  59   fun k => (FC.homologyShortComplex k).homology
  60 
  61 @[simp]
  62 lemma homologyObj_apply (FC : FilteredComplex C) (k : ℤ) :
  63     FC.homologyObj k = (FC.homologyShortComplex k).homology := rfl
  64 
  65 /-! ### The induced filtration on homology -/
  66 
  67 noncomputable def homologyFiltration (FC : FilteredComplex C) :
  68     Filtration FC.homologyObj where
  69   F s k :=
  70     let S := FC.homologyShortComplex k
  71     let I := kernelSubobject S.g ⊓ FC.filtration.F s k
  72     let hzero : I.arrow ≫ S.g = 0 := by
  73       rw [show I.arrow = Subobject.ofLE I (kernelSubobject S.g) inf_le_left ≫
  74         (kernelSubobject S.g).arrow from (Subobject.ofLE_arrow inf_le_left).symm]
  75       rw [Category.assoc, kernelSubobject_arrow_comp, comp_zero]
  76     imageSubobject (S.liftCycles I.arrow hzero ≫ S.homologyπ)
  77   mono := by
  78     intro s k
  79     let S := FC.homologyShortComplex k
  80     let I₁ := kernelSubobject S.g ⊓ FC.filtration.F (s + 1) k
  81     let I₀ := kernelSubobject S.g ⊓ FC.filtration.F s k
  82     have hle : I₁ ≤ I₀ := inf_le_inf_left _ (FC.filtration.mono s k)
  83     have hzero₁ : I₁.arrow ≫ S.g = 0 := by
  84       rw [show I₁.arrow = Subobject.ofLE I₁ (kernelSubobject S.g) inf_le_left ≫
  85         (kernelSubobject S.g).arrow from (Subobject.ofLE_arrow inf_le_left).symm]
  86       rw [Category.assoc, kernelSubobject_arrow_comp, comp_zero]
  87     have hzero₀ : I₀.arrow ≫ S.g = 0 := by
  88       rw [show I₀.arrow = Subobject.ofLE I₀ (kernelSubobject S.g) inf_le_left ≫
  89         (kernelSubobject S.g).arrow from (Subobject.ofLE_arrow inf_le_left).symm]
  90       rw [Category.assoc, kernelSubobject_arrow_comp, comp_zero]
  91     have hlift : S.liftCycles I₁.arrow hzero₁ =
  92         Subobject.ofLE I₁ I₀ hle ≫ S.liftCycles I₀.arrow hzero₀ := by
  93       apply (cancel_mono S.iCycles).mp
  94       rw [S.liftCycles_i, Category.assoc, S.liftCycles_i]
  95       exact (Subobject.ofLE_arrow hle).symm
  96     have hlift' : S.liftCycles I₁.arrow hzero₁ ≫ S.homologyπ =
  97         Subobject.ofLE I₁ I₀ hle ≫
  98           (S.liftCycles I₀.arrow hzero₀ ≫ S.homologyπ) := by
  99       rw [← Category.assoc, hlift]
 100     change imageSubobject (S.liftCycles I₁.arrow hzero₁ ≫ S.homologyπ) ≤
 101       imageSubobject (S.liftCycles I₀.arrow hzero₀ ≫ S.homologyπ)
 102     rw [hlift']
 103     exact imageSubobject_comp_le _ _

 107 /-- The differential from degree `k + 1` to degree `k`.  The canonical chain
 108 complex already indexes its target by `k`, so no degree transport is needed. -/
 109 noncomputable def dToK (FC : FilteredComplex C) (k : ℤ) :
 110     FC.complex.X (k + 1) ⟶ FC.complex.X k :=
 111   FC.complex.d (k + 1) k
 112 
 113 /-- The `r`-cycle subobject in the associated graded piece. -/
 114 noncomputable def cycleSubobject (FC : FilteredComplex C)
 115     (s k : ℤ) (r : WithTop ℕ) :
 116     Subobject (FC.filtration.associatedGraded s k) :=
 117   let πV := FC.filtration.toAssociatedGraded s k
 118   match r with
 119   | ⊤ =>
 120     let f := (FC.filtration.F s k).arrow ≫ FC.complex.d k (k - 1)
 121     imageSubobject ((kernelSubobject f).arrow ≫ πV)
 122   | (n : ℕ) =>
 123     let f := (FC.filtration.F s k).arrow ≫ FC.complex.d k (k - 1) ≫
 124       cokernel.π ((FC.filtration.F (s + ↑n) (k - 1)).arrow)
 125     imageSubobject ((kernelSubobject f).arrow ≫ πV)
 126 
 127 /-- The `r`-boundary subobject in the associated graded piece. -/
 128 noncomputable def boundarySubobject (FC : FilteredComplex C)
 129     (s k : ℤ) (r : WithTop ℕ) :
 130     Subobject (FC.filtration.associatedGraded s k) :=
 131   let πV := FC.filtration.toAssociatedGraded s k
 132   match r with
 133   | ⊤ =>
 134     let imgD := imageSubobject (FC.dToK k)
 135     let I := imgD ⊓ FC.filtration.F s k
 136     imageSubobject (Subobject.ofLE I (FC.filtration.F s k) inf_le_right ≫ πV)
 137   | (n : ℕ) =>
 138     let imgD := imageSubobject
 139       ((FC.filtration.F (s - ↑n + 1) (k + 1)).arrow ≫ FC.dToK k)
 140     let I := imgD ⊓ FC.filtration.F s k
 141     imageSubobject (Subobject.ofLE I (FC.filtration.F s k) inf_le_right ≫ πV)
 142 
 143 /-- The transported differential squares to zero. -/
 144 theorem dToK_comp_d (FC : FilteredComplex C) (k : ℤ) :
 145     FC.dToK k ≫ FC.complex.d k (k - 1) = 0 := by
 146   exact FC.complex.d_comp_d (k + 1) k (k - 1)

 148 /-- Boundaries are cycles in every associated graded piece. -/
 149 theorem B_le_Z_aux (FC : FilteredComplex C) (s k : ℤ)
 150     (r : WithTop ℕ) :
 151     FC.boundarySubobject s k r ≤ FC.cycleSubobject s k r := by
 152   rcases r with _ | n
 153   · simp only [boundarySubobject, cycleSubobject]
 154     let ι := Subobject.ofLE (FC.filtration.F (s + 1) k)
 155       (FC.filtration.F s k) (FC.filtration.mono s k)
 156     let πV := cokernel.π ι
 157     let imgD := imageSubobject (FC.dToK k)
 158     let I := imgD ⊓ FC.filtration.F s k
 159     let oI := Subobject.ofLE I (FC.filtration.F s k) inf_le_right
 160     let f := (FC.filtration.F s k).arrow ≫ FC.complex.d k (k - 1)
 161     have h_zero : oI ≫ f = 0 := by
 162       change Subobject.ofLE I (FC.filtration.F s k) inf_le_right ≫
 163         (FC.filtration.F s k).arrow ≫ FC.complex.d k (k - 1) = 0
 164       rw [← Category.assoc, Subobject.ofLE_arrow]
 165       rw [show I.arrow =
 166           Subobject.ofLE I imgD inf_le_left ≫ imgD.arrow
 167         from (Subobject.ofLE_arrow inf_le_left).symm,
 168         Category.assoc]
 169       have h : imgD.arrow ≫ FC.complex.d k (k - 1) = 0 := by
 170         rw [← cancel_epi (factorThruImageSubobject (FC.dToK k))]
 171         rw [comp_zero, ← Category.assoc, imageSubobject_arrow_comp]
 172         exact FC.dToK_comp_d k
 173       rw [h, comp_zero]
 174     have heq : oI ≫ πV = factorThruKernelSubobject f oI h_zero ≫
 175         (kernelSubobject f).arrow ≫ πV := by
 176       rw [← Category.assoc, factorThruKernelSubobject_comp_arrow]
 177     calc imageSubobject (oI ≫ πV)
 178         = imageSubobject (factorThruKernelSubobject f oI h_zero ≫
 179             (kernelSubobject f).arrow ≫ πV) := by rw [heq]
 180       _ ≤ imageSubobject ((kernelSubobject f).arrow ≫ πV) :=
 181           imageSubobject_comp_le _ _
 182   · simp only [boundarySubobject, cycleSubobject]
 183     let ι := Subobject.ofLE (FC.filtration.F (s + 1) k)
 184       (FC.filtration.F s k) (FC.filtration.mono s k)
 185     let πV := cokernel.π ι
 186     let imgD := imageSubobject
 187       ((FC.filtration.F (s - ↑n + 1) (k + 1)).arrow ≫ FC.dToK k)
 188     let I := imgD ⊓ FC.filtration.F s k
 189     let oI := Subobject.ofLE I (FC.filtration.F s k) inf_le_right
 190     let f := (FC.filtration.F s k).arrow ≫ FC.complex.d k (k - 1) ≫
 191       cokernel.π ((FC.filtration.F (s + ↑n) (k - 1)).arrow)
 192     have h_zero : oI ≫ f = 0 := by
 193       change Subobject.ofLE I (FC.filtration.F s k) inf_le_right ≫
 194         ((FC.filtration.F s k).arrow ≫ FC.complex.d k (k - 1) ≫
 195           cokernel.π ((FC.filtration.F (s + ↑n) (k - 1)).arrow)) = 0
 196       simp only [← Category.assoc]
 197       rw [Subobject.ofLE_arrow]
 198       rw [show I.arrow =
 199           Subobject.ofLE I imgD inf_le_left ≫ imgD.arrow
 200         from (Subobject.ofLE_arrow inf_le_left).symm]
 201       simp only [Category.assoc]
 202       have hd : imgD.arrow ≫ FC.complex.d k (k - 1) = 0 := by
 203         rw [← cancel_epi (factorThruImageSubobject
 204           ((FC.filtration.F (s - ↑n + 1) (k + 1)).arrow ≫ FC.dToK k))]
 205         rw [comp_zero, ← Category.assoc, imageSubobject_arrow_comp,
 206           Category.assoc, FC.dToK_comp_d, comp_zero]
 207       rw [show imgD.arrow ≫ FC.complex.d k (k - 1) ≫ cokernel.π _ =
 208         (imgD.arrow ≫ FC.complex.d k (k - 1)) ≫ cokernel.π _
 209         from (Category.assoc _ _ _).symm,
 210         hd, zero_comp, comp_zero]
 211     have heq : oI ≫ πV = factorThruKernelSubobject f oI h_zero ≫
 212         (kernelSubobject f).arrow ≫ πV := by
 213       rw [← Category.assoc, factorThruKernelSubobject_comp_arrow]
 214     calc imageSubobject (oI ≫ πV)
 215         = imageSubobject (factorThruKernelSubobject f oI h_zero ≫
 216             (kernelSubobject f).arrow ≫ πV) := by rw [heq]
 217       _ ≤ imageSubobject ((kernelSubobject f).arrow ≫ πV) :=
 218           imageSubobject_comp_le _ _

 220 /-- A chain map between filtered complexes which preserves every filtration
 221 level.  The factorisations are kept as explicit data: this is the categorical
 222 version of a filtered chain map and does not identify a filtered subobject with
 223 its ambient object. -/
 224 structure Morphism (FC GD : FilteredComplex C) where
 225   /-- The underlying Mathlib chain map. -/
 226   map : FC.complex ⟶ GD.complex
 227   /-- Factorisations of every component through the target filtration. -/
 228   preserves : ∀ (s k : ℤ),
 229     ∃ φ : Subobject.underlying.obj (FC.filtration.F s k) ⟶
 230         Subobject.underlying.obj (GD.filtration.F s k),
 231       φ ≫ (GD.filtration.F s k).arrow =
 232         (FC.filtration.F s k).arrow ≫ map.f k

 238 @[ext]
 239 lemma ext {f g : Morphism FC GD} (h : f.map = g.map) : f = g := by
 240   cases f
 241   cases g
 242   cases h
 243   rfl

 245 /-- View a filtered chain map as a filtered morphism of the underlying graded
 246 objects. -/
 247 def toFilteredMorphism (f : Morphism FC GD) :
 248     InternalFilteredMorphism FC.filtration GD.filtration where
 249   map := fun k => f.map.f k
 250   preserves := f.preserves
 251 
 252 /-- The identity filtered chain map. -/
 253 def id (FC : FilteredComplex C) : Morphism FC FC where
 254   map := 𝟙 FC.complex
 255   preserves := fun s k => ⟨𝟙 _, by simp⟩
 256 
 257 /-- Composition of filtered chain maps. -/
 258 def comp (f : Morphism FC GD) (g : Morphism GD GE) : Morphism FC GE where
 259   map := f.map ≫ g.map
 260   preserves := fun s k => by
 261     refine ⟨(f.preserves s k).choose ≫ (g.preserves s k).choose, ?_⟩
 262     simp only [HomologicalComplex.comp_f, Category.assoc]
 263     rw [(g.preserves s k).choose_spec, ← Category.assoc,
 264       (f.preserves s k).choose_spec]
 265     simp only [Category.assoc]
 266 
 267 end Morphism

 287 /-- The differential induced on a fixed associated graded piece. -/
 288 noncomputable def associatedGradedDifferential (FC : FilteredComplex C)
 289     (s k : ℤ) :
 290     FC.filtration.associatedGraded s k ⟶
 291       FC.filtration.associatedGraded s (k - 1) := by
 292   unfold Filtration.associatedGraded
 293   exact cokernel.desc _
 294     ((FC.differential_preserves s k).choose ≫
 295       cokernel.π (Subobject.ofLE (FC.filtration.F (s + 1) (k - 1))
 296         (FC.filtration.F s (k - 1)) (FC.filtration.mono s (k - 1))))
 297     (by
 298       have factor :
 299           Subobject.ofLE (FC.filtration.F (s + 1) k) (FC.filtration.F s k)
 300               (FC.filtration.mono s k) ≫
 301             (FC.differential_preserves s k).choose =
 302           (FC.differential_preserves (s + 1) k).choose ≫
 303             Subobject.ofLE (FC.filtration.F (s + 1) (k - 1))
 304               (FC.filtration.F s (k - 1)) (FC.filtration.mono s (k - 1)) := by
 305         have mono_arrow : Mono (FC.filtration.F s (k - 1)).arrow := inferInstance
 306         have lhs_eq :
 307             (Subobject.ofLE (FC.filtration.F (s + 1) k) (FC.filtration.F s k)
 308                 (FC.filtration.mono s k) ≫
 309               (FC.differential_preserves s k).choose) ≫
 310                 (FC.filtration.F s (k - 1)).arrow =
 311               (FC.filtration.F (s + 1) k).arrow ≫ FC.complex.d k (k - 1) := by
 312           rw [Category.assoc, (FC.differential_preserves s k).choose_spec,
 313             ← Category.assoc, Subobject.ofLE_arrow]
 314         have rhs_eq :
 315             ((FC.differential_preserves (s + 1) k).choose ≫
 316               Subobject.ofLE (FC.filtration.F (s + 1) (k - 1))
 317                 (FC.filtration.F s (k - 1)) (FC.filtration.mono s (k - 1))) ≫
 318                 (FC.filtration.F s (k - 1)).arrow =
 319               (FC.filtration.F (s + 1) k).arrow ≫ FC.complex.d k (k - 1) := by
 320           rw [Category.assoc, Subobject.ofLE_arrow,
 321             (FC.differential_preserves (s + 1) k).choose_spec]
 322         exact mono_arrow.right_cancellation _ _ (lhs_eq.trans rhs_eq.symm)
 323       rw [← Category.assoc, factor, Category.assoc, cokernel.condition, comp_zero])
 324 
 325 /-- The associated graded differential squares to zero. -/
 326 theorem associatedGradedDifferential_sq (FC : FilteredComplex C)
 327     (s k : ℤ) :
 328     FC.associatedGradedDifferential s k ≫
 329       FC.associatedGradedDifferential s (k - 1) = 0 := by
 330   unfold associatedGradedDifferential Filtration.associatedGraded
 331   apply (cancel_epi (cokernel.π (Subobject.ofLE
 332     (FC.filtration.F (s + 1) k) (FC.filtration.F s k)
 333     (FC.filtration.mono s k)))).mp
 334   simp only [comp_zero, id_eq]
 335   rw [← Category.assoc, cokernel.π_desc, Category.assoc, cokernel.π_desc]
 336   suffices h : (FC.differential_preserves s k).choose ≫
 337       (FC.differential_preserves s (k - 1)).choose = 0 by
 338     rw [← Category.assoc, h, zero_comp]
 339   have mono_arrow : Mono (FC.filtration.F s (k - 1 - 1)).arrow := inferInstance
 340   apply mono_arrow.right_cancellation
 341   rw [zero_comp, Category.assoc,
 342     (FC.differential_preserves s (k - 1)).choose_spec,
 343     ← Category.assoc, (FC.differential_preserves s k).choose_spec,
 344     Category.assoc, FC.complex.d_comp_d, comp_zero]

 346 /-- The chosen restrictions of the original differential square to zero on
 347 every filtration level. -/
 348 lemma differential_preserves_sq (FC : FilteredComplex C) (s k : ℤ) :
 349     (FC.differential_preserves s k).choose ≫
 350         (FC.differential_preserves s (k - 1)).choose = 0 := by
 351   have mono_arrow : Mono (FC.filtration.F s (k - 1 - 1)).arrow := inferInstance
 352   apply mono_arrow.right_cancellation
 353   rw [zero_comp, Category.assoc,
 354     (FC.differential_preserves s (k - 1)).choose_spec,
 355     ← Category.assoc, (FC.differential_preserves s k).choose_spec,
 356     Category.assoc, FC.complex.d_comp_d, comp_zero]

 358 lemma differential_preserves_sq_transport
 359     (FC : FilteredComplex C) (s a b c : ℤ)
 360     (hab : a - 1 = b) (hbc : b - 1 = c) :
 361     ((FC.differential_preserves s a).choose ≫
 362         eqToHom (congrArg (fun k =>
 363           Subobject.underlying.obj (FC.filtration.F s k)) hab)) ≫
 364       (FC.differential_preserves s b).choose ≫
 365         eqToHom (congrArg (fun k =>
 366           Subobject.underlying.obj (FC.filtration.F s k)) hbc) = 0 := by
 367   subst b
 368   subst c
 369   simpa only [eqToHom_refl, Category.comp_id, Category.assoc] using
 370     FC.differential_preserves_sq s a

 372 /-- The chain complex given by one filtration level. -/
 373 noncomputable def filteredChainComplex (FC : FilteredComplex C) (s : ℤ) :
 374     ChainComplex C ℤ :=
 375   ChainComplex.of
 376     (fun k => Subobject.underlying.obj (FC.filtration.F s k))
 377     (fun k => (FC.differential_preserves s (k + 1)).choose ≫
 378       eqToHom (congrArg (fun j =>
 379         Subobject.underlying.obj (FC.filtration.F s j))
 380         (by omega : (k + 1) - 1 = k)))
 381     (fun k => by
 382       apply differential_preserves_sq_transport FC s (k + 1 + 1) (k + 1) k <;>
 383         omega)

 399 /-- The canonical inclusion between two filtration levels is a chain map on
 400 the restricted complexes (before the harmless integer-index transports are
 401 inserted by `ChainComplex.of`). -/
 402 lemma filtration_inclusion_comp_differential (FC : FilteredComplex C)
 403     {t s : ℤ} (h : t ≤ s) (k : ℤ) :
 404     FC.filtration.inclusion h (k + 1) ≫
 405         (FC.differential_preserves t (k + 1)).choose =
 406       (FC.differential_preserves s (k + 1)).choose ≫
 407         FC.filtration.inclusion h (k + 1 - 1) := by
 408   apply (cancel_mono (FC.filtration.F t (k + 1 - 1)).arrow).mp
 409   calc
 410     (FC.filtration.inclusion h (k + 1) ≫
 411           (FC.differential_preserves t (k + 1)).choose) ≫
 412         (FC.filtration.F t (k + 1 - 1)).arrow =
 413       FC.filtration.inclusion h (k + 1) ≫
 414         ((FC.differential_preserves t (k + 1)).choose ≫
 415           (FC.filtration.F t (k + 1 - 1)).arrow) := by
 416       simp only [Category.assoc]
 417     _ = FC.filtration.inclusion h (k + 1) ≫
 418         ((FC.filtration.F t (k + 1)).arrow ≫
 419           FC.complex.d (k + 1) (k + 1 - 1)) := by
 420       rw [(FC.differential_preserves t (k + 1)).choose_spec]
 421     _ = (FC.filtration.F s (k + 1)).arrow ≫
 422           FC.complex.d (k + 1) (k + 1 - 1) := by
 423       rw [← Category.assoc, FC.filtration.inclusion_arrow]
 424     _ = ((FC.differential_preserves s (k + 1)).choose ≫
 425           (FC.filtration.F s (k + 1 - 1)).arrow) := by
 426       rw [(FC.differential_preserves s (k + 1)).choose_spec]
 427     _ = ((FC.differential_preserves s (k + 1)).choose ≫
 428           FC.filtration.inclusion h (k + 1 - 1)) ≫
 429           (FC.filtration.F t (k + 1 - 1)).arrow := by
 430       rw [Category.assoc, FC.filtration.inclusion_arrow]

 432 /-- The inclusion of one filtration level into a lower level, regarded as a
 433 chain map between the restricted complexes.  The target of the map is the
 434 level `t` complex, so the hypothesis is oriented as `t ≤ s`. -/
 435 noncomputable def filtrationInclusionChainMap (FC : FilteredComplex C)
 436     {t s : ℤ} (h : t ≤ s) :
 437     FC.filteredChainComplex s ⟶ FC.filteredChainComplex t :=
 438   ChainComplex.ofHom
 439     (fun k => FC.filtration.inclusion h k)
 440     (fun k => by
 441       dsimp [filteredChainComplex, ChainComplex.of.d]
 442       simp only [if_true, Category.id_comp]
 443       -- First use preservation of the ambient differential.  What remains is
 444       -- just the canonical transport from `(k + 1) - 1` to `k`.
 445       rw [← Category.assoc, FC.filtration_inclusion_comp_differential h k]
 446       apply (cancel_mono (FC.filtration.F t k).arrow).mp
 447       let hk : (k + 1) - 1 = k := by omega
 448       have ht := FC.filtration.transport_arrow t hk
 449       have hs := FC.filtration.transport_arrow s hk

 619 /-- The filtration-level factorisations of a filtered chain map commute with
 620 the restricted differentials. -/
 621 lemma preserves_differential (f : Morphism FC GD) (s k : ℤ) :
 622     (f.preserves s k).choose ≫
 623         (GD.differential_preserves s k).choose =
 624       (FC.differential_preserves s k).choose ≫
 625         (f.preserves s (k - 1)).choose := by
 626   have hF := (f.preserves s k).choose_spec
 627   have hF' := (f.preserves s (k - 1)).choose_spec
 628   have hG := (GD.differential_preserves s k).choose_spec
 629   have hD := (FC.differential_preserves s k).choose_spec
 630   have hchain := f.map.comm k (k - 1)
 631   apply (cancel_mono (GD.filtration.F s (k - 1)).arrow).mp
 632   calc
 633     ((f.preserves s k).choose ≫
 634         (GD.differential_preserves s k).choose) ≫
 635         (GD.filtration.F s (k - 1)).arrow =
 636       (f.preserves s k).choose ≫
 637         ((GD.differential_preserves s k).choose ≫
 638           (GD.filtration.F s (k - 1)).arrow) := by simp only [Category.assoc]
 639     _ = (f.preserves s k).choose ≫
 640         ((GD.filtration.F s k).arrow ≫ GD.complex.d k (k - 1)) := by
 641       rw [hG]
 642     _ = ((FC.filtration.F s k).arrow ≫ f.map.f k) ≫
 643         GD.complex.d k (k - 1) := by
 644       rw [← Category.assoc, hF]
 645     _ = ((FC.filtration.F s k).arrow ≫ FC.complex.d k (k - 1)) ≫
 646         f.map.f (k - 1) := by
 647       rw [Category.assoc, hchain, ← Category.assoc]
 648     _ = ((FC.differential_preserves s k).choose ≫
 649         (FC.filtration.F s (k - 1)).arrow) ≫ f.map.f (k - 1) := by
 650       rw [hD]
 651     _ = (FC.differential_preserves s k).choose ≫
 652         ((f.preserves s (k - 1)).choose ≫
 653           (GD.filtration.F s (k - 1)).arrow) := by
 654       simpa only [Category.assoc] using
 655         congrArg (fun q => (FC.differential_preserves s k).choose ≫ q) hF'.symm
 656     _ = ((FC.differential_preserves s k).choose ≫
 657         (f.preserves s (k - 1)).choose) ≫
 658         (GD.filtration.F s (k - 1)).arrow := by simp only [Category.assoc]

 746 filtration-level chain diagrams. -/
 747 noncomputable def Morphism.filteredChainDiagramNatTrans (f : Morphism FC GD) :
 748     FC.filteredChainDiagram ⟶ GD.filteredChainDiagram where
 749   app s := f.filteredChainMap s
 750   naturality {s t} α := by
 751     apply HomologicalComplex.Hom.ext
 752     funext k
 753     change FC.filtration.inclusion α.le k ≫ (f.preserves t k).choose =
 754       (f.preserves s k).choose ≫ GD.filtration.inclusion α.le k
 755     apply (cancel_mono (GD.filtration.F t k).arrow).mp

 757       (FC.filtration.inclusion α.le k ≫ (f.preserves t k).choose) ≫

 760             ((f.preserves t k).choose ≫ (GD.filtration.F t k).arrow) := by
 761         simp only [Category.assoc]
 762       _ = FC.filtration.inclusion α.le k ≫
 763             ((FC.filtration.F t k).arrow ≫ f.map.f k) := by
 764         rw [(f.preserves t k).choose_spec]
 765       _ = (FC.filtration.F s k).arrow ≫ f.map.f k := by
 766         rw [← Category.assoc, FC.filtration.inclusion_arrow]
 767       _ = ((f.preserves s k).choose ≫
 768             (GD.filtration.F s k).arrow) := by
 769         rw [(f.preserves s k).choose_spec]
 770       _ = ((f.preserves s k).choose ≫

 903   subst b
 904   simpa only [eqToHom_refl, Category.comp_id, Category.assoc] using
 905     f.associatedGradedMap_comm s a

 907 /-- The chain map on associated graded complexes induced by a filtered chain
 908 map. -/
 909 noncomputable def Morphism.associatedGradedChainMap (f : Morphism FC GD) (s : ℤ) :
 910     FC.associatedGradedComplex s ⟶ GD.associatedGradedComplex s :=
 911   ChainComplex.ofHom
 912     (fun k => f.associatedGradedMap s k)
 913     (fun k => by
 914       dsimp [associatedGradedComplex, ChainComplex.of.d]
 915       simp only [Category.id_comp]
 916       apply associatedGradedMap_comm_transport f s (k + 1) k
 917       omega)
 918 
 919 @[simp]
 920 lemma Morphism.associatedGradedChainMap_id (FC : FilteredComplex C) (s : ℤ) :
 921     (Morphism.id FC).associatedGradedChainMap s = 𝟙 _ := by
 922   apply HomologicalComplex.Hom.ext
 923   funext k
 924   change (Morphism.id FC).associatedGradedMap s k = 𝟙 _
 925   exact Morphism.associatedGradedMap_id FC s k

 927 @[simp]
 928 lemma Morphism.associatedGradedChainMap_comp (f : Morphism FC GD)
 929     (g : Morphism GD GE) (s : ℤ) :
 930     (Morphism.comp f g).associatedGradedChainMap s =
 931       f.associatedGradedChainMap s ≫ g.associatedGradedChainMap s := by
 932   apply HomologicalComplex.Hom.ext
 933   funext k
 934   change (Morphism.comp f g).associatedGradedMap s k =
 935     f.associatedGradedMap s k ≫ g.associatedGradedMap s k
 936   exact Morphism.associatedGradedMap_comp f g s k
