   8 /-!
   9 # Spectral-sequence assembly for a filtered complex
  10 
  11 The data are supplied by `FilteredComplex.toPreSS`; this layer discharges the
  12 three spectral-sequence laws using the finite-page proofs.
  13 -/
  14 
  15 namespace KIP126.Core.SpectralSequence
  16 
  17 open CategoryTheory CategoryTheory.Limits
  18 
  19 universe u v
  20 
  21 variable {C : Type u} [Category.{v} C] [Abelian C]
  22 
  23 namespace FilteredComplex
  24 
  25 set_option maxHeartbeats 6400000 in
  26 /-- The nested-subobject spectral sequence canonically built from a bounded
  27 filtered complex. -/
  28 noncomputable def toSpectralSequence (FC : FilteredComplex C)
  29     (bnd : FC.IsBounded) : SpectralSequence C (ℤ × ℤ) :=
  30   SpectralSequence.ofPreSS (FC.toPreSS bnd)
  31     (fun r ⟨s, k⟩ => by
  32       by_cases hr : 0 ≤ r
  33       · obtain ⟨n, rfl⟩ := Int.eq_ofNat_of_zero_le hr
  34         simp only [FC.toPreSS_d_nat bnd]
  35         change FC.pageDifferential s k n ≫
  36           FC.pageDifferential (s + ↑n) (k - 1) n = 0
  37         exact FC.pageDifferential_comp s k n
  38       · dsimp only [FilteredComplex.toPreSS]
  39         rw [dif_neg hr]
  40         exact zero_comp)
  41     (fun r ⟨s, k⟩ hr => by
  42       have hr0 : 0 ≤ r := by
  43         simpa only [FilteredComplex.toPreSS] using hr
  44       obtain ⟨n, rfl⟩ := Int.eq_ofNat_of_zero_le hr0
  45       simp only [FC.toPreSS_d_nat bnd]
  46       change kernelSubobject (FC.pageDifferential s k n) =
  47         imageSubobject (Subobject.ofLE
  48           (FC.cycleSubobject s k ↑(n + 1)) (FC.cycleSubobject s k ↑n)
  49           (FC.cycleSubobject_antitone s k
  50             (by exact_mod_cast Nat.le_succ n)) ≫ FC.pageπ s k ↑n)
  51       apply le_antisymm
  52       · exact pageDifferential_Z_succ_le FC s k n
  53       · exact pageDifferential_Z_succ_ge FC s k n)
  54     (fun r ⟨s, k⟩ hr => by
  55       have hr0 : 0 ≤ r := by
  56         simpa only [FilteredComplex.toPreSS] using hr
  57       obtain ⟨n, rfl⟩ := Int.eq_ofNat_of_zero_le hr0
  58       simp only [FC.toPreSS_d_nat bnd]
  59       change imageSubobject (FC.pageDifferential s k n) =
  60         imageSubobject (Subobject.ofLE
  61           (FC.boundarySubobject (s + ↑n) (k - 1) ↑(n + 1))
  62           (FC.cycleSubobject (s + ↑n) (k - 1) ↑n)
  63           (le_trans (FC.B_le_Z_aux (s + ↑n) (k - 1) ↑(n + 1))
  64             (FC.cycleSubobject_antitone (s + ↑n) (k - 1)
  65               (by exact_mod_cast Nat.le_succ n))) ≫
  66           FC.pageπ (s + ↑n) (k - 1) ↑n)
  67       exact FC.pageDifferential_B_succ s k n)
  68 
  69 end FilteredComplex
  70 
  71 end KIP126.Core.SpectralSequence
