  22 /-- Historical two-sided boundedness record, with its original field names. -/
  23 structure IsBounded (FC : FilteredComplex C) where
  24   /-- Degreewise lower bound. -/
  25   lo : ℤ → ℤ
  26   /-- Degreewise upper bound. -/
  27   hi : ℤ → ℤ
  28   /-- The bounds occur in the expected order. -/
  29   lo_le_hi : ∀ k, lo k ≤ hi k
  30   /-- Low filtration levels are the whole object. -/
  31   boundedBelow : ∀ k s, s ≤ lo k → FC.fil s k = ⊤
  32   /-- High filtration levels are zero. -/
  33   boundedAbove : ∀ k s, hi k ≤ s → FC.fil s k = ⊥
  34 
  35 /-- A filtered representative `xl` lifts the associated-graded element `x`. -/
  36 def IsLift (FC : FilteredComplex C) {T : C} (s k : ℤ)
  37     (xl : T ⟶ Subobject.underlying.obj (FC.fil s k))
  38     (x : T ⟶ FC.assocGraded s k) : Prop :=
  39   xl ≫ FC.filToAssocGraded s k = x
  40 
  41 /-- A strict filtered-lift relation between associated-graded representatives. -/
  42 def RepresentativeRelation (FC : FilteredComplex C)
  43     (r : ℤ) (hr : 0 ≤ r) (s k : ℤ) {T : C}
  44     (x : T ⟶ FC.filtration.associatedGraded s k)
  45     (y : T ⟶ FC.filtration.associatedGraded (s + r) (k - 1)) : Prop :=
  46   ∃ (xl : T ⟶ Subobject.underlying.obj (FC.filtration.F s k))
  47     (yl : T ⟶ Subobject.underlying.obj (FC.filtration.F (s + r) (k - 1))),
  48     xl ≫ FC.filtration.toAssociatedGraded s k = x ∧
  49     yl ≫ FC.filtration.toAssociatedGraded (s + r) (k - 1) = y ∧
  50     xl ≫ FC.filDiff s k =
  51       yl ≫ FC.filtration.inclusion (by omega) (k - 1)
  52 
  53 end FilteredComplex
