  10 /-!
  11 # Historical filtered-complex data API
  12 
  13 This module exposes the data names used by `KIPBase.SpectralSequence.FilteredComplex`
  14 on top of KIP126's canonical Mathlib `ChainComplex`-based implementation.  It does
  15 not introduce a second filtered-complex structure: the old fields are compatibility
  16 accessors for `complex` and `filtration`.
  17 -/
  18 
  19 namespace KIP126.Core.SpectralSequence
  20 
  21 open CategoryTheory CategoryTheory.Limits
  22 
  23 universe u v
  24 
  25 variable {C : Type u} [Category.{v} C] [Abelian C]
  26 
  27 namespace FilteredComplex
  28 
  29 /-- Historical name for the underlying graded object. -/
  30 @[reducible] def A (FC : FilteredComplex C) : ℤ → C :=
  31   FC.complex.X
  32 
  33 /-- Historical one-index presentation of the chain differential. -/
  34 noncomputable def d (FC : FilteredComplex C) (k : ℤ) :
  35     FC.A k ⟶ FC.A (k - 1) :=
  36   FC.complex.d k (k - 1)
  37 
  38 /-- Historical name for the filtration subobjects. -/
  39 @[reducible] def fil (FC : FilteredComplex C) (s k : ℤ) : Subobject (FC.A k) :=
  40   FC.filtration.F s k
  41 
  42 /-- The chosen restriction of the differential to one filtration level. -/
  43 noncomputable def filDiff (FC : FilteredComplex C) (s k : ℤ) :
  44     Subobject.underlying.obj (FC.fil s k) ⟶
  45       Subobject.underlying.obj (FC.fil s (k - 1)) :=
  46   (FC.differential_preserves s k).choose
  47 
  48 /-- Historical name for an associated-graded piece. -/
  49 @[reducible] noncomputable def assocGraded (FC : FilteredComplex C) (s k : ℤ) : C :=
  50   FC.filtration.associatedGraded s k
  51 
  52 /-- Historical name for the quotient map onto an associated-graded piece. -/
  53 @[reducible] noncomputable def filToAssocGraded (FC : FilteredComplex C) (s k : ℤ) :
  54     Subobject.underlying.obj (FC.fil s k) ⟶ FC.assocGraded s k :=
  55   FC.filtration.toAssociatedGraded s k
  56 
  57 /-- Historical name for the differential induced on the associated graded. -/
  58 @[reducible] noncomputable def assocGradedDiff (FC : FilteredComplex C) (s k : ℤ) :
  59     FC.assocGraded s k ⟶ FC.assocGraded s (k - 1) :=
  60   FC.associatedGradedDifferential s k
  61 
  62 end FilteredComplex
  63 
  64 end KIP126.Core.SpectralSequence
