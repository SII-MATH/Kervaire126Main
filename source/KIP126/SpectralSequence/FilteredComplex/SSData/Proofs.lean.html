  22 /-- Actual filtered lifts give a representative-level differential relation. -/
  23 theorem representativeRelation_of_lifts
  24     (FC : FilteredComplex C) (r : ℤ) (hr : 0 ≤ r) (s k : ℤ)
  25     {T : C}
  26     (xl : T ⟶ Subobject.underlying.obj (FC.filtration.F s k))
  27     (yl : T ⟶ Subobject.underlying.obj (FC.filtration.F (s + r) (k - 1)))
  28     (hd : xl ≫ FC.filDiff s k =
  29       yl ≫ FC.filtration.inclusion (by omega) (k - 1)) :
  30     RepresentativeRelation FC r hr s k
  31       (xl ≫ FC.filtration.toAssociatedGraded s k)
  32       (yl ≫ FC.filtration.toAssociatedGraded (s + r) (k - 1)) := by
  33   exact ⟨xl, yl, rfl, rfl, hd⟩

  68 /-- Historical square-zero statement for the one-index differential. -/
  69 theorem d_comp_d (FC : FilteredComplex C) (k : ℤ) :
  70     FC.d k ≫ FC.d (k - 1) = 0 := by
  71   change FC.complex.d k (k - 1) ≫
  72     FC.complex.d (k - 1) ((k - 1) - 1) = 0
  73   exact FC.complex.d_comp_d k (k - 1) ((k - 1) - 1)
  74 
  75 /-- Naturality of the chain differential under transport of its target index. -/
  76 theorem complex_d_comp_eqToHom (FC : FilteredComplex C)
  77     (i : ℤ) {j j' : ℤ} (h : j = j') :
  78     FC.complex.d i j ≫ eqToHom (congrArg FC.complex.X h) =
  79       FC.complex.d i j' := by
  80   subst j'
  81   simp

  83 /-- Historical successor antitonicity statement for the filtration. -/
  84 theorem fil_anti (FC : FilteredComplex C) (s k : ℤ) :
  85     FC.fil (s + 1) k ≤ FC.fil s k :=
  86   FC.filtration.mono s k

  88 /-- Historical filtration-preservation witness. -/
  89 theorem d_preserves_fil (FC : FilteredComplex C) (s k : ℤ) :
  90     ∃ φ : Subobject.underlying.obj (FC.fil s k) ⟶
  91         Subobject.underlying.obj (FC.fil s (k - 1)),
  92       φ ≫ (FC.fil s (k - 1)).arrow = (FC.fil s k).arrow ≫ FC.d k :=
  93   FC.differential_preserves s k

  95 /-- Compatibility of the associated-graded differential with the quotient maps. -/
  96 theorem assocGradedDiff_compat (FC : FilteredComplex C) (s k : ℤ) :
  97     FC.filToAssocGraded s k ≫ FC.assocGradedDiff s k =
  98       FC.filDiff s k ≫ FC.filToAssocGraded s (k - 1) :=
  99   FC.toAssociatedGraded_comp_associatedGradedDifferential s k

 101 /-- Transport the associated-graded differential along equality of degrees. -/
 102 theorem assocGradedDiff_cast (FC : FilteredComplex C)
 103     (s : ℤ) {k₁ k₂ : ℤ} (hk : k₁ = k₂) :
 104     FC.assocGradedDiff s k₁ =
 105       eqToHom (by rw [hk]) ≫ FC.assocGradedDiff s k₂ ≫ eqToHom (by rw [hk]) := by
 106   subst hk
 107   simp

 109 /-- The associated-graded differential squares to zero. -/
 110 theorem assocGradedDiff_sq (FC : FilteredComplex C) (s k : ℤ) :
 111     FC.assocGradedDiff s k ≫ FC.assocGradedDiff s (k - 1) = 0 :=
 112   FC.associatedGradedDifferential_sq s k

 114 /-- Antitonicity between arbitrary filtration indices. -/
 115 theorem fil_anti_of_le (FC : FilteredComplex C) {a b : ℤ} (k : ℤ)
 116     (hab : a ≤ b) : FC.fil b k ≤ FC.fil a k :=

 119 /-- Two lifts of the same associated-graded element differ through the next
 120 filtration level.  This discharges the corresponding historical placeholder. -/
 121 theorem isLift_sub_factors (FC : FilteredComplex C) {T : C}
 122     (s k : ℤ) {xl1 xl2 : T ⟶ Subobject.underlying.obj (FC.fil s k)}
 123     {x : T ⟶ FC.assocGraded s k}
 124     (h₁ : FC.IsLift s k xl1 x) (h₂ : FC.IsLift s k xl2 x) :
 125     Subobject.Factors (FC.fil (s + 1) k)
 126       ((xl1 - xl2) ≫ (FC.fil s k).arrow) := by
 127   let ι := Subobject.ofLE (FC.fil (s + 1) k) (FC.fil s k) (FC.fil_anti s k)
 128   have hzero : (xl1 - xl2) ≫ FC.filToAssocGraded s k = 0 := by
 129     rw [Preadditive.sub_comp, h₁, h₂, sub_self]
 130   let z := Abelian.monoLift ι (xl1 - xl2) hzero
 131   have hz : z ≫ (FC.fil (s + 1) k).arrow =
 132       (xl1 - xl2) ≫ (FC.fil s k).arrow := by
 133     calc
 134     z ≫ (FC.fil (s + 1) k).arrow =
 135         (z ≫ ι) ≫ (FC.fil s k).arrow := by
 136           rw [Category.assoc, Subobject.ofLE_arrow]
 137     _ = (xl1 - xl2) ≫ (FC.fil s k).arrow := by
 138       rw [Abelian.monoLift_comp]
 139   rw [← hz]
 140   exact Subobject.factors_comp_arrow z
