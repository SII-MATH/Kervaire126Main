  25 /-- 第零页的边缘没有非零广义元素。 -/
  26 theorem boundary_zero_apply (FC : FilteredComplex C)
  27     (s k : ℤ) {T : C} [Projective T]
  28     {v : T ⟶ FC.assocGraded s (k - 1)}
  29     (hv : Subobject.Factors (FC.boundarySubobject s (k - 1) 0) v) :
  30     v = 0 := by
  31   have htmp := FC.lift_boundary s (k - 1) 0 hv
  32   rw [show s - (↑(0 : ℕ) : ℤ) + 1 = s + 1 by omega] at htmp
  33   obtain ⟨a₀, b, hdb₀, hbb⟩ := htmp
  34   let e : Subobject.underlying.obj (FC.fil (s + 1) k) =
  35       Subobject.underlying.obj (FC.fil (s + 1) (k - 1 + 1)) := by
  36     rw [show k - 1 + 1 = k by omega]
  37   let a := a₀ ≫ eqToHom e.symm
  38   have he : eqToHom e ≫
  39       ((FC.fil (s + 1) (k - 1 + 1)).arrow ≫ FC.dToK (k - 1)) =
  40       (FC.fil (s + 1) k).arrow ≫ FC.d k := by
  41     have htransport : ∀ {i j : ℤ} (h : i = j),
  42         eqToHom (congrArg
  43           (fun q => Subobject.underlying.obj (FC.fil (s + 1) q)) h) ≫
  44           ((FC.fil (s + 1) j).arrow ≫ FC.complex.d j (k - 1)) =
  45         (FC.fil (s + 1) i).arrow ≫ FC.complex.d i (k - 1) := by
  46       intro i j h
  47       subst j
  48       simp
  49     have hk : k = k - 1 + 1 := by omega
  50     have heq : e = congrArg
  51         (fun q => Subobject.underlying.obj (FC.fil (s + 1) q)) hk :=
  52       Subsingleton.elim _ _
  53     rw [heq]
  54     simpa only [FilteredComplex.dToK, FilteredComplex.d] using htransport hk
  55   have hc : (a₀ ≫ eqToHom e.symm) ≫ eqToHom e = a₀ := by
  56     rw [Category.assoc, eqToHom_trans, eqToHom_refl, Category.comp_id]
  57   have hdb : a ≫ (FC.fil (s + 1) k).arrow ≫ FC.d k =
  58       b ≫ (FC.fil s (k - 1)).arrow := by
  59     calc
  60       a ≫ (FC.fil (s + 1) k).arrow ≫ FC.d k =
  61           a ≫ (eqToHom e ≫
  62             ((FC.fil (s + 1) (k - 1 + 1)).arrow ≫ FC.dToK (k - 1))) := by
  63               rw [he]
  64       _ = a₀ ≫ (FC.fil (s + 1) (k - 1 + 1)).arrow ≫ FC.dToK (k - 1) := by
  65         simp only [a, ← Category.assoc]
  66         rw [hc]
  67       _ = b ≫ (FC.fil s (k - 1)).arrow := hdb₀
  68   let j := Subobject.ofLE (FC.fil (s + 1) (k - 1))
  69     (FC.fil s (k - 1)) (FC.fil_anti s (k - 1))
  70   let c := a ≫ FC.filDiff (s + 1) k
  71   have hbc : b = c ≫ j := by
  72     apply (cancel_mono (FC.fil s (k - 1)).arrow).mp
  73     calc
  74       b ≫ (FC.fil s (k - 1)).arrow =
  75           a ≫ (FC.fil (s + 1) k).arrow ≫ FC.d k := hdb.symm
  76       _ = c ≫ (FC.fil (s + 1) (k - 1)).arrow := by
  77         simp only [c, Category.assoc, filDiff, fil, d,
  78           (FC.differential_preserves (s + 1) k).choose_spec]
  79       _ = (c ≫ j) ≫ (FC.fil s (k - 1)).arrow := by
  80         simp only [Category.assoc, j, Subobject.ofLE_arrow]
  81   calc
  82     v = b ≫ FC.filToAssocGraded s (k - 1) := hbb.symm
  83     _ = c ≫ j ≫ FC.filToAssocGraded s (k - 1) := by
  84       simp only [hbc, Category.assoc]
  85     _ = 0 := by
  86       have hj : j ≫ FC.filToAssocGraded s (k - 1) = 0 := by
  87         dsimp only [j, FilteredComplex.filToAssocGraded]
  88         exact cokernel.condition _
  89       rw [hj, comp_zero]

  91 /-- 若严格过滤微分的目标在相应页不是边缘，则它给出本质微分关系。 -/
  92 theorem essentialRelation_of_filtered_lift
  93     (FC : FilteredComplex C) (bnd : FC.IsBounded)
  94     (t p k : ℤ) (m : ℕ) (ht : t + ↑m = p) {T : C} [Projective T]
  95     (a : T ⟶ Subobject.underlying.obj (FC.fil t k))
  96     (b : T ⟶ Subobject.underlying.obj (FC.fil p (k - 1)))
  97     (hdb : a ≫ (FC.fil t k).arrow ≫ FC.d k =
  98       b ≫ (FC.fil p (k - 1)).arrow)
  99     (hn : ¬ Subobject.Factors (FC.boundarySubobject p (k - 1) (↑m))
 100       (b ≫ FC.filToAssocGraded p (k - 1))) :
 101     EssentialDifferentialRelation (FC.toSpectralSequence bnd)
 102         (↑m) ⟨t, k⟩
 103         (Eq.mpr (congrArg (fun X : C => T ⟶ X)
 104           (show (((FC.toSpectralSequence bnd).ssData ⟨t, k⟩).V =
 105             FC.assocGraded t k) from rfl)) (a ≫ FC.filToAssocGraded t k))
 106         (Eq.mpr (congrArg (fun X : C => T ⟶ X)
 107           (show (((FC.toSpectralSequence bnd).ssData
 108             (⟨t, k⟩ + (FC.toSpectralSequence bnd).diffDeg (↑m))).V =
 109             FC.assocGraded p (k - 1)) by rw [← ht]; rfl))
 110               (b ≫ FC.filToAssocGraded p (k - 1))) := by
 111   subst p
 112   let j := Subobject.ofLE (FC.fil (t + ↑m) (k - 1))
 113     (FC.fil t (k - 1)) (FC.fil_anti_of_le (k - 1) (by omega))
 114   have hd : a ≫ FC.filDiff t k = b ≫ j := by
 115     apply (cancel_mono (FC.fil t (k - 1)).arrow).mp
 116     calc
 117       (a ≫ FC.filDiff t k) ≫ (FC.fil t (k - 1)).arrow =
 118           a ≫ (FC.fil t k).arrow ≫ FC.d k := by
 119             simp only [Category.assoc, filDiff, fil, d,
 120               (FC.differential_preserves t k).choose_spec]
 121       _ = b ≫ (FC.fil (t + ↑m) (k - 1)).arrow := hdb
 122       _ = (b ≫ j) ≫ (FC.fil t (k - 1)).arrow := by
 123         simp only [Category.assoc, j, Subobject.ofLE_arrow]
 124   constructor
 125   · exact (Solutions.differentialRelation_iff_representativeRelation
 126       FC bnd m t k _ _).2 ⟨a, b, rfl, rfl, hd⟩
 127   · change ¬ Subobject.Factors
 128       (FC.boundarySubobject (t + ↑m) (k - 1) (↑m))
 129         (b ≫ FC.filToAssocGraded (t + ↑m) (k - 1))
 130     exact hn

 132 /-- 非零有限页边缘来自更高源过滤层的本质微分。 -/
 133 theorem essential_ancestor_of_nonzero_boundary
 134     (FC : FilteredComplex C) (bnd : FC.IsBounded) (k : ℤ)
 135     (m : ℕ) (t p : ℤ) (ht : t + (m : ℤ) = p)
 136     {T : C} [Projective T]
 137     (z : T ⟶ FC.assocGraded p (k - 1)) (hz : z ≠ 0)
 138     (hb : (FC.boundarySubobject p (k - 1) (↑m)).Factors z) :
 139     ∃ (u : ℤ) (j : ℕ) (huj : u + (j : ℤ) = p)
 140       (x' : T ⟶ FC.assocGraded u k),
 141       t < u ∧ EssentialDifferentialRelation (FC.toSpectralSequence bnd)
 142         (↑j) ⟨u, k⟩
 143         (Eq.mpr (congrArg (fun X : C => T ⟶ X)
 144           (show (((FC.toSpectralSequence bnd).ssData ⟨u, k⟩).V =
 145             FC.assocGraded u k) from rfl)) x')
 146         (Eq.mpr (congrArg (fun X : C => T ⟶ X)
 147           (show (((FC.toSpectralSequence bnd).ssData
 148             (⟨u, k⟩ + (FC.toSpectralSequence bnd).diffDeg (↑j))).V =
 149             FC.assocGraded p (k - 1)) by rw [← huj]; rfl)) z) := by
 150   induction m generalizing t p with
 151   | zero =>
 152     cases ht
 153     have hz₀ : z = 0 := by
 154       simpa only [Int.cast_zero, add_zero] using
 155         (FC.boundary_zero_apply (t + (0 : ℤ)) k hb)
 156     exact (hz hz₀).elim
 157   | succ n ih =>
 158     cases ht
 159     obtain ⟨a, b, hdb, hbb⟩ :=
 160       FC.lift_boundary_at_differential t k (n + 1) hb
 161     have ht' : (t + 1) + (n : ℤ) = t + ((n + 1 : ℕ) : ℤ) := by omega
 162     by_cases hbn : (FC.boundarySubobject (t + ((n + 1 : ℕ) : ℤ))
 163         (k - 1) (↑n)).Factors z
 164     · obtain ⟨u, j, huj, x', htu, hess⟩ :=
 165         ih (t + 1) (t + ((n + 1 : ℕ) : ℤ)) ht' z hz hbn
 166       exact ⟨u, j, huj, x', by omega, hess⟩
 167     · have hnot : ¬ (FC.boundarySubobject (t + ((n + 1 : ℕ) : ℤ))
 168           (k - 1) (↑n)).Factors
 169           (b ≫ FC.filToAssocGraded (t + ((n + 1 : ℕ) : ℤ)) (k - 1)) := by
 170         rwa [hbb]
 171       have hess := FC.essentialRelation_of_filtered_lift
 172         bnd (t + 1) (t + ((n + 1 : ℕ) : ℤ)) k n ht' a b hdb hnot
 173       rw [hbb] at hess
 174       exact ⟨t + 1, n, ht', a ≫ FC.filToAssocGraded (t + 1) k,
 175         by omega, hess⟩

 177 /-- 同一有限页微分的两个不同环境目标代表元产生精确 crossing。 -/
 178 theorem differentialRelation_crossed_of_two_exact
 179     (FC : FilteredComplex C) (bnd : FC.IsBounded) (n : ℕ) (s k : ℤ) {T : C}
 180     [Projective T]
 181     {x : T ⟶ FC.assocGraded s k}
 182     {y₁ y₂ : T ⟶ FC.assocGraded (s + ↑n) (k - 1)}
 183     (h₁ : DifferentialRelation (FC.toSpectralSequence bnd) n ⟨s, k⟩ x y₁)
 184     (h₂ : DifferentialRelation (FC.toSpectralSequence bnd) n ⟨s, k⟩ x y₂)
 185     (hne : y₁ ≠ y₂) :
 186     RelationCrossedByAt (FC.toSpectralSequence bnd)
 187       (fun q => q.1) n ⟨s, k⟩ x y₁ h₁ := by
 188   have hdiff : y₁ - y₂ ≠ 0 := sub_ne_zero.mpr hne
 189   have hboundary := DifferentialRelation.targets_sub_factors_boundary
 190     (FC.toSpectralSequence bnd) n ⟨s, k⟩ h₁ h₂
 191   change Subobject.Factors
 192     (FC.boundarySubobject (s + ↑n) (k - 1) (↑n)) (y₁ - y₂) at hboundary
 193   obtain ⟨u, j, huj, x', hsu, hess⟩ :=
 194     FC.essential_ancestor_of_nonzero_boundary bnd k n s (s + ↑n)
 195       (by rfl) (y₁ - y₂) hdiff hboundary
 196   refine ⟨u - s, by omega, (↑j), ⟨u, k⟩, _, _, ?_, hess, ?_⟩
 197   · change u = s + (u - s)
 198     omega
 199   · change u + ↑j = s + ↑n
 200     exact huj

 202 /-- 同一有限页微分的两个不同目标代表元产生 crossing。 -/
 203 theorem differentialRelation_crossed_of_two
 204     (FC : FilteredComplex C) (bnd : FC.IsBounded) (n : ℕ) (s k : ℤ) {T : C}
 205     [Projective T]
 206     {x : T ⟶ FC.assocGraded s k}
 207     {y₁ y₂ : T ⟶ FC.assocGraded (s + ↑n) (k - 1)}
 208     (h₁ : DifferentialRelation (FC.toSpectralSequence bnd) n ⟨s, k⟩ x y₁)
 209     (h₂ : DifferentialRelation (FC.toSpectralSequence bnd) n ⟨s, k⟩ x y₂)
 210     (hne : y₁ ≠ y₂) :
 211     RelationCrossedBy (FC.toSpectralSequence bnd)
 212       (fun q => q.1) n ⟨s, k⟩ x y₁ h₁ :=
 213   RelationCrossedByAt.toCrossed _ _ _ _
 214     (FC.differentialRelation_crossed_of_two_exact bnd n s k h₁ h₂ hne)

 216 /-- 两个严格代表元关系具有不同目标时，其对应的页微分关系产生精确 crossing。 -/
 217 theorem representativeRelation_crossed_of_two_exact
 218     (FC : FilteredComplex C) (bnd : FC.IsBounded) (n : ℕ) (s k : ℤ) {T : C}
 219     [Projective T]
 220     {x : T ⟶ FC.assocGraded s k}
 221     {y₁ y₂ : T ⟶ FC.assocGraded (s + ↑n) (k - 1)}
 222     (h₁ : FC.RepresentativeRelation n (Int.natCast_nonneg n) s k x y₁)
 223     (h₂ : FC.RepresentativeRelation n (Int.natCast_nonneg n) s k x y₂)
 224     (hne : y₁ ≠ y₂) :
 225     RelationCrossedByAt (FC.toSpectralSequence bnd)
 226       (fun q => q.1) n ⟨s, k⟩ x y₁
 227         ((Solutions.differentialRelation_iff_representativeRelation
 228           FC bnd n s k x y₁).2 h₁) := by
 229   let d₁ := (Solutions.differentialRelation_iff_representativeRelation
 230     FC bnd n s k x y₁).2 h₁
 231   let d₂ := (Solutions.differentialRelation_iff_representativeRelation
 232     FC bnd n s k x y₂).2 h₂
 233   exact FC.differentialRelation_crossed_of_two_exact bnd n s k d₁ d₂ hne

 251 /-- 若一个代表元级微分关系没有 crossing，则同一源的严格目标代表元唯一。 -/
 252 theorem representativeRelation_target_eq_of_not_crossed
 253     (FC : FilteredComplex C) (bnd : FC.IsBounded) (n : ℕ) (s k : ℤ) {T : C}
 254     [Projective T]
 255     {x : T ⟶ FC.assocGraded s k}
 256     {y₁ y₂ : T ⟶ FC.assocGraded (s + ↑n) (k - 1)}
 257     (h₁ : FC.RepresentativeRelation n (Int.natCast_nonneg n) s k x y₁)
 258     (h₂ : FC.RepresentativeRelation n (Int.natCast_nonneg n) s k x y₂)
 259     (hnc : ¬ RelationCrossedBy (FC.toSpectralSequence bnd)
 260       (fun q => q.1) n ⟨s, k⟩ x y₁
 261         ((Solutions.differentialRelation_iff_representativeRelation
 262           FC bnd n s k x y₁).2 h₁)) :
 263     y₁ = y₂ := by
 264   by_contra hne
 265   exact hnc (FC.representativeRelation_crossed_of_two bnd n s k h₁ h₂ hne)
