   7 /-!
   8 # Historical homology target of a filtered complex
   9 
  10 The current KIP126 homology target uses Mathlib's normalized chain degrees.
  11 This layer retains the definitionally indexed target used by the historical
  12 weak-convergence proof, without changing the canonical `homologyObj` API.
  13 -/
  14 
  15 namespace KIP126.Core.SpectralSequence
  16 
  17 open CategoryTheory CategoryTheory.Limits
  18 
  19 universe u v
  20 
  21 variable {C : Type u} [Category.{v} C] [Abelian C]
  22 
  23 namespace FilteredComplex
  24 
  25 /-- The historical presentation of the homology object at degree `k`. -/
  26 @[reducible] noncomputable def homologySSObj (FC : FilteredComplex C) (k : ℤ) : C :=
  27   (ShortComplex.mk (FC.d (k + 1)) (FC.d (k + 1 - 1))
  28     (FC.d_comp_d (k + 1))).homology
  29 
  30 /-- The filtration on the historical homology presentation used by the
  31 nested-subobject convergence construction. -/
  32 noncomputable def homologySSFiltration (FC : FilteredComplex C) :
  33     Filtration FC.homologySSObj where
  34   F := fun s k =>
  35     let S := ShortComplex.mk (FC.d (k + 1)) (FC.d (k + 1 - 1))
  36       (FC.d_comp_d (k + 1))
  37     let I := kernelSubobject S.g ⊓ FC.fil s (k + 1 - 1)
  38     let h_zero : I.arrow ≫ S.g = 0 := by
  39       rw [show I.arrow = Subobject.ofLE I (kernelSubobject S.g)
  40           inf_le_left ≫ (kernelSubobject S.g).arrow
  41         from (Subobject.ofLE_arrow inf_le_left).symm]
  42       rw [Category.assoc, kernelSubobject_arrow_comp, comp_zero]
  43     imageSubobject (S.liftCycles I.arrow h_zero ≫ S.homologyπ)
  44   mono := fun s k => by
  45     dsimp only []
  46     set S := ShortComplex.mk (FC.d (k + 1)) (FC.d (k + 1 - 1))
  47       (FC.d_comp_d (k + 1))
  48     set I1 := kernelSubobject S.g ⊓ FC.fil (s + 1) (k + 1 - 1)
  49     set I0 := kernelSubobject S.g ⊓ FC.fil s (k + 1 - 1)
  50     have hle : I1 ≤ I0 := inf_le_inf_left _ (FC.fil_anti s (k + 1 - 1))
  51     have h_zero_1 : I1.arrow ≫ S.g = 0 := by
  52       rw [show I1.arrow = Subobject.ofLE I1 (kernelSubobject S.g) inf_le_left ≫
  53         (kernelSubobject S.g).arrow from (Subobject.ofLE_arrow inf_le_left).symm]
  54       rw [Category.assoc, kernelSubobject_arrow_comp, comp_zero]
  55     have h_zero_0 : I0.arrow ≫ S.g = 0 := by
  56       rw [show I0.arrow = Subobject.ofLE I0 (kernelSubobject S.g) inf_le_left ≫
  57         (kernelSubobject S.g).arrow from (Subobject.ofLE_arrow inf_le_left).symm]
  58       rw [Category.assoc, kernelSubobject_arrow_comp, comp_zero]
  59     have hlift : S.liftCycles I1.arrow h_zero_1 =
  60         Subobject.ofLE I1 I0 hle ≫ S.liftCycles I0.arrow h_zero_0 := by
  61       apply (cancel_mono S.iCycles).mp
  62       rw [S.liftCycles_i, Category.assoc, S.liftCycles_i]
  63       exact (Subobject.ofLE_arrow hle).symm
  64     have hlift2 : S.liftCycles I1.arrow h_zero_1 ≫ S.homologyπ =
  65         Subobject.ofLE I1 I0 hle ≫
  66           (S.liftCycles I0.arrow h_zero_0 ≫ S.homologyπ) := by
  67       rw [← Category.assoc, hlift]
  68     conv_lhs => rw [show S.liftCycles I1.arrow _ ≫ S.homologyπ =
  69       Subobject.ofLE I1 I0 hle ≫
  70         (S.liftCycles I0.arrow _ ≫ S.homologyπ) from hlift2]
  71     exact imageSubobject_comp_le _ _
  72 
  73 end FilteredComplex
  74 
  75 end KIP126.Core.SpectralSequence
