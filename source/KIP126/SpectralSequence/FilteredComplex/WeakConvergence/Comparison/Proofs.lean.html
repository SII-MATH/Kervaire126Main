  10 # Infinity-page comparison for weak convergence
  11 
  12 This proof identifies the infinity page of the bounded filtered-complex
  13 spectral sequence with the associated graded of its homology filtration.
  14 The convergence record is assembled in the parent `Data.lean` module.

  25 /-- The infinity page is isomorphic to the associated graded of filtered homology. -/
  26 theorem FilteredComplex.weakConvergenceIso_nonempty (FC : FilteredComplex C)
  27     (bnd : FC.IsBounded) (s k : ℤ) :
  28     Nonempty ((FC.toSSData bnd s k).eInfty ≅
  29       FC.homologySSFiltration.associatedGraded s k) := by
  30     refine ⟨?_⟩
  31     unfold SSData.eInfty SSData.page Filtration.associatedGraded
  32     -- Goal: cokernel(ofLE (B⊤) (Z⊤) _) ≅ cokernel(ofLE (F_{s+1}H_k) (F_sH_k) _)
  33     -- Abbreviations
  34     let S := ShortComplex.mk (FC.d (k + 1)) (FC.d (k + 1 - 1)) (FC.d_comp_d (k + 1))
  35     let e_deg : FC.A k = FC.A (k + 1 - 1) := congr_arg FC.A (show k = k + 1 - 1 by omega)
  36     let ι_fil := Subobject.ofLE (FC.fil (s + 1) k) (FC.fil s k) (FC.fil_anti s k)
  37     let πV := cokernel.π ι_fil
  38     let f_Z := (FC.fil s k).arrow ≫ FC.d k
  39     let kerZ := kernelSubobject f_Z
  40     -- φ_to_X2 : kerZ.underlying → S.X₂ (= FC.A(k+1-1))
  41     let φ_to_X2 := kerZ.arrow ≫ (FC.fil s k).arrow ≫ eqToHom e_deg
  42     have h_lift : φ_to_X2 ≫ S.g = 0 := by
  43       change (kerZ.arrow ≫ (FC.fil s k).arrow ≫ eqToHom e_deg) ≫ S.g = 0
  44       simp only [Category.assoc]
  45       have : eqToHom e_deg ≫ S.g =
  46           FC.d k ≫ eqToHom (congr_arg FC.A (show k - 1 = (k + 1 - 1) - 1 by omega)) := by
  47         simp [S]
  48       rw [this, ← Category.assoc, ← Category.assoc,
  49         show (kerZ.arrow ≫ (FC.fil s k).arrow) ≫ FC.d k = kerZ.arrow ≫ f_Z
  50           from by simp only [f_Z, Category.assoc],
  51         kernelSubobject_arrow_comp, zero_comp]
  52     -- φ_pre : kerZ.underlying → S.homology
  53     let φ_pre := S.liftCycles φ_to_X2 h_lift ≫ S.homologyπ
  54     -- I_s for the homologyFiltration
  55     let I_s := kernelSubobject S.g ⊓ FC.fil s (k + 1 - 1)
  56     have h_zero_s : I_s.arrow ≫ S.g = 0 := by
  57       rw [show I_s.arrow = Subobject.ofLE I_s (kernelSubobject S.g)
  58           inf_le_left ≫ (kernelSubobject S.g).arrow
  59         from (Subobject.ofLE_arrow inf_le_left).symm,
  60         Category.assoc, kernelSubobject_arrow_comp, comp_zero]
  61     let gen_s := S.liftCycles I_s.arrow h_zero_s ≫ S.homologyπ
  62     -- Show φ_to_X2 factors through I_s.arrow, so φ_pre factors through gen_s
  63     -- This means imageSubobject(φ_pre) ≤ imageSubobject(gen_s) = F_s
  64     have h_factors_I : I_s.Factors φ_to_X2 := by
  65       rw [show I_s = kernelSubobject S.g ⊓ FC.fil s (k + 1 - 1) from rfl,
  66           Subobject.inf_factors φ_to_X2]
  67       refine ⟨kernelSubobject_factors S.g φ_to_X2 h_lift, ?_⟩
  68       -- Need: (FC.fil s (k+1-1)).Factors φ_to_X2
  69       -- φ_to_X2 = kerZ.arrow ≫ (fil s k).arrow ≫ eqToHom(e_deg)
  70       -- We use: (fil s k).arrow ≫ eqToHom = eqToHom_sub ≫ (fil s (k+1-1)).arrow
  71       -- so φ_to_X2 = (kerZ.arrow ≫ eqToHom_sub) ≫ (fil s (k+1-1)).arrow
  72       let e_sub := congr_arg (fun j => Subobject.underlying.obj (FC.fil s j))
  73           (show k = k + 1 - 1 by omega)
  74       have h_transport : (FC.fil s k).arrow ≫ eqToHom e_deg = eqToHom e_sub ≫
  75           (FC.fil s (k + 1 - 1)).arrow := by
  76         -- Use the Mono property: compose both sides with the identity on A(k+1-1)
  77         -- and show they agree. Actually, we can use eqToHom_naturality.
  78         -- eqToHom at object level is functorial, so the naturality square commutes.
  79         -- For `FC.fil s` applied to equal indices, the arrow commutes with eqToHom.
  80         -- We prove this by induction on the proof of equality.
  81         -- Since we can't subst directly, use proof irrelevance + a general lemma.
  82         have : ∀ (a b : ℤ) (h : a = b),
  83             (FC.fil s a).arrow ≫ eqToHom (congr_arg FC.A h) =
  84             eqToHom (congr_arg (fun j => Subobject.underlying.obj (FC.fil s j)) h) ≫
  85             (FC.fil s b).arrow := by
  86           intro a b h; subst h; simp
  87         exact this k (k + 1 - 1) (by omega)
  88       have h_φ_eq : φ_to_X2 = (kerZ.arrow ≫ eqToHom e_sub) ≫
  89           (FC.fil s (k + 1 - 1)).arrow := by
  90         change kerZ.arrow ≫ (FC.fil s k).arrow ≫ eqToHom e_deg =
  91           (kerZ.arrow ≫ eqToHom e_sub) ≫ (FC.fil s (k + 1 - 1)).arrow
  92         rw [Category.assoc, h_transport, ← Category.assoc]
  93       rw [h_φ_eq]
  94       exact Subobject.factors_comp_arrow _
  95     -- Factor: kerZ → I_s
  96     let kerZ_to_Is := Subobject.factorThru I_s φ_to_X2 h_factors_I
  97     -- Key property: kerZ_to_Is ≫ I_s.arrow = φ_to_X2
  98     have h_kerZ_to_Is : kerZ_to_Is ≫ I_s.arrow = φ_to_X2 :=
  99       Subobject.factorThru_arrow I_s φ_to_X2 h_factors_I
 100     -- liftCycles is functorial: kerZ_to_Is ≫ gen_s = φ_pre
 101     have h_func : kerZ_to_Is ≫ gen_s = φ_pre := by
 102       change kerZ_to_Is ≫ (S.liftCycles I_s.arrow h_zero_s ≫ S.homologyπ) =
 103         S.liftCycles φ_to_X2 h_lift ≫ S.homologyπ
 104       rw [← Category.assoc]
 105       congr 1
 106       apply (cancel_mono S.iCycles).mp
 107       rw [Category.assoc, S.liftCycles_i, S.liftCycles_i, h_kerZ_to_Is]
 108     -- imageSubobject(φ_pre) ≤ imageSubobject(gen_s) = F_s
 109     have h_img_le : imageSubobject φ_pre ≤ imageSubobject gen_s := by
 110       rw [show φ_pre = kerZ_to_Is ≫ gen_s from h_func.symm]
 111       exact imageSubobject_comp_le _ _
 112     -- φ_pre factors through F_s = imageSubobject(gen_s)
 113     have h_fac_Fs : (imageSubobject gen_s).Factors φ_pre := by
 114       have : φ_pre = (kerZ_to_Is ≫ factorThruImageSubobject gen_s) ≫
 115           (imageSubobject gen_s).arrow := by
 116         rw [Category.assoc, imageSubobject_arrow_comp]
 117         exact h_func.symm
 118       rw [this]
 119       exact Subobject.factors_comp_arrow _
 120     let φ_to_Fs := Subobject.factorThru (imageSubobject gen_s) φ_pre h_fac_Fs
 121     have h_φ_to_Fs : φ_to_Fs ≫ (imageSubobject gen_s).arrow = φ_pre :=
 122       Subobject.factorThru_arrow _ _ h_fac_Fs
 123     -- Compose with cokernel projection to get kerZ → RHS
 124     -- RHS = cokernel(ofLE F_{s+1} F_s _)
 125     -- We need the cokernel.π for the RHS
 126     -- But first, F_s and F_{s+1} are from homologyFiltration, not our local gen_s/gen_s1.
 127     -- They should be definitionally equal (or propositionally equal)
 128     -- to imageSubobject gen_s / gen_s1.
 129     -- Let's show they match.
 130     -- ======================================
 131     -- CONSTRUCTION OF THE CONVERGENCE ISOMORPHISM
 132     -- E_∞ = Z_⊤/B_⊤ ≅ gr^s H_k = F_s H_k / F_{s+1} H_k
 133     --
 134     -- Forward: E_∞ → gr^s H_k via φ_to_Fs
 135     -- Inverse: gr^s H_k → E_∞ via kerZ.arrow ≫ πV
 136     -- ======================================
 137     -- Step 1: Forward map on Z_⊤
 138     -- We build fwd_Z : Z_⊤.underlying → gr^s H_k (the RHS)
 139     -- by descending φ_to_Fs ≫ cokernel.π through the epi
 140     -- factorThruImageSubobject(kerZ.arrow ≫ πV) : kerZ → Z_⊤
 141     -- The RHS cokernel projection
 142     let πRHS := cokernel.π (Subobject.ofLE
 143       (FC.homologySSFiltration.F (s + 1) k)
 144       (FC.homologySSFiltration.F s k)
 145       (FC.homologySSFiltration.mono s k))
 146     -- The composed map kerZ → gr^s H_k
 147     let fwd_kerZ := φ_to_Fs ≫ πRHS
 148     -- The epi from kerZ to Z_⊤
 149     let e_Z := factorThruImageSubobject (kerZ.arrow ≫ πV)
 150     -- Descent condition: kernel.ι(e_Z) ≫ fwd_kerZ = 0
 151     -- This says: if x ∈ kerZ with kerZ.arrow(x) ∈ F^{s+1}, then φ_to_Fs(x) ∈ F_{s+1} H_k
 152     have h_desc_fwd : kernel.ι e_Z ≫ fwd_kerZ = 0 := by
 153       -- kernel.ι e_Z ≫ (kerZ.arrow ≫ πV) = 0 (kernel condition + image factorization)
 154       have h_ker_πV : kernel.ι e_Z ≫ kerZ.arrow ≫ πV = 0 := by
 155         have h0 : kernel.ι e_Z ≫ e_Z = 0 := kernel.condition e_Z
 156         have h1 : factorThruImageSubobject (kerZ.arrow ≫ πV) ≫
 157             (imageSubobject (kerZ.arrow ≫ πV)).arrow =
 158             kerZ.arrow ≫ πV := by simp [imageSubobject_arrow_comp]
 159         calc kernel.ι e_Z ≫ kerZ.arrow ≫ πV
 160             = kernel.ι e_Z ≫ (e_Z ≫
 161                 (imageSubobject (kerZ.arrow ≫ πV)).arrow) := by rw [h1]
 162           _ = (kernel.ι e_Z ≫ e_Z) ≫ (imageSubobject (kerZ.arrow ≫ πV)).arrow :=
 163                 (Category.assoc _ _ _).symm
 164           _ = 0 ≫ _ := by rw [h0]
 165           _ = 0 := zero_comp
 166       -- Lift through ι_fil
 167       have h_ker_πV' : (kernel.ι e_Z ≫ kerZ.arrow) ≫ cokernel.π ι_fil = 0 := by
 168         rw [Category.assoc]; exact h_ker_πV
 169       set lift_s1 := Abelian.monoLift ι_fil (kernel.ι e_Z ≫ kerZ.arrow) h_ker_πV'
 170       have h_lift_s1 : lift_s1 ≫ ι_fil = kernel.ι e_Z ≫ kerZ.arrow :=
 171         Abelian.monoLift_comp _ _ _
 172       -- kernel.ι e_Z ≫ φ_to_X2 factors through I_{s+1} = ker(S.g) ∩ fil(s+1, k+1-1)
 173       set I_s1 := kernelSubobject S.g ⊓ FC.fil (s + 1) (k + 1 - 1)
 174       have h_zero_s1 : I_s1.arrow ≫ S.g = 0 := by
 175         rw [show I_s1.arrow = Subobject.ofLE I_s1 (kernelSubobject S.g)
 176             inf_le_left ≫ (kernelSubobject S.g).arrow
 177           from (Subobject.ofLE_arrow inf_le_left).symm,
 178           Category.assoc, kernelSubobject_arrow_comp, comp_zero]
 179       set gen_s1 := S.liftCycles I_s1.arrow h_zero_s1 ≫ S.homologyπ
 180       -- Key factorization: kernel.ι e_Z ≫ φ_to_X2 lies in I_{s+1}
 181       -- φ_to_X2 = kerZ.arrow ≫ (FC.fil s k).arrow ≫ eqToHom e_deg
 182       -- kernel.ι e_Z ≫ kerZ.arrow = lift_s1 ≫ ι_fil (h_lift_s1)
 183       -- So kernel.ι e_Z ≫ φ_to_X2 = lift_s1 ≫ ι_fil ≫ (fil s k).arrow ≫ eqToHom
 184       --   = lift_s1 ≫ (fil(s+1,k)).arrow ≫ eqToHom
 185       --   = (lift_s1 ≫ eqToHom_sub) ≫ (fil(s+1,k+1-1)).arrow
 186       -- which factors through fil(s+1,k+1-1).
 187       -- Also kernel.ι e_Z ≫ φ_to_X2 ≫ S.g = 0 (from h_lift).
 188       -- So it's in ker(S.g) ∩ fil(s+1,k+1-1) = I_{s+1}.
 189       -- Then kernel.ι e_Z ≫ φ_pre = α ≫ gen_s1 (liftCycles functoriality)
 190       -- = (α ≫ fTI gen_s1) ≫ (F_{s+1}).arrow
 191       -- And kernel.ι e_Z ≫ φ_to_Fs ≫ (F_s).arrow = kernel.ι e_Z ≫ φ_pre
 192       -- So kernel.ι e_Z ≫ φ_to_Fs = (α ≫ fTI gen_s1) ≫ ofLE(F_{s+1}, F_s) (mono cancel)
 193       -- Step B: factor kernel.ι e_Z ≫ φ_to_X2 through I_{s+1}
 194       have h_ι_arrow : ι_fil ≫ (FC.fil s k).arrow = (FC.fil (s + 1) k).arrow :=
 195         Subobject.ofLE_arrow (FC.fil_anti s k)
 196       have h_transport_s1 : (FC.fil (s + 1) k).arrow ≫ eqToHom e_deg =
 197           eqToHom (congr_arg (fun j => Subobject.underlying.obj (FC.fil (s + 1) j))
 198             (show k = k + 1 - 1 by omega)) ≫
 199           (FC.fil (s + 1) (k + 1 - 1)).arrow := by
 200         have : ∀ (a b : ℤ) (h : a = b),
 201             (FC.fil (s + 1) a).arrow ≫ eqToHom (congr_arg FC.A h) =
 202             eqToHom (congr_arg (fun j => Subobject.underlying.obj (FC.fil (s + 1) j)) h) ≫
 203             (FC.fil (s + 1) b).arrow := by
 204           intro a b h; subst h; simp
 205         exact this k (k + 1 - 1) (by omega)
 206       have h_ker_lift_sg : (kernel.ι e_Z ≫ φ_to_X2) ≫ S.g = 0 := by
 207         rw [Category.assoc]; exact (h_lift ▸ comp_zero)
 208       have h_ker_in_fil_s1 :
 209           (FC.fil (s + 1) (k + 1 - 1)).Factors (kernel.ι e_Z ≫ φ_to_X2) := by
 210         have h_eq : kernel.ι e_Z ≫ φ_to_X2 =
 211             (lift_s1 ≫ eqToHom (congr_arg (fun j => Subobject.underlying.obj (FC.fil (s + 1) j))
 212               (show k = k + 1 - 1 by omega))) ≫
 213             (FC.fil (s + 1) (k + 1 - 1)).arrow := by
 214           change kernel.ι e_Z ≫ (kerZ.arrow ≫ (FC.fil s k).arrow ≫ eqToHom e_deg) = _
 215           rw [show kernel.ι e_Z ≫ (kerZ.arrow ≫ (FC.fil s k).arrow ≫ eqToHom e_deg) =
 216               (kernel.ι e_Z ≫ kerZ.arrow) ≫ ((FC.fil s k).arrow ≫ eqToHom e_deg) by
 217             simp only [Category.assoc]]
 218           rw [h_lift_s1.symm]
 219           rw [show (lift_s1 ≫ ι_fil) ≫ ((FC.fil s k).arrow ≫ eqToHom e_deg) =
 220               lift_s1 ≫ (ι_fil ≫ (FC.fil s k).arrow) ≫ eqToHom e_deg by
 221             simp only [Category.assoc]]
 222           rw [h_ι_arrow, h_transport_s1]
 223           simp only [Category.assoc]
 224         rw [h_eq]; exact Subobject.factors_comp_arrow _
 225       have h_fac_I_s1 : I_s1.Factors (kernel.ι e_Z ≫ φ_to_X2) := by
 226         rw [show I_s1 = kernelSubobject S.g ⊓ FC.fil (s + 1) (k + 1 - 1) from rfl,
 227             Subobject.inf_factors]
 228         exact ⟨kernelSubobject_factors S.g _ h_ker_lift_sg, h_ker_in_fil_s1⟩
 229       let ker_to_I_s1 := Subobject.factorThru I_s1 (kernel.ι e_Z ≫ φ_to_X2) h_fac_I_s1
 230       have h_ker_to_I_s1 : ker_to_I_s1 ≫ I_s1.arrow = kernel.ι e_Z ≫ φ_to_X2 :=
 231         Subobject.factorThru_arrow _ _ _
 232       -- Step C: liftCycles functoriality
 233       have h_func_s1 : ker_to_I_s1 ≫ gen_s1 = kernel.ι e_Z ≫ φ_pre := by
 234         change ker_to_I_s1 ≫ (S.liftCycles I_s1.arrow h_zero_s1 ≫ S.homologyπ) =
 235           kernel.ι e_Z ≫ (S.liftCycles φ_to_X2 h_lift ≫ S.homologyπ)
 236         rw [← Category.assoc, ← Category.assoc]; congr 1
 237         apply (cancel_mono S.iCycles).mp
 238         rw [Category.assoc, S.liftCycles_i, Category.assoc, S.liftCycles_i, h_ker_to_I_s1]
 239       -- Step D: factor through ofLE(F_{s+1}, F_s)
 240       have h_Fs1_le_Fs : FC.homologySSFiltration.F (s + 1) k ≤
 241           FC.homologySSFiltration.F s k :=
 242         FC.homologySSFiltration.mono s k
 243       have h_ofLE_arrow : Subobject.ofLE (FC.homologySSFiltration.F (s + 1) k)
 244           (FC.homologySSFiltration.F s k) h_Fs1_le_Fs ≫
 245           (FC.homologySSFiltration.F s k).arrow =
 246           (FC.homologySSFiltration.F (s + 1) k).arrow :=
 247         Subobject.ofLE_arrow h_Fs1_le_Fs
 248       have h_factor_ofLE : kernel.ι e_Z ≫ φ_to_Fs =
 249           (ker_to_I_s1 ≫ factorThruImageSubobject gen_s1) ≫
 250           Subobject.ofLE (FC.homologySSFiltration.F (s + 1) k)
 251             (FC.homologySSFiltration.F s k) h_Fs1_le_Fs := by
 252         apply (cancel_mono (imageSubobject gen_s).arrow).mp
 253         rw [Category.assoc, h_φ_to_Fs]
 254         -- LHS: kernel.ι e_Z ≫ φ_pre
 255         -- RHS: ((ker_to_I_s1 ≫ fTI gen_s1) ≫ ofLE) ≫ (imageSubobject gen_s).arrow
 256         -- Rewrite RHS to ker_to_I_s1 ≫ gen_s1 = kernel.ι e_Z ≫ φ_pre
 257         have h_rhs : ((ker_to_I_s1 ≫ factorThruImageSubobject gen_s1) ≫
 258             Subobject.ofLE (FC.homologySSFiltration.F (s + 1) k)
 259               (FC.homologySSFiltration.F s k) h_Fs1_le_Fs) ≫
 260             (imageSubobject gen_s).arrow =
 261             ker_to_I_s1 ≫ gen_s1 := by
 262           -- (imageSubobject gen_s).arrow = (FC.homologySSFiltration.F s k).arrow definitionally
 263           change ((ker_to_I_s1 ≫ factorThruImageSubobject gen_s1) ≫
 264             Subobject.ofLE (FC.homologySSFiltration.F (s + 1) k)
 265               (FC.homologySSFiltration.F s k) h_Fs1_le_Fs) ≫
 266             (FC.homologySSFiltration.F s k).arrow = ker_to_I_s1 ≫ gen_s1
 267           simp only [Category.assoc]
 268           dsimp only [FilteredComplex.homologySSFiltration] at h_ofLE_arrow ⊢
 269           rw [h_ofLE_arrow]
 270           -- Now: ker_to_I_s1 ≫ fTI gen_s1 ≫ (imageSubobject gen_s1).arrow
 271           -- = ker_to_I_s1 ≫ gen_s1
 272           -- (FC.homologySSFiltration.F (s+1) k).arrow = (imageSubobject gen_s1).arrow definitionally
 273           change ker_to_I_s1 ≫ factorThruImageSubobject gen_s1 ≫
 274             (imageSubobject gen_s1).arrow = ker_to_I_s1 ≫ gen_s1
 275           rw [imageSubobject_arrow_comp]
 276         dsimp only [FilteredComplex.homologySSFiltration] at h_rhs ⊢
 277         rw [h_rhs]
 278         exact h_func_s1.symm
 279       -- Step E: compose with πRHS = cokernel.π → 0
 280       change kernel.ι e_Z ≫ (φ_to_Fs ≫ πRHS) = 0
 281       dsimp only [FilteredComplex.homologySSFiltration] at h_factor_ofLE ⊢
 282       rw [← Category.assoc, h_factor_ofLE, Category.assoc, Category.assoc]
 283       erw [show Subobject.ofLE (FC.homologySSFiltration.F (s + 1) k)
 284             (FC.homologySSFiltration.F s k) h_Fs1_le_Fs ≫ πRHS = 0
 285         from cokernel.condition _]
 286       erw [comp_zero, comp_zero]
 287       rfl
 288     -- Descended map on Z_⊤
 289     let fwd_Z := Abelian.epiDesc e_Z fwd_kerZ h_desc_fwd
 290     -- Property: e_Z ≫ fwd_Z = fwd_kerZ
 291     have h_fwd_Z : e_Z ≫ fwd_Z = fwd_kerZ := Abelian.comp_epiDesc _ _ _
 292     -- Step 2: Show fwd_Z kills B_⊤
 293     -- Need: ofLE(B_⊤, Z_⊤, _) ≫ fwd_Z = 0
 294     have h_fwd_kills_B : Subobject.ofLE
 295         ((FC.toSSData bnd s k).B ⊤) ((FC.toSSData bnd s k).Z ⊤)
 296         ((FC.toSSData bnd s k).B_le_Z ⊤) ≫ fwd_Z = 0 := by
 297       -- === Proof that fwd_Z kills B_⊤ ===
 298       -- Local boundary notation
 299       let imgD := imageSubobject (FC.dToK k)
 300       let I_B := imgD ⊓ FC.fil s k
 301       let oI := Subobject.ofLE I_B (FC.fil s k) inf_le_right
 302       -- oI ≫ f_Z = 0 (boundary elements are cycles)
 303       have h_oI_zero : oI ≫ f_Z = 0 := by
 304         change Subobject.ofLE I_B (FC.fil s k) inf_le_right ≫
 305           (FC.fil s k).arrow ≫ FC.d k = 0
 306         rw [← Category.assoc, Subobject.ofLE_arrow]
 307         rw [show I_B.arrow =
 308             Subobject.ofLE I_B imgD inf_le_left ≫ imgD.arrow
 309           from (Subobject.ofLE_arrow inf_le_left).symm,
 310           Category.assoc]
 311         have : imgD.arrow ≫ FC.d k = 0 := by
 312           rw [← cancel_epi (factorThruImageSubobject (FC.dToK k))]
 313           rw [comp_zero, ← Category.assoc, imageSubobject_arrow_comp]
 314           exact FC.dToK_comp_d k
 315         rw [this, comp_zero]
 316       -- Factor oI through kerZ
 317       let oI_lift := factorThruKernelSubobject f_Z oI h_oI_zero
 318       have h_oI_lift : oI_lift ≫ kerZ.arrow = oI :=
 319         factorThruKernelSubobject_comp_arrow f_Z oI h_oI_zero
 320       -- Step A: factorThruImageSubobject(oI ≫ πV) ≫ ofLE(B,Z) = oI_lift ≫ e_Z
 321       have h_fti_ofLE : factorThruImageSubobject (oI ≫ πV) ≫
 322           Subobject.ofLE ((FC.toSSData bnd s k).B ⊤) ((FC.toSSData bnd s k).Z ⊤)
 323           ((FC.toSSData bnd s k).B_le_Z ⊤) = oI_lift ≫ e_Z := by
 324         apply (cancel_mono ((FC.toSSData bnd s k).Z ⊤).arrow).mp
 325         dsimp only [FilteredComplex.toSSData, FilteredComplex.cycleSubobject,
 326           FilteredComplex.boundarySubobject] at *
 327         erw [Category.assoc, Subobject.ofLE_arrow]
 328         change factorThruImageSubobject (oI ≫ πV) ≫
 329           ((FC.toSSData bnd s k).B ⊤).arrow = (oI_lift ≫ e_Z) ≫ _
 330         rw [show ((FC.toSSData bnd s k).B ⊤).arrow =
 331             (imageSubobject (oI ≫ πV)).arrow from rfl]
 332         erw [imageSubobject_arrow_comp]
 333         rw [Category.assoc]
 334         erw [show e_Z ≫ ((FC.toSSData bnd s k).Z ⊤).arrow =
