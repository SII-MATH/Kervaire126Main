   8 /-!
   9 # Nested-subobject spectral sequence of a filtered complex
  10 
  11 This is the construction layer connecting KIP126's canonical filtered complex
  12 to the migrated `SSData`/`PreSS`/`SpectralSequence` API.  Its proof fields reuse
  13 the already separated finite-page results; no property is postulated.
  14 -/

  16 namespace KIP126.Core.SpectralSequence

  18 open CategoryTheory CategoryTheory.Limits

  20 universe u v

  22 variable {C : Type u} [Category.{v} C] [Abelian C]
  23 
  24 namespace FilteredComplex

  26 /-- The nested cycle/boundary data of a bounded filtered complex at `(s,k)`. -/
  27 @[reducible] noncomputable def toSSData (FC : FilteredComplex C)
  28     (bnd : FC.IsBounded) (s k : ℤ) : SSData C where
  29   V := FC.assocGraded s k
  30   Z := FC.cycleSubobject s k
  31   B := FC.boundarySubobject s k
  32   Z_anti := FC.cycleSubobject_antitone s k
  33   B_mono := FC.boundarySubobject_monotone s k
  34   Z_zero := FC.cycleSubobject_zero s k
  35   B_le_Z := FC.B_le_Z_aux s k
  36   Z_top_greatest := FC.cycleSubobject_top_greatest
  37     bnd.toFiltration.toIsBoundedAbove s k
  38   B_top_least := FC.boundarySubobject_top_least
  39     bnd.toFiltration.toIsBoundedBelow s k
  40 
  41 set_option maxHeartbeats 6400000 in
  42 /-- The data-only pre-spectral sequence of a bounded filtered complex. -/
  43 noncomputable def toPreSS (FC : FilteredComplex C)
  44     (bnd : FC.IsBounded) : PreSS C (ℤ × ℤ) where
  45   r₀ := 0
  46   ssData := fun ⟨s, k⟩ => FC.toSSData bnd s k
  47   diffDeg := fun r => (r, -1)
  48   d := fun r ⟨s, k⟩ =>
  49     if hr : 0 ≤ r then
  50       eqToHom (by
  51         have hnat : (r - 0).toNat = r.toNat := by omega
  52         rw [hnat]
  53         simp [SSData.page, FilteredComplex.toSSData,
  54           FilteredComplex.pageObj]) ≫
  55         FC.pageDifferential s k r.toNat ≫
  56         eqToHom (by
  57           have hnat : (r - 0).toNat = r.toNat := by omega
  58           have hrnat : (r.toNat : ℤ) = r := Int.toNat_of_nonneg hr
  59           rw [hnat, hrnat]
  60           simp [SSData.page, FilteredComplex.toSSData,
  61             FilteredComplex.pageObj]
  62           rw [show k - 1 = k + -1 by omega])
  63     else 0
  64 
  65 end FilteredComplex
  66 
  67 end KIP126.Core.SpectralSequence
