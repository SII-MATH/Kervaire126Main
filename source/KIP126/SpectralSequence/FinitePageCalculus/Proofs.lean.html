  27 public theorem represents_unique
  28     {E : KIP126.Core.SpectralSequence (ModuleCat.{v} ℤ) (ℤ × ℤ)} {r : ℤ} {p : ℤ × ℤ} {x : E.Page 2 p} {a b : E.Page r p}
  29     (ha : RepresentsOnPage E r p x a) (hb : RepresentsOnPage E r p x b) : a = b := by
  30   obtain ⟨hr, z, hz, hza⟩ := ha
  31   obtain ⟨_, z', hz', hzb⟩ := hb
  32   let D := E.ssData p
  33   let n : WithTop ℕ := ↑(2 - E.r₀).toNat
  34   let m : WithTop ℕ := ↑(r - E.r₀).toNat
  35   have hnm : n ≤ m := by
  36     dsimp only [n, m]
  37     exact_mod_cast (show (2-E.r₀).toNat ≤ (r-E.r₀).toNat by omega)
  38   apply sub_eq_zero.mp
  39   rw [← hza, ← hzb, ← map_sub]
  40   apply (subobject_cokernel_π_eq_zero_iff (D.B m) (D.Z m) (D.B_le_Z m) _).mpr
  41   have heq : D.pageπ n ((Subobject.ofLE _ _ (D.Z_anti hnm)) (z-z')) = 0 := by
  42     change (Subobject.ofLE _ _ (D.Z_anti hnm) ≫ D.pageπ n) (z-z') = 0
  43     rw [map_sub, hz, hz', sub_self]
  44   have hh := (subobject_cokernel_π_eq_zero_iff (D.B n) (D.Z n) (D.B_le_Z n) _).mp heq
  45   have he : (D.Z n).arrow ((Subobject.ofLE _ _ (D.Z_anti hnm)) (z-z')) =
  46       (D.Z m).arrow (z-z') := ConcreteCategory.congr_hom (Subobject.ofLE_arrow (D.Z_anti hnm)) _
  47   rw [he] at hh
  48   exact (ModuleCat.subobjectModule D.V).monotone (D.B_mono hnm) hh

  53     {a : E.Page r p} (ha : RepresentsOnPage E r p x a)
  54     (ht : ReachesPage E t p x) : E.d r p a = 0 := by

  56   let D := E.ssData p

  59   let ns : WithTop ℕ := ↑((r-E.r₀).toNat+1)
  60   have hrs : nr ≤ ns := by
  61     dsimp only [nr, ns]
  62     exact_mod_cast (show (r-E.r₀).toNat ≤ (r-E.r₀).toNat+1 by omega)

  64     dsimp only [ns, nt]
  65     exact_mod_cast (show (r-E.r₀).toNat+1 ≤ (t-E.r₀).toNat by omega)
  66   let zr := (Subobject.ofLE (D.Z nt) (D.Z nr) (D.Z_anti (hrs.trans hst))) z

  68     refine ⟨ha.1, zr, ?_, rfl⟩
  69     dsimp only [zr]
  70     rw [← CategoryTheory.comp_apply, ← Category.assoc, Subobject.ofLE_comp_ofLE]

  74   rw [← subobjectModule_kernel, E.Z_succ r p hr0, subobjectModule_image]
  75   refine ⟨(Subobject.ofLE (D.Z nt) (D.Z ns) (D.Z_anti hst)) z, ?_⟩

  87     (E.ssData (p+E.diffDeg r)).pageπ ↑((r-E.r₀).toNat+1) z = 0 ↔

  96   let j := Subobject.ofLE (D.B ↑(n+1)) (D.Z ↑n)
  97     ((D.B_le_Z _).trans (D.Z_anti hnm))
  98   rw [← subobjectModule_image, E.B_succ r p hr, subobjectModule_image]
  99   change D.pageπ ↑(n+1) z = 0 ↔ (i ≫ D.pageπ ↑n) z ∈ LinearMap.range (j ≫ D.pageπ ↑n).hom

 108     congr 2
 109     dsimp only [i,j]
 110     rw [← Category.assoc, Subobject.ofLE_comp_ofLE]

 119       change (j ≫ (D.Z ↑n).arrow) v ∈ _

 121       rw [Subobject.ofLE_arrow]
 122       exact ⟨v, rfl⟩
 123     have hh := ((ModuleCat.subobjectModule D.V) (D.B ↑(n+1))).add_mem hb' hv'
 124     rw [map_sub, sub_add_cancel] at hh
 125     change (i ≫ (D.Z ↑n).arrow) z ∈ _ at hh
 126     dsimp only [i] at hh
 127     rwa [Subobject.ofLE_arrow] at hh

 130 public theorem page_generated_from_representatives
 131     {E : KIP126.Core.SpectralSequence (ModuleCat.{v} ℤ) (ℤ × ℤ)} {r : ℤ} {p : ℤ × ℤ} (hr : 2 ≤ r)
 132     {N : ℕ} (e : E.Page 2 p ≃ₗ[ℤ] (Fin N →₀ KIP126.Core.Algebra.F2))
 133     (v : Fin N → E.Page r p)
 134     (hv : ∀ i, RepresentsOnPage E r p (e.symm (Finsupp.single i 1)) (v i))
 135     (a : E.Page r p) :
 136     ∃ c : Fin N → KIP126.Core.Algebra.F2, a = ∑ i, if c i = 0 then 0 else v i := by
 137   classical
 138   have hadd {x y : E.Page 2 p} {xr yr : E.Page r p}
 139       (hx : RepresentsOnPage E r p x xr) (hy : RepresentsOnPage E r p y yr) :
 140       RepresentsOnPage E r p (x+y) (xr+yr) := by
 141     obtain ⟨hr, zx, hx, hxr⟩ := hx
 142     obtain ⟨_, zy, hy, hyr⟩ := hy
 143     exact ⟨hr, zx+zy, by rw [map_add,hx,hy], by rw [map_add,hxr,hyr]⟩
 144   let D := E.ssData p
 145   let n : WithTop ℕ := ↑(r-E.r₀).toNat
 146   haveI : Epi (D.pageπ n) := inferInstanceAs (Epi (cokernel.π _))
 147   obtain ⟨z, hz⟩ := (ModuleCat.epi_iff_surjective (D.pageπ n)).mp inferInstance a
 148   let x : E.Page 2 p := (Subobject.ofLE _ _ (D.Z_anti (by
 149     dsimp only [n]
 150     exact_mod_cast (show (2-E.r₀).toNat ≤ (r-E.r₀).toNat by omega))) ≫ D.pageπ _) z
 151   have hxa : RepresentsOnPage E r p x a := ⟨hr,z,rfl,hz⟩
 152   let c : Fin N → KIP126.Core.Algebra.F2 := e x
 153   have hsum (s : Finset (Fin N)) :
 154       RepresentsOnPage E r p
 155         (∑ i ∈ s, if c i = 0 then 0 else e.symm (Finsupp.single i 1))
 156         (∑ i ∈ s, if c i = 0 then 0 else v i) := by
 157     induction s using Finset.induction_on with
 158     | empty => simpa using (RepresentsOnPage.zero (E := E) (p := p) hr)
 159     | @insert i s hi ih =>
 160       simp only [Finset.sum_insert hi]
 161       apply hadd _ ih
 162       by_cases hc : c i = 0
 163       · simpa only [hc,ite_true] using (RepresentsOnPage.zero (E := E) (p := p) hr)
 164       · simpa only [hc,ite_false] using hv i
 165   have hx : (∑ i : Fin N, if c i = 0 then 0 else e.symm (Finsupp.single i 1)) = x := by
 166     apply e.injective
 167     rw [map_sum]
 168     ext i
 169     simp only [Finsupp.finsetSum_apply]
 170     rw [Finset.sum_eq_single i]
 171     · change (e (if c i = 0 then 0 else e.symm (Finsupp.single i 1))) i = c i
 172       generalize c i = b
 173       fin_cases b
 174       · change (e 0) i = 0
 175         simp
 176       · change (e (e.symm (Finsupp.single i 1))) i = 1
 177         simp
 178     · intro j hj hji
 179       by_cases hc : c j = 0
 180       · simp [hc]
 181       · simp [hc,hji]
 182     · simp
 183   refine ⟨c, represents_unique (E:=E) (r:=r) (p:=p) (x:=x) hxa ?_⟩
 184   have hh := hsum Finset.univ
 185   simpa only [hx] using hh
