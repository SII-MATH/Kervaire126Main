 140 /-! Internal trust audit. This module exports nothing. It inspects private
 141 implementation declarations as well as the public interface. -/
 142 open Lean Elab Command
 143 set_option maxHeartbeats 0
 144 set_option maxRecDepth 8192
 145 run_cmd do
 146   let env ← getEnv
 147   let mut checked : Nat := 0
 148   for (name, info) in env.constants.toList do
 149     let some idx := env.getModuleIdxFor? name | continue
 150     let mod := env.header.moduleNames[idx.toNat]!
 151     unless (`KIP126).isPrefixOf mod do continue
 152     if info.isAxiom then
 153       throwError "Project axiom declaration: {name}"
 154     let axioms ← collectAxioms name
 155     let unexpected := axioms.filter fun n =>
 156       n != `propext && n != `Classical.choice && n != `Quot.sound
 157     unless unexpected.isEmpty do
 158       throwError "Unexpected axiom dependency in {name}: {unexpected}"
 159     checked := checked + 1
 160   logInfo m!"SpectralSequence internal trust audit: {checked} declarations; only propext, Classical.choice, Quot.sound permitted"
