  21 /-- Every actual E₃ element has an E₂ label with the same common cycle
  22 representative. No nonvanishing or basis assumption is required. -/
  23 public theorem exists_representative (p : ℤ × ℤ) (y : E.Page 3 p) :
  24     ∃ x : E.Page 2 p, RepresentsOnPage E 3 p x y := by
  25   let D := E.ssData p
  26   let n : WithTop ℕ := ↑(2 - E.r₀).toNat
  27   let m : WithTop ℕ := ↑(3 - E.r₀).toNat
  28   let a := Subobject.ofLE (D.Z m) (D.Z n) (D.Z_anti (by
  29     dsimp only [n, m]
  30     exact_mod_cast (show (2 - E.r₀).toNat ≤ (3 - E.r₀).toNat by omega))) ≫ D.pageπ n
  31   letI : Epi (D.pageπ m) := coequalizer.π_epi
  32   obtain ⟨z, hz⟩ := (ModuleCat.epi_iff_surjective (D.pageπ m)).mp inferInstance y
  33   exact ⟨a z, by decide, z, rfl, hz⟩

  35 /-- A full E₂ label admits an E₃ representative exactly when its actual d₂
  36 is zero. This ranges over every degree and every label. -/
  37 public theorem reachesPage_iff_d2_eq_zero (hstart : E.r₀ ≤ 2) (p : ℤ × ℤ)
  38     (x : E.Page 2 p) : ReachesPage E 3 p x ↔ E.d 2 p x = 0 := by
  39   have hm : (3 - E.r₀).toNat = (2 - E.r₀).toNat + 1 := by omega
  40   have hk := congrArg (ModuleCat.subobjectModule (E.Page 2 p)) (E.Z_succ 2 p hstart)
  41   rw [subobjectModule_kernel, subobjectModule_image] at hk
  42   have hindexed (m : ℕ) (hnm : (2 - E.r₀).toNat ≤ m)
  43       (heq : m = (2 - E.r₀).toNat + 1) :
  44       LinearMap.ker (E.d 2 p).hom = LinearMap.range
  45         (Subobject.ofLE ((E.ssData p).Z ↑m) ((E.ssData p).Z ↑(2 - E.r₀).toNat)
  46           ((E.ssData p).Z_anti (by exact_mod_cast hnm)) ≫
  47           (E.ssData p).pageπ ↑(2 - E.r₀).toNat).hom := by
  48     subst m
  49     exact hk
  50   have hk := hindexed (3 - E.r₀).toNat (by omega) hm
  51   constructor
  52   · rintro ⟨y, _, z, hx, _⟩
  53     have hmem : x ∈ LinearMap.ker (E.d 2 p).hom := by
  54       rw [hk]
  55       exact ⟨z, hx⟩
  56     exact hmem
  57   · intro hx
  58     have hmem : x ∈ LinearMap.ker (E.d 2 p).hom := hx
  59     rw [hk] at hmem
  60     obtain ⟨z, hz⟩ := hmem
  61     exact ⟨(E.ssData p).pageπ ↑(3 - E.r₀).toNat z, by decide, z, hz, rfl⟩

  63 theorem represents_sub {r : ℤ} {p : ℤ × ℤ}
  64     {a b : E.Page 2 p} {x y : E.Page r p}
  65     (ha : RepresentsOnPage E r p a x) (hb : RepresentsOnPage E r p b y) :
  66     RepresentsOnPage E r p (a - b) (x - y) := by
  67   obtain ⟨hr, z, hza, hzx⟩ := ha
  68   obtain ⟨_, w, hwb, hwy⟩ := hb
  69   exact ⟨hr, z - w, by simp only [map_sub, hza, hwb],
  70     by simp only [map_sub, hzx, hwy]⟩

  72 /-- Cumulative second-page boundaries have the zero third-page representative. -/
  73 public theorem boundary_represents_zero {p : ℤ × ℤ} {x : E.Page 2 p}
  74     (h : IsBoundaryBy E 2 p x) : RepresentsOnPage E 3 p x 0 := by
  75   let D := E.ssData p
  76   let n : WithTop ℕ := ↑(2 - E.r₀).toNat
  77   let m : WithTop ℕ := ↑(2 + 1 - E.r₀).toNat
  78   obtain ⟨_, b, hb⟩ := h
  79   refine ⟨by decide, Subobject.ofLE (D.B m) (D.Z m) (D.B_le_Z m) b, ?_, ?_⟩
  80   · change (Subobject.ofLE (D.B m) (D.Z m) (D.B_le_Z m) ≫
  81         Subobject.ofLE (D.Z m) (D.Z n) _ ≫ D.pageπ n) b = x
  82     simpa only [D, m, n, ← Category.assoc, Subobject.ofLE_comp_ofLE] using hb
  83   · exact cokernel.condition_apply _ b

  85 /-- The cumulative boundary condition after d₂ is precisely the image of
  86 that actual d₂. Indexing the target by `p + diffDeg 2` avoids choosing any
  87 unrelated grading transport; as p varies this covers every target degree. -/
  88 public theorem isBoundaryBy_iff_mem_d2_range (hstart : E.r₀ ≤ 2) (p : ℤ × ℤ)
  89     (x : E.Page 2 (p + E.diffDeg 2)) :
  90     IsBoundaryBy E 2 (p + E.diffDeg 2) x ↔ x ∈ LinearMap.range (E.d 2 p).hom := by
  91   have hm : (2 + 1 - E.r₀).toNat = (2 - E.r₀).toNat + 1 := by omega
  92   have hb := congrArg (ModuleCat.subobjectModule (E.Page 2 (p + E.diffDeg 2)))
  93     (E.B_succ 2 p hstart)
  94   rw [subobjectModule_image, subobjectModule_image] at hb
  95   have hindexed (m : ℕ) (hnm : (2 - E.r₀).toNat ≤ m)
  96       (heq : m = (2 - E.r₀).toNat + 1) :
  97       LinearMap.range (E.d 2 p).hom = LinearMap.range
  98         (Subobject.ofLE ((E.ssData (p + E.diffDeg 2)).B ↑m)
  99           ((E.ssData (p + E.diffDeg 2)).Z ↑(2 - E.r₀).toNat)
 100           (((E.ssData (p + E.diffDeg 2)).B_le_Z ↑m).trans
 101             ((E.ssData (p + E.diffDeg 2)).Z_anti (by exact_mod_cast hnm))) ≫
 102           (E.ssData (p + E.diffDeg 2)).pageπ ↑(2 - E.r₀).toNat).hom := by
 103     subst m
 104     exact hb
 105   have hb := hindexed (2 + 1 - E.r₀).toNat (by omega) hm
 106   constructor
 107   · rintro ⟨_, b, hx⟩
 108     rw [hb]
 109     refine ⟨b, ?_⟩
 110     simpa only [← Category.assoc, Subobject.ofLE_comp_ofLE] using hx
 111   · intro hx
 112     rw [hb] at hx
 113     obtain ⟨b, hx⟩ := hx
 114     exact ⟨by decide, b, by
 115       simpa only [← Category.assoc, Subobject.ofLE_comp_ofLE] using hx⟩

 117 theorem represents_add {r : ℤ} {p : ℤ × ℤ}
 118     {a b : E.Page 2 p} {x y : E.Page r p}
 119     (ha : RepresentsOnPage E r p a x) (hb : RepresentsOnPage E r p b y) :
 120     RepresentsOnPage E r p (a + b) (x + y) := by
 121   obtain ⟨hr, z, hza, hzx⟩ := ha
 122   obtain ⟨_, w, hwb, hwy⟩ := hb
 123   exact ⟨hr, z + w, by simp only [map_add, hza, hwb],
 124     by simp only [map_add, hzx, hwy]⟩

 126 /-- Two E₂ labels represent the same specified E₃ element exactly when
 127 their difference is an incoming actual d₂ boundary. Only one label is
 128 assumed to represent that element; the converse constructs the other. -/
 129 public theorem represents_iff_sub_mem_d2_range (hstart : E.r₀ ≤ 2) (p : ℤ × ℤ)
 130     {a b : E.Page 2 (p + E.diffDeg 2)} {y : E.Page 3 (p + E.diffDeg 2)}
 131     (ha : RepresentsOnPage E 3 (p + E.diffDeg 2) a y) :
 132     RepresentsOnPage E 3 (p + E.diffDeg 2) b y ↔
 133       b - a ∈ LinearMap.range (E.d 2 p).hom := by
 134   constructor
 135   · intro hb
 136     apply (isBoundaryBy_iff_mem_d2_range hstart p (b - a)).mp
 137     exact (RepresentsOnPage.eq_zero_iff_isBoundaryBy (r := 2) (by decide)
 138       (represents_sub hb ha)).mp (sub_self y)
 139   · intro hba
 140     have hzero := boundary_represents_zero
 141       ((isBoundaryBy_iff_mem_d2_range hstart p (b - a)).mpr hba)
 142     simpa only [sub_add_cancel, zero_add] using represents_add hzero ha

 144 /-- Equality of the actual third-page classes is detected by the complete
 145 incoming d₂ image, without a finite-coordinate restriction. -/
 146 public theorem equal_iff_sub_mem_d2_range (hstart : E.r₀ ≤ 2) (p : ℤ × ℤ)
 147     {a b : E.Page 2 (p + E.diffDeg 2)} {x y : E.Page 3 (p + E.diffDeg 2)}
 148     (ha : RepresentsOnPage E 3 (p + E.diffDeg 2) a x)
 149     (hb : RepresentsOnPage E 3 (p + E.diffDeg 2) b y) :
 150     x = y ↔ a - b ∈ LinearMap.range (E.d 2 p).hom := by
 151   calc
 152     x = y ↔ IsBoundaryBy E 2 (p + E.diffDeg 2) (a - b) := by
 153       have h : x - y = 0 ↔ IsBoundaryBy E 2 (p + E.diffDeg 2) (a - b) :=
 154         RepresentsOnPage.eq_zero_iff_isBoundaryBy (r := 2) (by decide)
 155           (represents_sub ha hb)
 156       exact (sub_eq_zero (a := x) (b := y)).symm.trans h
 157     _ ↔ a - b ∈ LinearMap.range (E.d 2 p).hom :=
 158       isBoundaryBy_iff_mem_d2_range hstart p (a - b)
