  13 # Actual page representatives and cumulative boundaries

  25 /-- An actual quotient-page element is zero exactly when its cycle representative
  26 comes from the actual boundary subobject. -/
  27 public theorem SSData.pageπ_eq_zero_iff (D : SSData (ModuleCat.{v} R)) (n : WithTop ℕ)
  28     (z : (Subobject.underlying.obj (D.Z n) : ModuleCat R)) :
  29     D.pageπ n z = 0 ↔ ∃ b : (Subobject.underlying.obj (D.B n) : ModuleCat R),
  30       Subobject.ofLE (D.B n) (D.Z n) (D.B_le_Z n) b = z := by
  31   let i := Subobject.ofLE (D.B n) (D.Z n) (D.B_le_Z n)
  32   have h := ShortComplex.cokernelSequence_exact i
  33   constructor
  34   · exact (ShortComplex.moduleCat_exact_iff _).mp h z
  35   · rintro ⟨b, rfl⟩
  36     exact cokernel.condition_apply i b

  38 /-- Existence of a continuation and vanishing on every continuation give a
  39 differential equation with the zero label. -/
  40 public theorem ReachesPage.hasDifferential_zero
  41     {E : SpectralSequence (ModuleCat.{v} R) (ℤ × ℤ)} {r : ℤ} {p q : ℤ × ℤ}
  42     {x : E.Page 2 p} (hr : 2 ≤ r) (hdeg : p + E.diffDeg r = q)
  43     (hx : ReachesPage E r p x) (hz : DifferentialVanishesOn E r p x) :
  44     HasDifferential E r p q x 0 := by
  45   obtain ⟨xr, hxr⟩ := hx
  46   refine ⟨hdeg, xr, 0, hxr, RepresentsOnPage.zero hr, ?_⟩
  47   simp only [ModuleCat.comp_apply, hz xr hxr, map_zero]

  49 /-- Two later-cycle representatives equal on an earlier page remain equal on
  50 the later page: the earlier boundaries are included in the later boundaries. -/
  51 public theorem SSData.page_eq_of_early_page_eq (D : SSData (ModuleCat.{v} R))
  52     (n m : WithTop ℕ) (hnm : n ≤ m)
  53     (z w : (Subobject.underlying.obj (D.Z m) : ModuleCat R))
  54     (h : (Subobject.ofLE (D.Z m) (D.Z n) (D.Z_anti hnm) ≫ D.pageπ n) z =
  55       (Subobject.ofLE (D.Z m) (D.Z n) (D.Z_anti hnm) ≫ D.pageπ n) w) :
  56     D.pageπ m z = D.pageπ m w := by
  57   let a := Subobject.ofLE (D.Z m) (D.Z n) (D.Z_anti hnm)
  58   have hz : D.pageπ n (a (z - w)) = 0 := by
  59     change (a ≫ D.pageπ n) (z - w) = 0
  60     rw [map_sub, h, sub_self]
  61   obtain ⟨b, hb⟩ := (D.pageπ_eq_zero_iff n (a (z - w))).mp hz
  62   let bm := Subobject.ofLE (D.B n) (D.B m) (D.B_mono hnm) b
  63   have hm : Subobject.ofLE (D.B m) (D.Z m) (D.B_le_Z m) bm = z - w := by
  64     apply (ModuleCat.mono_iff_injective a).mp inferInstance
  65     change (Subobject.ofLE (D.B n) (D.B m) (D.B_mono hnm) ≫
  66       Subobject.ofLE (D.B m) (D.Z m) (D.B_le_Z m) ≫ a) b = a (z - w)
  67     simpa only [a, Subobject.ofLE_comp_ofLE] using hb
  68   have hm0 := (D.pageπ_eq_zero_iff m (z - w)).mpr ⟨bm, hm⟩
  69   simpa only [map_sub, sub_eq_zero] using hm0

  71 /-- A fixed E₂ label has at most one actual continuation on any later page.
  72 This does not assert that a continuation exists. -/
  73 public theorem RepresentsOnPage.unique
  74     {E : SpectralSequence (ModuleCat.{v} R) (ℤ × ℤ)} {r : ℤ} {p : ℤ × ℤ}
  75     {x : E.Page 2 p} {yr zr : E.Page r p}
  76     (hy : RepresentsOnPage E r p x yr) (hz : RepresentsOnPage E r p x zr) :
  77     yr = zr := by
  78   obtain ⟨hr, y, hxy, hy⟩ := hy
  79   obtain ⟨_, z, hxz, hz⟩ := hz
  80   have hnm : (↑(2 - E.r₀).toNat : WithTop ℕ) ≤ ↑(r - E.r₀).toNat := by
  81     exact_mod_cast (show (2 - E.r₀).toNat ≤ (r - E.r₀).toNat by omega)
  82   exact hy.symm.trans (((E.ssData p).page_eq_of_early_page_eq _ _ hnm y z
  83     (hxy.trans hxz.symm)).trans hz)

  85 /-- A represented class on page `r + 1` is zero exactly when its E₂ label is
  86 in the actual cumulative boundary subobject after page `r`. -/
  87 public theorem RepresentsOnPage.eq_zero_iff_isBoundaryBy
  88     {E : SpectralSequence (ModuleCat.{v} R) (ℤ × ℤ)} {r : ℤ} {p : ℤ × ℤ}
  89     {x : E.Page 2 p} {yr : E.Page (r + 1) p} (hr : 2 ≤ r)
  90     (hrep : RepresentsOnPage E (r + 1) p x yr) :
  91     yr = 0 ↔ IsBoundaryBy E r p x := by
  92   let D := E.ssData p
  93   let n : WithTop ℕ := ↑(2 - E.r₀).toNat
  94   let m : WithTop ℕ := ↑(r + 1 - E.r₀).toNat
  95   have hnm : n ≤ m := by
  96     dsimp [n, m]
  97     exact_mod_cast (show (2 - E.r₀).toNat ≤ (r + 1 - E.r₀).toNat by omega)
  98   constructor
  99   · intro hy0
 100     obtain ⟨_, z, hx, hy⟩ := hrep
 101     have hz : D.pageπ m z = 0 := hy.trans hy0
 102     obtain ⟨b, hb⟩ := (D.pageπ_eq_zero_iff m z).mp hz
 103     refine ⟨hr, b, ?_⟩
 104     rw [← hb] at hx
 105     change (Subobject.ofLE (D.B m) (D.B ⊤) (D.B_mono le_top) ≫
 106       Subobject.ofLE (D.B ⊤) (D.Z ⊤) (D.B_le_Z ⊤) ≫
 107       Subobject.ofLE (D.Z ⊤) (D.Z n) (D.Z_anti le_top) ≫ D.pageπ n) b = x
 108     change (Subobject.ofLE (D.B m) (D.Z m) (D.B_le_Z m) ≫
 109       Subobject.ofLE (D.Z m) (D.Z n) (D.Z_anti hnm) ≫ D.pageπ n) b = x at hx
 110     simpa only [← Category.assoc, Subobject.ofLE_comp_ofLE] using hx
 111   · rintro ⟨_, b, hb⟩
 112     have hrep0 : RepresentsOnPage E (r + 1) p x 0 := by
 113       refine ⟨by omega,
 114         Subobject.ofLE (D.B m) (D.Z m) (D.B_le_Z m) b, ?_, ?_⟩
 115       · change (Subobject.ofLE (D.B m) (D.Z m) (D.B_le_Z m) ≫
 116           Subobject.ofLE (D.Z m) (D.Z n) (D.Z_anti hnm) ≫ D.pageπ n) b = x
 117         simpa only [D, m, n, ← Category.assoc, Subobject.ofLE_comp_ofLE] using hb
 118       · exact cokernel.condition_apply _ b
 119     exact hrep.unique hrep0

 121 /-- A differential equation with the zero E₂ source label can only target an
 122 E₂ label already in the cumulative boundary space before that differential.
 123 The page bound is exactly the range in which `IsBoundaryBy (r - 1)` is defined.
 124 This uses actual common representatives, not a program boundary-span test. -/
 125 public theorem HasDifferential.isBoundaryBy_of_source_zero
 126     {E : SpectralSequence (ModuleCat.{v} R) (ℤ × ℤ)} {r : ℤ} {p q : ℤ × ℤ}
 127     {y : E.Page 2 q} (hr : 3 ≤ r) (h : HasDifferential E r p q 0 y) :
 128     IsBoundaryBy E (r - 1) q y := by
 129   obtain ⟨k, rfl⟩ : ∃ k : ℤ, r = k + 1 := ⟨r - 1, by omega⟩
 130   obtain ⟨hdeg, xr, yr, hx, hy, hd⟩ := h
 131   have hx0 : xr = 0 := hx.unique (RepresentsOnPage.zero (by omega))
 132   have hy0 : yr = 0 := by
 133     simpa only [hx0, ModuleCat.comp_apply, map_zero] using hd.symm
 134   simpa only [add_sub_cancel_right] using
 135     (hy.eq_zero_iff_isBoundaryBy (by omega)).mp hy0

 137 /-- An actual earlier-boundary exclusion refutes a zero-source equation.
 138 A database diagnostic asserting this exclusion is not a proof of the premise. -/
 139 public theorem not_hasDifferential_zero_of_not_isBoundaryBy
 140     {E : SpectralSequence (ModuleCat.{v} R) (ℤ × ℤ)} {r : ℤ} {p q : ℤ × ℤ}
 141     {y : E.Page 2 q} (hr : 3 ≤ r) (hy : ¬ IsBoundaryBy E (r - 1) q y) :
 142     ¬ HasDifferential E r p q 0 y :=
 143   fun h => hy (h.isBoundaryBy_of_source_zero hr)

 145 /-- A label excluded from the actual cumulative boundary space has a nonzero
 146 next-page continuation whenever a continuation exists. -/
 147 public theorem ReachesPage.survives_of_not_isBoundaryBy
 148     {E : SpectralSequence (ModuleCat.{v} R) (ℤ × ℤ)} {r : ℤ} {p : ℤ × ℤ}
 149     {x : E.Page 2 p} (hr : 2 ≤ r) (hx : ReachesPage E (r + 1) p x)
 150     (hboundary : ¬ IsBoundaryBy E r p x) : SurvivesTo E (r + 1) p x := by
 151   obtain ⟨xr, hxr⟩ := hx
 152   exact ⟨xr, hxr, fun hz => hboundary ((hxr.eq_zero_iff_isBoundaryBy hr).mp hz)⟩

 154 /-- Actual target exhaustiveness and exclusion of its nonzero alternative
 155 prove the zero differential once the source has an actual continuation. -/
 156 public theorem hasDifferential_zero_of_excluded_target
 157     {E : SpectralSequence (ModuleCat.{v} R) (ℤ × ℤ)} {r : ℤ} {p : ℤ × ℤ}
 158     {x : E.Page 2 p} (hr : 2 ≤ r) (hx : ReachesPage E r p x)
 159     (y : E.Page 2 (p + E.diffDeg r))
 160     (htargets : DifferentialTargets E r p x y)
 161     (hexclude : ¬ HasNonzeroDifferential E r p (p + E.diffDeg r) x y) :
 162     HasDifferential E r p (p + E.diffDeg r) x 0 := by
 163   exact hx.hasDifferential_zero hr rfl
 164     ((differentialVanishesOn_iff_not_hasNonzeroDifferential y htargets).mpr hexclude)
