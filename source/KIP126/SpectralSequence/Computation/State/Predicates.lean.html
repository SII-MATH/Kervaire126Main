   8 /-!
   9 # Mathematical state predicates for computation interfaces
  10 
  11 These predicates only use the actual internal spectral sequence and its
  12 common representatives. They do not interpret a raw program row. In
  13 particular, unknown program values are not mathematical assertions, and
  14 permanent-cycle membership does not require a nonzero E∞ class.
  15 -/
  16 
  17 namespace KIP126.Core.SpectralSequence
  18 
  19 open CategoryTheory
  20 
  21 universe u v
  22 
  23 variable {R : Type u} [Ring R]
  24 
  25 /-- A label has a representative on the stated page; its class there may
  26 be zero. This is the cycle information stored by an unresolved outgoing
  27 staircase entry, not nonzero survival. -/
  28 public def ReachesPage
  29     (E : SpectralSequence (ModuleCat.{v} R) (ℤ × ℤ))
  30     (r : ℤ) (p : ℤ × ℤ) (x : E.Page 2 p) : Prop :=
  31   ∃ xr : E.Page r p, RepresentsOnPage E r p x xr
  32 
  33 /-- The E₂ label is in the boundary subobject after page `r`. This
  34 allows an earlier hit, a sum of boundaries, and zero; an unresolved
  35 incoming staircase entry does not specify one nonzero differential. -/
  36 public def IsBoundaryBy
  37     (E : SpectralSequence (ModuleCat.{v} R) (ℤ × ℤ))
  38     (r : ℤ) (p : ℤ × ℤ) (x : E.Page 2 p) : Prop :=
  39   2 ≤ r ∧
  40     let D := E.ssData p
  41     let n : WithTop ℕ := ↑(2 - E.r₀).toNat
  42     let m : WithTop ℕ := ↑(r + 1 - E.r₀).toNat
  43     ∃ b : (Subobject.underlying.obj (D.B m) : ModuleCat R),
  44       (Subobject.ofLE (D.B m) (D.B ⊤) (D.B_mono le_top) ≫
  45         Subobject.ofLE (D.B ⊤) (D.Z ⊤) (D.B_le_Z ⊤) ≫
  46         Subobject.ofLE (D.Z ⊤) (D.Z n) (D.Z_anti le_top) ≫ D.pageπ n) b = x
  47 
  48 /-- A common permanent-cycle representative of the given E₂ class.
  49 Its E∞ image may be zero. This deliberately omits the nonvanishing
  50 condition in `NonzeroSurvival`. -/
  51 public def IsPermanentCycle
  52     (E : SpectralSequence (ModuleCat.{v} R) (ℤ × ℤ))
  53     (p : ℤ × ℤ) (x : E.Page 2 p) : Prop :=
  54   let D := E.ssData p
  55   let n : WithTop ℕ := ↑(2 - E.r₀).toNat
  56   ∃ z : (Subobject.underlying.obj (D.Z ⊤) : ModuleCat R),
  57     (Subobject.ofLE (D.Z ⊤) (D.Z n) (D.Z_anti le_top) ≫ D.pageπ n) z = x
  58 
  59 /-- Membership in the final boundary subobject, expressed at the
  60 specified E₂ label. Zero is allowed, and no finite hitting page or
  61 nonzero incoming differential is selected by this predicate. -/
  62 public def IsFinalBoundary
  63     (E : SpectralSequence (ModuleCat.{v} R) (ℤ × ℤ))
  64     (p : ℤ × ℤ) (x : E.Page 2 p) : Prop :=
  65   let D := E.ssData p
  66   let n : WithTop ℕ := ↑(2 - E.r₀).toNat
  67   ∃ b : (Subobject.underlying.obj (D.B ⊤) : ModuleCat R),
  68     (Subobject.ofLE (D.B ⊤) (D.Z ⊤) (D.B_le_Z ⊤) ≫
  69       Subobject.ofLE (D.Z ⊤) (D.Z n) (D.Z_anti le_top) ≫ D.pageπ n) b = x
  70 
  71 /-- Actual continuation and zero outgoing differential on every page
  72 in the specified nonempty window. Incoming boundaries and zero classes
  73 are allowed. Unlike a universal implication on continuations, this
  74 condition asserts that the continuations exist. -/
  75 public def IsCycleOnWindow
  76     (E : SpectralSequence (ModuleCat.{v} R) (ℤ × ℤ))
  77     (first last : ℤ) (p : ℤ × ℤ) (x : E.Page 2 p) : Prop :=
  78   2 ≤ first ∧ first ≤ last ∧
  79     ∀ r : ℤ, first ≤ r → r ≤ last →
  80       ∃ xr : E.Page r p, RepresentsOnPage E r p x xr ∧ IsPageCycle E r p xr
  81 
  82 /-- Nonzero continuation on all pages of the specified nonempty
  83 window. This only concerns the stated window, not E∞ survival. -/
  84 public def SurvivesOnWindow
  85     (E : SpectralSequence (ModuleCat.{v} R) (ℤ × ℤ))
  86     (first last : ℤ) (p : ℤ × ℤ) (x : E.Page 2 p) : Prop :=
  87   2 ≤ first ∧ first ≤ last ∧
  88     ∀ r : ℤ, first ≤ r → r ≤ last → SurvivesTo E r p x
  89 
  90 /-- No nonzero incoming hit in the specified nonempty window. This
  91 does not assert existence of continuations or absence of outgoing
  92 differentials, and is not an unrestricted permanence claim. -/
  93 public def NeverHitOnWindow
  94     (E : SpectralSequence (ModuleCat.{v} R) (ℤ × ℤ))
  95     (first last : ℤ) (p : ℤ × ℤ) (x : E.Page 2 p) : Prop :=
  96   2 ≤ first ∧ first ≤ last ∧
  97     ∀ r : ℤ, first ≤ r → r ≤ last → ¬ HitOnPage E r p x
  98 
  99 /-- Every outgoing differential at this bidegree is zero on every page
 100 from E₂ onward. This permits a zero page and does not assert survival of a label. -/
 101 public def NoOutgoingAt
 102     (E : SpectralSequence (ModuleCat.{v} R) (ℤ × ℤ)) (s t : ℤ) : Prop :=
 103   ∀ r : ℤ, 2 ≤ r → ∀ x : E.Page r (s, t), E.d r (s, t) x = 0
 104 
 105 end KIP126.Core.SpectralSequence
