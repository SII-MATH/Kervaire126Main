  12 /-- Kernel membership for an actual page element, with no survival claim. -/
  13 public def IsPageCycle {R : Type u} [Ring R]
  14     (E : SpectralSequence (ModuleCat.{v} R) (ℤ × ℤ))
  15     (r : ℤ) (p : ℤ × ℤ) (x : E.Page r p) : Prop := E.d r p x = 0
  16 
  17 /-- Membership in the image of the actual page differential with source
  18 degree `p`. The target degree is `p + E.diffDeg r`. Zero is allowed: this is
  19 boundary membership, not the assertion of a nonzero incoming hit. -/
  20 public def IsPageBoundary {R : Type u} [Ring R]
  21     (E : SpectralSequence (ModuleCat.{v} R) (ℤ × ℤ))
  22     (r : ℤ) (p : ℤ × ℤ) (y : E.Page r (p + E.diffDeg r)) : Prop :=
  23   ∃ x : E.Page r p, E.d r p x = y
  24 
  25 /-- A later-page class represented by the given E₂ class. This is not a map
  26 from all of E₂ to Eᵣ. A common cycle representative is required. -/
  27 public def RepresentsOnPage {R : Type u} [Ring R]
  28     (E : SpectralSequence (ModuleCat.{v} R) (ℤ × ℤ))
  29     (r : ℤ) (p : ℤ × ℤ) (x : E.Page 2 p) (y : E.Page r p) : Prop :=
  30   ∃ hr : 2 ≤ r,
  31     let D := E.ssData p
  32     let n : WithTop ℕ := ↑(2 - E.r₀).toNat
  33     let m : WithTop ℕ := ↑(r - E.r₀).toNat
  34     ∃ z : (Subobject.underlying.obj (D.Z m) : ModuleCat R),
  35       (Subobject.ofLE (D.Z m) (D.Z n) (D.Z_anti (by
  36         change (↑(2 - E.r₀).toNat : WithTop ℕ) ≤ ↑(r - E.r₀).toNat
  37         exact_mod_cast (show (2 - E.r₀).toNat ≤ (r - E.r₀).toNat by omega))) ≫
  38         D.pageπ n) z = x ∧ D.pageπ m z = y
  39 
  40 /-- Survival *to* Eᵣ means a nonzero class there, not that dᵣ vanishes. -/
  41 public def SurvivesTo {R : Type u} [Ring R]
  42     (E : SpectralSequence (ModuleCat.{v} R) (ℤ × ℤ))
  43     (r : ℤ) (p : ℤ × ℤ) (x : E.Page 2 p) : Prop :=
  44   ∃ y : E.Page r p, RepresentsOnPage E r p x y ∧ y ≠ 0
  45 
  46 /-- No nonzero continuation is an incoming differential target. This alone
  47 does not exclude an outgoing differential on x. -/
  48 public def NeverHit {R : Type u} [Ring R]
  49     (E : SpectralSequence (ModuleCat.{v} R) (ℤ × ℤ))
  50     (p : ℤ × ℤ) (x : E.Page 2 p) : Prop :=
  51   ∀ (r : ℤ) (_ : 2 ≤ r) (q : ℤ × ℤ) (h : q + E.diffDeg r = p)
  52     (y : E.Page r p), RepresentsOnPage E r p x y → y ≠ 0 →
  53     ∀ z : E.Page r q, (E.d r q ≫ eqToHom (congrArg (E.Page r) h)) z ≠ y
  54 
  55 /-- Vanishing on all continuations of a fixed E₂ class on one page.
  56 This predicate alone does not assert that any continuation exists. -/
  57 public def DifferentialVanishesOn {R : Type u} [Ring R]
  58     (E : SpectralSequence (ModuleCat.{v} R) (ℤ × ℤ))
  59     (r : ℤ) (p : ℤ × ℤ) (x : E.Page 2 p) : Prop :=
  60   ∀ xr : E.Page r p, RepresentsOnPage E r p x xr → E.d r p xr = 0
  61 
  62 /-- Every differential on a continuation is zero or represented by the
  63 specified target. The nonzero alternative is not assumed to occur. -/
  64 public def DifferentialTargets {R : Type u} [Ring R]
  65     (E : SpectralSequence (ModuleCat.{v} R) (ℤ × ℤ))
  66     (r : ℤ) (p : ℤ × ℤ) (x : E.Page 2 p)
  67     (y : E.Page 2 (p + E.diffDeg r)) : Prop :=
  68   ∀ xr : E.Page r p, RepresentsOnPage E r p x xr →
  69     E.d r p xr = 0 ∨
  70       ∃ yr : E.Page r (p + E.diffDeg r),
  71         RepresentsOnPage E r (p + E.diffDeg r) y yr ∧ E.d r p xr = yr
  72 
  73 /-- An equation for the existing differential between specified E₂ labels.
  74 Both labels must admit a common-cycle representative on page r. Zero targets
  75 are allowed; later-page nonvanishing is the stronger predicate below.
  76 Ported from the useful local-differential interface in KIPBase. -/
  77 public def HasDifferential {R : Type u} [Ring R]
  78     (E : SpectralSequence (ModuleCat.{v} R) (ℤ × ℤ))
  79     (r : ℤ) (p q : ℤ × ℤ) (x : E.Page 2 p) (y : E.Page 2 q) : Prop :=
  80   ∃ h : p + E.diffDeg r = q,
  81     ∃ (xr : E.Page r p) (yr : E.Page r q),
  82       RepresentsOnPage E r p x xr ∧ RepresentsOnPage E r q y yr ∧
  83         (E.d r p ≫ eqToHom (congrArg (E.Page r) h)) xr = yr
  84 
  85 /-- A nonzero differential between specified E₂ representatives, with both
  86 classes lifted to the actual r-th page and the degree transport made explicit. -/
  87 public def HasNonzeroDifferential {R : Type u} [Ring R]
  88     (E : SpectralSequence (ModuleCat.{v} R) (ℤ × ℤ))
  89     (r : ℤ) (p q : ℤ × ℤ) (x : E.Page 2 p) (y : E.Page 2 q) : Prop :=
  90   ∃ h : p + E.diffDeg r = q,
  91     ∃ (xr : E.Page r p) (yr : E.Page r q),
  92       RepresentsOnPage E r p x xr ∧ RepresentsOnPage E r q y yr ∧
  93         (E.d r p ≫ eqToHom (congrArg (E.Page r) h)) xr = yr ∧ yr ≠ 0
  94 
  95 /-- Some differential hits a nonzero r-page continuation of x. -/
  96 public def HitOnPage {R : Type u} [Ring R]
  97     (E : SpectralSequence (ModuleCat.{v} R) (ℤ × ℤ))
  98     (r : ℤ) (p : ℤ × ℤ) (x : E.Page 2 p) : Prop :=
  99   ∃ (q : ℤ × ℤ) (h : q + E.diffDeg r = p) (z : E.Page r q)
 100     (y : E.Page r p), RepresentsOnPage E r p x y ∧ y ≠ 0 ∧
 101       (E.d r q ≫ eqToHom (congrArg (E.Page r) h)) z = y
