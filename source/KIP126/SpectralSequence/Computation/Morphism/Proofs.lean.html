  27 /-- A cycle-and-boundary-preserving ambient map carries common representatives
  28 through any reindexing of the degrees, provided the starting pages agree.
  29 Both displayed maps are the canonical quotient maps of this same `SSDataMorphism`.
  30 No additivity of the reindexing or compatibility with differentials is needed. -/
  31 public theorem SSDataMorphism.representsOnPage_reindexed
  32     {E E' : SpectralSequence (ModuleCat.{v} R) (ℤ × ℤ)}
  33     {τ : (ℤ × ℤ) → (ℤ × ℤ)}
  34     (f : SSDataMorphism (ℤ × ℤ) E.ssData (fun p => E'.ssData (τ p)))
  35     (hstart : E.r₀ = E'.r₀) {r : ℤ} {p : ℤ × ℤ}
  36     {x : E.Page 2 p} {y : E.Page r p} (h : RepresentsOnPage E r p x y) :
  37     RepresentsOnPage E' r (τ p)
  38       ((f.pageMap p (↑(2 - E.r₀).toNat : WithTop ℕ) ≫ eqToHom
  39         (congrArg (fun a : ℤ => (E'.ssData (τ p)).page
  40           (↑(2 - a).toNat : WithTop ℕ)) hstart)) x)
  41       ((f.pageMap p (↑(r - E.r₀).toNat : WithTop ℕ) ≫ eqToHom
  42         (congrArg (fun a : ℤ => (E'.ssData (τ p)).page
  43           (↑(r - a).toNat : WithTop ℕ)) hstart)) y) := by
  44   rcases E with ⟨⟨r₀, D, δ, d⟩, dd, Zs, Bs⟩
  45   rcases E' with ⟨⟨r₀', D', δ', d'⟩, dd', Zs', Bs'⟩
  46   dsimp at hstart
  47   subst r₀'
  48   obtain ⟨hr, z, hx, hy⟩ := h
  49   let n : WithTop ℕ := ↑(2 - r₀).toNat
  50   let m : WithTop ℕ := ↑(r - r₀).toNat
  51   have hnm : n ≤ m := by
  52     dsimp [n, m]
  53     exact_mod_cast (show (2 - r₀).toNat ≤ (r - r₀).toNat by omega)
  54   refine ⟨hr, f.cycleMap p m z, ?_, ?_⟩
  55   · have hs := f.cycleMap_ofLE_assoc p hnm ((D' (τ p)).pageπ n)
  56     have hp := f.pageπ_pageMap p n
  57     have heq :
  58         (Subobject.ofLE ((D p).Z m) ((D p).Z n) ((D p).Z_anti hnm) ≫
  59           (D p).pageπ n) ≫ f.pageMap p n =
  60         f.cycleMap p m ≫
  61           Subobject.ofLE ((D' (τ p)).Z m) ((D' (τ p)).Z n)
  62             ((D' (τ p)).Z_anti hnm) ≫ (D' (τ p)).pageπ n := by
  63       rw [Category.assoc, hp]
  64       exact hs
  65     have hz := congrArg (fun a => a z) heq
  66     simpa only [ModuleCat.comp_apply, hx, eqToHom_refl, Category.comp_id, n, m]
  67       using hz.symm
  68   · have hz := congrArg (fun a => a z) (f.pageπ_pageMap p m)
  69     simpa only [ModuleCat.comp_apply, hy, eqToHom_refl, Category.comp_id, n, m]
  70       using hz.symm

  72 /-- Differential equations transfer along a reindexed ambient morphism whose
  73 induced quotient maps commute with the actual differentials. The reindexing
  74 need only respect each differential translation; it need not be additive.
  75 The commutation premise is a morphism law to be proved for any actual model. -/
  76 public theorem SSDataMorphism.hasDifferential_reindexed
  77     {E E' : SpectralSequence (ModuleCat.{v} R) (ℤ × ℤ)}
  78     {τ : (ℤ × ℤ) → (ℤ × ℤ)}
  79     (f : SSDataMorphism (ℤ × ℤ) E.ssData (fun p => E'.ssData (τ p)))
  80     (hstart : E.r₀ = E'.r₀)
  81     (hdegree : ∀ (r : ℤ) (p : ℤ × ℤ),
  82       τ (p + E.diffDeg r) = τ p + E'.diffDeg r)
  83     (hcomm :
  84       let Φ : ∀ (r : ℤ) (p : ℤ × ℤ), E.Page r p ⟶ E'.Page r (τ p) :=
  85         fun r p => f.pageMap p (↑(r - E.r₀).toNat : WithTop ℕ) ≫ eqToHom
  86           (congrArg (fun a : ℤ => (E'.ssData (τ p)).page
  87             (↑(r - a).toNat : WithTop ℕ)) hstart)
  88       ∀ (r : ℤ) (p : ℤ × ℤ),
  89         Φ r p ≫ E'.d r (τ p) =
  90           E.d r p ≫ Φ r (p + E.diffDeg r) ≫
  91             eqToHom (congrArg (E'.Page r) (hdegree r p)))
  92     {r : ℤ} {p q : ℤ × ℤ} {x : E.Page 2 p} {y : E.Page 2 q}
  93     (h : HasDifferential E r p q x y) :
  94     HasDifferential E' r (τ p) (τ q)
  95       ((f.pageMap p (↑(2 - E.r₀).toNat : WithTop ℕ) ≫ eqToHom
  96         (congrArg (fun a : ℤ => (E'.ssData (τ p)).page
  97           (↑(2 - a).toNat : WithTop ℕ)) hstart)) x)
  98       ((f.pageMap q (↑(2 - E.r₀).toNat : WithTop ℕ) ≫ eqToHom
  99         (congrArg (fun a : ℤ => (E'.ssData (τ q)).page
 100           (↑(2 - a).toNat : WithTop ℕ)) hstart)) y) := by
 101   let Φ : ∀ (r : ℤ) (p : ℤ × ℤ), E.Page r p ⟶ E'.Page r (τ p) :=
 102     fun r p => f.pageMap p (↑(r - E.r₀).toNat : WithTop ℕ) ≫ eqToHom
 103       (congrArg (fun a : ℤ => (E'.ssData (τ p)).page
 104         (↑(r - a).toNat : WithTop ℕ)) hstart)
 105   change ∀ (r : ℤ) (p : ℤ × ℤ),
 106     Φ r p ≫ E'.d r (τ p) = E.d r p ≫ Φ r (p + E.diffDeg r) ≫
 107       eqToHom (congrArg (E'.Page r) (hdegree r p)) at hcomm
 108   change HasDifferential E' r (τ p) (τ q) (Φ 2 p x) (Φ 2 q y)
 109   obtain ⟨hdeg, xr, yr, hx, hy, hd⟩ := h
 110   have hdeg' : τ p + E'.diffDeg r = τ q :=
 111     (hdegree r p).symm.trans (congrArg τ hdeg)
 112   refine ⟨hdeg', Φ r p xr, Φ r q yr, ?_, ?_, ?_⟩
 113   · exact f.representsOnPage_reindexed hstart hx
 114   · exact f.representsOnPage_reindexed hstart hy
 115   · have heq : Φ r p ≫ E'.d r (τ p) ≫ eqToHom (congrArg (E'.Page r) hdeg') =
 116         (E.d r p ≫ eqToHom (congrArg (E.Page r) hdeg)) ≫ Φ r q := by
 117       rw [← Category.assoc, hcomm, Category.assoc]
 118       subst q
 119       simp
 120     have hz := congrArg (fun a => a xr) heq
 121     simpa only [ModuleCat.comp_apply, hd] using hz

 123 /-- A common cycle representative maps to a common cycle representative. -/
 124 public theorem SpectralSequenceMorphism.representsOnPage
 125     {E E' : SpectralSequence (ModuleCat.{v} R) (ℤ × ℤ)}
 126     (f : SpectralSequenceMorphism E E') {r : ℤ} {p : ℤ × ℤ}
 127     {x : E.Page 2 p} {y : E.Page r p} (h : RepresentsOnPage E r p x y) :
 128     RepresentsOnPage E' r p (f.pageMap 2 p x) (f.pageMap r p y) := by
 129   convert f.toSSDataMorphism.representsOnPage_reindexed (E := E) (E' := E')
 130     (τ := id) f.r₀_eq h using 1 <;> rfl

 132 /-- Differential equations between named E₂ representatives are natural.
 133 The image equation can have zero source or target. -/
 134 public theorem SpectralSequenceMorphism.hasDifferential
 135     {E E' : SpectralSequence (ModuleCat.{v} R) (ℤ × ℤ)}
 136     (f : SpectralSequenceMorphism E E') {r : ℤ} {p q : ℤ × ℤ}
 137     {x : E.Page 2 p} {y : E.Page 2 q} (h : HasDifferential E r p q x y) :
 138     HasDifferential E' r p q (f.pageMap 2 p x) (f.pageMap 2 q y) := by
 139   have hdegree : ∀ (r : ℤ) (p : ℤ × ℤ),
 140       id (p + E.diffDeg r) = id p + E'.diffDeg r := by
 141     intro r p
 142     rw [f.diffDeg_eq]
 143     rfl
 144   have hf := f.toSSDataMorphism.hasDifferential_reindexed (E := E) (E' := E')
 145     (τ := id) f.r₀_eq hdegree (fun r p => by
 146       convert f.pageMap_comm_d r p using 1 <;> rfl) h
 147   convert hf using 1 <;> rfl
