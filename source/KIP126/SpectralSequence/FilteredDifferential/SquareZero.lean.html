  14 open CategoryTheory CategoryTheory.Limits

  16 universe u v

  18 variable {C : Type u} [Category.{v} C] [Abelian C]

  20 open DifferentialHelpers

  22 set_option backward.isDefEq.respectTransparency false in
  23 theorem pageDifferential_comp (FC : FilteredComplex C)
  24     (s k : ℤ) (n : ℕ) :
  25     FC.pageDifferential s k n ≫ FC.pageDifferential (s + ↑n) (k - 1) n = 0 := by
  26   set f₁ := Subobject.ofLE (FC.boundarySubobject s k ↑n) (FC.cycleSubobject s k ↑n)
  27     (FC.B_le_Z_aux s k ↑n)
  28   haveI : Epi (cokernel.π f₁) := inferInstance
  29   rw [show FC.pageDifferential s k n ≫ FC.pageDifferential (s + ↑n) (k - 1) n =
  30     FC.pageDifferential s k n ≫ FC.pageDifferential (s + ↑n) (k - 1) n from rfl]
  31   rw [← cancel_epi (cokernel.π f₁), comp_zero, ← Category.assoc]
  32   erw [cokernel.π_desc]
  33   set f_n₁ := (FC.filtration.F s k).arrow ≫ FC.complex.d k (k - 1) ≫ cokernel.π ((FC.filtration.F (s + ↑n) (k - 1)).arrow)
  34   set kerZ₁ := kernelSubobject f_n₁
  35   set ι₁ := Subobject.ofLE (FC.filtration.F (s + 1) k) (FC.filtration.F s k) (FC.filtration.mono s k)
  36   set πV₁ := cokernel.π ι₁
  37   set p₁ := factorThruImageSubobject (kerZ₁.arrow ≫ πV₁)
  38   haveI : Epi p₁ := inferInstance
  39   rw [← cancel_epi p₁, comp_zero, ← Category.assoc]
  40   erw [Abelian.comp_epiDesc]
  41   rw [Category.assoc]
  42   erw [cokernel.π_desc]
  43   rw [Category.assoc]
  44   erw [Abelian.comp_epiDesc]
  45   simp only [Category.assoc]
  46   erw [show ∀ {A' B' C' D' : C} (f : A' ⟶ B') (g : B' ⟶ C') (h : C' ⟶ D'),
  47     f ≫ (g ≫ h) = (f ≫ g) ≫ h from fun f g h => (Category.assoc f g h).symm]
  48   suffices h_zero : _ ≫ _ = (0 : _ ⟶ Subobject.underlying.obj (kernelSubobject
  49     ((FC.filtration.F (s + ↑n + ↑n) (k - 1 - 1)).arrow ≫ FC.complex.d (k - 1 - 1) (k - 1 - 1 - 1) ≫
  50       cokernel.π ((FC.filtration.F (s + ↑n + ↑n + ↑n) (k - 1 - 1 - 1)).arrow)))) by
  51     erw [h_zero, zero_comp]
  52   apply (inferInstance : Mono (kernelSubobject
  53     ((FC.filtration.F (s + ↑n + ↑n) (k - 1 - 1)).arrow ≫ FC.complex.d (k - 1 - 1) (k - 1 - 1 - 1) ≫
  54       cokernel.π ((FC.filtration.F (s + ↑n + ↑n + ↑n) (k - 1 - 1 - 1)).arrow))).arrow).right_cancellation
  55   simp only [zero_comp, Category.assoc]
  56   erw [factorThruKernelSubobject_comp_arrow]
  57   apply (inferInstance : Mono (FC.filtration.F (s + ↑n + ↑n) (k - 1 - 1)).arrow).right_cancellation
  58   simp only [zero_comp, Category.assoc]
  59   erw [Abelian.monoLift_comp]
  60   conv_lhs => erw [← Category.assoc, factorThruKernelSubobject_comp_arrow]
  61   conv_lhs => erw [← Category.assoc, Abelian.monoLift_comp]
  62   erw [Category.assoc, Category.assoc, FC.complex.d_comp_d k, comp_zero, comp_zero]
  63 
  64 end KIP126.Core.SpectralSequence.FilteredComplex
