  20 theorem kernelSubobject_epiDesc' {C' : Type*} [Category C'] [Abelian C']
  21     {A' B' D' : C'} (p : A' ⟶ B') [Epi p] (g : A' ⟶ D')
  22     (hg : kernel.ι p ≫ g = 0) :
  23     kernelSubobject (Abelian.epiDesc p g hg) =
  24       imageSubobject ((kernelSubobject g).arrow ≫ p) := by
  25   apply le_antisymm
  26   · -- (≤): ker(epiDesc) ≤ image((ker g).arrow ≫ p)
  27     set K := kernelSubobject (Abelian.epiDesc p g hg) with hK_def
  28     -- Form the pullback of p (epi) and K.arrow (mono)
  29     have hpb_comm : pullback.fst p K.arrow ≫ p = pullback.snd p K.arrow ≫ K.arrow :=
  30       pullback.condition
  31     haveI : Mono (pullback.fst p K.arrow) :=
  32       pullback.fst_of_mono (f := p) (g := K.arrow)
  33     haveI : Epi (pullback.snd p K.arrow) :=
  34       Abelian.epi_pullback_of_epi_f p K.arrow
  35     -- Show pullback.fst ≫ g = 0
  36     have h_fst_g_zero : pullback.fst p K.arrow ≫ g = 0 := by
  37       calc pullback.fst p K.arrow ≫ g
  38           = pullback.fst p K.arrow ≫ (p ≫ Abelian.epiDesc p g hg) :=
  39             by rw [Abelian.comp_epiDesc]
  40         _ = (pullback.fst p K.arrow ≫ p) ≫ Abelian.epiDesc p g hg :=
  41             (Category.assoc _ _ _).symm
  42         _ = (pullback.snd p K.arrow ≫ K.arrow) ≫ Abelian.epiDesc p g hg :=
  43             by rw [hpb_comm]
  44         _ = pullback.snd p K.arrow ≫ (K.arrow ≫ Abelian.epiDesc p g hg) :=
  45             Category.assoc _ _ _
  46         _ = pullback.snd p K.arrow ≫ 0 := by rw [kernelSubobject_arrow_comp]
  47         _ = 0 := comp_zero
  48     -- Factor through ker(g)
  49     set φ := factorThruKernelSubobject g (pullback.fst p K.arrow) h_fst_g_zero
  50     have hφ_spec : φ ≫ (kernelSubobject g).arrow = pullback.fst p K.arrow :=
  51       factorThruKernelSubobject_comp_arrow g _ _
  52     -- Key equation: φ ≫ ker(g).arrow ≫ p = pullback.snd ≫ K.arrow
  53     have h_key_eq : φ ≫ (kernelSubobject g).arrow ≫ p =
  54         pullback.snd p K.arrow ≫ K.arrow := by
  55       rw [show φ ≫ (kernelSubobject g).arrow ≫ p =
  56         (φ ≫ (kernelSubobject g).arrow) ≫ p from (Category.assoc _ _ _).symm,
  57         hφ_spec, hpb_comm]
  58     -- image(pullback.snd ≫ K.arrow) = K (since snd is epi, K.arrow is mono)
  59     have h_img_epi_comp : ∀ {X₁ X₂ X₃ : C'} (e : X₁ ⟶ X₂) [Epi e] (h : X₂ ⟶ X₃),
  60         imageSubobject (e ≫ h) = imageSubobject h := by
  61       intro X₁ X₂ X₃ e _ h
  62       have hle := imageSubobject_comp_le e h
  63       haveI : Epi (Subobject.ofLE _ _ hle) := imageSubobject_comp_le_epi_of_epi e h
  64       haveI : IsIso (Subobject.ofLE _ _ hle) := isIso_of_mono_of_epi _
  65       exact le_antisymm hle (Subobject.le_of_comm (inv (Subobject.ofLE _ _ hle))
  66         (by rw [IsIso.inv_comp_eq]; exact (Subobject.ofLE_arrow hle).symm))
  67     have h_img_epi : imageSubobject (pullback.snd p K.arrow ≫ K.arrow) = K := by
  68       rw [h_img_epi_comp, imageSubobject_mono, Subobject.mk_arrow]
  69     calc K = imageSubobject (pullback.snd p K.arrow ≫ K.arrow) := h_img_epi.symm
  70       _ = imageSubobject (φ ≫ (kernelSubobject g).arrow ≫ p) := by rw [h_key_eq]
  71       _ ≤ imageSubobject ((kernelSubobject g).arrow ≫ p) :=
  72             imageSubobject_comp_le φ ((kernelSubobject g).arrow ≫ p)
  73   · -- (≥): image((ker g).arrow ≫ p) ≤ ker(epiDesc)
  74     apply le_kernelSubobject
  75     rw [← cancel_epi (factorThruImageSubobject ((kernelSubobject g).arrow ≫ p)),
  76         comp_zero, ← Category.assoc,
  77         imageSubobject_arrow_comp ((kernelSubobject g).arrow ≫ p),
  78         Category.assoc, Abelian.comp_epiDesc, kernelSubobject_arrow_comp]

  83 theorem kernelSubobject_cokernel_desc' {C' : Type*} [Category C'] [Abelian C']
  84     {A' B' D' : C'} (f : A' ⟶ B') (g : B' ⟶ D') (w : f ≫ g = 0) :
  85     kernelSubobject (cokernel.desc f g w) =
  86       imageSubobject ((kernelSubobject g).arrow ≫ cokernel.π f) := by
  87   apply le_antisymm
  88   · -- (≤): ker(desc) ≤ image(ker(g).arrow ≫ cokernel.π f)
  89     -- Pullback argument: pull back π (epi) along K.arrow (mono) where K = ker(desc).
  90     -- The pullback P has:
  91     --   pullback.fst : P → B' (mono, since K.arrow is mono)
  92     --   pullback.snd : P → K (epi, since π is epi)
  93     -- Then fst ≫ g = fst ≫ π ≫ desc = snd ≫ K.arrow ≫ desc = 0,
  94     -- so fst factors through ker(g). This shows image(ker(g).arrow ≫ π) ⊇ K.
  95     set K := kernelSubobject (cokernel.desc f g w) with hK_def
  96     set π := cokernel.π f with hπ_def
  97     set desc' := cokernel.desc f g w with hdesc'_def
  98     -- Form the pullback of π and K.arrow
  99     have hpb_comm : pullback.fst π K.arrow ≫ π = pullback.snd π K.arrow ≫ K.arrow :=
 100       pullback.condition
 101     -- pullback.fst is mono (since K.arrow is mono)
 102     haveI : Mono (pullback.fst π K.arrow) :=
 103       pullback.fst_of_mono (f := π) (g := K.arrow)
 104     -- pullback.snd is epi (since π is epi, in abelian category)
 105     haveI : Epi (pullback.snd π K.arrow) :=
 106       Abelian.epi_pullback_of_epi_f π K.arrow
 107     -- Show pullback.fst ≫ g = 0, so it factors through ker(g)
 108     have h_fst_g_zero : pullback.fst π K.arrow ≫ g = 0 := by
 109       calc pullback.fst π K.arrow ≫ g
 110           = pullback.fst π K.arrow ≫ (π ≫ desc') := by rw [hπ_def, cokernel.π_desc]
 111         _ = (pullback.fst π K.arrow ≫ π) ≫ desc' := (Category.assoc _ _ _).symm
 112         _ = (pullback.snd π K.arrow ≫ K.arrow) ≫ desc' := by rw [hpb_comm]
 113         _ = pullback.snd π K.arrow ≫ (K.arrow ≫ desc') := Category.assoc _ _ _
 114         _ = pullback.snd π K.arrow ≫ 0 := by rw [kernelSubobject_arrow_comp]
 115         _ = 0 := comp_zero
 116     -- Factor through ker(g)
 117     set φ := factorThruKernelSubobject g (pullback.fst π K.arrow) h_fst_g_zero
 118     have hφ_spec : φ ≫ (kernelSubobject g).arrow = pullback.fst π K.arrow :=
 119       factorThruKernelSubobject_comp_arrow g _ _
 120     -- Key equation: φ ≫ ker(g).arrow ≫ π = pullback.snd ≫ K.arrow
 121     have h_key_eq : φ ≫ (kernelSubobject g).arrow ≫ π = pullback.snd π K.arrow ≫ K.arrow := by
 122       rw [show φ ≫ (kernelSubobject g).arrow ≫ π =
 123         (φ ≫ (kernelSubobject g).arrow) ≫ π from (Category.assoc _ _ _).symm,
 124         hφ_spec, hpb_comm]
 125     -- Now: image(pullback.snd ≫ K.arrow) = image(K.arrow) since pullback.snd is epi
 126     -- Use: imageSubobject(e ≫ m) = imageSubobject(m) when e is epi
 127     -- (inline proof since the helper in Basic.lean is private)
 128     have h_img_epi_comp : ∀ {X₁ X₂ X₃ : C'} (e : X₁ ⟶ X₂) [Epi e] (h : X₂ ⟶ X₃),
 129         imageSubobject (e ≫ h) = imageSubobject h := by
 130       intro X₁ X₂ X₃ e _ h
 131       have hle := imageSubobject_comp_le e h
 132       haveI : Epi (Subobject.ofLE _ _ hle) := imageSubobject_comp_le_epi_of_epi e h
 133       haveI : IsIso (Subobject.ofLE _ _ hle) := isIso_of_mono_of_epi _
 134       exact le_antisymm hle (Subobject.le_of_comm (inv (Subobject.ofLE _ _ hle))
 135         (by rw [IsIso.inv_comp_eq]; exact (Subobject.ofLE_arrow hle).symm))
 136     have h_img_epi : imageSubobject (pullback.snd π K.arrow ≫ K.arrow) = K := by
 137       rw [h_img_epi_comp, imageSubobject_mono, Subobject.mk_arrow]
 138     -- Rewrite: image(φ ≫ ker(g).arrow ≫ π) = image(pullback.snd ≫ K.arrow) = K
 139     -- And: image(φ ≫ ker(g).arrow ≫ π) ≤ image(ker(g).arrow ≫ π) by imageSubobject_comp_le
 140     calc K = imageSubobject (pullback.snd π K.arrow ≫ K.arrow) := h_img_epi.symm
 141       _ = imageSubobject (φ ≫ (kernelSubobject g).arrow ≫ π) := by rw [h_key_eq]
 142       _ ≤ imageSubobject ((kernelSubobject g).arrow ≫ π) :=
 143             imageSubobject_comp_le φ ((kernelSubobject g).arrow ≫ π)
 144   · -- (≥): image(ker(g).arrow ≫ cokernel.π f) ≤ ker(desc)
 145     apply le_kernelSubobject
 146     rw [← cancel_epi (factorThruImageSubobject ((kernelSubobject g).arrow ≫ cokernel.π f)),
 147         comp_zero, ← Category.assoc,
 148         imageSubobject_arrow_comp ((kernelSubobject g).arrow ≫ cokernel.π f),
 149         Category.assoc, cokernel.π_desc, kernelSubobject_arrow_comp]

 151 theorem eqToHom_arrow_dToK_gen_local (FC : FilteredComplex C)
 152     (s : ℤ) (m k : ℤ) (hmk : m + 1 = k) :
 153     eqToHom (show Subobject.underlying.obj (FC.filtration.F s k) =
 154       Subobject.underlying.obj (FC.filtration.F s (m + 1)) by rw [hmk]) ≫
 155     ((FC.filtration.F s (m + 1)).arrow ≫ FC.dToK m) =
 156       (FC.filtration.F s k).arrow ≫ FC.complex.d k (k - 1) ≫
 157       eqToHom (congr_arg FC.complex.X (show k - 1 = m by omega)) := by
 158   subst hmk
 159   simp [FilteredComplex.dToK]

 161 theorem eqToHom_arrow_dToK_local (FC : FilteredComplex C)
 162     (s k : ℤ) :
 163     eqToHom (show Subobject.underlying.obj (FC.filtration.F s k) =
 164       Subobject.underlying.obj (FC.filtration.F s ((k - 1) + 1)) by
 165       rw [show (k - 1 : ℤ) + 1 = k from by omega]) ≫
 166     ((FC.filtration.F s ((k - 1) + 1)).arrow ≫ FC.dToK (k - 1)) =
 167       (FC.filtration.F s k).arrow ≫ FC.complex.d k (k - 1) := by
 168   rw [eqToHom_arrow_dToK_gen_local FC s (k - 1) k (by omega)]
 169   simp

 171 lemma imageSubobject_epi_comp'_local {C' : Type*} [Category C'] [Abelian C']
 172     {X₁ X₂ X₃ : C'} (e : X₁ ⟶ X₂) [Epi e] (f : X₂ ⟶ X₃) :
 173     imageSubobject (e ≫ f) = imageSubobject f := by
 174   apply le_antisymm (imageSubobject_comp_le e f)
 175   have hle := imageSubobject_comp_le e f
 176   haveI : Epi (Subobject.ofLE _ _ hle) := imageSubobject_comp_le_epi_of_epi e f
 177   haveI : IsIso (Subobject.ofLE _ _ hle) := isIso_of_mono_of_epi _
 178   exact Subobject.le_of_comm (inv (Subobject.ofLE _ _ hle))
 179     (by rw [IsIso.inv_comp_eq]; exact (Subobject.ofLE_arrow hle).symm)
