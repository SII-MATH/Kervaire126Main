  21 /-- Compute the canonical page differential on a cycle-kernel representative.
  22 If its actual differential is represented by `v` in the target filtration,
  23 the differential of its page class is the page class of `v`. -/
  24 theorem pageDifferential_on_cycle_kernel (FC : FilteredComplex C)
  25     (s k : ℤ) (n : ℕ) :
  26     let f := (FC.filtration.F s k).arrow ≫ FC.complex.d k (k - 1) ≫
  27       cokernel.π ((FC.filtration.F (s + ↑n) (k - 1)).arrow)
  28     let g := (FC.filtration.F (s + ↑n) (k - 1)).arrow ≫
  29       FC.complex.d (k - 1) (k - 1 - 1) ≫
  30       cokernel.π ((FC.filtration.F (s + ↑n + ↑n) (k - 1 - 1)).arrow)
  31     let K := kernelSubobject f
  32     let K' := kernelSubobject g
  33     ∀ {T : C} (u : T ⟶ Subobject.underlying.obj K)
  34       (v : T ⟶ Subobject.underlying.obj (FC.filtration.F (s + ↑n) (k - 1)))
  35       (_hv : u ≫ K.arrow ≫ (FC.filtration.F s k).arrow ≫ FC.complex.d k (k - 1) =
  36         v ≫ (FC.filtration.F (s + ↑n) (k - 1)).arrow)
  37       (hvg : v ≫ g = 0),
  38       (u ≫ factorThruImageSubobject
  39         (K.arrow ≫ FC.filtration.toAssociatedGraded s k)) ≫
  40           FC.pageπ s k (↑n) ≫ FC.pageDifferential s k n =
  41         (factorThruKernelSubobject g v hvg ≫
  42           factorThruImageSubobject
  43             (K'.arrow ≫ FC.filtration.toAssociatedGraded (s + ↑n) (k - 1))) ≫
  44           FC.pageπ (s + ↑n) (k - 1) (↑n) := by
  45   dsimp only
  46   intro T u v hv hvg
  47   unfold FilteredComplex.pageDifferential
  48   simp only [Category.assoc]
  49   erw [cokernel.π_desc, Abelian.comp_epiDesc]
  50   let f := (FC.filtration.F s k).arrow ≫ FC.complex.d k (k - 1) ≫
  51     cokernel.π ((FC.filtration.F (s + ↑n) (k - 1)).arrow)
  52   let g := (FC.filtration.F (s + ↑n) (k - 1)).arrow ≫
  53     FC.complex.d (k - 1) (k - 1 - 1) ≫
  54     cokernel.π ((FC.filtration.F (s + ↑n + ↑n) (k - 1 - 1)).arrow)
  55   let K := kernelSubobject f
  56   let K' := kernelSubobject g
  57   have hfactor : (K.arrow ≫ (FC.filtration.F s k).arrow ≫
  58       FC.complex.d k (k - 1)) ≫
  59       cokernel.π ((FC.filtration.F (s + ↑n) (k - 1)).arrow) = 0 := by
  60     simpa only [f, Category.assoc] using kernelSubobject_arrow_comp f
  61   let lift_n := Abelian.monoLift (FC.filtration.F (s + ↑n) (k - 1)).arrow
  62     (K.arrow ≫ (FC.filtration.F s k).arrow ≫ FC.complex.d k (k - 1)) hfactor
  63   have hcycle : lift_n ≫ g = 0 := by
  64     calc
  65       lift_n ≫ g =
  66           ((lift_n ≫ (FC.filtration.F (s + ↑n) (k - 1)).arrow) ≫
  67             FC.complex.d (k - 1) (k - 1 - 1)) ≫
  68               cokernel.π ((FC.filtration.F (s + ↑n + ↑n) (k - 1 - 1)).arrow) := by
  69                 simp only [g, Category.assoc]
  70       _ = ((K.arrow ≫ (FC.filtration.F s k).arrow ≫ FC.complex.d k (k - 1)) ≫
  71             FC.complex.d (k - 1) (k - 1 - 1)) ≫
  72               cokernel.π ((FC.filtration.F (s + ↑n + ↑n) (k - 1 - 1)).arrow) := by
  73                 rw [Abelian.monoLift_comp]
  74       _ = 0 := by
  75         simp only [Category.assoc, FC.complex.d_comp_d, comp_zero, zero_comp]
  76   have hlift : u ≫ lift_n = v := by
  77     apply (cancel_mono (FC.filtration.F (s + ↑n) (k - 1)).arrow).mp
  78     rw [Category.assoc, Abelian.monoLift_comp]
  79     exact hv
  80   have hker : u ≫ factorThruKernelSubobject g lift_n hcycle =
  81       factorThruKernelSubobject g v hvg := by
  82     apply (cancel_mono K'.arrow).mp
  83     calc
  84       (u ≫ factorThruKernelSubobject g lift_n hcycle) ≫ K'.arrow =
  85           u ≫ lift_n := by
  86             rw [Category.assoc, factorThruKernelSubobject_comp_arrow]
  87       _ = v := hlift
  88       _ = factorThruKernelSubobject g v hvg ≫ K'.arrow := by
  89             rw [factorThruKernelSubobject_comp_arrow]
  90   simpa only [Category.assoc, FilteredComplex.pageπ] using
  91     congrArg (fun z : T ⟶ Subobject.underlying.obj K' =>
  92       z ≫ factorThruImageSubobject
  93         (K'.arrow ≫ FC.filtration.toAssociatedGraded (s + ↑n) (k - 1)) ≫
  94         FC.pageπ (s + ↑n) (k - 1) (↑n)) hker

  96 /-- A strict filtered differential equation produces compatible source and
  97 target cycle representatives whose quotient classes satisfy the canonical
  98 page differential relation. -/
  99 theorem pageDifferential_of_strict_lifts (FC : FilteredComplex C)
 100     (s k : ℤ) (n : ℕ) {T : C}
 101     (xl : T ⟶ Subobject.underlying.obj (FC.filtration.F s k))
 102     (yl : T ⟶ Subobject.underlying.obj (FC.filtration.F (s + ↑n) (k - 1)))
 103     (hd : xl ≫ (FC.differential_preserves s k).choose =
 104       yl ≫ FC.filtration.inclusion (by omega) (k - 1)) :
 105     ∃ (zx : T ⟶ Subobject.underlying.obj (FC.cycleSubobject s k (↑n)))
 106       (zy : T ⟶ Subobject.underlying.obj
 107         (FC.cycleSubobject (s + ↑n) (k - 1) (↑n))),
 108       zx ≫ (FC.cycleSubobject s k (↑n)).arrow =
 109         xl ≫ FC.filtration.toAssociatedGraded s k ∧
 110       zy ≫ (FC.cycleSubobject (s + ↑n) (k - 1) (↑n)).arrow =
 111         yl ≫ FC.filtration.toAssociatedGraded (s + ↑n) (k - 1) ∧
 112       zx ≫ FC.pageπ s k (↑n) ≫ FC.pageDifferential s k n =
 113         zy ≫ FC.pageπ (s + ↑n) (k - 1) (↑n) := by
 114   let f := (FC.filtration.F s k).arrow ≫ FC.complex.d k (k - 1) ≫
 115     cokernel.π ((FC.filtration.F (s + ↑n) (k - 1)).arrow)
 116   let g := (FC.filtration.F (s + ↑n) (k - 1)).arrow ≫
 117     FC.complex.d (k - 1) (k - 1 - 1) ≫
 118     cokernel.π ((FC.filtration.F (s + ↑n + ↑n) (k - 1 - 1)).arrow)
 119   let K := kernelSubobject f
 120   let K' := kernelSubobject g
 121   have hamb : xl ≫ (FC.filtration.F s k).arrow ≫ FC.complex.d k (k - 1) =
 122       yl ≫ (FC.filtration.F (s + ↑n) (k - 1)).arrow := by
 123     calc
 124       xl ≫ (FC.filtration.F s k).arrow ≫ FC.complex.d k (k - 1) =
 125           xl ≫ (FC.differential_preserves s k).choose ≫
 126             (FC.filtration.F s (k - 1)).arrow := by
 127               simpa only [Category.assoc] using
 128                 congrArg (fun q => xl ≫ q)
 129                   (FC.differential_preserves s k).choose_spec.symm
 130       _ = yl ≫ FC.filtration.inclusion (by omega) (k - 1) ≫
 131             (FC.filtration.F s (k - 1)).arrow := by
 132               simpa only [Category.assoc] using
 133                 congrArg (fun q => q ≫ (FC.filtration.F s (k - 1)).arrow) hd
 134       _ = yl ≫ (FC.filtration.F (s + ↑n) (k - 1)).arrow := by
 135         simpa only [Category.assoc] using
 136           congrArg (fun q => yl ≫ q)
 137             (FC.filtration.inclusion_arrow (by omega) (k - 1))
 138   have hxl0 : xl ≫ f = 0 := by
 139     calc
 140       xl ≫ f =
 141           (xl ≫ (FC.filtration.F s k).arrow ≫ FC.complex.d k (k - 1)) ≫
 142             cokernel.π ((FC.filtration.F (s + ↑n) (k - 1)).arrow) := by
 143               simp only [f, Category.assoc]
 144       _ = (yl ≫ (FC.filtration.F (s + ↑n) (k - 1)).arrow) ≫
 145             cokernel.π ((FC.filtration.F (s + ↑n) (k - 1)).arrow) := by rw [hamb]
 146       _ = 0 := by rw [Category.assoc, cokernel.condition, comp_zero]
 147   have hyl0 : yl ≫ g = 0 := by
 148     calc
 149       yl ≫ g =
 150           ((yl ≫ (FC.filtration.F (s + ↑n) (k - 1)).arrow) ≫
 151             FC.complex.d (k - 1) (k - 1 - 1)) ≫
 152               cokernel.π ((FC.filtration.F (s + ↑n + ↑n) (k - 1 - 1)).arrow) := by
 153                 simp only [g, Category.assoc]
 154       _ = ((xl ≫ (FC.filtration.F s k).arrow ≫ FC.complex.d k (k - 1)) ≫
 155             FC.complex.d (k - 1) (k - 1 - 1)) ≫
 156               cokernel.π ((FC.filtration.F (s + ↑n + ↑n) (k - 1 - 1)).arrow) := by
 157                 rw [hamb]
 158       _ = 0 := by
 159         simp only [Category.assoc, FC.complex.d_comp_d, comp_zero, zero_comp]
 160   let u := factorThruKernelSubobject f xl hxl0
 161   let v := factorThruKernelSubobject g yl hyl0
 162   let zx := u ≫ factorThruImageSubobject
 163     (K.arrow ≫ FC.filtration.toAssociatedGraded s k)
 164   let zy := v ≫ factorThruImageSubobject
 165     (K'.arrow ≫ FC.filtration.toAssociatedGraded (s + ↑n) (k - 1))
 166   refine ⟨zx, zy, ?_, ?_, ?_⟩
 167   · change zx ≫
 168       (imageSubobject (K.arrow ≫ FC.filtration.toAssociatedGraded s k)).arrow = _
 169     simp only [zx, Category.assoc, imageSubobject_arrow_comp]
 170     simpa only [Category.assoc] using
 171       congrArg (fun q => q ≫ FC.filtration.toAssociatedGraded s k)
 172         (factorThruKernelSubobject_comp_arrow f xl hxl0)
 173   · change zy ≫ (imageSubobject
 174       (K'.arrow ≫ FC.filtration.toAssociatedGraded (s + ↑n) (k - 1))).arrow = _
 175     simp only [zy, Category.assoc, imageSubobject_arrow_comp]
 176     simpa only [Category.assoc] using
 177       congrArg (fun q => q ≫ FC.filtration.toAssociatedGraded (s + ↑n) (k - 1))
 178         (factorThruKernelSubobject_comp_arrow g yl hyl0)
 179   · have hv : u ≫ K.arrow ≫ (FC.filtration.F s k).arrow ≫
 180         FC.complex.d k (k - 1) =
 181         yl ≫ (FC.filtration.F (s + ↑n) (k - 1)).arrow := by
 182       calc
 183         u ≫ K.arrow ≫ (FC.filtration.F s k).arrow ≫ FC.complex.d k (k - 1) =
 184             xl ≫ (FC.filtration.F s k).arrow ≫ FC.complex.d k (k - 1) := by
 185               simpa only [Category.assoc] using
 186                 congrArg (fun q => q ≫ (FC.filtration.F s k).arrow ≫
 187                   FC.complex.d k (k - 1))
 188                   (factorThruKernelSubobject_comp_arrow f xl hxl0)
 189         _ = yl ≫ (FC.filtration.F (s + ↑n) (k - 1)).arrow := hamb
 190     exact FC.pageDifferential_on_cycle_kernel s k n u yl hv hyl0
