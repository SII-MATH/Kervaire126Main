  23 /-- The image of the finite-page differential is the page image of the next
  24     boundary subobject included in the current cycle subobject.  This is the
  25     canonical finite-page `B_succ` relation, adapted from the historical
  26     filtered-complex proof. -/
  27 theorem pageDifferential_B_succ (FC : FilteredComplex C)
  28     (s k : ℤ) (n : ℕ) :
  29     imageSubobject (FC.pageDifferential s k n) =
  30       imageSubobject (Subobject.ofLE
  31         (FC.boundarySubobject (s + ↑n) (k - 1) ↑(n + 1))
  32         (FC.cycleSubobject (s + ↑n) (k - 1) ↑n)
  33         (le_trans (FC.B_le_Z_aux (s + ↑n) (k - 1) ↑(n + 1))
  34           (FC.cycleSubobject_antitone (s + ↑n) (k - 1)
  35             (by exact_mod_cast Nat.le_succ n))) ≫
  36         FC.pageπ (s + ↑n) (k - 1) ↑n) := by
  37   -- === Reconstruct the internal abbreviations of pageDifferential ===
  38   set ι_s := Subobject.ofLE (FC.filtration.F (s + 1) k) (FC.filtration.F s k) (FC.filtration.mono s k) with hι_s_def
  39   set πV := FC.filtration.toAssociatedGraded s k with hπV_def
  40   set f_n := (FC.filtration.F s k).arrow ≫ FC.complex.d k (k - 1) ≫
  41     cokernel.π ((FC.filtration.F (s + ↑n) (k - 1)).arrow) with hf_n_def
  42   set kerZ := kernelSubobject f_n with hkerZ_def
  43   set ι_t := Subobject.ofLE (FC.filtration.F (s + ↑n + 1) (k - 1)) (FC.filtration.F (s + ↑n) (k - 1))
  44     (FC.filtration.mono (s + ↑n) (k - 1)) with hι_t_def
  45   set πV' := FC.filtration.toAssociatedGraded (s + ↑n) (k - 1) with hπV'_def
  46   set f_n' := (FC.filtration.F (s + ↑n) (k - 1)).arrow ≫ FC.complex.d (k - 1) (k - 1 - 1) ≫
  47     cokernel.π ((FC.filtration.F (s + ↑n + ↑n) (k - 1 - 1)).arrow) with hf_n'_def
  48   set kerZ' := kernelSubobject f_n' with hkerZ'_def
  49   -- Reconstruct lift_n, lift_to_kerZ', to_Z_n_t, ψ, p, h_on_Zn
  50   have h_ker_fn : kerZ.arrow ≫ f_n = 0 := kernelSubobject_arrow_comp f_n
  51   have h_factor_zero : (kerZ.arrow ≫ (FC.filtration.F s k).arrow ≫ FC.complex.d k (k - 1)) ≫
  52       cokernel.π ((FC.filtration.F (s + ↑n) (k - 1)).arrow) = 0 := by
  53     simp only [Category.assoc] at h_ker_fn ⊢; exact h_ker_fn
  54   set lift_n := Abelian.monoLift (FC.filtration.F (s + ↑n) (k - 1)).arrow
  55     (kerZ.arrow ≫ (FC.filtration.F s k).arrow ≫ FC.complex.d k (k - 1)) h_factor_zero with hlift_n_def
  56   have h_lift_spec : lift_n ≫ (FC.filtration.F (s + ↑n) (k - 1)).arrow =
  57       kerZ.arrow ≫ (FC.filtration.F s k).arrow ≫ FC.complex.d k (k - 1) := Abelian.monoLift_comp _ _ _
  58   have h_lift_in_kerZ' : lift_n ≫ f_n' = 0 := by
  59     calc lift_n ≫ f_n'
  60         = ((lift_n ≫ (FC.filtration.F (s + ↑n) (k - 1)).arrow) ≫ FC.complex.d (k - 1) (k - 1 - 1)) ≫
  61             cokernel.π ((FC.filtration.F (s + ↑n + ↑n) (k - 1 - 1)).arrow) := by
  62           simp only [hf_n'_def, Category.assoc]
  63       _ = ((kerZ.arrow ≫ (FC.filtration.F s k).arrow ≫ FC.complex.d k (k - 1)) ≫ FC.complex.d (k - 1) (k - 1 - 1)) ≫
  64             cokernel.π ((FC.filtration.F (s + ↑n + ↑n) (k - 1 - 1)).arrow) := by rw [h_lift_spec]
  65       _ = 0 := by
  66           rw [show (kerZ.arrow ≫ (FC.filtration.F s k).arrow ≫ FC.complex.d k (k - 1)) ≫ FC.complex.d (k - 1) (k - 1 - 1) =
  67             kerZ.arrow ≫ (FC.filtration.F s k).arrow ≫ (FC.complex.d k (k - 1) ≫ FC.complex.d (k - 1) (k - 1 - 1)) from by
  68             simp only [Category.assoc], FC.complex.d_comp_d k]; simp only [comp_zero, zero_comp]
  69   set lift_to_kerZ' := factorThruKernelSubobject f_n' lift_n h_lift_in_kerZ'
  70     with hlift_to_kerZ'_def
  71   have h_ltk_spec : lift_to_kerZ' ≫ kerZ'.arrow = lift_n :=
  72     factorThruKernelSubobject_comp_arrow f_n' lift_n h_lift_in_kerZ'
  73   set to_Z_n_t := lift_to_kerZ' ≫ factorThruImageSubobject (kerZ'.arrow ≫ πV')
  74     with hto_Z_n_t_def
  75   let ψ := to_Z_n_t ≫ FC.pageπ (s + ↑n) (k - 1) ↑n
  76   have hψ_def : ψ = to_Z_n_t ≫ FC.pageπ (s + ↑n) (k - 1) ↑n := rfl
  77   let p := factorThruImageSubobject (kerZ.arrow ≫ πV)
  78   have hp_def : p = factorThruImageSubobject (kerZ.arrow ≫ πV) := rfl
  79   haveI : Epi p := inferInstance
  80   -- h_ker_p_ψ : kernel.ι p ≫ ψ = 0 (from pageDifferential construction)
  81   -- We need this for epiDesc. Reconstruct the proof.
  82   have h_ker_p_ψ : kernel.ι p ≫ ψ = 0 := by
  83     -- This follows the same proof as in pageDifferential (lines 699-908).
  84     -- kernel.ι p ≫ to_Z_n_t factors through ofLE(B_n_t, Z_n_t), and cokernel kills it.
  85     set B_n_t' := FC.boundarySubobject (s + ↑n) (k - 1) ↑n
  86     set Z_n_t' := FC.cycleSubobject (s + ↑n) (k - 1) ↑n
  87     have hB_le_Z' := FC.B_le_Z_aux (s + ↑n) (k - 1) ↑n
  88     -- to_Z_n_t ≫ Z_n_t'.arrow = lift_n ≫ πV'
  89     have h_to_Z_comp' : to_Z_n_t ≫ Z_n_t'.arrow = lift_n ≫ πV' := by
  90       change (lift_to_kerZ' ≫ factorThruImageSubobject (kerZ'.arrow ≫ πV')) ≫
  91         (imageSubobject (kerZ'.arrow ≫ πV')).arrow = lift_n ≫ πV'
  92       simp only [Category.assoc, imageSubobject_arrow_comp]
  93       rw [show lift_to_kerZ' ≫ kerZ'.arrow ≫ πV' =
  94         (lift_to_kerZ' ≫ kerZ'.arrow) ≫ πV' from (Category.assoc _ _ _).symm, h_ltk_spec]
  95     -- kernel.ι p ≫ lift_n ≫ πV' factors through B_n_t'
  96     have h_comp_eq' : kernel.ι p ≫ lift_n ≫ (FC.filtration.F (s + ↑n) (k - 1)).arrow =
  97         kernel.ι p ≫ kerZ.arrow ≫ (FC.filtration.F s k).arrow ≫ FC.complex.d k (k - 1) := by
  98       simp only [h_lift_spec]
  99     set imgD' := imageSubobject ((FC.filtration.F (s + 1) ((k - 1) + 1)).arrow ≫ FC.dToK (k - 1))
 100     have h_fac_imgD' : imgD'.Factors
 101         (kernel.ι p ≫ lift_n ≫ (FC.filtration.F (s + ↑n) (k - 1)).arrow) := by
 102       rw [h_comp_eq']
 103       have h1 : kernel.ι p ≫ (kerZ.arrow ≫ πV) = 0 := by
 104         have h_p_Z : p ≫ (imageSubobject (kerZ.arrow ≫ πV)).arrow = kerZ.arrow ≫ πV :=
 105           imageSubobject_arrow_comp (kerZ.arrow ≫ πV)
 106         calc kernel.ι p ≫ (kerZ.arrow ≫ πV)
 107             = kernel.ι p ≫ (p ≫ (imageSubobject (kerZ.arrow ≫ πV)).arrow) := by rw [h_p_Z]
 108           _ = (kernel.ι p ≫ p) ≫ (imageSubobject (kerZ.arrow ≫ πV)).arrow := by
 109               simp only [Category.assoc]
 110           _ = 0 ≫ (imageSubobject (kerZ.arrow ≫ πV)).arrow := by rw [kernel.condition]
 111           _ = 0 := zero_comp
 112       have h1' : (kernel.ι p ≫ kerZ.arrow) ≫ cokernel.π ι_s = 0 := by
 113         rw [Category.assoc]; exact h1
 114       set α' := Abelian.monoLift ι_s (kernel.ι p ≫ kerZ.arrow) h1'
 115       have hα'_spec : α' ≫ ι_s = kernel.ι p ≫ kerZ.arrow := Abelian.monoLift_comp _ _ _
 116       have h_ι_arrow' : ι_s ≫ (FC.filtration.F s k).arrow = (FC.filtration.F (s + 1) k).arrow :=
 117         Subobject.ofLE_arrow (FC.filtration.mono s k)
 118       have h_rewrite' : kernel.ι p ≫ kerZ.arrow ≫ (FC.filtration.F s k).arrow ≫ FC.complex.d k (k - 1) =
 119           α' ≫ ((FC.filtration.F (s + 1) k).arrow ≫ FC.complex.d k (k - 1)) := by
 120         conv_lhs => rw [show kernel.ι p ≫ kerZ.arrow ≫ (FC.filtration.F s k).arrow ≫ FC.complex.d k (k - 1) =
 121           ((kernel.ι p ≫ kerZ.arrow) ≫ (FC.filtration.F s k).arrow) ≫ FC.complex.d k (k - 1) from by
 122           simp only [Category.assoc]]
 123         rw [← hα'_spec, show (α' ≫ ι_s) ≫ (FC.filtration.F s k).arrow = α' ≫ (ι_s ≫ (FC.filtration.F s k).arrow) from
 124           Category.assoc _ _ _, h_ι_arrow', Category.assoc]
 125       rw [h_rewrite']
 126       apply Subobject.factors_of_factors_right
 127       have h_transport' : eqToHom (show Subobject.underlying.obj (FC.filtration.F (s + 1) k) =
 128           Subobject.underlying.obj (FC.filtration.F (s + 1) ((k - 1) + 1)) by
 129           rw [show (k - 1 : ℤ) + 1 = k from by omega]) ≫
 130           ((FC.filtration.F (s + 1) ((k - 1) + 1)).arrow ≫ FC.dToK (k - 1)) =
 131           (FC.filtration.F (s + 1) k).arrow ≫ FC.complex.d k (k - 1) :=
 132         eqToHom_arrow_dToK_local FC (s + 1) k
 133       rw [← h_transport', Subobject.mk_factors_iff]
 134       exact ⟨eqToHom (by rw [show (k - 1 : ℤ) + 1 = k from by omega]) ≫
 135         factorThruImage ((FC.filtration.F (s + 1) ((k - 1) + 1)).arrow ≫ FC.dToK (k - 1)), by simp⟩
 136     have h_fac_fil' : (FC.filtration.F (s + ↑n) (k - 1)).Factors
 137         (kernel.ι p ≫ lift_n ≫ (FC.filtration.F (s + ↑n) (k - 1)).arrow) :=
 138       Subobject.factors_of_factors_right _ (Subobject.factors_comp_arrow lift_n)
 139     set I' := imgD' ⊓ FC.filtration.F (s + ↑n) (k - 1)
 140     have h_fac_I' : I'.Factors (kernel.ι p ≫ lift_n ≫ (FC.filtration.F (s + ↑n) (k - 1)).arrow) :=
 141       (Subobject.inf_factors _).mpr ⟨h_fac_imgD', h_fac_fil'⟩
 142     set δ' := I'.factorThru _ h_fac_I'
 143     have hδ'_spec : δ' ≫ I'.arrow = kernel.ι p ≫ lift_n ≫ (FC.filtration.F (s + ↑n) (k - 1)).arrow :=
 144       Subobject.factorThru_arrow _ _ _
 145     have h_δ'_ofLE : δ' ≫ Subobject.ofLE I' (FC.filtration.F (s + ↑n) (k - 1)) inf_le_right =
 146         kernel.ι p ≫ lift_n := by
 147       apply (inferInstance : Mono (FC.filtration.F (s + ↑n) (k - 1)).arrow).right_cancellation
 148       rw [Category.assoc, Subobject.ofLE_arrow, hδ'_spec]
 149       simp only [Category.assoc]
 150     have h_imgD_bnd' : imageSubobject ((FC.filtration.F (s + ↑n - ↑n + 1) ((k - 1) + 1)).arrow ≫
 151         FC.dToK (k - 1)) = imgD' := by
 152       rw [show (s : ℤ) + (↑n : ℤ) - (↑n : ℤ) + 1 = s + 1 from by omega]
 153     have h_I_bnd' : imageSubobject ((FC.filtration.F (s + ↑n - ↑n + 1) ((k - 1) + 1)).arrow ≫
 154         FC.dToK (k - 1)) ⊓ FC.filtration.F (s + ↑n) (k - 1) = I' := by
 155       rw [h_imgD_bnd']
 156     have h_bnd_fac' : B_n_t'.Factors (kernel.ι p ≫ lift_n ≫ πV') := by
 157       rw [show kernel.ι p ≫ lift_n ≫ πV' = (kernel.ι p ≫ lift_n) ≫ πV'
 158         from (Category.assoc _ _ _).symm, ← h_δ'_ofLE, Category.assoc]
 159       apply Subobject.factors_of_factors_right
 160       set I_bnd' := imageSubobject ((FC.filtration.F (s + ↑n - ↑n + 1) ((k - 1) + 1)).arrow ≫
 161         FC.dToK (k - 1)) ⊓ FC.filtration.F (s + ↑n) (k - 1)
 162       have h_I_le_Ibnd' : I' ≤ I_bnd' := le_of_eq h_I_bnd'.symm
 163       have h_ofLE_factor' : Subobject.ofLE I' (FC.filtration.F (s + ↑n) (k - 1)) inf_le_right =
 164           Subobject.ofLE I' I_bnd' h_I_le_Ibnd' ≫
 165           Subobject.ofLE I_bnd' (FC.filtration.F (s + ↑n) (k - 1)) inf_le_right := by
 166         apply (inferInstance : Mono (FC.filtration.F (s + ↑n) (k - 1)).arrow).right_cancellation
 167         simp only [Category.assoc, Subobject.ofLE_arrow]
 168       rw [h_ofLE_factor', Category.assoc]
 169       apply Subobject.factors_of_factors_right
 170       change (imageSubobject
 171         (Subobject.ofLE I_bnd' (FC.filtration.F (s + ↑n) (k - 1)) inf_le_right ≫ πV')).Factors
 172         (Subobject.ofLE I_bnd' (FC.filtration.F (s + ↑n) (k - 1)) inf_le_right ≫ πV')
 173       rw [Subobject.mk_factors_iff]
 174       exact ⟨factorThruImage _, by simp⟩
 175     -- Factor kernel.ι p ≫ to_Z_n_t through ofLE(B_n_t', Z_n_t')
 176     have h_ker_to_Z_fac' : B_n_t'.Factors (kernel.ι p ≫ to_Z_n_t ≫ Z_n_t'.arrow) := by
 177       rw [h_to_Z_comp']; exact h_bnd_fac'
 178     set γ' := B_n_t'.factorThru _ h_ker_to_Z_fac'
 179     have hγ'_spec : γ' ≫ B_n_t'.arrow = kernel.ι p ≫ to_Z_n_t ≫ Z_n_t'.arrow :=
 180       Subobject.factorThru_arrow _ _ _
 181     have h_factor_B' : kernel.ι p ≫ to_Z_n_t =
 182         γ' ≫ Subobject.ofLE B_n_t' Z_n_t' hB_le_Z' := by
 183       apply (inferInstance : Mono Z_n_t'.arrow).right_cancellation
 184       calc
 185         (kernel.ι p ≫ to_Z_n_t) ≫ Z_n_t'.arrow =
 186             kernel.ι p ≫ to_Z_n_t ≫ Z_n_t'.arrow := Category.assoc _ _ _
 187         _ = γ' ≫ B_n_t'.arrow := hγ'_spec.symm
 188         _ = (γ' ≫ Subobject.ofLE B_n_t' Z_n_t' hB_le_Z') ≫ Z_n_t'.arrow := by
 189           rw [Category.assoc, Subobject.ofLE_arrow]
 190     have h_cok' : Subobject.ofLE B_n_t' Z_n_t' hB_le_Z' ≫ FC.pageπ (s + ↑n) (k - 1) ↑n = 0 := by
 191       rw [FilteredComplex.pageπ]
 192       exact cokernel.condition _
 193     rw [hψ_def]
 194     calc kernel.ι p ≫ to_Z_n_t ≫ FC.pageπ (s + ↑n) (k - 1) ↑n
 195         = (kernel.ι p ≫ to_Z_n_t) ≫ FC.pageπ (s + ↑n) (k - 1) ↑n := (Category.assoc _ _ _).symm
 196       _ = (γ' ≫ Subobject.ofLE B_n_t' Z_n_t' hB_le_Z') ≫ FC.pageπ (s + ↑n) (k - 1) ↑n := by rw [h_factor_B']; rfl
 197       _ = γ' ≫ (Subobject.ofLE B_n_t' Z_n_t' hB_le_Z' ≫ FC.pageπ (s + ↑n) (k - 1) ↑n) := Category.assoc _ _ _
 198       _ = γ' ≫ 0 := by rw [h_cok']
 199       _ = 0 := comp_zero
 200   set h_on_Zn := Abelian.epiDesc p ψ h_ker_p_ψ with hh_on_Zn_def
 201   -- h_on_Zn : Z_n_s.underlying → target page, with p ≫ h_on_Zn = ψ
 202   -- === Step 1: Reduce LHS ===
 203   -- pageDifferential = cokernel.desc(ofLE_source, h_on_Zn, h_B_zero)
 204   -- pageπ_source ≫ pageDifferential = h_on_Zn (cokernel.π_desc)
 205   -- pageπ_source is epi, so imageSubobject(pageDifferential) = imageSubobject(h_on_Zn)
 206   -- p ≫ h_on_Zn = ψ, p is epi, so imageSubobject(h_on_Zn) = imageSubobject(ψ)
 207   -- Therefore: imageSubobject(pageDifferential) = imageSubobject(ψ)
 208   -- Step 1a: Show pageDifferential matches cokernel.desc _ h_on_Zn _
 209   -- Need to reconstruct h_B_zero from pageDifferential definition.
 210   -- Instead of tracking h_B_zero, use a calc chain:
 211   -- imageSubobject(pageDiff) = imageSubobject(pageπ_s ≫ pageDiff) [epi_comp']
 212   --   = imageSubobject(h_on_Zn) [cokernel.π_desc]
 213   --   = imageSubobject(p ≫ h_on_Zn) [epi_comp']
 214   --   = imageSubobject(ψ) [comp_epiDesc]
 215   -- Use the approach: first show imageSubobject(pageDiff) = imageSubobject(ψ) indirectly.
 216   -- Approach: Use erw to access cokernel.π_desc and Abelian.comp_epiDesc, but
 217   -- contain pollution in a have block.
 218   have h_img_pageDiff_eq_ψ : imageSubobject (FC.pageDifferential s k n) =
 219       imageSubobject ψ := by
 220     -- Step 1: imageSubobject(pageπ_s ≫ pageDiff) = imageSubobject(pageDiff)
 221     set pageπ_s := FC.pageπ s k ↑n with hpageπ_s_def
 222     haveI : Epi pageπ_s := by
 223       change Epi (cokernel.π (Subobject.ofLE
 224         (FC.boundarySubobject s k ↑n) (FC.cycleSubobject s k ↑n)
 225         (FC.B_le_Z_aux s k ↑n)))
 226       infer_instance
 227     rw [← imageSubobject_epi_comp'_local pageπ_s (FC.pageDifferential s k n)]
 228     -- Goal: imageSubobject(pageπ_s ≫ pageDiff) = imageSubobject(ψ)
 229     -- Step 2: pageπ_s ≫ pageDiff = h_on_Zn
 230     erw [cokernel.π_desc]
 231     -- Goal: imageSubobject(h_on_Zn) = imageSubobject(ψ)
 232     -- Step 3: imageSubobject(p ≫ h_on_Zn) = imageSubobject(h_on_Zn)
 233     erw [← hh_on_Zn_def]
 234     erw [← imageSubobject_epi_comp'_local
 235       (factorThruImageSubobject (kerZ.arrow ≫ πV)) _]
 236     -- Goal: imageSubobject(p ≫ h_on_Zn) = imageSubobject(ψ)
 237     rw [Abelian.comp_epiDesc]
 238   rw [h_img_pageDiff_eq_ψ]
 239   -- Now goal: imageSubobject(ψ) = imageSubobject(ofLE(B_{n+1}, Z_n) ≫ FC.pageπ (s + ↑n) (k - 1) ↑n)
 240   -- === Step 2: Prove equality via le_antisymm ===
 241   apply le_antisymm
 242   -- === (≤) direction: imageSubobject(ψ) ≤ imageSubobject(ofLE(B_{n+1}, Z_n) ≫ FC.pageπ (s + ↑n) (k - 1) ↑n) ===
 243   · -- ψ = to_Z_n_t ≫ FC.pageπ (s + ↑n) (k - 1) ↑n
 244     -- to_Z_n_t = lift_to_kerZ' ≫ factorThruImageSubobject(kerZ'.arrow ≫ πV')
 245     -- to_Z_n_t : kerZ.underlying → Z_n_t.underlying
 246     -- We show: to_Z_n_t factors through ofLE(B_{n+1}, Z_n) via a left factor.
 247     -- Specifically: to_Z_n_t = γ ≫ ofLE(B_{n+1}, Z_n) where
 248     -- γ = factorThruImageSubobject(something) ≫ factorThruImage(...) etc.
 249     -- Actually we need lift_n to factor through I_bnd, then through B_{n+1}.
 250     -- Key: lift_n ≫ (FC.filtration.F (s+↑n) (k-1)).arrow = kerZ.arrow ≫ (FC.filtration.F s k).arrow ≫ d(k)
 251     -- The d-image kerZ.arrow ≫ fil(s)(k).arrow ≫ d(k) is in imgD (via transport)
 252     -- Also lift_n itself witnesses it's in fil(s+n)(k-1).
 253     -- So lift_n ≫ fil(s+n).arrow factors through I_bnd = imgD ⊓ fil(s+n).
 254     -- Then lift_n ≫ πV' = factorThru_I ≫ oI_bnd ≫ πV'
 255     -- and oI_bnd ≫ πV' generates B_{n+1}.
 256     -- === Setup I_bnd (boundary) at (s+n, k-1) with index n+1 ===
 257     -- B_{n+1} = boundarySubobject (s+↑n) (k-1) ↑(n+1)
 258     -- imgD_bnd = imageSubobject(fil((s+↑n) - ↑(n+1) + 1)((k-1)+1).arrow ≫ dToK(k-1))
 259     -- Note: (s+↑n) - ↑(n+1) + 1 = s, so
 260     -- imgD_bnd = imageSubobject(fil(s)((k-1)+1).arrow ≫ dToK(k-1))
 261     set imgD_bnd := imageSubobject ((FC.filtration.F (s + ↑n - ↑(n + 1) + 1) ((k - 1) + 1)).arrow ≫
 262       FC.dToK (k - 1)) with himgD_bnd_def
 263     set I_bnd := imgD_bnd ⊓ FC.filtration.F (s + ↑n) (k - 1) with hI_bnd_def
 264     set oI_bnd := Subobject.ofLE I_bnd (FC.filtration.F (s + ↑n) (k - 1)) inf_le_right
 265       with hoI_bnd_def
 266     -- B_{n+1} at target = imageSubobject(oI_bnd ≫ πV')
 267     -- Z_n at target = imageSubobject(kerZ'.arrow ≫ πV')
 268     let B_n1_t := FC.boundarySubobject (s + ↑n) (k - 1) ↑(n + 1)
 269     let Z_n_t := FC.cycleSubobject (s + ↑n) (k - 1) ↑n
 270     -- lift_n ≫ fil(s+n).arrow = kerZ.arrow ≫ fil(s).arrow ≫ d(k)
 271     -- This is in imgD_bnd because:
 272     -- fil(s).arrow ≫ d(k) = eqToHom ≫ fil(s)((k-1)+1).arrow ≫ dToK(k-1)
 273     -- and (s+↑n) - ↑(n+1) + 1 = s
 274     have h_idx_eq_s : (s : ℤ) + ↑n - ↑(n + 1) + 1 = s := by push_cast; omega
 275     -- imgD_bnd = imageSubobject(fil(s)((k-1)+1).arrow ≫ dToK(k-1))
 276     have h_imgD_eq : imgD_bnd =
 277         imageSubobject ((FC.filtration.F s ((k - 1) + 1)).arrow ≫ FC.dToK (k - 1)) := by
 278       rw [himgD_bnd_def, h_idx_eq_s]
 279     -- lift_n ≫ fil(s+n).arrow is in imgD_bnd
 280     have h_fac_imgD_bnd : imgD_bnd.Factors
 281         (lift_n ≫ (FC.filtration.F (s + ↑n) (k - 1)).arrow) := by
 282       rw [h_lift_spec]
