   7 /-!
   8 # Differential on quotient pages
   9 
  10 The differential is induced by the canonical filtered chain complex, by lifting
  11 through the cycle kernel and descending through its image and the boundary
  12 quotient. This adapts the completed construction in
  13 `KIPBase/SpectralSequence/FilteredComplex.lean` at `dc4a7d1`.
  14 Finite-page construction does not require boundedness of the filtration.
  15 -/

  25 theorem fil_arrow_eqToHom (FC : FilteredComplex C)
  26     (s : ℤ) (a b : ℤ) (h : a = b) :
  27     eqToHom (show Subobject.underlying.obj (FC.filtration.F s a) =
  28       Subobject.underlying.obj (FC.filtration.F s b) from by subst h; rfl) ≫
  29     (FC.filtration.F s b).arrow = (FC.filtration.F s a).arrow ≫
  30     eqToHom (show FC.complex.X a = FC.complex.X b from by subst h; rfl) := by
  31   subst h; simp

  33 /-- Transport lemma for the differential along an integer equality:
  34     `eqToHom ≫ FC.complex.d b (b - 1) = FC.complex.d a (a - 1) ≫ eqToHom`. -/
  35 theorem d_eqToHom' (FC : FilteredComplex C)
  36     (a b : ℤ) (h : a = b) :
  37     eqToHom (show FC.complex.X a = FC.complex.X b from by subst h; rfl) ≫
  38     FC.complex.d b (b - 1) = FC.complex.d a (a - 1) ≫
  39     eqToHom (show FC.complex.X (a - 1) = FC.complex.X (b - 1) from by subst h; rfl) := by
  40   subst h; simp

  42 /-- `eqToHom ≫ dToK(k-1) = d(k)`: the transported differential composed with eqToHom
  43     gives the original differential. (Generalized version with free variables.) -/
  44 theorem eqToHom_arrow_dToK_gen (FC : FilteredComplex C)
  45     (s : ℤ) (m k : ℤ) (hmk : m + 1 = k) :
  46     eqToHom (show Subobject.underlying.obj (FC.filtration.F s k) =
  47       Subobject.underlying.obj (FC.filtration.F s (m + 1)) by rw [hmk]) ≫
  48     ((FC.filtration.F s (m + 1)).arrow ≫ FC.dToK m) =
  49       (FC.filtration.F s k).arrow ≫ FC.complex.d k (k - 1) ≫
  50       eqToHom (congr_arg FC.complex.X (show k - 1 = m by omega)) := by
  51   subst hmk
  52   simp [FilteredComplex.dToK]

  54 /-- Specialized transport: `eqToHom ≫ F^s((k-1)+1).arrow ≫ dToK(k-1) = F^s(k).arrow ≫ d(k)`. -/
  55 theorem eqToHom_arrow_dToK (FC : FilteredComplex C)
  56     (s k : ℤ) :
  57     eqToHom (show Subobject.underlying.obj (FC.filtration.F s k) =
  58       Subobject.underlying.obj (FC.filtration.F s ((k - 1) + 1)) by
  59       rw [show (k - 1 : ℤ) + 1 = k from by omega]) ≫
  60     ((FC.filtration.F s ((k - 1) + 1)).arrow ≫ FC.dToK (k - 1)) =
  61       (FC.filtration.F s k).arrow ≫ FC.complex.d k (k - 1) := by
  62   rw [FC.eqToHom_arrow_dToK_gen s (k - 1) k (by omega)]
  63   simp

  65 set_option backward.isDefEq.respectTransparency false in
  66 /-- The induced differential on pages: `d_r : E_r^{s,k} → E_r^{s+r,k-1}`.
  67     This maps Z_r-cycles modulo B_r-boundaries at (s,k) to the same at (s+r,k-1),
  68     using the original differential d of the filtered complex.
  69 
  70     For `r = n : ℕ`:
  71     - Source: `E_n^{s,k} = Z_n^{s,k} / B_n^{s,k}` where
  72       `Z_n = { x ∈ F^s | dx ∈ F^{s+n} }` and `B_n = { dz | z ∈ F^{s-n+1}, dz ∈ F^s }`
  73     - Target: `E_n^{s+n,k-1} = Z_n^{s+n,k-1} / B_n^{s+n,k-1}`
  74     - Map: send class of x to class of dx -/
  75 @[no_expose] noncomputable def pageDifferential (FC : FilteredComplex C)
  76     (s k : ℤ) (n : ℕ) :
  77     FC.pageObj s k ↑n ⟶ FC.pageObj (s + ↑n) (k - 1) ↑n := by
  78   set ι_s := Subobject.ofLE (FC.filtration.F (s + 1) k) (FC.filtration.F s k) (FC.filtration.mono s k) with hι_s_def
  79   let πV := FC.filtration.toAssociatedGraded s k
  80   let f_n := (FC.filtration.F s k).arrow ≫ FC.complex.d k (k - 1) ≫
  81     cokernel.π ((FC.filtration.F (s + ↑n) (k - 1)).arrow)
  82   let kerZ := kernelSubobject f_n
  83   set ι_t := Subobject.ofLE (FC.filtration.F (s + ↑n + 1) (k - 1)) (FC.filtration.F (s + ↑n) (k - 1))
  84     (FC.filtration.mono (s + ↑n) (k - 1)) with hι_t_def
  85   let πV' := FC.filtration.toAssociatedGraded (s + ↑n) (k - 1)
  86   let f_n' := (FC.filtration.F (s + ↑n) (k - 1)).arrow ≫ FC.complex.d (k - 1) (k - 1 - 1) ≫
  87     cokernel.π ((FC.filtration.F (s + ↑n + ↑n) (k - 1 - 1)).arrow)
  88   let kerZ' := kernelSubobject f_n'
  89   -- === Step 1: Lift d-image through F^{s+n}(k-1) ===
  90   have h_ker_fn : kerZ.arrow ≫ f_n = 0 := kernelSubobject_arrow_comp f_n
  91   have h_factor_zero : (kerZ.arrow ≫ (FC.filtration.F s k).arrow ≫ FC.complex.d k (k - 1)) ≫
  92       cokernel.π ((FC.filtration.F (s + ↑n) (k - 1)).arrow) = 0 := by
  93     simp only [Category.assoc] at h_ker_fn ⊢; exact h_ker_fn
  94   set lift_n := Abelian.monoLift (FC.filtration.F (s + ↑n) (k - 1)).arrow
  95     (kerZ.arrow ≫ (FC.filtration.F s k).arrow ≫ FC.complex.d k (k - 1))
  96     h_factor_zero with hlift_n_def
  97   have h_lift_spec : lift_n ≫ (FC.filtration.F (s + ↑n) (k - 1)).arrow =
  98       kerZ.arrow ≫ (FC.filtration.F s k).arrow ≫ FC.complex.d k (k - 1) :=
  99     Abelian.monoLift_comp _ _ _
 100   -- === Step 2: Show lift_n factors through kerZ' (using d²=0) ===
 101   have h_lift_in_kerZ' : lift_n ≫ f_n' = 0 := by
 102     calc lift_n ≫ f_n'
 103         = lift_n ≫ ((FC.filtration.F (s + ↑n) (k - 1)).arrow ≫ FC.complex.d (k - 1) (k - 1 - 1) ≫
 104             cokernel.π ((FC.filtration.F (s + ↑n + ↑n) (k - 1 - 1)).arrow)) := rfl
 105       _ = ((lift_n ≫ (FC.filtration.F (s + ↑n) (k - 1)).arrow) ≫ FC.complex.d (k - 1) (k - 1 - 1)) ≫
 106             cokernel.π ((FC.filtration.F (s + ↑n + ↑n) (k - 1 - 1)).arrow) := by
 107           simp only [Category.assoc]
 108       _ = ((kerZ.arrow ≫ (FC.filtration.F s k).arrow ≫ FC.complex.d k (k - 1)) ≫ FC.complex.d (k - 1) (k - 1 - 1)) ≫
 109             cokernel.π ((FC.filtration.F (s + ↑n + ↑n) (k - 1 - 1)).arrow) := by
 110           rw [h_lift_spec]
 111       _ = (kerZ.arrow ≫ (FC.filtration.F s k).arrow ≫ (FC.complex.d k (k - 1) ≫ FC.complex.d (k - 1) (k - 1 - 1))) ≫
 112             cokernel.π ((FC.filtration.F (s + ↑n + ↑n) (k - 1 - 1)).arrow) := by
 113           simp only [Category.assoc]
 114       _ = 0 := by rw [FC.complex.d_comp_d k]; simp only [comp_zero, zero_comp]
 115   set lift_to_kerZ' := factorThruKernelSubobject f_n' lift_n h_lift_in_kerZ'
 116     with hlift_to_kerZ'_def
 117   -- === Step 3: Build ψ : kerZ.underlying → target page ===
 118   set to_Z_n_t := lift_to_kerZ' ≫ factorThruImageSubobject (kerZ'.arrow ≫ πV')
 119     with hto_Z_n_t_def
 120   set pageπ' := FC.pageπ (s + ↑n) (k - 1) ↑n with hpageπ'_def
 121   set ψ := to_Z_n_t ≫ pageπ' with hψ_def
 122   -- === Step 4: Descend ψ through epi (kerZ → Z_n_s) using Abelian.epiDesc ===
 123   set p := factorThruImageSubobject (kerZ.arrow ≫ πV) with hp_def
 124   haveI : Epi p := inferInstance
 125   have h_ker_p_ψ : kernel.ι p ≫ ψ = 0 := by
 126     -- === Step B: lift_n spec ===
 127     have h_comp_eq : kernel.ι p ≫ lift_n ≫ (FC.filtration.F (s + ↑n) (k - 1)).arrow =
 128         kernel.ι p ≫ kerZ.arrow ≫ (FC.filtration.F s k).arrow ≫ FC.complex.d k (k - 1) := by
 129       simp only [h_lift_spec]
 130     -- === Step C: Factor through imgD ⊓ F^{s+n}(k-1) ===
 131     set imgD := imageSubobject ((FC.filtration.F (s + 1) ((k - 1) + 1)).arrow ≫ FC.dToK (k - 1))
 132       with himgD_def
 133     have h_fac_imgD : imgD.Factors
 134         (kernel.ι p ≫ lift_n ≫ (FC.filtration.F (s + ↑n) (k - 1)).arrow) := by
 135       rw [h_comp_eq]
 136       have h1 : kernel.ι p ≫ (kerZ.arrow ≫ πV) = 0 := by
 137         have h_p_Z : p ≫ (imageSubobject (kerZ.arrow ≫ πV)).arrow = kerZ.arrow ≫ πV :=
 138           imageSubobject_arrow_comp (kerZ.arrow ≫ πV)
 139         calc kernel.ι p ≫ (kerZ.arrow ≫ πV)
 140             = kernel.ι p ≫ (p ≫ (imageSubobject (kerZ.arrow ≫ πV)).arrow) := by rw [h_p_Z]
 141           _ = (kernel.ι p ≫ p) ≫ (imageSubobject (kerZ.arrow ≫ πV)).arrow := by
 142               simp only [Category.assoc]
 143           _ = 0 ≫ (imageSubobject (kerZ.arrow ≫ πV)).arrow := by rw [kernel.condition]
 144           _ = 0 := zero_comp
 145       have h1' : (kernel.ι p ≫ kerZ.arrow) ≫ cokernel.π ι_s = 0 := by
 146         rw [Category.assoc]; exact h1
 147       set α := Abelian.monoLift ι_s (kernel.ι p ≫ kerZ.arrow) h1' with hα_def
 148       have hα_spec : α ≫ ι_s = kernel.ι p ≫ kerZ.arrow := Abelian.monoLift_comp _ _ _
 149       have h_ι_arrow : ι_s ≫ (FC.filtration.F s k).arrow = (FC.filtration.F (s + 1) k).arrow := by
 150         exact Subobject.ofLE_arrow (FC.filtration.mono s k)
 151       have h_rewrite : kernel.ι p ≫ kerZ.arrow ≫ (FC.filtration.F s k).arrow ≫ FC.complex.d k (k - 1) =
 152           α ≫ ((FC.filtration.F (s + 1) k).arrow ≫ FC.complex.d k (k - 1)) := by
 153         conv_lhs => rw [show kernel.ι p ≫ kerZ.arrow ≫ (FC.filtration.F s k).arrow ≫ FC.complex.d k (k - 1) =
 154           ((kernel.ι p ≫ kerZ.arrow) ≫ (FC.filtration.F s k).arrow) ≫ FC.complex.d k (k - 1) from by
 155           simp only [Category.assoc]]
 156         rw [← hα_spec, show (α ≫ ι_s) ≫ (FC.filtration.F s k).arrow = α ≫ (ι_s ≫ (FC.filtration.F s k).arrow) from
 157           Category.assoc _ _ _, h_ι_arrow, Category.assoc]
 158       rw [h_rewrite]
 159       apply Subobject.factors_of_factors_right
 160       have h_transport : eqToHom (show Subobject.underlying.obj (FC.filtration.F (s + 1) k) =
 161           Subobject.underlying.obj (FC.filtration.F (s + 1) ((k - 1) + 1)) by
 162           rw [show (k - 1 : ℤ) + 1 = k from by omega]) ≫
 163           ((FC.filtration.F (s + 1) ((k - 1) + 1)).arrow ≫ FC.dToK (k - 1)) =
 164           (FC.filtration.F (s + 1) k).arrow ≫ FC.complex.d k (k - 1) :=
 165         FC.eqToHom_arrow_dToK (s + 1) k
 166       rw [← h_transport, himgD_def, Subobject.mk_factors_iff]
 167       exact ⟨eqToHom (by rw [show (k - 1 : ℤ) + 1 = k from by omega]) ≫
 168         factorThruImage ((FC.filtration.F (s + 1) ((k - 1) + 1)).arrow ≫ FC.dToK (k - 1)), by simp⟩
 169     have h_fac_fil : (FC.filtration.F (s + ↑n) (k - 1)).Factors
 170         (kernel.ι p ≫ lift_n ≫ (FC.filtration.F (s + ↑n) (k - 1)).arrow) :=
 171       Subobject.factors_of_factors_right _ (Subobject.factors_comp_arrow lift_n)
 172     set I := imgD ⊓ FC.filtration.F (s + ↑n) (k - 1) with hI_def
 173     have h_fac_I : I.Factors (kernel.ι p ≫ lift_n ≫ (FC.filtration.F (s + ↑n) (k - 1)).arrow) :=
 174       (Subobject.inf_factors _).mpr ⟨h_fac_imgD, h_fac_fil⟩
 175     set δ := I.factorThru _ h_fac_I with hδ_def
 176     have hδ_spec : δ ≫ I.arrow = kernel.ι p ≫ lift_n ≫ (FC.filtration.F (s + ↑n) (k - 1)).arrow :=
 177       Subobject.factorThru_arrow _ _ _
 178     -- === Step D: δ ≫ ofLE(I, F^{s+n}) = kernel.ι p ≫ lift_n ===
 179     have h_δ_ofLE : δ ≫ Subobject.ofLE I (FC.filtration.F (s + ↑n) (k - 1)) inf_le_right =
 180         kernel.ι p ≫ lift_n := by
 181       apply (inferInstance : Mono (FC.filtration.F (s + ↑n) (k - 1)).arrow).right_cancellation
 182       rw [Category.assoc, Subobject.ofLE_arrow, hδ_spec]
 183       simp only [Category.assoc]
 184     -- === Step E: Match boundary definition with I ===
 185     have h_imgD_bnd : imageSubobject ((FC.filtration.F (s + ↑n - ↑n + 1) ((k - 1) + 1)).arrow ≫
 186         FC.dToK (k - 1)) = imgD := by
 187       rw [show (s : ℤ) + (↑n : ℤ) - (↑n : ℤ) + 1 = s + 1 from by omega]
 188     have h_I_bnd : imageSubobject ((FC.filtration.F (s + ↑n - ↑n + 1) ((k - 1) + 1)).arrow ≫
 189         FC.dToK (k - 1)) ⊓ FC.filtration.F (s + ↑n) (k - 1) = I := by
 190       rw [h_imgD_bnd]

 193     -- === Step F: Show B_n_t.Factors (kernel.ι p ≫ lift_n ≫ πV') ===
 194 
 195     have h_bnd_fac : (FC.boundarySubobject (s + ↑n) (k - 1) ↑n).Factors
 196         (kernel.ι p ≫ lift_n ≫ πV') := by
 197       rw [show kernel.ι p ≫ lift_n ≫ πV' = (kernel.ι p ≫ lift_n) ≫ πV'
 198         from (Category.assoc _ _ _).symm,
 199         ← h_δ_ofLE, Category.assoc]
 200       apply Subobject.factors_of_factors_right
 201       set I_bnd := imageSubobject ((FC.filtration.F (s + ↑n - ↑n + 1) ((k - 1) + 1)).arrow ≫
 202         FC.dToK (k - 1)) ⊓ FC.filtration.F (s + ↑n) (k - 1) with hI_bnd_def
 203       have h_I_le_Ibnd : I ≤ I_bnd := le_of_eq h_I_bnd.symm
 204       have h_ofLE_factor : Subobject.ofLE I (FC.filtration.F (s + ↑n) (k - 1)) inf_le_right =
 205           Subobject.ofLE I I_bnd h_I_le_Ibnd ≫
 206           Subobject.ofLE I_bnd (FC.filtration.F (s + ↑n) (k - 1)) inf_le_right := by
 207         apply (inferInstance : Mono (FC.filtration.F (s + ↑n) (k - 1)).arrow).right_cancellation
 208         simp only [Category.assoc, Subobject.ofLE_arrow]
 209       rw [h_ofLE_factor, Category.assoc]
 210       apply Subobject.factors_of_factors_right
 211       change (imageSubobject
 212         (Subobject.ofLE I_bnd (FC.filtration.F (s + ↑n) (k - 1)) inf_le_right ≫ πV')).Factors
 213         (Subobject.ofLE I_bnd (FC.filtration.F (s + ↑n) (k - 1)) inf_le_right ≫ πV')
 214       rw [Subobject.mk_factors_iff]
 215       exact ⟨factorThruImage _, by simp⟩
 216     -- === Step G: Show kernel.ι p ≫ ψ = 0 ===
 217     set B_n_t := FC.boundarySubobject (s + ↑n) (k - 1) ↑n with hB_n_t_def
 218     let Z_n_t := FC.cycleSubobject (s + ↑n) (k - 1) ↑n
 219     have hZ_n_t_def : Z_n_t = FC.cycleSubobject (s + ↑n) (k - 1) ↑n := rfl
 220     have hB_le_Z := FC.B_le_Z_aux (s + ↑n) (k - 1) ↑n
 221     have h_to_Z_comp : to_Z_n_t ≫ Z_n_t.arrow = lift_n ≫ πV' := by
 222       change (lift_to_kerZ' ≫ factorThruImageSubobject (kerZ'.arrow ≫ πV')) ≫
 223         (imageSubobject (kerZ'.arrow ≫ πV')).arrow = lift_n ≫ πV'
 224       simp only [Category.assoc, imageSubobject_arrow_comp]
 225       rw [show lift_to_kerZ' ≫ kerZ'.arrow ≫ πV' =
 226         (lift_to_kerZ' ≫ kerZ'.arrow) ≫ πV' from (Category.assoc _ _ _).symm]
 227       rw [show lift_to_kerZ' ≫ kerZ'.arrow = lift_n from by
 228         rw [hlift_to_kerZ'_def]
 229         exact factorThruKernelSubobject_comp_arrow f_n' lift_n h_lift_in_kerZ']
 230     have h_ker_to_Z_fac : B_n_t.Factors (kernel.ι p ≫ to_Z_n_t ≫ Z_n_t.arrow) := by
 231       rw [h_to_Z_comp]
 232       exact h_bnd_fac
 233     set γ := B_n_t.factorThru _ h_ker_to_Z_fac with hγ_def
 234     have hγ_spec : γ ≫ B_n_t.arrow = kernel.ι p ≫ to_Z_n_t ≫ Z_n_t.arrow :=
 235       Subobject.factorThru_arrow _ _ _
 236     have h_factor_B : kernel.ι p ≫ to_Z_n_t =
 237         γ ≫ Subobject.ofLE B_n_t Z_n_t hB_le_Z := by
 238       apply (inferInstance : Mono Z_n_t.arrow).right_cancellation
 239       calc
 240         (kernel.ι p ≫ to_Z_n_t) ≫ Z_n_t.arrow =
 241             kernel.ι p ≫ to_Z_n_t ≫ Z_n_t.arrow := Category.assoc _ _ _
 242         _ = γ ≫ B_n_t.arrow := hγ_spec.symm
 243         _ = (γ ≫ Subobject.ofLE B_n_t Z_n_t hB_le_Z) ≫ Z_n_t.arrow := by
 244           rw [Category.assoc, Subobject.ofLE_arrow]
 245     have h_cok : Subobject.ofLE B_n_t Z_n_t hB_le_Z ≫ pageπ' = 0 := by
 246       rw [show pageπ' = FC.pageπ (s + ↑n) (k - 1) ↑n from rfl]
 247       change Subobject.ofLE B_n_t Z_n_t hB_le_Z ≫
 248         cokernel.π (Subobject.ofLE (FC.boundarySubobject (s + ↑n) (k - 1) ↑n)
 249           (FC.cycleSubobject (s + ↑n) (k - 1) ↑n)
 250           (FC.B_le_Z_aux (s + ↑n) (k - 1) ↑n)) = 0
 251       exact cokernel.condition _
 252     rw [hψ_def]
 253     calc kernel.ι p ≫ to_Z_n_t ≫ pageπ'
 254         = (kernel.ι p ≫ to_Z_n_t) ≫ pageπ' := (Category.assoc _ _ _).symm
 255       _ = (γ ≫ Subobject.ofLE B_n_t Z_n_t hB_le_Z) ≫ pageπ' := by
 256         rw [← h_factor_B]
 257       _ = γ ≫ (Subobject.ofLE B_n_t Z_n_t hB_le_Z ≫ pageπ') := Category.assoc _ _ _
 258       _ = γ ≫ 0 := by rw [h_cok]
 259       _ = 0 := comp_zero
 260   set h_on_Zn := Abelian.epiDesc p ψ h_ker_p_ψ with hh_on_Zn_def
 261   -- === Step 5: Descend through source page cokernel ===
 262   have h_B_zero : Subobject.ofLE (FC.boundarySubobject s k ↑n)
 263       (FC.cycleSubobject s k ↑n)
 264       (FC.B_le_Z_aux s k ↑n) ≫ h_on_Zn = 0 := by
 265     let imgD_s := imageSubobject ((FC.filtration.F (s - ↑n + 1) (k + 1)).arrow ≫ FC.dToK k)
 266     let I_s := imgD_s ⊓ FC.filtration.F s k
 267     let oI_s := Subobject.ofLE I_s (FC.filtration.F s k) inf_le_right
 268     have h_B_eq : FC.boundarySubobject s k ↑n = imageSubobject (oI_s ≫ πV) := by
 269       simp [FilteredComplex.boundarySubobject, imgD_s, I_s, oI_s, πV]
 270     have h_Z_eq : FC.cycleSubobject s k ↑n = imageSubobject (kerZ.arrow ≫ πV) := by
 271       simp [FilteredComplex.cycleSubobject, kerZ, f_n, πV]
 272     -- === Step 1: oI_s ≫ f_n = 0 (boundary elements satisfy the cycle condition) ===
 273     have h_zero_s : oI_s ≫ f_n = 0 := by
 274       change Subobject.ofLE I_s (FC.filtration.F s k) inf_le_right ≫
 275         ((FC.filtration.F s k).arrow ≫ FC.complex.d k (k - 1) ≫
 276           cokernel.π ((FC.filtration.F (s + ↑n) (k - 1)).arrow)) = 0
 277       simp only [← Category.assoc]
 278       rw [Subobject.ofLE_arrow]
 279       rw [show I_s.arrow =
 280           Subobject.ofLE I_s imgD_s inf_le_left ≫ imgD_s.arrow
 281         from (Subobject.ofLE_arrow inf_le_left).symm]
 282       simp only [Category.assoc]
 283       have hd : imgD_s.arrow ≫ FC.complex.d k (k - 1) = 0 := by
 284         rw [← cancel_epi (factorThruImageSubobject
 285           ((FC.filtration.F (s - ↑n + 1) (k + 1)).arrow ≫ FC.dToK k))]
 286         rw [comp_zero, ← Category.assoc, imageSubobject_arrow_comp,
 287           Category.assoc, FC.dToK_comp_d, comp_zero]
 288       rw [show imgD_s.arrow ≫ FC.complex.d k (k - 1) ≫ cokernel.π _ =
 289         (imgD_s.arrow ≫ FC.complex.d k (k - 1)) ≫ cokernel.π _
 290         from (Category.assoc _ _ _).symm,
 291         hd, zero_comp, comp_zero]
 292     -- === Step 2: Lift oI_s to kerZ ===
 293     set σ_s := factorThruKernelSubobject f_n oI_s h_zero_s with hσ_s_def
 294     have hσ_s_spec : σ_s ≫ kerZ.arrow = oI_s :=
 295       factorThruKernelSubobject_comp_arrow f_n oI_s h_zero_s
 296     -- === Step 3: σ_s ≫ p ≫ Z_n.arrow = oI_s ≫ πV ===
 297     have hp_spec : p ≫ (FC.cycleSubobject s k ↑n).arrow = kerZ.arrow ≫ πV := by
 298       change factorThruImageSubobject (kerZ.arrow ≫ πV) ≫
 299         (imageSubobject (kerZ.arrow ≫ πV)).arrow = kerZ.arrow ≫ πV
 300       exact imageSubobject_arrow_comp (kerZ.arrow ≫ πV)
 301     -- === Step 4: Cancel epi to reduce to σ_s ≫ ψ = 0 ===
 302     set q := factorThruImageSubobject (oI_s ≫ πV) ≫
 303       eqToHom (congrArg (fun X : Subobject (FC.filtration.associatedGraded s k) =>
