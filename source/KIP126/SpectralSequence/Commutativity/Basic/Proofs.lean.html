  19 /-- The ambient object of the extension spectral sequence is the associated
  20 graded object of its two-term filtered complex. -/
  21 theorem ess_ssData_V {ω : Type w} [AddCommGroup ω] [DecidableEq ω]
  22     {E₁ E₂ : SpectralSequence C ω} {ω' : Type w}
  23     {A₁ A₂ : ω' → C} {F₁ : Filtration A₁} {F₂ : Filtration A₂}
  24     {conv₁ : Convergence E₁ A₁ F₁} {conv₂ : Convergence E₂ A₂ F₂}
  25     {cm : ConvergenceMorphism conv₁ conv₂}
  26     {bnd₁ : F₁.IsBounded} {bnd₂ : F₂.IsBounded}
  27     (ext : BoundedExtensionSS conv₁ conv₂ cm bnd₁ bnd₂)
  28     (t : ω') (s k : ℤ) :
  29     (((ext.complex t).toSpectralSequence (ext.bounded t)).ssData ⟨s, k⟩).V =
  30       (ext.complex t).assocGraded s k :=
  31   rfl

  33 /-- The extension spectral sequence has differential degree `(r, -1)`. -/
  34 theorem ess_diffDeg {ω : Type w} [AddCommGroup ω] [DecidableEq ω]
  35     {E₁ E₂ : SpectralSequence C ω} {ω' : Type w}
  36     {A₁ A₂ : ω' → C} {F₁ : Filtration A₁} {F₂ : Filtration A₂}
  37     {conv₁ : Convergence E₁ A₁ F₁} {conv₂ : Convergence E₂ A₂ F₂}
  38     {cm : ConvergenceMorphism conv₁ conv₂}
  39     {bnd₁ : F₁.IsBounded} {bnd₂ : F₂.IsBounded}
  40     (ext : BoundedExtensionSS conv₁ conv₂ cm bnd₁ bnd₂)
  41     (t : ω') (r : ℤ) :
  42     ((ext.complex t).toSpectralSequence (ext.bounded t)).diffDeg r = (r, -1) :=
  43   rfl
