  13 open CategoryTheory

  15 universe u v w
  16 
  17 variable {C : Type u} [Category.{v} C] [Abelian C]
  18 
  19 /-- Transport an associated-graded target from the fourth convergence
  20 reindexing to the second one. -/
  21 noncomputable def detectionGradedProj
  22     {ω : Type w} [AddCommGroup ω] [DecidableEq ω]
  23     {E₂ E₄ : SpectralSequence C ω} {ω' : Type w}
  24     {A₂ A₄ : ω' → C} {F₂ : Filtration A₂} {F₄ : Filtration A₄}
  25     {conv₂ : Convergence E₂ A₂ F₂} {conv₄ : Convergence E₄ A₄ F₄}
  26     (cmq : ConvergenceMorphism conv₂ conv₄) (k : ω) :
  27     F₄.associatedGraded (conv₄.reindex k).1 (conv₄.reindex k).2 ⟶
  28       F₄.associatedGraded (conv₂.reindex k).1 (conv₂.reindex k).2 :=
  29   eqToHom (by rw [cmq.reindex_eq])
  30 
  31 end KIP126.Core.SpectralSequence
