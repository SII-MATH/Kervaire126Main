  16 universe u v w

  18 variable {C : Type u} [Category.{v} C] [Abelian C]
  19 
  20 /-- A commutative square in the category of convergent spectral sequences. -/
  21 structure ConvergingSSSquare {ω : Type w} [AddCommGroup ω] [DecidableEq ω]
  22     {ω' : Type w} where
  23   V₁ : ConvergingSS C ω ω'
  24   V₂ : ConvergingSS C ω ω'
  25   V₃ : ConvergingSS C ω ω'
  26   V₄ : ConvergingSS C ω ω'
  27   f : V₁ ⟶ V₂
  28   p : V₁ ⟶ V₃
  29   q : V₂ ⟶ V₄
  30   g : V₃ ⟶ V₄
  31   comm : f ≫ q = p ≫ g
  32 
  33 /-- Historical componentwise presentation of a commutative square, including
  34 bounded extension data on its four edges. -/
  35 structure HomotopyCommSquare {ω : Type w} [AddCommGroup ω] [DecidableEq ω]
  36     {E₁ E₂ E₃ E₄ : SpectralSequence C ω} {ω' : Type w}
  37     {A₁ A₂ A₃ A₄ : ω' → C}
  38     {F₁ : Filtration A₁} {F₂ : Filtration A₂}
  39     {F₃ : Filtration A₃} {F₄ : Filtration A₄}
  40     (conv₁ : Convergence E₁ A₁ F₁) (conv₂ : Convergence E₂ A₂ F₂)
  41     (conv₃ : Convergence E₃ A₃ F₃) (conv₄ : Convergence E₄ A₄ F₄) where
  42   cmf : ConvergenceMorphism conv₁ conv₂
  43   cmp : ConvergenceMorphism conv₁ conv₃
  44   cmq : ConvergenceMorphism conv₂ conv₄
  45   cmg : ConvergenceMorphism conv₃ conv₄
  46   bnd₁ : F₁.IsBounded
  47   bnd₂ : F₂.IsBounded
  48   bnd₃ : F₃.IsBounded
  49   bnd₄ : F₄.IsBounded
  50   extf : BoundedExtensionSS conv₁ conv₂ cmf bnd₁ bnd₂
  51   extp : BoundedExtensionSS conv₁ conv₃ cmp bnd₁ bnd₃
  52   extq : BoundedExtensionSS conv₂ conv₄ cmq bnd₂ bnd₄
  53   extg : BoundedExtensionSS conv₃ conv₄ cmg bnd₃ bnd₄
  54   abutment_comm : ∀ k', cmf.aMap k' ≫ cmq.aMap k' = cmp.aMap k' ≫ cmg.aMap k'
  55   eInfty_comm : ∀ k, cmf.eMap k ≫ cmq.eMap k = cmp.eMap k ≫ cmg.eMap k
  56 
  57 /-- Convert the historical componentwise square to a categorical square. -/
  58 def HomotopyCommSquare.toSquare {ω : Type w} [AddCommGroup ω] [DecidableEq ω]
  59     {E₁ E₂ E₃ E₄ : SpectralSequence C ω} {ω' : Type w}
  60     {A₁ A₂ A₃ A₄ : ω' → C}
  61     {F₁ : Filtration A₁} {F₂ : Filtration A₂}
  62     {F₃ : Filtration A₃} {F₄ : Filtration A₄}
  63     {conv₁ : Convergence E₁ A₁ F₁} {conv₂ : Convergence E₂ A₂ F₂}
  64     {conv₃ : Convergence E₃ A₃ F₃} {conv₄ : Convergence E₄ A₄ F₄}
  65     (sq : HomotopyCommSquare conv₁ conv₂ conv₃ conv₄) :
  66     ConvergingSSSquare (C := C) (ω := ω) (ω' := ω') where
  67   V₁ := ⟨E₁, A₁, F₁, conv₁⟩
  68   V₂ := ⟨E₂, A₂, F₂, conv₂⟩
  69   V₃ := ⟨E₃, A₃, F₃, conv₃⟩
  70   V₄ := ⟨E₄, A₄, F₄, conv₄⟩
  71   f := sq.cmf
  72   p := sq.cmp
  73   q := sq.cmq
  74   g := sq.cmg
  75   comm := ConvergenceMorphism.ext
  76     (funext fun k => sq.eInfty_comm k)
  77     (funext fun k' => sq.abutment_comm k')
  78 
  79 end KIP126.Core.SpectralSequence
