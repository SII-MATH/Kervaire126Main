   9 open CategoryTheory CategoryTheory.Limits
  10 universe u v
  11 variable {R : Type u} [Ring R] {M : ModuleCat.{v} R}
  12 
  13 /-- Concrete representatives for the categorical subobject of a submodule. -/
  14 noncomputable def submoduleUnderlyingIso (Z : Submodule R M) :
  15     Subobject.underlying.obj ((ModuleCat.subobjectModule M).symm Z) ≅
  16       ModuleCat.of R Z :=
  17   Subobject.underlyingIso (ModuleCat.ofHom Z.subtype)
  18 
  19 /-- The internal categorical quotient of two submodules agrees with the
  20 ordinary module quotient, with explicitly chosen representative maps. -/
  21 noncomputable def submoduleCokernelIso (B Z : Submodule R M) (h : B ≤ Z) :
  22     cokernel (Subobject.ofLE
  23       ((ModuleCat.subobjectModule M).symm B)
  24       ((ModuleCat.subobjectModule M).symm Z)
  25       ((ModuleCat.subobjectModule _).symm.monotone h)) ≅
  26         ModuleCat.of R (Z ⧸ B.comap Z.subtype) :=
  27   cokernel.mapIso _ (ModuleCat.ofHom (Submodule.inclusion h))
  28     (Subobject.underlyingIso (ModuleCat.ofHom B.subtype))
  29     (Subobject.underlyingIso (ModuleCat.ofHom Z.subtype))
  30     (submodule_ofLE_underlyingIso B Z h) ≪≫
  31   ModuleCat.cokernelIsoRangeQuotient (ModuleCat.ofHom (Submodule.inclusion h)) ≪≫
  32   (Submodule.quotEquivOfEq _ _ (by
  33     exact Submodule.range_inclusion B Z h)).toModuleIso
