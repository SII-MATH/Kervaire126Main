  13 /-- Inclusions preserve the specified concrete representatives. -/
  14 @[reassoc] theorem submoduleUnderlyingIso_inv_ofLE (B Z : Submodule R M) (h : B ≤ Z) :
  15     (submoduleUnderlyingIso B).inv ≫
  16       Subobject.ofLE ((ModuleCat.subobjectModule M).symm B)
  17         ((ModuleCat.subobjectModule M).symm Z)
  18         ((ModuleCat.subobjectModule M).symm.monotone h) =
  19     ModuleCat.ofHom (Submodule.inclusion h) ≫ (submoduleUnderlyingIso Z).inv := by
  20   apply (cancel_mono (submoduleUnderlyingIso Z).hom).mp
  21   simp only [Category.assoc, Iso.inv_hom_id, Category.comp_id]
  22   unfold submoduleUnderlyingIso
  23   rw [submodule_ofLE_underlyingIso B Z h]
  24   exact Iso.inv_hom_id_assoc _ _

  26 set_option backward.isDefEq.respectTransparency false in
  27 /-- The cokernel comparison sends each actual representative to its quotient class. -/
  28 @[reassoc] theorem submoduleCokernelIso_π (B Z : Submodule R M) (h : B ≤ Z) :
  29     (Subobject.underlyingIso (ModuleCat.ofHom Z.subtype)).inv ≫
  30       cokernel.π (Subobject.ofLE ((ModuleCat.subobjectModule M).symm B)
  31         ((ModuleCat.subobjectModule M).symm Z)
  32         ((ModuleCat.subobjectModule M).symm.monotone h)) ≫
  33       (submoduleCokernelIso B Z h).hom =
  34         ModuleCat.ofHom (B.comap Z.subtype).mkQ := by
  35   simp only [submoduleCokernelIso, Iso.trans_hom, cokernel.mapIso_hom,
  36     cokernel.map, Category.assoc, cokernel.π_desc_assoc, Iso.inv_hom_id_assoc]
  37   rw [← Category.assoc, ModuleCat.cokernel_π_cokernelIsoRangeQuotient_hom]
  38   ext x
  39   exact Submodule.quotEquivOfEq_mk _ _ (Submodule.range_inclusion B Z h) x

  41 /-- Inverse ambient maps with actual lifts preserve and reflect membership.
  42 The subobjects are the existing ones; no alternate representative object is chosen. -/
  43 theorem subobject_mem_iff_of_inverse {N : ModuleCat.{v} R}
  44     (e : M ≃ₗ[R] N) (S : Subobject M) (T : Subobject N)
  45     (hf : ∃ f : Subobject.underlying.obj S ⟶ Subobject.underlying.obj T,
  46       f ≫ T.arrow = S.arrow ≫ ModuleCat.ofHom e.toLinearMap)
  47     (hg : ∃ g : Subobject.underlying.obj T ⟶ Subobject.underlying.obj S,
  48       g ≫ S.arrow = T.arrow ≫ ModuleCat.ofHom e.symm.toLinearMap) (x : M) :
  49     e x ∈ (ModuleCat.subobjectModule N) T ↔ x ∈ (ModuleCat.subobjectModule M) S := by
  50   obtain ⟨f, hf⟩ := hf
  51   obtain ⟨g, hg⟩ := hg
  52   constructor
  53   · rintro ⟨a, ha⟩
  54     refine ⟨g a, ?_⟩
  55     have h := congrArg (fun k => k.hom a) hg
  56     change S.arrow (g a) = e.symm (T.arrow a) at h
  57     simpa only [ha, e.symm_apply_apply] using h
  58   · rintro ⟨a, ha⟩
  59     refine ⟨f a, ?_⟩
  60     have h := congrArg (fun k => k.hom a) hf
  61     change T.arrow (f a) = e (S.arrow a) at h
  62     simpa only [ha] using h

  65 /-- Descend an ambient linear equivalence through the existing cycle and
  66 boundary subobjects. This adapts KIPBase's `quotientAddEquivOfAddEquiv` to
  67 the canonical categorical quotient, and records its effect on representatives. -/
  68 theorem subobject_quotient_iso_of_linearEquiv {N : ModuleCat.{v} R}
  69     (B Z : Subobject M) (B' Z' : Subobject N) (h : B ≤ Z) (h' : B' ≤ Z')
  70     (e : M ≃ₗ[R] N)
  71     (hB : ∀ x, e x ∈ (ModuleCat.subobjectModule N) B' ↔ x ∈ (ModuleCat.subobjectModule M) B)
  72     (hZ : ∀ x, e x ∈ (ModuleCat.subobjectModule N) Z' ↔ x ∈ (ModuleCat.subobjectModule M) Z) :
  73     ∃ q : cokernel (Subobject.ofLE B Z h) ≅ cokernel (Subobject.ofLE B' Z' h'),
  74       ∀ a : (Subobject.underlying.obj Z : ModuleCat R),
  75         ∃ b : (Subobject.underlying.obj Z' : ModuleCat R),
  76           Z'.arrow b = e (Z.arrow a) ∧
  77           q.hom (cokernel.π (Subobject.ofLE B Z h) a) =
  78             cokernel.π (Subobject.ofLE B' Z' h') b := by
  79   have restrict (S : Subobject M) (T : Subobject N)
  80       (he : ∀ x, e x ∈ (ModuleCat.subobjectModule N) T ↔ x ∈ (ModuleCat.subobjectModule M) S) :
  81       ∃ v : Subobject.underlying.obj S ≅ Subobject.underlying.obj T,
  82         ∀ a, T.arrow (v.hom a) = e (S.arrow a) := by
  83     let a := LinearEquiv.ofInjective S.arrow.hom ((ModuleCat.mono_iff_injective _).mp inferInstance)
  84     let b := LinearEquiv.ofInjective T.arrow.hom ((ModuleCat.mono_iff_injective _).mp inferInstance)
  85     have hm : ((ModuleCat.subobjectModule M) S).map e.toLinearMap =
  86         (ModuleCat.subobjectModule N) T := by
  87       ext y
  88       constructor
  89       · rintro ⟨x, hx, rfl⟩
  90         exact (he x).2 hx
  91       · intro hy
  92         refine ⟨e.symm y, (he (e.symm y)).1 ?_, e.apply_symm_apply y⟩
  93         simpa only [e.apply_symm_apply] using hy
  94     let c := e.ofSubmodules ((ModuleCat.subobjectModule M) S) ((ModuleCat.subobjectModule N) T) hm
  95     refine ⟨((a.trans c).trans b.symm).toModuleIso, ?_⟩
  96     intro x
  97     exact congrArg Subtype.val (b.apply_symm_apply (c (a x)))
  98   obtain ⟨b, hb⟩ := restrict B B' hB
  99   obtain ⟨z, hz⟩ := restrict Z Z' hZ
 100   have comm : Subobject.ofLE B Z h ≫ z.hom = b.hom ≫ Subobject.ofLE B' Z' h' := by
 101     apply (cancel_mono Z'.arrow).1
 102     apply ModuleCat.hom_ext
 103     apply LinearMap.ext
 104     intro x
 105     change Z'.arrow (z.hom (Subobject.ofLE B Z h x)) =
 106       Z'.arrow (Subobject.ofLE B' Z' h' (b.hom x))
 107     rw [hz]
 108     have hs := congrArg (fun f => f.hom x) (Subobject.ofLE_arrow h)
 109     have ht := congrArg (fun f => f.hom (b.hom x)) (Subobject.ofLE_arrow h')
 110     change Z.arrow (Subobject.ofLE B Z h x) = B.arrow x at hs
 111     change Z'.arrow (Subobject.ofLE B' Z' h' (b.hom x)) = B'.arrow (b.hom x) at ht
 112     rw [hs, ht, hb]
 113   let q := cokernel.mapIso (Subobject.ofLE B Z h) (Subobject.ofLE B' Z' h') b z comm
 114   refine ⟨q, fun a => ⟨z.hom a, hz a, ?_⟩⟩
 115   have hq : cokernel.π (Subobject.ofLE B Z h) ≫ q.hom =
 116       z.hom ≫ cokernel.π (Subobject.ofLE B' Z' h') := by
 117     simp only [q, cokernel.mapIso_hom, cokernel.map, cokernel.π_desc]
 118   exact congrArg (fun f => f.hom a) hq
