  20 /-- A spectral sequence degenerates at `N` when all differentials from page `N` vanish. -/
  21 public def DegeneratesAt
  22     {ι : Type w} [AddCommGroup ι] [DecidableEq ι]
  23     (E : SpectralSequence C ι) (N : ℤ) : Prop :=
  24   ∀ (r : ℤ), N ≤ r → ∀ (k : ι), E.d r k = 0
