  15 /-!
  16 # Nested-subobject spectral-sequence data
  17 
  18 This is the axiom-free KIP126 migration of the data declarations in
  19 `KIPBase/SpectralSequence/Basic.lean`.  It deliberately does not import the
  20 historical `KIPBase` library or Mathlib's spectral-sequence structure.
  21 
  22 At each grading, `SSData` records the classical nested chain
  23 
  24 `V ⩾ Z₀ ⩾ Z₁ ⩾ ⋯ ⩾ Z∞ ⩾ B∞ ⩾ ⋯ ⩾ B₁ ⩾ B₀`,
  25 
  26 and defines `E_r = Z_r / B_r`.  This file contains data and operations only;
  27 predicates and named proofs live in the adjacent layers.
  28 -/

  36 /-- Nested cycle and boundary subobjects at one grading of a spectral sequence. -/
  37 public structure SSData (C : Type u) [Category.{v} C] [Abelian C] where
  38   /-- Ambient object. -/
  39   V : C
  40   /-- The decreasing family of cycle subobjects. -/
  41   Z : WithTop ℕ → Subobject V
  42   /-- The increasing family of boundary subobjects. -/
  43   B : WithTop ℕ → Subobject V
  44   /-- Cycles decrease with the page. -/
  45   Z_anti : Antitone Z
  46   /-- Boundaries increase with the page. -/
  47   B_mono : Monotone B
  48   /-- Every element is a zero-cycle. -/
  49   Z_zero : Z 0 = ⊤
  50   /-- Boundaries are cycles on every page. -/
  51   B_le_Z : ∀ r, B r ≤ Z r
  52   /-- `Z ⊤` is the infimum of the finite cycle stages. -/
  53   Z_top_greatest : ∀ (X : Subobject V), (∀ i : ℕ, X ≤ Z ↑i) → X ≤ Z ⊤
  54   /-- `B ⊤` is the supremum of the finite boundary stages. -/
  55   B_top_least : ∀ (X : Subobject V), (∀ i : ℕ, B ↑i ≤ X) → B ⊤ ≤ X
  56 
  57 variable {C : Type u} [Category.{v} C] [Abelian C]
  58 
  59 /-- The page `E_r = Z_r / B_r`. -/
  60 public noncomputable def SSData.page (D : SSData C) (r : WithTop ℕ) : C :=
  61   cokernel (Subobject.ofLE (D.B r) (D.Z r) (D.B_le_Z r))
  62 
  63 /-- The infinity page `Z_∞ / B_∞`. -/
  64 public noncomputable def SSData.eInfty (D : SSData C) : C := D.page ⊤
  65 
  66 /-- The quotient projection `Z_r ⟶ E_r`. -/
  67 public noncomputable def SSData.pageπ (D : SSData C) (r : WithTop ℕ) :
  68     Subobject.underlying.obj (D.Z r) ⟶ D.page r :=
  69   cokernel.π (Subobject.ofLE (D.B r) (D.Z r) (D.B_le_Z r))

  71 /-- A graded complex in an abelian category with a fixed differential degree. -/
  72 public structure GradedComplex (C : Type u) [Category.{v} C] [Abelian C]
  73     (ι : Type w) [AddCommGroup ι] where
  74   /-- Graded object. -/
  75   obj : ι → C
  76   /-- Degree of the differential. -/
  77   d_deg : ι
  78   /-- Differential. -/
  79   d : (k : ι) → obj k ⟶ obj (k + d_deg)
  80   /-- The differential squares to zero. -/
  81   d_sq : ∀ k, d k ≫ d (k + d_deg) = 0
  82 
  83 /-- The short complex centered at grading `k`. -/
  84 public noncomputable def GradedComplex.shortComplex
  85     {ι : Type w} [AddCommGroup ι]
  86     (G : GradedComplex C ι) (k : ι) : ShortComplex C :=
  87   { X₁ := G.obj (k - G.d_deg)
  88     X₂ := G.obj k
  89     X₃ := G.obj (k + G.d_deg)
  90     f := G.d (k - G.d_deg) ≫ eqToHom (show G.obj (k - G.d_deg + G.d_deg) = G.obj k by
  91       congr 1
  92       abel)
  93     g := G.d k
  94     zero := by
  95       simp only [Category.assoc]
  96       have key : eqToHom (show G.obj (k - G.d_deg + G.d_deg) = G.obj k by
  97           congr 1
  98           abel) ≫ G.d k =
  99         G.d (k - G.d_deg + G.d_deg) ≫ eqToHom
 100           (show G.obj (k - G.d_deg + G.d_deg + G.d_deg) = G.obj (k + G.d_deg) by
 101             congr 1
 102             abel) := by
 103         rw [eqToHom_comp_iff]
 104         simp
 105       rw [key, ← Category.assoc, G.d_sq, zero_comp] }
 106 
 107 /-- Homology of a graded complex at grading `k`. -/
 108 public noncomputable def GradedComplex.homology
 109     {ι : Type w} [AddCommGroup ι]
 110     (G : GradedComplex C ι) (k : ι) : C :=
 111   (G.shortComplex k).homology

 113 /-- The data-only part of a spectral sequence. -/
 114 public structure PreSS (C : Type u) [Category.{v} C] [Abelian C]
 115     (ι : Type w) [AddCommGroup ι] [DecidableEq ι] where
 116   /-- First displayed page. -/
 117   r₀ : ℤ
 118   /-- Nested-subobject data at each grading. -/
 119   ssData : ι → SSData C
 120   /-- Degree of the differential on each page. -/
 121   diffDeg : ℤ → ι
 122   /-- Page differential. -/
 123   d : (r : ℤ) → (k : ι) →
 124     ((ssData k).page ↑(r - r₀).toNat ⟶
 125       (ssData (k + diffDeg r)).page ↑(r - r₀).toNat)
 126 
 127 /-- The page object derived from a `PreSS`. -/
 128 @[reducible] public noncomputable def PreSS.Page
 129     {ι : Type w} [AddCommGroup ι] [DecidableEq ι]
 130     (E : PreSS C ι) (r : ℤ) (k : ι) : C :=
 131   (E.ssData k).page ↑(r - E.r₀).toNat

 133 /-- A bare family of maps between the ambient objects of two `SSData` families. -/
 134 public structure UnderlyingMorphism
 135     (ι : Type w) [AddCommGroup ι] [DecidableEq ι]
 136     (D D' : ι → SSData C) where
 137   /-- Map on each ambient object. -/
 138   φ : ∀ (k : ι), (D k).V ⟶ (D' k).V
 139 
 140 /-- A morphism preserving the cycle and boundary towers. -/
 141 public structure SSDataMorphism
 142     (ι : Type w) [AddCommGroup ι] [DecidableEq ι]
 143     (D D' : ι → SSData C) extends UnderlyingMorphism ι D D' where
 144   /-- The ambient maps preserve cycles. -/
 145   preserves_Z : ∀ (k : ι) (r : WithTop ℕ),
 146     ∃ (lift : Subobject.underlying.obj ((D k).Z r) ⟶
 147       Subobject.underlying.obj ((D' k).Z r)),
 148       lift ≫ ((D' k).Z r).arrow = ((D k).Z r).arrow ≫ φ k
 149   /-- The ambient maps preserve boundaries. -/
 150   preserves_B : ∀ (k : ι) (r : WithTop ℕ),
 151     ∃ (lift : Subobject.underlying.obj ((D k).B r) ⟶
 152       Subobject.underlying.obj ((D' k).B r)),
 153       lift ≫ ((D' k).B r).arrow = ((D k).B r).arrow ≫ φ k
 154 
 155 /-- Forget the preservation witnesses of an `SSDataMorphism`. -/
 156 def SSDataMorphism.toUnderlying
 157     {ι : Type w} [AddCommGroup ι] [DecidableEq ι]
 158     {D D' : ι → SSData C} (f : SSDataMorphism ι D D') :
 159     UnderlyingMorphism ι D D' where
 160   φ := f.φ
 161 
 162 /-- Identity on the ambient objects and their cycle and boundary towers. -/
 163 public def SSDataMorphism.id
 164     {ι : Type w} [AddCommGroup ι] [DecidableEq ι] (D : ι → SSData C) :
 165     SSDataMorphism ι D D where
 166   φ := fun _ => 𝟙 _
 167   preserves_Z := fun _ _ => ⟨𝟙 _, by simp⟩
 168   preserves_B := fun _ _ => ⟨𝟙 _, by simp⟩
 169 
 170 /-- Composition of ambient maps preserving both towers. -/
 171 public def SSDataMorphism.comp
 172     {ι : Type w} [AddCommGroup ι] [DecidableEq ι]
 173     {D D' D'' : ι → SSData C}
 174     (f : SSDataMorphism ι D D') (g : SSDataMorphism ι D' D'') :
 175     SSDataMorphism ι D D'' where
 176   φ := fun k => f.φ k ≫ g.φ k
 177   preserves_Z := fun k r =>
 178     ⟨(f.preserves_Z k r).choose ≫ (g.preserves_Z k r).choose, by
 179       rw [Category.assoc, (g.preserves_Z k r).choose_spec,
 180         ← Category.assoc, (f.preserves_Z k r).choose_spec, Category.assoc]⟩
 181   preserves_B := fun k r =>
 182     ⟨(f.preserves_B k r).choose ≫ (g.preserves_B k r).choose, by
 183       rw [Category.assoc, (g.preserves_B k r).choose_spec,
 184         ← Category.assoc, (f.preserves_B k r).choose_spec, Category.assoc]⟩
 185 
 186 /-- The restriction of the ambient map to cycles.  The subobject arrow is mono,
 187 so the preservation witness determines this map uniquely. -/
 188 public noncomputable def SSDataMorphism.cycleMap
 189     {ι : Type w} [AddCommGroup ι] [DecidableEq ι]
 190     {D D' : ι → SSData C} (f : SSDataMorphism ι D D')
 191     (k : ι) (r : WithTop ℕ) :
 192     Subobject.underlying.obj ((D k).Z r) ⟶ Subobject.underlying.obj ((D' k).Z r) :=
 193   (f.preserves_Z k r).choose
 194 
 195 /-- The canonical quotient map induced by the ambient map on `Z_r / B_r`.
 196 This includes the infinity page and makes no independent choice of page maps. -/
 197 public noncomputable def SSDataMorphism.pageMap
 198     {ι : Type w} [AddCommGroup ι] [DecidableEq ι]
 199     {D D' : ι → SSData C} (f : SSDataMorphism ι D D')
 200     (k : ι) (r : WithTop ℕ) : (D k).page r ⟶ (D' k).page r :=
 201   cokernel.map
 202     (Subobject.ofLE ((D k).B r) ((D k).Z r) ((D k).B_le_Z r))
 203     (Subobject.ofLE ((D' k).B r) ((D' k).Z r) ((D' k).B_le_Z r))
 204     (f.preserves_B k r).choose (f.cycleMap k r)
 205     (by
 206       apply (cancel_mono ((D' k).Z r).arrow).mp
 207       simp only [Category.assoc, SSDataMorphism.cycleMap,
 208         (f.preserves_Z k r).choose_spec, Subobject.ofLE_arrow,
 209         (f.preserves_B k r).choose_spec, Subobject.ofLE_arrow_assoc])
 210 
 211 /-- The same quotient map with the displayed page numbering transported along
 212 equality of starting pages. -/
 213 public noncomputable def SSDataMorphism.pageMapAt
 214     {ι : Type w} [AddCommGroup ι] [DecidableEq ι]
 215     {E E' : PreSS C ι} (f : SSDataMorphism ι E.ssData E'.ssData)
 216     (h : E.r₀ = E'.r₀) (r : ℤ) (k : ι) : E.Page r k ⟶ E'.Page r k :=
 217   f.pageMap k ↑(r - E.r₀).toNat ≫ eqToHom (by rw [h])
 218 
 219 /-- A morphism of pre-spectral sequences. -/
 220 public structure PreSSMorphism
 221     {ι : Type w} [AddCommGroup ι] [DecidableEq ι]
 222     (E E' : PreSS C ι) extends SSDataMorphism ι E.ssData E'.ssData where
 223   /-- Ordinary morphisms use the same displayed page numbering. -/
 224   r₀_eq : E.r₀ = E'.r₀
 225   /-- Ordinary morphisms preserve the differential degree on each page. -/
 226   diffDeg_eq : E.diffDeg = E'.diffDeg
 227   /-- The induced maps on pages commute with the differentials. -/
 228   comm_d : ∀ (r : ℤ) (k : ι),
 229     toSSDataMorphism.pageMapAt r₀_eq r k ≫ E'.d r k =
 230       E.d r k ≫ toSSDataMorphism.pageMapAt r₀_eq r (k + E.diffDeg r) ≫
 231         eqToHom (by rw [diffDeg_eq])
 232 
 233 /-- Forget differential compatibility from a `PreSSMorphism`. -/
 234 public def PreSSMorphism.ssDataMorphism
 235     {ι : Type w} [AddCommGroup ι] [DecidableEq ι]
 236     {E E' : PreSS C ι} (f : PreSSMorphism E E') :
 237     SSDataMorphism ι E.ssData E'.ssData where
 238   φ := f.φ
 239   preserves_Z := f.preserves_Z
 240   preserves_B := f.preserves_B
 241 
 242 /-- The canonical page map of a pre-spectral-sequence morphism. -/
 243 public noncomputable def PreSSMorphism.pageMap
 244     {ι : Type w} [AddCommGroup ι] [DecidableEq ι]
 245     {E E' : PreSS C ι} (f : PreSSMorphism E E') (r : ℤ) (k : ι) :
 246     E.Page r k ⟶ E'.Page r k :=
 247   f.toSSDataMorphism.pageMapAt f.r₀_eq r k

 257 /-- A nested-subobject spectral sequence. -/
 258 public structure SpectralSequence (C : Type u) [Category.{v} C] [Abelian C]
 259     (ι : Type w) [AddCommGroup ι] [DecidableEq ι]
 260     extends SpectralSequence.PreSS C ι where
 261   /-- Every page differential squares to zero. -/
 262   d_comp_d : ∀ (r : ℤ) (k : ι), d r k ≫ d r (k + diffDeg r) = 0
 263   /-- The kernel of `d_r` is the image of `Z_(r+1)` on page `r`. -/
 264   Z_succ : ∀ (r : ℤ) (k : ι) (_ : r₀ ≤ r),
 265     let n := (r - r₀).toNat
 266     kernelSubobject (d r k) =
 267       imageSubobject (Subobject.ofLE
 268         ((ssData k).Z ↑(n + 1)) ((ssData k).Z ↑n)
 269         ((ssData k).Z_anti (by exact_mod_cast Nat.le_succ n)) ≫
 270         (ssData k).pageπ ↑n)
 271   /-- The image of `d_r` is the image of `B_(r+1)` on page `r`. -/
 272   B_succ : ∀ (r : ℤ) (k : ι) (_ : r₀ ≤ r),
 273     let n := (r - r₀).toNat
 274     imageSubobject (d r k) =
 275       imageSubobject (Subobject.ofLE
 276         ((ssData (k + diffDeg r)).B ↑(n + 1))
 277         ((ssData (k + diffDeg r)).Z ↑n)
 278         (le_trans ((ssData (k + diffDeg r)).B_le_Z ↑(n + 1))
 279           ((ssData (k + diffDeg r)).Z_anti (by exact_mod_cast Nat.le_succ n))) ≫
 280         (ssData (k + diffDeg r)).pageπ ↑n)
 281 
 282 end KIP126.Core

 292 /-- Assemble a spectral sequence from its data and explicit compatibility proofs. -/
 293 noncomputable def ofPreSS
 294     {ι : Type w} [AddCommGroup ι] [DecidableEq ι]
 295     (P : PreSS C ι)
 296     (d_comp_d : ∀ (r : ℤ) (k : ι),
 297       P.d r k ≫ P.d r (k + P.diffDeg r) = 0)
 298     (Z_succ : ∀ (r : ℤ) (k : ι) (_ : P.r₀ ≤ r),
 299       let n := (r - P.r₀).toNat
 300       kernelSubobject (P.d r k) =
 301         imageSubobject (Subobject.ofLE
 302           ((P.ssData k).Z ↑(n + 1)) ((P.ssData k).Z ↑n)
 303           ((P.ssData k).Z_anti (by exact_mod_cast Nat.le_succ n)) ≫
 304           (P.ssData k).pageπ ↑n))
 305     (B_succ : ∀ (r : ℤ) (k : ι) (_ : P.r₀ ≤ r),
 306       let n := (r - P.r₀).toNat
 307       imageSubobject (P.d r k) =
 308         imageSubobject (Subobject.ofLE
 309           ((P.ssData (k + P.diffDeg r)).B ↑(n + 1))
 310           ((P.ssData (k + P.diffDeg r)).Z ↑n)
 311           (le_trans ((P.ssData (k + P.diffDeg r)).B_le_Z ↑(n + 1))
 312             ((P.ssData (k + P.diffDeg r)).Z_anti (by exact_mod_cast Nat.le_succ n))) ≫
 313           (P.ssData (k + P.diffDeg r)).pageπ ↑n)) :
 314     SpectralSequence C ι where
 315   toPreSS := P
 316   d_comp_d := d_comp_d
 317   Z_succ := Z_succ
 318   B_succ := B_succ

 332 /-- The canonical quotient map `R/P ⟶ R/Q` for `P ≤ Q ≤ R`. -/
 333 noncomputable def Subobject.cokernelDesc_ofLE {V : C}
 334     (P Q R : Subobject V) (hPQ : P ≤ Q) (hQR : Q ≤ R)
 335     (hPR : P ≤ R := le_trans hPQ hQR) :
 336     cokernel (Subobject.ofLE P R hPR) ⟶ cokernel (Subobject.ofLE Q R hQR) :=
 337   cokernel.desc _ (cokernel.π (Subobject.ofLE Q R hQR)) (by
 338     have : Subobject.ofLE P R hPR =
 339         Subobject.ofLE P Q hPQ ≫ Subobject.ofLE Q R hQR :=
 340       (Subobject.ofLE_comp_ofLE P Q R hPQ hQR).symm
 341     rw [this, Category.assoc, cokernel.condition, comp_zero])
 342 
 343 /-- The canonical map `Q/P ⟶ R/P` for `P ≤ Q ≤ R`. -/
 344 noncomputable def Subobject.cokernelMap_ofLE {V : C}
 345     (P Q R : Subobject V) (hPQ : P ≤ Q) (hQR : Q ≤ R)
 346     (hPR : P ≤ R := le_trans hPQ hQR) :
 347     cokernel (Subobject.ofLE P Q hPQ) ⟶ cokernel (Subobject.ofLE P R hPR) :=
 348   cokernel.desc _
 349     (Subobject.ofLE Q R hQR ≫ cokernel.π (Subobject.ofLE P R hPR)) (by
 350       rw [← Category.assoc, Subobject.ofLE_comp_ofLE, cokernel.condition])
 351 
 352 /-- The third-isomorphism identification `(R/P)/(Q/P) ≅ R/Q`. -/
 353 noncomputable def Subobject.thirdIso {V : C}
 354     (P Q R : Subobject V) (hPQ : P ≤ Q) (hQR : Q ≤ R)
 355     (hPR : P ≤ R := le_trans hPQ hQR) :
 356     cokernel (Subobject.cokernelMap_ofLE P Q R hPQ hQR hPR) ≅
 357       cokernel (Subobject.ofLE Q R hQR) :=
 358   { hom := cokernel.desc _ (Subobject.cokernelDesc_ofLE P Q R hPQ hQR hPR) (by
 359       ext
 360       simp only [Subobject.cokernelMap_ofLE, Subobject.cokernelDesc_ofLE,
 361         cokernel.π_desc_assoc, comp_zero]
 362       rw [Category.assoc, cokernel.π_desc, cokernel.condition])
 363     inv := cokernel.desc _ (cokernel.π (Subobject.ofLE P R hPR) ≫
 364         cokernel.π (Subobject.cokernelMap_ofLE P Q R hPQ hQR hPR)) (by
 365       set f := Subobject.cokernelMap_ofLE P Q R hPQ hQR hPR
 366       have h1 : cokernel.π (Subobject.ofLE P Q hPQ) ≫ f =
 367           Subobject.ofLE Q R hQR ≫ cokernel.π (Subobject.ofLE P R hPR) :=
 368         cokernel.π_desc _ _ _
 369       calc
 370         Subobject.ofLE Q R hQR ≫ cokernel.π (Subobject.ofLE P R hPR) ≫ cokernel.π f =
 371             (Subobject.ofLE Q R hQR ≫ cokernel.π (Subobject.ofLE P R hPR)) ≫
 372               cokernel.π f := by rw [Category.assoc]
 373         _ = (cokernel.π (Subobject.ofLE P Q hPQ) ≫ f) ≫ cokernel.π f := by rw [h1]
 374         _ = cokernel.π (Subobject.ofLE P Q hPQ) ≫ (f ≫ cokernel.π f) := by
 375           rw [Category.assoc]
 376         _ = cokernel.π (Subobject.ofLE P Q hPQ) ≫ 0 := by rw [cokernel.condition]
 377         _ = 0 := comp_zero)
 378     hom_inv_id := by
 379       ext
 380       simp only [Category.comp_id, cokernel.π_desc,
 381         Subobject.cokernelDesc_ofLE, cokernel.π_desc_assoc]
 382     inv_hom_id := by
 383       ext
 384       simp only [Category.comp_id, Category.assoc, cokernel.π_desc_assoc,
 385         Subobject.cokernelDesc_ofLE, cokernel.π_desc] }

 393 /-- A morphism of spectral sequences.  Its page maps are derived from `φ`,
 394 not supplied independently. -/
 395 public structure SpectralSequenceMorphism
 396     {ι : Type w} [AddCommGroup ι] [DecidableEq ι]
 397     (E E' : SpectralSequence C ι) where
 398   /-- Map on each ambient object. -/
 399   φ : ∀ (k : ι), (E.ssData k).V ⟶ (E'.ssData k).V
 400   /-- The ambient maps preserve cycles. -/
 401   preserves_Z : ∀ (k : ι) (r : WithTop ℕ),
 402     ∃ (lift : Subobject.underlying.obj ((E.ssData k).Z r) ⟶
 403       Subobject.underlying.obj ((E'.ssData k).Z r)),
 404       lift ≫ ((E'.ssData k).Z r).arrow = ((E.ssData k).Z r).arrow ≫ φ k
 405   /-- The ambient maps preserve boundaries. -/
 406   preserves_B : ∀ (k : ι) (r : WithTop ℕ),
 407     ∃ (lift : Subobject.underlying.obj ((E.ssData k).B r) ⟶
 408       Subobject.underlying.obj ((E'.ssData k).B r)),
 409       lift ≫ ((E'.ssData k).B r).arrow = ((E.ssData k).B r).arrow ≫ φ k
 410   /-- Ordinary morphisms have the same displayed page numbering. -/
 411   r₀_eq : E.r₀ = E'.r₀
 412   /-- Ordinary morphisms have the same differential degrees. -/
 413   diffDeg_eq : E.diffDeg = E'.diffDeg
 414   /-- The canonical quotient maps commute with the differentials. -/
 415   comm_d : ∀ (r : ℤ) (k : ι),
 416     let f : SSDataMorphism ι E.ssData E'.ssData :=
 417       { φ := φ, preserves_Z := preserves_Z, preserves_B := preserves_B }
 418     f.pageMapAt (E := E.toPreSS) (E' := E'.toPreSS) r₀_eq r k ≫ E'.d r k =
 419       E.d r k ≫ f.pageMapAt (E := E.toPreSS) (E' := E'.toPreSS)
 420         r₀_eq r (k + E.diffDeg r) ≫ eqToHom (by rw [diffDeg_eq])

 429   preserves_B := f.preserves_B

 431 /-- The same morphism between the underlying pre-spectral sequences. -/
 432 public def SpectralSequenceMorphism.toPreSSMorphism
 433     {ι : Type w} [AddCommGroup ι] [DecidableEq ι]
 434     {E E' : SpectralSequence C ι} (f : SpectralSequenceMorphism E E') :
 435     PreSSMorphism E.toPreSS E'.toPreSS where
 436   toSSDataMorphism := f.toSSDataMorphism
 437   r₀_eq := f.r₀_eq
 438   diffDeg_eq := f.diffDeg_eq
 439   comm_d := f.comm_d

 441 /-- The canonical map on each displayed page. -/
 442 public noncomputable def SpectralSequenceMorphism.pageMap
 443     {ι : Type w} [AddCommGroup ι] [DecidableEq ι]
 444     {E E' : SpectralSequence C ι} (f : SpectralSequenceMorphism E E')
 445     (r : ℤ) (k : ι) : E.Page r k ⟶ E'.Page r k :=
 446   f.toPreSSMorphism.pageMap r k
 447 
 448 /-- The map induced by a spectral-sequence morphism on the infinity page. -/
 449 public noncomputable def SpectralSequenceMorphism.eInftyMap
 450     {ι : Type w} [AddCommGroup ι] [DecidableEq ι]
 451     {E E' : SpectralSequence C ι} (f : SpectralSequenceMorphism E E') (k : ι) :
 452     (E.ssData k).eInfty ⟶ (E'.ssData k).eInfty :=
 453   f.toSSDataMorphism.pageMap k ⊤

 455 /-- A spectral sequence packaged with a convenient infinity-page accessor. -/
 456 structure EInftyData (C : Type u) [Category.{v} C] [Abelian C]
 457     (ι : Type w) [AddCommGroup ι] [DecidableEq ι] where

 462 noncomputable def EInftyData.EInfty

 465     (eData : EInftyData C ι) (k : ι) : C :=
