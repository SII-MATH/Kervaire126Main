  23 /-- Bottom-page boundaries lie in the cycles on every page. -/
  24 public theorem SSData.B_bot_le_Z (D : SSData C) (r : WithTop ℕ) : D.B ⊥ ≤ D.Z r :=
  25   le_trans (D.B_mono bot_le) (D.B_le_Z r)

  27 /-- If cycles and boundaries agree, the corresponding quotient page is zero. -/
  28 public theorem SSData.page_isZero_of_eq
  29     (D : SSData C) (r : WithTop ℕ) (h : D.B r = D.Z r) :
  30     IsZero (D.page r) := by
  31   unfold SSData.page
  32   have : IsIso (Subobject.ofLE (D.B r) (D.Z r) (D.B_le_Z r)) := by
  33     rw [← Subobject.isoOfEq_hom _ _ h]
  34     infer_instance
  35   exact isZero_cokernel_of_epi _

  37 /-- If a quotient page is zero, its cycles and boundaries agree. -/
  38 public theorem SSData.eq_of_page_isZero
  39     (D : SSData C) (r : WithTop ℕ) (h : IsZero (D.page r)) :
  40     D.B r = D.Z r := by
  41   unfold SSData.page at h
  42   have hepi : Epi (Subobject.ofLE (D.B r) (D.Z r) (D.B_le_Z r)) := by
  43     rwa [Preadditive.epi_iff_isZero_cokernel]
  44   haveI : IsIso (Subobject.ofLE (D.B r) (D.Z r) (D.B_le_Z r)) :=
  45     isIso_of_mono_of_epi _
  46   apply le_antisymm (D.B_le_Z r)
  47   exact Subobject.le_of_comm
  48     (inv (Subobject.ofLE (D.B r) (D.Z r) (D.B_le_Z r)))
  49     (by simp [Subobject.ofLE_arrow])

  51 /-- A spectral sequence with zero pages at one grading has zero infinity page there. -/
  52 public theorem eInfty_isZero_of_page_isZero
  53     {C : Type*} [Category C] [Abelian C]
  54     {α : Type*} [AddCommGroup α] [DecidableEq α]
  55     (E : SpectralSequence C α) (k : α)
  56     (h : ∀ r : ℤ, E.r₀ ≤ r → IsZero (E.Page r k)) :
  57     IsZero ((E.ssData k).eInfty) := by
  58   unfold SSData.eInfty
  59   apply SSData.page_isZero_of_eq
  60   set D := E.ssData k
  61   apply le_antisymm
  62   · exact D.B_le_Z ⊤
  63   · have hpage0 : IsZero (D.page (↑(0 : ℕ))) := by
  64       have hfirst := h E.r₀ (le_refl _)
  65       simp only [SpectralSequence.Page, SSData.page] at hfirst
  66       have hrr : (E.r₀ - E.r₀).toNat = 0 := by omega
  67       rw [hrr] at hfirst
  68       exact hfirst
  69     have hBZ : D.B ↑(0 : ℕ) = D.Z ↑(0 : ℕ) :=
  70       D.eq_of_page_isZero _ hpage0
  71     calc
  72       D.Z ⊤ ≤ D.Z ↑(0 : ℕ) := D.Z_anti le_top
  73       _ = D.B ↑(0 : ℕ) := hBZ.symm
  74       _ ≤ D.B ⊤ := D.B_mono le_top

 256 /-- The composite of the canonical quotient maps commutes with differentials. -/
 257 public theorem PreSSMorphism.comp_comm_d
 258     {ι : Type w} [AddCommGroup ι] [DecidableEq ι]
 259     {E E' E'' : PreSS C ι} (f : PreSSMorphism E E') (g : PreSSMorphism E' E'')
 260     (r : ℤ) (k : ι) :
 261     (f.toSSDataMorphism.comp g.toSSDataMorphism).pageMapAt
 262         (f.r₀_eq.trans g.r₀_eq) r k ≫ E''.d r k =
 263       E.d r k ≫ (f.toSSDataMorphism.comp g.toSSDataMorphism).pageMapAt
 264         (f.r₀_eq.trans g.r₀_eq) r (k + E.diffDeg r) ≫
 265           eqToHom (by rw [f.diffDeg_eq, g.diffDeg_eq]) := by
 266   rcases E with ⟨r₀, D, δ, d⟩
 267   rcases E' with ⟨r₀', D', δ', d'⟩
 268   rcases E'' with ⟨r₀'', D'', δ'', d''⟩
 269   rcases f with ⟨f, hf, hδf, hdf⟩
 270   rcases g with ⟨g, hg, hδg, hdg⟩
 271   dsimp at hf hg hδf hδg
 272   subst r₀'
 273   subst r₀''
 274   subst δ'
 275   subst δ''
 276   simp only [SSDataMorphism.pageMapAt, eqToHom_refl, Category.comp_id,
 277     SSDataMorphism.pageMap_comp] at hdf hdg ⊢
 278   rw [Category.assoc, hdg, ← Category.assoc, hdf, Category.assoc]

 285 /-- Composing an epimorphism on the left does not change the image subobject. -/
 286 theorem imageSubobject_epi_comp
 287     {X₁ X₂ X₃ : C} (e : X₁ ⟶ X₂) [Epi e] (f : X₂ ⟶ X₃) :
 288     imageSubobject (e ≫ f) = imageSubobject f := by
 289   apply le_antisymm (imageSubobject_comp_le e f)
 290   have hle := imageSubobject_comp_le e f
 291   haveI : Epi (Subobject.ofLE _ _ hle) := imageSubobject_comp_le_epi_of_epi e f
 292   haveI : IsIso (Subobject.ofLE _ _ hle) := isIso_of_mono_of_epi _
 293   exact Subobject.le_of_comm (inv (Subobject.ofLE _ _ hle))
 294     (by rw [IsIso.inv_comp_eq]; exact (Subobject.ofLE_arrow hle).symm)

 306 /-- Factorization identity for the canonical map `Q/P ⟶ R/P`. -/
 307 theorem factor_cokernelMap
 308     {V : C} (P Q R : Subobject V) (hPQ : P ≤ Q) (hQR : Q ≤ R)
 309     (hPR : P ≤ R := le_trans hPQ hQR) :
 310     Subobject.ofLE Q R hQR ≫ cokernel.π (Subobject.ofLE P R hPR) =
 311       cokernel.π (Subobject.ofLE P Q hPQ) ≫
 312         Subobject.cokernelMap_ofLE P Q R hPQ hQR hPR := by
 313   simp [Subobject.cokernelMap_ofLE, cokernel.π_desc]

 315 /-- The canonical map `Q/P ⟶ R/P` is a monomorphism. -/
 316 instance cokernelMap_ofLE_mono
 317     {V : C} (P Q R : Subobject V) (hPQ : P ≤ Q) (hQR : Q ≤ R)
 318     (hPR : P ≤ R := le_trans hPQ hQR) :
 319     Mono (Subobject.cokernelMap_ofLE P Q R hPQ hQR hPR) := by
 320   open CategoryTheory.Abelian.Pseudoelement in
 321   refine mono_of_zero_of_map_zero _ fun a ha => ?_
 322   obtain ⟨q, hq⟩ := pseudo_surjective_of_epi
 323     (cokernel.π (Subobject.ofLE P Q hPQ)) a
 324   rw [← hq] at ha ⊢
 325   have ha' : pseudoApply (cokernel.π (Subobject.ofLE P R hPR))
 326       (pseudoApply (Subobject.ofLE Q R hQR) q) = 0 := by
 327     have h1 := (Abelian.Pseudoelement.comp_apply
 328       (Subobject.ofLE Q R hQR) (cokernel.π (Subobject.ofLE P R hPR)) q).symm
 329     rw [h1, factor_cokernelMap P Q R hPQ hQR hPR,
 330       Abelian.Pseudoelement.comp_apply]
 331     exact ha
 332   have hexact : (ShortComplex.mk (Subobject.ofLE P R hPR)
 333       (cokernel.π (Subobject.ofLE P R hPR)) (cokernel.condition _)).Exact :=
 334     ShortComplex.cokernelSequence_exact (Subobject.ofLE P R hPR)
 335   obtain ⟨p, hp⟩ := pseudo_exact_of_exact hexact _ ha'
 336   have hp' : pseudoApply (Subobject.ofLE Q R hQR)
 337       (pseudoApply (Subobject.ofLE P Q hPQ) p) =
 338       pseudoApply (Subobject.ofLE Q R hQR) q := by
 339     rw [← Abelian.Pseudoelement.comp_apply, Subobject.ofLE_comp_ofLE]
 340     exact hp
 341   have hinj := pseudo_injective_of_mono (Subobject.ofLE Q R hQR) hp'
 342   rw [← hinj, ← Abelian.Pseudoelement.comp_apply, cokernel.condition, zero_apply]

 346 /-- Equal page classes of cycle representatives differ by a boundary in the ambient object. -/
 347 public theorem SSData.sub_factors_boundary_of_page_eq (D : SSData C)
 348     (n : WithTop ℕ) {T : C}
 349     (u v : T ⟶ Subobject.underlying.obj (D.Z n))
 350     (h : u ≫ D.pageπ n = v ≫ D.pageπ n) :
 351     Subobject.Factors (D.B n)
 352       (u ≫ (D.Z n).arrow - v ≫ (D.Z n).arrow) := by
 353   let i := Subobject.ofLE (D.B n) (D.Z n) (D.B_le_Z n)
 354   have hz : (u - v) ≫ cokernel.π i = 0 := by
 355     change (u - v) ≫ D.pageπ n = 0
 356     rw [Preadditive.sub_comp, h, sub_self]
 357   let b := Abelian.monoLift i (u - v) hz
 358   have hb : b ≫ (D.B n).arrow =
 359       u ≫ (D.Z n).arrow - v ≫ (D.Z n).arrow := by
 360     calc
 361       b ≫ (D.B n).arrow = b ≫ i ≫ (D.Z n).arrow := by
 362         rw [Subobject.ofLE_arrow]
 363       _ = (u - v) ≫ (D.Z n).arrow := by
 364         rw [← Category.assoc, Abelian.monoLift_comp]
 365       _ = u ≫ (D.Z n).arrow - v ≫ (D.Z n).arrow := by
 366         rw [Preadditive.sub_comp]
 367   rw [← hb]
 368   exact Subobject.factors_comp_arrow b

 370 /-- A nonzero page boundary has a first finite stage at which it enters the boundary tower. -/
 371 public theorem SSData.first_boundary_page (D : SSData C) {T : C}
 372     (v : T ⟶ D.V) (n : ℕ)
 373     (hn : Subobject.Factors (D.B (↑n)) v)
 374     (hzero : ¬ Subobject.Factors (D.B (↑(0 : ℕ))) v) :
 375     ∃ m : ℕ, m < n ∧
 376       Subobject.Factors (D.B (↑(m + 1))) v ∧
 377       ¬ Subobject.Factors (D.B (↑m)) v := by
 378   classical
 379   let P : ℕ → Prop := fun j => Subobject.Factors (D.B (↑j)) v
 380   have hex : ∃ j : ℕ, P j := ⟨n, hn⟩
 381   let p := Nat.find hex
 382   have hp : P p := Nat.find_spec hex

 384     by_contra h

 404 theorem subobject_eq_of_quotient_epi
 405     {C : Type*} [Category C] [Abelian C] {V : C}
 406     (B P Q : Subobject V) (hBP : B ≤ P) (hPQ : P ≤ Q)
 407     [Epi (Subobject.ofLE P Q hPQ ≫
 408       cokernel.π (Subobject.ofLE B Q (hBP.trans hPQ)))] : P = Q := by
 409   let i := Subobject.ofLE P Q hPQ
 410   let b := Subobject.ofLE B Q (hBP.trans hPQ)
 411   have hi : Epi i := by
 412     apply (Preadditive.epi_iff_cancel_zero i).mpr
 413     intro R g hg
 414     have hbg : b ≫ g = 0 := by
 415       change Subobject.ofLE B Q _ ≫ g = 0
 416       rw [← Subobject.ofLE_comp_ofLE B P Q hBP hPQ, Category.assoc]
 417       change Subobject.ofLE B P hBP ≫ (i ≫ g) = 0
 418       rw [hg, comp_zero]
 419     let q := cokernel.desc b g hbg
 420     have hq : q = 0 := by
 421       apply zero_of_epi_comp (i ≫ cokernel.π b)
 422       rw [Category.assoc, cokernel.π_desc]
 423       exact hg
 424     calc
 425       g = cokernel.π b ≫ q := (cokernel.π_desc b g hbg).symm
 426       _ = 0 := by rw [hq, comp_zero]
 427   haveI : Epi (Subobject.ofLE P Q hPQ) := hi
 428   haveI : IsIso (Subobject.ofLE P Q hPQ) := isIso_of_mono_of_epi _
 429   exact le_antisymm hPQ
 430     (Subobject.le_of_comm (inv (Subobject.ofLE P Q hPQ))
 431       (by simp [Subobject.ofLE_arrow]))

 433 /-- A zero outgoing differential preserves the cycle subobject. -/
 434 public theorem cycles_succ_of_zero
 435     {C : Type*} [Category C] [Abelian C]
 436     {ι : Type*} [AddCommGroup ι] [DecidableEq ι]
 437     (E : SpectralSequence C ι) (r : ℤ) (hr : E.r₀ ≤ r)
 438     (k : ι) (hd : E.d r k = 0) :
 439     (E.ssData k).Z ↑((r - E.r₀).toNat + 1) =
 440       (E.ssData k).Z ↑(r - E.r₀).toNat := by
 441   let n := (r - E.r₀).toNat
 442   let D := E.ssData k
 443   have hz := E.Z_succ r k hr
 444   rw [hd, kernelSubobject_zero] at hz
 445   let i := Subobject.ofLE (D.Z ↑(n + 1)) (D.Z ↑n)
 446     (D.Z_anti (by exact_mod_cast Nat.le_succ n))
 447   let p := i ≫ D.pageπ ↑n
 448   have hp : imageSubobject p = ⊤ := hz.symm
 449   haveI : IsIso (imageSubobject p).arrow :=
 450     (Subobject.isIso_arrow_iff_eq_top _).mpr hp
 451   haveI : Epi p := by
 452     rw [← imageSubobject_arrow_comp p]
 453     infer_instance
 454   let hBP : D.B ↑n ≤ D.Z ↑(n + 1) :=
 455     (D.B_mono (by exact_mod_cast Nat.le_succ n)).trans (D.B_le_Z _)
 456   let hPQ : D.Z ↑(n + 1) ≤ D.Z ↑n :=
 457     D.Z_anti (by exact_mod_cast Nat.le_succ n)
 458   have hEpi : Epi (Subobject.ofLE (D.Z ↑(n + 1)) (D.Z ↑n) hPQ ≫
 459       cokernel.π (Subobject.ofLE (D.B ↑n) (D.Z ↑n) (hBP.trans hPQ))) := by
 460     change Epi p
 461     infer_instance
 462   exact @subobject_eq_of_quotient_epi C _ _ D.V
 463     (D.B ↑n) (D.Z ↑(n + 1)) (D.Z ↑n) hBP hPQ hEpi

 465 /-- A zero differential leaves its target boundary subobject unchanged. -/
 466 public theorem boundaries_succ_of_zero
 467     {C : Type*} [Category C] [Abelian C]
 468     {ι : Type*} [AddCommGroup ι] [DecidableEq ι]
 469     (E : SpectralSequence C ι) (r : ℤ) (hr : E.r₀ ≤ r)
 470     (k : ι) (hd : E.d r k = 0) :
 471     (E.ssData (k + E.diffDeg r)).B ↑((r - E.r₀).toNat + 1) =
 472       (E.ssData (k + E.diffDeg r)).B ↑(r - E.r₀).toNat := by
 473   let n := (r - E.r₀).toNat
 474   let D := E.ssData (k + E.diffDeg r)
 475   have hb := E.B_succ r k hr
 476   rw [hd, imageSubobject_zero] at hb
 477   let b := Subobject.ofLE (D.B ↑n) (D.Z ↑n) (D.B_le_Z _)
 478   let j := Subobject.ofLE (D.B ↑(n + 1)) (D.Z ↑n)
 479     ((D.B_le_Z _).trans (D.Z_anti (by exact_mod_cast Nat.le_succ n)))
 480   have hj : j ≫ cokernel.π b = 0 := by
 481     have hp : imageSubobject (j ≫ cokernel.π b) = ⊥ := hb.symm
 482     have ha : (imageSubobject (j ≫ cokernel.π b)).arrow = 0 := by
 483       rw [hp]
 484       exact Subobject.bot_arrow
 485     rw [← imageSubobject_arrow_comp (j ≫ cokernel.π b), ha, comp_zero]
 486   apply le_antisymm _ (D.B_mono (by exact_mod_cast Nat.le_succ n))
 487   apply Subobject.le_of_comm (Abelian.monoLift b j hj)
 488   have hcomp := congrArg (fun f => f ≫ (D.Z ↑n).arrow)
 489     (Abelian.monoLift_comp b j hj)
 490   simpa only [Category.assoc, b, j, Subobject.ofLE_arrow] using hcomp

 493 /-- Vanishing of all later outgoing differentials stabilizes the cycles
 494 at this grading, including the actual intersection at infinity. -/
 495 public theorem cycles_top_eq_of_d_eq_zero
 496     {C : Type*} [Category C] [Abelian C]
 497     {ι : Type*} [AddCommGroup ι] [DecidableEq ι]
 498     (E : SpectralSequence C ι) (N : ℤ) (hN : E.r₀ ≤ N) (k : ι)
 499     (hd : ∀ r : ℤ, N ≤ r → E.d r k = 0) :
 500     (E.ssData k).Z ⊤ = (E.ssData k).Z ↑(N - E.r₀).toNat := by
 501   let n₀ := (N - E.r₀).toNat
 502   let D := E.ssData k
 503   have hn₀ : (n₀ : ℤ) = N - E.r₀ := Int.toNat_of_nonneg (by omega)
 504   have htail : ∀ m : ℕ, D.Z (↑(n₀ + m) : WithTop ℕ) = D.Z ↑n₀ := by
 505     intro m
 506     induction m with
 507     | zero => simp
 508     | succ m ih =>
 509       have hz := cycles_succ_of_zero E (E.r₀ + (n₀ + m : ℕ)) (by omega) k
 510         (hd _ (by rw [Nat.cast_add, hn₀]; omega))
 511       simp only [add_sub_cancel_left, Int.toNat_natCast] at hz
 512       exact hz.trans ih
 513   apply le_antisymm (D.Z_anti le_top)
 514   apply D.Z_top_greatest
 515   intro n
 516   by_cases hn : n₀ ≤ n
 517   · have h := (htail (n - n₀)).ge
 518     simpa only [Nat.add_sub_of_le hn] using h
 519   · exact D.Z_anti (by exact_mod_cast Nat.le_of_lt (Nat.lt_of_not_ge hn))

 521 /-- Vanishing of all later incoming differentials stabilizes the boundaries
 522 at this grading, including the actual union at infinity. -/
 523 public theorem boundaries_top_eq_of_d_eq_zero
 524     {C : Type*} [Category C] [Abelian C]
 525     {ι : Type*} [AddCommGroup ι] [DecidableEq ι]
 526     (E : SpectralSequence C ι) (N : ℤ) (hN : E.r₀ ≤ N) (k : ι)
 527     (hd : ∀ r : ℤ, N ≤ r → E.d r (k - E.diffDeg r) = 0) :
 528     (E.ssData k).B ⊤ = (E.ssData k).B ↑(N - E.r₀).toNat := by
 529   let n₀ := (N - E.r₀).toNat
 530   let D := E.ssData k
 531   have hn₀ : (n₀ : ℤ) = N - E.r₀ := Int.toNat_of_nonneg (by omega)
 532   have htail : ∀ m : ℕ, D.B (↑(n₀ + m) : WithTop ℕ) = D.B ↑n₀ := by
 533     intro m
 534     induction m with
 535     | zero => simp
 536     | succ m ih =>
 537       have hb := boundaries_succ_of_zero E (E.r₀ + (n₀ + m : ℕ)) (by omega)
 538         (k - E.diffDeg (E.r₀ + (n₀ + m : ℕ)))
 539         (hd _ (by rw [Nat.cast_add, hn₀]; omega))
 540       simp only [add_sub_cancel_left, Int.toNat_natCast] at hb
 541       have hb' : D.B ↑(n₀ + m + 1) = D.B ↑(n₀ + m) :=
 542         (congrArg (fun j => (E.ssData j).B ↑(n₀ + m + 1) =
 543           (E.ssData j).B ↑(n₀ + m))
 544           (sub_add_cancel k (E.diffDeg (E.r₀ + (n₀ + m : ℕ))))).mp hb
 545       exact hb'.trans ih
 546   apply le_antisymm _ (D.B_mono le_top)
 547   apply D.B_top_least
 548   intro n
 549   by_cases hn : n₀ ≤ n
 550   · have h := (htail (n - n₀)).le
 551     simpa only [Nat.add_sub_of_le hn] using h
 552   · exact D.B_mono (by exact_mod_cast Nat.le_of_lt (Nat.lt_of_not_ge hn))
