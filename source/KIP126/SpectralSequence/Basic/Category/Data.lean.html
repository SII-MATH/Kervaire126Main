  20 /-- The category of graded `SSData` families and their underlying morphisms. -/
  21 instance {ι : Type w} [AddCommGroup ι] [DecidableEq ι] :
  22     Category.{max w v} (GradedSSData C ι) where
  23   Hom D D' := UnderlyingMorphism ι D.data D'.data
  24   id _ := ⟨fun _ => 𝟙 _⟩
  25   comp f g := ⟨fun k => f.φ k ≫ g.φ k⟩
  26   id_comp f := UnderlyingMorphism.ext (funext fun k => Category.id_comp (f.φ k))
  27   comp_id f := UnderlyingMorphism.ext (funext fun k => Category.comp_id (f.φ k))
  28   assoc f g h :=
  29     UnderlyingMorphism.ext (funext fun k => Category.assoc (f.φ k) (g.φ k) (h.φ k))
  30 
  31 /-- The category of pre-spectral sequences. -/
  32 public instance {ι : Type w} [AddCommGroup ι] [DecidableEq ι] :
  33     Category.{max w v} (PreSS C ι) where
  34   Hom E E' := PreSSMorphism E E'
  35   id E := {
  36     toSSDataMorphism := SSDataMorphism.id E.ssData
  37     r₀_eq := rfl
  38     diffDeg_eq := rfl
  39     comm_d := PreSSMorphism.id_comm_d E }
  40   comp f g := {
  41     toSSDataMorphism := f.toSSDataMorphism.comp g.toSSDataMorphism
  42     r₀_eq := f.r₀_eq.trans g.r₀_eq
  43     diffDeg_eq := f.diffDeg_eq.trans g.diffDeg_eq
  44     comm_d := PreSSMorphism.comp_comm_d f g }
  45   id_comp f := PreSSMorphism.ext (funext fun k => Category.id_comp (f.φ k))
  46   comp_id f := PreSSMorphism.ext (funext fun k => Category.comp_id (f.φ k))
  47   assoc f g h :=
  48     PreSSMorphism.ext (funext fun k => Category.assoc (f.φ k) (g.φ k) (h.φ k))
  49 
  50 /-- Forget a pre-spectral sequence to its graded family of nested-subobject data. -/
  51 def PreSS.forget {ι : Type w} [AddCommGroup ι] [DecidableEq ι] :
  52     PreSS C ι ⥤ GradedSSData C ι where
  53   obj E := ⟨E.ssData⟩
  54   map f := ⟨f.φ⟩
  55   map_id _ := rfl
  56   map_comp _ _ := rfl
  57 
  58 /-- The forgetful functor from `PreSS` is faithful. -/
  59 instance {ι : Type w} [AddCommGroup ι] [DecidableEq ι] :
  60     Functor.Faithful (PreSS.forget (C := C) (ι := ι)) where
  61   map_injective h := PreSSMorphism.ext (congrArg UnderlyingMorphism.φ h)

  63 /-- The category of nested-subobject spectral sequences. -/
  64 public instance {ι : Type w} [AddCommGroup ι] [DecidableEq ι] :
  65     Category.{max w v} (SpectralSequence C ι) where
  66   Hom E E' := SpectralSequenceMorphism E E'
  67   id E := {
  68     φ := (SSDataMorphism.id E.ssData).φ
  69     preserves_Z := (SSDataMorphism.id E.ssData).preserves_Z
  70     preserves_B := (SSDataMorphism.id E.ssData).preserves_B
  71     r₀_eq := rfl
  72     diffDeg_eq := rfl
  73     comm_d := PreSSMorphism.id_comm_d E.toPreSS }
  74   comp f g := {
  75     φ := (f.toSSDataMorphism.comp g.toSSDataMorphism).φ
  76     preserves_Z := (f.toSSDataMorphism.comp g.toSSDataMorphism).preserves_Z
  77     preserves_B := (f.toSSDataMorphism.comp g.toSSDataMorphism).preserves_B
  78     r₀_eq := f.r₀_eq.trans g.r₀_eq
  79     diffDeg_eq := f.diffDeg_eq.trans g.diffDeg_eq
  80     comm_d := PreSSMorphism.comp_comm_d f.toPreSSMorphism g.toPreSSMorphism }
  81   id_comp f := SpectralSequenceMorphism.ext
  82     (funext fun k => Category.id_comp (f.φ k))
  83   comp_id f := SpectralSequenceMorphism.ext
  84     (funext fun k => Category.comp_id (f.φ k))
  85   assoc f g h := SpectralSequenceMorphism.ext
  86     (funext fun k => Category.assoc (f.φ k) (g.φ k) (h.φ k))

  88 /-- Spectral-sequence morphisms are exactly morphisms of the underlying `PreSS`. -/
  89 def SpectralSequenceMorphism.equivPreSSMorphism
  90     {ι : Type w} [AddCommGroup ι] [DecidableEq ι]
  91     (E E' : SpectralSequence C ι) :
  92     SpectralSequenceMorphism E E' ≃ PreSSMorphism E.toPreSS E'.toPreSS where
  93   toFun f := f.toPreSSMorphism
  94   invFun g := {
  95     φ := g.φ
  96     preserves_Z := g.preserves_Z
  97     preserves_B := g.preserves_B
  98     r₀_eq := g.r₀_eq
  99     diffDeg_eq := g.diffDeg_eq
 100     comm_d := g.comm_d }
 101   left_inv _ := rfl
 102   right_inv _ := rfl

 104 /-- Inclusion of spectral sequences into pre-spectral sequences. -/
 105 def inclusion
 106     {ι : Type w} [AddCommGroup ι] [DecidableEq ι] :
 107     SpectralSequence C ι ⥤ PreSS C ι where
 108   obj E := E.toPreSS
 109   map f := SpectralSequenceMorphism.equivPreSSMorphism _ _ f
 110   map_id _ := PreSSMorphism.ext rfl
 111   map_comp _ _ := PreSSMorphism.ext rfl
 112 
 113 /-- The inclusion of spectral sequences is faithful. -/
 114 instance {ι : Type w} [AddCommGroup ι] [DecidableEq ι] :
 115     Functor.Faithful (SpectralSequence.inclusion (C := C) (ι := ι)) where
 116   map_injective h := SpectralSequenceMorphism.ext (congrArg (fun m => m.φ) h)
 117 
 118 /-- The inclusion of spectral sequences is full. -/
 119 instance {ι : Type w} [AddCommGroup ι] [DecidableEq ι] :
 120     Functor.Full (SpectralSequence.inclusion (C := C) (ι := ι)) where
 121   map_surjective g :=
 122     ⟨(SpectralSequenceMorphism.equivPreSSMorphism _ _).symm g,
 123       (SpectralSequenceMorphism.equivPreSSMorphism _ _).apply_symm_apply g⟩
 124 
 125 /-- The inclusion of spectral sequences is fully faithful. -/
 126 noncomputable def inclusionFullyFaithful
 127     {ι : Type w} [AddCommGroup ι] [DecidableEq ι] :
 128     (SpectralSequence.inclusion (C := C) (ι := ι)).FullyFaithful :=
 129   Functor.FullyFaithful.ofFullyFaithful _
