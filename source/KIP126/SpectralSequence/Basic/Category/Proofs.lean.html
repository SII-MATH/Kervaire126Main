  20 @[simp]
  21 public theorem PreSSMorphism.pageMap_id
  22     {ι : Type w} [AddCommGroup ι] [DecidableEq ι]
  23     (E : PreSS C ι) (r : ℤ) (k : ι) :
  24     (𝟙 E : E ⟶ E).pageMap r k = 𝟙 _ :=
  25   SSDataMorphism.pageMapAt_id E r k
  26 
  27 @[simp]
  28 public theorem PreSSMorphism.pageMap_comp
  29     {ι : Type w} [AddCommGroup ι] [DecidableEq ι]
  30     {E E' E'' : PreSS C ι} (f : E ⟶ E') (g : E' ⟶ E'') (r : ℤ) (k : ι) :
  31     (f ≫ g).pageMap r k = f.pageMap r k ≫ g.pageMap r k :=
  32   SSDataMorphism.pageMapAt_comp f.toSSDataMorphism g.toSSDataMorphism
  33     f.r₀_eq g.r₀_eq r k

  64 /-- A square commutes exactly when its maps commute at every underlying grading. -/
  65 public theorem commSq_iff_underlying
  66     {ι : Type w} [AddCommGroup ι] [DecidableEq ι]
  67     {W X Y Z : SpectralSequence C ι}
  68     (f : W ⟶ X) (g : W ⟶ Y) (h : X ⟶ Z) (i : Y ⟶ Z) :
  69     SpectralSequence.CommSq f g h i ↔
  70       ∀ (k : ι), f.φ k ≫ h.φ k = g.φ k ≫ i.φ k := by
  71   constructor
  72   · intro hsq k
  73     exact congrFun (congrArg SpectralSequenceMorphism.φ hsq.w) k
  74   · intro hsq
  75     constructor
  76     exact SpectralSequenceMorphism.ext (funext hsq)
