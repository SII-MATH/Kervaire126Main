   8 /-!
   9 # Representative criterion for spectral-sequence morphisms
  10 
  11 An ambient map which preserves cycles, sends zero representatives to zero on
  12 the target page, and satisfies the representative differential equation
  13 determines the canonical quotient maps.  This is the KIP126 formulation of
  14 the useful proved part of the historical KIPBase criterion; it works with
  15 KIP126's nested-subobject pages and does not introduce an independent page-map
  16 field.
  17 -/

  19 namespace KIP126.Core.SpectralSequence

  21 open CategoryTheory CategoryTheory.Limits
  22 
  23 universe u v w
  24 
  25 variable {C : Type u} [Category.{v} C] [Abelian C]
  26 variable {ι : Type w} [AddCommGroup ι] [DecidableEq ι]
  27 
  28 structure DifferentialMorphismCriterion
  29     (E E' : SpectralSequence C ι)
  30     (f : UnderlyingMorphism ι E.ssData E'.ssData) where
  31   r₀_eq : E.r₀ = E'.r₀
  32   diffDeg_eq : E.diffDeg = E'.diffDeg
  33   survives : ∀ (k : ι) (n : WithTop ℕ),
  34     ∃ zLift : Subobject.underlying.obj ((E.ssData k).Z n) ⟶
  35         Subobject.underlying.obj ((E'.ssData k).Z n),
  36       zLift ≫ ((E'.ssData k).Z n).arrow =
  37         ((E.ssData k).Z n).arrow ≫ f.φ k
  38   zero_differential : ∀ (k : ι) (n : WithTop ℕ),
  39     let β := Subobject.ofLE ((E.ssData k).B n)
  40       ((E.ssData k).Z n) ((E.ssData k).B_le_Z n)
  41     β ≫ (survives k n).choose ≫ (E'.ssData k).pageπ n = 0
  42   differential : ∀ (r : ℤ) (k : ι),
  43       let n : WithTop ℕ := ↑(r - E.r₀).toNat
  44       let n' : WithTop ℕ := ↑(r - E'.r₀).toNat
  45       let hn : n = n' := congrArg
  46         (fun r₀ => (↑(r - r₀).toNat : WithTop ℕ)) r₀_eq
  47       let hk : k + E.diffDeg r = k + E'.diffDeg r :=
  48         congrArg (fun d => k + d) (congrFun diffDeg_eq r)
  49       ∀ (T : C)
  50         (x : T ⟶ Subobject.underlying.obj ((E.ssData k).Z n))
  51         (y : T ⟶ Subobject.underlying.obj
  52           ((E.ssData (k + E.diffDeg r)).Z n)),
  53         x ≫ (E.ssData k).pageπ n ≫ E.d r k =
  54           y ≫ (E.ssData (k + E.diffDeg r)).pageπ n →
  55         x ≫ (survives k n).choose ≫ (E'.ssData k).pageπ n ≫
  56           eqToHom (by rw [hn]) ≫ E'.d r k =
  57         y ≫ (survives (k + E.diffDeg r) n).choose ≫
  58           (E'.ssData (k + E.diffDeg r)).pageπ n ≫
  59           eqToHom (by rw [hk, hn])

  61 noncomputable def DifferentialMorphismCriterion.toSSDataMorphism
  62     {E E' : SpectralSequence C ι}
  63     {f : UnderlyingMorphism ι E.ssData E'.ssData}
  64     (H : DifferentialMorphismCriterion E E' f) :
  65     SSDataMorphism ι E.ssData E'.ssData := {
  66   φ := f.φ
  67   preserves_Z := H.survives
  68   preserves_B := by
  69     intro k n
  70     let D := E.ssData k
  71     let D' := E'.ssData k
  72     let β := Subobject.ofLE (D.B n) (D.Z n) (D.B_le_Z n)
  73     let β' := Subobject.ofLE (D'.B n) (D'.Z n) (D'.B_le_Z n)
  74     let zLift := (H.survives k n).choose
  75     have hz : zLift ≫ (D'.Z n).arrow = (D.Z n).arrow ≫ f.φ k :=
  76       (H.survives k n).choose_spec
  77     have hzero : (β ≫ zLift) ≫ D'.pageπ n = 0 := by
  78       simpa only [Category.assoc] using H.zero_differential k n
  79     let bLift := (ShortComplex.exact_cokernel β').lift (β ≫ zLift) hzero
  80     refine ⟨bLift, ?_⟩
  81     calc
  82       bLift ≫ (D'.B n).arrow =
  83           (bLift ≫ β') ≫ (D'.Z n).arrow := by
  84             rw [Category.assoc, Subobject.ofLE_arrow]
  85       _ = (β ≫ zLift) ≫ (D'.Z n).arrow := by
  86             rw [ShortComplex.Exact.lift_f]
  87       _ = (D.B n).arrow ≫ f.φ k := by
  88             rw [Category.assoc, hz, ← Category.assoc,
  89               Subobject.ofLE_arrow]
  90 }

  92 noncomputable def DifferentialMorphismCriterion.toMorphism
  93     {E E' : SpectralSequence C ι}
  94     {f : UnderlyingMorphism ι E.ssData E'.ssData}
  95     (H : DifferentialMorphismCriterion E E' f) :
  96     SpectralSequenceMorphism E E' := by
  97   let F := H.toSSDataMorphism
  98   let P : PreSSMorphism E.toPreSS E'.toPreSS := {
  99     toSSDataMorphism := F
 100     r₀_eq := H.r₀_eq
 101     diffDeg_eq := H.diffDeg_eq
 102     comm_d := by
 103       intro r k
 104       let n : WithTop ℕ := ↑(r - E.r₀).toNat
 105       let hn : n = (↑(r - E'.r₀).toNat : WithTop ℕ) :=
 106         congrArg (fun r₀ => (↑(r - r₀).toNat : WithTop ℕ)) H.r₀_eq
 107       let hk : k + E.diffDeg r = k + E'.diffDeg r :=
 108         congrArg (fun d => k + d) (congrFun H.diffDeg_eq r)
 109       let p := (E.ssData k).pageπ n
 110       let q := (E.ssData (k + E.diffDeg r)).pageπ n
 111       haveI : Epi p := inferInstanceAs (Epi (cokernel.π
 112         (Subobject.ofLE ((E.ssData k).B n) ((E.ssData k).Z n)
 113           ((E.ssData k).B_le_Z n))))
 114       haveI : Epi q := inferInstanceAs (Epi (cokernel.π
 115         (Subobject.ofLE ((E.ssData (k + E.diffDeg r)).B n)
 116           ((E.ssData (k + E.diffDeg r)).Z n)
 117           ((E.ssData (k + E.diffDeg r)).B_le_Z n))))
 118       let g := p ≫ E.d r k
 119       let P := pullback g q
 120       haveI : Epi (pullback.fst g q) := inferInstance
 121       have hrep := H.differential r k P (pullback.fst g q) (pullback.snd g q)
 122         (pullback.condition)
 123       have hπ (j : ι) :
 124           F.cycleMap j n = (H.survives j n).choose := rfl
 125       apply (cancel_epi p).mp
 126       apply (cancel_epi (pullback.fst g q)).mp
 127       dsimp only [P, g, p, q] at hrep ⊢
 128       simp only [SSDataMorphism.pageMapAt] at ⊢
 129       simp only [← Category.assoc] at ⊢
 130       rw [Category.assoc (pullback.fst g q) p (E.d r k)] at ⊢
 131       rw [pullback.condition] at ⊢
 132       rw [Category.assoc (pullback.fst g q) p (F.pageMap k n)] at ⊢
 133       rw [Category.assoc (pullback.snd g q) q
 134         (F.pageMap (k + E.diffDeg r) n)] at ⊢
 135       rw [F.pageπ_pageMap, F.pageπ_pageMap] at ⊢
 136       rw [hπ k, hπ (k + E.diffDeg r)] at ⊢
 137       convert hrep using 1 <;> simp only [Category.assoc, eqToHom_trans] <;>
 138         dsimp only [g, q, p, n]
 139   }
 140   exact {
 141     φ := P.φ
 142     preserves_Z := P.preserves_Z
 143     preserves_B := P.preserves_B
 144     r₀_eq := P.r₀_eq
 145     diffDeg_eq := P.diffDeg_eq
 146     comm_d := P.comm_d }

 148 theorem DifferentialMorphismCriterion.exists_morphism
 149     {E E' : SpectralSequence C ι}
 150     {f : UnderlyingMorphism ι E.ssData E'.ssData}
 151     (H : DifferentialMorphismCriterion E E' f) :
 152     ∃ F : SpectralSequenceMorphism E E', ∀ k, F.φ k = f.φ k := by
 153   exact ⟨H.toMorphism, fun _ => rfl⟩
