  26 /-- The homology of page `r` at grading `k` is canonically isomorphic to page `r + 1`. -/
  27 @[no_expose] public noncomputable def pageHomologyIso
  28     {C : Type u} [Category.{v} C] [Abelian C]
  29     {ι : Type w} [AddCommGroup ι] [DecidableEq ι]
  30     (E : SpectralSequence C ι) (r : ℤ) (k : ι) (hr : E.r₀ ≤ r) :
  31     E.Page (r + 1) k ≅ (E.pageShortComplex r (k - E.diffDeg r)).homology := by
  32   set n := (r - E.r₀).toNat with hn_def
  33   have hn1 : (r + 1 - E.r₀).toNat = n + 1 := by omega
  34   set S := E.pageShortComplex r (k - E.diffDeg r) with hS_def
  35   set k' := k - E.diffDeg r + E.diffDeg r
  36   set D' := E.ssData k'
  37   have hk_eq : k' = k := sub_add_cancel k (E.diffDeg r)
  38   have hBn_le_Bn1 : D'.B ↑n ≤ D'.B ↑(n + 1) :=
  39     D'.B_mono (by exact_mod_cast Nat.le_succ n)
  40   have hBn1_le_Zn1 : D'.B ↑(n + 1) ≤ D'.Z ↑(n + 1) := D'.B_le_Z _
  41   have hZn1_le_Zn : D'.Z ↑(n + 1) ≤ D'.Z ↑n :=
  42     D'.Z_anti (by exact_mod_cast Nat.le_succ n)
  43   have hBn_le_Zn1 : D'.B ↑n ≤ D'.Z ↑(n + 1) :=
  44     le_trans hBn_le_Bn1 hBn1_le_Zn1
  45   set i := Subobject.cokernelMap_ofLE
  46     (D'.B ↑n) (D'.Z ↑(n + 1)) (D'.Z ↑n)
  47     hBn_le_Zn1 hZn1_le_Zn with hi_def
  48   set π_map := Subobject.cokernelDesc_ofLE
  49     (D'.B ↑n) (D'.B ↑(n + 1)) (D'.Z ↑(n + 1))
  50     hBn_le_Bn1 hBn1_le_Zn1 with hπ_def
  51   have wi : i ≫ S.g = 0 := by
  52     apply comp_eq_zero_of_image_le_kernel
  53     change imageSubobject i ≤ kernelSubobject (E.d r k')
  54     rw [E.Z_succ r k' hr,
  55       show Subobject.ofLE (D'.Z ↑(n + 1)) (D'.Z ↑n)
  56           (D'.Z_anti (by exact_mod_cast Nat.le_succ n)) ≫ D'.pageπ ↑n =
  57         cokernel.π (Subobject.ofLE (D'.B ↑n) (D'.Z ↑(n + 1)) hBn_le_Zn1) ≫ i
  58         from factor_cokernelMap _ _ _ _ _,
  59       imageSubobject_epi_comp]
  60   have heq_sub : Subobject.mk i = Subobject.mk (kernel.ι (E.d r k')) := by
  61     change Subobject.mk i = kernelSubobject (E.d r k')
  62     rw [E.Z_succ r k' hr,
  63       show Subobject.ofLE (D'.Z ↑(n + 1)) (D'.Z ↑n)
  64           (D'.Z_anti (by exact_mod_cast Nat.le_succ n)) ≫ D'.pageπ ↑n =
  65         cokernel.π (Subobject.ofLE (D'.B ↑n) (D'.Z ↑(n + 1)) hBn_le_Zn1) ≫ i
  66         from factor_cokernelMap _ _ _ _ _,
  67       imageSubobject_epi_comp, imageSubobject_mono]
  68   have hi : IsLimit (KernelFork.ofι i wi) := by
  69     apply (kernelIsKernel (E.d r k')).ofIsoLimit
  70     exact Fork.ext (Subobject.isoOfMkEqMk i (kernel.ι (E.d r k')) heq_sub).symm
  71       (Subobject.ofMkLEMk_comp heq_sub.ge)
  72   set j := Subobject.cokernelMap_ofLE
  73     (D'.B ↑n) (D'.B ↑(n + 1)) (D'.Z ↑(n + 1))
  74     hBn_le_Bn1 hBn1_le_Zn1 with hj_def
  75   set f_lift := hi.lift (KernelFork.ofι S.f (by exact S.zero)) with hf_lift_def
  76   have hj_π : j ≫ π_map = 0 := by
  77     simp only [j, π_map, Subobject.cokernelMap_ofLE, Subobject.cokernelDesc_ofLE]
  78     ext
  79     simp only [cokernel.π_desc_assoc, cokernel.π_desc, comp_zero,
  80       Category.assoc, cokernel.condition]
  81   have hBn_le_Zn : D'.B ↑(n + 1) ≤ D'.Z ↑n :=
  82     le_trans hBn1_le_Zn1 hZn1_le_Zn
  83   have hfactor :
  84       Subobject.ofLE (D'.B ↑(n + 1)) (D'.Z ↑n) hBn_le_Zn ≫ D'.pageπ ↑n =
  85         cokernel.π (Subobject.ofLE (D'.B ↑n) (D'.B ↑(n + 1)) hBn_le_Bn1) ≫
  86           j ≫ i := by
  87     change Subobject.ofLE (D'.B ↑(n + 1)) (D'.Z ↑n) hBn_le_Zn ≫
  88       cokernel.π (Subobject.ofLE (D'.B ↑n) (D'.Z ↑n) (D'.B_le_Z ↑n)) = _
  89     rw [(Subobject.ofLE_comp_ofLE
  90       (D'.B ↑(n + 1)) (D'.Z ↑(n + 1)) (D'.Z ↑n)
  91       hBn1_le_Zn1 hZn1_le_Zn).symm,
  92       Category.assoc,
  93       factor_cokernelMap
  94         (D'.B ↑n) (D'.Z ↑(n + 1)) (D'.Z ↑n) hBn_le_Zn1 hZn1_le_Zn,
  95       ← Category.assoc
  96         (Subobject.ofLE (D'.B ↑(n + 1)) (D'.Z ↑(n + 1)) hBn1_le_Zn1),
  97       factor_cokernelMap
  98         (D'.B ↑n) (D'.B ↑(n + 1)) (D'.Z ↑(n + 1))
  99         hBn_le_Bn1 hBn1_le_Zn1,
 100       Category.assoc]
 101   have hfi : f_lift ≫ i = S.f :=
 102     hi.fac (KernelFork.ofι S.f (by exact S.zero)) WalkingParallelPair.zero
 103   have hB := E.B_succ r (k - E.diffDeg r) hr
 104   have him : imageSubobject (f_lift ≫ i) = imageSubobject (j ≫ i) := by
 105     rw [hfi]
 106     change imageSubobject (E.d r (k - E.diffDeg r)) = _
 107     rw [hB, hfactor, imageSubobject_epi_comp]
 108   have hjim : imageSubobject (j ≫ i) = Subobject.mk (j ≫ i) :=
 109     imageSubobject_mono _
 110   have hle : imageSubobject (f_lift ≫ i) ≤ Subobject.mk (j ≫ i) :=
 111     him ▸ hjim ▸ le_refl _
 112   have hfact_ji : (Subobject.mk (j ≫ i)).Factors (f_lift ≫ i) := by
 113     apply Subobject.factors_of_le _ hle
 114     have hself := imageSubobject_factors_comp_self (f := f_lift ≫ i) (𝟙 _)
 115     simpa using hself
 116   set φ_ji := (Subobject.mk (j ≫ i)).factorThru (f_lift ≫ i) hfact_ji
 117   have hφ_ji : φ_ji ≫ (Subobject.mk (j ≫ i)).arrow = f_lift ≫ i :=
 118     Subobject.factorThru_arrow _ _ _
 119   have harrow_ji : (Subobject.mk (j ≫ i)).arrow =
 120       (Subobject.underlyingIso (j ≫ i)).hom ≫ (j ≫ i) :=
 121     (Subobject.underlyingIso_hom_comp_eq_mk (j ≫ i)).symm
 122   set ψ := φ_ji ≫ (Subobject.underlyingIso (j ≫ i)).hom with hψ_def
 123   have hψ_j : ψ ≫ j = f_lift := by
 124     apply (cancel_mono i).mp
 125     rw [Category.assoc]
 126     change ψ ≫ (j ≫ i) = f_lift ≫ i
 127     change (φ_ji ≫ (Subobject.underlyingIso (j ≫ i)).hom) ≫ (j ≫ i) = f_lift ≫ i
 128     rw [Category.assoc, ← harrow_ji, hφ_ji]
 129   have wπ : f_lift ≫ π_map = 0 := by
 130     calc
 131       f_lift ≫ π_map = (ψ ≫ j) ≫ π_map := by rw [hψ_j]
 132       _ = ψ ≫ (j ≫ π_map) := by rw [Category.assoc]
 133       _ = ψ ≫ 0 := by rw [hj_π]
 134       _ = 0 := comp_zero
 135   have hπ : IsColimit (CokernelCofork.ofπ π_map wπ) := by
 136     have hφ_eq : φ_ji = factorThruImageSubobject (f_lift ≫ i) ≫
 137         (Subobject.isoOfEq _ _ (him.trans hjim)).hom := by
 138       apply (cancel_mono (Subobject.mk (j ≫ i)).arrow).mp
 139       rw [hφ_ji, Category.assoc, Subobject.isoOfEq_hom, Subobject.ofLE_arrow,
 140         imageSubobject_arrow_comp]
 141     have hψ_epi : Epi ψ := by
 142       rw [hψ_def, hφ_eq, Category.assoc]
 143       exact epi_comp _ _
 144     have hπ_factor : cokernel.π j ≫
 145         (Subobject.thirdIso
 146           (D'.B ↑n) (D'.B ↑(n + 1)) (D'.Z ↑(n + 1))
 147           hBn_le_Bn1 hBn1_le_Zn1).hom = π_map :=
 148       cokernel.π_desc _ _ _
 149     set thirdIso := Subobject.thirdIso
 150       (D'.B ↑n) (D'.B ↑(n + 1)) (D'.Z ↑(n + 1))
 151       hBn_le_Bn1 hBn1_le_Zn1 with hthirdIso_def
 152     have hjs : ∀ (s : Cofork f_lift 0), j ≫ Cofork.π s = 0 := by
 153       intro s
 154       haveI := hψ_epi
 155       apply zero_of_epi_comp ψ
 156       rw [← Category.assoc, hψ_j]
 157       have hs := s.condition
 158       simp only [zero_comp] at hs
 159       exact hs
 160     exact Cofork.IsColimit.mk _
 161       (fun s => thirdIso.inv ≫ cokernel.desc j (Cofork.π s) (hjs s))
 162       (fun s => by
 163         change π_map ≫ thirdIso.inv ≫ cokernel.desc j (Cofork.π s) (hjs s) =
 164           Cofork.π s
 165         rw [show π_map = cokernel.π j ≫ thirdIso.hom from hπ_factor.symm,
 166           Category.assoc, thirdIso.hom_inv_id_assoc, cokernel.π_desc])
 167       (fun s m hm => by
 168         change m = thirdIso.inv ≫ cokernel.desc j (Cofork.π s) (hjs s)
 169         rw [← cancel_epi thirdIso.hom, thirdIso.hom_inv_id_assoc]
 170         apply (cancel_epi (cokernel.π j)).mp
 171         rw [cokernel.π_desc, ← Category.assoc, hπ_factor]
 172         exact hm)
 173   set h : S.LeftHomologyData := {
 174     K := cokernel (Subobject.ofLE (D'.B ↑n) (D'.Z ↑(n + 1)) hBn_le_Zn1)
 175     H := D'.page ↑(n + 1)
 176     i := i
 177     π := π_map
 178     wi := wi
 179     hi := hi
 180     wπ := wπ
 181     hπ := hπ
 182   }
 183   have hPageH : E.Page (r + 1) k = h.H := by
 184     change (E.ssData k).page ↑((r + 1 - E.r₀).toNat) = D'.page ↑(n + 1)
 185     rw [hn1]
 186     congr 1
 187     exact hk_eq.symm ▸ rfl
 188   exact eqToIso hPageH ≪≫ h.homologyIso.symm
 189 
 190 end KIP126.Core.SpectralSequence
