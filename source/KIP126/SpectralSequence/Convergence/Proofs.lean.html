  26 /-- A filtration representative has zero graded class precisely when its
  27 actual ambient element belongs to the next filtration level. -/
  28 public theorem Filtration.toAssociatedGraded_eq_zero_iff
  29     {R : Type u} [Ring R] {ω : Type w} {A : ω → ModuleCat.{v} R}
  30     (F : Filtration A) (s : ℤ) (k : ω)
  31     (x : (Subobject.underlying.obj (F.F s k) : ModuleCat R)) :
  32     F.toAssociatedGraded s k x = 0 ↔
  33       (F.F s k).arrow x ∈ (ModuleCat.subobjectModule (A k)) (F.F (s + 1) k) :=
  34   subobject_cokernel_π_eq_zero_iff (F.F (s + 1) k) (F.F s k) (F.mono s k) x

  36 /-- Detection by zero is equivalent to lifting one filtration level deeper. -/
  37 public theorem detect_zero
  38     {ω : Type w} [AddCommGroup ω] [DecidableEq ω]
  39     {E : SpectralSequence C ω} {ω' : Type w} {A : ω' → C} {F : Filtration A}
  40     (conv : Convergence E A F)
  41     {T : C} {k : ω}
  42     (x : T ⟶ Subobject.underlying.obj
  43       (F.F (conv.reindex k).1 (conv.reindex k).2)) :
  44     Detects conv (0 : T ⟶ (E.ssData k).eInfty) x ↔
  45       ∃ (x' : T ⟶ Subobject.underlying.obj
  46         (F.F ((conv.reindex k).1 + 1) (conv.reindex k).2)),
  47         x' ≫ Subobject.ofLE
  48           (F.F ((conv.reindex k).1 + 1) (conv.reindex k).2)
  49           (F.F (conv.reindex k).1 (conv.reindex k).2)
  50           (F.mono (conv.reindex k).1 (conv.reindex k).2) = x := by
  51   simp only [Detects, Filtration.toAssociatedGraded, Filtration.associatedGraded,
  52     Limits.zero_comp]
  53   constructor
  54   · intro h
  55     exact ⟨Abelian.monoLift _ x (by rw [h]), Abelian.monoLift_comp _ x (by rw [h])⟩
  56   · rintro ⟨x', hx'⟩
  57     rw [← hx', Category.assoc, cokernel.condition, Limits.comp_zero]

  59 /-- Two representatives detected by the same class differ by a deeper-filtration lift. -/
  60 public theorem detect_difference
  61     {ω : Type w} [AddCommGroup ω] [DecidableEq ω]
  62     {E : SpectralSequence C ω} {ω' : Type w} {A : ω' → C} {F : Filtration A}
  63     (conv : Convergence E A F)
  64     {T : C} {k : ω}
  65     (y : T ⟶ (E.ssData k).eInfty)
  66     (x x' : T ⟶ Subobject.underlying.obj
  67       (F.F (conv.reindex k).1 (conv.reindex k).2)) :
  68     (Detects conv y x ∧ Detects conv y x') ↔
  69       (Detects conv y x ∧
  70         ∃ (z : T ⟶ Subobject.underlying.obj
  71           (F.F ((conv.reindex k).1 + 1) (conv.reindex k).2)),
  72           z ≫ Subobject.ofLE
  73             (F.F ((conv.reindex k).1 + 1) (conv.reindex k).2)
  74             (F.F (conv.reindex k).1 (conv.reindex k).2)
  75             (F.mono (conv.reindex k).1 (conv.reindex k).2) = x - x') := by
  76   constructor
  77   · rintro ⟨hx, hx'⟩
  78     refine ⟨hx, ?_⟩
  79     have key : (x - x') ≫
  80         F.toAssociatedGraded (conv.reindex k).1 (conv.reindex k).2 = 0 := by
  81       rw [Detects] at hx hx'
  82       rw [Preadditive.sub_comp, sub_eq_zero]
  83       exact hx.symm.trans hx'
  84     simp only [Filtration.toAssociatedGraded] at key
  85     exact ⟨Abelian.monoLift _ (x - x') key,
  86       Abelian.monoLift_comp _ (x - x') key⟩
  87   · rintro ⟨hx, z, hz⟩
  88     refine ⟨hx, ?_⟩
  89     change y ≫ (conv.iso k).hom =
  90       x' ≫ F.toAssociatedGraded (conv.reindex k).1 (conv.reindex k).2
  91     have hd : Detects conv y x := hx
  92     rw [Detects] at hd
  93     have hx'eq : x' = x - z ≫ Subobject.ofLE
  94         (F.F ((conv.reindex k).1 + 1) (conv.reindex k).2)
  95         (F.F (conv.reindex k).1 (conv.reindex k).2)
  96         (F.mono (conv.reindex k).1 (conv.reindex k).2) := by
  97       rw [hz, sub_sub_cancel]
  98     have hz0 : (z ≫ Subobject.ofLE
  99         (F.F ((conv.reindex k).1 + 1) (conv.reindex k).2)
 100         (F.F (conv.reindex k).1 (conv.reindex k).2)
 101         (F.mono (conv.reindex k).1 (conv.reindex k).2)) ≫
 102         F.toAssociatedGraded (conv.reindex k).1 (conv.reindex k).2 = 0 := by
 103       dsimp only [Filtration.toAssociatedGraded, Filtration.associatedGraded]
 104       rw [Category.assoc, cokernel.condition, comp_zero]
 105       rfl
 106     rw [hx'eq, Preadditive.sub_comp, hd, hz0, sub_zero]

 116 /-- Eventual vanishing implies separatedness. The stronger premise uses
 117 the eventual-vanishing predicate on the same filtration. -/
 118 theorem Filtration.isHausdorff_of_isEventuallyZero
 119     {ω : Type w} {A : ω → C} {F : Filtration A}
 120     (h : F.IsEventuallyZero) : F.IsHausdorff := by
 121   intro k S hS
 122   obtain ⟨s, hs⟩ := h k
 123   change F.F s k = ⊥ at hs
 124   exact le_antisymm (by simpa only [hs] using hS s) bot_le

 126 /-- A bounded-above filtration is Hausdorff. -/
 127 public theorem Filtration.IsBoundedAbove.toIsHausdorff
 128     {ω : Type w} {A : ω → C} {F : Filtration A}
 129     (h : F.IsBoundedAbove) : F.IsHausdorff := by
 130   apply Filtration.isHausdorff_of_isEventuallyZero
 131   intro k
 132   exact ⟨h.hi k, h.boundedAbove k (h.hi k) le_rfl⟩

 135 @[ext]
 136 public theorem ConvergenceMorphism.ext
 137     {ω : Type w} [AddCommGroup ω] [DecidableEq ω]
 138     {E₁ E₂ : SpectralSequence C ω} {ω' : Type w}

 140     {conv₁ : Convergence E₁ A₁ F₁} {conv₂ : Convergence E₂ A₂ F₂}
 141     {f g : ConvergenceMorphism conv₁ conv₂}
 142     (he : f.eMap = g.eMap) (ha : f.aMap = g.aMap) : f = g := by
 143   cases f with | mk f_data f_re f_iso => ?_
 144   cases f_data with | mk f_e f_a f_fil => ?_

 146   cases g_data with | mk g_e g_a g_fil => ?_
 147   subst he
 148   subst ha
 149   rfl

 199 /-- Explicit identity lifts induce the identity on associated graded pieces. -/
 200 theorem Filtration.inducedGradedMapOfMap_id
 201     {ω' : Type w} {A : ω' → C} (F : Filtration A) (s : ℤ) (k' : ω')
 202     (hw : ∀ (s : ℤ) (k' : ω'),
 203       Subobject.ofLE (F.F (s + 1) k') (F.F s k') (F.mono s k') ≫ 𝟙 _ =
 204         (𝟙 _ : Subobject.underlying.obj (F.F (s + 1) k') ⟶ _) ≫
 205           Subobject.ofLE (F.F (s + 1) k') (F.F s k') (F.mono s k')) :

 296 /-- Associated-graded maps respect composition of convergence morphisms. -/
 297 theorem Filtration.inducedAssocGradedMap_comp
 298     {ω : Type w} [AddCommGroup ω] [DecidableEq ω]
 299     {E₁ E₂ E₃ : SpectralSequence C ω} {ω' : Type w}
 300     {A₁ A₂ A₃ : ω' → C}
 301     {F₁ : Filtration A₁} {F₂ : Filtration A₂} {F₃ : Filtration A₃}
 302     {conv₁ : Convergence E₁ A₁ F₁} {conv₂ : Convergence E₂ A₂ F₂}
 303     {conv₃ : Convergence E₃ A₃ F₃}
 304     (f : ConvergenceMorphism conv₁ conv₂) (g : ConvergenceMorphism conv₂ conv₃)
 305     (s : ℤ) (k' : ω') :
 306     Filtration.inducedAssocGradedMap (fun k' => f.aMap k' ≫ g.aMap k')
 307         (ConvergenceMorphismData.fcComp f g) s k' =
 308       Filtration.inducedAssocGradedMap f.aMap f.filtration_compat s k' ≫
 309         Filtration.inducedAssocGradedMap g.aMap g.filtration_compat s k' := by
 310   rw [Filtration.inducedAssocGradedMap_eq_inducedGradedMapOfMap
 311     (fun k' => f.aMap k' ≫ g.aMap k') (ConvergenceMorphismData.fcComp f g)]
 312   rw [Filtration.inducedAssocGradedMap_eq_inducedGradedMapOfMap f.aMap f.filtration_compat]
 313   rw [Filtration.inducedAssocGradedMap_eq_inducedGradedMapOfMap g.aMap g.filtration_compat]
 314   have h : (fun s k' => (ConvergenceMorphismData.fcComp f g s k').choose) =
 315       fun s k' => (f.filtration_compat s k').choose ≫
 316         (g.filtration_compat s k').choose := by
 317     funext s k'
 318     apply (cancel_mono ((F₃.F s k').arrow)).mp
 319     rw [Category.assoc _ _ ((F₃.F s k').arrow),
 320       (ConvergenceMorphismData.fcComp f g s k').choose_spec,
 321       (g.filtration_compat s k').choose_spec]
 322     conv_rhs => rw [← Category.assoc _ _ (g.aMap k'),
 323       (f.filtration_compat s k').choose_spec]
 324     exact (Category.assoc _ _ _).symm
 325   rw [Filtration.inducedGradedMapOfMap_congr
 326     (Filtration.choose_compat _ (ConvergenceMorphismData.fcComp f g))
 327     (fun s k' => by
 328       apply (cancel_mono ((F₃.F s k').arrow)).mp
 329       simp only [Category.assoc]
 330       rw [(g.filtration_compat s k').choose_spec,
 331         ← Category.assoc _ _ (g.aMap k'),
 332         (f.filtration_compat s k').choose_spec,
 333         ← Category.assoc _ _ (g.aMap k'),
 334         ← Category.assoc _ ((F₁.F s k').arrow) (f.aMap k'),
 335         Subobject.ofLE_arrow, Subobject.ofLE_arrow,
 336         (g.filtration_compat (s + 1) k').choose_spec,
 337         ← Category.assoc _ _ (g.aMap k'),
 338         (f.filtration_compat (s + 1) k').choose_spec]) h s k']
 339   exact Filtration.inducedGradedMapOfMap_comp _ _
 340     (Filtration.choose_compat f.aMap f.filtration_compat)
 341     (Filtration.choose_compat g.aMap g.filtration_compat) _ s k'

 360 /-- Compatibility of convergence isomorphisms is stable under composition. -/
 361 theorem ConvergenceMorphism.iso_compat_comp
 362     {ω : Type w} [AddCommGroup ω] [DecidableEq ω]
 363     {ω' : Type w} {X Y Z : ConvergingSS C ω ω'}
 364     (f : ConvergenceMorphism X.conv Y.conv) (g : ConvergenceMorphism Y.conv Z.conv)
 365     (k : ω) :
 366     (f.eMap k ≫ g.eMap k) ≫ (Z.conv.iso k).hom ≫
 367         Z.F.transportGraded ((congrFun (f.reindex_eq.trans g.reindex_eq) k).symm) =
 368       (X.conv.iso k).hom ≫
 369         Filtration.inducedAssocGradedMap (fun k' => f.aMap k' ≫ g.aMap k')
 370           (ConvergenceMorphismData.fcComp f g)
 371           (X.conv.reindex k).1 (X.conv.reindex k).2 := by
 372   have hsplit :
 373       Z.F.transportGraded ((congrFun (f.reindex_eq.trans g.reindex_eq) k).symm) =
 374         Z.F.transportGraded ((congrFun g.reindex_eq k).symm) ≫
 375           Z.F.transportGraded ((congrFun f.reindex_eq k).symm) := by
 376     rw [Filtration.transportGraded_trans]
 377   have hnat :
