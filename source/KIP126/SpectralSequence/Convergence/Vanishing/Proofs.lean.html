   9 /-! Vanishing of associated grades raises the filtration of the same
  10 object. This extracts the filtration argument from the historical Adams
  11 divisibility proof without importing its Adams objects or assumptions. -/

  13 namespace KIP126.Core.SpectralSequence
  14 
  15 open CategoryTheory CategoryTheory.Limits
  16 
  17 universe u v w
  18 variable {C : Type u} [Category.{v} C] [Abelian C]
  19   {ω : Type w} {A : ω → C} (F : Filtration A)
  20 
  21 /-- A zero associated grade identifies adjacent filtration subobjects. -/
  22 public theorem Filtration.eq_succ_of_associatedGraded_isZero (s : ℤ) (k : ω)
  23     (h : IsZero (F.associatedGraded s k)) : F.F s k = F.F (s + 1) k := by
  24   let i := Subobject.ofLE (F.F (s + 1) k) (F.F s k) (F.mono s k)
  25   haveI : Epi i := Abelian.epi_of_cokernel_π_eq_zero i (h.eq_of_tgt _ _)
  26   haveI : IsIso i := isIso_of_mono_of_epi i
  27   apply le_antisymm
  28   · apply Subobject.le_of_comm (inv i)
  29     rw [← Subobject.ofLE_arrow (F.mono s k)]
  30     exact IsIso.inv_hom_id_assoc i _
  31   · exact F.mono s k
  32 
  33 /-- Vanishing on a finite interval identifies its endpoint filtrations.
  34 No separation or convergence hypothesis is required. -/
  35 public theorem Filtration.eq_of_associatedGraded_isZero (s t : ℤ) (k : ω)
  36     (hst : s ≤ t)
  37     (h : ∀ j : ℤ, s ≤ j → j < t → IsZero (F.associatedGraded j k)) :
  38     F.F s k = F.F t k := by
  39   have step : ∀ n : ℕ, s + n ≤ t → F.F s k = F.F (s + n) k := by
  40     intro n hn
  41     induction n with
  42     | zero => simp
  43     | succ n ih =>
  44       rw [show s + ((n + 1 : ℕ) : ℤ) = (s + n) + 1 by omega]
  45       exact (ih (by omega)).trans
  46         (F.eq_succ_of_associatedGraded_isZero _ k (h _ (by omega) (by omega)))
  47   simpa only [show s + ((t - s).toNat : ℤ) = t by omega] using
  48     step (t - s).toNat (by omega)
