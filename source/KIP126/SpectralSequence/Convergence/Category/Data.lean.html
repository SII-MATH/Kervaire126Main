  14 open CategoryTheory

  16 universe u v w
  17 
  18 variable {C : Type u} [Category.{v} C] [Abelian C]
  19 
  20 /-- The category whose objects are convergent nested-subobject spectral sequences. -/
  21 public noncomputable instance
  22     {ω : Type w} [AddCommGroup ω] [DecidableEq ω] {ω' : Type w} :
  23     Category.{max w v} (ConvergingSS C ω ω') where
  24   Hom X Y := ConvergenceMorphism X.conv Y.conv
  25   id X := ⟨
  26     { eMap := fun _ => 𝟙 _
  27       aMap := fun _ => 𝟙 _
  28       filtration_compat := by exact X.F.fcId },
  29     rfl,
  30     fun k => by
  31       simp only [Category.id_comp, Filtration.transportGraded_self,
  32         Category.comp_id,
  33         X.F.inducedAssocGradedMap_id (X.conv.reindex k).1 (X.conv.reindex k).2]⟩
  34   comp {X Y Z} f g := ⟨
  35     { eMap := fun k => f.eMap k ≫ g.eMap k
  36       aMap := fun k' => f.aMap k' ≫ g.aMap k'
  37       filtration_compat := by exact ConvergenceMorphismData.fcComp f g },
  38     f.reindex_eq.trans g.reindex_eq,
  39     fun k => by
  40       show (f.eMap k ≫ g.eMap k) ≫ (Z.conv.iso k).hom ≫
  41           Z.F.transportGraded
  42             ((congrFun (f.reindex_eq.trans g.reindex_eq) k).symm) =
  43         (X.conv.iso k).hom ≫
  44           Filtration.inducedAssocGradedMap (fun k' => f.aMap k' ≫ g.aMap k')
  45             (ConvergenceMorphismData.fcComp f g)
  46             (X.conv.reindex k).1 (X.conv.reindex k).2
  47       exact ConvergenceMorphism.iso_compat_comp f g k⟩
  48   id_comp f := ConvergenceMorphism.ext
  49     (funext fun _ => Category.id_comp _) (funext fun _ => Category.id_comp _)
  50   comp_id f := ConvergenceMorphism.ext
  51     (funext fun _ => Category.comp_id _) (funext fun _ => Category.comp_id _)
  52   assoc f g h := ConvergenceMorphism.ext
  53     (funext fun _ => Category.assoc _ _ _) (funext fun _ => Category.assoc _ _ _)
  54 
  55 end KIP126.Core.SpectralSequence
