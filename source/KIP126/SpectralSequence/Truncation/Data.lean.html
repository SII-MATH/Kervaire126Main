   8 /-!
   9 # Truncation data for nested-subobject filtrations
  10 
  11 The historical API is retained, while the quotient objects and transition
  12 maps are delegated to KIP126's canonical algebra-filtration implementation.
  13 -/
  14 
  15 namespace KIP126.Core.SpectralSequence
  16 
  17 open CategoryTheory CategoryTheory.Limits
  18 
  19 universe u v w
  20 
  21 variable {C : Type u} [Category.{v} C] [Abelian C]
  22 
  23 /-- The truncated object `A(k) / F^(s₀+1)(A(k))`. -/
  24 @[reducible] noncomputable def Filtration.truncatedObj
  25     {ω : Type w} {A : ω → C} (fil : Filtration A) (s₀ : ℤ) (k : ω) : C :=
  26   fil.quotientAt (s₀ + 1) k
  27 
  28 /-- The quotient projection to the historical truncation level. -/
  29 @[reducible] noncomputable def Filtration.truncationProj
  30     {ω : Type w} {A : ω → C} (fil : Filtration A) (s₀ : ℤ) (k : ω) :
  31     A k ⟶ fil.truncatedObj s₀ k :=
  32   fil.quotientProjection (s₀ + 1) k
  33 
  34 /-- The image filtration induced on a truncated object. -/
  35 @[reducible] noncomputable def Filtration.truncatedFiltration
  36     {ω : Type w} {A : ω → C} (fil : Filtration A) (s₀ : ℤ) :
  37     Filtration (fil.truncatedObj s₀) :=
  38   fil.quotientFiltration (s₀ + 1)
  39 
  40 /-- Transition `A/F^(s₁+1) ⟶ A/F^(s₀+1)` for `s₀ ≤ s₁`. -/
  41 noncomputable def Filtration.truncationTransition
  42     {ω : Type w} {A : ω → C} (fil : Filtration A)
  43     {s₀ s₁ : ℤ} (h : s₀ ≤ s₁) (k : ω) :
  44     fil.truncatedObj s₁ k ⟶ fil.truncatedObj s₀ k :=
  45   fil.quotientTransition (show s₀ + 1 ≤ s₁ + 1 by omega) k
  46 
  47 /-- Map on truncated target objects induced by a convergence morphism. -/
  48 noncomputable def ConvergenceMorphism.truncatedAMap
  49     {ω : Type w} [AddCommGroup ω] [DecidableEq ω]
  50     {E₁ E₂ : SpectralSequence C ω} {ω' : Type w}
  51     {A₁ A₂ : ω' → C} {F₁ : Filtration A₁} {F₂ : Filtration A₂}
  52     {conv₁ : Convergence E₁ A₁ F₁} {conv₂ : Convergence E₂ A₂ F₂}
  53     (cm : ConvergenceMorphism conv₁ conv₂) (s₀ : ℤ) (k' : ω') :
  54     F₁.truncatedObj s₀ k' ⟶ F₂.truncatedObj s₀ k' := by
  55   change cokernel ((F₁.F (s₀ + 1) k').arrow) ⟶
  56     cokernel ((F₂.F (s₀ + 1) k').arrow)
  57   exact cokernel.desc ((F₁.F (s₀ + 1) k').arrow)
  58     (cm.aMap k' ≫ cokernel.π ((F₂.F (s₀ + 1) k').arrow))
  59     (by
  60       obtain ⟨φ, hφ⟩ := cm.filtration_compat (s₀ + 1) k'
  61       rw [← Category.assoc, ← hφ, Category.assoc, cokernel.condition, comp_zero])
  62 
  63 end KIP126.Core.SpectralSequence
