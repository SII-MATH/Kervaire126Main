   9 /-- Adams 谱序列的有效显示页；下界作为数据的一部分保存。 -/
  10 public structure AdamsPage where
  11   value : ℕ
  12   valid : 2 ≤ value
  13   deriving DecidableEq
  14 
  15 namespace AdamsPage
  16 
  17 /-- Adams 谱序列的第二页。 -/
  18 public def two : AdamsPage := ⟨2, le_rfl⟩
  19 
  20 public def toNat (r : AdamsPage) : ℕ := r.value
  21 
  22 public def toInt (r : AdamsPage) : ℤ := r.value
  23 
  24 /-- 从自然数构造 Adams 页时必须显式提供有效性证明。 -/
  25 public def ofNat (r : ℕ) (hr : 2 ≤ r) : AdamsPage := ⟨r, hr⟩
  26 
  27 /-- 整数页只在证明有效下界后才能进入 Adams 页接口。 -/
  28 public def ofInt (r : ℤ) (hr : 2 ≤ r) : AdamsPage :=
  29   ⟨r.toNat, by simpa using Int.toNat_le_toNat hr⟩
  30 
  31 end AdamsPage
  32 
  33 /-- 扩张谱序列的长度；零长度是合法的，因此不带 Adams 页下界。 -/
  34 public structure ExtensionLength where
  35   value : ℕ
  36   deriving DecidableEq
  37 
  38 namespace ExtensionLength
  39 
  40 public def zero : ExtensionLength := ⟨0⟩
  41 
  42 public def ofNat (n : ℕ) : ExtensionLength := ⟨n⟩
  43 
  44 public def toNat (n : ExtensionLength) : ℕ := n.value
  45 
  46 public def toInt (n : ExtensionLength) : ℤ := n.value
  47 
  48 /-- 只有长度至少为二时，才能显式把扩张长度解释为 Adams 页。 -/
  49 public def toAdamsPage (n : ExtensionLength) (hn : 2 ≤ n.value) : AdamsPage :=
  50   ⟨n.value, hn⟩
  51 
  52 end ExtensionLength
  53 
  54 namespace AdamsPage
  55 
  56 /-- 把 Adams 页数值显式解释为扩张长度。 -/
  57 public def toExtensionLength (r : AdamsPage) : ExtensionLength := ⟨r.value⟩
  58 
  59 end AdamsPage

  61 public structure PageLevelConvention where
  62   firstPage : ℕ
  63   admissibleFrom : ℤ
  64   page : ℕ → ℤ
  65   cycleLevel : ℕ → ℤ
  66   quotientExponent : ℕ → ℤ
  67   page_first : page firstPage = admissibleFrom
  68   page_succ : ∀ r, page (r + 1) = page r + 1
  69   cycle_succ : ∀ r, cycleLevel (r + 1) = cycleLevel r + 1
  70   quotient_succ : ∀ r, quotientExponent (r + 1) = quotientExponent r + 1
  71   cycleLevel_eq_quotientExponent : ∀ r,
  72     cycleLevel r = quotientExponent r
  73 
  74 namespace PageLevelConvention
  75 
  76 /-- 在 Adams 页类型上读取底层整数页。 -/
  77 public def pageAtAdams (P : PageLevelConvention) (r : AdamsPage) : ℤ :=
  78   P.page r.toNat
  79 
  80 /-- 在扩张长度类型上读取底层整数页。 -/
  81 public def pageAtExtension (P : PageLevelConvention) (n : ExtensionLength) : ℤ :=
  82   P.page n.toNat
  83 
  84 end PageLevelConvention
