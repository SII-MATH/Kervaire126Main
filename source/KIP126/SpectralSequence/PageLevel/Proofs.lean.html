  14 /-- 有效 Adams 页转换成整数后仍满足第二页下界。 -/
  15 public theorem two_le_toInt (r : AdamsPage) : (2 : ℤ) ≤ r.toInt := by
  16   rcases r with ⟨r, hr⟩
  17   change (2 : ℤ) ≤ (r : ℤ)
  18   exact_mod_cast hr
  19 
  20 end AdamsPage

  24 public theorem page_ge (P : PageLevelConvention) {r : ℕ}
  25     (h : P.firstPage ≤ r) : P.admissibleFrom ≤ P.page r := by
  26   induction r, h using Nat.le_induction with
  27   | base => simp [P.page_first]
  28   | succ r _ ih => rw [P.page_succ]; omega
