  22 /-- The filtered map carried by this convergence morphism. -/
  23 public def ConvergenceMorphism.filteredMap
  24     {ι : Type w} [AddCommGroup ι] [DecidableEq ι]
  25     {E E' : SpectralSequence C ι}
  26     {c : Convergence E A F} {c' : Convergence E' B G}
  27     (m : ConvergenceMorphism c c') : FilteredMorphism F G where
  28   map := m.aMap
  29   compat := m.filtration_compat

  31 /-- Quotient projection commutes with the restriction of the same filtered map. -/
  32 public theorem FilteredMorphism.quotient_naturality (f : FilteredMorphism F G)
  33     (s : ℤ) (k : ω) :
  34     F.toAssociatedGraded s k ≫ f.inducedGrMap s k =
  35       (f.compat s k).choose ≫ G.toAssociatedGraded s k := by
  36   simp only [Filtration.toAssociatedGraded, FilteredMorphism.inducedGrMap,
  37     Filtration.inducedAssocGradedMap, cokernel.map, cokernel.π_desc]

  41 /-- The actual representatives of an extension of nonnegative length. -/
  42 public def Relation (f : FilteredMorphism F G) (k : ω) (s n : ℤ)
  43     {T : C} (x : T ⟶ F.associatedGraded s k)
  44     (y : T ⟶ G.associatedGraded (s + n) k) : Prop :=
  45   0 ≤ n ∧ ∃ (a : T ⟶ Subobject.underlying.obj (F.F s k))
  46     (b : T ⟶ Subobject.underlying.obj (G.F (s + n) k)),
  47     a ≫ F.toAssociatedGraded s k = x ∧
  48     b ≫ G.toAssociatedGraded (s + n) k = y ∧
  49     a ≫ (F.F s k).arrow ≫ f.map k = b ≫ (G.F (s + n) k).arrow
  50 
  51 /-- Homogeneous target ambiguity of the same representative equation. -/
  52 public def TargetIndeterminacy (f : FilteredMorphism F G) (k : ω) (s n : ℤ)
  53     (T : C) : Set (T ⟶ G.associatedGraded (s + n) k) :=
  54   {z | Relation f k s n (0 : T ⟶ F.associatedGraded s k) z}
  55 
  56 /-- All targets differing from `y` by an actual homogeneous solution. -/
  57 public def TargetCoset (f : FilteredMorphism F G) (k : ω) (s n : ℤ)
  58     {T : C} (y : T ⟶ G.associatedGraded (s + n) k) :
  59     Set (T ⟶ G.associatedGraded (s + n) k) :=
  60   {z | z - y ∈ TargetIndeterminacy f k s n T}
  61 
  62 /-- Essentiality is nonvanishing modulo the complete target ambiguity. -/
  63 public def Essential (f : FilteredMorphism F G) (k : ω) (s n : ℤ)
  64     {T : C} (x : T ⟶ F.associatedGraded s k)
  65     (y : T ⟶ G.associatedGraded (s + n) k) : Prop :=
  66   Relation f k s n x y ∧ 0 ∉ TargetCoset f k s n y

  72 public theorem Relation.add (h : Relation f k s n x y)
  73     (h' : Relation f k s n x' y') : Relation f k s n (x + x') (y + y') := by
  74   obtain ⟨hn, a, b, ha, hb, he⟩ := h
  75   obtain ⟨_, a', b', ha', hb', he'⟩ := h'
  76   refine ⟨hn, a + a', b + b', ?_, ?_, ?_⟩
  77   · simp only [Preadditive.add_comp, ha, ha']
  78   · simp only [Preadditive.add_comp, hb, hb']
  79   · simp only [Preadditive.add_comp, he, he']
  80 
  81 public theorem Relation.sub (h : Relation f k s n x y)
  82     (h' : Relation f k s n x' y') : Relation f k s n (x - x') (y - y') := by
  83   obtain ⟨hn, a, b, ha, hb, he⟩ := h
  84   obtain ⟨_, a', b', ha', hb', he'⟩ := h'
  85   refine ⟨hn, a - a', b - b', ?_, ?_, ?_⟩
  86   · simp only [Preadditive.sub_comp, ha, ha']
  87   · simp only [Preadditive.sub_comp, hb, hb']
  88   · simp only [Preadditive.sub_comp, he, he']

  90 /-- Once a target is known, its coset consists exactly of all targets of the same source. -/
  91 public theorem Relation.mem_targetCoset_iff (h : Relation f k s n x y) :
  92     z ∈ TargetCoset f k s n y ↔ Relation f k s n x z := by
  93   constructor
  94   · intro hz
  95     have hh : Relation f k s n 0 (z - y) := hz
  96     simpa only [zero_add, sub_add_cancel] using hh.add h
  97   · intro hz
  98     change Relation f k s n 0 (z - y)
  99     simpa only [sub_self] using hz.sub h

 101 /-- A known extension is essential exactly when the same source has no zero target. -/
 102 public theorem essential_iff (h : Relation f k s n x y) :
 103     Essential f k s n x y ↔ ¬ Relation f k s n x 0 := by
 104   simp only [Essential, h, true_and, h.mem_targetCoset_iff]

 106 /-- The ambiguity consists exactly of images of higher-source-filtration
 107 corrections, with their actual target lifts and quotient labels. -/
 108 public theorem mem_indeterminacy_iff :
 109     z ∈ TargetIndeterminacy f k s n T ↔
 110       0 ≤ n ∧ ∃ (a : T ⟶ Subobject.underlying.obj (F.F (s + 1) k))
 111         (b : T ⟶ Subobject.underlying.obj (G.F (s + n) k)),
 112         b ≫ G.toAssociatedGraded (s + n) k = z ∧
 113         a ≫ (F.F (s + 1) k).arrow ≫ f.map k =
 114           b ≫ (G.F (s + n) k).arrow := by
 115   let i := Subobject.ofLE (F.F (s + 1) k) (F.F s k) (F.mono s k)
 116   constructor
 117   · rintro ⟨hn, a, b, ha, hb, he⟩
 118     have ha0 : a ≫ cokernel.π i = 0 := ha
 119     let a' := Abelian.monoLift i a ha0
 120     have ha' : a' ≫ i = a := Abelian.monoLift_comp i a ha0
 121     refine ⟨hn, a', b, hb, ?_⟩
 122     rw [← ha'] at he
 123     simpa only [Category.assoc, i, Subobject.ofLE_arrow_assoc] using he
 124   · rintro ⟨hn, a, b, hb, he⟩
 125     refine ⟨hn, a ≫ i, b, ?_, hb, ?_⟩
 126     · change (a ≫ i) ≫ cokernel.π i = 0
 127       rw [Category.assoc, cokernel.condition, comp_zero]
 128     · simpa only [Category.assoc, i, Subobject.ofLE_arrow_assoc] using he

 130 theorem zero_lifts_iff (f : FilteredMorphism F G) (k : ω) (s t : ℤ) (h : t = s)
 131     {T : C} (x : T ⟶ F.associatedGraded s k)
 132     (y : T ⟶ G.associatedGraded t k) :
 133     (∃ (a : T ⟶ Subobject.underlying.obj (F.F s k))
 134       (b : T ⟶ Subobject.underlying.obj (G.F t k)),
 135       a ≫ F.toAssociatedGraded s k = x ∧ b ≫ G.toAssociatedGraded t k = y ∧
 136       a ≫ (F.F s k).arrow ≫ f.map k = b ≫ (G.F t k).arrow) ↔
 137       (∃ a : T ⟶ Subobject.underlying.obj (F.F s k),
 138         a ≫ F.toAssociatedGraded s k = x) ∧
 139       x ≫ f.inducedGrMap s k = y ≫ G.transportGraded (congrArg (fun q => (q, k)) h) := by
 140   subst t
 141   simp only [Filtration.transportGraded_self, Category.comp_id]
 142   constructor
 143   · rintro ⟨a, b, ha, hb, he⟩
 144     refine ⟨⟨a, ha⟩, ?_⟩
 145     have hab : a ≫ (f.compat s k).choose = b := by
 146       apply (cancel_mono (G.F s k).arrow).mp
 147       rw [Category.assoc, (f.compat s k).choose_spec]
 148       exact he
 149     rw [← ha, Category.assoc, f.quotient_naturality,
 150       ← Category.assoc, hab, hb]
 151   · rintro ⟨⟨a, ha⟩, h⟩
 152     refine ⟨a, a ≫ (f.compat s k).choose, ha, ?_, ?_⟩
 153     · rw [Category.assoc, ← f.quotient_naturality, ← Category.assoc, ha, h]
 154     · rw [Category.assoc, (f.compat s k).choose_spec]

 156 /-- At length zero the representative equation is the induced graded map equation.
 157 The source lifting condition is retained for arbitrary test objects. -/
 158 public theorem relation_zero_iff (f : FilteredMorphism F G) (k : ω) (s : ℤ)
 159     {T : C} (x : T ⟶ F.associatedGraded s k)
 160     (y : T ⟶ G.associatedGraded (s + 0) k) :
 161     Relation f k s 0 x y ↔
 162       (∃ a : T ⟶ Subobject.underlying.obj (F.F s k),
 163         a ≫ F.toAssociatedGraded s k = x) ∧
 164       x ≫ f.inducedGrMap s k =
 165         y ≫ G.transportGraded (show (s + 0, k) = (s, k) by simp) := by
 166   unfold Relation
 167   rw [and_iff_right (show (0 : ℤ) ≤ 0 from le_rfl)]
 168   exact zero_lifts_iff f k s (s + 0) (add_zero s) x y

 170 /-- Projective test objects supply the lifting needed for the zero-length formula. -/
 171 public theorem relation_zero_iff_of_projective (f : FilteredMorphism F G)
 172     (k : ω) (s : ℤ) {T : C} [Projective T]
 173     (x : T ⟶ F.associatedGraded s k)
 174     (y : T ⟶ G.associatedGraded (s + 0) k) :
 175     Relation f k s 0 x y ↔
 176       x ≫ f.inducedGrMap s k = y ≫ G.transportGraded (show (s + 0, k) = (s, k) by simp) := by
 177   rw [relation_zero_iff]
 178   haveI : Epi (F.toAssociatedGraded s k) := inferInstanceAs (Epi (cokernel.π _))
 179   exact and_iff_right ⟨Projective.factorThru x (F.toAssociatedGraded s k),
 180     Projective.factorThru_comp x (F.toAssociatedGraded s k)⟩
