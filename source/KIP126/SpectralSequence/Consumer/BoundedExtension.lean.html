  13 /-! The bounded extension sequence of a specified filtered map. The filtered
  14 complex is an implementation detail; all public types use the original map,

  21 universe u v w
  22 variable {C : Type u} [Category.{v} C] [Abelian C]
  23   {ω : Type w} {A B : ω → C} {F : Filtration A} {G : Filtration B}

  25 /-- The canonical ESS; boundedness is explicit and applies to these filtrations. -/
  26 @[no_expose] public noncomputable def sequence (f : FilteredMorphism F G)
  27     (hF : F.IsBounded) (hG : G.IsBounded) (k : ω) :
  28     SpectralSequence C (ℤ × ℤ) :=
  29   (underlyingComplex f.map f.compat k).toSpectralSequence
  30     (underlyingComplexBounded f.map f.compat k hF hG)
  31 
  32 public theorem sequence_firstPage (f : FilteredMorphism F G)
  33     (hF : F.IsBounded) (hG : G.IsBounded) (k : ω) :
  34     (sequence f hF hG k).r₀ = 0 := by rfl
  35 
  36 public theorem sequence_diffDeg (f : FilteredMorphism F G)
  37     (hF : F.IsBounded) (hG : G.IsBounded) (k : ω) (n : ℤ) :
  38     (sequence f hF hG k).diffDeg n = (n, -1) := by rfl
  39 
  40 /-- Source ambient object is the associated grade of the same source filtration. -/
  41 @[no_expose] public noncomputable def sourceIso (f : FilteredMorphism F G)
  42     (hF : F.IsBounded) (hG : G.IsBounded) (k : ω) (s : ℤ) :
  43     ((sequence f hF hG k).ssData (s, 1)).V ≅ F.associatedGraded s k :=
  44   Iso.refl _
  45 
  46 /-- Target ambient object is the associated grade at the actual differential target. -/
  47 @[no_expose] public noncomputable def targetIso (f : FilteredMorphism F G)
  48     (hF : F.IsBounded) (hG : G.IsBounded) (k : ω) (s n : ℤ) :
  49     ((sequence f hF hG k).ssData ((s, 1) + (sequence f hF hG k).diffDeg n)).V ≅
  50       G.associatedGraded (s + n) k :=
  51   Iso.refl _
  52 
  53 theorem relation_iff_fiber (f : FilteredMorphism F G) (k : ω) (s n : ℤ)
  54     {T : C} (x : T ⟶ F.associatedGraded s k)
  55     (y : T ⟶ G.associatedGraded (s + n) k) :
  56     Relation f k s n x y ↔
  57       0 ≤ n ∧ Nonempty (FilteredComplex.Solutions.Fiber
  58         (underlyingComplex f.map f.compat k) n s 1 x y) := by
  59   constructor
  60   · rintro ⟨hn, a, b, ha, hb, he⟩
  61     refine ⟨hn, ⟨⟨(a, b), (FilteredComplex.Solutions.mem_fiber_iff _).mpr ?_⟩⟩⟩
  62     refine ⟨ha, hb, ?_⟩
  63     rw [underlyingComplex_d_one]
  64     exact he
  65   · rintro ⟨hn, ⟨⟨z, hz⟩⟩⟩
  66     obtain ⟨ha, hb, he⟩ := (FilteredComplex.Solutions.mem_fiber_iff z).mp hz
  67     refine ⟨hn, z.1, z.2, ?_⟩
  68     refine ⟨ha, hb, ?_⟩
  69     rw [underlyingComplex_d_one] at he
  70     exact he

  73 /-- The actual ESS differential relation is precisely the original filtered-map
  74 equation. Projectivity is retained when arbitrary morphism-valued labels are used. -/
  75 public theorem differentialRelation_iff (f : FilteredMorphism F G)
  76     (hF : F.IsBounded) (hG : G.IsBounded) (k : ω) (s : ℤ) (n : ℕ)
  77     {T : C} [Projective T] (x : T ⟶ F.associatedGraded s k)
  78     (y : T ⟶ G.associatedGraded (s + (n : ℤ)) k) :
  79     DifferentialRelation (sequence f hF hG k) n (s, 1)
  80       (x ≫ (sourceIso f hF hG k s).inv)
  81       (y ≫ (targetIso f hF hG k s n).inv) ↔ Relation f k s n x y := by
  82   rw [relation_iff_fiber, and_iff_right (Int.natCast_nonneg n)]
  83   change DifferentialRelation
  84     ((underlyingComplex f.map f.compat k).toSpectralSequence
  85       (underlyingComplexBounded f.map f.compat k hF hG)) n (s, 1)
  86     (x ≫ 𝟙 (F.associatedGraded s k)) (y ≫ 𝟙 (G.associatedGraded (s + ↑n) k)) ↔ _
  87   rw [Category.comp_id, Category.comp_id]
  88   exact
  89     FilteredComplex.Solutions.differentialRelation_iff_nonempty_fiber
  90       (underlyingComplex f.map f.compat k)
  91       (underlyingComplexBounded f.map f.compat k hF hG) n s 1 x y
