   7 /-! Transport for the actual filtered-map equation.  Associated graded maps,
   8 complete target ambiguity, and essentiality are derived from levelwise
   9 isomorphisms.  No boundedness is imposed on the representative equation. -/

  19 /-- An isomorphism of two specified filtered pieces, retaining actual
  20 representatives in their ambient objects. -/
  21 public structure FilteredPieceIso (F : Filtration A) (F' : Filtration A')
  22     (k : ω) (k' : ω') where
  23   ambient : A k ≅ A' k'
  24   level : ∀ s, Subobject.underlying.obj (F.F s k) ≅
  25     Subobject.underlying.obj (F'.F s k')
  26   level_arrow : ∀ s, (level s).hom ≫ (F'.F s k').arrow =
  27     (F.F s k).arrow ≫ ambient.hom
  28 
  29 namespace FilteredPieceIso
  30 variable {k : ω} {k' : ω'}

  32 public noncomputable def symm (e : FilteredPieceIso F F' k k') : FilteredPieceIso F' F k' k where
  33   ambient := e.ambient.symm
  34   level s := (e.level s).symm
  35   level_arrow s := by
  36     change (e.level s).inv ≫ (F.F s k).arrow =
  37       (F'.F s k').arrow ≫ e.ambient.inv
  38     apply (cancel_epi (e.level s).hom).mp
  39     simp only [← Category.assoc, Iso.hom_inv_id, Category.id_comp]
  40     rw [e.level_arrow]
  41     simp
  42 
  43 omit [Abelian C] in
  44 /-- Inclusion maps commute because both composites are the same actual
  45 subobject inclusion in the ambient object. -/
  46 public theorem inclusion_comm (e : FilteredPieceIso F F' k k') (s : ℤ) :
  47     Subobject.ofLE _ _ (F.mono s k) ≫ (e.level s).hom =
  48       (e.level (s+1)).hom ≫ Subobject.ofLE _ _ (F'.mono s k') := by
  49   apply (cancel_mono (F'.F s k').arrow).mp
  50   simp only [Category.assoc, e.level_arrow, Subobject.ofLE_arrow_assoc,
  51     Subobject.ofLE_arrow]
  52 
  53 public noncomputable def gradedIso (e : FilteredPieceIso F F' k k') (s : ℤ) :
  54     F.associatedGraded s k ≅ F'.associatedGraded s k' :=
  55   cokernel.mapIso (Subobject.ofLE _ _ (F.mono s k))
  56     (Subobject.ofLE _ _ (F'.mono s k')) (e.level (s+1)) (e.level s)
  57     (e.inclusion_comm s)
  58 
  59 public theorem projection_comm (e : FilteredPieceIso F F' k k') (s : ℤ) :
  60     F.toAssociatedGraded s k ≫ (e.gradedIso s).hom =
  61       (e.level s).hom ≫ F'.toAssociatedGraded s k' := by
  62   dsimp only [gradedIso, Filtration.toAssociatedGraded, cokernel.mapIso]
  63   exact cokernel.π_desc _ _ _
  64 
  65 public theorem projection_inv_comm (e : FilteredPieceIso F F' k k') (s : ℤ) :
  66     F'.toAssociatedGraded s k' ≫ (e.gradedIso s).inv =
  67       (e.level s).inv ≫ F.toAssociatedGraded s k := by
  68   apply (cancel_epi (e.level s).hom).mp
  69   rw [← Category.assoc, ← e.projection_comm]
  70   simp
  71 end FilteredPieceIso

  73 /-- A commuting square between the actual filtered maps, together with
  74 isomorphisms on every source and target filtration level. -/
  75 public structure FilteredMapTransport (f : FilteredMorphism F G)
  76     (f' : FilteredMorphism F' G') (k : ω) (k' : ω') where
  77   source : FilteredPieceIso F F' k k'
  78   target : FilteredPieceIso G G' k k'
  79   map_comm : source.ambient.hom ≫ f'.map k' = f.map k ≫ target.ambient.hom

  87 /-- The chosen representatives transport along the actual level isomorphisms. -/
  88 public theorem representative_equation_iff
  89     (a : T ⟶ Subobject.underlying.obj (F.F s k))
  90     (b : T ⟶ Subobject.underlying.obj (G.F (s+n) k)) :
  91     a ≫ (F.F s k).arrow ≫ f.map k = b ≫ (G.F (s+n) k).arrow ↔
  92       (a ≫ (e.source.level s).hom) ≫ (F'.F s k').arrow ≫ f'.map k' =
  93         (b ≫ (e.target.level (s+n)).hom) ≫ (G'.F (s+n) k').arrow := by
  94   have hsource := e.source.level_arrow s
  95   have htarget := e.target.level_arrow (s+n)
  96   simp only [Category.assoc]
  97   rw [← Category.assoc (e.source.level s).hom, hsource, Category.assoc, e.map_comm]
  98   rw [htarget]
  99   constructor
 100   · intro h
 101     simpa only [Category.assoc] using congrArg (fun z => z ≫ e.target.ambient.hom) h
 102   · intro h
 103     apply (cancel_mono e.target.ambient.hom).mp
 104     simpa only [Category.assoc] using h

 106 public theorem relation_iff
 107     (x : T ⟶ F.associatedGraded s k)
 108     (y : T ⟶ G.associatedGraded (s+n) k) :
 109     FilteredMapExtension.Relation f k s n x y ↔
 110       FilteredMapExtension.Relation f' k' s n
 111         (x ≫ (e.source.gradedIso s).hom) (y ≫ (e.target.gradedIso (s+n)).hom) := by
 112   constructor
 113   · rintro ⟨hn, a, b, ha, hb, he⟩
 114     refine ⟨hn, a ≫ (e.source.level s).hom,
 115       b ≫ (e.target.level (s+n)).hom, ?_, ?_,
 116       (e.representative_equation_iff a b).mp he⟩
 117     · rw [Category.assoc, ← e.source.projection_comm, ← Category.assoc, ha]
 118     · rw [Category.assoc, ← e.target.projection_comm, ← Category.assoc, hb]
 119   · rintro ⟨hn, a, b, ha, hb, he⟩
 120     refine ⟨hn, a ≫ (e.source.level s).inv,
 121       b ≫ (e.target.level (s+n)).inv, ?_, ?_, ?_⟩
 122     · rw [Category.assoc, ← e.source.projection_inv_comm, ← Category.assoc, ha]
 123       simp
 124     · rw [Category.assoc, ← e.target.projection_inv_comm, ← Category.assoc, hb]
 125       simp
 126     · apply (e.representative_equation_iff _ _).mpr
 127       simpa using he

 129 /-- Complete homogeneous ambiguity, rather than a selected correction subgroup. -/
 130 public theorem indeterminacy_iff (z : T ⟶ G.associatedGraded (s+n) k) :
 131     z ∈ FilteredMapExtension.TargetIndeterminacy f k s n T ↔
 132       z ≫ (e.target.gradedIso (s+n)).hom ∈
 133         FilteredMapExtension.TargetIndeterminacy f' k' s n T := by
 134   simpa only [FilteredMapExtension.TargetIndeterminacy, Set.mem_setOf_eq,
 135     zero_comp] using e.relation_iff (0 : T ⟶ F.associatedGraded s k) z
 136 
 137 public theorem targetCoset_iff (y z : T ⟶ G.associatedGraded (s+n) k) :
 138     z ∈ FilteredMapExtension.TargetCoset f k s n y ↔
 139       z ≫ (e.target.gradedIso (s+n)).hom ∈
 140         FilteredMapExtension.TargetCoset f' k' s n
 141           (y ≫ (e.target.gradedIso (s+n)).hom) := by
 142   simpa only [FilteredMapExtension.TargetCoset, Set.mem_setOf_eq,
 143     Preadditive.sub_comp] using e.indeterminacy_iff (z-y)

 145 public theorem essential_iff (x : T ⟶ F.associatedGraded s k)
 146     (y : T ⟶ G.associatedGraded (s+n) k) :
 147     FilteredMapExtension.Essential f k s n x y ↔
 148       FilteredMapExtension.Essential f' k' s n
 149         (x ≫ (e.source.gradedIso s).hom) (y ≫ (e.target.gradedIso (s+n)).hom) := by
 150   unfold FilteredMapExtension.Essential
 151   rw [e.relation_iff]
 152   have h := e.targetCoset_iff y (0 : T ⟶ G.associatedGraded (s+n) k)
 153   simpa only [zero_comp] using and_congr_right (fun _ => not_congr h)

 155 /-- An actual higher-source-filtration essential extension crossing the
 156 selected target level. Nonnegative lengths and the same test object remain explicit. -/
 157 public def NoCrossing (f : FilteredMorphism F G) (k : ω) (s n : ℤ) (T : C) : Prop :=
 158   ¬ ∃ (a m : ℤ), 0 < a ∧ 0 ≤ m ∧ a + m ≤ n ∧
 159     ∃ (x : T ⟶ F.associatedGraded (s+a) k)
 160       (y : T ⟶ G.associatedGraded ((s+a)+m) k),
 161       FilteredMapExtension.Essential f k (s+a) m x y
 162 
 163 include e in
 164 public theorem noCrossing_iff (s n : ℤ) (T : C) :
 165     NoCrossing f k s n T ↔ NoCrossing f' k' s n T := by
 166   apply not_congr
 167   constructor
 168   · rintro ⟨a, m, ha, hm, hmn, x, y, he⟩
 169     exact ⟨a, m, ha, hm, hmn, _, _, (e.essential_iff x y).mp he⟩
 170   · rintro ⟨a, m, ha, hm, hmn, x, y, he⟩
 171     refine ⟨a, m, ha, hm, hmn, x ≫ (e.source.gradedIso (s+a)).inv,
 172       y ≫ (e.target.gradedIso ((s+a)+m)).inv, ?_⟩
 173     apply (e.essential_iff _ _).mpr
 174     simpa using he

 176 /-- ESS differential transport is available when both actual filtrations
 177 are bounded and the test object is projective. General representative
 178 transport above does not silently assert those hypotheses. -/
 179 public theorem differentialRelation_iff
 180     (hF : F.IsBounded) (hG : G.IsBounded)
 181     (hF' : F'.IsBounded) (hG' : G'.IsBounded)
 182     (s : ℤ) (n : ℕ) [Projective T]
 183     (x : T ⟶ F.associatedGraded s k)
 184     (y : T ⟶ G.associatedGraded (s+(n:ℤ)) k) :
 185     DifferentialRelation (FilteredMapExtension.sequence f hF hG k) n (s,1)
 186       (x ≫ (FilteredMapExtension.sourceIso f hF hG k s).inv)
 187       (y ≫ (FilteredMapExtension.targetIso f hF hG k s n).inv) ↔
 188     DifferentialRelation (FilteredMapExtension.sequence f' hF' hG' k') n (s,1)
 189       (x ≫ (e.source.gradedIso s).hom ≫
 190         (FilteredMapExtension.sourceIso f' hF' hG' k' s).inv)
 191       (y ≫ (e.target.gradedIso (s+(n:ℤ))).hom ≫
 192         (FilteredMapExtension.targetIso f' hF' hG' k' s n).inv) := by
 193   rw [FilteredMapExtension.differentialRelation_iff]
 194   rw [← Category.assoc x, ← Category.assoc y,
 195     FilteredMapExtension.differentialRelation_iff]
 196   exact e.relation_iff x y
