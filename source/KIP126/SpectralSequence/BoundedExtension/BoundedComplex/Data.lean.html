  13 
  14 open CategoryTheory
  15 
  16 universe u v
  17 
  18 variable {C : Type u} [Category.{v} C] [Abelian C]
  19 
  20 /-- A filtered complex packaged with the historical two-sided boundedness
  21 witness. -/
  22 structure BoundedFilteredComplex
  23     (C : Type u) [Category.{v} C] [Abelian C] where
  24   /-- The underlying filtered complex. -/
  25   FC : FilteredComplex C
  26   /-- Its filtration is bounded in every chain degree. -/
  27   bnd : FC.IsBounded
  28 
  29 /-- Bounded filtered complexes form the full subcategory of filtered
  30 complexes on bounded objects. -/
  31 noncomputable instance : Category (BoundedFilteredComplex C) where
  32   Hom X Y := FilteredComplexMorphism X.FC Y.FC
  33   id X := FilteredComplex.Morphism.id X.FC
  34   comp f g := FilteredComplex.Morphism.comp f g
  35   id_comp f := by
  36     apply FilteredComplex.Morphism.ext
  37     simp [FilteredComplex.Morphism.comp, FilteredComplex.Morphism.id]
  38   comp_id f := by
  39     apply FilteredComplex.Morphism.ext
  40     simp [FilteredComplex.Morphism.comp, FilteredComplex.Morphism.id]
  41   assoc f g h := by
  42     apply FilteredComplex.Morphism.ext
  43     simp [FilteredComplex.Morphism.comp]
  44 
  45 end KIP126.Core.SpectralSequence
