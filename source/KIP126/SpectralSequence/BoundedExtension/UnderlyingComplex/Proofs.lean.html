  85           (underlyingComplex aMap hcompat t).differential_preserves s 1 |>.choose_spec

  87       (hcompat s t).choose_spec.symm

 142 lemma subobject_eq_bot_of_isZero {X : C} (hX : IsZero X)
 143     (P : Subobject X) : P = ⊥ := by
 144   have hP_zero : IsZero (Subobject.underlying.obj P) := hX.of_mono P.arrow
 145   apply le_antisymm
 146   · apply Subobject.le_of_comm (hP_zero.to_ _)
 147     apply hX.eq_of_tgt
 148   · exact bot_le

 153 /-- Bounded target filtrations induce a bounded filtration on the two-term complex. -/
 154 noncomputable def underlyingComplexBounded
 155     {A₁ A₂ : ω' → C} {F₁ : Filtration A₁} {F₂ : Filtration A₂}
 156     (aMap : ∀ k', A₁ k' ⟶ A₂ k')
 157     (hcompat : ∀ (s : ℤ) (k' : ω'),
 158       ∃ φ : Subobject.underlying.obj (F₁.F s k') ⟶
 159           Subobject.underlying.obj (F₂.F s k'),
 160         φ ≫ (F₂.F s k').arrow = (F₁.F s k').arrow ≫ aMap k')
 161     (t : ω') (bnd₁ : F₁.IsBounded) (bnd₂ : F₂.IsBounded) :
 162     (underlyingComplex aMap hcompat t).IsBounded where
 163   lo := fun k => if k = 1 then bnd₁.lo t else if k = 0 then bnd₂.lo t else 0
 164   hi := fun k => if k = 1 then bnd₁.hi t else if k = 0 then bnd₂.hi t else 0
 165   lo_le_hi := fun k => by
 166     by_cases h₁ : k = 1
 167     · simp only [h₁, ↓reduceIte]
 168       exact bnd₁.lo_le_hi t
 169     · by_cases h₀ : k = 0
 170       · simp only [h₀, ↓reduceIte]
 171         exact bnd₂.lo_le_hi t
 172       · simp [h₁, h₀]
 173   boundedBelow := fun k s hs => by
 174     simp only [FilteredComplex.fil]
 175     unfold underlyingComplex
 176     by_cases h₁ : k = 1
 177     · subst k
 178       simpa [twoTermFil, BoundedExtension.twoTermFil] using
 179         bnd₁.boundedBelow t s hs
 180     · by_cases h₀ : k = 0
 181       · subst k
 182         simpa [twoTermFil, BoundedExtension.twoTermFil] using
 183           bnd₂.boundedBelow t s hs
 184       · simp [twoTermFil, BoundedExtension.twoTermFil, h₁, h₀]
 185   boundedAbove := fun k s hs => by
 186     simp only [FilteredComplex.fil]
 187     unfold underlyingComplex
 188     by_cases h₁ : k = 1
 189     · subst k
 190       simpa [twoTermFil, BoundedExtension.twoTermFil] using
 191         bnd₁.boundedAbove t s hs
 192     · by_cases h₀ : k = 0
 193       · subst k
 194         simpa [twoTermFil, BoundedExtension.twoTermFil] using
 195           bnd₂.boundedAbove t s hs
 196       · simp only [if_neg h₁, if_neg h₀] at hs
 197         have hz : IsZero
 198             ((BoundedExtension.twoTermComplex (A₁ t) (A₂ t) (aMap t)).X k) := by
 199           rw [BoundedExtension.twoTermComplex_X]
 200           rw [BoundedExtension.twoTermObj_other (A₁ t) (A₂ t) k h₁ h₀]
 201           exact IsInitial.isZero initialIsInitial
 202         simp [twoTermFil, BoundedExtension.twoTermFil, h₁, h₀]
 203         exact subobject_eq_bot_of_isZero hz ⊤
