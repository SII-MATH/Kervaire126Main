  10 # The filtered two-term complex of a filtration-compatible map

  13 namespace KIP126.Core.SpectralSequence
  14 
  15 open CategoryTheory CategoryTheory.Limits
  16 
  17 universe u v w
  18 
  19 variable {C : Type u} [Category.{v} C] [Abelian C]
  20 variable {ω' : Type w}
  21 
  22 set_option backward.defeqAttrib.useBackward true in
  23 set_option backward.isDefEq.respectTransparency false in
  24 set_option maxHeartbeats 2000000 in
  25 /-- The two-term filtered complex `A₁(t) ⟶ A₂(t)` at a fixed stem. -/
  26 noncomputable def underlyingComplex
  27     {A₁ A₂ : ω' → C} {F₁ : Filtration A₁} {F₂ : Filtration A₂}
  28     (aMap : ∀ k', A₁ k' ⟶ A₂ k')
  29     (hcompat : ∀ (s : ℤ) (k' : ω'),
  30       ∃ φ : Subobject.underlying.obj (F₁.F s k') ⟶
  31           Subobject.underlying.obj (F₂.F s k'),
  32         φ ≫ (F₂.F s k').arrow = (F₁.F s k').arrow ≫ aMap k')
  33     (t : ω') : FilteredComplex C where
  34   complex := BoundedExtension.twoTermComplex (A₁ t) (A₂ t) (aMap t)
  35   filtration := {
  36     F := twoTermFil (fun s => F₁.F s t) (fun s => F₂.F s t)
  37     mono := fun s k => by
  38       by_cases h₁ : k = 1
  39       · subst k
  40         simpa [BoundedExtension.twoTermComplex, BoundedExtension.twoTermObj,
  41           twoTermFil, BoundedExtension.twoTermFil] using F₁.mono s t
  42       · by_cases h₀ : k = 0
  43         · subst k
  44           simpa [BoundedExtension.twoTermComplex, BoundedExtension.twoTermObj,
  45             twoTermFil, BoundedExtension.twoTermFil] using F₂.mono s t
  46         · simp [twoTermFil, BoundedExtension.twoTermFil, h₁, h₀] }
  47   differential_preserves := fun s k => by
  48     by_cases h : k = 1
  49     · subst k
  50       refine ⟨(hcompat s t).choose, ?_⟩
  51       simpa [BoundedExtension.twoTermDiff, BoundedExtension.twoTermFil,
  52         twoTermFil] using (hcompat s t).choose_spec
  53     · refine ⟨0, ?_⟩
  54       simp [BoundedExtension.twoTermDiff, BoundedExtension.twoTermFil,
  55         twoTermFil, h]
  56 
  57 end KIP126.Core.SpectralSequence
