  22 /-- The spectral sequence of the filtered two-term complex. -/
  23 noncomputable def BoundedExtensionSS.ess
  24     {E₁ E₂ : SpectralSequence C ω}
  25     {A₁ A₂ : ω' → C} {F₁ : Filtration A₁} {F₂ : Filtration A₂}
  26     {conv₁ : Convergence E₁ A₁ F₁} {conv₂ : Convergence E₂ A₂ F₂}
  27     {cm : ConvergenceMorphism conv₁ conv₂}
  28     {bnd₁ : F₁.IsBounded} {bnd₂ : F₂.IsBounded}
  29     (ext : BoundedExtensionSS conv₁ conv₂ cm bnd₁ bnd₂) (t : ω') :
  30     SpectralSequence C (ℤ × ℤ) :=
  31   (ext.complex t).toSpectralSequence (ext.bounded t)
  32 
  33 /-- Weak convergence of the extension spectral sequence to the historical
  34 homology presentation of its two-term complex. -/
  35 noncomputable def BoundedExtensionSS.weakConvergence
  36     {E₁ E₂ : SpectralSequence C ω}
  37     {A₁ A₂ : ω' → C} {F₁ : Filtration A₁} {F₂ : Filtration A₂}
  38     {conv₁ : Convergence E₁ A₁ F₁} {conv₂ : Convergence E₂ A₂ F₂}
  39     {cm : ConvergenceMorphism conv₁ conv₂}
  40     {bnd₁ : F₁.IsBounded} {bnd₂ : F₂.IsBounded}
  41     (ext : BoundedExtensionSS conv₁ conv₂ cm bnd₁ bnd₂) (t : ω') :
  42     Convergence (ext.ess t)
  43       (ext.complex t).homologySSObj (ext.complex t).homologySSFiltration :=
  44   (ext.complex t).weakConvergence (ext.bounded t)
  45 
  46 /-- The `E₀` object in chain degree `1`. -/
  47 noncomputable def BoundedExtensionSS.e0PageAtOne
  48     {E₁ E₂ : SpectralSequence C ω}
  49     {A₁ A₂ : ω' → C} {F₁ : Filtration A₁} {F₂ : Filtration A₂}
  50     {conv₁ : Convergence E₁ A₁ F₁} {conv₂ : Convergence E₂ A₂ F₂}
  51     {cm : ConvergenceMorphism conv₁ conv₂}
  52     {bnd₁ : F₁.IsBounded} {bnd₂ : F₂.IsBounded}
  53     (ext : BoundedExtensionSS conv₁ conv₂ cm bnd₁ bnd₂)
  54     (t : ω') (s : ℤ) : C :=
  55   (ext.complex t).assocGraded s 1
  56 
  57 /-- The `E₀` object in chain degree `0`. -/
  58 noncomputable def BoundedExtensionSS.e0PageAtZero
  59     {E₁ E₂ : SpectralSequence C ω}
  60     {A₁ A₂ : ω' → C} {F₁ : Filtration A₁} {F₂ : Filtration A₂}
  61     {conv₁ : Convergence E₁ A₁ F₁} {conv₂ : Convergence E₂ A₂ F₂}
  62     {cm : ConvergenceMorphism conv₁ conv₂}
  63     {bnd₁ : F₁.IsBounded} {bnd₂ : F₂.IsBounded}
  64     (ext : BoundedExtensionSS conv₁ conv₂ cm bnd₁ bnd₂)
  65     (t : ω') (s : ℤ) : C :=
  66   (ext.complex t).assocGraded s 0
  67 
  68 /-- The page-zero differential from chain degree `1` to chain degree `0`. -/
  69 noncomputable def BoundedExtensionSS.d0
  70     {E₁ E₂ : SpectralSequence C ω}
  71     {A₁ A₂ : ω' → C} {F₁ : Filtration A₁} {F₂ : Filtration A₂}
  72     {conv₁ : Convergence E₁ A₁ F₁} {conv₂ : Convergence E₂ A₂ F₂}
  73     {cm : ConvergenceMorphism conv₁ conv₂}
  74     {bnd₁ : F₁.IsBounded} {bnd₂ : F₂.IsBounded}
  75     (ext : BoundedExtensionSS conv₁ conv₂ cm bnd₁ bnd₂)
  76     (t : ω') (s : ℤ) :
  77     ext.e0PageAtOne t s ⟶ ext.e0PageAtZero t s :=
  78   (ext.complex t).assocGradedDiff s 1
