  51 /-- The page-zero differential is the associated-graded map induced by the
  52 target map of the convergence morphism. -/
  53 theorem BoundedExtensionSS.d0_eq_inducedAssocGradedMap
  54     {E₁ E₂ : SpectralSequence C ω}
  55     {A₁ A₂ : ω' → C} {F₁ : Filtration A₁} {F₂ : Filtration A₂}
  56     {conv₁ : Convergence E₁ A₁ F₁} {conv₂ : Convergence E₂ A₂ F₂}
  57     {cm : ConvergenceMorphism conv₁ conv₂}
  58     {bnd₁ : F₁.IsBounded} {bnd₂ : F₂.IsBounded}
  59     (ext : BoundedExtensionSS conv₁ conv₂ cm bnd₁ bnd₂)
  60     (t : ω') (s : ℤ) :
  61     ext.d0 t s =
  62       eqToHom (ext.e0PageAtOne_eq t s) ≫
  63         Filtration.inducedAssocGradedMap cm.aMap cm.filtration_compat s t ≫
  64         eqToHom (ext.e0PageAtZero_eq t s).symm := by
  65   have hOne : eqToHom (ext.e0PageAtOne_eq t s) = 𝟙 _ := by rfl
  66   have hZero : eqToHom (ext.e0PageAtZero_eq t s).symm = 𝟙 _ := by rfl
  67   rw [hOne, hZero, Category.id_comp, Category.comp_id]
  68   change (underlyingComplex cm.aMap cm.filtration_compat t).assocGradedDiff s 1 =
  69     Filtration.inducedAssocGradedMap cm.aMap cm.filtration_compat s t
  70   letI : Epi (F₁.toAssociatedGraded s t) := by
  71     unfold Filtration.toAssociatedGraded
  72     infer_instance
  73   apply (cancel_epi (F₁.toAssociatedGraded s t)).mp
  74   rw [underlyingComplex_assocGradedDiff_compat_one]
  75   unfold Filtration.inducedAssocGradedMap cokernel.map
  76   simp only [Filtration.toAssociatedGraded, cokernel.π_desc]
