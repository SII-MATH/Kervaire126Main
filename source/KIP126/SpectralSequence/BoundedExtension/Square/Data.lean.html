   8 open CategoryTheory
   9 universe u v
  10 variable {C : Type u} [Category.{v} C] [Abelian C]
  11 
  12 /-- The degreewise map of two-term complexes determined by a square. -/
  13 noncomputable def twoTermMap {X₁ X₂ Y₁ Y₂ : C}
  14     (a : X₁ ⟶ Y₁) (b : X₂ ⟶ Y₂) (k : ℤ) :
  15     twoTermObj X₁ X₂ k ⟶ twoTermObj Y₁ Y₂ k :=
  16   if h : k = 1 then
  17     eqToHom (by simp [twoTermObj, h]) ≫ a ≫ eqToHom (by simp [twoTermObj, h])
  18   else if h : k = 0 then
  19     eqToHom (by simp [twoTermObj, h]) ≫ b ≫ eqToHom (by simp [twoTermObj, h])
  20   else 0
  21 
  22 end KIP126.Core.SpectralSequence.BoundedExtension
