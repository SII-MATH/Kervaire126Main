  19 /-- A commuting square of actual filtered morphisms induces its actual
  20 two-term filtered chain map. No map of pages is chosen independently. -/
  21 noncomputable def underlyingComplexMap
  22     (f : KIP126.Core.SpectralSequence.FilteredMorphism F₁ F₂)
  23     (g : KIP126.Core.SpectralSequence.FilteredMorphism G₁ G₂)
  24     (a : KIP126.Core.SpectralSequence.FilteredMorphism F₁ G₁)
  25     (b : KIP126.Core.SpectralSequence.FilteredMorphism F₂ G₂)
  26     (h : ∀ i, a.map i ≫ g.map i = f.map i ≫ b.map i) (i : ι) :
  27     FilteredComplex.Morphism (underlyingComplex f.map f.compat i)
  28       (underlyingComplex g.map g.compat i) :=
  29   FilteredComplexMorphism.mk (BoundedExtension.twoTermMap (a.map i) (b.map i))
  30     (fun k => by
  31       simpa only [FilteredComplex.d, underlyingComplex,
  32         BoundedExtension.twoTermComplex_d] using
  33           BoundedExtension.twoTermMap_comm (f.map i) (g.map i) (a.map i) (b.map i) (h i) k)
  34     (fun s k => by
  35       by_cases hk : k = 1
  36       · subst k
  37         refine ⟨(a.compat s i).choose, ?_⟩
  38         simpa [underlyingComplex, FilteredComplex.fil, FilteredComplex.A,
  39           twoTermFil, BoundedExtension.twoTermFil, BoundedExtension.twoTermObj,
  40           BoundedExtension.twoTermMap] using (a.compat s i).choose_spec
  41       · by_cases hk₀ : k = 0
  42         · subst k
  43           refine ⟨(b.compat s i).choose, ?_⟩
  44           simpa [underlyingComplex, FilteredComplex.fil, FilteredComplex.A,
  45             twoTermFil, BoundedExtension.twoTermFil, BoundedExtension.twoTermObj,
  46             BoundedExtension.twoTermMap] using (b.compat s i).choose_spec
  47         · refine ⟨0, ?_⟩
  48           simp [underlyingComplex, BoundedExtension.twoTermMap, hk, hk₀])
  49 
  50 end KIP126.Core.SpectralSequence
