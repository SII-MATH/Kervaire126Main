   8 open CategoryTheory
   9 universe u v
  10 variable {C : Type u} [Category.{v} C] [Abelian C]
  11 
  12 theorem twoTermMap_comm {X₁ X₂ Y₁ Y₂ : C}
  13     (f : X₁ ⟶ X₂) (g : Y₁ ⟶ Y₂) (a : X₁ ⟶ Y₁) (b : X₂ ⟶ Y₂)
  14     (h : a ≫ g = f ≫ b) (k : ℤ) :
  15     twoTermMap a b k ≫ twoTermDiff Y₁ Y₂ g k =
  16       twoTermDiff X₁ X₂ f k ≫ twoTermMap a b (k - 1) := by
  17   by_cases hk : k = 1
  18   · subst k
  19     simp [twoTermMap, twoTermDiff]
  20     rw [← Category.assoc a g, h, Category.assoc]
  21   · simp [twoTermDiff, hk]
  22 
  23 end KIP126.Core.SpectralSequence.BoundedExtension
