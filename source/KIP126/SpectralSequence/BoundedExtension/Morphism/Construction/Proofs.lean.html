  21 /-- The compatibility constructor exposes the supplied degreewise maps. -/
  22 @[simp]
  23 theorem mk_f {FC₁ FC₂ : FilteredComplex C}
  24     (f : ∀ k, FC₁.A k ⟶ FC₂.A k)
  25     (comm_d : ∀ k, f k ≫ FC₂.d k = FC₁.d k ≫ f (k - 1))
  26     (filt_compat : ∀ s k,
  27       ∃ φ : Subobject.underlying.obj (FC₁.fil s k) ⟶
  28           Subobject.underlying.obj (FC₂.fil s k),
  29         φ ≫ (FC₂.fil s k).arrow = (FC₁.fil s k).arrow ≫ f k)
  30     (k : ℤ) :
  31     (FilteredComplexMorphism.mk f comm_d filt_compat).f k = f k := by
  32   rfl
  33 
  34 end FilteredComplexMorphism
