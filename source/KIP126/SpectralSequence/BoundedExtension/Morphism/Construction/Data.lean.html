  15 universe u v

  17 variable {C : Type u} [Category.{v} C] [Abelian C]
  18 
  19 namespace FilteredComplexMorphism
  20 
  21 /-- Constructor with the argument order and one-index differential convention
  22 of the historical API. -/
  23 noncomputable def mk {FC₁ FC₂ : FilteredComplex C}
  24     (f : ∀ k, FC₁.A k ⟶ FC₂.A k)
  25     (comm_d : ∀ k, f k ≫ FC₂.d k = FC₁.d k ≫ f (k - 1))
  26     (filt_compat : ∀ s k,
  27       ∃ φ : Subobject.underlying.obj (FC₁.fil s k) ⟶
  28           Subobject.underlying.obj (FC₂.fil s k),
  29         φ ≫ (FC₂.fil s k).arrow = (FC₁.fil s k).arrow ≫ f k) :
  30     FilteredComplexMorphism FC₁ FC₂ where
  31   map := ChainComplex.ofHom f (fun k =>
  32     comm_d_transport f comm_d (by omega : k + 1 - 1 = k))
  33   preserves := filt_compat
  34 
  35 end FilteredComplexMorphism
  36 
  37 end KIP126.Core.SpectralSequence
