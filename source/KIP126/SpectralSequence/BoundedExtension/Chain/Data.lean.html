  18 universe u v w

  20 variable {C : Type u} [Category.{v} C] [Abelian C]
  21 variable {ω' : Type w}
  22 variable {ω : Type w} [AddCommGroup ω] [DecidableEq ω]
  23 
  24 /-- A composable pair of convergence morphisms `V₁ ⟶ V₂ ⟶ V₃`, together
  25 with boundedness of all three target filtrations. -/
  26 structure ThreeSpectraChain
  27     {E₁ E₂ E₃ : SpectralSequence C ω}
  28     {A₁ A₂ A₃ : ω' → C}
  29     {F₁ : Filtration A₁} {F₂ : Filtration A₂} {F₃ : Filtration A₃}
  30     (conv₁ : Convergence E₁ A₁ F₁)
  31     (conv₂ : Convergence E₂ A₂ F₂)
  32     (conv₃ : Convergence E₃ A₃ F₃) where
  33   /-- The first convergence morphism. -/
  34   cm₁₂ : ConvergenceMorphism conv₁ conv₂
  35   /-- The second convergence morphism. -/
  36   cm₂₃ : ConvergenceMorphism conv₂ conv₃
  37   /-- Boundedness of the first target filtration. -/
  38   bnd₁ : F₁.IsBounded
  39   /-- Boundedness of the second target filtration. -/
  40   bnd₂ : F₂.IsBounded
  41   /-- Boundedness of the third target filtration. -/
  42   bnd₃ : F₃.IsBounded
  43 
  44 end KIP126.Core.SpectralSequence
