  15 variable {C : Type u} [StableHomotopyCategory.{u, v} C]

  17   {Syn : Type w} [SyntheticCategory.{w, v} Syn]

  22 public def extensionSourceObject : SyntheticObject :=
  23   .shift (0, (normalizedExponent D.classicalTower f : ℤ)) (.nu X)
  24 public def finiteExtensionMap (q : ℕ) := XModLambdaN.map (D.normalizedMap X Y f).map q

  26 /-- Labels for any actual shifted ν object in a finite λ quotient. -/
  27 public def finiteNuLabel (X : ClassicalObject) (q : ℕ) (a s t : ℤ) (k : ℕ)
  28     (x : (D.classical.obj (X.obj D.auxiliary)).Page 2 (s,t)) :
  29     (D.family.obj ((SyntheticObject.quotient q (.shift (0, a) (.nu X))).obj
  30       D.nu D.auxiliary)).E₂ (s, t, t + a - k) :=
  31   D.family.pageMap (XModLambdaN.incl _ q) 2 (s, t, t + a - k)
  32     (D.comparison.nuE2 X a s t k x)
  33 
  34 /-- The unshifted target label uses the specified zero-shift isomorphism. -/
  35 public def targetNuLabel (Y : ClassicalObject) (s t : ℤ) (k : ℕ)
  36     (y : (D.classical.obj (Y.obj D.auxiliary)).Page 2 (s,t)) :
  37     (D.family.obj (D.nu.functor.obj (Y.obj D.auxiliary))).E₂ (s, t, t - k) :=
  38   D.family.pageMap (SyntheticCategory.biShift_zero.hom.app _) 2 (s, t, t - k)
  39     (by simpa [SyntheticAdamsSS.E₂] using D.comparison.nuE2 Y 0 s t k y)
  40 
  41 public def finiteTargetLabel (Y : ClassicalObject) (q : ℕ) (s t : ℤ) (k : ℕ)
  42     (y : (D.classical.obj (Y.obj D.auxiliary)).Page 2 (s,t)) :
  43     (D.family.obj ((SyntheticObject.quotient q (.nu Y)).obj D.nu D.auxiliary)).E₂
  44       (s, t, t - k) :=
  45   D.family.pageMap (XModLambdaN.incl _ q) 2 (s, t, t - k)
  46     (targetNuLabel D Y s t k y)
  47 
  48 /-- Only transports a proved degree equality. -/
  49 public def regradeE2 {A : SyntheticAdamsSS.{v}} {i j : Tridegree} (h : i = j)
  50     (x : A.E₂ i) : A.E₂ j := (eqToHom (congrArg A.E₂ h)) x
  51 
  52 public def finiteSourceForExtension (q : ℕ) (s t : ℤ) (x : (D.classical.obj (X.obj D.auxiliary)).Page 2 (s,t)) :
  53     (D.family.obj ((SyntheticObject.quotient q (extensionSourceObject D X Y f)).obj
  54       D.nu D.auxiliary)).E₂ (s, (t - s) + s, t + normalizedExponent D.classicalTower f) :=
  55   regradeE2 (by simp)
  56     (finiteNuLabel D X q (normalizedExponent D.classicalTower f) s t 0 x)
  57 
  58 public def finiteTargetForExtension (q : ℕ) (n s t : ℤ)
  59     (hn : (normalizedExponent D.classicalTower f : ℤ) ≤ n)
  60     (y : (D.classical.obj (Y.obj D.auxiliary)).Page 2 (s+n,t+n)) :
  61     (D.family.obj ((SyntheticObject.quotient q (.nu Y)).obj D.nu D.auxiliary)).E₂
  62       (s + n, (t - s) + (s + n), t + normalizedExponent D.classicalTower f) :=
  63   regradeE2 (by
  64     ext <;> simp only [Int.toNat_of_nonneg (sub_nonneg.mpr hn)] <;> omega)
  65     (finiteTargetLabel D Y q (s + n) (t + n) (n - normalizedExponent D.classicalTower f).toNat y)
  66 
  67 public def infiniteSourceForExtension (s t : ℤ) (x : (D.classical.obj (X.obj D.auxiliary)).Page 2 (s,t)) :
  68     (D.family.obj ((extensionSourceObject D X Y f).obj D.nu D.auxiliary)).E₂
  69       (s, (t - s) + s, t + normalizedExponent D.classicalTower f) :=
  70   regradeE2 (by simp) (D.comparison.nuE2 X (normalizedExponent D.classicalTower f) s t 0 x)
  71 
  72 public def infiniteTargetForExtension (n s t : ℤ)
  73     (hn : (normalizedExponent D.classicalTower f : ℤ) ≤ n)
  74     (y : (D.classical.obj (Y.obj D.auxiliary)).Page 2 (s+n,t+n)) :
  75     (D.family.obj (D.nu.functor.obj (Y.obj D.auxiliary))).E₂
  76       (s + n, (t - s) + (s + n), t + normalizedExponent D.classicalTower f) :=
  77   regradeE2 (by
  78     ext <;> simp only [Int.toNat_of_nonneg (sub_nonneg.mpr hn)] <;> omega)
  79     (targetNuLabel D Y (s + n) (t + n) (n - normalizedExponent D.classicalTower f).toNat y)
  80 end
