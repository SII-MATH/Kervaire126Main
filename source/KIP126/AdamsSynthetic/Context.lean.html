  26 /-- The structural part of a stable homotopy category.
  27 
  28 The fields are bundled so a concrete model can supply its additive, shifted,
  29 monoidal, and triangulated structures together.  In particular, this class
  30 does not assert the extra closedness, cofiber choices, or sign conventions
  31 used by particular constructions; those belong to separate witness records.
  32 -/
  33 public class StableHomotopyCategory (C : Type u) extends
  34     Category.{v} C,
  35     Preadditive C,
  36     HasShift C ℤ,
  37     MonoidalCategory C where
  38   /-- Every shift functor is additive. -/
  39   shiftAdditive : ∀ (n : ℤ), (shiftFunctor C n).Additive
  40   /-- The category has a zero object. -/
  41   hasZeroObject : HasZeroObject C
  42   /-- The category is pretriangulated. -/
  43   pretriangulated : Pretriangulated C
  44 
  45 attribute [instance] StableHomotopyCategory.shiftAdditive
  46 attribute [instance] StableHomotopyCategory.hasZeroObject
  47 attribute [reducible] StableHomotopyCategory.pretriangulated
  48 attribute [instance] StableHomotopyCategory.pretriangulated
  49

  52 /-! ### Explicit cofiber and tensor-triangulated witnesses -/
  53 
  54 /-- A chosen functorial cofiber construction on a stable category.
  55 
  56 This is data supplied by a concrete model.  Keeping it as a class preserves
  57 the convenient Mathlib notation without introducing a global postulate that
  58 would silently choose a cofiber for every map.
  59 -/
  60 public class HasFunctorialCofiber where
  61   cofib : {X Y : C} → (X ⟶ Y) → C
  62   cofibι : {X Y : C} → (f : X ⟶ Y) → Y ⟶ cofib f
  63   cofibδ : {X Y : C} → (f : X ⟶ Y) → cofib f ⟶ (shiftFunctor C (1 : ℤ)).obj X
  64   cofib_distinguished : ∀ {X Y : C} (f : X ⟶ Y),
  65     Triangle.mk f (cofibι f) (cofibδ f) ∈ distTriang C
  66   cofibMap : ∀ {X₁ Y₁ X₂ Y₂ : C} (f₁ : X₁ ⟶ Y₁) (f₂ : X₂ ⟶ Y₂)
  67     (α : X₁ ⟶ X₂) (β : Y₁ ⟶ Y₂), α ≫ f₂ = f₁ ≫ β →
  68     (cofib f₁ ⟶ cofib f₂)
  69   cofibMap_ι : ∀ {X₁ Y₁ X₂ Y₂ : C} (f₁ : X₁ ⟶ Y₁) (f₂ : X₂ ⟶ Y₂)
  70     (α : X₁ ⟶ X₂) (β : Y₁ ⟶ Y₂) (h : α ≫ f₂ = f₁ ≫ β),
  71     β ≫ cofibι f₂ = cofibι f₁ ≫ cofibMap f₁ f₂ α β h
  72   cofibMap_δ : ∀ {X₁ Y₁ X₂ Y₂ : C} (f₁ : X₁ ⟶ Y₁) (f₂ : X₂ ⟶ Y₂)
  73     (α : X₁ ⟶ X₂) (β : Y₁ ⟶ Y₂) (h : α ≫ f₂ = f₁ ≫ β),
  74     cofibMap f₁ f₂ α β h ≫ cofibδ f₂ =
  75       cofibδ f₁ ≫ (shiftFunctor C (1 : ℤ)).map α
  76 
  77 /-- The sphere spectrum is the monoidal unit. -/
  78 public abbrev SphereSpectrum : C := 𝟙_ C
  79 
  80 /-- The `n`-sphere is the `n`-fold shift of the sphere spectrum. -/
  81 public abbrev Sphere (n : ℤ) : C := (shiftFunctor C n).obj SphereSpectrum
  82 
  83 /-- Stable homotopy classes represented by maps from the `n`-sphere. -/
  84 public abbrev HomotopyGroup (n : ℤ) (X : C) : Type v := Sphere n ⟶ X
  85 
  86 /-- The selected mod-two coefficient object and unit on this same background.
  87 The record specifies its homotopy groups; it does not construct a spectrum. -/
  88 public structure ModTwoCoefficients (C : Type u) [StableHomotopyCategory.{u, v} C] where
  89   object : C
  90   pi0Equiv : HomotopyGroup 0 object ≃+ ZMod 2
  91   homotopy_vanishes : ∀ n : ℤ, n ≠ 0 → Subsingleton (HomotopyGroup n object)
  92 
  93 /-- The unit is the element `1` in the same specified zeroth homotopy group. -/
  94 public def ModTwoCoefficients.unit (H : ModTwoCoefficients C) :
  95     SphereSpectrum ⟶ H.object :=
  96   (shiftFunctorZero C ℤ).inv.app SphereSpectrum ≫ H.pi0Equiv.symm 1
  97 
  98 /-- The graded homotopy groups of a spectrum, as additive-group objects. -/
  99 public noncomputable def HomotopyGroups (X : C) : ℤ → AddCommGrpCat.{v} :=
 100   fun n => AddCommGrpCat.mk (HomotopyGroup n X)
 101

 104 /-- A chosen distinguished triangle `X → Y → Z → ΣX`. -/
 105 public structure HoCofiberSequence where
 106   X : C
 107   Y : C
 108   Z : C
 109   f : X ⟶ Y
 110   g : Y ⟶ Z
 111   h : Z ⟶ (shiftFunctor C (1 : ℤ)).obj X
 112   distinguished : Triangle.mk f g h ∈ distTriang C
 113 
 114 /-- The map on homotopy groups induced by postcomposition with `f`. -/
 115 public def inducedMap {X Y : C} (f : X ⟶ Y) (n : ℤ) :
 116     HomotopyGroup n X →+ HomotopyGroup n Y where
 117   toFun := fun α => α ≫ f
 118   map_zero' := Limits.zero_comp
 119   map_add' := fun a b => Preadditive.add_comp _ _ _ a b f

 121 end KIP126.StableHomotopy

 132 /-- A stable category with a coherent bigraded suspension and deformation map.
 133 
 134 The parent class supplies the additive, shifted, monoidal, and triangulated
 135 structure.  `biShift` records the extra synthetic grading, while `lam` is the
 136 natural transformation
 137 `Σ^(0,-1) ⟶ Id` used to define the λ-power quotients.
 138 -/
 139 public class SyntheticCategory (Syn : Type u) extends
 140     StableHomotopyCategory.{u, v} Syn where
 141   biShift : ℤ × ℤ → Syn ⥤ Syn
 142   biShift_comp : ∀ (a b : ℤ × ℤ),
 143     biShift a ⋙ biShift b ≅ biShift (a + b)
 144   biShift_zero : biShift (0, 0) ≅ 𝟭 Syn
 145   biShift_compat : ∀ n : ℤ, biShift (n, 0) ≅ shiftFunctor Syn n
 146   /-- Each bigraded suspension is fully faithful (the inverse suspension is
 147       supplied by the same coherence data in a concrete model). -/
 148   biShift_fullyFaithful : ∀ p : ℤ × ℤ, (biShift p).FullyFaithful
 149   lam : biShift (0, -1) ⟶ 𝟭 Syn
 150   biShift_tensor_comm : ∀ (p : ℤ × ℤ) (X Y : Syn),
 151     (biShift p).obj (MonoidalCategory.tensorObj X Y) ≅
 152       MonoidalCategory.tensorObj ((biShift p).obj X) Y
 153 
 154 variable {Syn : Type u} [SyntheticCategory.{u, v} Syn]
 155 
 156 /-- The `n`-fold composite of the deformation map `λ` on `X`.
 157 
 158 The recursive definition follows the coherence isomorphism supplied by
 159 `biShift_comp`; in particular `lambdaPow 0 X` is the chosen identity
 160 isomorphism from `Σ^(0,0)X` to `X`.
 161 -/
 162 public noncomputable def lambdaPow : (n : ℕ) → (X : Syn) →
 163     (SyntheticCategory.biShift (0, -(n : ℤ))).obj X ⟶ X
 164   | 0, X => SyntheticCategory.biShift_zero.hom.app X
 165   | n + 1, X => by
 166       have step1 :
 167           (SyntheticCategory.biShift ((0 : ℤ), -1) ⋙
 168             SyntheticCategory.biShift ((0 : ℤ), -(n : ℤ))).obj X ⟶ X :=
 169         (SyntheticCategory.biShift ((0 : ℤ), -(n : ℤ))).map
 170             (SyntheticCategory.lam.app X) ≫ lambdaPow n X
 171       have step2 :
 172           (SyntheticCategory.biShift ((0, -1) + (0, -(n : ℤ)))).obj X ⟶ X :=
 173         (SyntheticCategory.biShift_comp (0, -1) (0, -(n : ℤ))).inv.app X ≫ step1
 174       have heq :
 175           ((0 : ℤ), (-1 : ℤ)) + ((0 : ℤ), -(n : ℤ)) =
 176             ((0 : ℤ), -(↑(n + 1) : ℤ)) := by
 177         simp
 178       exact heq ▸ step2
 179 
 180 /-- The chosen cofiber of `λ_X`. -/
 181 public noncomputable def XModLambda (X : Syn) [HasFunctorialCofiber (C := Syn)] : Syn :=
 182   HasFunctorialCofiber.cofib (SyntheticCategory.lam.app X)
 183 
 184 /-- The chosen cofiber of `λⁿ_X`. -/
 185 public noncomputable def XModLambdaN (X : Syn) (n : ℕ)
 186     [HasFunctorialCofiber (C := Syn)] : Syn :=
 187   HasFunctorialCofiber.cofib (lambdaPow n X)

 189 /-- The explicit ν-interface from a stable category into the synthetic one.
 190 
 191 The compatibility fields are hypotheses supplied by a concrete construction;
 192 they are deliberately stored in a record instead of declared as axioms.
 193 -/
 194 public structure NuFunctorData (Stable : Type u) [StableHomotopyCategory.{u, v} Stable]
 195     (Syn : Type u') [SyntheticCategory.{u', v'} Syn] where
 196   functor : Stable ⥤ Syn
 197   additive : functor.Additive
 198   zeroIso : functor.obj 0 ≅ (0 : Syn)
 199   /-- The synthetic analogue of the ordinary sphere is the synthetic unit.
 200   This is explicit model data, not a claim that ν is strong monoidal on all
 201   spectra. Source: Pstrągowski, Definition 4.6 and its following discussion
 202   (`Source/Pst/source/synthetic_spectra.tex`, lines 1935–1938). -/
 203   unitIso : functor.obj (SphereSpectrum (C := Stable)) ≅ SphereSpectrum (C := Syn)
 204   suspensionIso : ∀ X : Stable,
 205     functor.obj ((shiftFunctor Stable (1 : ℤ)).obj X) ≅
 206       (SyntheticCategory.biShift (1, 1)).obj (functor.obj X)
 207 
 208 /-- The landing of a normalized synthetic connecting map:
 209 `Σ^(0,-1) ν(ΣX) ≅ Σ^(1,0) νX ≅ (νX)[1]`.
 210 Every comparison is supplied by the same ν datum and synthetic grading. -/
 211 public noncomputable def NuFunctorData.boundaryLandingIso
 212     {Stable : Type u} [StableHomotopyCategory.{u, v} Stable]
 213     {Syn : Type u'} [SyntheticCategory.{u', v'} Syn]
 214     (N : NuFunctorData Stable Syn) (X : Stable) :
 215     (SyntheticCategory.biShift (0, -1)).obj
 216         (N.functor.obj ((shiftFunctor Stable (1 : ℤ)).obj X)) ≅
 217       (shiftFunctor Syn (1 : ℤ)).obj (N.functor.obj X) :=
 218   (SyntheticCategory.biShift (0, -1)).mapIso (N.suspensionIso X) ≪≫
 219     (SyntheticCategory.biShift_comp (1, 1) (0, -1)).app (N.functor.obj X) ≪≫
 220     (SyntheticCategory.biShift_compat (Syn := Syn) 1).app (N.functor.obj X)
 221 
 222 end KIP126.Synthetic.Context
