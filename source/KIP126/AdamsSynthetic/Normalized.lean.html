  17 /-- Actual factorization through the target tower of this system. -/
  18 public def MapFiltrationAtLeast (P : ClassicalTowerPresentation unit A)
  19     {X Y : C} (f : X ⟶ Y) (k : ℕ) : Prop :=
  20   ∃ lift : X ⟶ (P.tower Y).stageAt k,
  21     lift ≫ (P.tower Y).projection k = f
  22 
  23 /-- The paper's normalization exponent, determined by the same target tower. -/
  24 public def normalizedExponent (P : ClassicalTowerPresentation unit A)
  25     {X Y : C} (f : X ⟶ Y) : ℕ := by
  26   classical
  27   exact if MapFiltrationAtLeast P f 1 then 1 else 0
  28 
  29 public theorem normalizedExponent_le_one (P : ClassicalTowerPresentation unit A)
  30     {X Y : C} (f : X ⟶ Y) : normalizedExponent P f ≤ 1 := by
  31   classical
  32   unfold normalizedExponent
  33   split <;> decide
  34 
  35 /-- Positive-target form of the actual λ power. -/
  36 public def lambdaToPositivePow (k : ℕ) (X : Syn) :
  37     X ⟶ (SyntheticCategory.biShift (0, (k : ℤ))).obj X := by
  38   have h : ((0 : ℤ), (k : ℤ)) + ((0 : ℤ), -(k : ℤ)) = (0, 0) := by
  39     ext <;> simp
  40   exact SyntheticCategory.biShift_zero.inv.app X ≫
  41     eqToHom (congrArg (fun p => (SyntheticCategory.biShift p).obj X) h.symm) ≫
  42     (SyntheticCategory.biShift_comp (0, (k : ℤ)) (0, -(k : ℤ))).inv.app X ≫
  43     lambdaPow k ((SyntheticCategory.biShift (0, (k : ℤ))).obj X)
  44 
  45 /-- A normalized lift of this very classical map, with its factorization.
  46 Its existence remains an input, never an axiom or an automatically chosen map. -/
  47 public structure NormalizedSyntheticMap (P : ClassicalTowerPresentation unit A)
  48     (N : NuFunctorData C Syn) {X Y : C} (f : X ⟶ Y) where
  49   map : (SyntheticCategory.biShift (0, (normalizedExponent P f : ℤ))).obj
  50     (N.functor.obj X) ⟶ N.functor.obj Y
  51   factorization : lambdaToPositivePow (normalizedExponent P f) (N.functor.obj X) ≫
  52     map = N.functor.map f
  53 
  54 end
