  10 /-! Mapping spectra and their Adams systems on one given closed background.
  11 The objects are actual internal hom objects, not freely chosen graded groups.
  12 Convergence is requested only for pairs of objects in the specified scope.
  13 Source: KIPBase `multiplicativeSS/Adams.lean`, baseline 5aa845241533. -/
  14 namespace KIP126.AdamsSynthetic
  15 open CategoryTheory CategoryTheory.Limits CategoryTheory.MonoidalCategory

  17 open KIP126.Core.SpectralSequence

  22 /-- The internal hom belonging to this supplied closed structure. -/
  23 public def mappingSpectrum (B : ClosedSymmetricTensorTriangulated C) (X Y : C) : C :=
  24   letI := B.closed
  25   (ihom X).obj Y
  26 
  27 /-- The canonical closed adjunction and the same tensor-shift comparison
  28 identify the homotopy groups with actual shifted stable maps. -/
  29 public def mappingHomotopyEquiv (B : ClosedSymmetricTensorTriangulated C)
  30     (n : ℤ) (X Y : C) :
  31     HomotopyGroup n (mappingSpectrum B X Y) ≃ ((shiftFunctor C n).obj X ⟶ Y) := by
  32   letI := B.closed
  33   letI := B.symmetric
  34   exact ((ihom.adjunction X).homEquiv (Sphere n) Y).symm |>.trans
  35     (Iso.homCongr ((β_ X (Sphere n)).trans (B.sphereTensorShift n X)) (Iso.refl Y))
  36 
  37 /-- At degree zero this is an actual map `X ⟶ Y`. -/
  38 public def mappingHomotopyZeroEquiv (B : ClosedSymmetricTensorTriangulated C) (X Y : C) :
  39     HomotopyGroup 0 (mappingSpectrum B X Y) ≃ (X ⟶ Y) :=
  40   (mappingHomotopyEquiv B 0 X Y).trans
  41     (Iso.homCongr ((shiftFunctorZero C ℤ).app X) (Iso.refl Y))
  42 
  43 /-- The sphere mapping spectrum is canonically isomorphic to the sphere.
  44 This replaces the old unjustified equality of chosen objects. -/
  45 public def sphereMappingIso (B : ClosedSymmetricTensorTriangulated C) :
  46     mappingSpectrum B SphereSpectrum SphereSpectrum ≅ (SphereSpectrum : C) := by
  47   letI := B.closed
  48   exact MonoidalClosed.unitIsoSelf (C := C) (X := (SphereSpectrum : C))
  49 
  50 /-- Its Adams comparison is induced by the one given Adams functor. -/
  51 public def sphereMappingAdamsIso (B : ClosedSymmetricTensorTriangulated C)
  52     (A : ClassicalAdamsFamily C) :
  53     A.obj (mappingSpectrum B SphereSpectrum SphereSpectrum) ≅ A.sphere :=
  54   A.functor.mapIso (sphereMappingIso B)

  59 /-- Convergence of the actual internal hom Adams sequences. `scope` may be
  60 finite spectra or a specified object closure; no global convergence follows. -/
  61 public structure MappingSpectrumSystem (B : ClosedSymmetricTensorTriangulated C)
  62     (P : ClassicalTowerPresentation unit A) (scope : C → Prop) where
  63   convergence : ∀ X Y : C, scope X → scope Y →
  64     ClassicalTowerConvergence P (mappingSpectrum B X Y)
  65 
  66 namespace MappingSpectrumSystem
  67 variable {B : ClosedSymmetricTensorTriangulated C}
  68   {P : ClassicalTowerPresentation unit A} {scope : C → Prop}
  69 
  70 public abbrev sequence (_S : MappingSpectrumSystem B P scope) (X Y : C) :=
  71   A.obj (mappingSpectrum B X Y)
  72 
  73 public def converging (S : MappingSpectrumSystem B P scope) (X Y : C)
  74     (hX : scope X) (hY : scope Y) : ConvergingSS (ModuleCat.{v} ℤ) (ℤ × ℤ) ℤ where
  75   E := S.sequence X Y
  76   A := AbstractAdamsTower.homotopy (mappingSpectrum B X Y)
  77   F := (P.tower (mappingSpectrum B X Y)).filtration
  78   conv := (S.convergence X Y hX hY).toConvergence
  79 
  80 public theorem reindex (S : MappingSpectrumSystem B P scope) (X Y : C)
  81     (hX : scope X) (hY : scope Y) (p : ℤ × ℤ) :
  82     (S.converging X Y hX hY).conv.reindex p = (p.1, p.2 - p.1) := rfl
  83 
  84 public def homotopyEquiv (_S : MappingSpectrumSystem B P scope) (n : ℤ) (X Y : C) :
  85     HomotopyGroup n (mappingSpectrum B X Y) ≃ ((shiftFunctor C n).obj X ⟶ Y) :=
  86   mappingHomotopyEquiv B n X Y

  88 end MappingSpectrumSystem
