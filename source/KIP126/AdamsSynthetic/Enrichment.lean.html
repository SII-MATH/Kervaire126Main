  10 /-! The deformation parameter has weight -1.  Accordingly the enrichment
  11 lives on the total weight-graded Hom, with degree-zero morphisms the actual
  12 underlying category.  A polynomial action on each ungraded Hom separately
  13 would erase that degree and is not the contract supplied here. -/
  14 namespace KIP126.AdamsSynthetic
  15 open CategoryTheory KIP126.Synthetic.Context
  16 universe u v
  17 variable {Syn : Type u} [SyntheticCategory.{u,v} Syn]
  18 
  19 public abbrev WeightHom (X Y : Syn) (w : ℤ) :=
  20   (SyntheticCategory.biShift (0,w)).obj X ⟶ Y
  21 
  22 public abbrev TotalWeightHom (X Y : Syn) := DirectSum ℤ (WeightHom X Y)
  23 
  24 /-- Inclusion of an actual morphism of specified weight into total Hom. -/
  25 public noncomputable def weightHomInclusion {X Y : Syn} (w : ℤ) :
  26     WeightHom X Y w →+ TotalWeightHom X Y :=
  27   DirectSum.of (WeightHom X Y) w
  28 
  29 /-- The actual composition, through the already selected suspension comparison. -/
  30 public noncomputable def weightHomComp {X Y Z : Syn} {a b : ℤ}
  31     (f : WeightHom X Y a) (g : WeightHom Y Z b) : WeightHom X Z (a+b) :=
  32   (biShiftAddIso (0,a) (0,b) (0,a+b) (by ext <;> simp)).inv.app X ≫
  33     (SyntheticCategory.biShift (0,b)).map f ≫ g
  34 
  35 /-- λ acts by the actual deformation map and lowers the source weight by one. -/
  36 public noncomputable def weightHomLambda {X Y : Syn} {w : ℤ}
  37     (f : WeightHom X Y w) : WeightHom X Y (w-1) :=
  38   (biShiftAddIso (0,w) (0,-1) (0,w-1) (by ext <;> simp [sub_eq_add_neg])).inv.app X ≫
  39     SyntheticCategory.lam.app ((SyntheticCategory.biShift (0,w)).obj X) ≫ f
  40 
  41 /-- An enrichment on the total actual graded Hom. Its scalar action is tied
  42 to λ, its composition to actual shifted composition, and its units to the
  43 specified zero-shift comparison. No global enriched category is postulated. -/
  44 public structure GradedLambdaEnrichment (Syn : Type u) [SyntheticCategory.{u,v} Syn] where
  45   coherence : BiShiftCoherence Syn
  46   shiftAdditive : ∀ p : ℤ × ℤ, (SyntheticCategory.biShift p : Syn ⥤ Syn).Additive
  47   module : ∀ X Y : Syn, Module (Polynomial ℤ) (TotalWeightHom X Y)
  48   lambda_smul : ∀ (X Y : Syn) (w : ℤ) (f : WeightHom X Y w),
  49     letI := module X Y
  50     (Polynomial.X : Polynomial ℤ) • weightHomInclusion w f =
  51       weightHomInclusion (w-1) (weightHomLambda f)
  52   compose : ∀ X Y Z : Syn,
  53     letI := module X Y
  54     letI := module Y Z
  55     letI := module X Z
  56     TotalWeightHom Y Z →ₗ[Polynomial ℤ]
  57       TotalWeightHom X Y →ₗ[Polynomial ℤ] TotalWeightHom X Z
  58   compose_homogeneous : ∀ (X Y Z : Syn) (a b : ℤ)
  59       (f : WeightHom X Y a) (g : WeightHom Y Z b),
  60     compose X Y Z (weightHomInclusion b g) (weightHomInclusion a f) =
  61       weightHomInclusion (a+b) (weightHomComp f g)
  62 
  63 namespace GradedLambdaEnrichment
  64 variable (E : GradedLambdaEnrichment Syn)
  65 
  66 /-- Ordinary morphisms are included using the chosen degree-zero identity. -/
  67 public noncomputable def underlyingHom {X Y : Syn} (f : X ⟶ Y) : TotalWeightHom X Y :=
  68   weightHomInclusion 0 (SyntheticCategory.biShift_zero.hom.app X ≫ f)
  69 
  70 /-- Scalar bilinearity is a consequence of the supplied linear composition. -/
  71 public theorem compose_lambda_right (X Y Z : Syn)
  72     (g : TotalWeightHom Y Z) (f : TotalWeightHom X Y) :
  73     letI := E.module X Y
  74     letI := E.module Y Z
  75     letI := E.module X Z
  76     E.compose X Y Z g ((Polynomial.X : Polynomial ℤ) • f) =
  77       (Polynomial.X : Polynomial ℤ) • E.compose X Y Z g f := by
  78   letI := E.module X Y
  79   letI := E.module Y Z
  80   letI := E.module X Z
  81   exact (E.compose X Y Z g).map_smul _ _
  82 
  83 public theorem compose_lambda_left (X Y Z : Syn)
  84     (g : TotalWeightHom Y Z) (f : TotalWeightHom X Y) :
  85     letI := E.module X Y
  86     letI := E.module Y Z
  87     letI := E.module X Z
  88     E.compose X Y Z ((Polynomial.X : Polynomial ℤ) • g) f =
  89       (Polynomial.X : Polynomial ℤ) • E.compose X Y Z g f := by
  90   letI := E.module X Y
  91   letI := E.module Y Z
  92   letI := E.module X Z
  93   exact LinearMap.congr_fun ((E.compose X Y Z).map_smul (Polynomial.X : Polynomial ℤ) g) f
  94 end GradedLambdaEnrichment
  95 end KIP126.AdamsSynthetic

 102 private theorem homogeneous_regrade {X Y Z : Syn} (a b m : ℤ) (hab : a+b=m)
 103     (f : WeightHom X Y a) (g : WeightHom Y Z b) :
 104     weightHomInclusion (a+b) (weightHomComp f g) =
 105       weightHomInclusion m ((biShiftAddIso (0,a) (0,b) (0,m)
 106         (by ext <;> simp [hab])).inv.app X ≫ (SyntheticCategory.biShift (0,b)).map f ≫ g) := by
 107   subst m
 108   rfl

 110 private theorem homogeneous_left_unit (coh : BiShiftCoherence Syn)
 111     {X Y : Syn} (a : ℤ) (f : WeightHom X Y a) :
 112     weightHomInclusion (a+0)
 113       (weightHomComp f (SyntheticCategory.biShift_zero.hom.app Y)) =
 114         weightHomInclusion a f := by
 115   rw [homogeneous_regrade a 0 a (add_zero a)]
 116   apply congrArg (weightHomInclusion a)
 117   rw [SyntheticCategory.biShift_zero.hom.naturality]
 118   rw [← coh.right_unit (0,a) X]
 119   exact Iso.inv_hom_id_app_assoc (biShiftAddIso (0,a) 0 (0,a) (add_zero _)) X f
 120 
 121 private theorem homogeneous_right_unit (coh : BiShiftCoherence Syn)
 122     {X Y : Syn} (a : ℤ) (f : WeightHom X Y a) :
 123     weightHomInclusion (0+a)
 124       (weightHomComp (SyntheticCategory.biShift_zero.hom.app X) f) =
 125         weightHomInclusion a f := by
 126   rw [homogeneous_regrade 0 a a (zero_add a)]
 127   apply congrArg (weightHomInclusion a)
 128   rw [← coh.left_unit (0,a) X]
 129   exact Iso.inv_hom_id_app_assoc (biShiftAddIso 0 (0,a) (0,a) (zero_add _)) X f

 131 private theorem homogeneous_assoc (coh : BiShiftCoherence Syn)
 132     {W X Y Z : Syn} (a b c : ℤ)
 133     (f : WeightHom W X a) (g : WeightHom X Y b) (h : WeightHom Y Z c) :
 134     weightHomInclusion ((a+b)+c) (weightHomComp (weightHomComp f g) h) =
 135       weightHomInclusion (a+(b+c)) (weightHomComp f (weightHomComp g h)) := by
 136   rw [homogeneous_regrade (a+b) c (a+(b+c)) (add_assoc a b c)]
 137   apply congrArg (weightHomInclusion (a+(b+c)))
 138   let eab := biShiftAddIso (Syn := Syn) (0,a) (0,b) (0,a+b) (by ext <;> simp)
 139   let ebc := biShiftAddIso (Syn := Syn) (0,b) (0,c) (0,b+c) (by ext <;> simp)
 140   let eabc := biShiftAddIso (Syn := Syn) (0,a+b) (0,c) (0,a+(b+c)) (by ext <;> simp [add_assoc])
 141   let eabc' := biShiftAddIso (Syn := Syn) (0,a) (0,b+c) (0,a+(b+c)) (by ext <;> simp)
 142   have hi : ((SyntheticCategory.biShift (0,c)).mapIso (eab.app W) ≪≫ eabc.app W) =
 143       (ebc.app ((SyntheticCategory.biShift (0,a)).obj W) ≪≫ eabc'.app W) := by
 144     apply Iso.ext
 145     exact coh.associativity (0,a) (0,b) (0,c) (0,a+b) (0,b+c) (0,a+(b+c))
 146       (by ext <;> simp) (by ext <;> simp)
 147       (by ext <;> simp [add_assoc]) (by ext <;> simp) W
 148   have hinv : eabc.inv.app W ≫ (SyntheticCategory.biShift (0,c)).map (eab.inv.app W) =
 149       eabc'.inv.app W ≫ ebc.inv.app ((SyntheticCategory.biShift (0,a)).obj W) :=
 150     congrArg Iso.inv hi
 151   change eabc.inv.app W ≫ (SyntheticCategory.biShift (0,c)).map
 152       (eab.inv.app W ≫ (SyntheticCategory.biShift (0,b)).map f ≫ g) ≫ h =
 153     eabc'.inv.app W ≫ (SyntheticCategory.biShift (0,b+c)).map f ≫
 154       ebc.inv.app X ≫ (SyntheticCategory.biShift (0,c)).map g ≫ h
 155   simp only [Functor.map_comp, Category.assoc]
 156   rw [← Category.assoc (eabc.inv.app W), hinv]
 157   rw [Category.assoc]
 158   have hn : (SyntheticCategory.biShift (0,b+c)).map f ≫ ebc.inv.app X =
 159       ebc.inv.app ((SyntheticCategory.biShift (0,a)).obj W) ≫
 160         (SyntheticCategory.biShift (0,c)).map ((SyntheticCategory.biShift (0,b)).map f) :=
 161     ebc.inv.naturality f
 162   rw [← Category.assoc (ebc.inv.app ((SyntheticCategory.biShift (0,a)).obj W)), ← hn]
 163   simp only [Category.assoc]

 167 /-- Units on arbitrary total Hom elements follow from the actual homogeneous
 168 composition and the supplied suspension unit coherence. -/
 169 public theorem left_unit (X Y : Syn) (f : TotalWeightHom X Y) :
 170     E.compose X Y Y (weightHomInclusion 0 (SyntheticCategory.biShift_zero.hom.app Y)) f = f := by
 171   letI := E.module X Y
 172   letI := E.module Y Y
 173   induction f using DirectSum.induction_on with
 174   | zero => exact map_zero _
 175   | of a f =>
 176     exact (E.compose_homogeneous X Y Y a 0 f _).trans (homogeneous_left_unit E.coherence a f)
 177   | add x y hx hy => simp only [map_add, hx, hy]
 178 
 179 public theorem right_unit (X Y : Syn) (f : TotalWeightHom X Y) :
 180     E.compose X X Y f (weightHomInclusion 0 (SyntheticCategory.biShift_zero.hom.app X)) = f := by
 181   letI := E.module X X
 182   letI := E.module X Y
 183   induction f using DirectSum.induction_on with
 184   | zero => simp only [map_zero, LinearMap.zero_apply]
 185   | of a f =>
 186     exact (E.compose_homogeneous X X Y 0 a _ f).trans (homogeneous_right_unit E.coherence a f)
 187   | add x y hx hy => simp only [map_add, LinearMap.add_apply, hx, hy]

 189 /-- Associativity is derived by additive induction from actual shifted-map
 190 composition and the same suspension associativity constraint. -/
 191 public theorem assoc (W X Y Z : Syn) (f : TotalWeightHom W X)
 192     (g : TotalWeightHom X Y) (h : TotalWeightHom Y Z) :
 193     E.compose W Y Z h (E.compose W X Y g f) =
 194       E.compose W X Z (E.compose X Y Z h g) f := by
 195   letI := E.module W X
 196   letI := E.module X Y
 197   letI := E.module Y Z
 198   letI := E.module W Y
 199   letI := E.module X Z
 200   letI := E.module W Z
 201   induction f using DirectSum.induction_on with
 202   | zero => simp only [map_zero]
 203   | add f f' hf hf' => simp only [map_add, hf, hf']
 204   | of a f =>
 205     induction g using DirectSum.induction_on with
 206     | zero => simp only [map_zero, LinearMap.zero_apply]
 207     | add g g' hg hg' => simp only [map_add, LinearMap.add_apply, hg, hg']
 208     | of b g =>
 209       induction h using DirectSum.induction_on with
 210       | zero => simp only [map_zero, LinearMap.zero_apply]
 211       | add h h' hh hh' => simp only [map_add, LinearMap.add_apply, hh, hh']
 212       | of c h =>
 213         change E.compose W Y Z (weightHomInclusion c h)
 214           (E.compose W X Y (weightHomInclusion b g) (weightHomInclusion a f)) =
 215           E.compose W X Z (E.compose X Y Z (weightHomInclusion c h) (weightHomInclusion b g))
 216             (weightHomInclusion a f)
 217         rw [E.compose_homogeneous, E.compose_homogeneous,
 218           E.compose_homogeneous, E.compose_homogeneous]
 219         exact homogeneous_assoc E.coherence a b c f g h
 220 end GradedLambdaEnrichment
 221 end KIP126.AdamsSynthetic
