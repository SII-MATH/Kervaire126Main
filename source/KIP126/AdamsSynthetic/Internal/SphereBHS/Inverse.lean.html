  26 /-- Nonempty finite terms with arbitrary adjacent restriction maps admit a
  27 compatible dependent inverse sequence. The point at the initial index is
  28 not prescribed, and the restriction maps need not be surjective. -/
  29 private theorem exists_compatible_of_finite (m₀ : ℕ)
  30     (A : (m : ℕ) → m₀ ≤ m → Type u)
  31     (restrict : ∀ (m : ℕ) (hm : m₀ ≤ m),
  32       A (m + 1) (Nat.le_succ_of_le hm) → A m hm)
  33     (hfinite : ∀ (m : ℕ) (hm : m₀ ≤ m), Finite (A m hm))
  34     (hne : ∀ (m : ℕ) (hm : m₀ ≤ m), Nonempty (A m hm)) :
  35     ∃ a : ∀ (m : ℕ) (hm : m₀ ≤ m), A m hm,
  36       ∀ (m : ℕ) (hm : m₀ ≤ m),
  37         restrict m hm (a (m + 1) (Nat.le_succ_of_le hm)) = a m hm := by
  38   classical
  39   let B (m : ℕ) := ∀ hm : m₀ ≤ m, A m hm
  40   let step (m : ℕ) (b : B (m + 1)) : B m :=
  41     fun hm => restrict m hm (b (Nat.le_succ_of_le hm))
  42   letI : ∀ m, Finite (B m) := fun m => by
  43     letI : ∀ hm : m₀ ≤ m, Finite (A m hm) := hfinite m
  44     exact Pi.finite
  45   letI : ∀ m, Nonempty (B m) :=
  46     fun m => ⟨fun hm => Classical.choice (hne m hm)⟩
  47   let π : {i j : ℕ} → i ≤ j → B j → B i :=
  48     fun {_ _} hij b => Nat.decreasingInduction (fun k _ => step k) b hij
  49   obtain ⟨a, ha⟩ := exists_seq_forall_proj_of_forall_finite π
  50     (fun {_} _ => Nat.decreasingInduction_self _ _)
  51     (by
  52       intro i j k hij hjk b
  53       exact (Nat.decreasingInduction_trans (motive := fun m (_ : m ≤ k) => B m)
  54         hij hjk (fun m _ => step m) b).symm)
  55     (fun _ _ => Set.toFinite _)
  56   refine ⟨fun m hm => a m hm, ?_⟩
  57   intro m hm
  58   have h := congrFun (ha (Nat.le_succ m)) hm
  59   simpa only [π, Nat.decreasingInduction_succ', step] using h

  61 /-- The `1 - shift` map of an inverse sequence of finite abelian groups is
  62 surjective, the algebraic vanishing criterion for its first derived limit.
  63 For each target sequence, apply finite inverse-system compactness to the
  64 affine restrictions; the original homomorphisms need not be surjective. -/
  65 private theorem surjective_sub_shift_of_finite (A : ℕ → Type u)
  66     [∀ n, AddCommGroup (A n)] [∀ n, Finite (A n)]
  67     (f : ∀ n, A (n + 1) →+ A n) :
  68     Function.Surjective (fun x : ∀ n, A n => fun n => x n - f n (x (n + 1))) := by
  69   intro b
  70   obtain ⟨x, hx⟩ := exists_compatible_of_finite 0 (fun n _ => A n)
  71     (fun n _ a => b n + f n a) (fun n _ => inferInstance)
  72     (fun n _ => ⟨0⟩)
  73   refine ⟨fun n => x n (Nat.zero_le n), ?_⟩
  74   funext n
  75   have h := hx n (Nat.zero_le n)
  76   dsimp only at h ⊢
  77   rw [← h, add_sub_cancel_right]

  80 private theorem finite_fiber_of_finite_ker
  81     {A : Type u} {B : Type v} [AddGroup A] [AddGroup B]
  82     (f : A →+ B) [Finite f.ker] (b : B) : Finite {a : A // f a = b} := by
  83   classical
  84   by_cases h : Nonempty {a : A // f a = b}
  85   · let a₀ := Classical.choice h
  86     let e : {a : A // f a = b} → f.ker := fun a =>
  87       ⟨a.val - a₀.val, by simp only [AddMonoidHom.mem_ker, map_sub, a.property, a₀.property, sub_self]⟩
  88     exact Finite.of_injective e (fun a a' h => Subtype.ext (sub_left_inj.mp (congrArg Subtype.val h)))
  89   · letI : IsEmpty {a : A // f a = b} := not_nonempty_iff.mp h
  90     infer_instance

  92 private theorem finite_fiber_comp
  93     {A : Type u} {B : Type v} {C : Type w}
  94     (f : B → C) (g : A → B)
  95     (hf : ∀ c, Finite {b // f b = c}) (hg : ∀ b, Finite {a // g a = b})
  96     (c : C) : Finite {a // f (g a) = c} := by
  97   letI : Finite {b // f b = c} := hf c
  98   letI : ∀ b : {b // f b = c}, Finite {a // g a = b.val} := fun b => hg b.val
  99   let e : {a // f (g a) = c} → Σ b : {b // f b = c}, {a // g a = b.val} :=
 100     fun a => ⟨⟨g a.val, a.property⟩, ⟨a.val, rfl⟩⟩
 101   exact Finite.of_injective e (fun a a' h => Subtype.ext (congrArg (fun z => z.2.val) h))

 104 private theorem finite_fiber_tower
 105     (G : ℕ → Type u) [∀ n, AddCommGroup (G n)]
 106     (p : ∀ a b, a ≤ b → G b →+ G a)
 107     (pid : ∀ a x, p a a (Nat.le_refl a) x = x)
 108     (pcomp : ∀ a b c (hab : a ≤ b) (hbc : b ≤ c) x,
 109       p a b hab (p b c hbc x) = p a c (hab.trans hbc) x)
 110     (hfinite : ∀ n, Finite (p n (n+1) (Nat.le_succ n)).ker)
 111     (a b : ℕ) (hab : a ≤ b) (x : G a) : Finite {y : G b // p a b hab y = x} := by
 112   induction b, hab using Nat.le_induction generalizing x with
 113   | base =>
 114     let he : {y : G a // p a a (Nat.le_refl a) y = x} → {y : G a // y = x} :=
 115       fun y => ⟨y.val, (pid a y.val).symm.trans y.property⟩
 116     exact Finite.of_injective he (by
 117       intro y y' h
 118       apply Subtype.ext
 119       exact congrArg (fun z : {y : G a // y = x} => z.val) h)
 120   | succ b hab ih =>
 121     letI := hfinite b
 122     have hf : ∀ z, Finite {y : G b // p a b hab y = z} := by
 123       intro z
 124       exact ih z
 125     have hg := finite_fiber_comp (p a b hab) (p b (b+1) (Nat.le_succ b)) hf
 126       (finite_fiber_of_finite_ker (p b (b+1) (Nat.le_succ b))) x
 127     exact Finite.of_injective
 128       (fun y : {y : G (b+1) // p a (b+1) (hab.trans (Nat.le_succ b)) y = x} =>
 129         (⟨y.val, (pcomp a b (b+1) hab (Nat.le_succ b) y.val).trans y.property⟩ :
 130           {y : G (b+1) // p a b hab (p b (b+1) (Nat.le_succ b) y) = x}))
 131       (by
 132         intro y y' h
 133         apply Subtype.ext
 134         exact congrArg (fun z : {y : G (b+1) // p a b hab (p b (b+1) (Nat.le_succ b) y) = x} => z.val) h)

 137 /-- A zero inverse limit together with finite adjacent kernels separates all image filtrations. -/
 138 private theorem separated_of_finite_ker
 139     (G : ℕ → Type u) [∀ n, AddCommGroup (G n)]
 140     (p : ∀ a b, a ≤ b → G b →+ G a)
 141     (pid : ∀ a x, p a a (Nat.le_refl a) x = x)
 142     (pcomp : ∀ a b c (hab : a ≤ b) (hbc : b ≤ c) x,
 143       p a b hab (p b c hbc x) = p a c (hab.trans hbc) x)
 144     (hfinite : ∀ n, Finite (p n (n+1) (Nat.le_succ n)).ker)
 145     (hzero : ∀ z : ∀ n, G n,
 146       (∀ n, p n (n+1) (Nat.le_succ n) (z (n+1)) = z n) → ∀ n, z n = 0)
 147     (a : ℕ) (x : G a)
 148     (hx : ∀ b (hab : a ≤ b), ∃ y, p a b hab y = x) : x = 0 := by
 149   classical
 150   let A (b : ℕ) (hab : a ≤ b) := {y : G b // p a b hab y = x}
 151   let step (b : ℕ) (hab : a ≤ b) : A (b+1) (Nat.le_succ_of_le hab) → A b hab :=
 152     fun y => ⟨p b (b+1) (Nat.le_succ b) y.val,
 153       (pcomp a b (b+1) hab (Nat.le_succ b) y.val).trans y.property⟩
 154   obtain ⟨u, hu⟩ := exists_compatible_of_finite a A step
 155     (fun b hab => finite_fiber_tower G p pid pcomp hfinite a b hab x)
 156     (fun b hab => let ⟨y, hy⟩ := hx b hab; ⟨⟨y, hy⟩⟩)
 157   have hua : (u a (Nat.le_refl a)).val = x :=
 158     (pid a _).symm.trans (u a (Nat.le_refl a)).property
 159   let z (b : ℕ) : G b := if hab : a ≤ b then (u b hab).val else p b a (by omega) x
 160   have hcompat : ∀ b, p b (b+1) (Nat.le_succ b) (z (b+1)) = z b := by
 161     intro b
 162     by_cases hab : a ≤ b
 163     · have hab' : a ≤ b+1 := Nat.le_succ_of_le hab
 164       simp only [z, dif_pos hab, dif_pos hab']
 165       exact congrArg Subtype.val (hu b hab)
 166     · by_cases hab' : a ≤ b+1
 167       · have hba : b+1 = a := by omega
 168         subst a
 169         simp only [z, dif_pos hab', dif_neg hab]
 170         rw [hua]
 171       · simp only [z, dif_neg hab, dif_neg hab']
 172         exact pcomp b (b+1) a (Nat.le_succ b) (by omega) x
 173   have hza := hzero z hcompat a
 174   simpa only [z, dif_pos (Nat.le_refl a), hua] using hza

 176 /-- Injectivity of the explicit difference operator kills compatible families. -/
 177 private theorem compatible_zero_of_difference_injective
 178     (G : ℕ → Type u) [∀ n, AddCommGroup (G n)]
 179     (step : ∀ n, G (n+1) →+ G n)
 180     (hinj : Function.Injective (fun z : ∀ n, G n => fun n => z n - step n (z (n+1))))
 181     (z : ∀ n, G n) (hz : ∀ n, step n (z (n+1)) = z n) : ∀ n, z n = 0 := by
 182   have he : z = 0 := hinj (by funext n; simp only [hz, sub_self, Pi.zero_apply, map_zero])
 183   intro n
 184   exact congrFun he n

 188 /-- The correction step in §9.1: solving the difference equation changes
 189 compatible representatives into one representative of all quotient classes. -/
 190 private theorem common_representative
 191     (G : ℕ → Type v) [∀ s, AddCommGroup (G s)]
 192     {A : Type w} [AddCommGroup A]
 193     (i : ∀ s, G (s+1) →+ G s) (p : ∀ s, G s →+ A)
 194     (hp : ∀ s y, p s (i s y) = p (s+1) y)
 195     (hdelta : Function.Surjective (fun z : ∀ s, G s => fun s => z s - i s (z (s+1))))
 196     (x : ℕ → A) (hx : ∀ s, ∃ y, p s y = x s - x (s+1)) :
 197     ∃ a : A, ∀ s, ∃ y, p s y = x s - a := by
 198   classical
 199   choose y hy using hx
 200   obtain ⟨z, hz⟩ := hdelta y
 201   have hz' (s : ℕ) : p s (z s) - p (s+1) (z (s+1)) = x s - x (s+1) := by
 202     have h := congrArg (p s) (congrFun hz s)
 203     simpa only [map_sub, hp, hy] using h
 204   have hw (s : ℕ) : x s - p s (z s) = x 0 - p 0 (z 0) := by
 205     induction s with
 206     | zero => rfl
 207     | succ s ih =>
 208       calc
 209         x (s+1) - p (s+1) (z (s+1)) = x s - p s (z s) := by
 210           rw [sub_eq_iff_eq_add.mp (hz' s)]
 211           abel
 212         _ = _ := ih
 213   refine ⟨x 0 - p 0 (z 0), fun s => ⟨z s, ?_⟩⟩
 214   rw [← hw s]
 215   abel
