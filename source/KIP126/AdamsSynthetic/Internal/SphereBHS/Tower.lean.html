  34 private def oneSubShift [HasProductsOfShape ℕ C] : ∏ᶜ T.stage ⟶ ∏ᶜ T.stage :=
  35   𝟙 _ - Pi.lift (fun s => Pi.π T.stage (s+1) ≫ T.map (Nat.le_succ s))
  36 
  37 private theorem hom_delta_bijective [HasProductsOfShape ℕ C]
  38     (hT : IsIso (oneSubShift T)) (B : C) :
  39     Function.Bijective (fun z : ∀ s, B ⟶ T.stage s =>
  40       fun s => z s - z (s+1) ≫ T.map (Nat.le_succ s)) := by
  41   letI := hT
  42   have proj (z : ∀ s, B ⟶ T.stage s) (s : ℕ) :
  43       (Pi.lift z ≫ oneSubShift T) ≫ Pi.π T.stage s =
  44         z s - z (s+1) ≫ T.map (Nat.le_succ s) := by
  45     simp [oneSubShift, Preadditive.comp_sub, Preadditive.sub_comp]
  46   constructor
  47   · intro z w h
  48     have he : Pi.lift z ≫ oneSubShift T = Pi.lift w ≫ oneSubShift T := by
  49       apply Pi.hom_ext
  50       intro s
  51       simpa only [proj] using congrFun h s
  52     have he' := (cancel_mono (oneSubShift T)).mp he
  53     funext s
  54     simpa using congrArg (fun f => f ≫ Pi.π T.stage s) he'
  55   · intro y
  56     let z : ∀ s, B ⟶ T.stage s :=
  57       fun s => Pi.lift y ≫ inv (oneSubShift T) ≫ Pi.π T.stage s
  58     have hl : Pi.lift z = Pi.lift y ≫ inv (oneSubShift T) := by
  59       apply Pi.hom_ext
  60       intro s
  61       simp [z]
  62     refine ⟨z, funext fun s => ?_⟩
  63     change z s - z (s+1) ≫ T.map (Nat.le_succ s) = y s
  64     rw [← proj z s, hl]
  65     simp

  67 private theorem separated_of_finite_kernels (n : ℤ)
  68     (hfinite : ∀ s, Finite (inducedMap (T.map (Nat.le_succ s)) n).ker)
  69     (hinj : Function.Injective (fun z : ∀ s, HomotopyGroup n (T.stage s) =>
  70       fun s => z s - z (s+1) ≫ T.map (Nat.le_succ s)))
  71     (a : ℕ) (x : HomotopyGroup n (T.stage a))
  72     (hx : ∀ b (hab : a ≤ b), x ∈ LinearMap.range
  73       (inducedMap (T.map hab) n).toIntLinearMap) : x = 0 := by
  74   apply separated_of_finite_ker (fun s => HomotopyGroup n (T.stage s))
  75     (fun a b hab => inducedMap (T.map hab) n)
  76     (fun a x => by simp only [T.map_self, inducedMap, AddMonoidHom.coe_mk,
  77       ZeroHom.coe_mk, Category.comp_id])
  78     (fun a b c hab hbc x => by
  79       change (x ≫ T.map hbc) ≫ T.map hab = _
  80       rw [Category.assoc, T.map_comp]
  81       rfl)
  82     hfinite ?_ a x hx
  83   intro z hz
  84   exact compatible_zero_of_difference_injective _
  85     (fun s => inducedMap (T.map (Nat.le_succ s)) n) hinj z hz

  87 private theorem filtration_element_separated (n : ℤ)
  88     (hfinite : ∀ s, Finite (inducedMap (T.map (Nat.le_succ s)) n).ker)
  89     (hinj : Function.Injective (fun z : ∀ s, HomotopyGroup n (T.stage s) =>
  90       fun s => z s - z (s+1) ≫ T.map (Nat.le_succ s)))
  91     (x : HomotopyGroup n X)
  92     (hx : ∀ s : ℕ, x ∈ T.filtrationSubmodule s n) : x = 0 := by
  93   have hz : x ≫ T.zeroIso.inv = 0 := by
  94     apply separated_of_finite_kernels T n hfinite hinj 0 (x ≫ T.zeroIso.inv)
  95     intro s hs
  96     obtain ⟨y, hy⟩ := hx s
  97     refine ⟨y, ?_⟩
  98     apply (cancel_mono T.zeroIso.hom).mp
  99     change y ≫ (T.map (Nat.zero_le s) ≫ T.zeroIso.hom) = x at hy
 100     change (y ≫ T.map hs) ≫ T.zeroIso.hom = (x ≫ T.zeroIso.inv) ≫ T.zeroIso.hom
 101     simpa only [Category.assoc, Iso.inv_hom_id, Category.comp_id] using hy
 102   have h := congrArg (fun z => z ≫ T.zeroIso.hom) hz
 103   simpa only [Category.assoc, Iso.inv_hom_id, Category.comp_id, zero_comp] using h

 105 private theorem filtration_separated
 106     (hsep : ∀ (n : ℤ) (x : HomotopyGroup n X),
 107       (∀ s : ℕ, x ∈ T.filtrationSubmodule s n) → x = 0)
 108     (n : ℤ) (S : Subobject (AbstractAdamsTower.homotopy X n))
 109     (hS : ∀ s : ℤ, S ≤ T.filtration.F s n) : S = ⊥ := by
 110   apply (ModuleCat.subobjectModule (AbstractAdamsTower.homotopy X n)).injective
 111   rw [OrderIso.map_bot]
 112   apply le_antisymm
 113   · intro x hx
 114     apply (Submodule.mem_bot ℤ).mpr
 115     apply hsep n x
 116     intro s
 117     have hs := (ModuleCat.subobjectModule _).monotone (hS (s : ℤ))
 118     have hx' := hs hx
 119     simpa only [AbstractAdamsTower.filtration, OrderIso.apply_symm_apply] using hx'
 120   · exact bot_le

 122 private theorem quotientProjection_eq_zero_iff
 123     (s n : ℤ) (x : AbstractAdamsTower.homotopy X n) :
 124     T.filtration.quotientProjection s n x = 0 ↔
 125       x ∈ T.filtrationSubmodule s n := by
 126   refine (cokernel_π_eq_zero_iff_mem_range
 127     (T.filtration.F s n).arrow x).trans ?_
 128   change x ∈ (ModuleCat.subobjectModule (AbstractAdamsTower.homotopy X n))
 129     ((ModuleCat.subobjectModule (AbstractAdamsTower.homotopy X n)).symm (T.filtrationSubmodule s n)) ↔ _
 130   rw [OrderIso.apply_symm_apply]

 132 /-- The prescribed cokernel cone is limiting, not just abstractly isomorphic
 133 to some complete group. Only surjectivity of the difference and separation
 134 at stage zero are needed for this step. -/
 135 private theorem adams_complete_of_difference (n : ℤ)
 136     (hdelta : Function.Surjective
 137       (fun z : ∀ s, HomotopyGroup n (T.stage s) =>
 138         fun s => z s - z (s+1) ≫ T.map (Nat.le_succ s)))
 139     (hsep : ∀ x : HomotopyGroup n X,
 140       (∀ s : ℕ, x ∈ T.filtrationSubmodule s n) → x = 0) :
 141     Nonempty (Filtration.CompletionWitness
 142       T.filtration n) := by
 143   classical
 144   let F := T.filtration
 145   let q (s : ℤ) := F.quotientProjection s n
 146   have hq (s : ℤ) : Function.Surjective (q s) := by
 147     apply (ModuleCat.epi_iff_surjective _).mp
 148     change Epi (cokernel.π _)
 149     infer_instance
 150   have hk (s : ℤ) (x : HomotopyGroup n X) :
 151       q s x = 0 ↔ x ∈ T.filtrationSubmodule s n :=
 152     quotientProjection_eq_zero_iff T s n x
 153   suffices ht : Nonempty (IsLimit ((forget (ModuleCat ℤ)).mapCone (F.quotientTowerCone n))) by
 154     exact ⟨⟨isLimitOfReflects (forget (ModuleCat ℤ)) (Classical.choice ht)⟩⟩
 155   rw [Types.isLimit_iff]
 156   intro c hc
 157   have hcn (s : ℕ) :
 158       F.quotientTransition (by omega : (s : ℤ) ≤ (s+1 : ℕ)) n (c (s+1 : ℕ)) = c s := by
 159     exact hc (homOfLE (show (s+1 : ℕ) ≤ (s : OrderDual ℤ) from by
 160       change (s : ℤ) ≤ (s+1 : ℕ); omega))
 161   choose x hx using fun s : ℕ => hq s (c s)
 162   change ℕ → HomotopyGroup n X at x
 163   have hdiff (s : ℕ) : ∃ y : HomotopyGroup n (T.stage s),
 164       inducedMap (T.projection (s : ℤ)) n y = x s - x (s+1) := by
 165     have hh : q s (x s - x (s+1)) = 0 := by
 166       rw [map_sub, hx, sub_eq_zero]
 167       calc
 168         c s = F.quotientTransition (by omega : (s : ℤ) ≤ (s+1 : ℕ)) n (q (s+1 : ℕ) (x (s+1))) := by
 169           rw [hx, hcn]
 170         _ = q s (x (s+1)) := by
 171           exact ConcreteCategory.congr_hom (F.quotientProjection_transition
 172             (by omega : (s : ℤ) ≤ (s+1 : ℕ)) n) _
 173     exact (hk s _).mp hh
 174   obtain ⟨a, ha⟩ := common_representative (A := HomotopyGroup n X)
 175     (fun s => HomotopyGroup n (T.stage s))
 176     (fun s => inducedMap (T.map (Nat.le_succ s)) n)
 177     (fun s => inducedMap (T.projection (s : ℤ)) n)
 178     (by
 179       intro s y
 180       change (y ≫ T.map (Nat.le_succ s)) ≫ T.projection (s : ℤ) = _
 181       rw [Category.assoc]
 182       exact congrArg (fun f => y ≫ f)
 183         (T.projection_compat (show (s : ℤ) ≤ (s+1 : ℕ) by omega))) hdelta x hdiff
 184   have han (s : ℕ) : q s a = c s := by
 185     have hmem : x s - a ∈ T.filtrationSubmodule s n := ha s
 186     have he := (hk s _).mpr hmem
 187     rw [map_sub, hx, sub_eq_zero] at he
 188     exact he.symm
 189   have haall (s : ℤ) : q s a = c s := by
 190     let k := s.toNat
 191     have hsk : s ≤ (k : ℤ) := by dsimp [k]; omega
 192     have hcs : F.quotientTransition hsk n (c k) = c s :=
 193       hc (homOfLE (show (k : OrderDual ℤ) ≤ s from hsk))
 194     rw [← hcs, ← han k]
 195     exact (ConcreteCategory.congr_hom (F.quotientProjection_transition hsk n) a).symm
 196   refine ⟨a, haall, ?_⟩
 197   intro b hb
 198   change HomotopyGroup n X at b
 199   change ∀ s : ℤ, q s b = c s at hb
 200   change (b : HomotopyGroup n X) = a
 201   apply sub_eq_zero.mp
 202   apply hsep
 203   intro s
 204   apply (hk s _).mp
 205   rw [map_sub]
 206   change q s b - q s a = 0
 207   rw [hb, haall, sub_self]

 210 /-- Conditional producer for the complete, separated actual tower filtration.
 211 The two hypotheses are retained; this is not an instance for Standard.Spectrum. -/
 212 private theorem complete_and_separated_of_acyclic_and_finite
 213     [HasProductsOfShape ℕ C] (hT : IsIso (oneSubShift T))
 214     (hfinite : ∀ (n : ℤ) (s : ℕ),
 215       Finite (inducedMap (T.map (Nat.le_succ s)) n).ker) :
 216     (∀ n : ℤ, Nonempty (Filtration.CompletionWitness T.filtration n)) ∧
 217       (∀ (n : ℤ) (S : Subobject (AbstractAdamsTower.homotopy X n)),
 218         (∀ s : ℤ, S ≤ T.filtration.F s n) → S = ⊥) := by
 219   have hd (n : ℤ) := hom_delta_bijective T hT (Sphere (C := C) n)
 220   have hs := fun n => filtration_element_separated T n (hfinite n) (hd n).injective
 221   exact ⟨fun n => adams_complete_of_difference T n (hd n).surjective (hs n),
 222     filtration_separated T hs⟩
 223 
 224 end
