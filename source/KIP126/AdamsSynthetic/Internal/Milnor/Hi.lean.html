  25 /-- The polynomial representative `[ξ₁^(2^i)]` of `hᵢ`, starting with `i = 0`. -/
  26 def hiPolynomial (i : ℕ) : TensorPower 1 :=
  27   MvPolynomial.X (0, 0) ^ (2 ^ i)
  28 
  29 end

  31 /- Adapted from KIP126/Def/Steenrod/MilnorCobar/Polynomial/Hi/Proofs.lean. -/

  34 open MvPolynomial
  35 
  36 /-- The representative has internal degree `2^i` and vanishing augmentation. -/
  37 theorem hiPolynomial_mem (i : ℕ) : IsCochain (2 ^ i) (hiPolynomial i) := by
  38   constructor
  39   · simpa [hiPolynomial, weight] using
  40       (MvPolynomial.isWeightedHomogeneous_X (R := F2) (@weight 1) (0, 0)).pow (2 ^ i)
  41   · change hiPolynomial i ∈ (⨅ slot : Fin 1, LinearMap.ker (augmentSlot slot).toLinearMap)
  42     rw [Submodule.mem_iInf]
  43     intro slot
  44     have hslot : slot = 0 := Subsingleton.elim _ _
  45     subst slot
  46     simp [LinearMap.mem_ker, augmentSlot, hiPolynomial]
  47 
  48 /-- Frobenius preserves primitiveness in characteristic two. -/
  49 theorem hiPolynomial_differential (i : ℕ) :
  50     differentialPolynomial 1 (hiPolynomial i) = 0 := by
  51   have hp (x y : TensorPower 2) :
  52       (x + y) ^ (2 ^ i) = x ^ (2 ^ i) + y ^ (2 ^ i) :=
  53     add_pow_char_pow x y 2 i
  54   simp [differentialPolynomial, hiPolynomial, insertLeft, insertRight,
  55     splitSlot, coproductGenerator, xi, Finset.sum_range_succ, hp,
  56     add_assoc, CharTwo.add_self_eq_zero, CharTwo.add_cancel_left]
  57 
  58 /-- Concatenating two `hᵢ` representatives gives internal degree `2^(i+1)`. -/
  59 theorem hiSquare_internalDegree (i : ℕ) : 2 ^ i + 2 ^ i = 2 ^ (i + 1) := by
  60   rw [pow_succ]
  61   omega
  62 
  63 @[simp] theorem hiPolynomial_six : hiPolynomial 6 = h6Polynomial := rfl
  64 
  65 end

  70 /-- The standard normalized cochain `[ξ₁^(2^i)]` in bidegree `(1, 2^i)`. -/
  71 def hiCochain (i : ℕ) : cochains 1 (2 ^ i) :=
  72   ⟨hiPolynomial i, hiPolynomial_mem i⟩
  73 
  74 /-- The actual concatenation square, reindexed to bidegree `(2, 2^(i+1))`. -/
  75 def hiSquareCochain (i : ℕ) : cochains 2 (2 ^ (i + 1)) :=
  76   reindex rfl (hiSquare_internalDegree i) (cup (hiCochain i) (hiCochain i))
  77 
  78 end
  79 
  80 /- Adapted from KIP126/Def/Steenrod/MilnorCobar/Hi/Proofs.lean. -/
  81 noncomputable section
  82 
  83 theorem hiCochain_isCycle (i : ℕ) : IsCycle (hiCochain i) := by
  84   apply Subtype.ext
  85   exact hiPolynomial_differential i
  86 
  87 theorem hiSquareCochain_isCycle (i : ℕ) : IsCycle (hiSquareCochain i) := by
  88   have transport (t t' : ℕ) (ht : t = t') (x : cochains 2 t) (hx : IsCycle x) :
  89       IsCycle (reindex rfl ht x) := by
  90     subst t'
  91     exact hx
  92   exact transport _ _ (hiSquare_internalDegree i) _
  93     (cup_isCycle _ _ (hiCochain_isCycle i) (hiCochain_isCycle i))
  94 
  95 @[simp] theorem hiCochain_six : hiCochain 6 = h6Cochain := rfl
  96 
  97 @[simp] theorem hiSquareCochain_six : hiSquareCochain 6 = h6SquareCochain := rfl
  98 
  99 end
