  26 /-- Embed a block of `s` slots starting at `offset` in `n` slots. -/
  27 def blockRename {s n : ℕ} (offset : ℕ) (h : offset + s ≤ n) :
  28     TensorPower s →ₐ[F2] TensorPower n :=
  29   MvPolynomial.rename fun a => (⟨offset + a.1.val, by have := a.1.isLt; omega⟩, a.2)
  30 
  31 end

  38 /-- Slot renaming respects the convention `ξ₀ = 1`. -/
  39 theorem rename_xi {s s' : ℕ} (f : Fin s → Fin s') (i : Fin s) (j : ℕ) :
  40     rename (fun a : Fin s × ℕ => (f a.1, a.2)) (xi i j) = xi (f i) j := by
  41   cases j <;> simp [xi]
  42 
  43 /-- A coproduct depends only on the two adjacent slots into which it lands. -/
  44 theorem rename_coproductGenerator {s s' : ℕ} (g : Fin (s + 1) → Fin (s' + 1))
  45     (i : Fin s) (k : Fin s') (h₁ : g i.castSucc = k.castSucc) (h₂ : g i.succ = k.succ)
  46     (j : ℕ) :
  47     rename (fun a : Fin (s + 1) × ℕ => (g a.1, a.2)) (coproductGenerator i j) =
  48       coproductGenerator k j := by
  49   simp only [coproductGenerator, map_sum, map_mul, map_pow, rename_xi, h₁, h₂]

  51 /-- Naturality of splitting with respect to an embedding of tensor slots. -/
  52 theorem splitSlot_rename {s s' : ℕ} (f : Fin s → Fin s')
  53     (g : Fin (s + 1) → Fin (s' + 1)) (i : Fin s) (k : Fin s')
  54     (hf : ∀ a, f a = k ↔ a = i)
  55     (h₁ : g i.castSucc = k.castSucc) (h₂ : g i.succ = k.succ)
  56     (hg : ∀ a, a ≠ i →
  57       g (if a < i then a.castSucc else a.succ) =
  58         if f a < k then (f a).castSucc else (f a).succ)
  59     (x : TensorPower s) :
  60     splitSlot k (rename (fun a : Fin s × ℕ => (f a.1, a.2)) x) =
  61       rename (fun a : Fin (s + 1) × ℕ => (g a.1, a.2)) (splitSlot i x) := by
  62   have he : (splitSlot k).comp (rename (fun a : Fin s × ℕ => (f a.1, a.2))) =
  63       (rename (fun a : Fin (s + 1) × ℕ => (g a.1, a.2))).comp (splitSlot i) := by
  64     ext a : 1
  65     simp only [AlgHom.comp_apply, rename_X, splitSlot, aeval_X, hf]
  66     by_cases ha : a.1 = i
  67     · simp only [if_pos ha]
  68       exact (rename_coproductGenerator g i k h₁ h₂ a.2).symm
  69     · simp only [if_neg ha, rename_X, hg a.1 ha]
  70   exact DFunLike.congr_fun he x
  71 
  72 /-- Splitting a slot outside an embedded tensor block only reindexes the block. -/
  73 theorem splitSlot_rename_disjoint {s s' : ℕ} (f : Fin s → Fin s')
  74     (g : Fin s → Fin (s' + 1)) (k : Fin s')
  75     (hf : ∀ a, f a ≠ k)
  76     (hg : ∀ a, (if f a < k then (f a).castSucc else (f a).succ) = g a)
  77     (x : TensorPower s) :
  78     splitSlot k (rename (fun a : Fin s × ℕ => (f a.1, a.2)) x) =
  79       rename (fun a : Fin s × ℕ => (g a.1, a.2)) x := by
  80   have he : (splitSlot k).comp (rename (fun a : Fin s × ℕ => (f a.1, a.2))) =
  81       rename (fun a : Fin s × ℕ => (g a.1, a.2)) := by
  82     ext a : 1
  83     simp [splitSlot, hf, hg]
  84   exact DFunLike.congr_fun he x

  86 /-- Concatenation embeds the two factors in consecutive tensor blocks. -/
  87 theorem cupPolynomial_blocks {s s' : ℕ} (x : TensorPower s) (y : TensorPower s') :
  88     cupPolynomial x y = blockRename 0 (by omega) x * blockRename s (by omega) y := by
  89   simp only [cupPolynomial, blockRename]
  90   congr 1
  91   congr 2
  92   funext a
  93   exact Prod.ext (Fin.ext (by simp)) rfl

  95 /-- Composition adds the starting positions of embedded blocks. -/
  96 theorem blockRename_comp {s m n : ℕ} (a b : ℕ) (ha : a + s ≤ m) (hb : b + m ≤ n)
  97     (x : TensorPower s) :
  98     blockRename b hb (blockRename a ha x) = blockRename (b + a) (by omega) x := by
  99   simp only [blockRename, rename_rename]
 100   congr 2
 101   funext i
 102   apply Prod.ext
 103   · apply Fin.ext
 104     dsimp
 105     omega
 106   · rfl

 140 /-- A coproduct taken inside an embedded block splits that block. -/
 141 theorem splitSlot_blockRename_inside {s n : ℕ} (a : ℕ) (ha : a + s ≤ n)
 142     (i : Fin s) (x : TensorPower s) :
 143     splitSlot (⟨a + i.val, by have := i.isLt; omega⟩ : Fin n) (blockRename a ha x) =
 144       blockRename a (by omega) (splitSlot i x) := by
 145   refine splitSlot_rename
 146     (fun j : Fin s => (⟨a + j.val, by have := j.isLt; omega⟩ : Fin n))
 147     (fun j : Fin (s + 1) => (⟨a + j.val, by have := j.isLt; omega⟩ : Fin (n + 1)))
 148     i ⟨a + i.val, by have := i.isLt; omega⟩ ?_ ?_ ?_ ?_ x
 149   · intro j
 150     simp only [Fin.ext_iff]
 151     omega
 152   · apply Fin.ext; rfl
 153   · apply Fin.ext; dsimp; omega
 154   · intro j hj
 155     have hlt : (⟨a + j.val, by have := j.isLt; omega⟩ : Fin n) <
 156         ⟨a + i.val, by have := i.isLt; omega⟩ ↔ j < i := by
 157       change a + j.val < a + i.val ↔ j.val < i.val
 158       omega
 159     simp only [hlt]
 160     split_ifs
 161     · apply Fin.ext; rfl
 162     · apply Fin.ext; dsimp; omega

 164 /-- A coproduct preceding a block shifts the whole block one slot to the right. -/
 165 theorem splitSlot_blockRename_before {s n : ℕ} (a : ℕ) (ha : a + s ≤ n)
 166     (i : Fin n) (hi : i.val < a) (x : TensorPower s) :
 167     splitSlot i (blockRename a ha x) = blockRename (a + 1) (by omega) x := by
 168   refine splitSlot_rename_disjoint
 169     (fun j : Fin s => (⟨a + j.val, by have := j.isLt; omega⟩ : Fin n))
 170     (fun j : Fin s => (⟨a + 1 + j.val, by have := j.isLt; omega⟩ : Fin (n + 1)))
 171     i ?_ ?_ x
 172   · intro j h
 173     have := congrArg Fin.val h
 174     dsimp at this
 175     omega
 176   · intro j
 177     have h : ¬(⟨a + j.val, by have := j.isLt; omega⟩ : Fin n) < i := by
 178       change ¬a + j.val < i.val
 179       omega
 180     rw [if_neg h]
 181     apply Fin.ext
 182     dsimp
 183     omega
 184 
 185 /-- A coproduct following a block leaves its position unchanged. -/
 186 theorem splitSlot_blockRename_after {s n : ℕ} (a : ℕ) (ha : a + s ≤ n)
 187     (i : Fin n) (hi : a + s ≤ i.val) (x : TensorPower s) :
 188     splitSlot i (blockRename a ha x) = blockRename a (by omega) x := by
 189   refine splitSlot_rename_disjoint
 190     (fun j : Fin s => (⟨a + j.val, by have := j.isLt; omega⟩ : Fin n))
 191     (fun j : Fin s => (⟨a + j.val, by have := j.isLt; omega⟩ : Fin (n + 1)))
 192     i ?_ ?_ x
 193   · intro j h
 194     have := congrArg Fin.val h
 195     have := j.isLt
 196     dsimp at *
 197     omega
 198   · intro j
 199     have h : (⟨a + j.val, by have := j.isLt; omega⟩ : Fin n) < i := by
 200       change a + j.val < i.val
 201       have := j.isLt
 202       omega
 203     rw [if_pos h]
 204     rfl

 206 /-- The characteristic-two Leibniz formula, with all terms embedded in the
 207 same tensor power so that no associativity casts are implicit. -/
 208 theorem differentialPolynomial_cup_blocks {s s' : ℕ} (x : TensorPower s) (y : TensorPower s') :
 209     differentialPolynomial (s + s') (cupPolynomial x y) =
 210       blockRename (n := s + s' + 1) 0 (by omega) (differentialPolynomial s x) *
 211         blockRename (s + 1) (by omega) y +
 212       blockRename 0 (by omega) x *
 213         blockRename s (by omega) (differentialPolynomial s' y) := by
 214   have hLL (i : Fin s) :
 215       splitSlot (i.castAdd s') (blockRename (n := s + s') 0 (by omega) x) =
 216         blockRename 0 (by omega) (splitSlot i x) := by
 217     have hk : (⟨0 + i.val, by have := i.isLt; omega⟩ : Fin (s + s')) = i.castAdd s' :=
 218       Fin.ext (by simp)
 219     simpa only [hk] using splitSlot_blockRename_inside 0 (by omega : 0 + s ≤ s + s') i x
 220   have hLR (i : Fin s) :
 221       splitSlot (i.castAdd s') (blockRename (n := s + s') s (by omega) y) =
 222         blockRename (s + 1) (by omega) y :=
 223     splitSlot_blockRename_before s (by omega) (i.castAdd s') i.isLt y
 224   have hRL (i : Fin s') :
 225       splitSlot (i.natAdd s) (blockRename (n := s + s') 0 (by omega) x) =
 226         blockRename 0 (by omega) x :=
 227     splitSlot_blockRename_after 0 (by omega) (i.natAdd s) (by dsimp; omega) x
 228   have hRR (i : Fin s') :
 229       splitSlot (i.natAdd s) (blockRename (n := s + s') s (by omega) y) =
 230         blockRename s (by omega) (splitSlot i y) :=
 231     splitSlot_blockRename_inside s (by omega) i y
 232   rw [cupPolynomial_blocks]
 233   simp only [differentialPolynomial, LinearMap.add_apply, LinearMap.sum_apply,
 234     AlgHom.toLinearMap_apply, map_add, map_sum, map_mul,
 235     insertLeft_blockRename, insertRight_blockRename, blockRename_insertLeft,
 236     blockRename_insertRight, Fin.sum_univ_add, hLL, hLR, hRL, hRR,
 237     add_mul, mul_add, Finset.sum_mul, Finset.mul_sum]
 238   abel_nf
 239   simp [CharTwo.two_eq_zero]
 240 
 241 end
