  24 /-- The coefficient field of this fixed cobar construction. -/
  25 abbrev F2 := ZMod 2
  26 
  27 /- Adapted from KIP126/Def/Steenrod/MilnorCobar/Polynomial/Data.lean. -/
  28 noncomputable section
  29 
  30 open MvPolynomial
  31 open scoped BigOperators
  32 
  33 /-- The `s`fold tensor power of the polynomial dual Steenrod algebra.
  34 The generator `(slot,j)` denotes `ξ_(j+1)` in that slot. -/
  35 abbrev TensorPower (s : ℕ) := MvPolynomial (Fin s × ℕ) F2
  36 
  37 /-- Milnor's grading, independent of the tensor slot. -/
  38 def weight {s : ℕ} (a : Fin s × ℕ) : ℕ := 2 ^ (a.2 + 1) - 1
  39 
  40 /-- Include the convention `ξ₀ = 1` in the polynomial formula. -/
  41 def xi {s : ℕ} (slot : Fin s) : ℕ → TensorPower s
  42   | 0 => 1
  43   | j + 1 => MvPolynomial.X (slot, j)
  44 
  45 /-- The Milnor coproduct, inserted at the specified adjacent tensor slots. -/
  46 def coproductGenerator {s : ℕ} (slot : Fin s) (j : ℕ) : TensorPower (s + 1) :=
  47   ∑ i ∈ Finset.range (j + 2),
  48     xi slot.castSucc (j + 1 - i) ^ (2 ^ i) * xi slot.succ i
  49 
  50 /-- Apply the coproduct in one tensor slot. -/
  51 def splitSlot {s : ℕ} (slot : Fin s) : TensorPower s →ₐ[F2] TensorPower (s + 1) :=
  52   MvPolynomial.aeval fun a =>
  53     if a.1 = slot then coproductGenerator slot a.2
  54     else MvPolynomial.X (if a.1 < slot then a.1.castSucc else a.1.succ, a.2)
  55 
  56 /-- Insert the coaugmentation at the left end. -/
  57 def insertLeft (s : ℕ) : TensorPower s →ₐ[F2] TensorPower (s + 1) :=
  58   MvPolynomial.rename fun a => (a.1.succ, a.2)
  59 
  60 /-- Insert the coaugmentation at the right end. -/
  61 def insertRight (s : ℕ) : TensorPower s →ₐ[F2] TensorPower (s + 1) :=
  62   MvPolynomial.rename fun a => (a.1.castSucc, a.2)
  63 
  64 /-- Augment one slot by sending all its positive-degree generators to zero. -/
  65 def augmentSlot {s : ℕ} (slot : Fin s) : TensorPower s →ₐ[F2] TensorPower s :=
  66   MvPolynomial.aeval fun a => if a.1 = slot then 0 else MvPolynomial.X a
  67 
  68 /-- Homogeneous normalized cobar cochains. -/
  69 def cochains (s t : ℕ) : Submodule F2 (TensorPower s) :=
  70   MvPolynomial.weightedHomogeneousSubmodule F2 weight t ⊓
  71     ⨅ slot : Fin s, LinearMap.ker (augmentSlot slot).toLinearMap
  72 
  73 /-- The characteristic-two cobar differential before restriction to normalized cochains. -/
  74 def differentialPolynomial (s : ℕ) : TensorPower s →ₗ[F2] TensorPower (s + 1) :=
  75   (insertLeft s).toLinearMap + (insertRight s).toLinearMap +
  76     ∑ slot : Fin s, (splitSlot slot).toLinearMap
  77 
  78 /-- Concatenation of tensor words, expressed by disjoint variable renamings. -/
  79 def cupPolynomial {s s' : ℕ} (x : TensorPower s) (y : TensorPower s') :
  80     TensorPower (s + s') :=
  81   MvPolynomial.rename (fun a : Fin s × ℕ => (a.1.castAdd s', a.2)) x *
  82     MvPolynomial.rename (fun a : Fin s' × ℕ => (a.1.natAdd s, a.2)) y
  83 
  84 /-- The polynomial representative `[ξ₁^64]`. -/
  85 def h6Polynomial : TensorPower 1 := MvPolynomial.X (0, 0) ^ 64
  86 
  87 end
