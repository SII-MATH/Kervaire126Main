  29 /-- The coproduct differential preserves normalization and internal degree. -/
  30 theorem differentialPolynomial_mem (s t : ℕ) (x : cochains s t) :
  31     IsCochain t (differentialPolynomial s x) := by
  32   constructor
  33   · change IsWeightedHomogeneous weight (differentialPolynomial s x.val) t
  34     simp only [differentialPolynomial, LinearMap.add_apply, LinearMap.sum_apply,
  35       AlgHom.toLinearMap_apply]
  36     exact ((homogeneous_rename_slots Fin.succ x.property.1).add
  37       (homogeneous_rename_slots Fin.castSucc x.property.1)).add
  38       (IsWeightedHomogeneous.sum Finset.univ (fun slot => splitSlot slot x.val) t
  39         (fun slot _ => homogeneous_splitSlot slot x.property.1))
  40   · change differentialPolynomial s x.val ∈
  41       (⨅ slot : Fin (s + 1), LinearMap.ker (augmentSlot slot).toLinearMap)
  42     rw [Submodule.mem_iInf]
  43     intro slot
  44     exact differentialPolynomial_normalized x.val
  45       (fun i => (Submodule.mem_iInf _).mp x.property.2 i) slot
  46 
  47 /-- Concatenation preserves normalization and adds the two internal degrees. -/
  48 theorem cupPolynomial_mem {s s' t t' : ℕ} (x : cochains s t) (y : cochains s' t') :
  49     IsCochain (t + t') (cupPolynomial x.val y.val) := by
  50   constructor
  51   · exact (homogeneous_rename_slots (Fin.castAdd s') x.property.1).mul
  52       (homogeneous_rename_slots (Fin.natAdd s) y.property.1)
  53   · change cupPolynomial x.val y.val ∈
  54       (⨅ slot : Fin (s + s'), LinearMap.ker (augmentSlot slot).toLinearMap)
  55     rw [Submodule.mem_iInf]
  56     intro slot
  57     refine Fin.addCases (fun i => ?_) (fun i => ?_) slot
  58     · change augmentSlot (i.castAdd s') (cupPolynomial x.val y.val) = 0
  59       have hx : augmentSlot i x.val = 0 := (Submodule.mem_iInf _).mp x.property.2 i
  60       rw [cupPolynomial, map_mul,
  61         augmentSlot_rename (Fin.castAdd s') (Fin.castAdd_injective s s'), hx, map_zero, zero_mul]
  62     · change augmentSlot (i.natAdd s) (cupPolynomial x.val y.val) = 0
  63       have hy : augmentSlot i y.val = 0 := (Submodule.mem_iInf _).mp y.property.2 i
  64       rw [cupPolynomial, map_mul,
  65         augmentSlot_rename (Fin.natAdd s) (Fin.natAdd_injective s' s), hy, map_zero, mul_zero]

  67 theorem h6Polynomial_mem : IsCochain 64 h6Polynomial := by
  68   constructor
  69   · simpa [h6Polynomial, weight] using
  70       (MvPolynomial.isWeightedHomogeneous_X (R := F2)
  71         (@weight 1) (0, 0)).pow 64
  72   · change h6Polynomial ∈ (⨅ slot : Fin 1, LinearMap.ker (augmentSlot slot).toLinearMap)
  73     rw [Submodule.mem_iInf]
  74     intro slot
  75     have hslot : slot = 0 := Subsingleton.elim _ _
  76     subst slot
  77     simp [LinearMap.mem_ker, augmentSlot, h6Polynomial]

  79 /-- The primitive power representative is closed, independently of the
  80 restriction of the differential to normalized cochains. -/
  81 theorem h6Polynomial_differential : differentialPolynomial 1 h6Polynomial = 0 := by
  82   have hp (x y : TensorPower 2) : (x + y) ^ 64 = x ^ 64 + y ^ 64 :=
  83     add_pow_char_pow x y 2 6
  84   simp [differentialPolynomial, h6Polynomial, insertLeft, insertRight,
  85     splitSlot, coproductGenerator, xi, Finset.sum_range_succ, hp,
  86     add_assoc, CharTwo.add_self_eq_zero, CharTwo.add_cancel_left]
  87 
  88 /-- The concatenated square is closed at the polynomial level. -/
  89 theorem h6SquarePolynomial_differential :
  90     differentialPolynomial 2 (cupPolynomial h6Polynomial h6Polynomial) = 0 := by
  91   have hp (x y : TensorPower 3) : (x + y) ^ 64 = x ^ 64 + y ^ 64 :=
  92     add_pow_char_pow x y 2 6
  93   simp [differentialPolynomial, cupPolynomial, h6Polynomial, insertLeft, insertRight,
  94     splitSlot, coproductGenerator, xi, Fin.sum_univ_succ, Finset.sum_range_succ, hp]
  95   ring_nf
  96   simp [CharTwo.two_eq_zero]

  98 end
