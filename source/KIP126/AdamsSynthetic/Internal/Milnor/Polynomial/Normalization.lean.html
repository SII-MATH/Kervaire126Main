  28 /-- Augmenting a renamed slot agrees with renaming after augmentation. -/
  29 theorem augmentSlot_rename {s s' : ℕ} (f : Fin s → Fin s')
  30     (hf : Function.Injective f) (slot : Fin s) (x : TensorPower s) :
  31     augmentSlot (f slot) (rename (fun a : Fin s × ℕ => (f a.1, a.2)) x) =
  32       rename (fun a : Fin s × ℕ => (f a.1, a.2)) (augmentSlot slot x) := by
  33   have h : (augmentSlot (f slot)).comp (rename (fun a : Fin s × ℕ => (f a.1, a.2))) =
  34       (rename (fun a : Fin s × ℕ => (f a.1, a.2))).comp (augmentSlot slot) := by
  35     ext a : 1
  36     by_cases ha : a.1 = slot <;> simp [augmentSlot, hf.eq_iff, ha]
  37   exact DFunLike.congr_fun h x

  39 @[simp] theorem augmentSlot_xi_same {s : ℕ} (slot : Fin s) (j : ℕ) :
  40     augmentSlot slot (xi slot j) = if j = 0 then 1 else 0 := by
  41   cases j <;> simp [xi, augmentSlot]

  43 theorem augmentSlot_xi_of_ne {s : ℕ} (slot other : Fin s) (h : other ≠ slot) (j : ℕ) :
  44     augmentSlot slot (xi other j) = xi other j := by
  45   cases j <;> simp [xi, augmentSlot, h]

  47 /-- The left counit identity on a Milnor generator. -/
  48 theorem augmentSlot_coproduct_left {s : ℕ} (slot : Fin s) (j : ℕ) :
  49     augmentSlot slot.castSucc (coproductGenerator slot j) = X (slot.succ, j) := by
  50   have hne : slot.succ ≠ slot.castSucc := by
  51     intro h
  52     have := congrArg Fin.val h
  53     simp only [Fin.val_succ, Fin.val_castSucc] at this
  54     omega
  55   simp only [coproductGenerator, map_sum, map_mul, map_pow,
  56     augmentSlot_xi_same, augmentSlot_xi_of_ne _ _ hne]
  57   rw [Finset.sum_eq_single (j + 1)]
  58   · simp [xi]
  59   · intro i hi hij
  60     have hi' : i < j + 1 := by have := Finset.mem_range.mp hi; omega
  61     simp [Nat.sub_ne_zero_of_lt hi']
  62   · intro h
  63     exact (h (Finset.mem_range.mpr (by omega))).elim

  65 /-- The right counit identity on a Milnor generator. -/
  66 theorem augmentSlot_coproduct_right {s : ℕ} (slot : Fin s) (j : ℕ) :
  67     augmentSlot slot.succ (coproductGenerator slot j) = X (slot.castSucc, j) := by
  68   have hne : slot.castSucc ≠ slot.succ := by
  69     intro h
  70     have := congrArg Fin.val h
  71     simp only [Fin.val_succ, Fin.val_castSucc] at this
  72     omega
  73   simp only [coproductGenerator, map_sum, map_mul, map_pow,
  74     augmentSlot_xi_same, augmentSlot_xi_of_ne _ _ hne]
  75   rw [Finset.sum_eq_single 0]
  76   · simp [xi]
  77   · intro i hi hi0
  78     simp [hi0]
  79   · intro h
  80     exact (h (Finset.mem_range.mpr (by omega))).elim

  82 /-- Augmenting a disjoint slot leaves a coproduct generator unchanged. -/
  83 theorem augmentSlot_coproduct_other {s : ℕ} (slot : Fin s) (other : Fin (s + 1))
  84     (h₁ : slot.castSucc ≠ other) (h₂ : slot.succ ≠ other) (j : ℕ) :
  85     augmentSlot other (coproductGenerator slot j) = coproductGenerator slot j := by
  86   simp only [coproductGenerator, map_sum, map_mul, map_pow,
  87     augmentSlot_xi_of_ne _ _ h₁, augmentSlot_xi_of_ne _ _ h₂]

  89 /-- Left augmentation of a split slot inserts a unit in that slot. -/
  90 theorem augmentSlot_splitSlot_left {s : ℕ} (slot : Fin s) (x : TensorPower s) :
  91     augmentSlot slot.castSucc (splitSlot slot x) =
  92       rename (fun a : Fin s × ℕ => (slot.castSucc.succAbove a.1, a.2)) x := by
  93   have he : (augmentSlot slot.castSucc).comp (splitSlot slot) =
  94       rename (fun a : Fin s × ℕ => (slot.castSucc.succAbove a.1, a.2)) := by
  95     ext a : 1
  96     simp only [AlgHom.comp_apply, splitSlot, aeval_X, rename_X]
  97     by_cases ha : a.1 = slot
  98     · simp [ha, augmentSlot_coproduct_left]
  99     · rw [if_neg ha]
 100       by_cases hlt : a.1 < slot
 101       · have hne : a.1.castSucc ≠ slot.castSucc := fun h => ha (Fin.castSucc_injective _ h)
 102         simp [hlt, augmentSlot, hne, Fin.succAbove_castSucc_of_lt slot a.1 hlt]
 103       · have hne : a.1.succ ≠ slot.castSucc := by
 104           intro h
 105           have := congrArg Fin.val h
 106           simp only [Fin.val_succ, Fin.val_castSucc] at this
 107           have := Fin.not_lt.mp hlt
 108           omega
 109         simp [hlt, augmentSlot, hne,
 110           Fin.succAbove_castSucc_of_le slot a.1 (Fin.not_lt.mp hlt)]
 111   exact DFunLike.congr_fun he x

 113 /-- Right augmentation of a split slot inserts a unit after that slot. -/
 114 theorem augmentSlot_splitSlot_right {s : ℕ} (slot : Fin s) (x : TensorPower s) :
 115     augmentSlot slot.succ (splitSlot slot x) =
 116       rename (fun a : Fin s × ℕ => (slot.succ.succAbove a.1, a.2)) x := by
 117   have he : (augmentSlot slot.succ).comp (splitSlot slot) =
 118       rename (fun a : Fin s × ℕ => (slot.succ.succAbove a.1, a.2)) := by
 119     ext a : 1
 120     simp only [AlgHom.comp_apply, splitSlot, aeval_X, rename_X]
 121     by_cases ha : a.1 = slot
 122     · simp [ha, augmentSlot_coproduct_right]
 123     · rw [if_neg ha]
 124       by_cases hlt : a.1 < slot
 125       · have hne : a.1.castSucc ≠ slot.succ := by
 126           intro h
 127           have := congrArg Fin.val h
 128           simp only [Fin.val_succ, Fin.val_castSucc] at this
 129           omega
 130         simp [hlt, augmentSlot, hne, Fin.succAbove_succ_of_le slot a.1 (Fin.le_of_lt hlt)]
 131       · have hgt : slot < a.1 := lt_of_le_of_ne (Fin.not_lt.mp hlt) (Ne.symm ha)
 132         have hne : a.1.succ ≠ slot.succ := fun h => ha (Fin.succ_injective _ h)
 133         simp [hlt, augmentSlot, hne, Fin.succAbove_succ_of_lt slot a.1 hgt]
 134   exact DFunLike.congr_fun he x

 136 /-- Augmentation in an earlier slot commutes with splitting a later slot. -/
 137 theorem augmentSlot_splitSlot_of_lt {s : ℕ} (slot other : Fin s) (hlt : other < slot)
 138     (x : TensorPower s) :
 139     augmentSlot other.castSucc (splitSlot slot x) = splitSlot slot (augmentSlot other x) := by
 140   have he : (augmentSlot other.castSucc).comp (splitSlot slot) =
 141       (splitSlot slot).comp (augmentSlot other) := by
 142     ext a : 1
 143     by_cases ha : a.1 = slot
 144     · have h₁ : slot.castSucc ≠ other.castSucc := by
 145         intro h; have := Fin.castSucc_injective _ h; subst other; exact (lt_irrefl _ hlt)
 146       have h₂ : slot.succ ≠ other.castSucc := by
 147         intro h; have := congrArg Fin.val h
 148         simp only [Fin.val_succ, Fin.val_castSucc] at this
 149         omega
 150       have hso : slot ≠ other := ne_of_gt hlt
 151       simp only [AlgHom.comp_apply, splitSlot, augmentSlot, aeval_X, ha, if_pos,
 152         hso, if_false]
 153       exact augmentSlot_coproduct_other slot other.castSucc h₁ h₂ a.2
 154     · have hiff : (if a.1 < slot then a.1.castSucc else a.1.succ) = other.castSucc ↔ a.1 = other := by
 155         split_ifs with h
 156         · exact Fin.castSucc_inj
 157         · constructor
 158           · intro heq
 159             have := congrArg Fin.val heq
 160             simp only [Fin.val_succ, Fin.val_castSucc] at this
 161             have := Fin.not_lt.mp h
 162             omega
 163           · intro heq; exact (h (heq ▸ hlt)).elim
 164       by_cases hao : a.1 = other <;>
 165         simp [AlgHom.comp_apply, splitSlot, augmentSlot, ha, hiff, hao,
 166           hlt, ne_of_lt hlt]
 167   exact DFunLike.congr_fun he x

 169 /-- Augmentation in a later slot commutes with splitting an earlier slot. -/
 170 theorem augmentSlot_splitSlot_of_gt {s : ℕ} (slot other : Fin s) (hlt : slot < other)
 171     (x : TensorPower s) :
 172     augmentSlot other.succ (splitSlot slot x) = splitSlot slot (augmentSlot other x) := by
 173   have he : (augmentSlot other.succ).comp (splitSlot slot) =
 174       (splitSlot slot).comp (augmentSlot other) := by
 175     ext a : 1
 176     by_cases ha : a.1 = slot
 177     · have h₁ : slot.castSucc ≠ other.succ := by
 178         intro h; have := congrArg Fin.val h
 179         simp only [Fin.val_succ, Fin.val_castSucc] at this
 180         omega
 181       have h₂ : slot.succ ≠ other.succ := by
 182         intro h; have := Fin.succ_injective _ h; subst other; exact (lt_irrefl _ hlt)
 183       have hso : slot ≠ other := ne_of_lt hlt
 184       simp only [AlgHom.comp_apply, splitSlot, augmentSlot, aeval_X, ha, if_pos,
 185         hso, if_false]
 186       exact augmentSlot_coproduct_other slot other.succ h₁ h₂ a.2
 187     · have hiff : (if a.1 < slot then a.1.castSucc else a.1.succ) = other.succ ↔ a.1 = other := by
 188         split_ifs with h
 189         · constructor
 190           · intro heq
 191             have := congrArg Fin.val heq
 192             simp only [Fin.val_succ, Fin.val_castSucc] at this
 193             omega
 194           · intro heq; exact (not_lt_of_ge (le_of_lt hlt) (heq ▸ h)).elim
 195         · exact Fin.succ_inj
 196       by_cases hao : a.1 = other <;>
 197         simp [AlgHom.comp_apply, splitSlot, augmentSlot, ha, hiff, hao,
 198           ne_of_gt hlt, not_lt_of_ge (le_of_lt hlt)]
 199   exact DFunLike.congr_fun he x

 201 /-- The two adjacent cofaces give the only surviving terms after augmentation. -/
 202 theorem augmentSlot_splitSlot_normalized {s : ℕ} (x : TensorPower s)
 203     (hx : ∀ slot, augmentSlot slot x = 0) (j : Fin (s + 1)) (i : Fin s) :
 204     augmentSlot j (splitSlot i x) =
 205       (if j = i.castSucc then rename (fun a : Fin s × ℕ => (j.succAbove a.1, a.2)) x else 0) +
 206       (if j = i.succ then rename (fun a : Fin s × ℕ => (j.succAbove a.1, a.2)) x else 0) := by
 207   by_cases h₁ : j = i.castSucc
 208   · subst j
 209     have hne : i.castSucc ≠ i.succ := by
 210       intro h; have := congrArg Fin.val h; simp only [Fin.val_castSucc, Fin.val_succ] at this; omega
 211     simp [hne, augmentSlot_splitSlot_left]
 212   · by_cases h₂ : j = i.succ
 213     · subst j
 214       simp [h₁, augmentSlot_splitSlot_right]
 215     · simp only [if_neg h₁, if_neg h₂, zero_add]
 216       by_cases hlt : j.val < i.val
 217       · let k : Fin s := ⟨j.val, lt_trans hlt i.isLt⟩
 218         have hj : j = k.castSucc := Fin.ext rfl
 219         rw [hj, augmentSlot_splitSlot_of_lt i k hlt, hx, map_zero]
 220       · have hji : j.val ≠ i.val := fun h => h₁ (Fin.ext h)
 221         have hji' : j.val ≠ i.val + 1 := fun h => h₂ (Fin.ext h)
 222         have hjpos : 0 < j.val := by omega
 223         let k : Fin s := ⟨j.val - 1, by have := j.isLt; omega⟩
 224         have hj : j = k.succ := Fin.ext (by dsimp [k]; omega)
 225         have hk : i < k := by change i.val < j.val - 1; omega
 226         rw [hj, augmentSlot_splitSlot_of_gt i k hk, hx, map_zero]

 228 /-- Renaming into slots disjoint from the augmented slot is unchanged. -/
 229 theorem augmentSlot_rename_disjoint {s s' : ℕ} (f : Fin s → Fin s') (slot : Fin s')
 230     (hf : ∀ i, f i ≠ slot) (x : TensorPower s) :
 231     augmentSlot slot (rename (fun a : Fin s × ℕ => (f a.1, a.2)) x) =
 232       rename (fun a : Fin s × ℕ => (f a.1, a.2)) x := by
 233   have he : (augmentSlot slot).comp (rename (fun a : Fin s × ℕ => (f a.1, a.2))) =
 234       rename (fun a : Fin s × ℕ => (f a.1, a.2)) := by
 235     ext a : 1
 236     simp [augmentSlot, hf]
 237   exact DFunLike.congr_fun he x

 239 /-- Only augmentation at the newly inserted left endpoint survives. -/
 240 theorem augmentSlot_insertLeft_normalized {s : ℕ} (x : TensorPower s)
 241     (hx : ∀ slot, augmentSlot slot x = 0) (j : Fin (s + 1)) :
 242     augmentSlot j (insertLeft s x) =
 243       if j = 0 then rename (fun a : Fin s × ℕ => (j.succAbove a.1, a.2)) x else 0 := by
 244   refine Fin.cases ?_ (fun i => ?_) j
 245   · simpa [insertLeft] using augmentSlot_rename_disjoint Fin.succ 0 (fun i => Fin.succ_ne_zero i) x
 246   · simp only [insertLeft, augmentSlot_rename Fin.succ (Fin.succ_injective s), hx, map_zero,
 247       Fin.succ_ne_zero, if_false]
 248 
 249 /-- Only augmentation at the newly inserted right endpoint survives. -/
 250 theorem augmentSlot_insertRight_normalized {s : ℕ} (x : TensorPower s)
 251     (hx : ∀ slot, augmentSlot slot x = 0) (j : Fin (s + 1)) :
 252     augmentSlot j (insertRight s x) =
 253       if j = Fin.last s then rename (fun a : Fin s × ℕ => (j.succAbove a.1, a.2)) x else 0 := by
 254   refine Fin.lastCases ?_ (fun i => ?_) j
 255   · simpa [insertRight] using augmentSlot_rename_disjoint Fin.castSucc (Fin.last s)
 256       (fun i => Fin.castSucc_ne_last i) x
 257   · simp only [insertRight, augmentSlot_rename Fin.castSucc (Fin.castSucc_injective s), hx, map_zero,
 258       Fin.castSucc_ne_last, if_false]

 260 /-- The cobar differential preserves the intersection of augmentation kernels. -/
 261 theorem differentialPolynomial_normalized {s : ℕ} (x : TensorPower s)
 262     (hx : ∀ slot, augmentSlot slot x = 0) (j : Fin (s + 1)) :
 263     augmentSlot j (differentialPolynomial s x) = 0 := by
 264   classical
 265   let q := rename (fun a : Fin s × ℕ => (j.succAbove a.1, a.2)) x
 266   have hleft : (if j = 0 then q else 0) + ∑ i : Fin s, (if j = i.succ then q else 0) = q := by
 267     rw [← Fin.sum_univ_succ (fun k : Fin (s + 1) => if j = k then q else 0)]
 268     simp
 269   have hright : (∑ i : Fin s, (if j = i.castSucc then q else 0)) +
 270       (if j = Fin.last s then q else 0) = q := by
 271     rw [← Fin.sum_univ_castSucc (fun k : Fin (s + 1) => if j = k then q else 0)]
 272     simp
 273   simp only [differentialPolynomial, LinearMap.add_apply, LinearMap.sum_apply,
 274     AlgHom.toLinearMap_apply, map_add, map_sum,
 275     augmentSlot_insertLeft_normalized x hx, augmentSlot_insertRight_normalized x hx,
 276     augmentSlot_splitSlot_normalized x hx, Finset.sum_add_distrib]
 277   change (if j = 0 then q else 0) + (if j = Fin.last s then q else 0) +
 278     ((∑ i : Fin s, if j = i.castSucc then q else 0) +
 279       ∑ i : Fin s, if j = i.succ then q else 0) = 0
 280   calc
 281     _ = ((∑ i : Fin s, if j = i.castSucc then q else 0) + (if j = Fin.last s then q else 0)) +
 282         ((if j = 0 then q else 0) + ∑ i : Fin s, if j = i.succ then q else 0) := by abel
 283     _ = q + q := by rw [hleft, hright]
 284     _ = 0 := CharTwo.add_self_eq_zero q
