  36 /-- Renaming tensor slots does not change internal degree. -/
  37 theorem homogeneous_rename_slots {s s' t : ℕ} (f : Fin s → Fin s')
  38     {x : TensorPower s} (hx : IsWeightedHomogeneous weight x t) :
  39     IsWeightedHomogeneous weight (rename (fun a : Fin s × ℕ => (f a.1, a.2)) x) t := by
  40   induction hx using IsWeightedHomogeneous.induction_on with
  41   | zero => simpa using isWeightedHomogeneous_zero F2 (@weight s') t
  42   | add x y hx hy ihx ihy => simpa using ihx.add ihy
  43   | monomial d r hd =>
  44     rw [rename_monomial]
  45     apply isWeightedHomogeneous_monomial
  46     rw [← hd]
  47     simp only [Finsupp.weight_apply, weight, smul_eq_mul]
  48     rw [Finsupp.sum_mapDomain_index (by simp) (by intros; simp [add_mul])]
  49 
  50 /-- Substitution of homogeneous polynomials of the prescribed generator
  51 weights preserves the weighted degree. -/
  52 theorem homogeneous_aeval {σ τ : Type*} {R : Type*} [CommSemiring R]
  53     (w : σ → ℕ) (w' : τ → ℕ) (f : σ → MvPolynomial τ R)
  54     (hf : ∀ a, IsWeightedHomogeneous w' (f a) (w a))
  55     {x : MvPolynomial σ R} {t : ℕ} (hx : IsWeightedHomogeneous w x t) :
  56     IsWeightedHomogeneous w' (aeval f x) t := by
  57   classical
  58   induction hx using IsWeightedHomogeneous.induction_on with
  59   | zero => simpa using isWeightedHomogeneous_zero R w' t
  60   | add x y hx hy ihx ihy => simpa using ihx.add ihy
  61   | monomial d r hd =>
  62     rw [aeval_monomial]
  63     have h := (IsWeightedHomogeneous.prod d.support
  64       (fun a => f a ^ d a) (fun a => d a * w a)
  65       (fun a _ => by simpa [nsmul_eq_mul] using (hf a).pow (d a))).C_mul r
  66     have hd' : ∑ a ∈ d.support, d a * w a = t := by
  67       simpa [Finsupp.weight_apply, Finsupp.sum, smul_eq_mul] using hd
  68     simpa only [Finsupp.prod, hd', MvPolynomial.algebraMap_eq] using h
  69 
  70 /-- The convention `ξ₀ = 1` respects Milnor's grading. -/
  71 theorem homogeneous_xi {s : ℕ} (slot : Fin s) (j : ℕ) :
  72     IsWeightedHomogeneous weight (xi slot j) (2 ^ j - 1) := by
  73   cases j with
  74   | zero => simpa [xi] using isWeightedHomogeneous_one F2 (@weight s)
  75   | succ j => exact isWeightedHomogeneous_X (R := F2) weight (slot, j)
  76 
  77 /-- Each summand of Milnor's coproduct has the original generator's degree. -/
  78 theorem homogeneous_coproductGenerator {s : ℕ} (slot : Fin s) (j : ℕ) :
  79     IsWeightedHomogeneous weight (coproductGenerator slot j) (2 ^ (j + 1) - 1) := by
  80   apply IsWeightedHomogeneous.sum
  81   intro i hi
  82   have hi' : i ≤ j + 1 := by
  83     have := Finset.mem_range.mp hi
  84     omega
  85   have he : (2 ^ i) * (2 ^ (j + 1 - i) - 1) + (2 ^ i - 1) = 2 ^ (j + 1) - 1 := by
  86     have hp : 2 ^ (j + 1 - i) * 2 ^ i = 2 ^ (j + 1) := by
  87       rw [← pow_add, Nat.sub_add_cancel hi']
  88     have h₁ : 1 ≤ 2 ^ (j + 1 - i) := Nat.one_le_pow _ _ (by omega)
  89     have h₂ : 1 ≤ 2 ^ i := Nat.one_le_pow _ _ (by omega)
  90     have h₃ : 1 ≤ 2 ^ (j + 1) := Nat.one_le_pow _ _ (by omega)
  91     nlinarith [Nat.sub_add_cancel h₁, Nat.sub_add_cancel h₂, Nat.sub_add_cancel h₃]
  92   simpa only [nsmul_eq_mul, Nat.cast_id, he] using
  93     ((homogeneous_xi slot.castSucc (j + 1 - i)).pow (2 ^ i)).mul
  94       (homogeneous_xi slot.succ i)
  95 
  96 /-- Splitting a tensor slot by the coproduct preserves internal degree. -/
  97 theorem homogeneous_splitSlot {s t : ℕ} (slot : Fin s) {x : TensorPower s}
  98     (hx : IsWeightedHomogeneous weight x t) :
  99     IsWeightedHomogeneous weight (splitSlot slot x) t := by
 100   apply homogeneous_aeval weight weight _ _ hx
 101   intro a
 102   by_cases h : a.1 = slot
 103   · simpa only [if_pos h, weight] using homogeneous_coproductGenerator slot a.2
 104   · simp only [if_neg h]
 105     exact isWeightedHomogeneous_X (R := F2) weight _
 106 
 107 end
