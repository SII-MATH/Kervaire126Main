  26 /-- The actual normalized cobar differential. -/
  27 def differential (s t : ℕ) : cochains s t →ₗ[F2] cochains (s + 1) t :=
  28   ((differentialPolynomial s).comp (cochains s t).subtype).codRestrict _
  29     (differentialPolynomial_mem s t)
  30 
  31 /-- The cochain product used to define the square. -/
  32 def cup {s s' t t' : ℕ} (x : cochains s t) (y : cochains s' t') :
  33     cochains (s + s') (t + t') :=
  34   ⟨cupPolynomial x.val y.val, cupPolynomial_mem x y⟩
  35 
  36 /-- The standard normalized cocycle representing `h₆`. -/
  37 def h6Cochain : cochains 1 64 := ⟨h6Polynomial, h6Polynomial_mem⟩
  38 
  39 /-- The square is the actual cochain concatenation product. -/
  40 def h6SquareCochain : cochains 2 128 := cup h6Cochain h6Cochain
  41 
  42 end

  47 /-- A normalized cochain is a cocycle when its cobar differential is zero. -/
  48 def IsCycle {s t : ℕ} (x : cochains s t) : Prop :=
  49   differential s t x = 0
  50 
  51 end

  57     (x x' : cochains s t) (y : cochains s' t') :
  58     cup (x + x') y = cup x y + cup x' y := by
  59   apply Subtype.ext

  93 /-- The canonical linear equivalence between equal cochain bidegrees. -/
  94 def reindex {s s' t t' : ℕ} (hs : s = s') (ht : t = t') :
  95     cochains s t ≃ₗ[F2] cochains s' t' := by
  96   subst s'
  97   subst t'
  98   exact LinearEquiv.refl _ _
  99 
 100 end

 126 /-- Concatenation satisfies the Leibniz identity over `𝔽₂`. -/
 127 theorem differential_cup {s s' t t' : ℕ} (x : cochains s t) (y : cochains s' t') :
 128     differential (s + s') (t + t') (cup x y) =
 129       reindex (by omega) rfl (cup (differential s t x) y) +
 130       reindex (by omega) rfl (cup x (differential s' t' y)) := by
 131   apply Subtype.ext
 132   change differentialPolynomial (s + s') (cupPolynomial x.val y.val) =
 133     (reindex _ _ (cup (differential s t x) y)).val +
 134     (reindex _ _ (cup x (differential s' t' y))).val
 135   rw [reindex_cup_val, reindex_cup_val]
 136   exact differentialPolynomial_cup_blocks x.val y.val

 148 /-- Products of cocycles are cocycles. -/
 149 theorem cup_isCycle {s s' t t' : ℕ} (x : cochains s t) (y : cochains s' t')
 150     (hx : IsCycle x) (hy : IsCycle y) : IsCycle (cup x y) := by
 151   change differential _ _ (cup x y) = 0
 152   rw [differential_cup, hx, hy, cup_zero_left, cup_zero_right, map_zero, map_zero, zero_add]
 153 
 154 end
