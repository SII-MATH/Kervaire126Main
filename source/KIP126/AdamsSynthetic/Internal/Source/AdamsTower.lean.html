  10 /-! The coefficient-unit Adams tower is constructed by actual iterated fibers.
  11 The stage and transition maps are not supplied by an implementation record.
  12 The proved rotation comparison identifies the chosen cofiber of a fiber inclusion
  13 with its original target, including the original map out of the source. -/
  14

  18 open CategoryTheory CategoryTheory.MonoidalCategory CategoryTheory.Pretriangulated
  19 open KIP126.StableHomotopy

  21 universe u v

  23 variable {C : Type u} [StableHomotopyCategory.{u, v} C]
  24   [HasFunctorialCofiber (C := C)]
  25 
  26 private def fiber {X Y : C} (f : X ⟶ Y) : C :=
  27   (HasFunctorialCofiber.cofib f)⟦(-1 : ℤ)⟧
  28 
  29 private def fiberι {X Y : C} (f : X ⟶ Y) : fiber f ⟶ X :=
  30   (HasFunctorialCofiber.cofibδ f)⟦(-1 : ℤ)⟧' ≫
  31     (shiftFunctorCompIsoId C (1 : ℤ) (-1 : ℤ) (by omega)).hom.app X
  32 
  33 private def adamsUnit {H : C} (unit : SphereSpectrum ⟶ H) (X : C) : X ⟶ H ⊗ X :=
  34   (λ_ X).inv ≫ unit ▷ X
  35 
  36 private def adamsStage {H : C} (unit : SphereSpectrum ⟶ H) (X : C) : ℕ → C
  37   | 0 => X
  38   | s + 1 => fiber (adamsUnit unit (adamsStage unit X s))
  39 
  40 private def adamsStep {H : C} (unit : SphereSpectrum ⟶ H) (X : C) (s : ℕ) :
  41     adamsStage unit X (s + 1) ⟶ adamsStage unit X s :=
  42   fiberι (adamsUnit unit (adamsStage unit X s))
  43 
  44 private def adamsComposite {H : C} (unit : SphereSpectrum ⟶ H) (X : C) (s : ℕ) :
  45     (length : ℕ) → (adamsStage unit X (s + length) ⟶ adamsStage unit X s)
  46   | 0 => 𝟙 _
  47   | length + 1 => adamsStep unit X (s + length) ≫ adamsComposite unit X s length
  48 
  49 private def adamsMap {H : C} (unit : SphereSpectrum ⟶ H) (X : C)
  50     (s t : ℕ) (h : s ≤ t) : adamsStage unit X t ⟶ adamsStage unit X s :=
  51   eqToHom (congrArg (adamsStage unit X) (by omega : t = s + (t - s))) ≫
  52     adamsComposite unit X s (t - s)
  53 
  54 variable {H : C} (unit : SphereSpectrum ⟶ H) (X : C)
  55 
  56 @[simp] private theorem adamsMap_self (s : ℕ) :
  57     adamsMap unit X s s le_rfl = 𝟙 _ := by
  58   apply eq_of_heq
  59   unfold adamsMap
  60   rw [eqToHom_comp_heq_iff, Nat.sub_self]
  61   rfl
  62 
  63 private theorem adamsMap_succ (s t : ℕ) (h : s ≤ t) :
  64     adamsMap unit X s (t + 1) (by omega) =
  65       adamsStep unit X t ≫ adamsMap unit X s t h := by
  66   obtain ⟨n, rfl⟩ := Nat.exists_eq_add_of_le h
  67   apply eq_of_heq
  68   simp only [adamsMap, eqToHom_comp_heq_iff]
  69   have h : adamsComposite unit X s (s + n + 1 - s) ≍
  70       adamsStep unit X (s + n) ≫ adamsComposite unit X s n := by
  71     have hn : s + n + 1 - s = n + 1 := by omega
  72     rw [hn]
  73     rfl
  74   apply h.trans
  75   apply CategoryTheory.heq_comp rfl rfl rfl (HEq.rfl)
  76   symm
  77   rw [eqToHom_comp_heq_iff, Nat.add_sub_cancel_left]
  78 
  79 private theorem adamsMap_comp (s t z : ℕ) (hst : s ≤ t) (htz : t ≤ z) :
  80     adamsMap unit X t z htz ≫ adamsMap unit X s t hst =
  81       adamsMap unit X s z (by omega) := by
  82   induction z, htz using Nat.le_induction with
  83   | base => simp
  84   | succ z htz ih =>
  85     rw [adamsMap_succ unit X t z htz,
  86       adamsMap_succ unit X s z (by omega), Category.assoc, ih]

  89 /-- The rotation comparison for the actual chosen fiber/cofiber, with the
  90 source-to-target map fixed. The negative first isomorphism retains the
  91 positive convention used in `fiberι`. -/
  92 private theorem fiberLayerComparison_exists {A B : C} (f : A ⟶ B) :
  93     Nonempty {e : HasFunctorialCofiber.cofib (fiberι f) ≅ B //
  94       HasFunctorialCofiber.cofibι (fiberι f) ≫ e.hom = f} := by
  95   let T := (Triangle.mk f (HasFunctorialCofiber.cofibι f)
  96     (HasFunctorialCofiber.cofibδ f)).invRotate
  97   have hT : T ∈ distTriang C :=
  98     inv_rot_of_distTriang _ (HasFunctorialCofiber.cofib_distinguished f)
  99   have hfirst : T.mor₁ = -fiberι f := by
 100     simp only [T, Triangle.invRotate, Triangle.mk, fiberι]
 101     rfl
 102   let e : Triangle.mk (fiberι f) (HasFunctorialCofiber.cofibι (fiberι f))
 103       (HasFunctorialCofiber.cofibδ (fiberι f)) ≅ T :=
 104     isoTriangleOfIso₁₂ _ _ (HasFunctorialCofiber.cofib_distinguished (fiberι f)) hT
 105       (-(Iso.refl _)) (Iso.refl _) (by
 106         simp only [Triangle.mk, hfirst]
 107         change fiberι f ≫ 𝟙 A = (-𝟙 (fiber f)) ≫ (-fiberι f)
 108         simp)
 109   refine ⟨⟨Triangle.π₃.mapIso e, ?_⟩⟩
 110   have h := e.hom.comm₂
 111   change HasFunctorialCofiber.cofibι (fiberι f) ≫ e.hom.hom₃ = 𝟙 A ≫ f at h
 112   exact h.trans (Category.id_comp f)

 114 private noncomputable def adamsLayerComparison (s : ℕ) :
 115     {e : HasFunctorialCofiber.cofib (adamsMap unit X s (s + 1) (Nat.le_succ s)) ≅
 116       H ⊗ adamsStage unit X s //
 117       HasFunctorialCofiber.cofibι (adamsMap unit X s (s + 1) (Nat.le_succ s)) ≫ e.hom =
 118         adamsUnit unit (adamsStage unit X s)} := by
 119   have hstep : adamsMap unit X s (s + 1) (Nat.le_succ s) =
 120       fiberι (adamsUnit unit (adamsStage unit X s)) := by
 121     rw [adamsMap_succ unit X s s le_rfl, adamsMap_self, Category.comp_id]
 122     rfl
 123   rw [hstep]
 124   exact Classical.choice (fiberLayerComparison_exists (adamsUnit unit (adamsStage unit X s)))

 126 private noncomputable def canonicalTower : AbstractAdamsTower unit X where
 127   stage := adamsStage unit X
 128   map {s t} h := adamsMap unit X s t h
 129   map_self := adamsMap_self unit X
 130   map_comp {r s t} hrs hst := adamsMap_comp unit X r s t hrs hst
 131   zeroIso := Iso.refl X
 132   layerIso s := (adamsLayerComparison unit X s).val
 133   layer_unit s := (adamsLayerComparison unit X s).property
