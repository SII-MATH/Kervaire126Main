  15 /-!
  16 # Recognition of the actual source cones and tensor generators
  17 
  18 The tensor condition here concerns suspension spectra of actual good based
  19 spaces, with the compactly generated reduced smash and its actual maps. It
  20 does not claim an arbitrary-prespectrum derived smash or its descent through
  21 stable equivalences. That full extension is a separate construction task.
  22 
  23 The cofiber condition compares the selected cofiber with the explicit mapping
  24 cone, including its actual inclusion and boundary. No freely chosen cone or
  25 smash object is introduced as source data.
  26 -/
  27

  36 variable {C : Type u} [StableHomotopyCategory.{u,v} C]
  37   {H : ModTwoCoefficients C} (R : SourceRecognition C H)
  38 
  39 noncomputable def spaceObj (A : BasedSpace.{0}) : C :=
  40   R.equivalence.functor.obj (sourceLocalization.obj (Prespectrum.suspensionSpectrum A))
  41 
  42 noncomputable def spaceMap {A B : BasedSpace.{0}} (f : A ⟶ B) : R.spaceObj A ⟶ R.spaceObj B :=
  43   R.equivalence.functor.map (sourceLocalization.map (Prespectrum.suspensionSpectrumMap f))
  44 
  45 /-- The unit isomorphism uses the same actual suspension spectrum of the
  46 discrete based two-point space and the fixed source sphere comparison. -/
  47 noncomputable def spaceUnit : R.spaceObj BasedSpace.zeroSphere ≅ SphereSpectrum :=
  48   R.equivalence.functor.mapIso (sourceLocalization.mapIso
  49     (eqToIso Prespectrum.suspensionSpectrum_zeroSphere)) ≪≫ R.sphereIso
  50 
  51 end SourceRecognition

  53 variable {C : Type u} [StableHomotopyCategory.{u,v} C]
  54   {H : ModTwoCoefficients C}
  55 
  56 /-- Recognition of the selected tensor on good suspension-spectrum
  57 generators only. Full derived-smash extension is not asserted by this type. -/
  58 structure SourceGeneratorTensorRecognition (R : SourceRecognition C H) where
  59   smash : ∀ (A B : BasedSpace.{0}), A.IsGood → B.IsGood →
  60     (R.spaceObj A ⊗ R.spaceObj B ≅ R.spaceObj (A.smash B))
  61   natural : ∀ {A B A' B' : BasedSpace.{0}} (f : A ⟶ A') (g : B ⟶ B')
  62       (hA : A.IsGood) (hB : B.IsGood) (hA' : A'.IsGood) (hB' : B'.IsGood),
  63     (R.spaceMap f ⊗ₘ R.spaceMap g) ≫ (smash A' B' hA' hB').hom =
  64       (smash A B hA hB).hom ≫ R.spaceMap (BasedSpace.smashMap f g)
  65   unit : ∀ (A : BasedSpace.{0}) (hA : A.IsGood),
  66     (smash BasedSpace.zeroSphere A BasedSpace.good_zeroSphere hA).hom ≫
  67       R.spaceMap (BasedSpace.smashUnit A hA) =
  68         (R.spaceUnit.hom ▷ R.spaceObj A) ≫ (λ_ (R.spaceObj A)).hom
  69 
  70 /-- Recognition of the actual reduced mapping cone for well-based inputs,
  71 including the same inclusion and connecting map as the fixed cofiber. -/
  72 structure SourceCofiberRecognition [HasFunctorialCofiber (C := C)]
  73     (R : SourceRecognition C H) where
  74   cone : ∀ {X Y : Prespectrum.{0}} (f : X ⟶ Y),
  75     X.IsWellBased → Y.IsWellBased →
  76       (R.equivalence.functor.obj (sourceLocalization.obj (Prespectrum.cone f)) ≅
  77         HasFunctorialCofiber.cofib
  78           (R.equivalence.functor.map (sourceLocalization.map f)))
  79   inclusion : ∀ {X Y : Prespectrum.{0}} (f : X ⟶ Y)
  80       (hX : X.IsWellBased) (hY : Y.IsWellBased),
  81     R.equivalence.functor.map (sourceLocalization.map (Prespectrum.coneInclusion f)) ≫
  82       (cone f hX hY).hom =
  83         HasFunctorialCofiber.cofibι (R.equivalence.functor.map (sourceLocalization.map f))
  84   boundary : ∀ {X Y : Prespectrum.{0}} (f : X ⟶ Y)
  85       (hX : X.IsWellBased) (hY : Y.IsWellBased),
  86     R.equivalence.functor.map (sourceLocalization.map (Prespectrum.coneBoundary f)) ≫
  87       (R.suspensionIso X hX).hom =
  88         (cone f hX hY).hom ≫
  89           HasFunctorialCofiber.cofibδ
  90             (R.equivalence.functor.map (sourceLocalization.map f))
  91 
  92 end KIP126.AdamsSynthetic.Source
