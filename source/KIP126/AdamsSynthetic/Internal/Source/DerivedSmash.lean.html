  14 /-!
  15 # The actual even-indexed prespectrum smash and its derived comparison boundary
  16 
  17 The even level `2n` is `X_n ∧ Y_n`; the odd level is its suspension. The
  18 paired bond is fixed by the actual two input bonds on quotient representatives
  19 and its continuity is proved below. No tensor of the target category is used
  20 in defining this source smash.
  21 
  22 Good replacement, flatness, and identification with the selected target tensor
  23 are separate mathematical conditions on this explicit construction.
  24 -/

  47 /-- Two consecutive bonds, on their actual quotient representatives. -/
  48 theorem pairedBond_exists (X Y : Prespectrum.{0}) :
  49     ∃! b : C((X.level 0 |>.smash (Y.level 0)).suspension × I,
  50         (X.level 1).smash (Y.level 1)),
  51       (∀ x y s t,
  52         b (BasedSpace.suspensionProjection _ (BasedSpace.smashPoint _ _ x y,s),t) =
  53           BasedSpace.smashPoint _ _ (X.bond 0 (x,s)) (Y.bond 0 (y,t))) ∧
  54       (∀ t, b (((X.level 0).smash (Y.level 0)).suspension.point,t) =
  55         ((X.level 1).smash (Y.level 1)).point) ∧
  56       (∀ a, b (a,0) = ((X.level 1).smash (Y.level 1)).point) ∧
  57       (∀ a, b (a,1) = ((X.level 1).smash (Y.level 1)).point) := by
  58   have kc (S : CompHaus.{0}) (k : C(S, X.level 1 × Y.level 1)) :
  59       Continuous (fun s : S => BasedSpace.smashPoint (X.level 1) (Y.level 1) (k s).1 (k s).2) := by
  60     let ev : (Σ i : (T : CompHaus.{0}) × C(T, X.level 1 × Y.level 1), i.fst) →
  61         X.level 1 × Y.level 1 := fun q => q.1.2 q.2
  62     have hi : Continuous (fun s : S => (⟨⟨S,k⟩,s⟩ :
  63         Σ i : (T : CompHaus.{0}) × C(T, X.level 1 × Y.level 1), i.fst)) :=
  64       @continuous_sigmaMk _ (fun i : (T : CompHaus.{0}) × C(T, X.level 1 × Y.level 1) => i.fst)
  65         (fun _ => inferInstance) ⟨S,k⟩
  66     letI : TopologicalSpace (X.level 1 × Y.level 1) := TopologicalSpace.coinduced ev inferInstance
  67     exact continuous_quot_mk.comp ((show Continuous ev from continuous_coinduced_rng).comp hi)
  68   let family (p : X.level 0 × Y.level 0) :
  69       C(I, C(I, (X.level 1).smash (Y.level 1))) :=
  70     (⟨fun st : I × I => BasedSpace.smashPoint _ _ (X.bond 0 (p.1,st.1)) (Y.bond 0 (p.2,st.2)),
  71       kc (CompHaus.of (I × I)) ⟨_,
  72         ((X.bond 0).continuous.comp (continuous_const.prodMk continuous_fst)).prodMk
  73           ((Y.bond 0).continuous.comp (continuous_const.prodMk continuous_snd))⟩⟩ :
  74       C(I × I, (X.level 1).smash (Y.level 1))).curry
  75   have hf : Continuous[TopologicalSpace.compactlyGenerated.{0} (X.level 0 × Y.level 0),
  76       inferInstance] family := by
  77     refine @continuous_from_compactlyGenerated (X.level 0 × Y.level 0) _ _ _ family ?_
  78     intro S k
  79     apply ContinuousMap.continuous_of_continuous_uncurry
  80     apply ContinuousMap.continuous_of_continuous_uncurry
  81     exact kc (CompHaus.of ((S × I) × I)) ⟨fun p =>
  82       (X.bond 0 ((k p.1.1).1,p.1.2), Y.bond 0 ((k p.1.1).2,p.2)),
  83       ((X.bond 0).continuous.comp ((k.continuous.fst.comp continuous_fst.fst).prodMk
  84         continuous_fst.snd)).prodMk
  85       ((Y.bond 0).continuous.comp ((k.continuous.snd.comp continuous_fst.fst).prodMk
  86         continuous_snd))⟩
  87   let raw (p : (X.level 0 × Y.level 0) × (I × I)) : (X.level 1).smash (Y.level 1) :=
  88     BasedSpace.smashPoint _ _ (X.bond 0 (p.1.1,p.2.1)) (Y.bond 0 (p.1.2,p.2.2))
  89   have collapsed (p : X.level 0 × Y.level 0) (st : I × I)
  90       (hp : p.1 = (X.level 0).point ∨ p.2 = (Y.level 0).point) :
  91       raw (p,st) = ((X.level 1).smash (Y.level 1)).point := by
  92     rcases hp with h | h
  93     · dsimp [raw]
  94       rw [h, X.bond_point]
  95       exact BasedSpace.smashPoint_base_left _ _ _
  96     · dsimp [raw]
  97       rw [h, Y.bond_point]
  98       exact BasedSpace.smashPoint_base_right _ _ _
  99   have hr (st : I × I) : ∀ a b, BasedSpace.SmashRelation (X.level 0) (Y.level 0) a b →
 100       raw (a,st) = raw (b,st) :=
 101     fun a b h => (collapsed a st h.1).trans (collapsed b st h.2).symm
 102   let mid (p : (X.level 0).smash (Y.level 0) × (I × I)) :=
 103     Quot.lift (fun q => raw (q,p.2)) (hr p.2) p.1
 104   have hm : Continuous mid := by
 105     letI : TopologicalSpace (X.level 0 × Y.level 0) :=
 106       TopologicalSpace.compactlyGenerated.{0} (X.level 0 × Y.level 0)
 107     apply isQuotientMap_quot_mk.continuous_lift_prod_left
 108     exact ((⟨family,hf⟩ : C(X.level 0 × Y.level 0,
 109       C(I,C(I,(X.level 1).smash (Y.level 1))))).uncurry.uncurry.continuous).comp
 110       ((continuous_fst.prodMk continuous_snd.fst).prodMk continuous_snd.snd)
 111   have mid_left (s t : I) : mid (((X.level 0).smash (Y.level 0)).point,(s,t)) =
 112       ((X.level 1).smash (Y.level 1)).point := collapsed _ _ (Or.inl rfl)
 113   have mid_zero (a : (X.level 0).smash (Y.level 0)) (t : I) :
 114       mid (a,(0,t)) = ((X.level 1).smash (Y.level 1)).point := by
 115     refine Quot.inductionOn a ?_
 116     rintro ⟨x,y⟩
 117     change BasedSpace.smashPoint _ _ (X.bond 0 (x,0)) (Y.bond 0 (y,t)) = _
 118     rw [X.bond_zero]
 119     exact BasedSpace.smashPoint_base_left _ _ _
 120   have mid_one (a : (X.level 0).smash (Y.level 0)) (t : I) :
 121       mid (a,(1,t)) = ((X.level 1).smash (Y.level 1)).point := by
 122     refine Quot.inductionOn a ?_
 123     rintro ⟨x,y⟩
 124     change BasedSpace.smashPoint _ _ (X.bond 0 (x,1)) (Y.bond 0 (y,t)) = _
 125     rw [X.bond_one]
 126     exact BasedSpace.smashPoint_base_left _ _ _
 127   have collapse_outer (p : (X.level 0).smash (Y.level 0) × I) (t : I)
 128       (hp : BasedSpace.SuspensionCollapsed ((X.level 0).smash (Y.level 0)) p) :
 129       mid (p.1,(p.2,t)) = ((X.level 1).smash (Y.level 1)).point := by
 130     rcases hp with h | h | h
 131     · rw [h, mid_left]
 132     · rw [h, mid_zero]
 133     · rw [h, mid_one]
 134   have hrout (t : I) : ∀ a b,
 135       BasedSpace.SuspensionRelation ((X.level 0).smash (Y.level 0)) a b →
 136       mid (a.1,(a.2,t)) = mid (b.1,(b.2,t)) :=
 137     fun a b h => (collapse_outer a t h.1).trans (collapse_outer b t h.2).symm
 138   let b : C(((X.level 0).smash (Y.level 0)).suspension × I,
 139       (X.level 1).smash (Y.level 1)) :=
 140     ⟨fun p => Quot.lift (fun q => mid (q.1,(q.2,p.2))) (hrout p.2) p.1, by
 141       apply isQuotientMap_quot_mk.continuous_lift_prod_left
 142       exact hm.comp (continuous_fst.fst.prodMk (continuous_fst.snd.prodMk continuous_snd))⟩
 143   refine ⟨b, ⟨fun _ _ _ _ => rfl, fun t => mid_left 0 t, ?_, ?_⟩, ?_⟩
 144   · intro a
 145     refine Quot.inductionOn a ?_
 146     rintro ⟨a,s⟩
 147     refine Quot.inductionOn a ?_
 148     rintro ⟨x,y⟩
 149     change BasedSpace.smashPoint _ _ (X.bond 0 (x,s)) (Y.bond 0 (y,0)) = _
 150     rw [Y.bond_zero]
 151     exact BasedSpace.smashPoint_base_right _ _ _
 152   · intro a
 153     refine Quot.inductionOn a ?_
 154     rintro ⟨a,s⟩
 155     refine Quot.inductionOn a ?_
 156     rintro ⟨x,y⟩
 157     change BasedSpace.smashPoint _ _ (X.bond 0 (x,s)) (Y.bond 0 (y,1)) = _
 158     rw [Y.bond_one]
 159     exact BasedSpace.smashPoint_base_right _ _ _
 160   · intro c hc
 161     apply ContinuousMap.ext
 162     rintro ⟨a,t⟩
 163     refine Quot.inductionOn a ?_
 164     rintro ⟨a,s⟩
 165     refine Quot.inductionOn a ?_
 166     rintro ⟨x,y⟩
 167     exact hc.1 x y s t

 175 def diagonalLevel (X Y : Prespectrum.{0}) : ℕ → BasedSpace.{0}
 176   | 0 => (X.level 0).smash (Y.level 0)
 177   | 1 => ((X.level 0).smash (Y.level 0)).suspension
 178   | n+2 => diagonalLevel X.tail Y.tail n

 180 noncomputable def diagonalBond (X Y : Prespectrum.{0}) :
 181     ∀ n, C(diagonalLevel X Y n × I, diagonalLevel X Y (n+1))
 182   | 0 => BasedSpace.suspensionProjection _
 183   | 1 => pairedBond X Y
 184   | n+2 => diagonalBond X.tail Y.tail n

 188 theorem diagonalBond_laws (X Y : Prespectrum.{0}) (n : ℕ) :
 189     (∀ t, diagonalBond X Y n ((diagonalLevel X Y n).point,t) =
 190       (diagonalLevel X Y (n+1)).point) ∧
 191     (∀ a, diagonalBond X Y n (a,0) = (diagonalLevel X Y (n+1)).point) ∧
 192     (∀ a, diagonalBond X Y n (a,1) = (diagonalLevel X Y (n+1)).point) := by
 193   induction n using Nat.twoStepInduction generalizing X Y with
 194   | zero =>
 195     exact ⟨fun _ => BasedSpace.suspensionProjection_collapsed _ _ (Or.inl rfl),
 196       fun _ => BasedSpace.suspensionProjection_collapsed _ _ (Or.inr (Or.inl rfl)),
 197       fun _ => BasedSpace.suspensionProjection_collapsed _ _ (Or.inr (Or.inr rfl))⟩
 198   | one => exact (pairedBond_exists X Y).exists.choose_spec.2
 199   | more n ih _ => exact ih X.tail Y.tail

 203 noncomputable def diagonalSmash (X Y : Prespectrum.{0}) : Prespectrum.{0} where
 204   level := diagonalLevel X Y
 205   bond := diagonalBond X Y
 206   bond_point n := (diagonalBond_laws X Y n).1
 207   bond_zero n := (diagonalBond_laws X Y n).2.1
 208   bond_one n := (diagonalBond_laws X Y n).2.2

 210 noncomputable def diagonalMapLevel {X Y X' Y' : Prespectrum.{0}}
 211     (f : X ⟶ X') (g : Y ⟶ Y') :
 212     ∀ n, diagonalLevel X Y n ⟶ diagonalLevel X' Y' n
 213   | 0 => BasedSpace.smashMap ⟨f.level 0, f.point 0⟩ ⟨g.level 0, g.point 0⟩
 214   | 1 => BasedSpace.suspendMap
 215       (BasedSpace.smashMap ⟨f.level 0, f.point 0⟩ ⟨g.level 0, g.point 0⟩)
 216   | n+2 => diagonalMapLevel (tailMap f) (tailMap g) n

 220 theorem diagonalMapLevel_bond {X Y X' Y' : Prespectrum.{0}}
 221     (f : X ⟶ X') (g : Y ⟶ Y') (n : ℕ) (a : diagonalLevel X Y n) (t : I) :
 222     (diagonalMapLevel f g (n+1)).map (diagonalBond X Y n (a,t)) =
 223       diagonalBond X' Y' n ((diagonalMapLevel f g n).map a,t) := by
 224   induction n using Nat.twoStepInduction generalizing X Y X' Y' with
 225   | zero => rfl
 226   | one =>
 227     refine Quot.inductionOn a ?_
 228     rintro ⟨b,s⟩
 229     refine Quot.inductionOn b ?_
 230     rintro ⟨x,y⟩
 231     have hsm {A B A' B' : BasedSpace.{0}} (p : A ⟶ A') (q : B ⟶ B') (x : A) (y : B) :
 232         (BasedSpace.smashMap p q).map (BasedSpace.smashPoint A B x y) =
 233           BasedSpace.smashPoint A' B' (p.map x) (q.map y) :=
 234       (BasedSpace.smashMap_exists p q).exists.choose_spec x y
 235     have hp (U V : Prespectrum.{0}) (x : U.level 0) (y : V.level 0) (s t : I) :
 236         pairedBond U V (BasedSpace.suspensionProjection _
 237           (BasedSpace.smashPoint _ _ x y,s),t) =
 238         BasedSpace.smashPoint _ _ (U.bond 0 (x,s)) (V.bond 0 (y,t)) :=
 239       (pairedBond_exists U V).exists.choose_spec.1 x y s t
 240     change (BasedSpace.smashMap ⟨f.level 1, f.point 1⟩ ⟨g.level 1, g.point 1⟩).map
 241         (pairedBond X Y (BasedSpace.suspensionProjection _
 242           (BasedSpace.smashPoint _ _ x y,s),t)) =
 243       pairedBond X' Y' (BasedSpace.suspensionProjection _
 244         ((BasedSpace.smashMap ⟨f.level 0, f.point 0⟩ ⟨g.level 0, g.point 0⟩).map
 245           (BasedSpace.smashPoint _ _ x y),s),t)
 246     rw [hp, hsm, hsm, hp]
 247     change BasedSpace.smashPoint _ _ (f.level 1 (X.bond 0 (x,s)))
 248         (g.level 1 (Y.bond 0 (y,t))) = _
 249     rw [f.bond, g.bond]
 250   | more n ih _ => exact ih (tailMap f) (tailMap g) a

 265 /-- A replacement is an actual point-set functor and actual stable comparison
 266 map, with the precise levelwise CW-type hypotheses used by the smash. -/
 267 structure SourceReplacementData where
 268   replacement : Prespectrum.{0} ⥤ Prespectrum.{0}
 269   comparison : replacement ⟶ 𝟭 _
 270   good : ∀ X, (replacement.obj X).IsGood
 271   stable : ∀ X, Prespectrum.stableEquivalences (comparison.app X)
 272 
 273 /-- Flatness is requested for this specified replacement and this actual
 274 two-variable map. It is not inferred merely from a replacement label. -/
 275 def SourceReplacementFlatness (P : SourceReplacementData) : Prop :=
 276   ∀ {X Y X' Y' : Prespectrum.{0}} (f : X ⟶ X') (g : Y ⟶ Y'),
 277     Prespectrum.stableEquivalences f → Prespectrum.stableEquivalences g →
 278       Prespectrum.stableEquivalences
 279         (Prespectrum.diagonalSmashMap (P.replacement.map f) (P.replacement.map g))

 295 /-- Full-object recognition against the independently constructed source
 296 smash of the specified replacements. This extends the object comparison to
 297 every prespectrum, rather than supplying a second freely chosen source tensor.
 298 
 299 The source unitor and its agreement with the generator normalization are not
 300 asserted here; they require an actual source unitor construction. -/
 301 structure SourceDerivedSmashRecognition
 302     {C : Type u} [StableHomotopyCategory.{u,v} C] {H : ModTwoCoefficients C}
 303     (R : SourceRecognition C H) (P : SourceReplacementData) where
 304   comparison : ∀ (X Y : Prespectrum.{0}),
 305     (R.equivalence.functor.obj (sourceLocalization.obj X) ⊗
 306       R.equivalence.functor.obj (sourceLocalization.obj Y) ≅
 307         R.equivalence.functor.obj (sourceLocalization.obj (P.derivedSmash X Y)))
 308   natural : ∀ {X Y X' Y' : Prespectrum.{0}} (f : X ⟶ X') (g : Y ⟶ Y'),
 309     (R.equivalence.functor.map (sourceLocalization.map f) ⊗ₘ
 310         R.equivalence.functor.map (sourceLocalization.map g)) ≫
 311       (comparison X' Y').hom =
 312         (comparison X Y).hom ≫
 313           R.equivalence.functor.map (sourceLocalization.map (P.derivedSmashMap f g))
 314 
 315 end KIP126.AdamsSynthetic.Source
