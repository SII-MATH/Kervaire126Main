  15 /-!
  16 # Recognition of the fixed classical background by its concrete source
  17 
  18 This is a mathematical comparison with the ordinary stable localization of
  19 the explicit sequential prespectra. It does not redefine that localization
  20 as a completed category. Every comparison uses the same equivalence, actual
  21 point-set maps, and the selected coefficient object and unit.
  22 
  23 The record contains no Adams page, differential, named Adams class, survival
  24 claim, cofiber comparison, or tensor comparison. The latter two require their
  25 own actual source constructions.
  26 -/

  35 /-- Recognition of one background by the explicit ordinary stable source,
  36 including its sphere, shifts, integer stable homotopy, and mod-two unit. -/
  37 structure SourceRecognition (C : Type u) [StableHomotopyCategory.{u,v} C]
  38     (H : ModTwoCoefficients C) where
  39   equivalence : SourceCategory.{0} ≌ C
  40   sphereIso : equivalence.functor.obj (sourceLocalization.obj Prespectrum.sphere) ≅
  41     SphereSpectrum
  42   suspensionIso : ∀ (X : Prespectrum.{0}), X.IsWellBased →
  43     (equivalence.functor.obj (sourceLocalization.obj X.suspension) ≅
  44       (shiftFunctor C (1 : ℤ)).obj
  45         (equivalence.functor.obj (sourceLocalization.obj X)))
  46   suspension_natural : ∀ {X Y : Prespectrum.{0}} (f : X ⟶ Y)
  47       (hX : X.IsWellBased) (hY : Y.IsWellBased),
  48     equivalence.functor.map (sourceLocalization.map (Prespectrum.suspensionMap f)) ≫
  49       (suspensionIso Y hY).hom =
  50         (suspensionIso X hX).hom ≫ (shiftFunctor C (1 : ℤ)).map
  51           (equivalence.functor.map (sourceLocalization.map f))
  52   desuspensionIso : ∀ X : Prespectrum.{0},
  53     equivalence.functor.obj (sourceLocalization.obj X.desuspension) ≅
  54       (shiftFunctor C (-1 : ℤ)).obj
  55         (equivalence.functor.obj (sourceLocalization.obj X))
  56   desuspension_natural : ∀ {X Y : Prespectrum.{0}} (f : X ⟶ Y),
  57     equivalence.functor.map (sourceLocalization.map (Prespectrum.desuspensionMap f)) ≫
  58       (desuspensionIso Y).hom =
  59         (desuspensionIso X).hom ≫ (shiftFunctor C (-1 : ℤ)).map
  60           (equivalence.functor.map (sourceLocalization.map f))
  61   homotopy : ∀ (X : Prespectrum.{0}) (d : ℤ),
  62     HomotopyGroup d (equivalence.functor.obj (sourceLocalization.obj X)) ≃
  63       Prespectrum.StableHomotopy X d
  64   homotopy_natural : ∀ {X Y : Prespectrum.{0}} (f : X ⟶ Y) (d : ℤ)
  65       (a : HomotopyGroup d (equivalence.functor.obj (sourceLocalization.obj X))),
  66     homotopy Y d (a ≫ equivalence.functor.map (sourceLocalization.map f)) =
  67       Prespectrum.stableHomotopyMap f d (homotopy X d a)
  68   homotopy_zero : ∀ (X : Prespectrum.{0}) (d : ℤ) (n r : ℕ)
  69       (h : (r : ℤ) = d + n),
  70     homotopy X d 0 = Prespectrum.loopClass X d n r h GenLoop.const
  71   /-- Stable addition is the actual concatenation of cubical loops. Two
  72   coordinates ensure the displayed representative lies in the abelian range. -/
  73   homotopy_add : ∀ (X : Prespectrum.{0}) (d : ℤ) (n r : ℕ)
  74       (h : ((r + 2 : ℕ) : ℤ) = d + n)
  75       (p q : GenLoop (Fin (r + 2)) (X.level n) (X.level n).point),
  76     (homotopy X d).symm
  77         (Prespectrum.loopClass X d n (r + 2) h (GenLoop.transAt 0 p q)) =
  78       (homotopy X d).symm (Prespectrum.loopClass X d n (r + 2) h p) +
  79         (homotopy X d).symm (Prespectrum.loopClass X d n (r + 2) h q)
  80   /-- The identity of the selected sphere has the actual positive point-set
  81   generator, fixing the normalization of the source sphere comparison. -/
  82   sphere_identity : homotopy Prespectrum.sphere 0
  83       ((shiftFunctorZero C ℤ).hom.app SphereSpectrum ≫ sphereIso.inv) =
  84     Prespectrum.sphereGenerator
  85   sourceHF2 : Prespectrum.{0}
  86   sourceHF2Pi0 : Prespectrum.StableHomotopy sourceHF2 0 ≃ ZMod 2
  87   sourceHF2Vanishes : ∀ d : ℤ, d ≠ 0 →
  88     Subsingleton (Prespectrum.StableHomotopy sourceHF2 d)
  89   hf2Iso : equivalence.functor.obj (sourceLocalization.obj sourceHF2) ≅ H.object
  90   /-- The source coefficient coordinate is the selected target coordinate on
  91   the same homotopy comparison, not a separately named two-element set. -/
  92   hf2_pi0 : ∀ a : HomotopyGroup 0
  93       (equivalence.functor.obj (sourceLocalization.obj sourceHF2)),
  94     H.pi0Equiv (a ≫ hf2Iso.hom) = sourceHF2Pi0 (homotopy sourceHF2 0 a)
  95   sourceUnit : Prespectrum.sphere ⟶ sourceHF2
  96   /-- The actual point-set unit represents this same selected coefficient
  97   unit under the fixed sphere and coefficient comparisons. -/
  98   unit_compat : equivalence.functor.map (sourceLocalization.map sourceUnit) ≫
  99       hf2Iso.hom = sphereIso.hom ≫ H.unit

 103 variable {C : Type u} [StableHomotopyCategory.{u,v} C]
 104   {H : ModTwoCoefficients C} (R : SourceRecognition C H)
 105 
 106 /-- The coefficient zero coordinate is forced by the same target zero and
 107 the same homotopy comparison; it is not a further field to be supplied. -/
 108 theorem sourceHF2Pi0_zero (n r : ℕ) (h : (r : ℤ) = (0 : ℤ) + n) :
 109     R.sourceHF2Pi0 (Prespectrum.loopClass R.sourceHF2 0 n r h GenLoop.const) = 0 := by
 110   rw [← R.homotopy_zero R.sourceHF2 0 n r h, ← R.hf2_pi0]
 111   simp
 112 
 113 /-- The chosen actual source unit sends the actual positive sphere generator
 114 to the coefficient one in the same specified coordinate. -/
 115 theorem sourceUnit_pi0 :
 116     R.sourceHF2Pi0 (Prespectrum.stableHomotopyMap R.sourceUnit 0
 117       Prespectrum.sphereGenerator) = 1 := by
 118   let a : HomotopyGroup 0 (R.equivalence.functor.obj
 119       (sourceLocalization.obj Prespectrum.sphere)) :=
 120     (shiftFunctorZero C ℤ).hom.app SphereSpectrum ≫ R.sphereIso.inv
 121   have ha : R.homotopy Prespectrum.sphere 0 a = Prespectrum.sphereGenerator :=
 122     R.sphere_identity
 123   rw [← ha, ← R.homotopy_natural R.sourceUnit 0 a, ← R.hf2_pi0]
 124   simp [a, Category.assoc, R.unit_compat, ModTwoCoefficients.unit]
 125   rw [← Category.assoc]
 126   erw [Iso.hom_inv_id_app]
 127   rw [Category.id_comp]
 128   exact H.pi0Equiv.apply_symm_apply 1
 129 
 130 end SourceRecognition
