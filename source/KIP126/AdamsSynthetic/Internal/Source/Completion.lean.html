  15 /-!
  16 # HF₂ cohomology localization of the same recognized ordinary source
  17 
  18 Every detector is the inverse image of an integer shift of the same selected
  19 HF₂ object under the same source equivalence. This uses the recognized stable
  20 shift, without asserting that an unrestricted point-set reduced suspension is
  21 a derived suspension. A morphism is inverted precisely when precomposition
  22 induces a bijection into every one of these fixed integer detectors.
  23 
  24 The category below is the actual categorical localization at that explicit
  25 class of maps. Its definition does not identify this localization with a
  26 reflective subcategory, Moore(2)-completion, or the paper's completion of
  27 connective spectra of finite type; those require further comparison proofs.
  28 -/
  29 
  30 namespace KIP126.AdamsSynthetic.Source
  31 
  32 open CategoryTheory KIP126.StableHomotopy
  33 
  34 universe u v
  35 
  36 noncomputable section
  37 
  38 variable {C : Type u} [StableHomotopyCategory.{u,v} C]
  39   {H : ModTwoCoefficients C}
  40 
  41 /-- A fixed integer-shifted HF₂ detector in the actual ordinary source. -/
  42 def shiftedHF2 (R : SourceRecognition C H) (d : ℤ) : SourceCategory.{0} :=
  43   R.equivalence.inverse.obj ((shiftFunctor C d).obj H.object)
  44 
  45 def shiftedHF2Iso (R : SourceRecognition C H) (d : ℤ) :
  46     R.equivalence.functor.obj (shiftedHF2 R d) ≅ (shiftFunctor C d).obj H.object :=
  47   R.equivalence.counitIso.app _
  48 
  49 /-- Equivalences detected by cohomology into every shift of the same HF₂. -/
  50 def HF2Equivalences (R : SourceRecognition C H) : MorphismProperty SourceCategory.{0} :=
  51   fun _X Y f => ∀ d : ℤ,
  52     Function.Bijective (fun a : Y ⟶ shiftedHF2 R d => f ≫ a)
  53 
  54 /-- The localization is a prescribed construction, not an arbitrary category
  55 supplied with a completion label. -/
  56 abbrev HF2LocalizedSource (R : SourceRecognition C H) := (HF2Equivalences R).Localization
  57 
  58 abbrev hf2Localization (R : SourceRecognition C H) :
  59     SourceCategory.{0} ⥤ HF2LocalizedSource R := (HF2Equivalences R).Q
  60 
  61 def completedSphere (R : SourceRecognition C H) : HF2LocalizedSource R :=
  62   (hf2Localization R).obj (sourceLocalization.obj Prespectrum.sphere)
  63 
  64 end
  65 
  66 end KIP126.AdamsSynthetic.Source
