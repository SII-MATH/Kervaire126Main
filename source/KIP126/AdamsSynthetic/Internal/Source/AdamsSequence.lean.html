  28 private noncomputable def towerI (T : AbstractAdamsTower unit X) (m s t : ℤ) (h : s ≤ t) :
  29     HomotopyGroup m (T.stageAt t) →ₗ[ℤ] HomotopyGroup m (T.stageAt s) :=
  30   (inducedMap (T.mapAt h) m).toIntLinearMap
  31 
  32 private theorem towerI_comp (T : AbstractAdamsTower unit X) (m s t z : ℤ)
  33     (hst : s ≤ t) (htz : t ≤ z) (a : HomotopyGroup m (T.stageAt z)) :
  34     towerI T m s t hst (towerI T m t z htz a) = towerI T m s z (hst.trans htz) a := by
  35   change (a ≫ T.mapAt htz) ≫ T.mapAt hst = a ≫ T.mapAt (hst.trans htz)
  36   rw [Category.assoc]
  37   congr 1
  38   exact T.map_comp _ _

  40 private abbrev cycleAmbient (T : AbstractAdamsTower unit X) (p : ℤ × ℤ) :=
  41   T.cycles 2 (by decide) p
  42 
  43 private noncomputable def finiteCycles (T : AbstractAdamsTower unit X) (p : ℤ × ℤ) (n : ℕ) :
  44     Submodule ℤ (cycleAmbient T p) :=
  45   (T.cycles (n + 2) (by omega) p).comap (T.cycles 2 (by decide) p).subtype
  46 
  47 private noncomputable def finiteBoundaries (T : AbstractAdamsTower unit X) (p : ℤ × ℤ)
  48     (n : ℕ) : Submodule ℤ (cycleAmbient T p) :=
  49   (T.boundaries (n + 2) (by omega) p).comap (T.cycles 2 (by decide) p).subtype

  51 private noncomputable def allCycles (T : AbstractAdamsTower unit X) (p : ℤ × ℤ) :
  52     WithTop ℕ → Submodule ℤ (cycleAmbient T p)
  53   | ⊤ => ⨅ n : ℕ, finiteCycles T p n
  54   | (n : ℕ) => finiteCycles T p n
  55 
  56 private noncomputable def allBoundaries (T : AbstractAdamsTower unit X) (p : ℤ × ℤ) :
  57     WithTop ℕ → Submodule ℤ (cycleAmbient T p)
  58   | ⊤ => ⨆ n : ℕ, finiteBoundaries T p n
  59   | (n : ℕ) => finiteBoundaries T p n

  61 private theorem finiteCycles_antitone (T : AbstractAdamsTower unit X) (p : ℤ × ℤ) :
  62     Antitone (finiteCycles T p) := by
  63   intro i j hij x hx
  64   obtain ⟨y, hy⟩ := hx
  65   refine ⟨towerI T (p.2-p.1-1) (p.1+(i+2:ℕ)) (p.1+(j+2:ℕ)) (by omega) y, ?_⟩
  66   change towerI T (p.2-p.1-1) (p.1+1) (p.1+(i+2:ℕ)) _ _ = _
  67   rw [towerI_comp]
  68   exact hy
  69 
  70 private theorem finiteBoundaries_monotone (T : AbstractAdamsTower unit X) (p : ℤ × ℤ) :
  71     Monotone (finiteBoundaries T p) := by
  72   intro i j hij x hx
  73   obtain ⟨y, hy, hj⟩ := hx
  74   refine ⟨y, ?_, hj⟩
  75   change towerI T (p.2-p.1) (p.1-(i+2:ℕ)+1) p.1 _ y = 0 at hy
  76   change towerI T (p.2-p.1) (p.1-(j+2:ℕ)+1) p.1 _ y = 0
  77   rw [← towerI_comp T (p.2-p.1) (p.1-(j+2:ℕ)+1)
  78     (p.1-(i+2:ℕ)+1) p.1 (by omega) (by omega), hy, map_zero]
  79 
  80 private theorem connecting_layerMap (T : AbstractAdamsTower unit X) (p : ℤ × ℤ)
  81     (a : HomotopyGroup (p.2-p.1) (T.stageAt p.1)) :
  82     T.connecting p (T.layerMap p a) = 0 := by
  83   let f := T.mapAt (show p.1 ≤ p.1+1 by omega)
  84   have hzero : HasFunctorialCofiber.cofibι f ≫ HasFunctorialCofiber.cofibδ f = 0 :=
  85     comp_distTriang_mor_zero₂₃ _ (HasFunctorialCofiber.cofib_distinguished f)
  86   change (AbstractAdamsTower.desuspend _ _)
  87     ((a ≫ HasFunctorialCofiber.cofibι f) ≫ HasFunctorialCofiber.cofibδ f) = 0
  88   rw [Category.assoc, hzero, comp_zero, map_zero]

  90 private theorem finiteBoundaries_le_cycles (T : AbstractAdamsTower unit X) (p : ℤ × ℤ)
  91     (i j : ℕ) : finiteBoundaries T p i ≤ finiteCycles T p j := by
  92   intro x hx
  93   obtain ⟨y, _, hj⟩ := hx
  94   change T.layerMap p y = x.val at hj
  95   change T.connecting p x.val ∈ LinearMap.range _
  96   rw [← hj, connecting_layerMap]
  97   exact Submodule.zero_mem _

  99 private theorem allCycles_antitone (T : AbstractAdamsTower unit X) (p : ℤ × ℤ) :
 100     Antitone (allCycles T p) := by
 101   intro i j hij
 102   cases i using WithTop.recTopCoe with
 103   | top => have h : j = ⊤ := top_le_iff.mp hij; subst j; exact le_rfl
 104   | coe i =>
 105     cases j using WithTop.recTopCoe with
 106     | top => exact iInf_le _ i
 107     | coe j => exact finiteCycles_antitone T p (WithTop.coe_le_coe.mp hij)
 108 
 109 private theorem allBoundaries_monotone (T : AbstractAdamsTower unit X) (p : ℤ × ℤ) :
 110     Monotone (allBoundaries T p) := by
 111   intro i j hij
 112   cases i using WithTop.recTopCoe with
 113   | top => have h : j = ⊤ := top_le_iff.mp hij; subst j; exact le_rfl
 114   | coe i =>
 115     cases j using WithTop.recTopCoe with
 116     | top => exact le_iSup (finiteBoundaries T p) i
 117     | coe j => exact finiteBoundaries_monotone T p (WithTop.coe_le_coe.mp hij)
 118 
 119 private theorem allBoundaries_le_cycles (T : AbstractAdamsTower unit X) (p : ℤ × ℤ)
 120     (r : WithTop ℕ) : allBoundaries T p r ≤ allCycles T p r := by
 121   cases r using WithTop.recTopCoe with
 122   | top => exact iSup_le fun i => le_iInf fun j => finiteBoundaries_le_cycles T p i j
 123   | coe n => exact finiteBoundaries_le_cycles T p n n

 125 private noncomputable def towerSSData (T : AbstractAdamsTower unit X) (p : ℤ × ℤ) :
 126     SSData (ModuleCat.{v} ℤ) where
 127   V := ModuleCat.of ℤ (cycleAmbient T p)
 128   Z r := (ModuleCat.subobjectModule _).symm (allCycles T p r)
 129   B r := (ModuleCat.subobjectModule _).symm (allBoundaries T p r)
 130   Z_anti := (ModuleCat.subobjectModule _).symm.monotone.comp_antitone (allCycles_antitone T p)
 131   B_mono := (ModuleCat.subobjectModule _).symm.monotone.comp (allBoundaries_monotone T p)
 132   Z_zero := by
 133     change (ModuleCat.subobjectModule (ModuleCat.of ℤ (cycleAmbient T p))).symm
 134       ((T.cycles 2 (by decide) p).comap (T.cycles 2 (by decide) p).subtype) = ⊤
 135     rw [Submodule.comap_subtype_self, OrderIso.map_top]
 136   B_le_Z r := (ModuleCat.subobjectModule _).symm.monotone (allBoundaries_le_cycles T p r)
 137   Z_top_greatest A h := by
 138     apply (ModuleCat.subobjectModule _).le_iff_le.mp
 139     change (ModuleCat.subobjectModule _) A ≤ (ModuleCat.subobjectModule _)
 140       ((ModuleCat.subobjectModule _).symm _)
 141     rw [OrderIso.apply_symm_apply]
 142     refine le_iInf fun n => ?_
 143     have hn := (ModuleCat.subobjectModule _).monotone (h n)
 144     rw [OrderIso.apply_symm_apply] at hn
 145     exact hn
 146   B_top_least A h := by
 147     apply (ModuleCat.subobjectModule _).le_iff_le.mp
 148     change (ModuleCat.subobjectModule _) ((ModuleCat.subobjectModule _).symm _) ≤
 149       (ModuleCat.subobjectModule _) A
 150     rw [OrderIso.apply_symm_apply]
 151     refine iSup_le fun n => ?_
 152     have hn := (ModuleCat.subobjectModule _).monotone (h n)
 153     rw [OrderIso.apply_symm_apply] at hn
 154     exact hn

 156 private theorem towerSSData_cycles (T : AbstractAdamsTower unit X) (p : ℤ × ℤ)
 157     (n : ℕ) (a : (towerSSData T p).V) :
 158     a ∈ (ModuleCat.subobjectModule _ ((towerSSData T p).Z n)) ↔
 159       a.val ∈ T.cycles (n + 2) (by omega) p := by
 160   change a ∈ (ModuleCat.subobjectModule (ModuleCat.of ℤ (cycleAmbient T p)))
 161     ((ModuleCat.subobjectModule (ModuleCat.of ℤ (cycleAmbient T p))).symm
 162       (finiteCycles T p n)) ↔ _
 163   rw [OrderIso.apply_symm_apply]
 164   rfl

 166 private theorem towerSSData_boundaries (T : AbstractAdamsTower unit X) (p : ℤ × ℤ)
 167     (n : ℕ) (a : (towerSSData T p).V) :
 168     a ∈ (ModuleCat.subobjectModule _ ((towerSSData T p).B n)) ↔

 176 /-- Every representative is evaluated through the same connecting map,
 177 the same actual tower lift and the same cofiber projection. -/
 178 private def TowerDifferentialRule (T : AbstractAdamsTower unit X) (r : ℤ)
 179     (hr : 2 ≤ r) (p : ℤ × ℤ)
 180     (d : (towerSSData T p).page (↑(r - 2).toNat : WithTop ℕ) ⟶
 181       (towerSSData T (p + (r,r-1))).page (↑(r - 2).toNat : WithTop ℕ)) : Prop :=
 182   ∀ (z : (Subobject.underlying.obj ((towerSSData T p).Z (↑(r - 2).toNat : WithTop ℕ)) :
 183         ModuleCat ℤ))
 184     (a : HomotopyGroup (p.2 - p.1 - 1) (T.stageAt (p.1 + r)))
 185     (z' : (Subobject.underlying.obj
 186         ((towerSSData T (p + (r,r-1))).Z (↑(r - 2).toNat : WithTop ℕ)) : ModuleCat ℤ)),
 187     T.connecting p ((((towerSSData T p).Z _).arrow z).val) =
 188       a ≫ T.mapAt (show p.1 + 1 ≤ p.1 + r by omega) →
 189     ((((towerSSData T (p + (r,r-1))).Z _).arrow z').val) =
 190       T.layerMap (p + (r,r-1))
 191         (Eq.mp (congrArg (fun m => HomotopyGroup m (T.stageAt (p.1 + r)))
 192           (by dsimp; omega : p.2 - p.1 - 1 = (p + (r,r-1)).2 - (p + (r,r-1)).1)) a) →
 193     d ((towerSSData T p).pageπ _ z) = (towerSSData T (p + (r,r-1))).pageπ _ z'

 195 /-- Existence and well-definedness of the linear map of quotient pages,
 196 with its actual lift/cofiber formula. This does not choose an arbitrary d. -/
 197 private theorem towerDifferential_exists (T : AbstractAdamsTower unit X) (r : ℤ)
 198     (hr : 2 ≤ r) (p : ℤ × ℤ) :
 199     ∃ d : (towerSSData T p).page (↑(r - 2).toNat : WithTop ℕ) ⟶
 200         (towerSSData T (p + (r,r-1))).page (↑(r - 2).toNat : WithTop ℕ),
 201       TowerDifferentialRule T r hr p d := by
 202   sorry

 204 private noncomputable def towerD (T : AbstractAdamsTower unit X) (r : ℤ) (p : ℤ × ℤ) :
 205     (towerSSData T p).page (↑(r - 2).toNat : WithTop ℕ) ⟶
 206       (towerSSData T (p + (r,r-1))).page (↑(r - 2).toNat : WithTop ℕ) :=
 207   if hr : 2 ≤ r then (towerDifferential_exists T r hr p).choose else 0
 208 
 209 private theorem towerD_rule (T : AbstractAdamsTower unit X) (r : ℤ) (hr : 2 ≤ r)
 210     (p : ℤ × ℤ) : TowerDifferentialRule T r hr p (towerD T r p) := by
 211   simpa only [towerD, dif_pos hr] using (towerDifferential_exists T r hr p).choose_spec

 213 private noncomputable def towerPreSS (T : AbstractAdamsTower unit X) :
 214     PreSS (ModuleCat.{v} ℤ) (ℤ × ℤ) where
 215   r₀ := 2
 216   ssData := towerSSData T
 217   diffDeg r := (r,r-1)
 218   d := towerD T
 219 
 220 private theorem towerD_comp (T : AbstractAdamsTower unit X) (r : ℤ) (p : ℤ × ℤ) :
 221     towerD T r p ≫ towerD T r (p + (r,r-1)) = 0 := by
 222   sorry
 223 
 224 private theorem towerD_kernel (T : AbstractAdamsTower unit X) (r : ℤ) (p : ℤ × ℤ)
 225     (hr : 2 ≤ r) :
 226     let n := (r - 2).toNat
 227     kernelSubobject (towerD T r p) =
 228       imageSubobject (Subobject.ofLE
 229         ((towerSSData T p).Z ↑(n + 1)) ((towerSSData T p).Z ↑n)
 230         ((towerSSData T p).Z_anti (by exact_mod_cast Nat.le_succ n)) ≫
 231         (towerSSData T p).pageπ ↑n) := by
 232   sorry
 233 
 234 private theorem towerD_image (T : AbstractAdamsTower unit X) (r : ℤ) (p : ℤ × ℤ)
 235     (hr : 2 ≤ r) :
 236     let n := (r - 2).toNat
 237     let D := towerSSData T (p + (r,r-1))
 238     imageSubobject (towerD T r p) =
 239       imageSubobject (Subobject.ofLE (D.B ↑(n + 1)) (D.Z ↑n)
 240         (le_trans (D.B_le_Z ↑(n + 1))
 241           (D.Z_anti (by exact_mod_cast Nat.le_succ n))) ≫ D.pageπ ↑n) := by
 242   sorry

 244 private noncomputable def towerSequence (T : AbstractAdamsTower unit X) :
 245     KIP126.Core.SpectralSequence (ModuleCat.{v} ℤ) (ℤ × ℤ) where
 246   toPreSS := towerPreSS T
 247   d_comp_d := towerD_comp T
 248   Z_succ := towerD_kernel T
 249   B_succ := towerD_image T

 251 private noncomputable def towerSequencePresentation (T : AbstractAdamsTower unit X) :
 252     TowerSequencePresentation unit T (towerSequence T).ssData
 253       (normalizedDifferential (towerSequence T) rfl (fun _ => rfl)) where
 254   ambient p := Iso.refl (ModuleCat.of ℤ (cycleAmbient T p))
 255   cycles := towerSSData_cycles T
 256   boundaries := towerSSData_boundaries T
 257   differential r hr p := towerD_rule T r hr p
