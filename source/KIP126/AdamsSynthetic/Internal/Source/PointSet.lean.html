  18 /-!
  19 # Concrete generators, smash maps, and cones in the ordinary source
  20 
  21 The reduced smash below is the actual quotient of the compactly generated
  22 product; suspension spectra use the fixed reduced suspensions, and mapping
  23 cones use the explicit point-set cone quotient with its inclusion and boundary.
  24 Only these constructions are claimed here. No arbitrary-prespectrum derived
  25 smash or descent of such a smash to stable localization is supplied.

  41 @[ext] structure Hom (X Y : BasedSpace.{u}) where
  42   map : C(X,Y)
  43   point : map X.point = Y.point

  45 instance : Category BasedSpace.{u} where
  46   Hom := Hom
  47   id X := ⟨ContinuousMap.id _, rfl⟩
  48   comp f g := ⟨g.map.comp f.map, by simp [f.point, g.point]⟩
  49   id_comp := by intros; ext; rfl
  50   comp_id := by intros; ext; rfl
  51   assoc := by intros; ext; rfl

  53 def suspendMap {X Y : BasedSpace.{u}} (f : X ⟶ Y) : X.suspension ⟶ Y.suspension where
  54   map := suspensionMap X Y f.map f.point
  55   point := suspensionProjection_collapsed _ _ (Or.inr (Or.inl rfl))

  57 def iterateSuspension (X : BasedSpace.{u}) : ℕ → BasedSpace.{u}
  58   | 0 => X
  59   | n+1 => (iterateSuspension X n).suspension

  61 def iterateSuspensionMap {X Y : BasedSpace.{u}} (f : X ⟶ Y) :
  62     ∀ n, iterateSuspension X n ⟶ iterateSuspension Y n
  63   | 0 => f
  64   | n+1 => suspendMap (iterateSuspensionMap f n)

  66 def SmashRelation (X Y : BasedSpace.{u}) (p q : X × Y) : Prop :=
  67   (p.1 = X.point ∨ p.2 = Y.point) ∧ (q.1 = X.point ∨ q.2 = Y.point)

  69 /-- Reduced smash in compactly generated spaces: k-ify the product before
  70 collapsing its wedge. -/
  71 def smash (X Y : BasedSpace.{u}) : BasedSpace.{u} where
  72   space := by
  73     letI : TopologicalSpace (X × Y) := TopologicalSpace.compactlyGenerated.{u} (X × Y)
  74     exact TopCat.of (Quot (SmashRelation X Y))
  75   point := Quot.mk _ (X.point,Y.point)
  76 
  77 def smashPoint (X Y : BasedSpace.{u}) (x : X) (y : Y) : X.smash Y := Quot.mk _ (x,y)
  78 
  79 theorem smashPoint_base_left (X Y : BasedSpace.{u}) (y : Y) :
  80     smashPoint X Y X.point y = (X.smash Y).point := Quot.sound ⟨Or.inl rfl, Or.inl rfl⟩
  81 
  82 theorem smashPoint_base_right (X Y : BasedSpace.{u}) (x : X) :
  83     smashPoint X Y x Y.point = (X.smash Y).point := Quot.sound ⟨Or.inr rfl, Or.inl rfl⟩

  85 /-- The representative formulas uniquely determine the induced based maps.
  86 Only their continuous quotient extension is a construction obligation. -/
  87 theorem smashMap_exists {X Y X' Y' : BasedSpace.{u}} (f : X ⟶ X') (g : Y ⟶ Y') :
  88     ∃! h : X.smash Y ⟶ X'.smash Y', ∀ x y,
  89       h.map (smashPoint X Y x y) = smashPoint X' Y' (f.map x) (g.map y) := by
  90   have hc : Continuous[TopologicalSpace.compactlyGenerated.{u} (X × Y),
  91       TopologicalSpace.compactlyGenerated.{u} (X' × Y')]
  92       (fun p : X × Y => (f.map p.1,g.map p.2)) := by
  93     refine @continuous_from_compactlyGenerated (X × Y) (X' × Y') _
  94       (TopologicalSpace.compactlyGenerated.{u} (X' × Y')) _ ?_
  95     intro S k
  96     let l : C(S, X' × Y') :=
  97       ⟨fun s => (f.map (k s).1,g.map (k s).2),
  98         (f.map.continuous.comp k.continuous.fst).prodMk
  99           (g.map.continuous.comp k.continuous.snd)⟩
 100     let ev : (Σ i : (T : CompHaus.{u}) × C(T, X' × Y'), i.fst) → X' × Y' :=
 101       fun q => q.1.2 q.2
 102     have hi : Continuous (fun s : S => (⟨⟨S,l⟩,s⟩ :
 103         Σ i : (T : CompHaus.{u}) × C(T, X' × Y'), i.fst)) :=
 104       @continuous_sigmaMk _ (fun i : (T : CompHaus.{u}) × C(T, X' × Y') => i.fst)
 105         (fun _ => inferInstance) ⟨S,l⟩
 106     letI : TopologicalSpace (X' × Y') := TopologicalSpace.coinduced ev inferInstance
 107     exact (show Continuous ev from continuous_coinduced_rng).comp hi
 108   let raw : X × Y → X'.smash Y' := fun p => smashPoint X' Y' (f.map p.1) (g.map p.2)
 109   have collapsed (p : X × Y) (hp : p.1 = X.point ∨ p.2 = Y.point) :
 110       raw p = (X'.smash Y').point := by
 111     rcases hp with h | h
 112     · dsimp [raw]
 113       rw [h, f.point]
 114       exact smashPoint_base_left _ _ _
 115     · dsimp [raw]
 116       rw [h, g.point]
 117       exact smashPoint_base_right _ _ _
 118   have hr : ∀ a b, SmashRelation X Y a b → raw a = raw b :=
 119     fun a b h => (collapsed a h.1).trans (collapsed b h.2).symm
 120   letI : TopologicalSpace (X × Y) := TopologicalSpace.compactlyGenerated.{u} (X × Y)
 121   letI : TopologicalSpace (X' × Y') := TopologicalSpace.compactlyGenerated.{u} (X' × Y')
 122   let h : X.smash Y ⟶ X'.smash Y' :=
 123     { map := ⟨Quot.lift raw hr, continuous_quot_lift hr (continuous_quot_mk.comp hc)⟩
 124       point := collapsed _ (Or.inl rfl) }
 125   refine ⟨h, fun _ _ => rfl, ?_⟩
 126   intro h' hh'
 127   apply Hom.ext
 128   apply ContinuousMap.ext
 129   intro z
 130   refine Quot.inductionOn z ?_
 131   intro p
 132   exact hh' p.1 p.2
 133 
 134 noncomputable def smashMap {X Y X' Y' : BasedSpace.{u}} (f : X ⟶ X') (g : Y ⟶ Y') :
 135     X.smash Y ⟶ X'.smash Y' := (smashMap_exists f g).exists.choose

 137 /-- Actual based CW homotopy type, using characteristic cells and weak
 138 CW topology from Mathlib, and homotopies fixed at the specified basepoint. -/
 139 def HasCWType (A : BasedSpace.{u}) : Prop :=
 140   ∃ Y : BasedSpace.{u}, T2Space Y ∧
 141     Nonempty (Topology.CWComplex (Set.univ : Set Y)) ∧ Y.IsWellBased ∧
 142     ∃ (f : A ⟶ Y) (g : Y ⟶ A),
 143       (g.map.comp f.map).HomotopicRel (ContinuousMap.id A) {A.point} ∧
 144       (f.map.comp g.map).HomotopicRel (ContinuousMap.id Y) {Y.point}
 145 
 146 /-- The derived-smash comparison is restricted to compactly generated,
 147 well-based spaces of actual CW homotopy type. Merely being well-based does
 148 not supply the flatness hypothesis used in MMSS, Proposition 11.7. -/
 149 def IsGood (A : BasedSpace.{u}) : Prop :=
 150   UCompactlyGeneratedSpace.{u} A ∧ A.IsWellBased ∧ A.HasCWType

 155 theorem smashUnit_exists (A : BasedSpace.{0}) (hA : A.IsGood) :
 156     ∃! f : zeroSphere.smash A ⟶ A,
 157       ∀ a, f.map (smashPoint zeroSphere A (show zeroSphere from (1 : Fin 2)) a) = a := by
 158   classical
 159   letI : DiscreteTopology zeroSphere := inferInstanceAs (DiscreteTopology (Fin 2))
 160   let raw : zeroSphere × A → A := fun p => if p.1 = zeroSphere.point then A.point else p.2
 161   have hc : Continuous raw := by
 162     apply continuous_prod_of_discrete_left.mpr
 163     intro x
 164     dsimp [raw]
 165     split_ifs
 166     · exact continuous_const
 167     · exact continuous_id
 168   have collapsed (p : zeroSphere × A)
 169       (hp : p.1 = zeroSphere.point ∨ p.2 = A.point) : raw p = A.point := by
 170     rcases hp with h | h
 171     · simp [raw, h]
 172     · simp [raw, h]
 173   have hr : ∀ a b, SmashRelation zeroSphere A a b → raw a = raw b :=
 174     fun a b h => (collapsed a h.1).trans (collapsed b h.2).symm
 175   have hck : Continuous[TopologicalSpace.compactlyGenerated.{0} (zeroSphere × A),
 176       inferInstance] raw :=
 177     continuous_from_compactlyGenerated raw (fun _ k => hc.comp k.continuous)
 178   letI : TopologicalSpace (zeroSphere × A) :=
 179     TopologicalSpace.compactlyGenerated.{0} (zeroSphere × A)
 180   let f : zeroSphere.smash A ⟶ A :=
 181     { map := ⟨Quot.lift raw hr, continuous_quot_lift hr hck⟩
 182       point := collapsed _ (Or.inl rfl) }
 183   refine ⟨f, ?_, ?_⟩
 184   · intro a
 185     change raw ((1 : Fin 2), a) = a
 186     simp [raw, zeroSphere, Prespectrum.sphereLevel]
 187   · intro g hg
 188     apply Hom.ext
 189     apply ContinuousMap.ext
 190     intro z
 191     refine Quot.inductionOn z ?_
 192     rintro ⟨x,a⟩
 193     change g.map (smashPoint zeroSphere A x a) = raw (x,a)
 194     change Fin 2 at x
 195     fin_cases x
 196     · change g.map (smashPoint zeroSphere A zeroSphere.point a) = raw (zeroSphere.point,a)
 197       rw [smashPoint_base_left, g.point]
 198       simp [raw]
 199     · simpa [raw, zeroSphere, Prespectrum.sphereLevel] using hg a

 204 theorem good_zeroSphere : zeroSphere.IsGood := by
 205   classical
 206   letI : DiscreteTopology zeroSphere := inferInstanceAs (DiscreteTopology (Fin 2))
 207   have hwb : zeroSphere.IsWellBased := by
 208     refine ⟨isClosed_singleton, ?_⟩
 209     intro Y f p hp
 210     let H : C(zeroSphere × I,Y) := ContinuousMap.uncurry
 211       ⟨fun x => if x = zeroSphere.point then p else ContinuousMap.const I (f x),
 212         continuous_of_discreteTopology⟩
 213     refine ⟨H, ?_, ?_⟩
 214     · intro x
 215       by_cases hx : x = zeroSphere.point
 216       · simpa [H, hx] using hp
 217       · simp [H, hx]
 218     · intro t
 219       simp [H]
 220   refine ⟨inferInstance, hwb, zeroSphere, inferInstance, ⟨inferInstance⟩, hwb,
 221     𝟙 _, 𝟙 _, ?_, ?_⟩
 222   all_goals exact ContinuousMap.HomotopicRel.refl _

 228 def suspensionSpectrum (A : BasedSpace.{u}) : Prespectrum.{u} where
 229   level := A.iterateSuspension
 230   bond _n := BasedSpace.suspensionProjection _
 231   bond_point _ _ := BasedSpace.suspensionProjection_collapsed _ _ (Or.inl rfl)
 232   bond_zero _ _ := BasedSpace.suspensionProjection_collapsed _ _ (Or.inr (Or.inl rfl))
 233   bond_one _ _ := BasedSpace.suspensionProjection_collapsed _ _ (Or.inr (Or.inr rfl))

 235 theorem suspensionSpectrum_zeroSphere :
 236     suspensionSpectrum BasedSpace.zeroSphere = sphere := by
 237   have h : BasedSpace.iterateSuspension BasedSpace.zeroSphere = sphereLevel := by
 238     funext n
 239     induction n with
 240     | zero => rfl
 241     | succ n ih => exact congrArg BasedSpace.suspension ih
 242   have ext (X Y : Prespectrum.{0}) (hlevel : X.level = Y.level)
 243       (hbond : ∀ n, HEq (X.bond n) (Y.bond n)) : X = Y := by
 244     cases X
 245     cases Y
 246     cases hlevel
 247     have hb := funext fun n => eq_of_heq (hbond n)
 248     cases hb
 249     rfl
 250   apply ext _ _ h
 251   intro n
 252   have hp (A B : BasedSpace.{0}) (hab : A = B) :
 253       HEq A.suspensionProjection B.suspensionProjection := by
 254     cases hab
 255     rfl
 256   exact hp _ _ (congrFun h n)

 264 /-- The reduced mapping-cone quotient at one level. -/
 265 inductive ConeRelation {X Y : Prespectrum.{u}} (f : X ⟶ Y) (n : ℕ) :
 266     Y.level n ⊕ (X.level n × I) → Y.level n ⊕ (X.level n × I) → Prop
 267   | start (x : X.level n) : ConeRelation f n (.inr (x,0)) (.inl (f.level n x))
 268   | finish (x : X.level n) : ConeRelation f n (.inr (x,1)) (.inl (Y.level n).point)
 269   | point (t : I) : ConeRelation f n (.inr ((X.level n).point,t)) (.inl (Y.level n).point)
 270 
 271 def coneLevel {X Y : Prespectrum.{u}} (f : X ⟶ Y) (n : ℕ) : BasedSpace.{u} where
 272   space := TopCat.of (Quot (ConeRelation f n))
 273   point := Quot.mk _ (.inl (Y.level n).point)
 274 
 275 def coneBondRepresentative {X Y : Prespectrum.{u}} (f : X ⟶ Y) (n : ℕ)
 276     (t : I) : Y.level n ⊕ (X.level n × I) → coneLevel f (n + 1)
 277   | .inl y => Quot.mk _ (.inl (Y.bond n (y,t)))
 278   | .inr (x,s) => Quot.mk _ (.inr (X.bond n (x,t),s))
 279 
 280 theorem coneBondRepresentative_respects {X Y : Prespectrum.{u}} (f : X ⟶ Y) (n : ℕ)
 281     (t : I) (a b) (h : ConeRelation f n a b) :
 282     coneBondRepresentative f n t a = coneBondRepresentative f n t b := by
 283   cases h with
 284   | start x =>
 285     change Quot.mk (ConeRelation f (n+1)) (.inr (X.bond n (x,t),0)) =
 286       Quot.mk (ConeRelation f (n+1)) (.inl (Y.bond n (f.level n x,t)))
 287     rw [← f.bond]
 288     exact Quot.sound (ConeRelation.start _)
 289   | finish x =>
 290     change Quot.mk (ConeRelation f (n+1)) (.inr (X.bond n (x,t),1)) =
 291       Quot.mk (ConeRelation f (n+1)) (.inl (Y.bond n ((Y.level n).point,t)))
 292     rw [Y.bond_point]
 293     exact Quot.sound (ConeRelation.finish _)
 294   | point s =>
 295     change Quot.mk (ConeRelation f (n+1)) (.inr (X.bond n ((X.level n).point,t),s)) =
 296       Quot.mk (ConeRelation f (n+1)) (.inl (Y.bond n ((Y.level n).point,t)))
 297     rw [X.bond_point, Y.bond_point]
 298     exact Quot.sound (ConeRelation.point s)

 300 def coneBond {X Y : Prespectrum.{u}} (f : X ⟶ Y) (n : ℕ)
 301     (p : coneLevel f n × I) : coneLevel f (n + 1) :=
 302   Quot.lift (coneBondRepresentative f n p.2)
 303     (coneBondRepresentative_respects f n p.2) p.1
 304 
 305 theorem continuous_coneBond {X Y : Prespectrum.{u}} (f : X ⟶ Y) (n : ℕ) :
 306     Continuous (coneBond f n) := by
 307   apply isQuotientMap_quot_mk.continuous_lift_prod_left
 308   have hleft : Continuous (fun p : Y.level n × I =>
 309       Quot.mk (ConeRelation f (n+1)) (.inl (Y.bond n p))) :=
 310     continuous_quot_mk.comp (continuous_inl.comp (Y.bond n).continuous)
 311   have hright : Continuous (fun p : (X.level n × I) × I =>
 312       Quot.mk (ConeRelation f (n+1)) (.inr (X.bond n (p.1.1,p.2),p.1.2))) :=
 313     continuous_quot_mk.comp (continuous_inr.comp
 314       (((X.bond n).continuous.comp (continuous_fst.fst.prodMk continuous_snd)).prodMk
 315         continuous_fst.snd))
 316   have hcont := (hleft.sumElim hright).comp
 317     (Homeomorph.sumProdDistrib (X := Y.level n) (Y := X.level n × I) (Z := I)).continuous
 318   apply hcont.congr
 319   rintro ⟨p,t⟩
 320   cases p <;> rfl

 322 def cone {X Y : Prespectrum.{u}} (f : X ⟶ Y) : Prespectrum.{u} where
 323   level := coneLevel f
 324   bond n := ⟨coneBond f n, continuous_coneBond f n⟩
 325   bond_point := by intro n t; change Quot.mk (ConeRelation f (n+1)) (Sum.inl (Y.bond n (_,t))) = _; rw [Y.bond_point]; rfl
 326   bond_zero := by
 327     intro n x
 328     refine Quot.inductionOn x ?_
 329     rintro (y | ⟨x,t⟩)
 330     · change Quot.mk (ConeRelation f (n+1)) (Sum.inl (Y.bond n (y,0))) = _
 331       rw [Y.bond_zero]; rfl
 332     · change Quot.mk (ConeRelation f (n+1)) (Sum.inr (X.bond n (x,0),t)) = _
 333       rw [X.bond_zero]
 334       exact Quot.sound (ConeRelation.point t)
 335   bond_one := by
 336     intro n x
 337     refine Quot.inductionOn x ?_
 338     rintro (y | ⟨x,t⟩)
 339     · change Quot.mk (ConeRelation f (n+1)) (Sum.inl (Y.bond n (y,1))) = _
 340       rw [Y.bond_one]; rfl
 341     · change Quot.mk (ConeRelation f (n+1)) (Sum.inr (X.bond n (x,1),t)) = _
 342       rw [X.bond_one]
 343       exact Quot.sound (ConeRelation.point t)

 351 def coneBoundaryRepresentative {X Y : Prespectrum.{u}} (f : X ⟶ Y) (n : ℕ) :
 352     Y.level n ⊕ (X.level n × I) → (X.level n).suspension
 353   | .inl _ => (X.level n).suspension.point
 354   | .inr p => BasedSpace.suspensionProjection _ p
 355 
 356 theorem coneBoundaryRepresentative_respects {X Y : Prespectrum.{u}} (f : X ⟶ Y)
 357     (n : ℕ) (a b) (h : ConeRelation f n a b) :
 358     coneBoundaryRepresentative f n a = coneBoundaryRepresentative f n b := by
 359   cases h with
 360   | start x =>
 361     exact BasedSpace.suspensionProjection_collapsed _ _ (Or.inr (Or.inl rfl))
 362   | finish x =>
 363     exact BasedSpace.suspensionProjection_collapsed _ _ (Or.inr (Or.inr rfl))
 364   | point t =>
 365     exact BasedSpace.suspensionProjection_collapsed _ _ (Or.inl rfl)
 366 
 367 def coneBoundaryLevel {X Y : Prespectrum.{u}} (f : X ⟶ Y) (n : ℕ) :
 368     coneLevel f n → (X.level n).suspension :=
 369   Quot.lift (coneBoundaryRepresentative f n) (coneBoundaryRepresentative_respects f n)
 370 
 371 theorem continuous_coneBoundaryLevel {X Y : Prespectrum.{u}} (f : X ⟶ Y) (n : ℕ) :
 372     Continuous (coneBoundaryLevel f n) := by
 373   exact continuous_quot_lift _ (continuous_sum_dom.mpr
 374     ⟨continuous_const, (BasedSpace.suspensionProjection (X.level n)).continuous⟩)

 376 def coneBoundary {X Y : Prespectrum.{u}} (f : X ⟶ Y) : cone f ⟶ X.suspension where
 377   level n := ⟨coneBoundaryLevel f n, continuous_coneBoundaryLevel f n⟩
 378   point := fun _ => rfl
 379   bond := by
 380     intro n x t
 381     refine Quot.inductionOn x ?_
 382     rintro (y | ⟨x,s⟩)
 383     · change (X.level (n+1)).suspension.point =
 384         BasedSpace.suspensionProjection _ (X.bond n ((X.level n).point,t),0)
 385       exact (BasedSpace.suspensionProjection_collapsed _ _ (Or.inr (Or.inl rfl))).symm
 386     · rfl
