  16 /-!
  17 # Concrete sequential-prespectrum source for the fixed Adams background
  18 
  19 Sequential prespectra are sequences of based spaces with continuous maps
  20 `X n × I → X (n+1)` constant on the basepoint cylinder and both ends.
  21 The stable homotopy sets below are the quotient of actual cubical based maps
  22 by based homotopy and these structure maps. Stable equivalences are precisely
  23 the maps inducing bijections in every integer stable degree. In particular,
  24 neither the category nor its weak equivalences are unspecified inputs.
  25 
  26 This is the ordinary, uncompleted stable homotopy source. Identification of a
  27 completed model must use an additional localization, not rename this sphere.
  28 
  29 Adapted from `KIP126/Def/StableHomotopy/Implementation/Prespectrum.lean` in
  30 SII-MATH/Kervaire126, develop commit
  31 `8cb35dad5f75ecc095e474df6c765baec9b175bf`. All declarations remain internal
  32 to AdamsSynthetic; public consumers receive the selected source recognition.
  33 -/

  42 /-- A space with a specified basepoint; the topology is part of `TopCat`. -/
  43 structure BasedSpace where
  44   space : TopCat.{u}
  45   point : space
  46 
  47 instance : CoeSort BasedSpace (Type u) := ⟨fun X => X.space⟩
  48 instance (X : BasedSpace) : TopologicalSpace X := X.space.str
  49 
  50 /-- Adjoint structure maps are written without choosing a loop-space model. -/
  51 structure Prespectrum where
  52   level : ℕ → BasedSpace.{u}
  53   bond : ∀ n, C(level n × I, level (n + 1))
  54   bond_point : ∀ n t, bond n ((level n).point, t) = (level (n + 1)).point
  55   bond_zero : ∀ n x, bond n (x, 0) = (level (n + 1)).point
  56   bond_one : ∀ n x, bond n (x, 1) = (level (n + 1)).point
  57 
  58 namespace Prespectrum

  60 /-- A levelwise continuous based map commuting with the actual bonds. -/
  61 @[ext]
  62 structure Hom (X Y : Prespectrum.{u}) where
  63   level : ∀ n, C(X.level n, Y.level n)
  64   point : ∀ n, level n (X.level n).point = (Y.level n).point
  65   bond : ∀ n x t, level (n + 1) (X.bond n (x,t)) = Y.bond n (level n x,t)
  66 
  67 instance : Category Prespectrum.{u} where
  68   Hom := Hom
  69   id X :=
  70     { level := fun _ => ContinuousMap.id _
  71       point := fun _ => rfl
  72       bond := fun _ _ _ => rfl }
  73   comp f g :=
  74     { level := fun n => (g.level n).comp (f.level n)
  75       point := by intro n; simp [f.point, g.point]
  76       bond := by intro n x t; simp only [ContinuousMap.comp_apply, f.bond, g.bond] }
  77   id_comp := by intros; ext; rfl
  78   comp_id := by intros; ext; rfl
  79   assoc := by intros; ext; rfl

  81 /-- An actual cubical representative in degree `d`, at a sufficiently high
  82 level. No homotopy group or stable stem is supplied as independent data. -/
  83 structure Representative (X : Prespectrum.{u}) (d : ℤ) where
  84   stage : ℕ
  85   dimension : ℕ
  86   degree : (dimension : ℤ) = d + stage
  87   loop : GenLoop (Fin dimension) (X.level stage) (X.level stage).point
  88 
  89 /-- Stabilization of a cubical loop by the displayed prespectrum bond.
  90 The new first coordinate is the structure-map interval. -/
  91 def stabilizeLoop (X : Prespectrum.{u}) (n r : ℕ)
  92     (p : GenLoop (Fin r) (X.level n) (X.level n).point) :
  93     GenLoop (Fin (r + 1)) (X.level (n + 1)) (X.level (n + 1)).point := by
  94   refine ⟨⟨fun t => X.bond n (p (fun i => t i.succ), t 0), ?_⟩, ?_⟩
  95   · exact (X.bond n).continuous.comp
  96       ((p.val.continuous.comp (continuous_pi fun i => continuous_apply i.succ)).prodMk
  97         (continuous_apply 0))
  98   · intro t ht
  99     obtain ⟨i, hi⟩ := ht
 100     induction i using Fin.cases with
 101     | zero =>
 102       rcases hi with h | h
 103       · simpa [h] using X.bond_zero n (p (fun i => t i.succ))
 104       · simpa [h] using X.bond_one n (p (fun i => t i.succ))
 105     | succ j =>
 106       have hp : p (fun i => t i.succ) = (X.level n).point := p.property _ ⟨j, hi⟩
 107       simpa [hp] using X.bond_point n (t 0)
 108 
 109 def Representative.next {X : Prespectrum.{u}} {d : ℤ}
 110     (a : Representative X d) : Representative X d where
 111   stage := a.stage + 1
 112   dimension := a.dimension + 1
 113   degree := by have := a.degree; omega
 114   loop := stabilizeLoop X a.stage a.dimension a.loop
 115 
 116 /-- Relations generating the colimit of the based homotopy sets. -/
 117 inductive StableRelation (X : Prespectrum.{u}) (d : ℤ) :
 118     Representative X d → Representative X d → Prop
 119   | homotopy (a : Representative X d)
 120       (p : GenLoop (Fin a.dimension) (X.level a.stage) (X.level a.stage).point)
 121       (h : GenLoop.Homotopic a.loop p) : StableRelation X d a { a with loop := p }
 122   | stabilization (a : Representative X d) : StableRelation X d a a.next
 123 
 124 -- Use the source construction's stable-homotopy name within this module.
 125 set_option linter.dupNamespace false in
 126 /-- The set underlying the usual integer-graded stable homotopy group. -/
 127 def StableHomotopy (X : Prespectrum.{u}) (d : ℤ) := Quot (StableRelation X d)
 128 
 129 def Representative.map {X Y : Prespectrum.{u}} (f : X ⟶ Y) {d : ℤ}
 130     (a : Representative X d) : Representative Y d where
 131   stage := a.stage
 132   dimension := a.dimension
 133   degree := a.degree
 134   loop := ⟨(f.level a.stage).comp a.loop.val, by
 135     intro t ht
 136     change f.level a.stage (a.loop t) = _
 137     exact (congrArg (fun x => f.level a.stage x) (a.loop.property t ht)).trans
 138       (f.point a.stage)⟩
 139 
 140 /-- Based homotopies and stabilization are respected by the actual maps.
 141 This proof obligation does not choose a new induced map. -/
 142 theorem Representative.map_relation {X Y : Prespectrum.{u}} (f : X ⟶ Y) {d : ℤ}
 143     {a b : Representative X d} (h : StableRelation X d a b) :
 144     StableRelation Y d (a.map f) (b.map f) := by
 145   cases h with
 146   | homotopy p h =>
 147     exact StableRelation.homotopy (a.map f) _
 148       (h.comp_continuousMap (f.level a.stage))
 149   | stabilization =>
 150     have hm : a.next.map f = (a.map f).next := by
 151       unfold Representative.next Representative.map
 152       congr 1
 153       apply Subtype.ext
 154       ext t
 155       exact f.bond a.stage _ _
 156     rw [hm]
 157     exact StableRelation.stabilization (a.map f)
 158 
 159 def stableHomotopyMap {X Y : Prespectrum.{u}} (f : X ⟶ Y) (d : ℤ) :
 160     StableHomotopy X d → StableHomotopy Y d :=
 161   Quot.map (Representative.map f) (fun _ _ h => Representative.map_relation f h)
 162 
 163 /-- Quotient class of a displayed cubical loop. -/
 164 def loopClass (X : Prespectrum.{u}) (d : ℤ) (n r : ℕ)
 165     (h : (r : ℤ) = d + n)
 166     (p : GenLoop (Fin r) (X.level n) (X.level n).point) : StableHomotopy X d :=
 167   Quot.mk _ ⟨n,r,h,p⟩
 168 
 169 /-- Stable weak equivalences, defined using the actual point-set system. -/
 170 def stableEquivalences : MorphismProperty Prespectrum.{u} :=
 171   fun _ _ f => ∀ d : ℤ, Function.Bijective (stableHomotopyMap f d)
 172 
 173 end Prespectrum
