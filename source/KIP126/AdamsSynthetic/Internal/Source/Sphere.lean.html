  14 /-!
  15 # The concrete sphere and shifts in the sequential-prespectrum source
  16 
  17 The zero sphere is the discrete based two-point space. Higher sphere levels
  18 are the actual reduced suspension quotients, with bonding maps the quotient
  19 projections. Suspension and desuspension are explicit operations on the same
  20 prespectra; they are not unspecified object choices.
  21 
  22 Adapted from `Implementation/PointSet.lean` and the padding desuspension in
  23 `Implementation/Generators.lean`, SII-MATH/Kervaire126 develop commit
  24 `8cb35dad5f75ecc095e474df6c765baec9b175bf`.
  25 -/

  36 /-- The actual basepoint inclusion is closed and has the homotopy extension
  37 property against continuous maps, the hypothesis needed for reduced suspension. -/
  38 def IsWellBased (X : BasedSpace.{u}) : Prop :=
  39   IsClosed ({X.point} : Set X) ∧
  40     ∀ (Y : TopCat.{u}) (f : C(X,Y)) (p : C(I,Y)), p 0 = f X.point →
  41       ∃ H : C(X × I,Y), (∀ x, H (x,0) = f x) ∧ (∀ t, H (X.point,t) = p t)
  42 
  43 def SuspensionCollapsed (X : BasedSpace.{u}) (p : X × I) : Prop :=
  44   p.1 = X.point ∨ p.2 = 0 ∨ p.2 = 1
  45 
  46 def SuspensionRelation (X : BasedSpace.{u}) (p q : X × I) : Prop :=
  47   SuspensionCollapsed X p ∧ SuspensionCollapsed X q
  48 
  49 def suspension (X : BasedSpace.{u}) : BasedSpace.{u} where
  50   space := TopCat.of (Quot (SuspensionRelation X))
  51   point := Quot.mk _ (X.point, 0)
  52 
  53 def suspensionProjection (X : BasedSpace.{u}) : C(X × I, X.suspension) :=
  54   ⟨Quot.mk _, continuous_quot_mk⟩
  55 
  56 theorem suspensionProjection_collapsed (X : BasedSpace.{u}) (p : X × I)
  57     (h : SuspensionCollapsed X p) : suspensionProjection X p = X.suspension.point :=
  58   Quot.sound ⟨h, Or.inl rfl⟩
  59 
  60 /-- Pointwise suspension of a based continuous family of maps. -/
  61 def suspensionFamily (X Y : BasedSpace.{u}) (f : C(X × I, Y))
  62     (hf : ∀ t, f (X.point,t) = Y.point) (p : X.suspension × I) : Y.suspension :=
  63   Quot.lift (fun q : X × I => suspensionProjection Y (f (q.1,p.2),q.2))
  64     (by
  65       intro a b hab
  66       apply Eq.trans _ (suspensionProjection_collapsed Y _ ?_).symm
  67       · apply suspensionProjection_collapsed
  68         rcases hab.1 with h | h | h
  69         · exact Or.inl (h ▸ hf p.2)
  70         · exact Or.inr (Or.inl h)
  71         · exact Or.inr (Or.inr h)
  72       · rcases hab.2 with h | h | h
  73         · exact Or.inl (h ▸ hf p.2)
  74         · exact Or.inr (Or.inl h)
  75         · exact Or.inr (Or.inr h)) p.1
  76 
  77 theorem continuous_suspensionFamily (X Y : BasedSpace.{u}) (f : C(X × I, Y))
  78     (hf : ∀ t, f (X.point,t) = Y.point) : Continuous (suspensionFamily X Y f hf) := by
  79   apply isQuotientMap_quot_mk.continuous_lift_prod_left
  80   exact (suspensionProjection Y).continuous.comp
  81     ((f.continuous.comp ((continuous_fst.fst).prodMk continuous_snd)).prodMk
  82       continuous_fst.snd)
  83 
  84 def suspensionMap (X Y : BasedSpace.{u}) (f : C(X,Y)) (hf : f X.point = Y.point) :
  85     C(X.suspension,Y.suspension) := by
  86   let raw : C(X × I,Y.suspension) :=
  87     (suspensionProjection Y).comp ⟨fun p => (f p.1,p.2),
  88       (f.continuous.comp continuous_fst).prodMk continuous_snd⟩
  89   have respects : ∀ a b, SuspensionRelation X a b → raw a = raw b := by
  90     intro a b hab
  91     have collapsed : ∀ p, SuspensionCollapsed X p → raw p = Y.suspension.point := by
  92       intro p hp
  93       apply suspensionProjection_collapsed
  94       rcases hp with h | h | h
  95       · left
  96         change f p.1 = Y.point
  97         rw [h]
  98         exact hf
  99       · exact Or.inr (Or.inl h)
 100       · exact Or.inr (Or.inr h)
 101     exact (collapsed a hab.1).trans (collapsed b hab.2).symm
 102   exact ⟨Quot.lift raw respects, continuous_quot_lift respects raw.continuous⟩
 103 
 104 end BasedSpace

 108 def IsWellBased (X : Prespectrum.{u}) : Prop := ∀ n, (X.level n).IsWellBased

 110 /-- The sphere prespectrum has level Sⁿ, formed by iterated actual reduced
 111 suspension of the discrete based two-point space. -/
 112 def sphereLevel : ℕ → BasedSpace
 113   | 0 => { space := TopCat.of (Fin 2), point := 0 }
 114   | n + 1 => (sphereLevel n).suspension
 115 
 116 def sphere : Prespectrum where
 117   level := sphereLevel
 118   bond n := BasedSpace.suspensionProjection (sphereLevel n)
 119   bond_point _n _t := BasedSpace.suspensionProjection_collapsed _ _ (Or.inl rfl)
 120   bond_zero _n _x := BasedSpace.suspensionProjection_collapsed _ _ (Or.inr (Or.inl rfl))
 121   bond_one _n _x := BasedSpace.suspensionProjection_collapsed _ _ (Or.inr (Or.inr rfl))
 122 
 123 /-- The positive generator of π₀ of the ordinary sphere: the non-basepoint
 124 of S⁰, viewed as a zero-dimensional cubical loop. -/
 125 def sphereGenerator : StableHomotopy sphere 0 :=
 126   loopClass sphere 0 0 0 (by norm_num)
 127     ⟨ContinuousMap.const _ (show sphere.level 0 from (1 : Fin 2)), by
 128       intro t ht
 129       obtain ⟨i, _⟩ := ht
 130       exact Fin.elim0 i⟩
 131 
 132 /-- Levelwise reduced suspension with the displayed interchange of the two
 133 interval variables. This is not an independently selected shift operation. -/
 134 def suspension (X : Prespectrum.{u}) : Prespectrum.{u} where
 135   level n := (X.level n).suspension
 136   bond n := ⟨BasedSpace.suspensionFamily _ _ (X.bond n) (X.bond_point n),
 137     BasedSpace.continuous_suspensionFamily _ _ (X.bond n) (X.bond_point n)⟩
 138   bond_point := by
 139     intro n t
 140     exact BasedSpace.suspensionProjection_collapsed _ _ (Or.inr (Or.inl rfl))
 141   bond_zero := by
 142     intro n x
 143     refine Quot.inductionOn x ?_
 144     intro p
 145     change BasedSpace.suspensionProjection _ (X.bond n (p.1,0),p.2) = _
 146     exact BasedSpace.suspensionProjection_collapsed _ _ (Or.inl (X.bond_zero n _))
 147   bond_one := by
 148     intro n x
 149     refine Quot.inductionOn x ?_
 150     intro p
 151     change BasedSpace.suspensionProjection _ (X.bond n (p.1,1),p.2) = _
 152     exact BasedSpace.suspensionProjection_collapsed _ _ (Or.inl (X.bond_one n _))
 153 
 154 def suspensionMap {X Y : Prespectrum.{u}} (f : X ⟶ Y) : X.suspension ⟶ Y.suspension where
 155   level n := BasedSpace.suspensionMap _ _ (f.level n) (f.point n)
 156   point := by
 157     intro n
 158     change BasedSpace.suspensionProjection _ (f.level n (X.level n).point,0) = _
 159     exact BasedSpace.suspensionProjection_collapsed _ _ (Or.inr (Or.inl rfl))
 160   bond := by
 161     intro n x t
 162     refine Quot.inductionOn x ?_
 163     intro p
 164     change BasedSpace.suspensionProjection _ (f.level (n+1) (X.bond n (p.1,t)),p.2) =
 165       BasedSpace.suspensionProjection _ (Y.bond n (f.level n p.1,t),p.2)
 166     rw [f.bond]
 167 
 168 /-- Padding with one zero level gives an actual desuspension prespectrum. -/
 169 def desuspension (X : Prespectrum.{u}) : Prespectrum.{u} where
 170   level
 171     | 0 => { space := TopCat.of PUnit, point := PUnit.unit }
 172     | n+1 => X.level n
 173   bond
 174     | 0 => ContinuousMap.const _ (X.level 0).point
 175     | n+1 => X.bond n
 176   bond_point := by intro n t; cases n <;> simp [X.bond_point]
 177   bond_zero := by intro n x; cases n <;> simp [X.bond_zero]
 178   bond_one := by intro n x; cases n <;> simp [X.bond_one]
 179 
 180 /-- The actual map on the padding desuspensions, with the unique zero-level map. -/
 181 def desuspensionMap {X Y : Prespectrum.{u}} (f : X ⟶ Y) :
 182     X.desuspension ⟶ Y.desuspension where
 183   level
 184     | 0 => ContinuousMap.id _
 185     | n + 1 => f.level n
 186   point := by
 187     intro n
 188     cases n with
 189     | zero => rfl
 190     | succ n => exact f.point n
 191   bond := by
 192     intro n x t
 193     cases n with
 194     | zero => exact f.point 0
 195     | succ n => exact f.bond n x t
 196 
 197 end Prespectrum
