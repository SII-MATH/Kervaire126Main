  28 private def homotopyDegreeEquiv (Y : C) {m n : ℤ} (h : m = n) :
  29     HomotopyGroup m Y ≃ₗ[ℤ] HomotopyGroup n Y := by
  30   subst n
  31   exact LinearEquiv.refl ℤ _
  32 
  33 private theorem homotopyDegreeEquiv_postcomp {Y Z : C} {m n : ℤ} (h : m = n)
  34     (a : HomotopyGroup m Y) (f : Y ⟶ Z) :
  35     homotopyDegreeEquiv Z h (a ≫ f) = homotopyDegreeEquiv Y h a ≫ f := by
  36   subst n
  37   rfl

  39 /-- The actual exact-couple first differential, including the ordinary
  40 homotopy-degree identification at its target. -/
  41 private noncomputable def towerFirstDifferential (T : AbstractAdamsTower unit X)
  42     (p : ℤ × ℤ) : T.FirstPage p →ₗ[ℤ] T.FirstPage (p.1+1,p.2) :=
  43   (T.layerMap (p.1+1,p.2)).comp
  44     ((homotopyDegreeEquiv (T.stageAt (p.1+1))
  45       (by dsimp; omega : p.2-p.1-1 = (p.1+1,p.2).2-(p.1+1,p.2).1)).toLinearMap.comp
  46         (T.connecting p))
  47 
  48 private theorem towerFirstDifferential_eq_zero_iff (T : AbstractAdamsTower unit X)
  49     (p : ℤ × ℤ) (x : T.FirstPage p) :
  50     towerFirstDifferential T p x = 0 ↔ x ∈ T.cycles 2 (by decide) p := by
  51   let hdegree : p.2-p.1-1 = p.2-(p.1+1) := by omega
  52   let f := T.mapAt (show p.1+1 ≤ (p.1+1)+1 by omega)
  53   let e := homotopyDegreeEquiv (T.stageAt (p.1+1)) hdegree
  54   let e' := homotopyDegreeEquiv (T.stageAt ((p.1+1)+1)) hdegree
  55   change e (T.connecting p x) ≫ HasFunctorialCofiber.cofibι f = 0 ↔
  56     ∃ a, a ≫ T.mapAt (show p.1+1 ≤ p.1+2 by omega) = T.connecting p x
  57   have hrange (q : ℤ) (hq : q = (p.1+1)+1) :
  58       (∃ a : HomotopyGroup (p.2-p.1-1) (T.stageAt q),
  59         a ≫ T.mapAt (show p.1+1 ≤ q by omega) = T.connecting p x) ↔
  60       ∃ a : HomotopyGroup (p.2-p.1-1) (T.stageAt ((p.1+1)+1)),
  61         a ≫ f = T.connecting p x := by
  62     subst q
  63     rfl
  64   rw [hrange (p.1+2) (by omega)]
  65   constructor
  66   · intro h
  67     obtain ⟨a,ha⟩ := Triangle.coyoneda_exact₂ _
  68       (HasFunctorialCofiber.cofib_distinguished f) (e (T.connecting p x)) h
  69     refine ⟨e'.symm a, ?_⟩
  70     apply e.injective
  71     change homotopyDegreeEquiv _ hdegree (e'.symm a ≫ f) = _
  72     rw [homotopyDegreeEquiv_postcomp]
  73     change e' (e'.symm a) ≫ f = _
  74     rw [LinearEquiv.apply_symm_apply]
  75     exact ha.symm
  76   · rintro ⟨a,ha⟩
  77     rw [← ha]
  78     change homotopyDegreeEquiv _ hdegree (a ≫ f) ≫ _ = 0
  79     rw [homotopyDegreeEquiv_postcomp, Category.assoc]
  80     have hzero : f ≫ HasFunctorialCofiber.cofibι f = 0 :=
  81       comp_distTriang_mor_zero₁₂ _ (HasFunctorialCofiber.cofib_distinguished f)
  82     rw [hzero, comp_zero]
  83 
  84 private def firstPageDegreeEquiv (T : AbstractAdamsTower unit X)
  85     {p q : ℤ × ℤ} (h : p = q) : T.FirstPage p ≃ₗ[ℤ] T.FirstPage q := by
  86   subst q
  87   exact LinearEquiv.refl ℤ _

  89 private noncomputable def towerFirstDifferentialNat (T : AbstractAdamsTower unit X)
  90     (s t : ℕ) : T.FirstPage ((s:ℤ),(t:ℤ)) →ₗ[ℤ]
  91       T.FirstPage (((s+1:ℕ):ℤ),(t:ℤ)) :=
  92   (firstPageDegreeEquiv T
  93     (show ((s:ℤ)+1,(t:ℤ)) = (((s+1:ℕ):ℤ),(t:ℤ)) by simp)).toLinearMap.comp
  94       (towerFirstDifferential T ((s:ℤ),(t:ℤ)))
  95 
  96 private theorem towerFirstDifferentialNat_eq_zero_iff (T : AbstractAdamsTower unit X)
  97     (s t : ℕ) (x : T.FirstPage ((s:ℤ),(t:ℤ))) :
  98     towerFirstDifferentialNat T s t x = 0 ↔
  99       x ∈ T.cycles 2 (by decide) ((s:ℤ),(t:ℤ)) := by
 100   let e := firstPageDegreeEquiv T
 101     (show ((s:ℤ)+1,(t:ℤ)) = (((s+1:ℕ):ℤ),(t:ℤ)) by simp)
 102   change e (towerFirstDifferential T ((s:ℤ),(t:ℤ)) x) = 0 ↔ _
 103   rw [LinearEquiv.map_eq_zero_iff, towerFirstDifferential_eq_zero_iff]

 105 private instance zeroCycleArrow_isIso (D : SSData (ModuleCat.{v} ℤ)) :
 106     IsIso (D.Z 0).arrow :=
 107   (Subobject.isIso_arrow_iff_eq_top _).mpr D.Z_zero

 109 private noncomputable def ambientToZeroPage (D : SSData (ModuleCat.{v} ℤ)) :
 110     D.V →ₗ[ℤ] D.page 0 :=
 111   (inv (D.Z 0).arrow ≫ D.pageπ 0).hom
 112 
 113 private theorem ambientToZeroPage_eq_zero_iff (D : SSData (ModuleCat.{v} ℤ))
 114     (a : D.V) : ambientToZeroPage D a = 0 ↔ a ∈ ModuleCat.subobjectModule _ (D.B 0) := by
 115   change D.pageπ 0 (inv (D.Z 0).arrow a) = 0 ↔ _
 116   have hz := subobject_cokernel_π_eq_zero_iff (D.B 0) (D.Z 0) (D.B_le_Z 0)
 117     (inv (D.Z 0).arrow a)
 118   change D.pageπ 0 (inv (D.Z 0).arrow a) = 0 ↔ _ at hz
 119   have h := congrArg (fun f => f a) (IsIso.inv_hom_id (D.Z 0).arrow)
 120   simp only [ModuleCat.comp_apply, ModuleCat.id_apply] at h
 121   exact hz.trans (by rw [h])
 122 
 123 private theorem ambientToZeroPage_surjective (D : SSData (ModuleCat.{v} ℤ)) :
 124     Function.Surjective (ambientToZeroPage D) := by
 125   letI : Epi (D.pageπ 0) := by dsimp only [SSData.pageπ]; infer_instance
 126   exact (ModuleCat.epi_iff_surjective (inv (D.Z 0).arrow ≫ D.pageπ 0)).mp inferInstance

 128 /-- The actual E₂ quotient map on the ambient module of second-page cycles. -/
 129 private noncomputable def towerE2ClassMap (T : AbstractAdamsTower unit X) (p : ℤ × ℤ) :
 130     T.cycles 2 (by decide) p →ₗ[ℤ] (towerSequence T).Page 2 p :=
 131   ambientToZeroPage (towerSSData T p)
 132 
 133 private noncomputable def towerE2ClassOfCycle (T : AbstractAdamsTower unit X)
 134     (p : ℤ × ℤ) (x : T.FirstPage p) (hx : x ∈ T.cycles 2 (by decide) p) :
 135     (towerSequence T).Page 2 p :=
 136   towerE2ClassMap T p ⟨x,hx⟩
 137 
 138 private theorem towerE2ClassOfCycle_eq_zero_iff (T : AbstractAdamsTower unit X)
 139     (p : ℤ × ℤ) (x : T.FirstPage p) (hx : x ∈ T.cycles 2 (by decide) p) :
 140     towerE2ClassOfCycle T p x hx = 0 ↔ x ∈ T.boundaries 2 (by decide) p := by
 141   exact (ambientToZeroPage_eq_zero_iff (towerSSData T p) ⟨x,hx⟩).trans
 142     (towerSSData_boundaries T p 0 ⟨x,hx⟩)
 143 
 144 private theorem towerE2ClassOfCycle_surjective (T : AbstractAdamsTower unit X)
 145     (p : ℤ × ℤ) (y : (towerSequence T).Page 2 p) :
 146     ∃ (x : T.FirstPage p) (hx : x ∈ T.cycles 2 (by decide) p),
 147       towerE2ClassOfCycle T p x hx = y := by
 148   obtain ⟨a,ha⟩ := ambientToZeroPage_surjective (towerSSData T p) y
 149   exact ⟨a.val,a.property,ha⟩
 150 
 151 private theorem towerE2ClassMap_eq_iff (T : AbstractAdamsTower unit X) (p : ℤ × ℤ)
 152     (x y : T.cycles 2 (by decide) p) :
 153     towerE2ClassMap T p x = towerE2ClassMap T p y ↔
 154       x.val-y.val ∈ T.boundaries 2 (by decide) p := by
 155   rw [← sub_eq_zero, ← map_sub]
 156   exact towerE2ClassOfCycle_eq_zero_iff T p (x-y).val (x-y).property

 158 private noncomputable def canonicalFirstDifferential (unit : SphereSpectrum ⟶ H)
 159     (X : C) (p : ℤ × ℤ) :
 160     (canonicalTower unit X).FirstPage p →ₗ[ℤ]
 161       (canonicalTower unit X).FirstPage (p.1+1,p.2) :=
 162   towerFirstDifferential (canonicalTower unit X) p

 164 private noncomputable def canonicalE2ClassOfCycle (unit : SphereSpectrum ⟶ H)
 165     (X : C) (p : ℤ × ℤ) (x : (canonicalTower unit X).FirstPage p)
 166     (hx : x ∈ (canonicalTower unit X).cycles 2 (by decide) p) :
 167     (canonicalSequence unit X).Page 2 p :=
 168   towerE2ClassOfCycle (canonicalTower unit X) p x hx

 170 private noncomputable def canonicalE2ClassOfFirstCocycle (unit : SphereSpectrum ⟶ H)
 171     (X : C) (p : ℤ × ℤ) (x : (canonicalTower unit X).FirstPage p)
 172     (hx : canonicalFirstDifferential unit X p x = 0) :
 173     (canonicalSequence unit X).Page 2 p :=
 174   canonicalE2ClassOfCycle unit X p x
 175     ((towerFirstDifferential_eq_zero_iff (canonicalTower unit X) p x).mp hx)
