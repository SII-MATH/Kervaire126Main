  22 public def FiniteExtension (r : ℕ) (n s t : ℤ)
  23     (x : (D.classical.obj (X.obj D.auxiliary)).Page 2 (s,t))
  24     (y : (D.classical.obj (Y.obj D.auxiliary)).Page 2 (s+n,t+n)) : Prop :=
  25   Nonempty (FiniteExtensionWitness D X Y f r n s t x y)
  26 
  27 public def FiniteExtensionWitness.Essential {r : ℕ} {n s t : ℤ} {x y}
  28     (W : FiniteExtensionWitness D X Y f r n s t x y) : Prop :=
  29   TowerEssentialExtension
  30     (D.convergentMap (.quotient (r - 1) (extensionSourceObject D X Y f))
  31       (.quotient (r - 1) (.nu Y)) (finiteExtensionMap D X Y f (r - 1)))
  32     (t - s, t + normalizedExponent D.classicalTower f) s n W.source W.target
  33 
  34 
  35 public def InfiniteExtension (n s t : ℤ)
  36     (x : (D.classical.obj (X.obj D.auxiliary)).Page 2 (s,t))
  37     (y : (D.classical.obj (Y.obj D.auxiliary)).Page 2 (s+n,t+n)) : Prop :=
  38   Nonempty (InfiniteExtensionWitness D X Y f n s t x y)
  39 
  40 public def InfiniteExtensionWitness.Essential {n s t : ℤ} {x y}
  41     (W : InfiniteExtensionWitness D X Y f n s t x y) : Prop :=
  42   TowerEssentialExtension
  43     (D.convergentMap (extensionSourceObject D X Y f) (.nu Y) (D.normalizedMap X Y f).map)
  44     (t - s, t + normalizedExponent D.classicalTower f) s n W.source W.target

  46 /-- The finite target page stays in the valid page-two range. -/
  47 public theorem FiniteExtensionWitness.target_page_ge_two {r : ℕ} {n s t : ℤ} {x y}
  48     (W : FiniteExtensionWitness D X Y f r n s t x y) :
  49     2 ≤ (r : ℤ) - n + normalizedExponent D.classicalTower f := by
  50   have h := W.length_upper
  51   omega
