  22 /-- The finite sphere/shift/cofiber closure, independently of the smaller
  23 configuration selected by the main proof. -/
  24 public inductive IsFiniteSpectrum : C → Prop where
  25   | sphere : IsFiniteSpectrum SphereSpectrum
  26   | shift {X : C} (n : ℤ) : IsFiniteSpectrum X → IsFiniteSpectrum (X⟦n⟧)
  27   | cofiber {X Y Z : C} {f : X ⟶ Y} {g : Y ⟶ Z} {h : Z ⟶ X⟦(1 : ℤ)⟧} :
  28       Triangle.mk f g h ∈ distTriang C →
  29       IsFiniteSpectrum X → IsFiniteSpectrum Y → IsFiniteSpectrum Z
  30 
  31 /-- Homology uses this coefficient object and the actual tensor product. -/
  32 public abbrev ModTwoHomology (H : ModTwoCoefficients C) (n : ℤ) (X : C) :=
  33   HomotopyGroup n (H.object ⊗ X)
  34 
  35 public def modTwoHomologyMap (H : ModTwoCoefficients C) {X Y : C}
  36     (f : X ⟶ Y) (n : ℤ) : ModTwoHomology H n X →+ ModTwoHomology H n Y :=
  37   inducedMap (H.object ◁ f) n
  38 
  39 public def ZeroOnModTwoHomology (H : ModTwoCoefficients C)
  40     {X Y : C} (f : X ⟶ Y) : Prop :=
  41   ∀ n, ∀ x : ModTwoHomology H n X, modTwoHomologyMap H f n x = 0
  42 
  43 /-- Positive-length factorizations through maps zero on the same HF₂ homology. -/
  44 public inductive AdamsMapFactorization (H : ModTwoCoefficients C) :
  45     {X Y : C} → ℕ → (X ⟶ Y) → Prop where
  46   | single {X Y : C} (f : X ⟶ Y) :
  47       ZeroOnModTwoHomology H f → AdamsMapFactorization H 1 f
  48   | prepend {X Y Z : C} {n : ℕ} (f : X ⟶ Y) (g : Y ⟶ Z) :
  49       ZeroOnModTwoHomology H f → AdamsMapFactorization H n g →
  50       AdamsMapFactorization H (n + 1) (f ≫ g)
  51 
  52 /-- The old lower-bound responsibility, without its redundant value wrapper. -/
  53 public def FactorizationFiltration (H : ModTwoCoefficients C)
  54     {X Y : C} (f : X ⟶ Y) (s : ℤ) : Prop :=
  55   ∀ n : ℕ, 0 < n → (n : ℤ) ≤ s → AdamsMapFactorization H n f

  57 variable {H : ModTwoCoefficients C} {A : ClassicalAdamsFamily C}

  59 /-- Integer filtration is the actual target-tower factorization; negative
  60 indices are stage zero, as in the tower itself. -/
  61 public def MapAdamsFiltration (P : ClassicalTowerPresentation H.unit A)
  62     {X Y : C} (f : X ⟶ Y) (s : ℤ) : Prop :=
  63   MapFiltrationAtLeast P f s.toNat
  64 
  65 public def ExactMapAdamsFiltration (P : ClassicalTowerPresentation H.unit A)
  66     {X Y : C} (f : X ⟶ Y) (s : ℤ) : Prop :=
  67   MapAdamsFiltration P f s ∧ ¬ MapAdamsFiltration P f (s + 1)
  68 
  69 /-- The comparison is an explicit obligation for the chosen coefficient tower. -/
  70 public structure FactorizationFiltrationComparison
  71     (P : ClassicalTowerPresentation H.unit A) : Prop where
  72   agrees : ∀ {X Y : C} (f : X ⟶ Y) (s : ℤ),
  73     FactorizationFiltration H f s ↔ MapAdamsFiltration P f s
  74 
  75 public theorem mapAdamsFiltration_zero (P : ClassicalTowerPresentation H.unit A)
  76     {X Y : C} (f : X ⟶ Y) : MapAdamsFiltration P f 0 := by
  77   refine ⟨f ≫ (P.tower Y).zeroIso.inv, ?_⟩
  78   change (f ≫ (P.tower Y).zeroIso.inv) ≫ (P.tower Y).projection 0 = f
  79   rw [AbstractAdamsTower.projection, AbstractAdamsTower.map_self]
  80   change (f ≫ (P.tower Y).zeroIso.inv) ≫
  81     (𝟙 _ ≫ (P.tower Y).zeroIso.hom) = f
  82   rw [Category.id_comp, Category.assoc, Iso.inv_hom_id, Category.comp_id]
  83 
  84 public theorem mapAdamsFiltration_mono (P : ClassicalTowerPresentation H.unit A)
  85     {X Y : C} (f : X ⟶ Y) {s t : ℤ} (hst : s ≤ t)
  86     (ht : MapAdamsFiltration P f t) : MapAdamsFiltration P f s := by
  87   obtain ⟨a, ha⟩ := ht
  88   refine ⟨a ≫ (P.tower Y).mapAt (show (s.toNat : ℤ) ≤ t.toNat by omega), ?_⟩
  89   rw [Category.assoc, AbstractAdamsTower.projection_compat]
  90   exact ha

  92 /-- Odd torsion includes zero. It must not be replaced by mere nonzeroness. -/
  93 public def IsOddTorsionMap {X Y : C} (f : X ⟶ Y) : Prop :=
  94   ∃ n : ℕ, Odd n ∧ 0 < n ∧ n • f = 0
  95 
  96 /-- Original foundational proof responsibility, retained as an explicit
  97 proof parameter. It is not an automatically supplied literature fact. -/
  98 public def FiniteMapSeparation (P : ClassicalTowerPresentation H.unit A) : Prop :=
  99   ∀ {X Y : C}, IsFiniteSpectrum X → IsFiniteSpectrum Y →
 100     ∀ f : X ⟶ Y, (∀ s : ℤ, MapAdamsFiltration P f s) → IsOddTorsionMap f
 101 
 102 public theorem separated_factorization (P : ClassicalTowerPresentation H.unit A)
 103     (comparison : FactorizationFiltrationComparison P) (separation : FiniteMapSeparation P)
 104     {X Y : C} (hX : IsFiniteSpectrum X) (hY : IsFiniteSpectrum Y) (f : X ⟶ Y)
 105     (hf : ∀ s : ℤ, FactorizationFiltration H f s) : IsOddTorsionMap f :=
 106   separation hX hY f (fun s => (comparison.agrees f s).mp (hf s))
 107 
 108 /-- Finite exact filtration is derived, with the finite-spectrum and
 109 non-odd-torsion conditions that the old unqualified statement omitted. -/
 110 public theorem finite_filtration_of_not_odd (P : ClassicalTowerPresentation H.unit A)
 111     (separation : FiniteMapSeparation P) {X Y : C}
 112     (hX : IsFiniteSpectrum X) (hY : IsFiniteSpectrum Y) (f : X ⟶ Y)
 113     (hf : ¬ IsOddTorsionMap f) : ∃ s : ℤ, ExactMapAdamsFiltration P f s := by
 114   classical
 115   have failure : ∃ n : ℕ, ¬ MapAdamsFiltration P f (n : ℤ) := by
 116     by_contra h
 117     push Not at h
 118     apply hf
 119     apply separation hX hY f
 120     intro s
 121     exact h s.toNat
 122   let n := Nat.find failure
 123   have hn : ¬ MapAdamsFiltration P f (n : ℤ) := Nat.find_spec failure
 124   have hnpos : 0 < n := by
 125     by_contra h
 126     have hn0 : n = 0 := by omega
 127     exact hn (by simpa [hn0] using mapAdamsFiltration_zero P f)
 128   refine ⟨((n - 1 : ℕ) : ℤ), ?_, ?_⟩
 129   · by_contra hprev
 130     have hle := Nat.find_min' failure hprev
 131     change n ≤ n - 1 at hle
 132     omega
 133   · have heq : ((n - 1 : ℕ) : ℤ) + 1 = (n : ℤ) := by omega
 134     simpa only [heq] using hn

 136 /-- Rational acyclicity expressed intrinsically by torsion of actual
 137 homotopy groups; no unspecified `True` hypothesis is used. -/
 138 public def RationallyAcyclic (X : C) : Prop :=
 139   ∀ n : ℤ, ∀ x : HomotopyGroup n X, ∃ k : ℕ, 0 < k ∧ k • x = 0
 140 
 141 /-- Odd multiplication is invertible on the actual homotopy groups. This
 142 excludes the odd torsion that otherwise survives every mod-two filtration. -/
 143 public def HomotopyTwoLocal (X : C) : Prop :=
 144   ∀ n : ℤ, ∀ k : ℕ, Odd k →
 145     Function.Bijective (fun x : HomotopyGroup n X => k • x)
 146 
 147 /-- A corrected boundedness input. A producer must justify this assertion
 148 on finite, rationally acyclic, two-local spectra in its chosen system. -/
 149 public structure AdamsBoundednessInput (P : ClassicalTowerPresentation H.unit A) : Prop where
 150   bounded : ∀ X : C, IsFiniteSpectrum X → RationallyAcyclic X → HomotopyTwoLocal X →
 151     ∀ i : ℤ, ∃ N : ℤ, ∀ s : ℤ, N < s → (P.tower X).filtrationSubmodule s i = ⊥
 152 
 153 public theorem AdamsBoundednessInput.excludes_nonzero
 154     {P : ClassicalTowerPresentation H.unit A} (B : AdamsBoundednessInput P)
 155     (X : C) (hX : IsFiniteSpectrum X) (hr : RationallyAcyclic X) (hl : HomotopyTwoLocal X)
 156     (i : ℤ) : ∃ N : ℤ, ∀ s : ℤ, N < s → ∀ x : HomotopyGroup i X,
 157       x ≠ 0 → x ∉ (P.tower X).filtrationSubmodule s i := by
 158   obtain ⟨N,hN⟩ := B.bounded X hX hr hl i
 159   refine ⟨N, fun s hs x hx hmem => ?_⟩
 160   rw [hN s hs] at hmem
 161   exact hx hmem
 162 
 163 public theorem filtration_contains_zero (P : ClassicalTowerPresentation H.unit A)
 164     (X : C) (s i : ℤ) : (0 : HomotopyGroup i X) ∈ (P.tower X).filtrationSubmodule s i :=
 165   Submodule.zero_mem _
