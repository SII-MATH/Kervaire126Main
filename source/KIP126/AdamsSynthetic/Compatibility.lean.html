  23 /-- The composition product through the given bigraded suspension. -/
  24 public def sphereProduct {m n k l : ℤ}
  25     (x : BiHom m n (S_0_0 : Syn)) (y : BiHom k l (S_0_0 : Syn)) :
  26     BiHom (m + k) (n + l) (S_0_0 : Syn) :=
  27   (SyntheticCategory.biShift_comp (m, n) (k, l)).inv.app S_0_0 ≫
  28     (SyntheticCategory.biShift (k, l)).map x ≫ y
  29 
  30 public def sphereAction {m n k l : ℤ} {X : Syn}
  31     (x : BiHom m n (S_0_0 : Syn)) (y : BiHom k l X) : BiHom (m + k) (n + l) X :=
  32   (SyntheticCategory.biShift_comp (m, n) (k, l)).inv.app S_0_0 ≫
  33     (SyntheticCategory.biShift (k, l)).map x ≫ y
  34 
  35 public def lambdaMultiply {m w : ℤ} {X : Syn} (k : ℕ)
  36     (x : BiHom m w X) : BiHom m (w - k) X :=
  37   eqToHom (by congr 1; simp) ≫
  38     (SyntheticCategory.biShift_comp (m, w) (0, -(k : ℤ))).inv.app S_0_0 ≫
  39       lambdaPow k (Smn m w) ≫ x
  40 
  41 public def homotopyRegrade {m m' w w' : ℤ} {X : Syn} (hm : m = m') (hw : w = w')
  42     (x : BiHom m w X) : BiHom m' w' X := by
  43   subst m'
  44   subst w'
  45   exact x
  46 
  47 public def biHomTargetIso (m n : ℤ) {X Y : Syn} (e : X ≅ Y) :
  48     BiHom m n X ≃+ BiHom m n Y where
  49   toFun x := x ≫ e.hom
  50   invFun y := y ≫ e.inv
  51   left_inv x := by simp
  52   right_inv y := by simp
  53   map_add' x y := by simp [Preadditive.add_comp]
  54 
  55 public def quotientClass [HasFunctorialCofiber (C := Syn)] {m n : ℤ} {X : Syn} (r : ℕ)
  56     (x : BiHom m n X) : BiHom m n (XModLambdaN X r) :=
  57   x ≫ XModLambdaN.incl X r
  58 end
  59 end KIP126.Synthetic.Context

  71 public abbrev FirstQuotientComparison (A : ClassicalAdamsFamily C)
  72     (N : NuFunctorData C Syn) (aux : AuxiliaryData C) :=
  73   ∀ (X : ClassicalObject) (a s t : ℤ),
  74     BiHom (t-s) (t+a)
  75       (XModLambdaN ((SyntheticCategory.biShift (0,a)).obj
  76         (N.functor.obj (X.obj aux))) 1) ≃ₗ[ℤ] (A.obj (X.obj aux)).Page 2 (s,t)

  78 /-- The sphere specialization is transported from the ν-sphere comparison
  79 using the unit iso and this same actual quotient functor. It is not another choice. -/
  80 public def sphereFirstQuotient {A : ClassicalAdamsFamily C}
  81     (N : NuFunctorData C Syn) {aux : AuxiliaryData C}
  82     (Q : LambdaQuotientFunctoriality (Syn := Syn))
  83     (comparison : FirstQuotientComparison A N aux) (s t : ℤ) :
  84     BiHom (t-s) t (XModLambdaN (S_0_0 : Syn) 1) ≃ₗ[ℤ] A.sphere.Page 2 (s,t) := by
  85   let e : (SyntheticCategory.biShift (0,0)).obj (N.functor.obj SphereSpectrum) ≅
  86       (S_0_0 : Syn) := SyntheticCategory.biShift_zero.app _ ≪≫ N.unitIso
  87   let q := (Q.functor 1).mapIso e
  88   let c : BiHom (t-s) t
  89       (XModLambdaN ((SyntheticCategory.biShift (0,0)).obj
  90         (N.functor.obj SphereSpectrum)) 1) ≃ₗ[ℤ] A.sphere.Page 2 (s,t) := by
  91     exact Eq.mp (congrArg (fun z : ℤ =>
  92       BiHom (t-s) z (XModLambdaN ((SyntheticCategory.biShift (0,0)).obj
  93         (N.functor.obj SphereSpectrum)) 1) ≃ₗ[ℤ] A.sphere.Page 2 (s,t))
  94       (Int.add_zero t)) (comparison .sphere 0 s t)
  95   exact (biHomTargetIso (t-s) t q.symm).toIntLinearEquiv.trans c

  97 /-- First-quotient and λ detection on the sphere, with the derived comparison. -/
  98 public structure SphereDetectionCompatible
  99     {A : ClassicalAdamsFamily C} {F : SyntheticAdamsFamily Syn}
 100     {N : NuFunctorData C Syn} {aux : AuxiliaryData C}
 101     (K : E2Comparison A F N aux)
 102     (Q : LambdaQuotientFunctoriality (Syn := Syn))
 103     (first : FirstQuotientComparison A N aux)
 104     {H : Syn} {unit : S_0_0 ⟶ H} {T : SyntheticTowerPresentation unit F}
 105     (cSphere : SyntheticTowerConvergence T (S_0_0 : Syn))
 106     (cQuotient : SyntheticTowerConvergence T (XModLambdaN (S_0_0 : Syn) 1)) : Prop where
 107   first_quotient : ∀ (s t : ℤ) (x : A.sphere.Page 2 (s,t)),
 108     Detects cQuotient (s,t,t)
 109       (F.pageMap (XModLambdaN.incl S_0_0 1) 2 (s,t,t)
 110         (by simpa only [SyntheticAdamsFamily.sphere, SyntheticAdamsSS.Page,
 111           Nat.cast_zero, sub_zero] using K.sphereE2 s t 0 x))
 112       ((sphereFirstQuotient N Q first s t).symm x)
 113   homotopy_lambda : ∀ (s t : ℤ) (k : ℕ) (x : BiHom (t-s) t (S_0_0 : Syn)),
 114     Detects cSphere (s,t,t-k)
 115       (K.sphereE2 s t k (sphereFirstQuotient N Q first s t (quotientClass 1 x)))
 116       (lambdaMultiply k x)
 117 
 118 /-- Detection respects the given E₂ product and actual composition product.
 119 The natural-number ranges and possible zero leading terms match the fixed source. -/
 120 public def MultiplicationCompatible
 121     {A : ClassicalAdamsFamily C} {F : SyntheticAdamsFamily Syn}
 122     {N : NuFunctorData C Syn} {aux : AuxiliaryData C}
 123     (K : E2Comparison A F N aux) (P : SphereE2Algebra A)
 124     {H : Syn} {unit : S_0_0 ⟶ H} {T : SyntheticTowerPresentation unit F}
 125     (cSphere : SyntheticTowerConvergence T (S_0_0 : Syn)) : Prop :=
 126   ∀ (s t s' t' k l : ℕ)
 127     (x : A.sphere.Page 2 (s,t)) (y : A.sphere.Page 2 (s',t'))
 128     (α : BiHom ((t : ℤ)-s) ((t : ℤ)-k) (S_0_0 : Syn))
 129     (β : BiHom ((t' : ℤ)-s') ((t' : ℤ)-l) (S_0_0 : Syn)),
 130     Detects cSphere (s,t,(t : ℤ)-k) (K.sphereE2 s t k x) α →
 131     Detects cSphere (s',t',(t' : ℤ)-l) (K.sphereE2 s' t' l y) β →
 132     Detects cSphere ((s : ℤ)+s', (t : ℤ)+t', (t : ℤ)+t'-(k+l : ℕ))
 133       (K.sphereE2 ((s : ℤ)+s') ((t : ℤ)+t') (k+l)
 134         (P.product (s,t) (s',t') x y))
 135       (homotopyRegrade (by dsimp; omega)
 136         (by dsimp; omega) (sphereProduct α β))

 138 /-- The sign uses topological degrees; no extra weight sign is inserted. -/
 139 public def SphereProductCommutative : Prop :=
 140   ∀ (m n k l : ℤ) (x : BiHom m n (S_0_0 : Syn)) (y : BiHom k l (S_0_0 : Syn)),
 141     sphereProduct x y = ((-1 : ℤ) ^ (m*k).natAbs) •
 142       homotopyRegrade (add_comm k m) (add_comm l n) (sphereProduct y x)
