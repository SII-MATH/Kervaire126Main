  20 /-- The given comparison evaluated at an actual weight. The nonnegative
  21 λ exponent is forced by `w ≤ t`; no new choice of standard class is made. -/
  22 public noncomputable def nuWeightLabel (D : Model C Syn) (X : ClassicalObject)
  23     (s t w : ℤ) (hw : w ≤ t)
  24     (x : (D.classical.obj (X.obj D.auxiliary)).Page 2 (s,t)) :
  25     (D.family.obj (D.nu.functor.obj (X.obj D.auxiliary))).E₂ (s,t,w) :=
  26   regradeE2 (by
  27     ext <;> simp [Int.toNat_of_nonneg (sub_nonneg.mpr hw)])
  28     (targetNuLabel D X s t (t-w).toNat x)
  29 
  30 /-- A coherent weight map on the actual ν(X) infinity pages. Compatibility
  31 with the specified E₂ labels identifies this map as the weight comparison,
  32 rather than an arbitrary map between unrelated page objects. -/
  33 public structure NuInfinityWeightMaps (D : Model C Syn) where
  34   map : ∀ (X : ClassicalObject) (s t w₁ w₂ : ℤ) (_hw : w₁ ≤ w₂) (_ht : w₂ ≤ t),
  35     ((D.family.obj (D.nu.functor.obj (X.obj D.auxiliary))).sequence.ssData (s,t,w₂)).eInfty ⟶
  36       ((D.family.obj (D.nu.functor.obj (X.obj D.auxiliary))).sequence.ssData (s,t,w₁)).eInfty
  37   map_self : ∀ (X : ClassicalObject) (s t w : ℤ) (hw : w ≤ t),
  38     map X s t w w le_rfl hw = 𝟙 _
  39   map_comp : ∀ (X : ClassicalObject) (s t w₁ w₂ w₃ : ℤ)
  40       (h₁₂ : w₁ ≤ w₂) (h₂₃ : w₂ ≤ w₃) (h₃ : w₃ ≤ t),
  41     map X s t w₂ w₃ h₂₃ h₃ ≫ map X s t w₁ w₂ h₁₂ (h₂₃.trans h₃) =
  42       map X s t w₁ w₃ (h₁₂.trans h₂₃) h₃
  43   naturality : ∀ (X Y : ClassicalObject) (f : X.obj D.auxiliary ⟶ Y.obj D.auxiliary)
  44       (s t w₁ w₂ : ℤ) (hw : w₁ ≤ w₂) (ht : w₂ ≤ t),
  45     (D.family.map (D.nu.functor.map f)).eInftyMap (s,t,w₂) ≫ map Y s t w₁ w₂ hw ht =
  46       map X s t w₁ w₂ hw ht ≫ (D.family.map (D.nu.functor.map f)).eInftyMap (s,t,w₁)
  47   representative : ∀ (X : ClassicalObject) (s t w₁ w₂ : ℤ)
  48       (hw : w₁ ≤ w₂) (ht : w₂ ≤ t)
  49       (x : (D.classical.obj (X.obj D.auxiliary)).Page 2 (s,t))
  50       (z : ((D.family.obj (D.nu.functor.obj (X.obj D.auxiliary))).sequence.ssData (s,t,w₂)).eInfty),
  51     HasInfinityRepresentative _ 2 (s,t,w₂) (nuWeightLabel D X s t w₂ ht x) z →
  52       HasInfinityRepresentative _ 2 (s,t,w₁) (nuWeightLabel D X s t w₁ (hw.trans ht) x)
  53         (map X s t w₁ w₂ hw ht z)
