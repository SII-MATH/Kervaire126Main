  23 /-- Selected E₂ labels on one pair of systems. The equations bind the
  24 chosen comparisons to the actual ν, unit isomorphism and λ maps. -/
  25 public structure E2Comparison
  26     (A : ClassicalAdamsFamily C) (F : SyntheticAdamsFamily Syn)
  27     (N : NuFunctorData C Syn) (aux : AuxiliaryData C) where
  28   nuE2 : ∀ (X : ClassicalObject) (a s t : ℤ) (k : ℕ),
  29     (A.obj (X.obj aux)).Page 2 (s,t) ≃ₗ[ℤ]
  30       (F.obj ((SyntheticCategory.biShift (0,a)).obj
  31         (N.functor.obj (X.obj aux)))).sequence.Page 2 (s,t,t+a-k)
  32   sphereE2 : ∀ (s t : ℤ) (k : ℕ),
  33     A.sphere.Page 2 (s,t) ≃ₗ[ℤ] F.sphere.sequence.Page 2 (s,t,t-k)
  34   sphere_nu : ∀ (s t : ℤ) (k : ℕ) (x : A.sphere.Page 2 (s,t)),
  35     (F.functor.map (SyntheticCategory.biShift_zero.hom.app _ ≫ N.unitIso.hom)).pageMap
  36       2 (s,t,t-k) (by simpa only [add_zero, ClassicalObject.obj, SyntheticAdamsFamily.obj] using nuE2 .sphere 0 s t k x) = sphereE2 s t k x
  37   nu_natural : ∀ (X Y : ClassicalObject) (f : X.obj aux ⟶ Y.obj aux)
  38       (a s t : ℤ) (k : ℕ) (x : (A.obj (X.obj aux)).Page 2 (s,t)),
  39     (F.functor.map ((SyntheticCategory.biShift (0,a)).map (N.functor.map f))).pageMap
  40       2 (s,t,t+a-k) (nuE2 X a s t k x) =
  41       nuE2 Y a s t k ((A.functor.map f).pageMap 2 (s,t) x)
  42   nu_lambda : ∀ (X : ClassicalObject) (s t : ℤ) (k : ℕ)
  43       (x : (A.obj (X.obj aux)).Page 2 (s,t)),
  44     HEq ((F.functor.map (SyntheticCategory.lam.app (N.functor.obj (X.obj aux)))).pageMap
  45       2 (s,t,t-1-k) (nuE2 X (-1) s t k x))
  46       (nuE2 X 0 s t (k+1) x)

  48 /-- Complete short exactness of the actual mod-two homology maps in every
  49 degree. The endpoint conditions are essential in Pstrągowski Lemma 4.23;
  50 middle exactness alone is insufficient for ν to preserve a cofiber triangle.
  51 Reviewed source: bbac1fc05843, StableHomotopy/Implementation/Data:474. -/
  52 public def HomologyShortExact (H : ModTwoCoefficients C)
  53     (T : CategoryTheory.Pretriangulated.Triangle C) : Prop :=
  54   ∀ n : ℤ,
  55     Function.Injective (inducedMap (MonoidalCategory.whiskerLeft H.object T.mor₁) n) ∧
  56     Function.Exact
  57       (inducedMap (MonoidalCategory.whiskerLeft H.object T.mor₁) n)
  58       (inducedMap (MonoidalCategory.whiskerLeft H.object T.mor₂) n) ∧
  59     Function.Surjective (inducedMap (MonoidalCategory.whiskerLeft H.object T.mor₂) n)
  60 
  61 /-- The same ν preserves this distinguished triangle under the complete
  62 homology short-exactness premise. This intentionally corrects the historical
  63 signature that omitted injectivity and surjectivity. The converse criterion
  64 is a separate literature result on the same coefficients and ν. -/
  65 public structure NuExactTriangleInput (H : ModTwoCoefficients C) (N : NuFunctorData C Syn) where
  66   preserves : ∀ (T : CategoryTheory.Pretriangulated.Triangle C),
  67     T ∈ distTriang C →
  68     HomologyShortExact H T →
  69     ∃ h : N.functor.obj T.obj₃ ⟶ (shiftFunctor Syn (1 : ℤ)).obj (N.functor.obj T.obj₁),
  70       CategoryTheory.Pretriangulated.Triangle.mk
  71         (N.functor.map T.mor₁) (N.functor.map T.mor₂) h ∈
  72           distTriang Syn
