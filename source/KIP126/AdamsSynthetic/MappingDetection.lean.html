   8 /-! Detection of maps belongs to their actual mapping spectrum. This corrects
   9 the old unqualified assertion about the target spectrum of an arbitrary map.
  10 Finite exact filtration uses the explicit separation obligation; an odd-torsion
  11 nonzero map is not silently treated as a nonzero E∞ class. -/

  24 /-- One common Z∞ representative with its nonzero E∞ image, and its image
  25 under the actual convergence identification. -/
  26 public structure NonzeroClassicalDetection {X : C} (c : ClassicalTowerConvergence P X)
  27     (p : ℤ × ℤ) (a : HomotopyGroup (p.2-p.1) X) where
  28   representative : (Subobject.underlying.obj (((A.obj X).ssData p).Z ⊤) : ModuleCat ℤ)
  29   nonzero : (((A.obj X).ssData p).pageπ ⊤) representative ≠ 0
  30   filteredClass : (Subobject.underlying.obj ((P.tower X).filtration.F p.1 (p.2-p.1)) : ModuleCat ℤ)
  31   homotopy_eq : ((P.tower X).filtration.F p.1 (p.2-p.1)).arrow filteredClass = a
  32   comparison_eq : (c.identification p).hom (((A.obj X).ssData p).pageπ ⊤ representative) =
  33     (P.tower X).filtration.toAssociatedGraded p.1 (p.2-p.1) filteredClass

  35 public def NonzeroClassicalDetection.pageElement {X : C} {c : ClassicalTowerConvergence P X}
  36     {p : ℤ × ℤ} {a : HomotopyGroup (p.2-p.1) X} (w : NonzeroClassicalDetection c p a) :
  37     (A.obj X).Page 2 p :=
  38   (Subobject.ofLE _ _ (((A.obj X).ssData p).Z_anti le_top) ≫
  39     ((A.obj X).ssData p).pageπ (↑(2-(A.obj X).r₀).toNat : WithTop ℕ)) w.representative

  41 public theorem NonzeroClassicalDetection.detects {X : C} {c : ClassicalTowerConvergence P X}
  42     {p : ℤ × ℤ} {a : HomotopyGroup (p.2-p.1) X} (w : NonzeroClassicalDetection c p a) :
  43     ClassicalDetects c p w.pageElement a :=
  44   ⟨w.representative, rfl, w.filteredClass, w.homotopy_eq, w.comparison_eq⟩

  46 /-- Exact actual filtration yields a nonzero E∞ representative under the
  47 given convergence. This is a consequence of the quotient and comparison,
  48 not an additional existence assumption. -/
  49 public theorem nonzeroClassicalDetection_of_exact {X : C}
  50     (c : ClassicalTowerConvergence P X) (p : ℤ × ℤ)
  51     (a : HomotopyGroup (p.2-p.1) X)
  52     (ha : a ∈ (P.tower X).filtrationSubmodule p.1 (p.2-p.1))
  53     (ha' : a ∉ (P.tower X).filtrationSubmodule (p.1+1) (p.2-p.1)) :
  54     Nonempty (NonzeroClassicalDetection c p a) := by
  55   have hmem : a ∈ ModuleCat.subobjectModule _
  56       ((P.tower X).filtration.F p.1 (p.2-p.1)) := by
  57     simpa only [AbstractAdamsTower.filtration, OrderIso.apply_symm_apply] using ha
  58   change ∃ b, ((P.tower X).filtration.F p.1 (p.2-p.1)).arrow.hom b = a at hmem
  59   obtain ⟨b,hb⟩ := hmem
  60   have hgraded : (P.tower X).filtration.toAssociatedGraded p.1 (p.2-p.1) b ≠ 0 := by
  61     intro hz
  62     have hnext := ((P.tower X).filtration.toAssociatedGraded_eq_zero_iff
  63       p.1 (p.2-p.1) b).mp hz
  64     apply ha'
  65     change ((P.tower X).filtration.F p.1 (p.2-p.1)).arrow.hom b ∈
  66       ModuleCat.subobjectModule _ ((P.tower X).filtration.F (p.1+1) (p.2-p.1)) at hnext
  67     rw [hb] at hnext
  68     simpa only [AbstractAdamsTower.filtration, OrderIso.apply_symm_apply] using hnext
  69   let e := (c.identification p).inv
  70     ((P.tower X).filtration.toAssociatedGraded p.1 (p.2-p.1) b)
  71   have he : (c.identification p).hom e =
  72       (P.tower X).filtration.toAssociatedGraded p.1 (p.2-p.1) b := by
  73     change ((c.identification p).inv ≫ (c.identification p).hom) _ = _
  74     rw [Iso.inv_hom_id]
  75     rfl
  76   have hene : e ≠ 0 := by
  77     intro hezero
  78     apply hgraded
  79     rw [← he, hezero, map_zero]
  80   haveI : Epi (((A.obj X).ssData p).pageπ ⊤) :=
  81     inferInstanceAs (Epi (cokernel.π _))
  82   obtain ⟨z,hz⟩ := (ModuleCat.epi_iff_surjective
  83     (((A.obj X).ssData p).pageπ ⊤)).mp inferInstance e
  84   exact ⟨⟨z, by simpa only [hz] using hene, b, hb, by rw [hz]; exact he⟩⟩

  86 /-- Actual stable maps are represented in the internal hom, with degree
  87 transport made explicit at the diagonal Adams bidegree. -/
  88 public def mappingClass (B : ClosedSymmetricTensorTriangulated C)
  89     {X Y : C} (f : X ⟶ Y) (s : ℤ) :
  90     HomotopyGroup (s-s) (mappingSpectrum B X Y) :=
  91   eqToHom (congrArg (Sphere (C := C)) (sub_self s)) ≫
  92     (mappingHomotopyZeroEquiv B X Y).symm f
  93 
  94 private theorem homotopy_filtration_transport {X : C}
  95     (n m s : ℤ) (h : n = m) (a : HomotopyGroup m X) :
  96     eqToHom (congrArg (Sphere (C := C)) h) ≫ a ∈
  97         (P.tower X).filtrationSubmodule s n ↔
  98       a ∈ (P.tower X).filtrationSubmodule s m := by
  99   subst m
 100   simp

 102 /-- The target-tower map filtration agrees with the actual internal-hom
 103 filtration, on finite spectra. This bridge must be supplied for the same P. -/
 104 public structure MappingFiltrationComparison (B : ClosedSymmetricTensorTriangulated C)
 105     (P : ClassicalTowerPresentation H.unit A) : Prop where
 106   agrees : ∀ {X Y : C}, IsFiniteSpectrum X → IsFiniteSpectrum Y →
 107     ∀ (f : X ⟶ Y) (s : ℤ), MapAdamsFiltration P f s ↔
 108       (mappingHomotopyZeroEquiv B X Y).symm f ∈
 109         (P.tower (mappingSpectrum B X Y)).filtrationSubmodule s 0

 111 /-- Consumer contract for detection, with all comparisons tied to the same
 112 coefficient tower, mapping spectra, finite scope and canonical convergence.
 113 It does not supply the separate filtration-separation proof. -/
 114 public structure MappingDetectionInput (B : ClosedSymmetricTensorTriangulated C)
 115     (P : ClassicalTowerPresentation H.unit A)
 116     (_S : MappingSpectrumSystem B P (IsFiniteSpectrum (C := C))) where
 117   factorizationComparison : FactorizationFiltrationComparison P
 118   mappingComparison : MappingFiltrationComparison B P
 119 
 120 /-- The nonzero detection witness is derived from exact filtration through
 121 the same mapping-spectrum comparison and canonical convergence. -/
 122 public theorem MappingDetectionInput.detects_exact
 123     {B : ClosedSymmetricTensorTriangulated C}
 124     {S : MappingSpectrumSystem B P (IsFiniteSpectrum (C := C))}
 125     (D : MappingDetectionInput B P S)
 126     {X Y : C} (hX : IsFiniteSpectrum X) (hY : IsFiniteSpectrum Y)
 127     (f : X ⟶ Y) (s : ℤ) (hs : ExactMapAdamsFiltration P f s) :
 128     Nonempty (NonzeroClassicalDetection (S.convergence X Y hX hY) (s,s) (mappingClass B f s)) := by
 129   apply nonzeroClassicalDetection_of_exact
 130   · have ha := (D.mappingComparison.agrees hX hY f s).mp hs.1
 131     exact (homotopy_filtration_transport (P := P) (s-s) 0 s (sub_self s) _).mpr ha
 132   · intro ha
 133     apply hs.2
 134     apply (D.mappingComparison.agrees hX hY f (s+1)).mpr
 135     exact (homotopy_filtration_transport (P := P) (s-s) 0 (s+1) (sub_self s) _).mp ha
 136 
 137 /-- The revised nonzero-map statement requires non-odd-torsion and finite
 138 objects; its nonzero E∞ class detects that very map in its mapping spectrum. -/
 139 public theorem MappingDetectionInput.detect_non_odd
 140     {B : ClosedSymmetricTensorTriangulated C}
 141     {S : MappingSpectrumSystem B P (IsFiniteSpectrum (C := C))}
 142     (D : MappingDetectionInput B P S) (separation : FiniteMapSeparation P)
 143     {X Y : C} (hX : IsFiniteSpectrum X) (hY : IsFiniteSpectrum Y)
 144     (f : X ⟶ Y) (hf : ¬ IsOddTorsionMap f) :
 145     ∃ s : ℤ, ExactMapAdamsFiltration P f s ∧
 146       Nonempty (NonzeroClassicalDetection (S.convergence X Y hX hY) (s,s) (mappingClass B f s)) := by
 147   obtain ⟨s,hs⟩ := finite_filtration_of_not_odd P separation hX hY f hf
 148   exact ⟨s,hs,D.detects_exact hX hY f s hs⟩
 149 
 150 end
