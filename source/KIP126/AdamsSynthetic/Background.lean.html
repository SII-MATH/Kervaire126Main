  19 /-- Closed symmetric tensor triangulated structure on this very background.
  20 The exactness statements use the supplied coherent shift comparisons, so a
  21 consumer need not independently assume that tensor or internal hom commutes
  22 with suspension. -/
  23 public structure ClosedSymmetricTensorTriangulated where
  24   symmetric : SymmetricCategory C
  25   closed : MonoidalClosed C
  26   tensorShift : ∀ W : C, (tensorRight W).CommShift ℤ
  27   tensorExact : ∀ W : C,
  28     letI := tensorShift W
  29     (tensorRight W).IsTriangulated
  30   ihomShift : letI := closed
  31     ∀ W : C, (ihom W).CommShift ℤ
  32   ihomExact : letI := closed
  33     ∀ W : C,
  34     letI := ihomShift W
  35     (ihom W).IsTriangulated
  36

  40 /-- Tensoring a shifted unit with an object is its actual integer shift.
  41 This comparison is derived from the supplied tensor shift coherence. -/
  42 public noncomputable def sphereTensorShift
  43     (B : ClosedSymmetricTensorTriangulated C) (n : ℤ) (X : C) :
  44     (Sphere n : C) ⊗ X ≅ (shiftFunctor C n).obj X := by
  45   letI := B.tensorShift X
  46   exact ((tensorRight X).commShiftIso n).app SphereSpectrum ≪≫
  47     (shiftFunctor C n).mapIso (λ_ X)
  48 
  49 /-- The sphere product comparison determined by this background. -/
  50 public noncomputable def smashSpheres
  51     (B : ClosedSymmetricTensorTriangulated C) (m n : ℤ) :
  52     (Sphere m : C) ⊗ Sphere n ≅ Sphere (m + n) :=
  53   B.sphereTensorShift m (Sphere n) ≪≫
  54     ((shiftFunctorAdd C n m).app SphereSpectrum).symm ≪≫
  55     eqToIso (congrArg (Sphere (C := C)) (add_comm n m))
  56 
  57 end ClosedSymmetricTensorTriangulated

  59 /-- The stable Koszul sign for the actual braiding and the sphere product
  60 comparisons supplied by `B`. It is a separate proof input, since symmetry
  61 alone does not imply the stable sign convention. -/
  62 public structure SphereTwistSign (B : ClosedSymmetricTensorTriangulated C) : Prop where
  63   twist : letI := B.symmetric
  64     ∀ r s : ℤ,
  65       (β_ (Sphere r : C) (Sphere s)).hom ≫ (B.smashSpheres s r).hom ≫
  66         eqToHom (congrArg (Sphere (C := C)) (add_comm s r)) =
  67       if Even (r * s) then (B.smashSpheres r s).hom
  68       else -(B.smashSpheres r s).hom
