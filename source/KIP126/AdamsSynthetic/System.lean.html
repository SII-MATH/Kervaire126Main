  24 /-- The synthetic coefficient unit is the image of this very HF₂ unit. -/

  70 /-- Compatibility on the selected object closure. No convergence on every
  71 synthetic object is requested, and every induced map is the actual map. -/
  72 public structure ComparisonCompatible (D : ModelData C Syn) : Prop where
  73   convergence_natural : ∀ (X Y : SyntheticObject)
  74       (f : X.obj D.nu D.auxiliary ⟶ Y.obj D.nu D.auxiliary),
  75     Nonempty (ConvergentMap D.family f (D.convergence X).toConvergence
  76       (D.convergence Y).toConvergence)
  77   first_quotient : ∀ (X : ClassicalObject) (a s t : ℤ)
  78       (x : (D.classical.obj (X.obj D.auxiliary)).Page 2 (s,t)),
  79     KIP126.AdamsSynthetic.Detects
  80       (D.convergence (.quotient 1 (.shift (0,a) (.nu X)))) (s,t,t+a)
  81       (D.family.pageMap
  82         (HasFunctorialCofiber.cofibι (lambdaPow 1
  83           ((SyntheticCategory.biShift (0,a)).obj (D.nu.functor.obj (X.obj D.auxiliary)))))
  84         2 (s,t,t+a) (by simpa only [Nat.cast_zero, sub_zero] using D.comparison.nuE2 X a s t 0 x))
  85       ((D.firstQuotient X a s t).symm x)

 101 /-- Naturality supplies one witness tied to this context and this map. -/
 102 public noncomputable def convergentMap (X Y : SyntheticObject)
 103     (f : X.obj D.nu D.auxiliary ⟶ Y.obj D.nu D.auxiliary) :
 104     ConvergentMap D.family f (D.convergence X).toConvergence
 105       (D.convergence Y).toConvergence :=
 106   Classical.choice (D.comparisonCompatible.convergence_natural X Y f)
