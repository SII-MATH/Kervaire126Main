  29 /-- Adams parity is the parity of the stem `t-s`. -/
  30 public def adamsStemParity : GradingParity where
  31   toFun p := ((p.2 - p.1 : ℤ) : ZMod 2)
  32   map_zero' := by simp
  33   map_add' p q := by
  34     simp only [Prod.fst_add, Prod.snd_add, Int.cast_sub, Int.cast_add]

  37 /-- All coherence laws on composition. The page, E∞, and filtered-target
  38 products are already related by `ConvergingSSPairing`; units are obtained from
  39 one common infinite-cycle representative and the actual identity map. -/
  40 public structure MappingComposition (S : MappingSpectrumSystem B P scope) where
  41   pairing : ∀ X Y Z : MappingObject scope,
  42     ConvergingSSPairing (S.hom X Y).underlying (S.hom Y Z).underlying (S.hom X Z).underlying
  43   parity : ∀ X Y Z, (pairing X Y Z).ssPairing.parity = adamsStemParity
  44   abutment_compatible : ∀ (X Y Z : MappingObject scope) (a b : ℤ)
  45       (x : (S.hom X Y).underlying.A a) (y : (S.hom Y Z).underlying.A b),
  46     mappingHomotopyEquiv B (a + b) X.val Z.val ((pairing X Y Z).abutmentPair a b x y) =
  47       (shiftFunctorAdd C a b).hom.app X.val ≫
  48         (shiftFunctor C b).map (mappingHomotopyEquiv B a X.val Y.val x) ≫
  49         mappingHomotopyEquiv B b Y.val Z.val y
  50   page_associative : ∀ (W X Y Z : MappingObject scope) (r : ℤ) (i j k : ℤ × ℤ)
  51       (x : (S.hom W X).underlying.E.Page r i)
  52       (y : (S.hom X Y).underlying.E.Page r j)
  53       (z : (S.hom Y Z).underlying.E.Page r k),
  54     HEq ((pairing W Y Z).ssPairing.pair r (i + j) k
  55       ((pairing W X Y).ssPairing.pair r i j x y) z)
  56       ((pairing W X Z).ssPairing.pair r i (j + k) x
  57         ((pairing X Y Z).ssPairing.pair r j k y z))
  58   identity : ∀ X : MappingObject scope, ConvergingSSUnit (S.hom X X).underlying rfl
  59   identity_actual : ∀ X : MappingObject scope,
  60     mappingHomotopyZeroEquiv B X.val X.val (identity X).abutment = 𝟙 X.val
  61   page_left_unital : ∀ (X Y : MappingObject scope) (r : ℤ) (k : ℤ × ℤ)
  62       (x : (S.hom X Y).underlying.E.Page r k),
  63     HEq ((pairing X X Y).ssPairing.pair r (0,0) k ((identity X).page r) x) x
  64   page_right_unital : ∀ (X Y : MappingObject scope) (r : ℤ) (k : ℤ × ℤ)
  65       (x : (S.hom X Y).underlying.E.Page r k),
  66     HEq ((pairing X Y Y).ssPairing.pair r k (0,0) x ((identity Y).page r)) x
  67 private theorem infinity_heq_of_page_heq
  68     (E : KIP126.Core.SpectralSequence (ModuleCat.{v} ℤ) (ℤ × ℤ))
  69     (p q : ℤ × ℤ) (hpq : p = q) (r : ℤ)
  70     (x : InfiniteCycles E p) (y : InfiniteCycles E q)
  71     (h : HEq (infinityToPage E r p x) (infinityToPage E r q y)) :
  72     HEq ((E.ssData p).pageπ ⊤ x) ((E.ssData q).pageπ ⊤ y) := by
  73   subst q
  74   exact heq_of_eq ((E.ssData p).infinity_projection_eq_of_page_projection_eq
  75     ↑(r-E.r₀).toNat x y (eq_of_heq h))

  77 private theorem MappingComposition.page_pair_infinity
  78     {S : MappingSpectrumSystem B P scope} (M : MappingComposition S)
  79     (X Y Z : MappingObject scope) (i j : ℤ × ℤ)
  80     (x : InfiniteCycles (S.hom X Y).underlying.E i)
  81     (y : InfiniteCycles (S.hom Y Z).underlying.E j) :
  82     (M.pairing X Y Z).ssPairing.pair 2 i j
  83       (infinityToPage (S.hom X Y).underlying.E 2 i x)
  84       (infinityToPage (S.hom Y Z).underlying.E 2 j y) =
  85     infinityToPage (S.hom X Z).underlying.E 2 (i+j)
  86       ((M.pairing X Y Z).cyclePairAtInfinity i j x y) :=
  87   (M.pairing X Y Z).finitePagePair_compatible 2 i j
  88     (le_of_eq (A.firstPage _)) (le_of_eq (A.firstPage _))
  89     (le_of_eq (A.firstPage _)) x y

  91 /-- The E∞ associativity law follows from page associativity on common
  92 infinite cycles and equality of their infinite quotient classes. -/
  93 public theorem MappingComposition.eInfty_associative
  94     {S : MappingSpectrumSystem B P scope} (M : MappingComposition S)
  95     (W X Y Z : MappingObject scope) (i j k : ℤ × ℤ)
  96     (x : ((S.hom W X).underlying.E.ssData i).eInfty)
  97     (y : ((S.hom X Y).underlying.E.ssData j).eInfty)
  98     (z : ((S.hom Y Z).underlying.E.ssData k).eInfty) :
  99     HEq ((M.pairing W Y Z).eInftyPair (i+j) k
 100       ((M.pairing W X Y).eInftyPair i j x y) z)
 101       ((M.pairing W X Z).eInftyPair i (j+k) x
 102         ((M.pairing X Y Z).eInftyPair j k y z)) := by
 103   haveI : Epi (((S.hom W X).underlying.E.ssData i).pageπ ⊤) :=
 104     inferInstanceAs (Epi (cokernel.π _))
 105   haveI : Epi (((S.hom X Y).underlying.E.ssData j).pageπ ⊤) :=
 106     inferInstanceAs (Epi (cokernel.π _))
 107   haveI : Epi (((S.hom Y Z).underlying.E.ssData k).pageπ ⊤) :=
 108     inferInstanceAs (Epi (cokernel.π _))
 109   obtain ⟨a,rfl⟩ := (ModuleCat.epi_iff_surjective
 110     (((S.hom W X).underlying.E.ssData i).pageπ ⊤)).mp inferInstance x
 111   obtain ⟨b,rfl⟩ := (ModuleCat.epi_iff_surjective
 112     (((S.hom X Y).underlying.E.ssData j).pageπ ⊤)).mp inferInstance y
 113   obtain ⟨c,rfl⟩ := (ModuleCat.epi_iff_surjective
 114     (((S.hom Y Z).underlying.E.ssData k).pageπ ⊤)).mp inferInstance z
 115   have h := M.page_associative W X Y Z 2 i j k
 116     (infinityToPage _ 2 i a) (infinityToPage _ 2 j b) (infinityToPage _ 2 k c)
 117   simp only [MappingComposition.page_pair_infinity] at h
 118   have hi := infinity_heq_of_page_heq (S.hom W Z).underlying.E
 119     ((i+j)+k) (i+(j+k)) (add_assoc i j k) 2 _ _ h
 120   simpa only [ConvergingSSPairing.eInftyPair_induced] using hi

 122 /-- The E∞ left unit is inherited from the page unit on the same common
 123 infinite-cycle representative. -/
 124 public theorem MappingComposition.eInfty_left_unital
 125     {S : MappingSpectrumSystem B P scope} (M : MappingComposition S)
 126     (X Y : MappingObject scope) (k : ℤ × ℤ)
 127     (x : ((S.hom X Y).underlying.E.ssData k).eInfty) :
 128     HEq ((M.pairing X X Y).eInftyPair (0,0) k (M.identity X).eInfty x) x := by
 129   haveI : Epi (((S.hom X Y).underlying.E.ssData k).pageπ ⊤) :=
 130     inferInstanceAs (Epi (cokernel.π _))
 131   obtain ⟨a,rfl⟩ := (ModuleCat.epi_iff_surjective
 132     (((S.hom X Y).underlying.E.ssData k).pageπ ⊤)).mp inferInstance x
 133   have h := M.page_left_unital X Y 2 k (infinityToPage _ 2 k a)
 134   simp only [ConvergingSSUnit.page, MappingComposition.page_pair_infinity] at h
 135   have hi := infinity_heq_of_page_heq (S.hom X Y).underlying.E
 136     ((0,0)+k) k (zero_add k) 2 _ _ h
 137   simpa only [ConvergingSSPairing.eInftyPair_induced, ConvergingSSUnit.eInfty] using hi

 143     (x : ((S.hom X Y).underlying.E.ssData k).eInfty) :
 144     HEq ((M.pairing X Y Y).eInftyPair k (0,0) x (M.identity Y).eInfty) x := by

 163 private theorem equivHEq {α β : ℤ → Type v} (e : ∀ n, α n ≃ β n)
 164     {m n : ℤ} (h : m = n) {x : α m} {y : α n}
 165     (hh : HEq (e m x) (e n y)) : HEq x y := by
 166   cases h
 167   exact heq_of_eq ((e m).injective (eq_of_heq hh))

 169 omit [HasFunctorialCofiber (C := C)] in
 170 private theorem shiftedLeftUnit {X Y : C} (n : ℤ) (f : (shiftFunctor C n).obj X ⟶ Y) :
 171     HEq ((shiftFunctorAdd C 0 n).hom.app X ≫
 172       (shiftFunctor C n).map ((shiftFunctorZero C ℤ).hom.app X) ≫ f) f := by
 173   apply homHEq (congrArg (fun i => (shiftFunctor C i).obj X) (zero_add n))
 174   rw [shiftFunctorAdd_zero_add_hom_app]
 175   simp only [Category.assoc, ← Functor.map_comp_assoc, Iso.inv_hom_id_app,
 176     Functor.id_obj]
 177   rw [(shiftFunctor C n).map_id X, Category.id_comp]

 179 omit [HasFunctorialCofiber (C := C)] in
 180 private theorem shiftedRightUnit {X Y : C} (n : ℤ) (f : (shiftFunctor C n).obj X ⟶ Y) :
 181     HEq ((shiftFunctorAdd C n 0).hom.app X ≫
 182       (shiftFunctor C (0:ℤ)).map f ≫ (shiftFunctorZero C ℤ).hom.app Y) f := by
 183   apply homHEq (congrArg (fun i => (shiftFunctor C i).obj X) (add_zero n))
 184   rw [shiftFunctorAdd_add_zero_hom_app]
 185   simp only [Category.assoc, (shiftFunctorZero C ℤ).hom.naturality,
 186     Iso.inv_hom_id_app_assoc, Functor.id_map]

 188 omit [HasFunctorialCofiber (C := C)] in
 189 private theorem shiftedAssoc {W X Y Z : C} (i j k : ℤ)
 190     (f : (shiftFunctor C i).obj W ⟶ X) (g : (shiftFunctor C j).obj X ⟶ Y)
 191     (h : (shiftFunctor C k).obj Y ⟶ Z) :
 192     HEq ((shiftFunctorAdd C (i+j) k).hom.app W ≫
 193       (shiftFunctor C k).map ((shiftFunctorAdd C i j).hom.app W ≫
 194         (shiftFunctor C j).map f ≫ g) ≫ h)
 195       ((shiftFunctorAdd C i (j+k)).hom.app W ≫
 196         (shiftFunctor C (j+k)).map f ≫
 197           (shiftFunctorAdd C j k).hom.app X ≫ (shiftFunctor C k).map g ≫ h) := by
 198   apply homHEq (congrArg (fun a => (shiftFunctor C a).obj W) (add_assoc i j k))
 199   simp only [Functor.map_comp, Category.assoc]
 200   rw [← Category.assoc ((shiftFunctorAdd C (i+j) k).hom.app W)
 201     ((shiftFunctor C k).map ((shiftFunctorAdd C i j).hom.app W))]
 202   rw [shiftFunctorAdd_assoc_hom_app]
 203   simp only [shiftFunctorAdd', Iso.trans_hom, NatTrans.comp_app, eqToIso.hom,
 204     eqToHom_app, Category.assoc]
 205   rw [(shiftFunctorAdd C j k).hom.naturality_assoc]
 206   rfl

 210 private theorem identityShift (X : MappingObject scope) :
 211     mappingHomotopyEquiv B 0 X.val X.val (M.identity X).abutment =
 212       (shiftFunctorZero C ℤ).hom.app X.val := by
 213   apply (Iso.homCongr ((shiftFunctorZero C ℤ).app X.val) (Iso.refl X.val)).injective
 214   simpa [mappingHomotopyZeroEquiv, Iso.homCongr] using M.identity_actual X

 216 /-- The left abutment unit is the actual identity map under the same closed comparison. -/
 217 public theorem abutment_left_unital (X Y : MappingObject scope) (n : ℤ)
 218     (x : (S.hom X Y).underlying.A n) :
 219     HEq ((M.pairing X X Y).abutmentPair 0 n (M.identity X).abutment x) x := by
 220   apply equivHEq (fun n => mappingHomotopyEquiv B n X.val Y.val) (zero_add n)
 221   rw [M.abutment_compatible, identityShift M]
 222   exact shiftedLeftUnit n _

 224 /-- The right abutment unit follows from the actual identity and shift coherence. -/
 225 public theorem abutment_right_unital (X Y : MappingObject scope) (n : ℤ)
 226     (x : (S.hom X Y).underlying.A n) :
 227     HEq ((M.pairing X Y Y).abutmentPair n 0 x (M.identity Y).abutment) x := by
 228   apply equivHEq (fun n => mappingHomotopyEquiv B n X.val Y.val) (add_zero n)
 229   rw [M.abutment_compatible, identityShift M]
 230   exact shiftedRightUnit n _

 232 /-- Abutment associativity follows from actual shifted-map composition and shift coherence. -/
 233 public theorem abutment_associative (W X Y Z : MappingObject scope) (i j k : ℤ)
 234     (x : (S.hom W X).underlying.A i) (y : (S.hom X Y).underlying.A j)
 235     (z : (S.hom Y Z).underlying.A k) :
 236     HEq ((M.pairing W Y Z).abutmentPair (i+j) k ((M.pairing W X Y).abutmentPair i j x y) z)
 237       ((M.pairing W X Z).abutmentPair i (j+k) x ((M.pairing X Y Z).abutmentPair j k y z)) := by
 238   apply equivHEq (fun n => mappingHomotopyEquiv B n W.val Z.val) (add_assoc i j k)
 239   rw [M.abutment_compatible, M.abutment_compatible, M.abutment_compatible, M.abutment_compatible]
 240   exact shiftedAssoc i j k _ _ _

 249 /-- The composition multiplication on sphere mappings agrees with the
 250 previously supplied sphere E₂ algebra, through the canonical mapping-sphere
 251 comparison. It is not a second freely chosen sphere multiplication. -/
 252 public structure MappingSphereCompatibility {S : MappingSpectrumSystem B P scope}
 253     (M : MappingComposition S) (E₂ : SphereE2Algebra A) (hSphere : scope SphereSpectrum) : Prop where
 254   product : ∀ (p q : ℤ × ℤ)
 255       (x : (S.hom ⟨SphereSpectrum,hSphere⟩ ⟨SphereSpectrum,hSphere⟩).underlying.E.Page 2 p)
 256       (y : (S.hom ⟨SphereSpectrum,hSphere⟩ ⟨SphereSpectrum,hSphere⟩).underlying.E.Page 2 q),
 257     (sphereMappingAdamsIso B A).hom.pageMap 2 (p + q)
 258       ((M.pairing ⟨SphereSpectrum,hSphere⟩ ⟨SphereSpectrum,hSphere⟩
 259         ⟨SphereSpectrum,hSphere⟩).ssPairing.pair 2 p q x y) =
 260       E₂.product p q ((sphereMappingAdamsIso B A).hom.pageMap 2 p x)
 261         ((sphereMappingAdamsIso B A).hom.pageMap 2 q y)
 262   unit : (sphereMappingAdamsIso B A).hom.pageMap 2 (0,0)
 263       ((M.identity ⟨SphereSpectrum,hSphere⟩).page 2) = E₂.unit
 264 end
