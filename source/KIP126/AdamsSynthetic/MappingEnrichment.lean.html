   8 /-! Explicit enrichment on the full converging-sequence category. Its Hom
   9 objects and composition agree with the actual mapping systems. Unit data
  10 are bound on finite pages, E∞ and the actual abutment as well. This repairs
  11 the older convergence-only ambient category, which forgot all finite pages.
  12 Source: KIPBase `multiplicativeSS/AdamsEnriched.lean`, baseline 5aa845241533. -/
  13 namespace KIP126.AdamsSynthetic
  14 open CategoryTheory CategoryTheory.MonoidalCategory
  15 open KIP126.StableHomotopy KIP126.Classical.Adams KIP126.Core.SpectralSequence
  16 universe u v
  17 noncomputable section
  18 variable {C : Type u} [StableHomotopyCategory.{u,v} C]
  19   [HasFunctorialCofiber (C := C)] {H : C} {unit : SphereSpectrum ⟶ H}
  20   {A : ClassicalAdamsFamily C} {B : ClosedSymmetricTensorTriangulated C}
  21   {P : ClassicalTowerPresentation unit A} {scope : C → Prop}

  23 /-- A supplied enrichment, rather than a globally postulated one. Every Hom
  24 is this system's actual converging mapping sequence; its full morphisms see
  25 all pages and the same E∞ comparison. -/
  26 public structure MappingEnrichment {S : MappingSpectrumSystem B P scope}
  27     (M : MappingComposition S) where
  28   monoidal : MonoidalCategory (ConvergingSequence.{0,v} ℤ)
  29   enriched : @EnrichedCategory (ConvergingSequence.{0,v} ℤ) _ monoidal (MappingObject scope)
  30   hom_eq : ∀ X Y : MappingObject scope, enriched.Hom X Y = S.hom X Y
  31   /-- Elementary tensor products for the composable mapping systems. These
  32   are requested only for these systems, whose differential degrees agree;
  33   arbitrary ambient converging sequences need not have compatible degrees. -/
  34   tensorPairing : letI := monoidal
  35     ∀ X Y Z : MappingObject scope,
  36       ConvergingSSPairing (S.hom X Y).underlying (S.hom Y Z).underlying
  37         ((S.hom X Y ⊗ S.hom Y Z).underlying)
  38   /-- Actual enriched composition takes elementary page tensors to the
  39   specified page products. No arbitrary equivalence of pairing records is
  40   used to give composition its mathematical meaning. -/
  41   composition_page : letI := monoidal
  42     ∀ (X Y Z : MappingObject scope) (r : ℤ) (i j : ℤ × ℤ)
  43       (x : (S.hom X Y).underlying.E.Page r i)
  44       (y : (S.hom Y Z).underlying.E.Page r j),
  45       let f : (S.hom X Y ⊗ S.hom Y Z) ⟶ S.hom X Z :=
  46         eqToHom (congrArg₂ (fun A B : ConvergingSequence.{0,v} ℤ => A ⊗ B)
  47           (hom_eq X Y).symm (hom_eq Y Z).symm) ≫ enriched.comp X Y Z ≫
  48             eqToHom (hom_eq X Z)
  49       f.sequence.pageMap r (i+j) ((tensorPairing X Y Z).ssPairing.pair r i j x y) =
  50         (M.pairing X Y Z).ssPairing.pair r i j x y
  51   composition_eInfty : letI := monoidal
  52     ∀ (X Y Z : MappingObject scope) (i j : ℤ × ℤ)
  53       (x : ((S.hom X Y).underlying.E.ssData i).eInfty)
  54       (y : ((S.hom Y Z).underlying.E.ssData j).eInfty),
  55       let f : (S.hom X Y ⊗ S.hom Y Z) ⟶ S.hom X Z :=
  56         eqToHom (congrArg₂ (fun A B : ConvergingSequence.{0,v} ℤ => A ⊗ B)
  57           (hom_eq X Y).symm (hom_eq Y Z).symm) ≫ enriched.comp X Y Z ≫
  58             eqToHom (hom_eq X Z)
  59       f.sequence.eInftyMap (i+j) ((tensorPairing X Y Z).eInftyPair i j x y) =
  60         (M.pairing X Y Z).eInftyPair i j x y
  61   composition_abutment : letI := monoidal
  62     ∀ (X Y Z : MappingObject scope) (i j : ℤ)
  63       (x : (S.hom X Y).underlying.A i) (y : (S.hom Y Z).underlying.A j),
  64       let f : (S.hom X Y ⊗ S.hom Y Z) ⟶ S.hom X Z :=
  65         eqToHom (congrArg₂ (fun A B : ConvergingSequence.{0,v} ℤ => A ⊗ B)
  66           (hom_eq X Y).symm (hom_eq Y Z).symm) ≫ enriched.comp X Y Z ≫
  67             eqToHom (hom_eq X Z)
  68       f.convergence.aMap (i+j) ((tensorPairing X Y Z).abutmentPair i j x y) =
  69         (M.pairing X Y Z).abutmentPair i j x y
  70   unit_reindex : monoidal.tensorUnit.underlying.conv.reindex (0,0) = (0,0)
  71   unit_data : ConvergingSSUnit monoidal.tensorUnit.underlying unit_reindex
  72   identity_page : letI := monoidal
  73     ∀ (X : MappingObject scope) (r : ℤ),
  74       let f : 𝟙_ (ConvergingSequence.{0,v} ℤ) ⟶ S.hom X X :=
  75         enriched.id X ≫ eqToHom (hom_eq X X)
  76       f.sequence.pageMap r (0,0) (unit_data.page r) = (M.identity X).page r
  77   identity_eInfty : letI := monoidal
  78     ∀ X : MappingObject scope,
  79       let f : 𝟙_ (ConvergingSequence.{0,v} ℤ) ⟶ S.hom X X :=
  80         enriched.id X ≫ eqToHom (hom_eq X X)
  81       f.sequence.eInftyMap (0,0) unit_data.eInfty = (M.identity X).eInfty
  82   identity_abutment : letI := monoidal
  83     ∀ X : MappingObject scope,
  84       let f : 𝟙_ (ConvergingSequence.{0,v} ℤ) ⟶ S.hom X X :=
  85         enriched.id X ≫ eqToHom (hom_eq X X)
  86       f.convergence.aMap 0 unit_data.abutment = (M.identity X).abutment
  87 end
  88 end KIP126.AdamsSynthetic
