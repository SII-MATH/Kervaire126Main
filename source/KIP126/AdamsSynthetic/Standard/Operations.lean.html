 109 private theorem infinityToPageAt_naturality
 110     {D D' : (ℤ × ℤ) → SSData (ModuleCat.{0} ℤ)}
 111     (g : SSDataMorphism (ℤ × ℤ) D D') (p : ℤ × ℤ)
 112     (n m : WithTop ℕ) (hn : n = m)
 113     (z : (Subobject.underlying.obj ((D p).Z ⊤) : ModuleCat ℤ)) :
 114     (g.pageMap p n ≫ eqToHom (congrArg (D' p).page hn))
 115       ((Subobject.ofLE _ _ ((D p).Z_anti (le_top : n ≤ ⊤)) ≫ (D p).pageπ n) z) =
 116       (Subobject.ofLE _ _ ((D' p).Z_anti (le_top : m ≤ ⊤)) ≫ (D' p).pageπ m)
 117         (g.cycleMap p ⊤ z) := by
 118   subst m
 119   simp only [eqToHom_refl, Category.comp_id]
 120   have h : (Subobject.ofLE _ _ ((D p).Z_anti (le_top : n ≤ ⊤)) ≫
 121       (D p).pageπ n) ≫ g.pageMap p n =
 122       g.cycleMap p ⊤ ≫ Subobject.ofLE _ _ ((D' p).Z_anti (le_top : n ≤ ⊤)) ≫
 123         (D' p).pageπ n := by
 124     rw [Category.assoc, g.pageπ_pageMap, ← Category.assoc,
 125       g.cycleMap_ofLE p (le_top : n ≤ ⊤)]
 126     exact Category.assoc _ _ _
 127   exact congrArg (fun m => m z) h

 138 public theorem mappingEnriched_composition_eInfty : letI := mappingEnriched_monoidal
 139     ∀ (X Y Z : MappingObject (IsFiniteSpectrum (C := Spectrum))) (i j : ℤ × ℤ)
 140       (x : ((mappingSystem.hom X Y).underlying.E.ssData i).eInfty)
 141       (y : ((mappingSystem.hom Y Z).underlying.E.ssData j).eInfty),
 142       let f : (mappingSystem.hom X Y ⊗ mappingSystem.hom Y Z) ⟶ mappingSystem.hom X Z :=
 143         eqToHom (congrArg₂ (fun A B : ConvergingSequence.{0,0} ℤ => A ⊗ B)
 144           (mappingEnriched_hom_eq X Y).symm (mappingEnriched_hom_eq Y Z).symm) ≫ mappingEnriched_enriched.comp X Y Z ≫
 145             eqToHom (mappingEnriched_hom_eq X Z)
 146       f.sequence.eInftyMap (i+j) ((mappingEnriched_tensorPairing X Y Z).eInftyPair i j x y) =
 147         (mappingComposition.pairing X Y Z).eInftyPair i j x y := by
 148   letI := mappingEnriched_monoidal
 149   intro X Y Z i j x y
 150   let f : (mappingSystem.hom X Y ⊗ mappingSystem.hom Y Z) ⟶ mappingSystem.hom X Z :=
 151     eqToHom (congrArg₂ (fun A B : ConvergingSequence.{0,0} ℤ => A ⊗ B)
 152       (mappingEnriched_hom_eq X Y).symm (mappingEnriched_hom_eq Y Z).symm) ≫
 153         mappingEnriched_enriched.comp X Y Z ≫ eqToHom (mappingEnriched_hom_eq X Z)
 154   let P := mappingEnriched_tensorPairing X Y Z
 155   let Q := mappingComposition.pairing X Y Z
 156   haveI : Epi (((mappingSystem.hom X Y).underlying.E.ssData i).pageπ ⊤) :=
 157     inferInstanceAs (Epi (cokernel.π _))
 158   haveI : Epi (((mappingSystem.hom Y Z).underlying.E.ssData j).pageπ ⊤) :=
 159     inferInstanceAs (Epi (cokernel.π _))
 160   obtain ⟨a, rfl⟩ := (ModuleCat.epi_iff_surjective
 161     (((mappingSystem.hom X Y).underlying.E.ssData i).pageπ ⊤)).mp inferInstance x
 162   obtain ⟨b, rfl⟩ := (ModuleCat.epi_iff_surjective
 163     (((mappingSystem.hom Y Z).underlying.E.ssData j).pageπ ⊤)).mp inferInstance y
 164   have h₀ (U V : MappingObject (IsFiniteSpectrum (C := Spectrum))) :
 165       (mappingSystem.hom U V).underlying.E.r₀ ≤ 2 :=
 166     le_of_eq (classical.firstPage _)
 167   have hT : (mappingSystem.hom X Y ⊗ mappingSystem.hom Y Z).underlying.E.r₀ ≤ 2 :=
 168     f.sequence.r₀_eq.le.trans (h₀ X Z)
 169   have h := mappingEnriched_composition_page X Y Z 2 i j
 170     (infinityToPage _ 2 i a) (infinityToPage _ 2 j b)
 171   change f.sequence.pageMap 2 (i+j)
 172       (P.ssPairing.pair 2 i j (infinityToPage _ 2 i a) (infinityToPage _ 2 j b)) =
 173     Q.ssPairing.pair 2 i j (infinityToPage _ 2 i a) (infinityToPage _ 2 j b) at h
 174   rw [P.finitePagePair_compatible 2 i j (h₀ X Y) (h₀ Y Z) hT,
 175     infinityToPage_naturality,
 176     Q.finitePagePair_compatible 2 i j (h₀ X Y) (h₀ Y Z) (h₀ X Z)] at h
 177   have hinf := ((mappingSystem.hom X Z).underlying.E.ssData (i+j)).infinity_projection_eq_of_page_projection_eq
 178     ↑(2 - (mappingSystem.hom X Z).underlying.E.r₀).toNat
 179     (f.sequence.toSSDataMorphism.cycleMap (i+j) ⊤ (P.cyclePairAtInfinity i j a b))
 180     (Q.cyclePairAtInfinity i j a b) h
 181   have htop := congrArg (fun m => m (P.cyclePairAtInfinity i j a b))
 182     (f.sequence.toSSDataMorphism.pageπ_pageMap (i+j) ⊤)
 183   change f.sequence.eInftyMap (i+j)
 184       (P.eInftyPair i j (((mappingSystem.hom X Y).underlying.E.ssData i).pageπ ⊤ a)
 185         (((mappingSystem.hom Y Z).underlying.E.ssData j).pageπ ⊤ b)) =
 186     Q.eInftyPair i j (((mappingSystem.hom X Y).underlying.E.ssData i).pageπ ⊤ a)
 187       (((mappingSystem.hom Y Z).underlying.E.ssData j).pageπ ⊤ b)
 188   rw [← P.eInftyPair_induced, ← Q.eInftyPair_induced]
 189   exact htop.trans hinf

 291     TotalWeightHom X Y →ₗ[ℤ] TotalWeightHom X Y :=

 297 private theorem totalLambda_of (X Y : SyntheticSpectrum) (w : ℤ) (f : WeightHom X Y w) :
 298     totalLambda X Y (weightHomInclusion w f) =
 299       weightHomInclusion (w-1) (weightHomLambda f) := by
 300   change DirectSum.toAddMonoid
 301     (fun v : ℤ => (DirectSum.of (WeightHom X Y) (v-1)).comp
 302       (Preadditive.leftComp Y
 303         (weightHomLambda (X := X) (𝟙 ((SyntheticCategory.biShift (0,v)).obj X)))))
 304     (DirectSum.of (WeightHom X Y) w f) =
 305       DirectSum.of (WeightHom X Y) (w-1) (weightHomLambda f)
 306   rw [DirectSum.toAddMonoid_of]
 307   change DirectSum.of (WeightHom X Y) (w-1)
 308       ((weightHomLambda (X := X) (𝟙 ((SyntheticCategory.biShift (0,w)).obj X))) ≫ f) =
 309     DirectSum.of (WeightHom X Y) (w-1) (weightHomLambda f)
 310   apply congrArg (DirectSum.of (WeightHom X Y) (w-1))
 311   dsimp only [weightHomLambda, Functor.id_obj, Functor.comp_obj]
 312   rw [Category.comp_id, Category.assoc]

 314 /-- The polynomial action obtained by evaluating X at actual graded λ. -/
 315 @[implicit_reducible] public def graded_module : ∀ X Y : SyntheticSpectrum, Module (Polynomial ℤ) (TotalWeightHom X Y) := by
 316   intro X Y
 317   exact
 318     (fun (M : Type) [AddCommGroup M] (D : M →+ M) =>
 319       let A := Module.End ℤ M

 321       let ring : NonAssocRing A := inferInstance

 324       let hom : Polynomial ℤ →+* A :=
 325         Polynomial.eval₂RingHom' (Int.castRingHom A) L
 326           (fun a => @Int.cast_commute A ring a L)
 327       Module.compHom M hom)

 330         (weightHomInclusion (w-1)).comp
 331           (Preadditive.leftComp Y

 356 private def totalComp (X Y Z : SyntheticSpectrum) :
 357     TotalWeightHom Y Z →+ (TotalWeightHom X Y →+ TotalWeightHom X Z) := by
 358   letI : ∀ p : ℤ × ℤ,
 359       (SyntheticCategory.biShift p : SyntheticSpectrum ⥤ SyntheticSpectrum).Additive :=
 360     exactness.shiftAdditive
 361   exact DirectSum.toAddMonoid fun b =>
 362     { toFun := fun g => DirectSum.toAddMonoid fun a =>
 363         { toFun := fun f => weightHomInclusion (a+b) (weightHomComp f g)
 364           map_zero' := by
 365             dsimp only [weightHomComp]
 366             rw [Functor.map_zero, Limits.zero_comp, Limits.comp_zero, map_zero]
 367           map_add' := by
 368             intro f f'
 369             dsimp only [weightHomComp]
 370             rw [Functor.map_add, Preadditive.add_comp, Preadditive.comp_add, map_add] }
 371       map_zero' := by
 372         apply DirectSum.addHom_ext
 373         intro a f
 374         change (DirectSum.toAddMonoid (γ := TotalWeightHom X Z) _)
 375           (DirectSum.of (WeightHom X Y) a f) = 0
 376         rw [DirectSum.toAddMonoid_of]
 377         change weightHomInclusion (a+b) (weightHomComp f (0 : WeightHom Y Z b)) = 0
 378         dsimp only [weightHomComp]
 379         rw [Limits.comp_zero, Limits.comp_zero, map_zero]
 380       map_add' := by
 381         intro g g'
 382         apply DirectSum.addHom_ext
 383         intro a f
 384         rw [AddMonoidHom.add_apply]
 385         change (DirectSum.toAddMonoid (γ := TotalWeightHom X Z) _)
 386           (DirectSum.of (WeightHom X Y) a f) =
 387           (DirectSum.toAddMonoid (γ := TotalWeightHom X Z) _)
 388             (DirectSum.of (WeightHom X Y) a f) +
 389           (DirectSum.toAddMonoid (γ := TotalWeightHom X Z) _)
 390             (DirectSum.of (WeightHom X Y) a f)
 391         rw [DirectSum.toAddMonoid_of, DirectSum.toAddMonoid_of, DirectSum.toAddMonoid_of]
 392         change weightHomInclusion (a+b) (weightHomComp f (g+g')) =
 393           weightHomInclusion (a+b) (weightHomComp f g) +
 394           weightHomInclusion (a+b) (weightHomComp f g')
 395         dsimp only [weightHomComp]
 396         rw [Preadditive.comp_add, Preadditive.comp_add, map_add] }

 410 private theorem totalComp_assoc (W X Y Z : SyntheticSpectrum)
 411     (f : TotalWeightHom W X) (g : TotalWeightHom X Y) (h : TotalWeightHom Y Z) :
 412     totalComp W Y Z h (totalComp W X Y g f) =
 413       totalComp W X Z (totalComp X Y Z h g) f := by
 414   induction f using DirectSum.induction_on with
 415   | zero => simp only [map_zero]
 416   | add f f' hf hf' => simp only [map_add, hf, hf']
 417   | of a f =>
 418     induction g using DirectSum.induction_on with
 419     | zero => simp only [map_zero, AddMonoidHom.zero_apply]
 420     | add g g' hg hg' => simp only [map_add, AddMonoidHom.add_apply, hg, hg']
 421     | of b g =>
 422       induction h using DirectSum.induction_on with
 423       | zero => simp only [map_zero, AddMonoidHom.zero_apply]
 424       | add h h' hh hh' => simp only [map_add, AddMonoidHom.add_apply, hh, hh']
 425       | of c h =>
 426         change totalComp W Y Z (weightHomInclusion c h)
 427           (totalComp W X Y (weightHomInclusion b g) (weightHomInclusion a f)) =
 428           totalComp W X Z (totalComp X Y Z (weightHomInclusion c h) (weightHomInclusion b g))
 429             (weightHomInclusion a f)
 430         rw [totalComp_of, totalComp_of, totalComp_of, totalComp_of]
 431         exact homogeneous_assoc quotients.coherence a b c f g h

 488 private theorem additive_polynomial_smul {M N : Type*} [AddCommGroup M] [AddCommGroup N]
 489     [Module (Polynomial ℤ) M] [Module (Polynomial ℤ) N]
 490     (f : M →+ N)
 491     (hX : ∀ m, f ((Polynomial.X : Polynomial ℤ) • m) = (Polynomial.X : Polynomial ℤ) • f m)
 492     (p : Polynomial ℤ) (m : M) : f (p • m) = p • f m := by
 493   induction p using Polynomial.induction_on generalizing m with
 494   | C a =>
 495     have ha : (Polynomial.C a : Polynomial ℤ) = (a : Polynomial ℤ) := Polynomial.C_eq_intCast a
 496     rw [ha]
 497     exact map_intCast_smul f (Polynomial ℤ) (Polynomial ℤ) a m
 498   | add p q hp hq => simp only [add_smul, map_add, hp, hq]
 499   | monomial n a ih =>
 500     rw [pow_succ, ← mul_assoc]
 501     rw [mul_smul, ih, hX]
 502     simp only [mul_smul]

 522 /-- Actual shifted composition, extended bilinearly over the polynomial action. -/
 523 @[no_expose] public def graded_compose : ∀ X Y Z : SyntheticSpectrum,
 524     letI := graded_module X Y
 525     letI := graded_module Y Z
 526     letI := graded_module X Z
 527     TotalWeightHom Y Z →ₗ[Polynomial ℤ]
 528       TotalWeightHom X Y →ₗ[Polynomial ℤ] TotalWeightHom X Z := by
 529   intro X Y Z
 530   letI := graded_module X Y
 531   letI := graded_module Y Z
 532   letI := graded_module X Z
 533   let C : TotalWeightHom Y Z →+ (TotalWeightHom X Y →+ TotalWeightHom X Z) :=
 534     totalComp X Y Z
 535   let F : TotalWeightHom Y Z →+ (TotalWeightHom X Y →ₗ[Polynomial ℤ] TotalWeightHom X Z) :=
 536     { toFun := fun g =>
 537         { toFun := C g
 538           map_add' := (C g).map_add
 539           map_smul' := by
 540             intro p f
 541             apply additive_polynomial_smul (C g) (fun f => ?_) p f
 542             change totalComp X Y Z g ((Polynomial.X : Polynomial ℤ) • f) =
 543               (Polynomial.X : Polynomial ℤ) • totalComp X Y Z g f
 544             rw [graded_X_smul, graded_X_smul]
 545             exact totalComp_lambda_right X Y Z f g }
 546       map_zero' := by
 547         apply LinearMap.ext
 548         intro f
 549         exact congrArg (fun h : TotalWeightHom X Y →+ TotalWeightHom X Z => h f) (map_zero (totalComp X Y Z))
 550       map_add' := by
 551         intro g g'
 552         apply LinearMap.ext
 553         intro f
 554         exact congrArg (fun h : TotalWeightHom X Y →+ TotalWeightHom X Z => h f) ((totalComp X Y Z).map_add g g') }
 555   exact
 556     { toFun := F
 557       map_add' := F.map_add
 558       map_smul' := by
 559         intro p g
 560         apply additive_polynomial_smul F (fun g => ?_) p g
 561         apply LinearMap.ext
 562         intro f
 563         change totalComp X Y Z ((Polynomial.X : Polynomial ℤ) • g) f =
 564           (Polynomial.X : Polynomial ℤ) • totalComp X Y Z g f
 565         rw [graded_X_smul, graded_X_smul]
 566         exact totalComp_lambda_left X Y Z f g }
