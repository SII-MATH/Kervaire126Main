  20 /-- The first-quotient label commutes with the positive weight shift and
  21 every quotient comparison satisfying the actual inclusion square. -/
  22 public theorem firstQuotient_positive_shift
  23     (X : ClassicalObject) (s t : ℤ)
  24     (x : (model.classical.obj (X.obj model.auxiliary)).Page 2 (s,t))
  25     (ε : XModLambdaN ((SyntheticCategory.biShift (0,1)).obj
  26         (model.nu.functor.obj (X.obj model.auxiliary))) 1 ≅
  27       (SyntheticCategory.biShift (0,1)).obj
  28         (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) 1))
  29     (hε : XModLambdaN.incl _ 1 ≫ ε.hom =
  30       (SyntheticCategory.biShift (0,1)).map (XModLambdaN.incl _ 1)) :
  31     homotopyRegrade (by simp) (by simp)
  32       (shiftHomotopyClass (0,1) (t-s,t)
  33         (homotopyRegrade rfl (by simp) ((model.firstQuotient X 0 s t).symm x) ≫
  34           XModLambdaN.map (SyntheticCategory.biShift_zero.hom.app _) 1)) ≫ ε.inv =
  35         (model.firstQuotient X 1 s t).symm x := by
  36   sorry

  38 /-- The first-quotient labels of a class and its actual classical suspension
  39 agree after the fixed ν suspension and quotient comparisons. -/
  40 public theorem firstQuotient_suspension
  41     (X : ClassicalObject) (s t : ℤ)
  42     (x : (model.classical.obj (X.obj model.auxiliary)).Page 2 (s,t))
  43     (sx : (model.classical.obj ((ClassicalObject.shift 1 X).obj model.auxiliary)).Page 2 (s,t+1))
  44     (ε : XModLambdaN ((SyntheticCategory.biShift (1,1)).obj
  45         (model.nu.functor.obj (X.obj model.auxiliary))) 1 ≅
  46       (SyntheticCategory.biShift (1,1)).obj
  47         (XModLambdaN (model.nu.functor.obj (X.obj model.auxiliary)) 1))
  48     (hε : XModLambdaN.incl _ 1 ≫ ε.hom =
  49       (SyntheticCategory.biShift (1,1)).map (XModLambdaN.incl _ 1))
  50     (hsx : (model.classicalSuspension X).DesuspendsClass s (t+1) sx (by simpa using x)) :
  51     (homotopyRegrade (w' := t+1) rfl (by simp)
  52         ((model.firstQuotient (.shift 1 X) 0 s (t+1)).symm sx) ≫
  53       XModLambdaN.map (SyntheticCategory.biShift_zero.hom.app _) 1) ≫
  54       XModLambdaN.map (model.nu.suspensionIso (X.obj model.auxiliary)).hom 1 ≫ ε.hom =
  55     homotopyRegrade (by simp; omega) (by simp)
  56       (shiftHomotopyClass (1,1) (t-s,t)
  57         (homotopyRegrade rfl (by simp) ((model.firstQuotient X 0 s t).symm x) ≫
  58           XModLambdaN.map (SyntheticCategory.biShift_zero.hom.app _) 1)) := by
  59   sorry
