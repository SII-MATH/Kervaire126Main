  16 private theorem page_subsingleton_of_ambient
  17     (D : KIP126.Core.SpectralSequence.SSData (ModuleCat.{0} ℤ))
  18     (r : WithTop ℕ) [Subsingleton D.V] : Subsingleton (D.page r) := by
  19   have hV : IsZero D.V := ModuleCat.isZero_of_subsingleton D.V
  20   have hZ : IsZero (Subobject.underlying.obj (D.Z r)) :=
  21     IsZero.of_mono (D.Z r).arrow hV
  22   letI : Epi (D.pageπ r) := coequalizer.π_epi
  23   exact ModuleCat.subsingleton_of_isZero (IsZero.of_epi (D.pageπ r) hZ)

  25 public theorem classical_page_negative (X : ClassicalObject) (r s t : ℤ)
  26     (hs : s < 0) :
  27     Subsingleton ((model.classical.obj (X.obj model.auxiliary)).Page r (s,t)) := by
  28   let T := model.classicalTower.tower (X.obj model.auxiliary)
  29   have hi (a b : ℕ) (hab : a ≤ b) (h : a = b) : IsIso (T.map hab) := by
  30     subst b
  31     rw [T.map_self]
  32     infer_instance
  33   let f := T.mapAt (show s ≤ s+1 by omega)
  34   haveI : IsIso f := hi s.toNat (s+1).toNat (by omega) (by omega)
  35   have hLayer : IsZero (T.layer s) :=
  36     Pretriangulated.Triangle.isZero₃_of_isIso₁
  37       (Pretriangulated.Triangle.mk f (HasFunctorialCofiber.cofibι f)
  38         (HasFunctorialCofiber.cofibδ f))
  39       (HasFunctorialCofiber.cofib_distinguished f) (show IsIso f from inferInstance)
  40   letI : Subsingleton (T.FirstPage (s,t)) := ⟨fun a b => hLayer.eq_of_tgt a b⟩
  41   let D := (model.classical.obj (X.obj model.auxiliary)).ssData (s,t)
  42   letI : Subsingleton D.V :=
  43     (model.classicalTower.sequence (X.obj model.auxiliary)).ambient (s,t)
  44       |>.toLinearEquiv.injective.subsingleton
  45   exact page_subsingleton_of_ambient D _

  47 public theorem sphere_page_filtration_zero (r : ℤ) (t : ℕ) (ht : t ≠ 0) :
  48     Subsingleton (model.classical.sphere.Page r (0,t)) := by
  49   let T := model.classicalTower.tower (SphereSpectrum (C := Spectrum))
  50   let e : T.layer 0 ≅ model.coefficients.object :=
  51     T.layerIso 0 ≪≫ (tensorLeft model.coefficients.object).mapIso T.zeroIso ≪≫
  52       ρ_ model.coefficients.object
  53   letI := model.coefficients.homotopy_vanishes (t : ℤ) (by exact_mod_cast ht)
  54   letI : Subsingleton (T.FirstPage (0,t)) := by
  55     change Subsingleton (HomotopyGroup (t : ℤ) (T.layer 0))
  56     exact (Iso.homCongr (Iso.refl (Sphere (C := Spectrum) (t : ℤ))) e).injective.subsingleton
  57   let D := model.classical.sphere.ssData (0,t)
  58   letI : Subsingleton D.V :=
  59     (model.classicalTower.sequence (SphereSpectrum (C := Spectrum))).ambient (0,t)
  60       |>.toLinearEquiv.injective.subsingleton
  61   exact page_subsingleton_of_ambient D _

  63 /-- Infinite HF₂ Adams filtration on the ordinary sphere implies odd torsion.
  64 The source is the actual finite shifted sphere and the target is the ordinary
  65 sphere. This follows from the finite-map separation producer obligation;
  66 it does not assert that these ordinary sphere classes vanish. -/
  67 public theorem classical_sphere_separated (m : ℤ)
  68     (a : HomotopyGroup (C := Spectrum) m SphereSpectrum)
  69     (ha : ∀ s : ℕ, a ∈ (model.classicalTower.tower SphereSpectrum).filtrationSubmodule s m) :
  70     KIP126.AdamsSynthetic.IsOddTorsionMap a := by
  71   apply finite_map_separation
  72     (IsFiniteSpectrum.shift m IsFiniteSpectrum.sphere) IsFiniteSpectrum.sphere a
  73   intro s
  74   exact ha s.toNat
