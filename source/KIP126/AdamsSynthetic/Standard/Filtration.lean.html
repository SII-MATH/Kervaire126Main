  64 public def homologyShift (X : Spectrum) :
  65     coefficients.object ⊗ X⟦(1 : ℤ)⟧ ≅ (coefficients.object ⊗ X)⟦(1 : ℤ)⟧ := by
  66   letI := closedBackground.symmetric
  67   letI := closedBackground.tensorShift coefficients.object
  68   exact (β_ coefficients.object (X⟦(1 : ℤ)⟧)) ≪≫
  69     ((tensorRight coefficients.object).commShiftIso (1 : ℤ)).app X ≪≫
  70     (shiftFunctor Spectrum (1 : ℤ)).mapIso (β_ coefficients.object X).symm
  71 
  72 private theorem desuspend_injective (X : Spectrum) (i : ℤ) :
  73     Function.Injective (AbstractAdamsTower.desuspend X i) := by
  74   let e := (shiftFunctorAdd' Spectrum i (-1) (i - 1) (by omega)).app SphereSpectrum
  75   let d := (shiftFunctorCompIsoId Spectrum 1 (-1) (by omega)).app X
  76   intro a b h
  77   change e.hom ≫ (shiftFunctor Spectrum (-1 : ℤ)).map a ≫ d.hom =
  78     e.hom ≫ (shiftFunctor Spectrum (-1 : ℤ)).map b ≫ d.hom at h
  79   apply (shiftFunctor Spectrum (-1 : ℤ)).map_injective
  80   apply (cancel_epi e.hom).mp
  81   apply (cancel_mono d.hom).mp
  82   simpa only [Category.assoc] using h

  84 private theorem homology_triangle_distinguished (T : HoCofiberSequence (C := Spectrum)) :
  85     Pretriangulated.Triangle.mk (coefficients.object ◁ T.f) (coefficients.object ◁ T.g)
  86       (coefficients.object ◁ T.h ≫ (homologyShift T.X).hom) ∈ distTriang Spectrum := by
  87   letI := closedBackground.symmetric
  88   letI := closedBackground.tensorShift coefficients.object
  89   letI := closedBackground.tensorExact coefficients.object
  90   let F := tensorRight coefficients.object
  91   refine Pretriangulated.isomorphic_distinguished _
  92     (F.map_distinguished _ T.distinguished) _ ?_
  93   refine Pretriangulated.Triangle.isoMk _ _ (β_ coefficients.object T.X)
  94     (β_ coefficients.object T.Y) (β_ coefficients.object T.Z) ?_ ?_ ?_
  95   · exact BraidedCategory.braiding_naturality_right _ T.f
  96   · exact BraidedCategory.braiding_naturality_right _ T.g
  97   · change (coefficients.object ◁ T.h ≫
  98         ((β_ coefficients.object (T.X⟦(1 : ℤ)⟧)).hom ≫
  99           ((tensorRight coefficients.object).commShiftIso (1 : ℤ)).hom.app T.X ≫
 100           (shiftFunctor Spectrum (1 : ℤ)).map (β_ coefficients.object T.X).inv)) ≫
 101         (shiftFunctor Spectrum (1 : ℤ)).map (β_ coefficients.object T.X).hom =
 102       (β_ coefficients.object T.Z).hom ≫
 103         (T.h ▷ coefficients.object ≫
 104           ((tensorRight coefficients.object).commShiftIso (1 : ℤ)).hom.app T.X)
 105     simp only [Category.assoc, ← Functor.map_comp, Iso.inv_hom_id,
 106       BraidedCategory.braiding_naturality_right_assoc]
 107     rw [(shiftFunctor Spectrum (1 : ℤ)).map_id ((tensorRight coefficients.object).obj T.X)]
 108     exact congrArg (fun k => (β_ coefficients.object T.Z).hom ≫
 109       T.h ▷ coefficients.object ≫ k)
 110       (Category.comp_id (((tensorRight coefficients.object).commShiftIso (1 : ℤ)).hom.app T.X))

 112 public theorem homology_exact (T : HoCofiberSequence (C := Spectrum)) (i : ℤ) :
 113     Function.Exact (modTwoHomologyMap coefficients T.g i)
 114       (modTwoConnecting coefficients homologyShift T i) := by
 115   let U := Pretriangulated.Triangle.mk (coefficients.object ◁ T.f)
 116     (coefficients.object ◁ T.g) (coefficients.object ◁ T.h ≫ (homologyShift T.X).hom)
 117   have hU : U ∈ distTriang Spectrum := homology_triangle_distinguished T
 118   intro z
 119   change AbstractAdamsTower.desuspend (coefficients.object ⊗ T.X) i
 120     (z ≫ U.mor₃) = 0 ↔ ∃ y, y ≫ U.mor₂ = z
 121   have hzero : AbstractAdamsTower.desuspend (coefficients.object ⊗ T.X) i
 122       (z ≫ U.mor₃) = 0 ↔ z ≫ U.mor₃ = 0 := by
 123     constructor
 124     · intro hz
 125       apply desuspend_injective (coefficients.object ⊗ T.X) i
 126       simpa only [map_zero] using hz
 127     · intro hz
 128       rw [hz, map_zero]
 129   rw [hzero]
 130   constructor
 131   · intro hz
 132     obtain ⟨y, hy⟩ := Pretriangulated.Triangle.coyoneda_exact₃ U hU z hz
 133     exact ⟨y, hy.symm⟩
 134   · rintro ⟨y, rfl⟩
 135     rw [Category.assoc, Pretriangulated.comp_distTriang_mor_zero₂₃ U hU, comp_zero]
 136 
 137 private def homologyExactness : ModTwoHomologyExactness coefficients :=
 138   ⟨homologyShift, homology_exact⟩
