 150 concatenation product in normalized cobar cohomology. -/

 152     (x : classical.sphere.Page 2 ((s : ℤ), (t : ℤ)))
 153     (y : classical.sphere.Page 2 ((s' : ℤ), (t' : ℤ))) :

 167 /-- The actual Main square is the cup square of the specified ξ₁⁶⁴ cocycle. -/
 168 public theorem standardH6Square_milnor :
 169     sphereE2_milnorEquiv 2 128 standardH6Square = milnorHiSquareClass 6 := by
 170   have h := sphereProduct_milnor 1 64 1 64 standardH6 standardH6
 171   simp only [standardH6Square_eq_product] at ⊢
 172   change sphereE2_milnorEquiv 2 128
 173     (sphereAlgebra.product (1,64) (1,64) standardH6 standardH6) = _
 174   change sphereE2_milnorEquiv 2 128
 175     (sphereAlgebra.product (1,64) (1,64) standardH6 standardH6) = _ at h
 176   rw [h]
 177   have hh : sphereE2_milnorEquiv 1 64 standardH6 = milnorHiClass 6 :=
 178     standardHi_milnor 6
 179   rw [hh]
 180   exact (Milnor.hiSquare_eq_cup 6).symm
