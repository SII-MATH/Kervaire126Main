  95 /-- An ambient element in every finite cycle module belongs to the specified limit subobject. -/
  96 private theorem mem_top_cycles (D : SSData (ModuleCat.{v} ℤ)) (a : D.V)
  97     (ha : ∀ n : ℕ, a ∈ ModuleCat.subobjectModule _ (D.Z n)) :
  98     a ∈ ModuleCat.subobjectModule _ (D.Z ⊤) := by
  99   have hspan : (ModuleCat.subobjectModule _).symm (Submodule.span ℤ {a}) ≤ D.Z ⊤ := by
 100     apply D.Z_top_greatest
 101     intro n
 102     apply (ModuleCat.subobjectModule _).le_iff_le.mp
 103     simp only [OrderIso.apply_symm_apply, Submodule.span_le, Set.singleton_subset_iff]
 104     exact ha n
 105   have h := (ModuleCat.subobjectModule _).monotone hspan
 106   apply h
 107   rw [OrderIso.apply_symm_apply]
 108   exact Submodule.subset_span (by simp)

 110 private theorem layerMap_mem_cycles {C : Type u} [StableHomotopyCategory.{u,v} C]
 111     [HasFunctorialCofiber (C := C)] {H : C} {unit : SphereSpectrum ⟶ H}
 112     {X : C} (T : AbstractAdamsTower unit X) (p : ℤ × ℤ)
 113     (a : HomotopyGroup (p.2-p.1) (T.stageAt p.1))
 114     (r : ℕ) (hr : 1 ≤ r) : T.layerMap p a ∈ T.cycles r hr p := by
 115   change T.connecting p (T.layerMap p a) ∈ LinearMap.range _
 116   have hz : T.connecting p (T.layerMap p a) = 0 := by
 117     change AbstractAdamsTower.desuspend _ _
 118       ((a ≫ HasFunctorialCofiber.cofibι _) ≫ HasFunctorialCofiber.cofibδ _) = 0
 119     have he : HasFunctorialCofiber.cofibι (T.mapAt (show p.1 ≤ p.1+1 by omega)) ≫
 120         HasFunctorialCofiber.cofibδ (T.mapAt (show p.1 ≤ p.1+1 by omega)) = 0 :=
 121       comp_distTriang_mor_zero₂₃ _ (HasFunctorialCofiber.cofib_distinguished _)
 122     rw [Category.assoc, he, comp_zero, map_zero]
 123   rw [hz]
 124   exact Submodule.zero_mem _

 126 /-- Actual tower lifts supply common infinite representatives without a separation premise. -/
 127 private theorem representative_of_lift {C : Type u} [StableHomotopyCategory.{u,v} C]
 128     [HasFunctorialCofiber (C := C)] {H : C} {unit : SphereSpectrum ⟶ H}
 129     {X : C} {T : AbstractAdamsTower unit X}
 130     {D : (ℤ × ℤ) → SSData (ModuleCat.{v} ℤ)}
 131     {d : ∀ (r : ℤ) (p : ℤ × ℤ), (D p).page (↑(r - 2).toNat : WithTop ℕ) ⟶
 132       (D (p + (r, r - 1))).page (↑(r - 2).toNat : WithTop ℕ)}
 133     (P : TowerSequencePresentation unit T D d) (p : ℤ × ℤ)
 134     (a : HomotopyGroup (p.2-p.1) (T.stageAt p.1)) :
 135     ∃ z : (Subobject.underlying.obj ((D p).Z ⊤) : ModuleCat ℤ),
 136       P.infiniteRepresentativeE1 p z = T.layerMap p a := by
 137   let a₂ : T.cycles 2 (by decide) p := ⟨T.layerMap p a, layerMap_mem_cycles T p a 2 (by decide)⟩
 138   let v : (D p).V := (P.ambient p).inv a₂
 139   have hv : ∀ n : ℕ, v ∈ ModuleCat.subobjectModule _ ((D p).Z n) := by
 140     intro n
 141     apply (P.cycles p n v).mpr
 142     have he : (P.ambient p).hom v = a₂ := by simp only [v, Iso.inv_hom_id_apply]
 143     rw [he]
 144     exact layerMap_mem_cycles T p a (n+2) (by omega)
 145   obtain ⟨z, hz⟩ := mem_top_cycles (D p) v hv
 146   refine ⟨z, ?_⟩
 147   change ((P.ambient p).hom (((D p).Z ⊤).arrow z)).val = T.layerMap p a
 148   rw [hz]
 149   simp only [v, Iso.inv_hom_id_apply, a₂]

 281 /-- Canonical identifications are natural on actual lifts; the graded quotient is epi. -/
 282 private theorem canonical_convergence_natural {Syn : Type u} [SyntheticCategory.{u,v} Syn]
 283     [HasFunctorialCofiber (C := Syn)] {H : Syn} {unit : S_0_0 ⟶ H}
 284     {F : SyntheticAdamsFamily Syn} (P : SyntheticTowerPresentation unit F)
 285     {X Y : Syn} (f : X ⟶ Y)
 286     (cx : SyntheticTowerConvergence P X) (cy : SyntheticTowerConvergence P Y) :
 287     Nonempty (ConvergentMap F f cx.toConvergence cy.toConvergence) := by
 288   let hpres : ∀ (s : ℤ) (p : ℤ × ℤ),
 289       ∃ φ : (Subobject.underlying.obj ((P.filtration X).F s p) : ModuleCat ℤ) ⟶
 290         (Subobject.underlying.obj ((P.filtration Y).F s p) : ModuleCat ℤ),
 291         φ ≫ ((P.filtration Y).F s p).arrow =
 292           ((P.filtration X).F s p).arrow ≫ syntheticHomotopyMap f p := by
 293     intro s p

 295         syntheticHomotopyMap f p (((P.filtration X).F s p).arrow b) ∈
 296           ModuleCat.subobjectModule _ ((P.filtration Y).F s p) := by
 297       intro b

 299       simp only [SyntheticTowerPresentation.filtration, OrderIso.apply_symm_apply]
 300       change (((P.filtration X).F s p).arrow b ≫ f) ∈ P.filtrationSubmodule Y s p
 301       exact hb
 302     exact ⟨restrict_subobject _ _ _ h, restrict_subobject_arrow _ _ _ h⟩
 303   refine ⟨{
 304     comparison := {
 305       eMap := (F.map f).eInftyMap
 306       aMap := syntheticHomotopyMap f
 307       filtration_compat := hpres
 308       reindex_eq := rfl
 309       iso_compat := ?_ }
 310     abutment_eq := rfl
 311     infinity_eq := rfl }⟩
 312   intro i
 313   dsimp only [SyntheticTowerConvergence.toConvergence]
 314   simp only [Filtration.transportGraded_self, Category.comp_id]
 315   apply (cancel_epi (cx.identification i).inv).mp
 316   letI : Epi ((P.filtration X).toAssociatedGraded i.1 (i.2.1-i.1,i.2.2)) := by
 317     change Epi (cokernel.π _)
 318     infer_instance
 319   apply (cancel_epi ((P.filtration X).toAssociatedGraded i.1 (i.2.1-i.1,i.2.2))).mp
 320   ext b
 321   obtain ⟨a, ha⟩ := (cx.canonical.1 i.1 (i.2.1-i.1) i.2.2
 322     (((P.filtration X).F i.1 (i.2.1-i.1,i.2.2)).arrow b)).mp
 323       (filtration_arrow_mem P X i.1 (i.2.1-i.1,i.2.2) b)
 324   obtain ⟨z, hz⟩ := representative_of_lift (P.sequence X i.2.2) (i.1,i.2.1) a
 325   have hx := cx.canonical.2 i a z hz b ha.symm
 326   have hxi : (cx.identification i).inv
 327       ((P.filtration X).toAssociatedGraded i.1 (i.2.1-i.1,i.2.2) b) =
 328       ((F.obj X).sequence.ssData i).pageπ ⊤ z := by
 329     rw [← hx, Iso.hom_inv_id_apply]
 330   let a' := a ≫ (P.map ((SyntheticCategory.biShift (0,-i.2.2)).map f)).stage i.1.toNat
 331   obtain ⟨z', hz'⟩ := representative_of_lift (P.sequence Y i.2.2) (i.1,i.2.1) a'
 332   have hm := mapped_representative_eq P f i a z hz z' hz'

 335       ((P.filtration X).F i.1 (i.2.1-i.1,i.2.2)).arrow b ≫ f :=
 336     ConcreteCategory.congr_hom (hpres i.1 (i.2.1-i.1,i.2.2)).choose_spec b
 337   have ha' : weightHomotopyMap (i.2.1-i.1) i.2.2 Y
 338       (((P.filtration Y).F i.1 (i.2.1-i.1,i.2.2)).arrow b') =
 339       a' ≫ (P.tower ((SyntheticCategory.biShift (0,-i.2.2)).obj Y)).projection i.1 := by
 340     rw [hb', weightMap_natural, ← ha]
 341     change (a ≫ (P.tower ((SyntheticCategory.biShift (0,-i.2.2)).obj X)).projection i.1) ≫
 342       (SyntheticCategory.biShift (0,-i.2.2)).map f =
 343       (a ≫ (P.map ((SyntheticCategory.biShift (0,-i.2.2)).map f)).stage i.1.toNat) ≫
 344         (P.tower ((SyntheticCategory.biShift (0,-i.2.2)).obj Y)).projection i.1
 345     calc
 346       _ = a ≫ ((P.tower ((SyntheticCategory.biShift (0,-i.2.2)).obj X)).projection i.1 ≫
