  26 /-- The canonical map of actual homotopy classes under the given bigraded shift. -/
  27 public def shiftHomotopyClass (δ p : ℤ × ℤ) {X : Syn} (a : BiHom p.1 p.2 X) :
  28     BiHom (p+δ).1 (p+δ).2 ((SyntheticCategory.biShift δ).obj X) :=
  29   (SyntheticCategory.biShift_comp p δ).inv.app S_0_0 ≫
  30     (SyntheticCategory.biShift δ).map a
  31 
  32 /-- Convergence at the stem-indexed infinity page, with only arithmetic reindexing. -/
  33 public def towerGradedIso {H : Syn} {unit : S_0_0 ⟶ H}
  34     {F : SyntheticAdamsFamily Syn} {P : SyntheticTowerPresentation unit F} {X : Syn}
  35     (c : SyntheticTowerConvergence P X) (p : ℤ × ℤ) (s : ℤ) :
  36     ((F.obj X).sequence.ssData (s,p.1+s,p.2)).eInfty ≅
  37       c.filtration.associatedGraded s p :=
  38   c.identification (s,p.1+s,p.2) ≪≫ eqToIso (by simp)
  39 
  40 /-- The infinity-page identification is determined by the same convergence
  41 and filtered-piece isomorphism; it is not another freely chosen comparison. -/
  42 public def towerInfinityIso {H : Syn} {unit : S_0_0 ⟶ H}
  43     {F : SyntheticAdamsFamily Syn} {P : SyntheticTowerPresentation unit F} {X Y : Syn}
  44     (cx : SyntheticTowerConvergence P X) (cy : SyntheticTowerConvergence P Y)
  45     {p q : ℤ × ℤ} (e : FilteredPieceIso cx.filtration cy.filtration p q) (s : ℤ) :
  46     ((F.obj X).sequence.ssData (s,p.1+s,p.2)).eInfty ≅
  47       ((F.obj Y).sequence.ssData (s,q.1+s,q.2)).eInfty :=
  48   towerGradedIso cx p s ≪≫ e.gradedIso s ≪≫ (towerGradedIso cy q s).symm

  50 /-- A uniform shift comparison on the selected closure. Page representatives
  51 are transported for every valid finite page, including representatives of
  52 infinite-page classes. Actual homotopy maps determine the ambient transport. -/
  53 public structure SyntheticESSShiftComparison (D : Model C Syn) (δ : ℤ × ℤ) where
  54   filtered : ∀ (X : SyntheticObject) (p : ℤ × ℤ),
  55     FilteredPieceIso (D.convergence X).filtration
  56       (D.convergence (.shift δ X)).filtration p (p+δ)
  57   ambient_eq : ∀ (X : SyntheticObject) (p : ℤ × ℤ)
  58       (a : BiHom p.1 p.2 (X.obj D.nu D.auxiliary)),
  59     (filtered X p).ambient.hom a = shiftHomotopyClass δ p a
  60   page : ∀ (X : SyntheticObject) (r : ℤ) (_hr : 2 ≤ r) (p : ℤ × ℤ) (s : ℤ),
  61     (D.family.obj (X.obj D.nu D.auxiliary)).Page r (s,p.1+s,p.2) ≅
  62       (D.family.obj ((SyntheticObject.shift δ X).obj D.nu D.auxiliary)).Page r
  63         (s,(p+δ).1+s,(p+δ).2)
  64   representative_iff : ∀ (X : SyntheticObject) (r : ℤ) (hr : 2 ≤ r)
  65       (p : ℤ × ℤ) (s : ℤ)
  66       (x : (D.family.obj (X.obj D.nu D.auxiliary)).Page r (s,p.1+s,p.2))
  67       (z : ((D.family.obj (X.obj D.nu D.auxiliary)).sequence.ssData (s,p.1+s,p.2)).eInfty),
  68     HasInfinityRepresentative _ r (s,p.1+s,p.2) x z ↔
  69       HasInfinityRepresentative _ r (s,(p+δ).1+s,(p+δ).2)
  70         ((page X r hr p s).hom x)
  71         ((towerInfinityIso (D.convergence X) (D.convergence (.shift δ X))
  72           (filtered X p) s).hom z)

  77 /-- The transport square is tied to the family maps of this very `g` and
  78 its shift, with the given convergence on both ends. -/
  79 public def transport (X Y : SyntheticObject)
  80     (g : X.obj D.nu D.auxiliary ⟶ Y.obj D.nu D.auxiliary) (p : ℤ × ℤ) :
  81     FilteredMapTransport (D.convergentMap X Y g).filteredMap
  82       (D.convergentMap (.shift δ X) (.shift δ Y)
  83         ((SyntheticCategory.biShift δ).map g)).filteredMap p (p+δ) where
  84   source := S.filtered X p
  85   target := S.filtered Y p
  86   map_comm := by
  87     simp only [ConvergentMap.filteredMap_map]
  88     ext a
  89     change syntheticHomotopyMap ((SyntheticCategory.biShift δ).map g) (p+δ)
  90         ((S.filtered X p).ambient.hom a) =
  91       (S.filtered Y p).ambient.hom (syntheticHomotopyMap g p a)
  92     rw [S.ambient_eq, S.ambient_eq]
  93     change shiftHomotopyClass δ p a ≫ (SyntheticCategory.biShift δ).map g =
  94       shiftHomotopyClass δ p (a ≫ g)
  95     simp [shiftHomotopyClass, Functor.map_comp, Category.assoc]

  97 /-- A representative equation shifts in both directions, with its complete
  98 source and target ambiguity handled by `transport.targetCoset_iff`. -/
  99 public theorem equation_iff (X Y : SyntheticObject)
 100     (g : X.obj D.nu D.auxiliary ⟶ Y.obj D.nu D.auxiliary) (p : ℤ × ℤ) (s n : ℤ)
 101     {T : ModuleCat.{v} ℤ}
 102     (x : T ⟶ (D.convergence X).filtration.associatedGraded s p)
 103     (y : T ⟶ (D.convergence Y).filtration.associatedGraded (s+n) p) :
 104     FilteredMapExtension.Relation (D.convergentMap X Y g).filteredMap p s n x y ↔
 105       FilteredMapExtension.Relation
 106         (D.convergentMap (.shift δ X) (.shift δ Y)
 107           ((SyntheticCategory.biShift δ).map g)).filteredMap (p+δ) s n
 108         (x ≫ ((S.filtered X p).gradedIso s).hom)
 109         (y ≫ ((S.filtered Y p).gradedIso (s+n)).hom) :=
 110   (S.transport X Y g p).relation_iff x y

 112 /-- The existing E∞ extension witnesses use the same transport.  This applies
 113 both to finite-quotient objects and to untruncated objects without imposing
 114 boundedness on their filtrations. -/
 115 public theorem towerExtensionEquation_iff (X Y : SyntheticObject)
 116     (g : X.obj D.nu D.auxiliary ⟶ Y.obj D.nu D.auxiliary) (p : ℤ × ℤ) (s n : ℤ)
 117     (x : ((D.family.obj (X.obj D.nu D.auxiliary)).sequence.ssData (s,p.1+s,p.2)).eInfty)
 118     (y : ((D.family.obj (Y.obj D.nu D.auxiliary)).sequence.ssData
 119       (s+n,p.1+(s+n),p.2)).eInfty) :
 120     TowerExtensionEquation (D.convergentMap X Y g) p s n x y ↔
 121       TowerExtensionEquation
 122         (D.convergentMap (.shift δ X) (.shift δ Y)
 123           ((SyntheticCategory.biShift δ).map g)) (p+δ) s n
 124         ((towerInfinityIso (D.convergence X) (D.convergence (.shift δ X))
 125           (S.filtered X p) s).hom x)
 126         ((towerInfinityIso (D.convergence Y) (D.convergence (.shift δ Y))
 127           (S.filtered Y p) (s+n)).hom y) := by
 128   change FilteredMapExtension.Relation _ p s n
 129       (elementMap x ≫ (towerGradedIso (D.convergence X) p s).hom)
 130       (elementMap y ≫ (towerGradedIso (D.convergence Y) p (s+n)).hom) ↔
 131     FilteredMapExtension.Relation _ (p+δ) s n
 132       (elementMap _ ≫ (towerGradedIso (D.convergence (.shift δ X)) (p+δ) s).hom)
 133       (elementMap _ ≫ (towerGradedIso (D.convergence (.shift δ Y)) (p+δ) (s+n)).hom)
 134   rw [(S.transport X Y g p).relation_iff]
 135   simp only [elementMap_comp, towerInfinityIso, Iso.trans_hom, Iso.symm_hom,
 136     Category.assoc, Iso.inv_hom_id, Category.comp_id, transport]

 140 /-- Boundary shift in the fixed bigraded closure. Its comparison with
 141 weight shift followed by ordinary suspension is supplied below. -/
 142 public def lambdaBoundaryShiftFunctor (n : ℕ) : Syn ⥤ Syn :=
 143   SyntheticCategory.biShift (1,-(n:ℤ))
 144 
 145 /-- The chosen bigraded realization is the same shift as the legacy
 146 `Σ ∘ Σ^(0,-n)`, up to the specified suspension comparison. -/
 147 public def lambdaBoundaryShiftIso (n : ℕ) :
 148     (SyntheticCategory.biShift (0,-(n:ℤ)) ⋙ shiftFunctor Syn (1:ℤ)) ≅
 149       (lambdaBoundaryShiftFunctor (Syn := Syn) n) :=
 150   Functor.isoWhiskerLeft (SyntheticCategory.biShift (0,-(n:ℤ)))
 151     (SyntheticCategory.biShift_compat (Syn := Syn) 1).symm ≪≫
 152       biShiftAddIso (0,-(n:ℤ)) (1,0) (1,-(n:ℤ)) (by ext <;> simp)
