  27   HasFunctorialCofiber.cofibδ (lambdaPow n X)

  29 /-- The map supplied by the same chosen cofiber construction. No identity
  30 or composition law for this choice is asserted by the minimal cofiber class. -/
  31 public noncomputable def XModLambdaN.map {X Y : Syn} (f : X ⟶ Y) (n : ℕ) :
  32     XModLambdaN X n ⟶ XModLambdaN Y n :=
  33   HasFunctorialCofiber.cofibMap (lambdaPow n X) (lambdaPow n Y)
  34     ((SyntheticCategory.biShift (0, -(n : ℤ))).map f) f (lambdaPow_naturality n f)

  39   Triangle.mk (lambdaPow n X) (XModLambdaN.incl X n) (XModLambdaN.proj X n)

 100 /-- The chosen cofiber map of the proved λ-power factorization square.
 101 Both its source and target are the existing `XModLambdaN` objects. This
 102 construction alone does not assert identity or composition laws for the
 103 choice of `HasFunctorialCofiber.cofibMap`. -/
 104 public noncomputable def XModLambdaN.restriction (coh : BiShiftCoherence Syn)
 105     (X : Syn) (i j : ℕ) (hij : i ≤ j) : XModLambdaN X j ⟶ XModLambdaN X i :=
 106   HasFunctorialCofiber.cofibMap (lambdaPow j X) (lambdaPow i X)
 107     (lambdaRestrictionSourceMap i j hij X) (𝟙 X)
 108     (by simpa only [lambdaDegree, Category.comp_id] using
 109       lambdaRestrictionSourceMap_commutes coh i j hij X)
 110 
 111 end KIP126.Synthetic.Context

 122 /-- A finite λ–ρ–δ triangle on the actual quotients, including its
 123 compatibility with multiplication by the existing `lambdaPow i X`. -/
 124 public structure LambdaRhoDeltaTriangle (X : Syn) (i j : ℕ) where
 125   i_pos : 0 < i
 126   i_lt_j : i < j
 127   lambdaMap :
 128     (SyntheticCategory.biShift (0, -(i : ℤ))).obj (XModLambdaN X (j - i)) ⟶
 129       XModLambdaN X j
 130   rho : XModLambdaN X j ⟶ XModLambdaN X i
 131   delta : XModLambdaN X i ⟶
 132     ((SyntheticCategory.biShift (0, -(i : ℤ))).obj
 133       (XModLambdaN X (j - i)))⟦(1 : ℤ)⟧
 134   distinguished : Triangle.mk lambdaMap rho delta ∈ distTriang Syn
 135   lambda_quotient :
 136     (SyntheticCategory.biShift (0, -(i : ℤ))).map (XModLambdaN.incl X (j - i)) ≫
 137         lambdaMap = lambdaPow i X ≫ XModLambdaN.incl X j
 138   rho_quotient : XModLambdaN.incl X j ≫ rho = XModLambdaN.incl X i
 139 
 140 /-- Restriction maps between the existing finite λ quotients with exact
 141 identity, composition, quotient, and triangle compatibility laws. -/
 142 public structure FiniteLambdaQuotientTower (X : Syn) where
 143   rho : ∀ (i j : ℕ), i ≤ j → (XModLambdaN X j ⟶ XModLambdaN X i)
 144   rho_id : ∀ i, rho i i le_rfl = 𝟙 (XModLambdaN X i)
 145   rho_comp : ∀ {k i j : ℕ} (hki : k ≤ i) (hij : i ≤ j),
 146     rho i j hij ≫ rho k i hki = rho k j (hki.trans hij)
 147   rho_quotient : ∀ {i j : ℕ} (hij : i ≤ j),
 148     XModLambdaN.incl X j ≫ rho i j hij = XModLambdaN.incl X i
 149   triangle : ∀ {i j : ℕ}, 0 < i → i < j → LambdaRhoDeltaTriangle X i j
 150   triangle_rho : ∀ {i j : ℕ} (hi : 0 < i) (hij : i < j),
 151     (triangle hi hij).rho = rho i j hij.le
 152 
 153 end KIP126.Synthetic.Context

 164 /-- All components are the actual `XModLambdaN.map f n`; this record supplies
 165 their compatibility with the specified restrictions and λ–ρ–δ triangles. -/
 166 public structure FiniteLambdaQuotientTower.Hom (TX : FiniteLambdaQuotientTower X)
 167     (TY : FiniteLambdaQuotientTower Y) (f : X ⟶ Y) : Prop where
 168   rho_naturality : ∀ {i j : ℕ} (hij : i ≤ j),
 169     TX.rho i j hij ≫ XModLambdaN.map f i =
 170       XModLambdaN.map f j ≫ TY.rho i j hij
 171   lambda_naturality : ∀ {i j : ℕ} (hi : 0 < i) (hij : i < j),
 172     (SyntheticCategory.biShift (0, -(i : ℤ))).map
 173           (XModLambdaN.map f (j - i)) ≫
 174         (TY.triangle hi hij).lambdaMap =
 175       (TX.triangle hi hij).lambdaMap ≫ XModLambdaN.map f j
 176   delta_naturality : ∀ {i j : ℕ} (hi : 0 < i) (hij : i < j),
 177     XModLambdaN.map f i ≫ (TY.triangle hi hij).delta =
 178       (TX.triangle hi hij).delta ≫
 179         (shiftFunctor Syn (1 : ℤ)).map
 180           ((SyntheticCategory.biShift (0, -(i : ℤ))).map
 181             (XModLambdaN.map f (j - i)))
 182 
 183 end KIP126.Synthetic.Context

 191 /-- The quotient functor uses exactly the selected cofiber maps. -/
 192 public noncomputable def LambdaQuotientFunctoriality.functor
 193     (Q : LambdaQuotientFunctoriality (Syn := Syn)) (n : ℕ) : Syn ⥤ Syn where
 194   obj X := XModLambdaN X n
 195   map f := XModLambdaN.map f n
 196   map_id X := Q.map_id X n
 197   map_comp f g := Q.map_comp f g n

 199 /-- Restrictions commute with the actual inclusion, by the cofiber square. -/
 200 public theorem XModLambdaN.restriction_incl (coh : BiShiftCoherence Syn)
 201     (X : Syn) (i j : ℕ) (hij : i ≤ j) :
 202     XModLambdaN.incl X j ≫ XModLambdaN.restriction coh X i j hij =
 203       XModLambdaN.incl X i := by
 204   have h := HasFunctorialCofiber.cofibMap_ι (lambdaPow j X) (lambdaPow i X)
 205     (lambdaRestrictionSourceMap i j hij X) (𝟙 X)
 206     (by simpa only [lambdaDegree, Category.comp_id] using
 207       lambdaRestrictionSourceMap_commutes coh i j hij X)
 208   simpa only [Category.id_comp, XModLambdaN.incl, XModLambdaN.restriction, XModLambdaN] using h.symm

 210 /-- The selected finite quotient system and its structural coherences.
 211 The restriction agreement rules out an unrelated family of identically named maps. -/
 212 public structure LambdaQuotientSystem (Syn : Type u) [SyntheticCategory.{u,v} Syn]
 213     [HasFunctorialCofiber (C := Syn)] where
 214   coherence : BiShiftCoherence Syn
 215   functoriality : LambdaQuotientFunctoriality (Syn := Syn)
 216   tower : ∀ X : Syn, FiniteLambdaQuotientTower X
 217   restriction : ∀ (X : Syn) (i j : ℕ) (hij : i ≤ j),
 218     (tower X).rho i j hij = XModLambdaN.restriction coherence X i j hij
 219   natural : ∀ {X Y : Syn} (f : X ⟶ Y), FiniteLambdaQuotientTower.Hom (tower X) (tower Y) f
 220 end KIP126.Synthetic.Context
