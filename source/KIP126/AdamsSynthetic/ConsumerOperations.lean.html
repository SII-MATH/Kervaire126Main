  23 open CategoryTheory KIP126.Synthetic.SpectralSequence
  24 universe v

  26 public def SyntheticRepresentsOnPage (E : SyntheticAdamsSS.{v}) (r : ℤ)
  27     (p : Tridegree) (x : E.E₂ p) (y : E.Page r p) : Prop :=
  28   ∃ hr : 2 ≤ r,
  29     let A := E.sequence.ssData p
  30     let n : WithTop ℕ := ↑(2-E.sequence.r₀).toNat
  31     let m : WithTop ℕ := ↑(r-E.sequence.r₀).toNat
  32     ∃ z : (Subobject.underlying.obj (A.Z m) : ModuleCat ℤ),
  33       (Subobject.ofLE _ _ (A.Z_anti (by
  34         dsimp [n,m]; exact_mod_cast (show (2-E.sequence.r₀).toNat ≤
  35           (r-E.sequence.r₀).toNat by omega))) ≫ A.pageπ n) z = x ∧ A.pageπ m z = y
  36 
  37 public def SyntheticHasDifferential (E : SyntheticAdamsSS.{v}) (r : ℤ)
  38     (p q : Tridegree) (x : E.E₂ p) (y : E.E₂ q) : Prop :=
  39   ∃ h : p+E.sequence.diffDeg r = q,
  40     ∃ (xr : E.Page r p) (yr : E.Page r q),
  41       SyntheticRepresentsOnPage E r p x xr ∧ SyntheticRepresentsOnPage E r q y yr ∧
  42       (E.sequence.d r p ≫ eqToHom (congrArg (E.Page r) h)) xr = yr
  43 end KIP126.AdamsSynthetic
