  25 /-- Canonical identification on lifts in the ONE supplied abstract tower.
  26 The cofiber image and the actual homotopy class use the same lift. -/
  27 public def ClassicalCanonicalIdentification (P : ClassicalTowerPresentation unit A) (X : C)
  28     (e : ∀ p : ℤ × ℤ, ((A.obj X).ssData p).eInfty ≅
  29       (P.tower X).filtration.associatedGraded p.1 (p.2-p.1)) : Prop :=
  30   ∀ (p : ℤ × ℤ) (a : HomotopyGroup (p.2-p.1) ((P.tower X).stageAt p.1))
  31     (z : (Subobject.underlying.obj (((A.obj X).ssData p).Z ⊤) : ModuleCat ℤ)),
  32     (P.sequence X).infiniteRepresentativeE1 p z = (P.tower X).layerMap p a →
  33     ∀ b : (Subobject.underlying.obj ((P.tower X).filtration.F p.1 (p.2-p.1)) : ModuleCat ℤ),
  34       ((P.tower X).filtration.F p.1 (p.2-p.1)).arrow b =
  35         a ≫ (P.tower X).projection p.1 →
  36       (e p).hom (((A.obj X).ssData p).pageπ ⊤ z) =
  37         (P.tower X).filtration.toAssociatedGraded p.1 (p.2-p.1) b
  38 
  39 /-- Convergence is requested only for this object. It does not construct a
  40 convergent instance or assert separation of its filtration. -/
  41 public structure ClassicalTowerConvergence (P : ClassicalTowerPresentation unit A) (X : C) where
  42   identification : ∀ p : ℤ × ℤ, ((A.obj X).ssData p).eInfty ≅
  43     (P.tower X).filtration.associatedGraded p.1 (p.2-p.1)
  44   canonical : ClassicalCanonicalIdentification P X identification

  46 public def ClassicalTowerConvergence.toConvergence
  47     {P : ClassicalTowerPresentation unit A} {X : C} (c : ClassicalTowerConvergence P X) :
  48     Convergence (A.obj X) (AbstractAdamsTower.homotopy X) (P.tower X).filtration where
  49   reindex p := (p.1,p.2-p.1)
  50   reindex_bijective := by
  51     constructor
  52     · rintro ⟨s,t⟩ ⟨s',t'⟩ h
  53       have hs := congrArg Prod.fst h
  54       have ht := congrArg Prod.snd h
  55       dsimp at hs ht
  56       ext <;> omega
  57     · rintro ⟨s,m⟩
  58       exact ⟨(s,m+s),by ext <;> simp⟩
  59   iso := c.identification

  61 public def ClassicalDetects {P : ClassicalTowerPresentation unit A} {X : C}
  62     (c : ClassicalTowerConvergence P X) (p : ℤ × ℤ) (x : (A.obj X).Page 2 p)
  63     (a : HomotopyGroup (p.2-p.1) X) : Prop :=
  64   ∃ z : (Subobject.underlying.obj (((A.obj X).ssData p).Z ⊤) : ModuleCat ℤ),
  65     (Subobject.ofLE _ _ (((A.obj X).ssData p).Z_anti le_top) ≫
  66       ((A.obj X).ssData p).pageπ (↑(2-(A.obj X).r₀).toNat : WithTop ℕ)) z = x ∧
  67     ∃ b : (Subobject.underlying.obj ((P.tower X).filtration.F p.1 (p.2-p.1)) : ModuleCat ℤ),
  68       ((P.tower X).filtration.F p.1 (p.2-p.1)).arrow b = a ∧
  69       (c.identification p).hom (((A.obj X).ssData p).pageπ ⊤ z) =
  70         (P.tower X).filtration.toAssociatedGraded p.1 (p.2-p.1) b
  71 end Classical

  78 public def syntheticHomotopyMap {X Y : Syn} (f : X ⟶ Y) (p : ℤ × ℤ) :
  79     syntheticHomotopy X p ⟶ syntheticHomotopy Y p :=
  80   ModuleCat.ofHom ((Preadditive.rightComp (Smn p.1 p.2) f).toIntLinearMap)
  81 
  82 /-- The fixed weight desuspension is dictated by the given grading. -/
  83 public def weightSourceIso (m w : ℤ) :
  84     (SyntheticCategory.biShift (0,-w)).obj (Smn m w (Syn := Syn)) ≅ Sphere (C := Syn) m :=
  85   (SyntheticCategory.biShift_comp (m,w) (0,-w)).app S_0_0 ≪≫
  86     eqToIso (by simp) ≪≫
  87     (SyntheticCategory.biShift_compat (Syn := Syn) m).app S_0_0
  88 
  89 public def weightHomotopyMap (m w : ℤ) (X : Syn) (a : BiHom m w X) :
  90     HomotopyGroup m ((SyntheticCategory.biShift (0,-w)).obj X) :=
  91   (weightSourceIso m w).inv ≫ (SyntheticCategory.biShift (0,-w)).map a

  94 public def filtrationSubmodule (P : SyntheticTowerPresentation unit F) (X : Syn)
  95     (s : ℤ) (p : ℤ × ℤ) : Submodule ℤ (BiHom p.1 p.2 X) :=
  96   LinearMap.range ((Preadditive.rightComp (Smn p.1 p.2)
  97     ((P.tower X).projection s)).toIntLinearMap)

  99 public theorem filtrationSubmodule_antitone (P : SyntheticTowerPresentation unit F)
 100     (X : Syn) (p : ℤ × ℤ) : Antitone (fun s => P.filtrationSubmodule X s p) := by
 101   intro s t h a ha
 102   obtain ⟨b, rfl⟩ := ha
 103   refine ⟨b ≫ (P.tower X).mapAt h, ?_⟩
 104   change (b ≫ (P.tower X).mapAt h) ≫ (P.tower X).projection s = b ≫ (P.tower X).projection t
 105   rw [Category.assoc, (P.tower X).projection_compat h]

 107 public def filtration (P : SyntheticTowerPresentation unit F) (X : Syn) :
 108     Filtration (syntheticHomotopy X) where
 109   F s p := (ModuleCat.subobjectModule _).symm (P.filtrationSubmodule X s p)
 110   mono s p := (ModuleCat.subobjectModule _).symm.monotone
 111     (P.filtrationSubmodule_antitone X p (by omega : s ≤ s+1))
 112 end SyntheticTowerPresentation

 114 /-- Both clauses are necessary: agreement of the actual weight-shifted tower
 115 filtrations, and the canonical cofiber/projection equation on the same lift. -/
 116 public def SyntheticCanonicalIdentification (P : SyntheticTowerPresentation unit F) (X : Syn)
 117     (e : ∀ i : Tridegree, ((F.obj X).sequence.ssData i).eInfty ≅
 118       (P.filtration X).associatedGraded i.1 (i.2.1-i.1,i.2.2)) : Prop :=
 119   (∀ (s m w : ℤ) (a : BiHom m w X),
 120     a ∈ P.filtrationSubmodule X s (m,w) ↔
 121       weightHomotopyMap m w X a ∈
 122         (P.tower ((SyntheticCategory.biShift (0,-w)).obj X)).filtrationSubmodule s m) ∧
 123   ∀ (i : Tridegree)
 124     (a : HomotopyGroup (i.2.1-i.1)
 125       ((P.tower ((SyntheticCategory.biShift (0,-i.2.2)).obj X)).stageAt i.1))
 126     (z : (Subobject.underlying.obj (((F.obj X).sequence.ssData i).Z ⊤) : ModuleCat ℤ)),
 127     (P.sequence X i.2.2).infiniteRepresentativeE1 (i.1,i.2.1) z =
 128       (P.tower ((SyntheticCategory.biShift (0,-i.2.2)).obj X)).layerMap (i.1,i.2.1) a →
 129     ∀ b : (Subobject.underlying.obj ((P.filtration X).F i.1 (i.2.1-i.1,i.2.2)) : ModuleCat ℤ),
 130       weightHomotopyMap (i.2.1-i.1) i.2.2 X
 131         (((P.filtration X).F i.1 (i.2.1-i.1,i.2.2)).arrow b) =
 132         a ≫ (P.tower ((SyntheticCategory.biShift (0,-i.2.2)).obj X)).projection i.1 →
 133       (e i).hom (((F.obj X).sequence.ssData i).pageπ ⊤ z) =
 134         (P.filtration X).toAssociatedGraded i.1 (i.2.1-i.1,i.2.2) b

 145 public def toConvergence {P : SyntheticTowerPresentation unit F} {X : Syn}
 146     (c : SyntheticTowerConvergence P X) :
 147     Convergence (F.obj X).sequence (syntheticHomotopy X) (P.filtration X) where
 148   reindex i := (i.1,i.2.1-i.1,i.2.2)
 149   reindex_bijective := by
 150     constructor
 151     · rintro ⟨s,t,w⟩ ⟨s',t',w'⟩ h
 152       change (s,t-s,w) = (s',t'-s',w') at h
 153       simp only [Prod.mk.injEq] at h
 154       obtain ⟨hs,ht,hw⟩ := h
 155       exact Prod.ext hs (Prod.ext (by dsimp; omega) hw)
 156     · rintro ⟨s,m,w⟩
 157       exact ⟨(s,m+s,w),by ext <;> simp⟩
 158   iso := c.identification
 159 end SyntheticTowerConvergence

 161 /-- A common infinite-cycle representative, including the zero case. -/
 162 public def HasInfinityRepresentative (A : SyntheticAdamsSS.{v}) (r : ℤ) (i : Tridegree)
 163     (x : A.Page r i) (e : (A.sequence.ssData i).eInfty) : Prop :=
 164   2 ≤ r ∧ ∃ z : (Subobject.underlying.obj ((A.sequence.ssData i).Z ⊤) : ModuleCat ℤ),
 165     (Subobject.ofLE _ _ ((A.sequence.ssData i).Z_anti le_top) ≫
 166       (A.sequence.ssData i).pageπ (↑(r-A.sequence.r₀).toNat : WithTop ℕ)) z = x ∧
 167       (A.sequence.ssData i).pageπ ⊤ z = e

 197 public theorem HasInfinityRepresentative.unique {A : SyntheticAdamsSS.{v}}
 198     {r : ℤ} {i : Tridegree} {x : A.Page r i}
 199     {e e' : (A.sequence.ssData i).eInfty}
 200     (h : HasInfinityRepresentative A r i x e)
 201     (h' : HasInfinityRepresentative A r i x e') : e = e' := by
 202   obtain ⟨_, z, hz, he⟩ := h
 203   obtain ⟨_, z', hz', he'⟩ := h'
 204   exact he.symm.trans
 205     (((A.sequence.ssData i).infinity_projection_eq_of_page_projection_eq
 206       (↑(r - A.sequence.r₀).toNat) z z' (hz.trans hz'.symm)).trans he')

 211 public theorem detects_add {x y : (F.obj X).E₂ i}
 212     {a b : BiHom (i.2.1-i.1) i.2.2 X}
 213     (ha : Detects c i x a) (hb : Detects c i y b) :
 214     Detects c i (x+y) (a+b) := by
 215   obtain ⟨e, ⟨hr, z, hz, he⟩, a', ha', hga⟩ := ha
 216   obtain ⟨e', ⟨_, z', hz', he'⟩, b', hb', hgb⟩ := hb
 217   refine ⟨e + e', ⟨hr, z + z', ?_, ?_⟩, a' + b', ?_, ?_⟩
 218   · rw [map_add, hz, hz']
 219   · rw [map_add, he, he']
 220   · rw [map_add, ha', hb']
 221   · rw [map_add, map_add, hga, hgb]

 223 private theorem filtrationAtLeast_of_graded_eq_zero (s m w : ℤ)
 224     (a : (Subobject.underlying.obj ((P.filtration X).F s (m,w)) : ModuleCat ℤ))
 225     (ha : (P.filtration X).toAssociatedGraded s (m,w) a = 0) :
 226     FiltrationAtLeast P (s+1) (((P.filtration X).F s (m,w)).arrow a) := by
 227   have hmem := ((P.filtration X).toAssociatedGraded_eq_zero_iff s (m,w) a).mp ha
 228   simpa only [SyntheticTowerPresentation.filtration, OrderIso.apply_symm_apply,
 229     FiltrationAtLeast] using hmem

 231 /-- A zero infinity class raises actual filtration; it does not imply the
 232 homotopy class itself vanishes. -/
 233 public theorem detects_zero_infinity_filtration
 234     {x : (F.obj X).E₂ i} {a : BiHom (i.2.1-i.1) i.2.2 X}
 235     (ha : Detects c i x a)
 236     (hx : HasInfinityRepresentative (F.obj X) 2 i x 0) :
 237     FiltrationAtLeast P (i.1+1) a := by
 238   obtain ⟨e, he, a', ha', hgr⟩ := ha
 239   have he0 := he.unique hx
 240   rw [he0, map_zero] at hgr
 241   rw [← ha']
 242   exact filtrationAtLeast_of_graded_eq_zero i.1 (i.2.1-i.1) i.2.2 a' hgr.symm

 244 public theorem detects_zero_filtration
 245     {a : BiHom (i.2.1-i.1) i.2.2 X} (ha : Detects c i 0 a) :
 246     FiltrationAtLeast P (i.1+1) a :=
 247   detects_zero_infinity_filtration c i ha

 250 /-- Equal leading labels determine a coset of the next filtration step. -/
 251 public theorem detects_sub_filtration {x : (F.obj X).E₂ i}
 252     {a b : BiHom (i.2.1-i.1) i.2.2 X}
 253     (ha : Detects c i x a) (hb : Detects c i x b) :
 254     FiltrationAtLeast P (i.1+1) (a-b) := by
 255   obtain ⟨e, he, a', ha', hga⟩ := ha
 256   obtain ⟨e', he', b', hb', hgb⟩ := hb
 257   have heq := he.unique he'
 258   have hzero : (P.filtration X).toAssociatedGraded i.1 (i.2.1-i.1,i.2.2) (a'-b') = 0 := by
 259     rw [map_sub, ← hga, ← hgb, heq, sub_self]
 260   have h := filtrationAtLeast_of_graded_eq_zero i.1 (i.2.1-i.1) i.2.2 (a'-b') hzero
 261   simpa only [map_sub, ha', hb'] using h
