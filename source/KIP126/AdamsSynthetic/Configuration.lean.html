   7 /-! Object closure for the selected proof route. Auxiliary objects are
   8 parameters with explicit maps, not asserted to have tmf's properties by
   9 their names. A(M) supplies the literature identifications/properties.
  10 The cofiber and its cell maps are derived from the supplied ν map. -/
  11 namespace KIP126.Kervaire.Route
  12 open CategoryTheory CategoryTheory.Pretriangulated KIP126.StableHomotopy KIP126.Synthetic.Context
  13 universe u v w
  14 variable {C : Type u} [StableHomotopyCategory.{u, v} C]
  15 
  16 public structure AuxiliaryData (C : Type u) [StableHomotopyCategory.{u, v} C] where
  17   /-- The classical η map used by the generalized Leibniz rule.
  18   Its identification with h₁ and with the synthetic η is an A(M) input. -/
  19   etaMap : Sphere (C := C) 1 ⟶ SphereSpectrum
  20   /-- The selected classical Hopf map; its h₂ detection is an A(M) input. -/
  21   nuMap : Sphere (C := C) 3 ⟶ SphereSpectrum
  22   /-- In the paper this pointed detecting spectrum is tmf. -/
  23   detector : C
  24   detectorUnit : SphereSpectrum ⟶ detector
  25 
  26 /-- No other spectra are needed in the selected Section 7 route when C(M)
  27 results are accepted; spectra used to prove C(M) are a separate dependency. -/
  28 public inductive ClassicalObject
  29   | sphere
  30   | nuCofiber
  31   | detector
  32   | shift (n : ℤ) (X : ClassicalObject)
  33 
  34 public noncomputable def ClassicalObject.obj [HasFunctorialCofiber (C := C)]
  35     (A : AuxiliaryData C) : ClassicalObject → C
  36   | .sphere => SphereSpectrum
  37   | .nuCofiber => HasFunctorialCofiber.cofib A.nuMap
  38   | .detector => A.detector
  39   | .shift n X => (shiftFunctor C n).obj (X.obj A)
  40 
  41 /-- Synthetic objects used by normalized maps, finite λ/ρ/δ triangles,
  42 and the untruncated argument. Closure includes all shifts and quotients,
  43 but does not demand convergence on every object of the ambient category. -/
  44 public inductive SyntheticObject
  45   | sphere
  46   | nu (X : ClassicalObject)
  47   | shift (p : ℤ × ℤ) (X : SyntheticObject)
  48   | quotient (n : ℕ) (X : SyntheticObject)
  49 
  50 public noncomputable def SyntheticObject.obj [HasFunctorialCofiber (C := C)]
  51     {Syn : Type w} [SyntheticCategory.{w, v} Syn] [HasFunctorialCofiber (C := Syn)]
  52     (N : NuFunctorData C Syn) (A : AuxiliaryData C) : SyntheticObject → Syn
  53   | .sphere => S_0_0
  54   | .nu X => N.functor.obj (X.obj A)
  55   | .shift p X => (SyntheticCategory.biShift p).obj (X.obj N A)
  56   | .quotient n X => XModLambdaN (X.obj N A) n
  57 
  58 /-- A distinguished triangle within the selected classical object closure.
  59 The connecting map lands in the actual suspension of X. -/
  60 public structure TriangleData [HasFunctorialCofiber (C := C)] (A : AuxiliaryData C) where
  61   X : ClassicalObject
  62   Y : ClassicalObject
  63   Z : ClassicalObject
  64   f : X.obj A ⟶ Y.obj A
  65   g : Y.obj A ⟶ Z.obj A
  66   h : Z.obj A ⟶ (ClassicalObject.shift 1 X).obj A
  67   distinguished : CategoryTheory.Pretriangulated.Triangle.mk f g h ∈
  68     distTriang C
  69 
  70 /-- The specific triangle used in the ν-extension argument. Neither its
  71 cofiber nor its bottom/top cell maps are independent choices. -/
  72 public noncomputable def AuxiliaryData.nuRouteTriangle [HasFunctorialCofiber (C := C)]
  73     (A : AuxiliaryData C) : TriangleData A where
  74   X := .shift 3 .sphere
  75   Y := .sphere
  76   Z := .nuCofiber
  77   f := A.nuMap
  78   g := HasFunctorialCofiber.cofibι A.nuMap
  79   h := HasFunctorialCofiber.cofibδ A.nuMap
  80   distinguished := HasFunctorialCofiber.cofib_distinguished A.nuMap
  81 end KIP126.Kervaire.Route
