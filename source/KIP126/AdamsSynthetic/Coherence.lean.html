  18 /-- The same recursive λ powers, constructed as natural transformations.
  19 Their components are proved equal to the existing `lambdaPow` separately. -/
  20 public noncomputable def lambdaPowNatTrans : (n : ℕ) →
  21     SyntheticCategory.biShift (0, -(n : ℤ)) ⟶ 𝟭 Syn
  22   | 0 => SyntheticCategory.biShift_zero.hom
  23   | n + 1 => by
  24       have step : SyntheticCategory.biShift ((0 : ℤ), -1) ⋙
  25           SyntheticCategory.biShift ((0 : ℤ), -(n : ℤ)) ⟶ 𝟭 Syn :=
  26         Functor.whiskerRight SyntheticCategory.lam
  27             (SyntheticCategory.biShift ((0 : ℤ), -(n : ℤ))) ≫
  28           lambdaPowNatTrans n
  29       have result : SyntheticCategory.biShift ((0, -1) + (0, -(n : ℤ))) ⟶ 𝟭 Syn :=
  30         (SyntheticCategory.biShift_comp (0, -1) (0, -(n : ℤ))).inv ≫ step
  31       have heq : ((0 : ℤ), (-1 : ℤ)) + ((0 : ℤ), -(n : ℤ)) =
  32           ((0 : ℤ), -(↑(n + 1) : ℤ)) := by simp
  33       exact heq ▸ result
  34 
  35 end KIP126.Synthetic.Context

  47 theorem transported_app {a b : ℤ × ℤ} (hab : a = b)
  48     (α : SyntheticCategory.biShift a ⟶ 𝟭 Syn) (X : Syn) :
  49     (hab ▸ α).app X = (hab ▸ α.app X) := by
  50   subst b
  51   rfl
  52 
  53 /-- This natural-transformation presentation does not choose a different λ power. -/
  54 @[simp] public theorem lambdaPowNatTrans_app (n : ℕ) (X : Syn) :
  55     (lambdaPowNatTrans n).app X = lambdaPow n X := by
  56   induction n with
  57   | zero => rfl
  58   | succ n ih =>
  59     simp only [lambdaPowNatTrans, lambdaPow, transported_app, NatTrans.comp_app,
  60       Functor.whiskerRight_app, ih]
  61 
  62 /-- Every existing λ-power is natural in the synthetic object. -/
  63 public theorem lambdaPow_naturality (n : ℕ) {X Y : Syn} (f : X ⟶ Y) :
  64     (SyntheticCategory.biShift (0, -(n : ℤ))).map f ≫ lambdaPow n Y =
  65       lambdaPow n X ≫ f := by
  66   have h := (lambdaPowNatTrans n).naturality f
  67   dsimp only [Functor.id_obj, Functor.id_map] at h
  68   simpa only [lambdaPowNatTrans_app] using h
  69 
  70 end KIP126.Synthetic.Context

  80 /-- The specified addition comparison, transported to a named total degree. -/
  81 public noncomputable def biShiftAddIso (a b c : ℤ × ℤ) (h : a + b = c) :
  82     SyntheticCategory.biShift a ⋙ SyntheticCategory.biShift b ≅
  83       (SyntheticCategory.biShift c : Syn ⥤ Syn) :=
  84   SyntheticCategory.biShift_comp a b ≪≫
  85     eqToIso (congrArg SyntheticCategory.biShift h)
  86 
  87 /-- The degree of the already defined `lambdaPow n`. -/
  88 public def lambdaDegree (n : ℕ) : ℤ × ℤ := (0, -(n : ℤ))
  89 
  90 /-- The existing addition comparison on the λ axis, with no new choices. -/
  91 public noncomputable def lambdaShiftAddIso (i j k : ℕ) (h : i + j = k) :
  92     SyntheticCategory.biShift (lambdaDegree i) ⋙
  93         SyntheticCategory.biShift (lambdaDegree j) ≅
  94       (SyntheticCategory.biShift (lambdaDegree k) : Syn ⥤ Syn) :=
  95   biShiftAddIso (lambdaDegree i) (lambdaDegree j) (lambdaDegree k)
  96     (by subst k; ext <;> simp [lambdaDegree, Nat.cast_add, add_comm])
  97 
  98 /-- The source map for restriction from the `j`th to the `i`th λ quotient.
  99 It uses multiplication by the existing `lambdaPow (j-i)`, through the
 100 specified addition comparison. Its commuting square is proved separately. -/
 101 public noncomputable def lambdaRestrictionSourceMap (i j : ℕ) (hij : i ≤ j) (X : Syn) :
 102     (SyntheticCategory.biShift (lambdaDegree j)).obj X ⟶
 103       (SyntheticCategory.biShift (lambdaDegree i)).obj X :=
 104   (lambdaShiftAddIso (j - i) i j (Nat.sub_add_cancel hij)).inv.app X ≫
 105     (SyntheticCategory.biShift (lambdaDegree i)).map (lambdaPow (j - i) X)
 106 
 107 end KIP126.Synthetic.Context

 117 /-- Associativity, units, and centrality of λ for the existing suspension
 118 comparisons. The named intermediate and final degrees only make every
 119 transport explicit; their equalities determine them uniquely. -/
 120 public structure BiShiftCoherence : Prop where
 121   associativity : ∀ (a b c ab bc total : ℤ × ℤ)
 122       (hab : a + b = ab) (hbc : b + c = bc)
 123       (habc : ab + c = total) (habc' : a + bc = total) (X : Syn),
 124     (SyntheticCategory.biShift c).map ((biShiftAddIso a b ab hab).hom.app X) ≫
 125         (biShiftAddIso ab c total habc).hom.app X =
 126       (biShiftAddIso b c bc hbc).hom.app ((SyntheticCategory.biShift a).obj X) ≫
 127         (biShiftAddIso a bc total habc').hom.app X
 128   left_unit : ∀ (a : ℤ × ℤ) (X : Syn),
 129     (biShiftAddIso 0 a a (zero_add a)).hom.app X =
 130       (SyntheticCategory.biShift a).map (SyntheticCategory.biShift_zero.hom.app X)
 131   right_unit : ∀ (a : ℤ × ℤ) (X : Syn),
 132     (biShiftAddIso a 0 a (add_zero a)).hom.app X =
 133       SyntheticCategory.biShift_zero.hom.app ((SyntheticCategory.biShift a).obj X)
 134   lambda_comm : ∀ (a : ℤ × ℤ) (X : Syn),
 135     SyntheticCategory.lam.app ((SyntheticCategory.biShift a).obj X) =
 136       (biShiftAddIso a (0, -1) (a + (0, -1)) rfl).hom.app X ≫
 137         (biShiftAddIso (0, -1) a (a + (0, -1)) (add_comm _ _)).inv.app X ≫
 138         (SyntheticCategory.biShift a).map (SyntheticCategory.lam.app X)
 139 
 140 end KIP126.Synthetic.Context

 152 theorem lambdaDegree_add (i j : ℕ) :
 153     lambdaDegree (i + j) = lambdaDegree i + lambdaDegree j := by
 154   ext <;> simp [lambdaDegree, Nat.cast_add, add_comm]

 156 /-- The associativity condition specialized to the existing λ suspensions. -/
 157 theorem BiShiftCoherence.lambdaShift_associativity (coh : BiShiftCoherence Syn)
 158     (i j k ij jk total : ℕ) (hij : i + j = ij) (hjk : j + k = jk)
 159     (htotal : ij + k = total) (htotal' : i + jk = total) (X : Syn) :
 160     (SyntheticCategory.biShift (lambdaDegree k)).map
 161           ((lambdaShiftAddIso i j ij hij).hom.app X) ≫
 162         (lambdaShiftAddIso ij k total htotal).hom.app X =
 163       (lambdaShiftAddIso j k jk hjk).hom.app
 164           ((SyntheticCategory.biShift (lambdaDegree i)).obj X) ≫
 165         (lambdaShiftAddIso i jk total htotal').hom.app X := by
 166   exact coh.associativity (lambdaDegree i) (lambdaDegree j) (lambdaDegree k)
 167     (lambdaDegree ij) (lambdaDegree jk) (lambdaDegree total) _ _ _ _ X

 169 theorem BiShiftCoherence.lambdaShift_zero_left (coh : BiShiftCoherence Syn)
 170     (n : ℕ) (X : Syn) :
 171     (lambdaShiftAddIso 0 n n (Nat.zero_add n)).hom.app X =
 172       (SyntheticCategory.biShift (lambdaDegree n)).map (lambdaPow 0 X) := by
 173   exact coh.left_unit (lambdaDegree n) X

 175 theorem comp_transport {a b : ℤ × ℤ} (h : a = b) (X Y : Syn)
 176     (f : (SyntheticCategory.biShift a).obj X ⟶ Y) :
 177     eqToHom (congrArg (fun p => (SyntheticCategory.biShift p).obj X) h) ≫

 182 /-- The defining successor equation, expressed through the same addition iso. -/
 183 public theorem lambdaPow_succ_comparison (n : ℕ) (X : Syn) :
 184     (lambdaShiftAddIso 1 n (n + 1) (by omega)).hom.app X ≫ lambdaPow (n + 1) X =
 185       (SyntheticCategory.biShift (lambdaDegree n)).map (SyntheticCategory.lam.app X) ≫
 186         lambdaPow n X := by
 187   simp only [lambdaShiftAddIso, biShiftAddIso, lambdaDegree,
 188     Iso.trans_hom, NatTrans.comp_app, eqToIso.hom, eqToHom_app, lambdaPow]
 189   rw [Category.assoc, comp_transport]
 190   exact Iso.hom_inv_id_app_assoc
 191     (SyntheticCategory.biShift_comp (0, -1) (0, -(n : ℤ))) X _
 192 
 193 theorem lambdaPow_succ_comparison_of_eq (n k : ℕ) (h : 1 + n = k) (X : Syn) :
 194     (lambdaShiftAddIso 1 n k h).hom.app X ≫ lambdaPow k X =
 195       (SyntheticCategory.biShift (lambdaDegree n)).map (SyntheticCategory.lam.app X) ≫
 196         lambdaPow n X := by
 197   have hk : k = n + 1 := by omega
 198   cases hk
 199   exact lambdaPow_succ_comparison n X

 201 /-- The chosen unit coherence identifies the first existing power with λ. -/
 202 public theorem lambdaPow_one (coh : BiShiftCoherence Syn) (X : Syn) :
 203     lambdaPow 1 X = SyntheticCategory.lam.app X := by
 204   apply (cancel_epi (SyntheticCategory.biShift_zero.hom.app
 205     ((SyntheticCategory.biShift (0, -1)).obj X))).mp
 206   calc
 207     _ = (lambdaShiftAddIso 1 0 1 rfl).hom.app X ≫ lambdaPow 1 X := by
 208       rw [show (lambdaShiftAddIso 1 0 1 rfl).hom.app X =
 209           SyntheticCategory.biShift_zero.hom.app
 210             ((SyntheticCategory.biShift (0, -1)).obj X) from
 211         coh.right_unit (0, -1) X]
 212     _ = (SyntheticCategory.biShift (lambdaDegree 0)).map
 213           (SyntheticCategory.lam.app X) ≫ lambdaPow 0 X :=
 214       lambdaPow_succ_comparison_of_eq 0 1 rfl X
 215     _ = _ := SyntheticCategory.biShift_zero.hom.naturality (SyntheticCategory.lam.app X)

 217 /-- Arbitrary power decomposition, proved for the original recursive powers.
 218 The final degree equality records all indexing transports explicitly. -/
 219 public theorem lambdaPow_add_comparison (coh : BiShiftCoherence Syn)
 220     (i j k : ℕ) (h : i + j = k) (X : Syn) :
 221     (lambdaShiftAddIso i j k h).hom.app X ≫ lambdaPow k X =
 222       (SyntheticCategory.biShift (lambdaDegree j)).map (lambdaPow i X) ≫
 223         lambdaPow j X := by
 224   induction i generalizing k with
 225   | zero =>
 226     have hk : k = j := by omega
 227     cases hk
 228     rw [coh.lambdaShift_zero_left]
 229   | succ i ih =>
 230     apply (cancel_epi ((SyntheticCategory.biShift (lambdaDegree j)).map
 231       ((lambdaShiftAddIso 1 i (i + 1) (by omega)).hom.app X))).mp
 232     calc
 233       _ = (lambdaShiftAddIso i j (i + j) rfl).hom.app
 234             ((SyntheticCategory.biShift (lambdaDegree 1)).obj X) ≫
 235           ((lambdaShiftAddIso 1 (i + j) k (by omega)).hom.app X ≫
 236             lambdaPow k X) := by
 237         rw [← Category.assoc,
 238           coh.lambdaShift_associativity 1 i j (i + 1) (i + j) k
 239             (by omega) rfl h (by omega) X]
 240         simp only [Category.assoc]
 241       _ = (lambdaShiftAddIso i j (i + j) rfl).hom.app
 242             ((SyntheticCategory.biShift (lambdaDegree 1)).obj X) ≫
 243           ((SyntheticCategory.biShift (lambdaDegree (i + j))).map
 244             (SyntheticCategory.lam.app X) ≫ lambdaPow (i + j) X) := by
 245         rw [lambdaPow_succ_comparison_of_eq]
 246       _ = (SyntheticCategory.biShift (lambdaDegree j)).map
 247             ((SyntheticCategory.biShift (lambdaDegree i)).map
 248               (SyntheticCategory.lam.app X)) ≫
 249           ((lambdaShiftAddIso i j (i + j) rfl).hom.app X ≫
 250             lambdaPow (i + j) X) := by
 251         rw [← Category.assoc, ← NatTrans.naturality]
 252         simp only [Functor.comp_map, Functor.id_obj, Category.assoc]
 253       _ = (SyntheticCategory.biShift (lambdaDegree j)).map
 254             ((SyntheticCategory.biShift (lambdaDegree i)).map
 255               (SyntheticCategory.lam.app X)) ≫
 256           ((SyntheticCategory.biShift (lambdaDegree j)).map (lambdaPow i X) ≫
 257             lambdaPow j X) := by rw [ih (i + j) rfl]
 258       _ = _ := by
 259         rw [← Category.assoc, ← Functor.map_comp, ← lambdaPow_succ_comparison,
 260           Functor.map_comp, Category.assoc]

 262 /-- The same decomposition with the comparison moved to the source. -/
 263 public theorem lambdaPow_add (coh : BiShiftCoherence Syn)
 264     (i j k : ℕ) (h : i + j = k) (X : Syn) :
 265     lambdaPow k X = (lambdaShiftAddIso i j k h).inv.app X ≫
 266       (SyntheticCategory.biShift (lambdaDegree j)).map (lambdaPow i X) ≫
 267         lambdaPow j X := by
 268   rw [← lambdaPow_add_comparison coh i j k h X]
 269   simp

 271 /-- The precise commuting square used to form actual quotient restrictions. -/
 272 public theorem lambdaRestrictionSourceMap_commutes (coh : BiShiftCoherence Syn)
 273     (i j : ℕ) (hij : i ≤ j) (X : Syn) :
 274     lambdaRestrictionSourceMap i j hij X ≫ lambdaPow i X = lambdaPow j X := by
 275   simpa only [lambdaRestrictionSourceMap, Category.assoc] using
 276     (lambdaPow_add coh (j - i) i j (Nat.sub_add_cancel hij) X).symm

 278 /-- The restriction square at an equal index has the identity source map.
 279 An identity law for the resulting cofiber map is a separate requirement on
 280 the chosen cofiber construction. -/
 281 public theorem lambdaRestrictionSourceMap_self (coh : BiShiftCoherence Syn)
 282     (i : ℕ) (X : Syn) :
 283     lambdaRestrictionSourceMap i i le_rfl X =
 284       𝟙 ((SyntheticCategory.biShift (lambdaDegree i)).obj X) := by
 285   have hzero (n : ℕ) (hn : n = 0) (hni : n + i = i) :
 286       (lambdaShiftAddIso n i i hni).inv.app X ≫
 287           (SyntheticCategory.biShift (lambdaDegree i)).map (lambdaPow n X) =
 288         𝟙 ((SyntheticCategory.biShift (lambdaDegree i)).obj X) := by
 289     cases hn
 290     rw [← coh.lambdaShift_zero_left i X]
 291     exact Iso.inv_hom_id_app _ X
 292   exact hzero (i - i) (Nat.sub_self i) (Nat.sub_add_cancel le_rfl)

 294 /-- The source map of the restriction square is natural before any cofiber
 295 choice is made. -/
 296 public theorem lambdaRestrictionSourceMap_naturality (i j : ℕ) (hij : i ≤ j)
 297     {X Y : Syn} (f : X ⟶ Y) :
 298     (SyntheticCategory.biShift (lambdaDegree j)).map f ≫
 299         lambdaRestrictionSourceMap i j hij Y =
 300       lambdaRestrictionSourceMap i j hij X ≫
 301         (SyntheticCategory.biShift (lambdaDegree i)).map f := by
 302   dsimp only [lambdaRestrictionSourceMap]
 303   rw [← Category.assoc, NatTrans.naturality, Category.assoc]
 304   dsimp only [Functor.comp_map, lambdaDegree]
 305   rw [← Functor.map_comp, lambdaPow_naturality, Functor.map_comp, Category.assoc]
