   8 /-! Actual sphere E₂ operations and designated standard classes.
   9 Squares are products in this same E₂. Nonvanishing is an explicit proposition
  10 at the indices a consumer needs, never a default assumption on the whole family. -/
  11 
  12 namespace KIP126.Classical.Adams
  13 open CategoryTheory KIP126.StableHomotopy
  14 universe u v
  15 variable {C : Type u} [StableHomotopyCategory.{u, v} C]

  17 public noncomputable abbrev ActualE2At (A : ClassicalAdamsFamily C) (p : Bidegree) :=
  18   A.sphere.Page 2 p
  19 
  20 /-- The multiplication and scalar structure on the given family's sphere E₂.
  21 The laws refer to this one product. `HEq` only transports equal degree sums. -/
  22 public structure SphereE2Algebra (A : ClassicalAdamsFamily C) where
  23   scalar : ∀ p : Bidegree, Module (ZMod 2) (ActualE2At A p)
  24   product : ∀ p q : Bidegree,
  25     ActualE2At A p →ₗ[ℤ] ActualE2At A q →ₗ[ℤ] ActualE2At A (p + q)
  26   unit : ActualE2At A (0, 0)
  27   assoc : ∀ (p q r : Bidegree) (x : ActualE2At A p) (y : ActualE2At A q)
  28       (z : ActualE2At A r),
  29     HEq (product (p + q) r (product p q x y) z)
  30       (product p (q + r) x (product q r y z))
  31   unit_left : ∀ (p : Bidegree) (x : ActualE2At A p),
  32     HEq (product (0, 0) p unit x) x
  33   unit_right : ∀ (p : Bidegree) (x : ActualE2At A p),
  34     HEq (product p (0, 0) x unit) x
  35

  39 public theorem product_smul_left (P : SphereE2Algebra A) (p q : Bidegree)
  40     (a : ZMod 2) (x : ActualE2At A p) (y : ActualE2At A q) :
  41     letI := P.scalar p
  42     letI := P.scalar (p + q)
  43     P.product p q (a • x) y = a • P.product p q x y := by
  44   letI := P.scalar p
  45   letI := P.scalar (p + q)
  46   let f : ActualE2At A p →+ ActualE2At A (p + q) :=
  47     { toFun := fun z => P.product p q z y
  48       map_zero' := by simp
  49       map_add' := by intros; simp }
  50   exact ZMod.map_smul f a x
  51 
  52 public theorem product_smul_right (P : SphereE2Algebra A) (p q : Bidegree)
  53     (a : ZMod 2) (x : ActualE2At A p) (y : ActualE2At A q) :
  54     letI := P.scalar q
  55     letI := P.scalar (p + q)
  56     P.product p q x (a • y) = a • P.product p q x y := by
  57   letI := P.scalar q
  58   letI := P.scalar (p + q)
  59   exact ZMod.map_smul (P.product p q x) a y
  60 end SphereE2Algebra

  62 public def hiDegree (i : ℕ) : Bidegree := (1, ((2 ^ i : ℕ) : ℤ))
  63 
  64 /-- Specified standard classes. Their identification with a source model is a
  65 producer obligation. No dimension or uniqueness condition is silently added. -/
  66 public structure StandardSphereClasses (A : ClassicalAdamsFamily C)
  67     (_P : SphereE2Algebra A) where
  68   hi : ∀ i : ℕ, ActualE2At A (hiDegree i)
  69 
  70 namespace StandardSphereClasses
  71 variable {A : ClassicalAdamsFamily C} {P : SphereE2Algebra A}
  72 
  73 public noncomputable abbrev standardHi (H : StandardSphereClasses A P) (i : ℕ) := H.hi i
  74 
  75 public noncomputable def standardHiSquare (H : StandardSphereClasses A P) (i : ℕ) :
  76     ActualE2At A (2, ((2 ^ (i + 1) : ℕ) : ℤ)) := by
  77   have hdegree : hiDegree i + hiDegree i = (2, ((2 ^ (i + 1) : ℕ) : ℤ)) := by
  78     ext <;> simp [hiDegree, pow_succ, mul_two]
  79   exact (eqToHom (congrArg (ActualE2At A) hdegree))
  80     (P.product (hiDegree i) (hiDegree i) (H.hi i) (H.hi i))
  81 
  82 /-- The square is exactly the same product, transported only along its degree sum. -/
  83 public theorem standardHiSquare_eq_product (H : StandardSphereClasses A P) (i : ℕ)
  84     (hdegree : hiDegree i + hiDegree i = (2, ((2 ^ (i + 1) : ℕ) : ℤ))) :
  85     H.standardHiSquare i = (eqToHom (congrArg (ActualE2At A) hdegree))
  86       (P.product (hiDegree i) (hiDegree i) (H.hi i) (H.hi i)) := rfl
  87 
  88 public noncomputable abbrev standardH6 (H : StandardSphereClasses A P) :
  89     ActualE2At A (1, 64) := H.standardHi 6
  90 
  91 public noncomputable abbrev standardH6Square (H : StandardSphereClasses A P) :
  92     ActualE2At A (2, 128) := H.standardHiSquare 6

  94 public theorem standardH6Square_eq_product (H : StandardSphereClasses A P) :
  95     H.standardH6Square = P.product (1, 64) (1, 64) H.standardH6 H.standardH6 := rfl

  97 /-- Required nonvanishing at one specified index, for use as an explicit input. -/
  98 public def HiNonzero (H : StandardSphereClasses A P) (i : ℕ) : Prop := H.hi i ≠ 0
  99 
 100 /-- Square nonvanishing is separate from the nonvanishing of the factor. -/
 101 public def HiSquareNonzero (H : StandardSphereClasses A P) (i : ℕ) : Prop :=
 102   H.standardHiSquare i ≠ 0
