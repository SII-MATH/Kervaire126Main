  19 /-- A convergence morphism tied to the actual map, on both abutment and E∞.
  20 The chosen source and target convergence data remain parameters of the type. -/
  21 public structure ConvergentMap (F : SyntheticAdamsFamily Syn)
  22     {X Y : Syn} (g : X ⟶ Y)
  23     {FX : Filtration (syntheticHomotopy X)} {FY : Filtration (syntheticHomotopy Y)}
  24     (cx : Convergence (F.obj X).sequence (syntheticHomotopy X) FX)
  25     (cy : Convergence (F.obj Y).sequence (syntheticHomotopy Y) FY) where
  26   comparison : ConvergenceMorphism cx cy
  27   abutment_eq : comparison.aMap = syntheticHomotopyMap g
  28   infinity_eq : comparison.eMap = (F.functor.map g).eInftyMap
  29 
  30 /-- An element as a map from the free rank-one module in the same universe. -/
  31 public noncomputable def elementMap {A : ModuleCat.{v} ℤ} (x : A) :
  32     ModuleCat.of ℤ (ULift.{v} ℤ) ⟶ A :=
  33   ModuleCat.ofHom ((LinearMap.toSpanSingleton ℤ A x).comp ULift.moduleEquiv.toLinearMap)

  41 /-- The filtered map uses the same `g`; compatibility is supplied by its
  42 convergence naturality witness, not by choosing another map. -/
  43 public noncomputable def filteredMap (M : ConvergentMap F g cx cy) : FilteredMorphism FX FY :=
  44   M.comparison.filteredMap
  45 
  46 @[simp] public theorem filteredMap_map (M : ConvergentMap F g cx cy) (p : ℤ × ℤ) :
  47     M.filteredMap.map p = syntheticHomotopyMap g p := congrFun M.abutment_eq p

  49 end ConvergentMap

  51 /-- Send an infinity-page arrow to the prescribed associated-graded piece.
  52 The equality transports indices of the same convergence, not its objects. -/
  53 public noncomputable def toGraded
  54     {E : KIP126.Core.SpectralSequence (ModuleCat.{v} ℤ) Tridegree}
  55     {A : (ℤ × ℤ) → ModuleCat.{v} ℤ} {G : Filtration A}
  56     (c : Convergence E A G) {i : Tridegree} {s : ℤ} {p : ℤ × ℤ}
  57     (h : c.reindex i = (s,p)) {T : ModuleCat.{v} ℤ}
  58     (x : T ⟶ (E.ssData i).eInfty) : T ⟶ G.associatedGraded s p :=
  59   x ≫ (c.iso i).hom ≫ G.transportGraded h

  67   (p : ℤ × ℤ) (s n : ℤ) {i j : Tridegree}
  68   (hi : cx.reindex i = (s,p)) (hj : cy.reindex j = (s+n,p))

  71 /-- The actual filtered extension equation, with the original nonnegative length. -/
  72 public def ExtensionEquation (x : T ⟶ ((F.obj X).sequence.ssData i).eInfty)
  73     (y : T ⟶ ((F.obj Y).sequence.ssData j).eInfty) : Prop :=
  74   FilteredMapExtension.Relation M.filteredMap p s n
  75     (toGraded cx hi x) (toGraded cy hj y)
  76 
  77 /-- Essentiality modulo the entire homogeneous target solution space. -/
  78 public def EssentialExtension (x : T ⟶ ((F.obj X).sequence.ssData i).eInfty)
  79     (y : T ⟶ ((F.obj Y).sequence.ssData j).eInfty) : Prop :=
  80   FilteredMapExtension.Essential M.filteredMap p s n
  81     (toGraded cx hi x) (toGraded cy hj y)
  82 
  83 /-- A consumer obtains actual lifts without expanding any filtered complex. -/
  84 public theorem extensionEquation_iff
  85     (x : T ⟶ ((F.obj X).sequence.ssData i).eInfty)
  86     (y : T ⟶ ((F.obj Y).sequence.ssData j).eInfty) :
  87     M.ExtensionEquation p s n hi hj x y ↔
  88       0 ≤ n ∧ ∃ (a : T ⟶ Subobject.underlying.obj (FX.F s p))
  89         (b : T ⟶ Subobject.underlying.obj (FY.F (s+n) p)),
  90         a ≫ FX.toAssociatedGraded s p = toGraded cx hi x ∧
  91         b ≫ FY.toAssociatedGraded (s+n) p = toGraded cy hj y ∧
  92         a ≫ (FX.F s p).arrow ≫ syntheticHomotopyMap g p = b ≫ (FY.F (s+n) p).arrow := by
  93   simp only [ExtensionEquation, FilteredMapExtension.Relation, filteredMap_map]
  94 
  95 /-- All targets for this source, expressed as the full homogeneous coset. -/
  96 public theorem targets_iff_coset
  97     (x : T ⟶ ((F.obj X).sequence.ssData i).eInfty)
  98     (y z : T ⟶ ((F.obj Y).sequence.ssData j).eInfty)
  99     (h : M.ExtensionEquation p s n hi hj x y) :
 100     M.ExtensionEquation p s n hi hj x z ↔
 101       toGraded cy hj z ∈ FilteredMapExtension.TargetCoset
 102         M.filteredMap p s n (toGraded cy hj y) :=
 103   (FilteredMapExtension.Relation.mem_targetCoset_iff h).symm
 104 
 105 public theorem essential_iff_no_zero_target
 106     (x : T ⟶ ((F.obj X).sequence.ssData i).eInfty)
 107     (y : T ⟶ ((F.obj Y).sequence.ssData j).eInfty)
 108     (h : M.ExtensionEquation p s n hi hj x y) :
 109     M.EssentialExtension p s n hi hj x y ↔
 110       ¬ M.ExtensionEquation p s n hi hj x 0 := by
 111   simpa only [EssentialExtension, ExtensionEquation, toGraded,
 112     Limits.zero_comp] using FilteredMapExtension.essential_iff h
 113 
 114 end ConvergentMap

 122 /-- Element form of the same extension relation, with canonical tower degrees.
 123 The equation remains meaningful without a bounded ESS construction. -/
 124 public def TowerExtensionEquation
 125     (M : ConvergentMap F g cx.toConvergence cy.toConvergence)
 126     (p : ℤ × ℤ) (s n : ℤ)
 127     (x : ((F.obj X).sequence.ssData (s,p.1+s,p.2)).eInfty)
 128     (y : ((F.obj Y).sequence.ssData (s+n,p.1+(s+n),p.2)).eInfty) : Prop :=
 129   M.ExtensionEquation p s n (by simp [SyntheticTowerConvergence.toConvergence])
 130     (by simp [SyntheticTowerConvergence.toConvergence]) (elementMap x) (elementMap y)
