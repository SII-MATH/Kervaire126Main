  34 /-- A homotopy pushout presented by a point-set pushout along a cofibration.
  35 All four comparison isomorphisms belong to the same localization functor. -/
  36 public def IsModelPushout {X Y Z W : C}
  37     (f : X ⟶ Y) (g : X ⟶ Z) (h : Y ⟶ W) (k : Z ⟶ W) : Prop :=
  38   ∃ (A B D E : P) (f' : A ⟶ B) (g' : A ⟶ D)
  39       (h' : B ⟶ E) (k' : D ⟶ E),
  40     cofibrations P f' ∧ IsPushout f' g' h' k' ∧
  41     ∃ (iX : L.obj A ≅ X) (iY : L.obj B ≅ Y)
  42       (iZ : L.obj D ≅ Z) (iW : L.obj E ≅ W),
  43       iX.hom ≫ f = L.map f' ≫ iY.hom ∧
  44       iX.hom ≫ g = L.map g' ≫ iZ.hom ∧
  45       iY.hom ≫ h = L.map h' ≫ iW.hom ∧
  46       iZ.hom ≫ k = L.map k' ≫ iW.hom
  47 
  48 /-- A homotopy pullback presented by a point-set pullback along a fibration. -/
  49 public def IsModelPullback {X Y Z W : C}
  50     (f : X ⟶ Y) (g : X ⟶ Z) (h : Y ⟶ W) (k : Z ⟶ W) : Prop :=
  51   ∃ (A B D E : P) (f' : A ⟶ B) (g' : A ⟶ D)
  52       (h' : B ⟶ E) (k' : D ⟶ E),
  53     fibrations P h' ∧ IsPullback f' g' h' k' ∧
  54     ∃ (iX : L.obj A ≅ X) (iY : L.obj B ≅ Y)
  55       (iZ : L.obj D ≅ Z) (iW : L.obj E ≅ W),
  56       iX.hom ≫ f = L.map f' ≫ iY.hom ∧
  57       iX.hom ≫ g = L.map g' ≫ iZ.hom ∧
  58       iY.hom ≫ h = L.map h' ≫ iW.hom ∧
  59       iZ.hom ≫ k = L.map k' ≫ iW.hom
  60 
  61 end LocalizationSquares

  63 /-- Stability on the actual localization target, including the model-square
  64 comparison contracts. The supplied derived suspension is identified with
  65 the already chosen shift, and model cofiber squares give distinguished
  66 triangles in the already chosen triangulation. -/
  67 public structure StableLocalization
  68     {P : Type u} [Category.{v} P] [ModelCategory P]
  69     {C : Type w} [StableHomotopyCategory.{w,z} C] (L : P ⥤ C) where
  70   localizes : L.IsLocalization (weakEquivalences P)
  71   suspension : C ⥤ C
  72   loop : C ⥤ C
  73   suspensionLoop : suspension ⊣ loop
  74   suspensionIso : suspension ≅ shiftFunctor C (1 : ℤ)
  75   bicartesian : ∀ {X Y Z W : C}
  76       (f : X ⟶ Y) (g : X ⟶ Z) (h : Y ⟶ W) (k : Z ⟶ W),
  77     f ≫ h = g ≫ k → (IsModelPushout L f g h k ↔ IsModelPullback L f g h k)
  78   pullbackComparison : ∀ {A B D E : P}
  79       (f : A ⟶ B) (g : A ⟶ D) (h : B ⟶ E) (k : D ⟶ E)
  80       (sq : f ≫ h = g ≫ k),
  81     fibrations P h →
  82     IsModelPullback L (L.map f) (L.map g) (L.map h) (L.map k) →
  83     IsIso (L.map (pullback.lift f g sq))
  84   pushoutComparison : ∀ {A B D E : P}
  85       (f : A ⟶ B) (g : A ⟶ D) (h : B ⟶ E) (k : D ⟶ E)
  86       (sq : f ≫ h = g ≫ k),
  87     cofibrations P f →
  88     IsModelPushout L (L.map f) (L.map g) (L.map h) (L.map k) →
  89     IsIso (L.map (pushout.desc h k sq))
  90   cofiberTriangle : ∀ {X Y Z : C} (f : X ⟶ Y) (g : Y ⟶ Z),
  91     IsModelPushout L f (0 : X ⟶ 0) g (0 : 0 ⟶ Z) →
  92     ∃ δ : Z ⟶ (shiftFunctor C (1 : ℤ)).obj X,
  93       Triangle.mk f g δ ∈ distTriang C

  99 /-- Stability of the unit follows from the supplied identification of
 100 derived suspension with the actual integer shift. -/
 101 public theorem unitIsIso (S : StableLocalization L) (X : C) :
 102     IsIso (S.suspensionLoop.unit.app X) := by
 103   letI : S.suspension.IsEquivalence :=
 104     Functor.isEquivalence_of_iso S.suspensionIso.symm
 105   infer_instance
 106 
 107 /-- The counit is also derived, rather than requested as a second proof
 108 input to the realization. -/
 109 public theorem counitIsIso (S : StableLocalization L) (X : C) :
 110     IsIso (S.suspensionLoop.counit.app X) := by
 111   letI : S.suspension.IsEquivalence :=
 112     Functor.isEquivalence_of_iso S.suspensionIso.symm
 113   infer_instance
 114 
 115 end StableLocalization

 133 /-- The actual weak-equivalence localization of the supplied point-set model. -/
 134 public abbrev Ho (R : StableModelRealization.{u,v,w,z} C) : Type u :=
 135   letI := R.pointSetCategory
 136   letI := R.pointSetModel
 137   (weakEquivalences R.PointSet).Localization
 138 
 139 /-- The same localization target, rather than a second unbound homotopy
 140 category. This equivalence follows from `localizes`. -/
 141 public noncomputable def homotopyEquivalence (R : StableModelRealization.{u,v,w,z} C) :
 142     letI := R.pointSetCategory
 143     letI := R.pointSetModel
 144     (weakEquivalences R.PointSet).Localization ≌ C := by
 145   letI := R.pointSetCategory
 146   letI := R.pointSetModel
 147   letI := R.stable.localizes
 148   exact Localization.equivalenceFromModel R.localization (weakEquivalences R.PointSet)
 149 
 150 /-- The canonical quotient followed by the comparison is the supplied
 151 localization functor, up to its canonical natural isomorphism. -/
 152 public noncomputable def localizationComparison
 153     (R : StableModelRealization.{u,v,w,z} C) :
 154     letI := R.pointSetCategory
 155     letI := R.pointSetModel
 156     (weakEquivalences R.PointSet).Q ⋙ R.homotopyEquivalence.functor ≅ R.localization := by
 157   letI := R.pointSetCategory
 158   letI := R.pointSetModel
 159   letI := R.stable.localizes
 160   exact Localization.qCompEquivalenceFromModelFunctorIso
 161     R.localization (weakEquivalences R.PointSet)
 162 
 163 end StableModelRealization

 165 /-- Generating classes for the supplied model structure, including the
 166 smallness hypotheses used for functorial replacement. -/
 167 public structure CofibrantGeneration (P : Type u)
 168     [Category.{v} P] [ModelCategory P] where
 169   generatingCofibrations : MorphismProperty P
 170   generatingTrivialCofibrations : MorphismProperty P
 171   cofibrationsRlp : generatingCofibrations.rlp = trivialFibrations P
 172   trivialCofibrationsRlp : generatingTrivialCofibrations.rlp = fibrations P
 173   cofibrationsSmall : MorphismProperty.HasSmallObjectArgument.{u} generatingCofibrations
 174   trivialCofibrationsSmall :
 175     MorphismProperty.HasSmallObjectArgument.{u} generatingTrivialCofibrations

 177 /-- The pushout-product and unit conditions for the same point-set tensor
 178 and model structure. The monoidal unit need not itself be cofibrant. -/
 179 public structure MonoidalModelCompatibility (P : Type u)
 180     [Category.{v} P] [ModelCategory P] [MonoidalCategory P] where
 181   pushoutProduct : ∀ {A B X Y : P} (f : A ⟶ B) (g : X ⟶ Y),
 182     cofibrations P f → cofibrations P g →
 183     cofibrations P (MonoidalCategory.Arrow.pushoutProduct.obj
 184       (Arrow.mk f) |>.obj (Arrow.mk g)).hom
 185   pushoutProductTrivialLeft : ∀ {A B X Y : P} (f : A ⟶ B) (g : X ⟶ Y),
 186     trivialCofibrations P f → cofibrations P g →
 187     trivialCofibrations P (MonoidalCategory.Arrow.pushoutProduct.obj
 188       (Arrow.mk f) |>.obj (Arrow.mk g)).hom
 189   pushoutProductTrivialRight : ∀ {A B X Y : P} (f : A ⟶ B) (g : X ⟶ Y),
 190     cofibrations P f → trivialCofibrations P g →
 191     trivialCofibrations P (MonoidalCategory.Arrow.pushoutProduct.obj
 192       (Arrow.mk f) |>.obj (Arrow.mk g)).hom
 193   unit : ∀ {QI : P} (q : QI ⟶ 𝟙_ P),
 194     IsCofibrant QI → weakEquivalences P q →
 195     ∀ X : P, IsCofibrant X → weakEquivalences P (q ▷ X)

 198 /-- Optional point-set realization of the classical background. Its model
 199 tensor is identified with the current homotopy-category tensor on cofibrant
 200 inputs, and the replacement of its unit is the current sphere. -/
 201 public structure SpectrumModelRealization (C : Type w)
 202     [StableHomotopyCategory.{w,z} C] extends StableModelRealization.{u,v,w,z} C where
 203   monoidal : MonoidalCategory PointSet
 204   symmetric : SymmetricCategory PointSet
 205   closed : MonoidalClosed PointSet
 206   limits : HasLimits.{v,u} PointSet
 207   colimits : HasColimits.{v,u} PointSet
 208   generation : CofibrantGeneration PointSet
 209   monoidalModel : MonoidalModelCompatibility PointSet
 210   tensorComparison : ∀ X Y : PointSet, IsCofibrant X → IsCofibrant Y →
 211     (localization.obj (X ⊗ Y) ≅ localization.obj X ⊗ localization.obj Y)
 212   tensorNaturality : ∀ {X X' Y Y' : PointSet} (f : X ⟶ X') (g : Y ⟶ Y')
 213       (hX : IsCofibrant X) (hX' : IsCofibrant X')
 214       (hY : IsCofibrant Y) (hY' : IsCofibrant Y'),
 215     localization.map (f ⊗ₘ g) ≫ (tensorComparison X' Y' hX' hY').hom =
 216       (tensorComparison X Y hX hY).hom ≫ (localization.map f ⊗ₘ localization.map g)
 217   unitComparison : ∀ {QI : PointSet} (q : QI ⟶ 𝟙_ PointSet),
 218     IsCofibrant QI → weakEquivalences PointSet q →
 219     (localization.obj QI ≅ SphereSpectrum (C := C))
