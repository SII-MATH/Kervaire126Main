  16 /-- The boundary is induced by this triangle's actual connecting arrow and
  17 this coefficient/shift comparison. It is not a freely chosen linear map. -/
  18 public def modTwoConnecting (H : ModTwoCoefficients C)
  19     (shiftIso : ∀ X : C, H.object ⊗ X⟦(1 : ℤ)⟧ ≅ (H.object ⊗ X)⟦(1 : ℤ)⟧)
  20     (T : HoCofiberSequence (C := C)) (i : ℤ) :
  21     ModTwoHomology H i T.Z →+ ModTwoHomology H (i-1) T.X :=
  22   (AbstractAdamsTower.desuspend (H.object ⊗ T.X) i).comp
  23     (inducedMap (H.object ◁ T.h ≫ (shiftIso T.X).hom) i)

  25 /-- Exactness on the same HF₂ homology and actual distinguished triangle.
  26 The cardinality bound is proved below rather than duplicated as an input. -/
  27 public structure ModTwoHomologyExactness (H : ModTwoCoefficients C) where
  28   shiftIso : ∀ X : C, H.object ⊗ X⟦(1 : ℤ)⟧ ≅ (H.object ⊗ X)⟦(1 : ℤ)⟧
  29   exact : ∀ (T : HoCofiberSequence (C := C)) (i : ℤ),
  30     Function.Exact (modTwoHomologyMap H T.g i) (modTwoConnecting H shiftIso T i)

  32 private def exactEmbedding {A B D : Type*} [AddCommGroup A] [AddCommGroup B]
  33     [AddCommGroup D] (f : A →+ B) (g : B →+ D) (h : Function.Exact f g) : B ↪ A × D := by
  34   classical
  35   let sectionMap : Set.range g → B := fun y => Classical.choose y.property
  36   have section_eq (y : Set.range g) : g (sectionMap y) = y.val :=
  37     Classical.choose_spec y.property
  38   let rep (b : B) := sectionMap ⟨g b, b, rfl⟩
  39   have hrep (b : B) : g (rep b) = g b := section_eq _
  40   have preimage (b : B) : ∃ a : A, f a = b - rep b :=
  41     (h (b - rep b)).mp (by rw [map_sub, hrep, sub_self])
  42   let lift (b : B) := Classical.choose (preimage b)
  43   have hlift (b : B) : f (lift b) = b - rep b := Classical.choose_spec (preimage b)
  44   refine ⟨fun b => (lift b, g b), ?_⟩
  45   intro b b' heq
  46   have hl : lift b = lift b' := congrArg Prod.fst heq
  47   have hg : g b = g b' := congrArg Prod.snd heq
  48   have hr : rep b = rep b' := congrArg sectionMap (Subtype.ext hg)
  49   calc
  50     b = f (lift b) + rep b := (eq_sub_iff_add_eq.mp (hlift b)).symm
  51     _ = f (lift b') + rep b' := by rw [hl, hr]
  52     _ = b' := eq_sub_iff_add_eq.mp (hlift b')
  53 
  54 private theorem exact_card_bound {A B D : Type*} [AddCommGroup A] [AddCommGroup B]
  55     [AddCommGroup D] (f : A →+ B) (g : B →+ D) (h : Function.Exact f g)
  56     (nA nD : ℕ) (hA : ∀ S : Finset A, S.card ≤ nA)
  57     (hD : ∀ S : Finset D, S.card ≤ nD) (S : Finset B) : S.card ≤ nA * nD := by
  58   classical
  59   let T := S.map (exactEmbedding f g h)
  60   have hsub : T ⊆ (T.image Prod.fst) ×ˢ (T.image Prod.snd) := by
  61     intro b hb
  62     exact Finset.mem_product.mpr
  63       ⟨Finset.mem_image.mpr ⟨b,hb,rfl⟩, Finset.mem_image.mpr ⟨b,hb,rfl⟩⟩
  64   calc
  65     S.card = T.card := (Finset.card_map _).symm
  66     _ ≤ ((T.image Prod.fst) ×ˢ (T.image Prod.snd)).card := Finset.card_le_card hsub
  67     _ = (T.image Prod.fst).card * (T.image Prod.snd).card := Finset.card_product _ _
  68     _ ≤ nA * nD := Nat.mul_le_mul (hA _) (hD _)
  69 
  70 /-- Correct cardinality version of the old cofiber bound. Cardinalities
  71 multiply; an additive bound would instead concern vector-space dimensions. -/
  72 public theorem ModTwoHomologyExactness.cofiber_card_bound
  73     {H : ModTwoCoefficients C} (E : ModTwoHomologyExactness H)
  74     (T : HoCofiberSequence (C := C)) (i : ℤ) (nX nY : ℕ)
  75     (hX : ∀ S : Finset (ModTwoHomology H (i-1) T.X), S.card ≤ nX)
  76     (hY : ∀ S : Finset (ModTwoHomology H i T.Y), S.card ≤ nY) :
  77     ∀ S : Finset (ModTwoHomology H i T.Z), S.card ≤ nX * nY := by
  78   intro S
  79   simpa [Nat.mul_comm] using exact_card_bound
  80     (modTwoHomologyMap H T.g i) (modTwoConnecting H E.shiftIso T i)
  81     (E.exact T i) nY nX hY hX S
