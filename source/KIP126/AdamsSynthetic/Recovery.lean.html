  20 /-- The actual component `λ_X : Σ^(0,-1)X ⟶ X` is invertible. -/
  21 public def IsLambdaInvertible : ObjectProperty Syn :=
  22   fun X => IsIso (SyntheticCategory.lam.app X)

  32 /-- The full subcategory cut out by invertibility of the existing λ map. -/
  33 public abbrev LambdaInvertibleObjects (Syn : Type u) [SyntheticCategory.{u, v} Syn] :=
  34   (IsLambdaInvertible (Syn := Syn)).FullSubcategory
  35 
  36 /-- The actual full-subcategory inclusion; its object map forgets only the proof. -/
  37 public abbrev lambdaInclusion (Syn : Type u) [SyntheticCategory.{u, v} Syn] :
  38     LambdaInvertibleObjects Syn ⥤ Syn :=
  39   (IsLambdaInvertible (Syn := Syn)).ι
  40 
  41 /-- A left adjoint to the specified inclusion. Its unit and universal
  42 property are determined by this adjunction, not chosen independently. -/
  43 public structure LambdaLocalization (Syn : Type u) [SyntheticCategory.{u, v} Syn] where
  44   reflector : Syn ⥤ LambdaInvertibleObjects Syn
  45   adjunction : reflector ⊣ lambdaInclusion Syn
  46 
  47 variable {Syn : Type u} [SyntheticCategory.{u, v} Syn]
  48 
  49 /-- The resulting reflective-subcategory witness, with the same reflector. -/
  50 public abbrev LambdaLocalization.reflective (L : LambdaLocalization Syn) :
  51     Reflective (lambdaInclusion Syn) where
  52   L := L.reflector
  53   adj := L.adjunction
  54 
  55 /-- Localization viewed as an endofunctor of the ambient synthetic category. -/
  56 public def LambdaLocalization.endofunctor (L : LambdaLocalization Syn) : Syn ⥤ Syn :=
  57   L.reflector ⋙ lambdaInclusion Syn
  58 
  59 /-- The unit of this specific adjunction. -/
  60 public def LambdaLocalization.unit (L : LambdaLocalization Syn) :
  61     𝟭 Syn ⟶ L.endofunctor := L.adjunction.unit
  62 
  63 /-- The uniquely determined factorization of a map to a λ-invertible object. -/
  64 public def LambdaLocalization.lift (L : LambdaLocalization Syn) {X Y : Syn}
  65     (hY : IsLambdaInvertible Y) (f : X ⟶ Y) : L.endofunctor.obj X ⟶ Y :=
  66   ((L.adjunction.homEquiv X ⟨Y, hY⟩).symm f).hom
  67 
  68 end KIP126.Synthetic.Context

  76 variable {C : Type u} [StableHomotopyCategory.{u, v} C]
  77   {Syn : Type u'} [SyntheticCategory.{u', v'} Syn]
  78 
  79 /-- The specified localization recovers the same classical category and ν. -/
  80 public structure LambdaRecovery (N : NuFunctorData C Syn) where
  81   localization : LambdaLocalization Syn
  82   equivalence : LambdaInvertibleObjects Syn ≌ C
  83   nuLocalization : N.functor ⋙ localization.reflector ≅ equivalence.inverse
  84 
  85 variable {N : NuFunctorData C Syn}
  86 
  87 /-- Underlying-spectrum realization, with the source and target fixed. -/
  88 public def LambdaRecovery.realization (R : LambdaRecovery N) : Syn ⥤ C :=
  89   R.localization.reflector ⋙ R.equivalence.functor
  90 
  91 /-- The spectral Yoneda embedding determined by the same equivalence. -/
  92 public def LambdaRecovery.spectralYoneda (R : LambdaRecovery N) : C ⥤ Syn :=
  93   R.equivalence.inverse ⋙ lambdaInclusion Syn
  94 
  95 /-- The spectral Yoneda embedding is fully faithful by the equivalence and
  96 the actual full-subcategory inclusion. -/
  97 public def LambdaRecovery.fullyFaithfulSpectralYoneda (R : LambdaRecovery N) :
  98     R.spectralYoneda.FullyFaithful :=
  99   R.equivalence.fullyFaithfulInverse.comp
 100     (IsLambdaInvertible (Syn := Syn)).fullyFaithfulι
 101 
 102 /-- Realization is left adjoint to spectral Yoneda, not to ν. -/
 103 public def LambdaRecovery.adjunction (R : LambdaRecovery N) :
 104     R.realization ⊣ R.spectralYoneda :=
 105   R.localization.adjunction.comp R.equivalence.toAdjunction
 106 
 107 /-- Applying realization after the specified ν recovers the identity functor. -/
 108 public def LambdaRecovery.nuRealizationIso (R : LambdaRecovery N) :
 109     N.functor ⋙ R.realization ≅ 𝟭 C :=
 110   (Functor.associator N.functor R.localization.reflector R.equivalence.functor).symm ≪≫
 111     Functor.isoWhiskerRight R.nuLocalization R.equivalence.functor ≪≫ R.equivalence.counitIso
 112 
 113 /-- The canonical map from ν to spectral Yoneda factors through the actual
 114 localization unit and the supplied natural comparison. -/
 115 public def LambdaRecovery.nuToSpectralYoneda (R : LambdaRecovery N) :
 116     N.functor ⟶ R.spectralYoneda :=
 117   Functor.whiskerLeft N.functor R.localization.unit ≫
 118     (Functor.isoWhiskerRight R.nuLocalization (lambdaInclusion Syn)).hom
 119 
 120 end KIP126.Synthetic.Context

 128 /-- The exactness conditions of the fixed review model, applied to the same
 129 specified suspension and realization functors. No extra ν exactness is inferred. -/
 130 public structure StructuralExactness (N : NuFunctorData C Syn) (R : LambdaRecovery N) where
 131   classicalTriangulated : IsTriangulated C
 132   syntheticTriangulated : IsTriangulated Syn
 133   shiftAdditive : ∀ p, (SyntheticCategory.biShift (Syn := Syn) p).Additive
 134   shiftCommShift : ∀ p, (SyntheticCategory.biShift (Syn := Syn) p).CommShift ℤ
 135   shiftExact : ∀ p, letI := shiftCommShift p;
 136     (SyntheticCategory.biShift (Syn := Syn) p).IsTriangulated
 137   realizationShift : R.realization.CommShift ℤ
 138   realizationExact : letI := realizationShift; R.realization.IsTriangulated
 139 end KIP126.AdamsSynthetic
