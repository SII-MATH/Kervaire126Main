  22 /-- An abstract presentation of the coefficient-unit fiber tower. Integer
  23 indices below zero are extended constantly, just as in the source model. -/
  24 public structure AbstractAdamsTower (X : C) where
  25   stage : ℕ → C
  26   map : ∀ {s t : ℕ}, s ≤ t → (stage t ⟶ stage s)
  27   map_self : ∀ s, map (le_refl s) = 𝟙 _
  28   map_comp : ∀ {r s t : ℕ} (hrs : r ≤ s) (hst : s ≤ t),
  29     map hst ≫ map hrs = map (hrs.trans hst)
  30   zeroIso : stage 0 ≅ X
  31   /-- The cofiber of a nonnegative step is the coefficient smash of that
  32   same stage. This assertion is intentionally not made at negative stages. -/
  33   layerIso : ∀ s : ℕ,
  34     HasFunctorialCofiber.cofib (map (Nat.le_succ s)) ≅ H ⊗ stage s
  35   layer_unit : ∀ s : ℕ,
  36     HasFunctorialCofiber.cofibι (map (Nat.le_succ s)) ≫ (layerIso s).hom =
  37       (λ_ (stage s)).inv ≫ unit ▷ (stage s)

  39 namespace AbstractAdamsTower
  40 variable {unit} {X : C} (T : AbstractAdamsTower unit X)
  41 public abbrev stageAt (s : ℤ) := T.stage s.toNat
  42 public def mapAt {s t : ℤ} (h : s ≤ t) : T.stageAt t ⟶ T.stageAt s :=
  43   T.map (show s.toNat ≤ t.toNat by omega)
  44 public def projection (s : ℤ) : T.stageAt s ⟶ X :=
  45   T.map (Nat.zero_le s.toNat) ≫ T.zeroIso.hom
  46 public def layer (s : ℤ) := HasFunctorialCofiber.cofib (T.mapAt (show s ≤ s + 1 by omega))
  47 public abbrev FirstPage (p : ℤ × ℤ) := HomotopyGroup (p.2 - p.1) (T.layer p.1)
  48 public def layerMap (p : ℤ × ℤ) :
  49     HomotopyGroup (p.2 - p.1) (T.stageAt p.1) →ₗ[ℤ] T.FirstPage p :=
  50   (inducedMap (HasFunctorialCofiber.cofibι (T.mapAt (show p.1 ≤ p.1 + 1 by omega)))
  51     (p.2 - p.1)).toIntLinearMap
  52 
  53 /-- Desuspension uses the shift structure of this category. -/
  54 public def desuspend (Y : C) (n : ℤ) :
  55     HomotopyGroup n (Y⟦(1 : ℤ)⟧) →+ HomotopyGroup (n - 1) Y where
  56   toFun z := (shiftFunctorAdd' C n (-1) (n - 1) (by omega)).hom.app SphereSpectrum ≫
  57     (shiftFunctor C (-1)).map z ≫ (shiftFunctorCompIsoId C 1 (-1) (by omega)).hom.app Y
  58   map_zero' := by simp only [Functor.map_zero, zero_comp, comp_zero]
  59   map_add' a b := by rw [Functor.map_add, Preadditive.add_comp, Preadditive.comp_add]
  60 
  61 public def connecting (p : ℤ × ℤ) : T.FirstPage p →ₗ[ℤ]
  62     HomotopyGroup (p.2 - p.1 - 1) (T.stageAt (p.1 + 1)) :=
  63   ((desuspend (T.stageAt (p.1 + 1)) (p.2 - p.1)).comp
  64     (inducedMap (HasFunctorialCofiber.cofibδ (T.mapAt (show p.1 ≤ p.1 + 1 by omega)))
  65       (p.2 - p.1))).toIntLinearMap
  66 
  67 public def cycles (r : ℕ) (hr : 1 ≤ r) (p : ℤ × ℤ) :
  68     Submodule ℤ (T.FirstPage p) :=
  69   (LinearMap.range (inducedMap (T.mapAt (show p.1 + 1 ≤ p.1 + r by omega))
  70     (p.2 - p.1 - 1)).toIntLinearMap).comap (T.connecting p)
  71 public def boundaries (r : ℕ) (hr : 1 ≤ r) (p : ℤ × ℤ) :
  72     Submodule ℤ (T.FirstPage p) :=
  73   (LinearMap.ker (inducedMap (T.mapAt (show p.1 - r + 1 ≤ p.1 by omega))
  74     (p.2 - p.1)).toIntLinearMap).map (T.layerMap p)
  75 
  76 public def homotopy (X : C) (m : ℤ) : ModuleCat.{v} ℤ :=
  77   ModuleCat.of ℤ (HomotopyGroup m X)
  78 public def filtrationSubmodule (s m : ℤ) : Submodule ℤ (HomotopyGroup m X) :=
  79   LinearMap.range (inducedMap (T.projection s) m).toIntLinearMap
  80 
  81 public theorem projection_compat {s t : ℤ} (h : s ≤ t) :
  82     T.mapAt h ≫ T.projection s = T.projection t := by
  83   simp only [projection, mapAt, ← Category.assoc, T.map_comp]
  84 
  85 public theorem filtrationSubmodule_antitone (m : ℤ) : Antitone (fun s => T.filtrationSubmodule s m) := by
  86   intro s t h a ha
  87   obtain ⟨b, rfl⟩ := ha
  88   refine ⟨b ≫ T.mapAt h, ?_⟩
  89   change (b ≫ T.mapAt h) ≫ T.projection s = b ≫ T.projection t
  90   rw [Category.assoc, T.projection_compat h]
  91 
  92 public def filtration : Filtration (homotopy X) where
  93   F s m := (ModuleCat.subobjectModule _).symm (T.filtrationSubmodule s m)
  94   mono s m := (ModuleCat.subobjectModule _).symm.monotone
  95     (T.filtrationSubmodule_antitone m (by omega : s ≤ s + 1))
  96 end AbstractAdamsTower

  98 /-- Comparison with a given page-two spectral sequence. Its ambient is the
  99 actual Z₂ in the tower's first page, and all Z/B subobjects are constrained.
 100 The last equation is the actual lift-and-cofiber differential, not a freely
 101 chosen page map. Both zero and nonzero representatives are included. -/
 102 public structure TowerSequencePresentation {X : C} (T : AbstractAdamsTower unit X)
 103     (D : (ℤ × ℤ) → SSData (ModuleCat.{v} ℤ))
 104     (d : ∀ (r : ℤ) (p : ℤ × ℤ), (D p).page (↑(r - 2).toNat : WithTop ℕ) ⟶
 105       (D (p + (r, r - 1))).page (↑(r - 2).toNat : WithTop ℕ)) where
 106   ambient : ∀ p, (D p).V ≅ ModuleCat.of ℤ (T.cycles 2 (by decide) p)
 107   cycles : ∀ p (n : ℕ) (a : (D p).V),
 108     a ∈ (ModuleCat.subobjectModule _ ((D p).Z n)) ↔
 109       ((ambient p).hom a).val ∈ T.cycles (n + 2) (by omega) p
 110   boundaries : ∀ p (n : ℕ) (a : (D p).V),
 111     a ∈ (ModuleCat.subobjectModule _ ((D p).B n)) ↔
 112       ((ambient p).hom a).val ∈ T.boundaries (n + 2) (by omega) p
 113   differential : ∀ (r : ℤ) (hr : 2 ≤ r) (p : ℤ × ℤ)
 114       (z : (Subobject.underlying.obj ((D p).Z (↑(r - 2).toNat : WithTop ℕ)) : ModuleCat ℤ))
 115       (a : HomotopyGroup (p.2 - p.1 - 1) (T.stageAt (p.1 + r)))
 116       (z' : (Subobject.underlying.obj ((D (p + (r, r - 1))).Z (↑(r - 2).toNat : WithTop ℕ)) : ModuleCat ℤ)),
 117     T.connecting p (((ambient p).hom (((D p).Z _).arrow z)).val) =
 118       a ≫ T.mapAt (show p.1 + 1 ≤ p.1 + r by omega) →
 119     (((ambient (p + (r, r - 1))).hom (((D (p + (r, r - 1))).Z _).arrow z')).val) =
 120       T.layerMap (p + (r, r - 1))
 121         (Eq.mp (congrArg (fun m => HomotopyGroup m (T.stageAt (p.1 + r)))
 122           (by dsimp; omega : p.2 - p.1 - 1 = (p + (r, r - 1)).2 - (p + (r, r - 1)).1)) a) →
 123     d r p ((D p).pageπ _ z) = (D (p + (r, r - 1))).pageπ _ z'

 135 /-- Tower morphisms fix the actual map at stage zero. -/
 136 public structure TowerMap {X Y : C} (T : AbstractAdamsTower unit X)
 137     (U : AbstractAdamsTower unit Y) (f : X ⟶ Y) where
 138   stage : ∀ n, T.stage n ⟶ U.stage n
 139   natural : ∀ {s t : ℕ} (h : s ≤ t), stage t ≫ U.map h = T.map h ≫ stage s
 140   zero : stage 0 ≫ U.zeroIso.hom = T.zeroIso.hom ≫ f
 141 
 142 namespace TowerMap
 143 variable {unit} {X Y : C} {T : AbstractAdamsTower unit X}
 144   {U : AbstractAdamsTower unit Y} {f : X ⟶ Y}
 145 public theorem projection (φ : TowerMap unit T U f) (s : ℤ) :
 146     φ.stage s.toNat ≫ U.projection s = T.projection s ≫ f := by
 147   dsimp [AbstractAdamsTower.projection]
 148   rw [← Category.assoc, φ.natural, Category.assoc, φ.zero, ← Category.assoc]
 149 
 150 /-- The cofiber-layer map induced by the same tower morphism. -/
 151 public def layer (φ : TowerMap unit T U f) (s : ℤ) : T.layer s ⟶ U.layer s :=
 152   HasFunctorialCofiber.cofibMap (T.mapAt (show s ≤ s + 1 by omega))
 153     (U.mapAt (show s ≤ s + 1 by omega)) (φ.stage (s + 1).toNat) (φ.stage s.toNat)
 154     (φ.natural (show s.toNat ≤ (s + 1).toNat by omega))
 155 end TowerMap

 157 /-- Differential on the page-two normalized quotient indices. -/
 158 public def normalizedDifferential
 159     (E : KIP126.Core.SpectralSequence (ModuleCat.{v} ℤ) (ℤ × ℤ))
 160     (h0 : E.r₀ = 2) (hd : ∀ r, E.diffDeg r = (r, r - 1))
 161     (r : ℤ) (p : ℤ × ℤ) :
 162     (E.ssData p).page (↑(r - 2).toNat : WithTop ℕ) ⟶
 163       (E.ssData (p + (r, r - 1))).page (↑(r - 2).toNat : WithTop ℕ) :=
 164   eqToHom (by simp only [h0]) ≫
 165     E.d r p ≫ eqToHom (by simp only [h0, hd])

 183 /-- Fixed-weight differential of the given synthetic family. -/
 184 public def fixedWeightDifferential (A : SyntheticAdamsSS.{v}) (w r : ℤ) (p : ℤ × ℤ) :
 185     (A.sequence.ssData (p.1,p.2,w)).page (↑(r - 2).toNat : WithTop ℕ) ⟶
 186       (A.sequence.ssData ((p + (r,r-1)).1,(p + (r,r-1)).2,w)).page
 187         (↑(r - 2).toNat : WithTop ℕ) :=
 188   eqToHom (by simp only [A.firstPage]) ≫
 189     A.sequence.d r (p.1,p.2,w) ≫ eqToHom (by
 190       simp only [A.firstPage, A.differentialDegree,
 191         syntheticAdamsRawShift, Prod.fst_add, Prod.snd_add]
 192       congr 2
 193       exact Prod.ext rfl (Prod.ext rfl (add_zero w)))

 195 /-- The chosen synthetic family is the same coefficient tower in every fixed
 196 weight. The comparison includes all cycles/boundaries and actual differentials;
 197 its naturality is tied to the actual stage and cofiber maps. -/
 198 public structure SyntheticTowerPresentation (F : SyntheticAdamsFamily Syn) where
 199   tower : ∀ X, AbstractAdamsTower synUnit X
 200   map : ∀ {X Y : Syn} (f : X ⟶ Y), TowerMap synUnit (tower X) (tower Y) f
 201   sequence : ∀ (X : Syn) (w : ℤ), TowerSequencePresentation synUnit
 202     (tower ((SyntheticCategory.biShift (0, -w)).obj X))
 203     (fun p => (F.obj X).sequence.ssData (p.1,p.2,w))
 204     (fixedWeightDifferential (F.obj X) w)
 205   /-- Naturality on E₂: raw first-page representatives may differ by a
 206   genuine initial boundary. This retains the source `natural_e2` contract
 207   without imposing a stronger equality before quotienting. -/
 208   natural_e2 : ∀ {X Y : Syn} (f : X ⟶ Y) (w : ℤ) (p : ℤ × ℤ)
 209       (a : ((F.obj X).sequence.ssData (p.1,p.2,w)).V),
 210     (((sequence Y w).ambient p).hom ((F.map f).φ (p.1,p.2,w) a)).val -
 211       (((sequence X w).ambient p).hom a).val ≫
 212         (map ((SyntheticCategory.biShift (0,-w)).map f)).layer p.1 ∈
 213       (tower ((SyntheticCategory.biShift (0,-w)).obj Y)).boundaries 2 (by decide) p
 214 end Synthetic
