  20 /-- Stage and cofiber comparisons for one object and its ordinary suspension. -/
  21 public structure ClassicalTowerSuspension (P : ClassicalTowerPresentation unit A) (X : C) where
  22   tower : ∀ s : ℤ, (P.tower (X⟦(1 : ℤ)⟧)).stageAt s ≅
  23     ((P.tower X).stageAt s)⟦(1 : ℤ)⟧
  24   tower_zero : (tower 0).hom ≫ (P.tower X).zeroIso.hom⟦(1 : ℤ)⟧' =
  25     (P.tower (X⟦(1 : ℤ)⟧)).zeroIso.hom
  26   tower_comm : ∀ (s t : ℤ) (hst : s ≤ t),
  27     (tower t).hom ≫ ((P.tower X).mapAt hst)⟦(1 : ℤ)⟧' =
  28       (P.tower (X⟦(1 : ℤ)⟧)).mapAt hst ≫ (tower s).hom
  29   layer : ∀ s : ℤ, (P.tower (X⟦(1 : ℤ)⟧)).layer s ≅
  30     ((P.tower X).layer s)⟦(1 : ℤ)⟧
  31   layer_inclusion : ∀ s : ℤ,
  32     HasFunctorialCofiber.cofibι
  33         ((P.tower (X⟦(1 : ℤ)⟧)).mapAt (show s ≤ s+1 by omega)) ≫
  34         (layer s).hom =
  35       (tower s).hom ≫ (HasFunctorialCofiber.cofibι
  36         ((P.tower X).mapAt (show s ≤ s+1 by omega)))⟦(1 : ℤ)⟧'
  37   layer_connecting : ∀ s : ℤ,
  38     (layer s).hom ≫ (HasFunctorialCofiber.cofibδ
  39         ((P.tower X).mapAt (show s ≤ s+1 by omega)))⟦(1 : ℤ)⟧' ≫
  40         (shiftFunctorComm C (1 : ℤ) (1 : ℤ)).hom.app ((P.tower X).stageAt (s+1)) =
  41       -(HasFunctorialCofiber.cofibδ
  42         ((P.tower (X⟦(1 : ℤ)⟧)).mapAt (show s ≤ s+1 by omega)) ≫
  43           (tower (s+1)).hom⟦(1 : ℤ)⟧')

  48 public def desuspendFirstPage (S : ClassicalTowerSuspension P X) (s t : ℤ) :
  49     (P.tower (X⟦(1 : ℤ)⟧)).FirstPage (s,t) →+
  50       (P.tower X).FirstPage (s,t-1) where
  51   toFun z :=
  52     eqToHom (congrArg (Sphere (C := C))
  53       (show (t-1)-s = (t-s)-1 by omega)) ≫
  54       AbstractAdamsTower.desuspend ((P.tower X).layer s) (t-s) (z ≫ (S.layer s).hom)
  55   map_zero' := by simp only [Limits.zero_comp, map_zero, Limits.comp_zero]
  56   map_add' a b := by simp only [Preadditive.add_comp, map_add, Preadditive.comp_add]
  57 
  58 /-- Desuspension of the actual common page-two representatives. Both labels
  59 come from the same supplied tower presentation and actual quotient projections. -/
  60 public def DesuspendsClass (S : ClassicalTowerSuspension P X) (s t : ℤ)
  61     (sx : (A.obj (X⟦(1 : ℤ)⟧)).Page 2 (s,t))
  62     (x : (A.obj X).Page 2 (s,t-1)) : Prop :=
  63   ∃ (a : (Subobject.underlying.obj (((A.obj (X⟦(1 : ℤ)⟧)).ssData (s,t)).Z 0) : ModuleCat.{v} ℤ))
  64     (b : (Subobject.underlying.obj (((A.obj X).ssData (s,t-1)).Z 0) : ModuleCat.{v} ℤ)),
  65     ((((A.obj (X⟦(1 : ℤ)⟧)).ssData (s,t)).pageπ 0) ≫
  66       eqToHom (by simp only [KIP126.Core.SpectralSequence.Page, A.firstPage, sub_self, Int.toNat_zero,
  67         Nat.cast_zero])) a = sx ∧
  68     ((((A.obj X).ssData (s,t-1)).pageπ 0) ≫
  69       eqToHom (by simp only [KIP126.Core.SpectralSequence.Page, A.firstPage, sub_self, Int.toNat_zero,
  70         Nat.cast_zero])) b = x ∧
  71     S.desuspendFirstPage s t
  72       ((((P.sequence (X⟦(1 : ℤ)⟧)).ambient (s,t)).hom
  73         ((((A.obj (X⟦(1 : ℤ)⟧)).ssData (s,t)).Z 0).arrow a)).val) =
  74       ((((P.sequence X).ambient (s,t-1)).hom
  75         ((((A.obj X).ssData (s,t-1)).Z 0).arrow b)).val)
