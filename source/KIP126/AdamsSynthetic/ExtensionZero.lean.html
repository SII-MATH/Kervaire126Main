   8 /-! Length-zero extensions are the E-infinity action of the same actual map.
   9 The general arrow version records projectivity; element consumers use the
  10 free rank-one module and need no extra hypothesis. -/
  11 namespace KIP126.AdamsSynthetic

  23 /-- The actual E∞ map, after the prescribed canonical identifications, is
  24 exactly the graded map of the same filtered homotopy map. -/
  25 public theorem ConvergentMap.infinity_action_graded
  26     (M : ConvergentMap F g cx.toConvergence cy.toConvergence) (i : Tridegree) :
  27     (cx.identification i).hom ≫
  28       M.filteredMap.inducedGrMap i.1 (i.2.1-i.1,i.2.2) =
  29       (F.map g).eInftyMap i ≫ (cy.identification i).hom := by
  30   have h := M.comparison.iso_compat i
  31   rw [M.infinity_eq] at h
  32   simpa only [SyntheticTowerConvergence.toConvergence, Filtration.transportGraded_self,
  33     Category.comp_id, ConvergentMap.filteredMap, ConvergenceMorphism.filteredMap,
  34     FilteredMorphism.inducedGrMap, SyntheticAdamsFamily.map] using h.symm
  35 
  36 /-- At length zero a projective test object supplies the source lift. -/
  37 public theorem ConvergentMap.extensionEquation_zero
  38     (M : ConvergentMap F g cx.toConvergence cy.toConvergence)
  39     (i : Tridegree) {T : ModuleCat.{v} ℤ} [Projective T]
  40     (x : T ⟶ ((F.obj X).sequence.ssData i).eInfty) :
  41     M.ExtensionEquation (i.2.1-i.1,i.2.2) i.1 0 rfl
  42       (by simp [SyntheticTowerConvergence.toConvergence])
  43       x (x ≫ (F.map g).eInftyMap i) := by
  44   unfold ConvergentMap.ExtensionEquation
  45   apply (FilteredMapExtension.relation_zero_iff_of_projective _ _ _ _ _).mpr
  46   simpa only [toGraded, SyntheticTowerConvergence.toConvergence,
  47     Filtration.transportGraded, Category.assoc, eqToHom_trans, eqToHom_refl,
  48     Category.comp_id] using
  49     congrArg (fun k => x ≫ k) (M.infinity_action_graded i)

  51 /-- The same zero-length equation with an explicitly chosen stem reindexing. -/
  52 public theorem ConvergentMap.extensionEquation_zero_reindex
  53     (M : ConvergentMap F g cx.toConvergence cy.toConvergence)
  54     (i : Tridegree) (p : ℤ × ℤ) (s : ℤ)
  55     (hi : cx.toConvergence.reindex i = (s,p))
  56     {T : ModuleCat.{v} ℤ} [Projective T]
  57     (x : T ⟶ ((F.obj X).sequence.ssData i).eInfty) :
  58     M.ExtensionEquation p s 0 hi
  59       (by simpa only [SyntheticTowerConvergence.toConvergence, add_zero] using hi)
  60       x (x ≫ (F.map g).eInftyMap i) := by
  61   have hs : i.1 = s := congrArg Prod.fst hi
  62   have hp : (i.2.1-i.1,i.2.2) = p := congrArg Prod.snd hi
  63   subst s
  64   subst p
  65   exact M.extensionEquation_zero i x

  67 private theorem extensionEquation_zero_transport
  68     (M : ConvergentMap F g cx.toConvergence cy.toConvergence)
  69     (i j : Tridegree) (p : ℤ × ℤ) (s : ℤ)
  70     (hi : cx.toConvergence.reindex i = (s,p))
  71     (hj : cy.toConvergence.reindex j = (s+0,p)) (hij : i=j)
  72     {T : ModuleCat.{v} ℤ} [Projective T]
  73     (x : T ⟶ ((F.obj X).sequence.ssData i).eInfty) :
  74     M.ExtensionEquation p s 0 hi hj x
  75       ((x ≫ (F.map g).eInftyMap i) ≫
  76         eqToHom (congrArg (fun k => ((F.obj Y).sequence.ssData k).eInfty) hij)) := by
  77   subst j
  78   simpa only [eqToHom_refl, Category.comp_id] using
  79     M.extensionEquation_zero_reindex i p s hi x

  81 public theorem elementMap_comp {A B : ModuleCat.{v} ℤ} (x : A) (f : A ⟶ B) :
  82     elementMap (f x) = elementMap x ≫ f := by

  86 /-- The element-level zero-length extension. Index transport only rewrites
  87 `s + 0`; the target is the actual family map on E∞, never a selected label. -/
  88 public theorem towerExtensionEquation_zero
  89     (M : ConvergentMap F g cx.toConvergence cy.toConvergence)
  90     (p : ℤ × ℤ) (s : ℤ)
  91     (x : ((F.obj X).sequence.ssData (s,p.1+s,p.2)).eInfty) :
  92     TowerExtensionEquation M p s 0 x
  93       ((eqToHom (by simp) :
  94         ((F.obj Y).sequence.ssData (s,p.1+s,p.2)).eInfty ⟶
  95         ((F.obj Y).sequence.ssData (s+0,p.1+(s+0),p.2)).eInfty)
  96         ((F.map g).eInftyMap (s,p.1+s,p.2) x)) := by
  97   unfold TowerExtensionEquation
  98   rw [elementMap_comp, elementMap_comp]
  99   exact extensionEquation_zero_transport M (s,p.1+s,p.2)
 100     (s+0,p.1+(s+0),p.2) p s _ _ (by simp) (elementMap x)
 101 end
