  21 /-! Concrete mathematical sources for the fixed Adams interfaces.
  22 The ordinary sequential-prespectrum source is kept distinct from completion.
  23 Milnor cohomology is the quotient for the specified polynomial differential;
  24 it is not an arbitrarily supplied graded object or a renamed Ext group.
  25 Only the source names needed to state fixed-background recognition are public.
  26 -/

 140 /-- Exactly the maps whose precomposition maps into every recognized integer

 143     {C : Type u} [StableHomotopyCategory.{u,v} C] {H : ModTwoCoefficients C}
 144     (R : ClassicalSourceRecognition C H) : MorphismProperty ClassicalSource :=
 145   Source.HF2Equivalences R

 147 public theorem classicalHF2Equivalences_iff
 148     {C : Type u} [StableHomotopyCategory.{u,v} C] {H : ModTwoCoefficients C}
 149     (R : ClassicalSourceRecognition C H) {X Y : ClassicalSource} (f : X ⟶ Y) :
 150     classicalHF2Equivalences R f ↔ ∀ d : ℤ,
 151       Function.Bijective (fun a : Y ⟶ R.shiftedCoefficient d => f ≫ a) := Iff.rfl
