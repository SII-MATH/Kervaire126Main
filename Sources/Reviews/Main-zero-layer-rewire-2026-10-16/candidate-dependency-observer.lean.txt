import Lean
open Lean Elab Command
set_option maxHeartbeats 0
set_option maxRecDepth 8192
run_cmd do
  let env ← getEnv
  let roots := #[`KIP126.Main.EtaExtensionDetection,`KIP126.Main.EtaExtensionDetectionAt,`KIP126.Main.H6SquarePermanent,`KIP126.Main.H6SquareSupportsD12,`KIP126.Main.Stem126D6Vanishes,`KIP126.Main.Theta5SquareNonzeroDetection,`KIP126.Main.Theta5SquareNonzeroDetectionAt,`KIP126.Main.ThetaChoice,`KIP126.Main.UChoice,`KIP126.Main.any_choice_criterion,`KIP126.Main.delta_h6_square_eq_lambda_eta_theta_square,`KIP126.Main.eta_extension_detection_choice_equivalence,`KIP126.Main.eta_extension_excluded_of_d6_vanishes,`KIP126.Main.h6_sq_permanent,`KIP126.Main.h6_square_differential_dichotomy_and_criterion,`KIP126.Main.theta5_square_nonzero_detection_choice_equivalence,`KIP126.Main.theta_square_three_possibilities]
  let mut pending := roots
  let mut seen : NameSet := {}
  let mut nodes : Array Json := #[]
  while !pending.isEmpty do
    let name := pending.back!
    pending := pending.pop
    if seen.contains name then continue
    seen := seen.insert name
    let some info := env.find? name | throwError "Missing dependency: {name}"
    let some idx := env.getModuleIdxFor? name | continue
    let mod := env.header.moduleNames[idx.toNat]!
    unless (`KIP126).isPrefixOf mod do continue
    let body := info.value? (allowOpaque := true)
    let deps := info.type.getUsedConstants ++ (body.map Expr.getUsedConstants).getD #[]
    nodes := nodes.push <| Json.mkObj [
      ("name", toJson name.toString),
      ("user_name", toJson (((privateToUserName? name).getD name).toString)),
      ("module", toJson mod.toString),
      ("is_axiom", toJson info.isAxiom),
      ("type_has_sorry", toJson info.type.hasSorry),
      ("body_has_sorry", toJson (body.any (·.hasSorry))),
      ("synthetic_sorry", toJson (info.type.hasSyntheticSorry || body.any (·.hasSyntheticSorry))),
      ("type_fingerprint", toJson (if roots.contains name || info.type.hasSorry || body.any (·.hasSorry) then toString (hash (toString info.type)) else "")),
      ("body_fingerprint", toJson (if info.type.hasSorry || body.any (·.hasSorry) then toString (hash (toString body)) else "")),
      ("dependencies", toJson (deps.map Name.toString))]
    pending := pending ++ deps
  liftIO <| IO.FS.writeFile "/tmp/Main-Actual-Dependency-Graph-Candidate.json"
    (Json.mkObj [("roots", toJson (roots.map Name.toString)),
      ("visited_constants", toJson seen.size), ("nodes", Json.arr nodes)]).compress
  logInfo m!"Main actual dependency graph: {nodes.size} project constants"
