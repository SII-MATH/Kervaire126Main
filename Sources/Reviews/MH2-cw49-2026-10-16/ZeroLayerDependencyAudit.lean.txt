open Lean Elab Command
set_option maxHeartbeats 0
set_option maxRecDepth 8192
run_cmd do
  let env ← getEnv
  let roots := #[`KIP126.ProofTools.Internal.Mahowald.generalized_mahowald_zero_layer_sphere,
    `KIP126.ProofTools.Internal.Mahowald.main_nu_mahowald_zero_layer]
  let mut pending : Array Name := #[]
  for sourceName in roots do
    let some (name, _) := env.constants.toList.find? (fun (name, _) =>
      ((privateToUserName? name).getD name) == sourceName)
      | throwError "Missing zero-layer theorem: {sourceName}"
    pending := pending.push name
  let ledger ← Lean.ofExcept <| Json.parse proofObligationLedger
  let rows ← Lean.ofExcept <| ledger.getObjValAs? (Array Json) "obligations"
  let mut seen : NameSet := {}
  let mut holes : Array Json := #[]
  while !pending.isEmpty do
    let name := pending.back!
    pending := pending.pop
    if seen.contains name then continue
    seen := seen.insert name
    let some info := env.find? name | throwError "Missing dependency: {name}"
    let some idx := env.getModuleIdxFor? name | continue
    let mod := env.header.moduleNames[idx.toNat]!
    unless (`KIP126).isPrefixOf mod do continue
    if info.isAxiom then throwError "Project axiom: {name}"
    let body := info.value? (allowOpaque := true)
    if info.type.hasSyntheticSorry || body.any (·.hasSyntheticSorry) then
      throwError "Synthetic sorry: {name}"
    if info.type.hasSorry || body.any (·.hasSorry) then
      if (`KIP126.Def.ProofTools).isPrefixOf mod then
        throwError "Local unfinished MH2/GL dependency: {name}"
      let some row := rows.find? (fun row => row.getObjValAs? String "name" == .ok name.toString)
        | throwError "Unregistered upstream proof responsibility: {name}"
      let ht ← Lean.ofExcept <| row.getObjValAs? String "type_fingerprint"
      let hb ← Lean.ofExcept <| row.getObjValAs? String "body_fingerprint"
      unless ht == toString (hash (toString info.type)) && hb == toString (hash (toString body)) do
        throwError "Changed upstream obligation: {name}"
      holes := holes.push <| Json.mkObj [("name", toJson name.toString), ("module", toJson mod.toString)]
    pending := pending ++ info.type.getUsedConstants
    if let some value := body then pending := pending ++ value.getUsedConstants
  liftIO <| IO.FS.writeFile "/tmp/MH2-zero-layer-dependencies.json"
    (Json.mkObj [("roots", toJson (roots.map Name.toString)),
      ("visited_constants", toJson seen.size), ("registered_upstream_obligations", Json.arr holes),
      ("local_pending_MH2_GL_dependencies", Json.arr #[]), ("project_axioms", Json.arr #[])]).pretty
  logInfo m!"Zero-layer dependency audit: {holes.size} registered upstream obligations; no local MH2/GL gap or project axiom"
