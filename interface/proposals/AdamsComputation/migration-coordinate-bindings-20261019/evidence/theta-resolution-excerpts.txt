private def enumerateMonomials {n : ℕ} (degree : Fin n → ℕ × ℕ)
    (candidates : List (Fin n)) : ℕ → ℕ → ℕ → ℕ → List (List (Fin n))
  | 0, _, s, t => if s = 0 ∧ t = 0 then [[]] else []
  | fuel + 1, lower, s, t =>
      if s = 0 ∧ t = 0 then [[]]
      else candidates.flatMap fun i =>
        if lower ≤ i.val ∧ (degree i).1 ≤ s ∧ (degree i).2 ≤ t then
          (enumerateMonomials degree candidates fuel i.val
            (s - (degree i).1) (t - (degree i).2)).map
              (fun m => i :: m)
        else []

private def monomialPowers {n : ℕ} (w : List (Fin n)) : Fin n →₀ ℕ :=
  (w.map fun i => Finsupp.single i 1).sum

private theorem enumerateMonomials_complete {n : ℕ} (degree : Fin n → ℕ × ℕ)

    (fuel lower s t : ℕ) (m : Fin n →₀ ℕ)

    ∃ w ∈ enumerateMonomials degree candidates fuel lower s t, monomialPowers w = m := by

      exact ⟨[], by simp [enumerateMonomials, hs, ht], rfl⟩

      have hp := positive i
      omega
    refine ⟨i :: w, ?_, ?_⟩

private theorem thetaSourceZero_of_monomials
    (hmono : ∀ (m : Fin generators.count →₀ ℕ),
      Finsupp.weight generators.degree m = (7, 70) →
      nativeProjection generators.count archiveSourceRelations (MvPolynomial.X ⟨18, by decide⟩) *
        nativeProjection generators.count archiveSourceRelations (MvPolynomial.monomial m 1) = 0)
    (x : sourceData.component 7 70) :
    sourceData.product 1 32 7 70 (sourceData.atom .h5) x = 0 := by
  apply Subtype.ext
  change nativeProjection generators.count archiveSourceRelations (MvPolynomial.X ⟨18, by decide⟩) * x.val = 0
  obtain ⟨p, hp, hx⟩ := x.property
  change p.IsWeightedHomogeneous generators.degree (7, 70) at hp
  change nativeProjection generators.count archiveSourceRelations p = x.val at hx
  rw [← hx]
  clear hx x
  induction hp using MvPolynomial.IsWeightedHomogeneous.induction_on with
  | zero => simp
  | add p q hp hq ihp ihq => simp only [map_add, mul_add, ihp, ihq, add_zero]
  | monomial m r hm =>
      have hmr : MvPolynomial.monomial m r = MvPolynomial.C r * MvPolynomial.monomial m 1 := by rw [MvPolynomial.C_mul_monomial, mul_one]
      rw [hmr, map_mul]
      calc
        _ = nativeProjection generators.count archiveSourceRelations (MvPolynomial.C r) *
            (nativeProjection generators.count archiveSourceRelations (MvPolynomial.X ⟨18, by decide⟩) *
            nativeProjection generators.count archiveSourceRelations (MvPolynomial.monomial m 1)) := by ring
        _ = 0 := by rw [hmono m hm, mul_zero]
